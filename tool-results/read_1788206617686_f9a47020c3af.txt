     1→// ============================================================
     2→// PLAN CONFIG — DB-backed, cached single source of truth for plans,
     3→// pricing, entitlements and usage limits.
     4→// ============================================================
     5→// The Client Billing API and the Platform Admin both read from (and
     6→// the admin mutates) this same service. `platform-data.ts` delegates
     7→// its `PLANS` / `getPlan` / `monthlyPrice` to the synchronous cache
     8→// exposed here, so an owner edit (savePlanConfig) propagates to the
     9→// client billing experience and to MRR on the next read.
    10→//
    11→// MULTI-CURRENCY: the plan carries ONE base price configuration —
    12→// priceMonthly / priceYearly denominated in the plan's default /
    13→// fallback `currency` (both edited directly in the Edit Plan modal).
    14→// The customer's currency is resolved SERVER-SIDE from their IP
    15→// geolocation (see country-pricing.ts:resolveCustomerPricing) when
    16→// the plan's `autoCurrency` flag is on; otherwise every customer is
    17→// billed in the plan default currency.
    18→//
    19→// pricesByCurrency is a DERIVED store (NOT edited per-plan in the
    20→// modal): the default currency's entry mirrors the base price on
    21→// every save, and the other entries hold the platform-level
    22→// regional prices (CountryPricing.regionalPrices + legacy entries)
    23→// — preserved across saves so customers in other currencies keep
    24→// seeing their configured price (e.g. "90 MAD").
    25→//
    26→// STRIPE: for each paid plan × supported currency × interval, a real
    27→// Stripe Price is created on sync and its ID persisted in
    28→// stripePriceIdsByCurrency. The default currency's IDs are mirrored
    29→// in stripePriceIdMonthly/Yearly for legacy callers. Checkout always
    30→// resolves the per-currency Stripe Price ID server-side — the
    31→// customer's currency determines which Price is charged.
    32→//
    33→// The cache is hydrated from the PlanConfig table on startup and
    34→// after every mutation. If the table is empty on first hydrate, the
    35→// DEFAULT configs are seeded.
    36→// ============================================================

    94→// -------------------- Entitlement normalization --------------------
    95→
    96→/** Normalize a raw entitlement key list (DB row or API input):
    97→ *  - legacy 'ai_content' → 'ai_platform' (it predates the two-key
    98→ *    split and always meant the platform-provided AI).
    99→ *  - Platform AI and Client's Own AI API are INDEPENDENT — both may
   100→ *    be present (no mutual exclusion to enforce).
   101→ *  - API Access is REMOVED as a feature: 'api_access' is a legacy
   102→ *    removed key (like custom_domains / white_label) — Client's Own
   103→ *    AI API already represents the client's own-provider/API access,
   104→ *    so the duplicate key is STRIPPED on load + save. Old DB rows
   105→ *    self-clean; stale API input never reaches the saved plan.
   106→ *  - legacy 'custom_domains' / 'white_label' are STRIPPED: site
   107→ *    domains and site branding are client-owned in this architecture
   108→ *    (every site carries its own domain + identity), so they are not
   109→ *    plan entitlements. Old DB rows self-clean on load + next save.
   110→ *  - dedupes while preserving order.
   111→ *  Non-editor keys (e.g. 'audit_log') pass through untouched. */
   112→export function normalizeEntitlementKeys(keys: readonly string[]): string[] {
   113→  // Legacy keys that are no longer plan entitlements — stripped on
   114→  // load/save so existing rows + stale API input clean themselves.
   115→  // 'api_access' joined them: Client's Own AI API already covers the
   116→  // client's own-API connectivity, so a separate API Access feature
   117→  // no longer exists anywhere (editor, cards, server checks).
   118→  const LEGACY_REMOVED_KEYS = new Set(['custom_domains', 'white_label', 'api_access']);
   119→  const out: string[] = [];
   120→  const seen = new Set<string>();
   121→  const hasPlatform = keys.includes('ai_platform') || keys.includes('ai_content');
   122→  for (const k of keys) {
   123→    if (typeof k !== 'string' || !k.trim()) continue;
   124→    if (k === 'ai_content') continue; // legacy → replaced by ai_platform below
   125→    if (LEGACY_REMOVED_KEYS.has(k)) continue; // not plan entitlements anymore
   126→    if (k === 'ai_platform' && hasPlatform) {
   127→      if (!seen.has('ai_platform')) out.push('ai_platform');
   128→      seen.add('ai_platform');
   129→      continue;
   130→    }
   131→    if (!seen.has(k)) {
   132→      out.push(k);
   133→      seen.add(k);
   134→    }
   135→  }
   136→  // Insert the platform AI key at the position where the legacy key
   137→  // (or an existing 'ai_platform') first appeared — append at the end
   138→  // when neither was present in the ordered loop above (defensive).
   139→  if (hasPlatform && !seen.has('ai_platform')) out.push('ai_platform');
   140→  return out;
   141→}

   157→  if (aiModeOfEntitlements(entitlements) === 'platform') return limits;

   165→/** Pick only the known limit fields from an untyped/partial input —
   166→ *  unknown keys (stale `aiWords` / `aiWordsPerMonth` / `automationRuns`
   167→ *  from older rows) never leak into the cache and are not re-serialized
   168→ *  on save. `aiWordsPerMonth` was REMOVED as a plan limit (AI usage is
   169→ *  metered by generations only): rows that still carry it self-clean on
   170→ *  load and the next save strips it permanently. */
   171→function pickLimits(parsed: unknown): PlanLimits {
   172→  const p = (parsed ?? {}) as Partial<PlanLimits>;
   173→  const num = (v: unknown): number => (typeof v === 'number' && !Number.isNaN(v) ? v : 0);
   174→  return {
   175→    maxSites: num(p.maxSites),
   176→    storageBytes: num(p.storageBytes),
   177→    aiArticlesPerMonth: num(p.aiArticlesPerMonth),
   178→    aiImagesPerMonth: num(p.aiImagesPerMonth),
   179→  };
   180→}

   182→export type BillingInterval = 'monthly' | 'yearly';
   183→
   184→/** The billing periods enabled on a plan (at least one, enforced).
   185→ *  A disabled period: not shown on the Client Billing page, rejected
   186→ *  at checkout (400 BILLING_PERIOD_NOT_ENABLED), and never gets a
   187→ *  Stripe Price created or selected. */
   188→export function enabledIntervalsOf(plan: {
   189→  billingMonthly: boolean;
   190→  billingYearly: boolean;
   191→}): BillingInterval[] {
   192→  const out: BillingInterval[] = [];
   193→  if (plan.billingMonthly) out.push('monthly');
   194→  if (plan.billingYearly) out.push('yearly');
   195→  return out;
   196→}
   197→
   198→/** Clamp a requested default cadence to an ENABLED period:
   199→ *  only monthly on → 'monthly'; only yearly on → 'yearly'; both on
   200→ *  → the requested value. (Neither enabled is rejected upstream —
   201→ *  here it falls back to the requested value.) */
   202→function normalizeInterval(
   203→  requested: BillingInterval,
   204→  billingMonthly: boolean,
   205→  billingYearly: boolean,
   206→): BillingInterval {
   207→  if (billingMonthly && !billingYearly) return 'monthly';
   208→  if (!billingMonthly && billingYearly) return 'yearly';
   209→  return requested;
   210→}

   434→/** Resolve the platform's default billing currency via a LAZY import of
   435→ *  country-pricing (which itself imports `getPlanConfigSync` from here).
   436→ *  The lazy `await import()` breaks the static circular dependency —
   437→ *  this module never statically imports country-pricing, so both load.
   438→ *  Falls back to 'CHF' (the schema default) if anything fails. */
   439→async function resolvePlatformDefaultCurrency(): Promise<string> {
   440→  try {
   441→    const { getPlatformDefaultCurrency } = await import('./country-pricing');
   442→    return await getPlatformDefaultCurrency();
   443→  } catch {
   444→    return 'CHF';
   445→  }
   446→}

   454→let _hydrated = false;

   456→const _subscribers = new Set<() => void>();

   458→function notify() {
   459→  for (const cb of _subscribers) {
   460→    try {
   461→      cb();
   462→    } catch {
   463→      // ignore subscriber errors
   464→    }
   465→  }
   466→}

   468→/** Register a callback fired after every cache refresh (hydrate / mutation). */
   469→export function subscribe(cb: () => void): () => void {
   470→  _subscribers.add(cb);
   471→  return () => _subscribers.delete(cb);
   472→}

   474→/** Synchronous access to the cached plan configs. Returns defaults before
   475→ *  the first hydrate completes (the defaults match the seeded DB rows, so
   476→ *  numbers are correct from the very first render). */
   477→export function getPlanConfigsSync(): PlanConfigData[] {
   478→  if (!_hydrated && !_hydrating) {
   479→    void hydrate();
   480→  }
   481→  return _cache;
   482→}

   507→  return base;
   508→}

   542→function rowToData(row: PlanConfigRow): PlanConfigData {
   543→  let features: string[] = [];
   544→  let entitlements: string[] = [];
   545→  try {
   546→    features = JSON.parse(row.features || '[]');
   547→  } catch {
   548→    features = [];
   549→  }
   550→  try {
   551→    entitlements = JSON.parse(row.entitlements || '[]');
   552→  } catch {
   553→    entitlements = [];
   554→  }
   555→  // Explicitly pick only the known limit fields. Older DB rows may still
   556→  // carry stale `aiWords` / `aiWordsPerMonth` / `automationRuns` keys in the
   557→  // limits JSON (aiWordsPerMonth was removed — AI usage is metered by
   558→  // generations only; those other keys never had real enforcement).
   559→  // Picking only the known fields prevents the stale keys from leaking
   560→  // back into the cache and being re-serialized on the next save.
   561→  let limits: PlanLimits = { ...DEFAULT_LIMITS };
   562→  try {
   563→    limits = pickLimits(JSON.parse(row.limits || '{}'));
   564→  } catch {
   565→    // keep defaults
   566→  }
   567→  // AI usage limits only exist for Platform AI plans — normalize the
   568→  // stored configuration to match the entitlement mode.
   569→  limits = zeroAiLimitsIfNotPlatform(limits, entitlements);
   570→  let pricesByCurrency: PricesByCurrency = {};
   571→  try {
   572→    const parsed = JSON.parse(row.pricesByCurrency || '{}') as Partial<PricesByCurrency>;
   573→    if (parsed && typeof parsed === 'object') {
   574→      for (const [cur, val] of Object.entries(parsed)) {
   575→        if (val && typeof val === 'object') {
   576→          const m = typeof (val as CurrencyPrice).monthly === 'number' ? (val as CurrencyPrice).monthly : 0;
   577→          const y = typeof (val as CurrencyPrice).yearly === 'number' ? (val as CurrencyPrice).yearly : 0;
   578→          pricesByCurrency[cur.toUpperCase()] = { monthly: m, yearly: y };
   579→        }
   580→      }
   581→    }
   582→  } catch {
   583→    pricesByCurrency = {};
   584→  }
   585→  let stripePriceIdsByCurrency: StripePriceIdsByCurrency = {};
   586→  try {
   587→    const parsed = JSON.parse(row.stripePriceIdsByCurrency || '{}') as Partial<StripePriceIdsByCurrency>;
   588→    if (parsed && typeof parsed === 'object') {
   589→      for (const [cur, val] of Object.entries(parsed)) {
   590→        if (val && typeof val === 'object') {
   591→          const m = typeof (val as CurrencyStripeIds).monthly === 'string' ? (val as CurrencyStripeIds).monthly : null;
   592→          const y = typeof (val as CurrencyStripeIds).yearly === 'string' ? (val as CurrencyStripeIds).yearly : null;
   593→          stripePriceIdsByCurrency[cur.toUpperCase()] = { monthly: m, yearly: y };
   594→        }
   595→      }
   596→    }
   597→  } catch {
   598→    stripePriceIdsByCurrency = {};
   599→  }
   600→  return {
   601→    planId: row.planId,
   602→    name: row.name,
   603→    priceMonthly: row.priceMonthly,
   604→    priceYearly: row.priceYearly,
   605→    currency: row.currency,
   606→    autoCurrency: row.autoCurrency ?? true,
   607→    pricesByCurrency,
   608→    stripePriceIdsByCurrency,
   609→    billingMonthly: row.billingMonthly ?? true,
   610→    billingYearly: row.billingYearly ?? true,
   611→    // Normalize the default cadence to an ENABLED period (defensive:
   612→    // legacy rows / stale values never advertise a disabled period).
   613→    interval: normalizeInterval(
   614→      (row.interval === 'yearly' ? 'yearly' : 'monthly') as BillingInterval,
   615→      row.billingMonthly ?? true,
   616→      row.billingYearly ?? true,
   617→    ),
   618→    isFree: row.isFree,
   619→    freePlanDurationDays: row.freePlanDurationDays,
   620→    stripePriceIdMonthly: row.stripePriceIdMonthly,
   621→    stripePriceIdYearly: row.stripePriceIdYearly,
   622→    active: row.active,
   623→    features,
   624→    // Normalize: legacy 'ai_content' → 'ai_platform'; the legacy
   625→    // 'api_access' key is stripped (no separate API Access feature —
   626→    // Client's Own AI API covers it); the two AI keys are independent
   627→    // (both may be present).
   628→    entitlements: normalizeEntitlementKeys(entitlements),
   629→    limits,
   630→    badgeVariant: row.badgeVariant,
   631→    sortOrder: row.sortOrder,
   632→  };
   633→}

   661→/**
   662→ * Migrate the legacy catalog to the new Free/Plus/Pro/Max shape. Idempotent.
   663→ *
   664→ * Legacy catalog (created by older bootstrap):
   665→ *   beta (free, active=false), pro, max, enterprise (stub)
   666→ *
   667→ * Migration steps (each guarded, can be re-run safely):
   668→ *  1. If `enterprise` exists AND no active subscription references it → delete it.
   669→ *     (Active subs that reference enterprise are preserved — the migration will
   670→ *      not break them; the plan row stays if needed. In practice, the demo
   671→ *      dataset has no DB subscriptions yet, so this is a clean delete.)
   672→ *  2. If `beta` exists → rename to `plus` (planId, name, badgeVariant) and
   673→ *     re-enable it (legacy `beta` was active=false in the demo DB). Also
   674→ *     shift sortOrder to 1 (Free takes 0).
   675→ *  3. If `free` does NOT exist → insert the Free plan at sortOrder 0.
   676→ *  4. Renumber sortOrder to match the canonical order: free=0, plus=1, pro=2, max=3.
   677→ *
   678→ * Returns a short report string for the bootstrap log.
   679→ */
   680→export async function migrateLegacyPlans(): Promise<string> {
   681→  const notes: string[] = [];
   682→
   683→  try {
   684→    // Step 1: delete Enterprise if safe
   685→    const enterprise = await db.planConfig.findUnique({ where: { planId: 'enterprise' } });
   686→    if (enterprise) {
   687→      const subCount = await db.subscription.count({ where: { planId: 'enterprise' } });
   688→      if (subCount === 0) {
   689→        await db.planConfig.delete({ where: { planId: 'enterprise' } });
   690→        notes.push('deleted enterprise plan (0 active subscriptions)');
   691→      } else {
   692→        notes.push(`kept enterprise plan (${subCount} active subscriptions reference it)`);
   693→      }
   694→    }
   695→
   696→    // Step 2: rename beta → plus (and refresh to a real paid tier)
   697→    const beta = await db.planConfig.findUnique({ where: { planId: 'beta' } });
   698→    if (beta) {
   699→      // Don't overwrite an existing plus row — if both exist, prefer the
   700→      // plus row and just delete the beta stub.
   701→      const plus = await db.planConfig.findUnique({ where: { planId: 'plus' } });
   702→      if (!plus) {
   703→        // Rename beta → plus AND refresh the data to the real DEFAULT
   704→        // Plus tier (paid plan with real entitlements + limits, not the
   705→        // legacy free-tier Beta stub). The old Beta was active=false and
   706→        // had no real customers, so this is a safe refresh.
   707→        const plusDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'plus')!;
   708→        await db.planConfig.update({
   709→          where: { planId: 'beta' },
   710→          data: {
   711→            ...dataToRow(plusDefault),
   712→            // Keep the original createdAt for continuity.
   713→            // (updatedAt is auto.)
   714→          },
   715→        });
   716→        notes.push('renamed + refreshed beta → plus (real paid tier)');
   717→      } else {
   718→        // Both beta and plus exist — delete the beta stub (migrate any
   719→        // legacy subscriptions on beta to plus first).
   720→        const betaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   721→        if (betaSubs > 0) {
   722→          await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   723→          notes.push(`migrated ${betaSubs} beta subscription(s) → plus`);
   724→        }
   725→        await db.planConfig.delete({ where: { planId: 'beta' } });
   726→        notes.push('deleted legacy beta stub (plus already exists)');
   727→      }
   728→    }
   729→
   730→    // Step 3: insert Free if missing
   731→    const free = await db.planConfig.findUnique({ where: { planId: 'free' } });
   732→    if (!free) {
   733→      const freeDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'free')!;
   734→      await db.planConfig.create({ data: dataToRow(freeDefault) });
   735→      notes.push('inserted free plan');
   736→    } else {
   737→      // Ensure free plan is properly flagged
   738→      if (!free.isFree || free.badgeVariant !== 'free' || free.sortOrder !== 0) {
   739→        await db.planConfig.update({
   740→          where: { planId: 'free' },
   741→          data: { isFree: true, badgeVariant: 'free', sortOrder: 0 },
   742→        });
   743→        notes.push('normalized free plan flags');
   744→      }
   745→    }
   746→
   747→    // Step 4: normalize sortOrder for all canonical plans
   748→    const canonicalOrder: Record<string, number> = { free: 0, plus: 1, pro: 2, max: 3 };
   749→    for (const [pid, order] of Object.entries(canonicalOrder)) {
   750→      const row = await db.planConfig.findUnique({ where: { planId: pid } });
   751→      if (row && row.sortOrder !== order) {
   752→        await db.planConfig.update({ where: { planId: pid }, data: { sortOrder: order } });
   753→      }
   754→    }
   755→
   756→    // Also migrate legacy subscriptions on 'beta' or 'enterprise' planId
   757→    // to the new ids (beta → plus, enterprise → max) so historical subs
   758→    // don't reference a missing plan.
   759→    const legacyBetaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   760→    if (legacyBetaSubs > 0) {
   761→      await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   762→      notes.push(`migrated ${legacyBetaSubs} legacy beta subscription(s) → plus`);
   763→    }
   764→    const legacyEnterpriseSubs = await db.subscription.count({ where: { planId: 'enterprise' } });
   765→    if (legacyEnterpriseSubs > 0) {
   766→      await db.subscription.updateMany({ where: { planId: 'enterprise' }, data: { planId: 'max' } });
   767→      notes.push(`migrated ${legacyEnterpriseSubs} legacy enterprise subscription(s) → max`);
   768→    }
   769→  } catch (err) {
   770→    notes.push(`migration error (non-fatal): ${(err as Error).message}`);
   771→  }
   772→
   773→  return notes.length > 0 ? notes.join('; ') : 'no migration needed';
   774→}

   776→/** Load all plan configs from the DB into the cache. Self-seeds the
   777→ *  DEFAULT_PLAN_CONFIGS when the table is empty. Runs the legacy
   778→ *  migration when the catalog still contains 'beta' or 'enterprise'.
   779→ *  Safe to call repeatedly. */
   780→export async function hydrate(): Promise<void> {
   781→  if (_hydrating) return _hydrating;
   782→  _hydrating = (async () => {
   783→    try {
   784→      const rows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   785→      if (rows.length === 0) {
   786→        // Self-seed so the platform is never without plans.
   787→        await db.planConfig.createMany({ data: DEFAULT_PLAN_CONFIGS.map(dataToRow) });
   788→        _cache = DEFAULT_PLAN_CONFIGS.map((p) => ({
   789→          ...p,
   790→          features: [...p.features],
   791→          entitlements: [...p.entitlements],
   792→          limits: { ...p.limits },
   793→        }));
   794→      } else {
   795→        // Check for legacy catalog (beta / enterprise) and migrate.
   796→        const hasLegacy =
   797→          rows.some((r) => r.planId === 'beta' || r.planId === 'enterprise') ||
   798→          !rows.some((r) => r.planId === 'free');
   799→        if (hasLegacy) {
   800→          await migrateLegacyPlans();
   801→          const migratedRows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   802→          _cache = migratedRows.map(rowToData);
   803→        } else {
   804→          _cache = rows.map(rowToData);
   805→        }
   806→      }
   807→      _hydrated = true;
   808→      notify();
   809→    } catch (err) {
   810→      // DB unavailable — keep defaults (still functional, just not persisted).
   811→      _hydrated = true;
   812→      notify();
   813→    } finally {
   814→      _hydrating = null;
   815→    }
   816→  })();
   817→  return _hydrating;
   818→}

   859→/** Merge a patch (Partial<PlanConfigData>) into the current data.
   860→ *  The Edit Plan modal sends the ONE base price configuration
   861→ *  (priceMonthly + priceYearly + currency + autoCurrency) — NOT a
   862→ *  per-currency matrix. Derivation rules:
   863→ *  - pricesByCurrency: starts from the current map (or the explicit
   864→ *    patch map when provided); whenever the base price / currency is
   865→ *    in the patch, the DEFAULT currency's entry is re-derived as
   866→ *    { monthly: priceMonthly, yearly: priceYearly }. Other currency
   867→ *    entries (platform regional / legacy prices) are PRESERVED.
   868→ *  - stripePriceIdsByCurrency: kept as-is when omitted (the Stripe
   869→ *    sync maintains it); the default-currency snapshots mirror the
   870→ *    default entry.
   871→ */
   872→function mergePlanPatch(current: PlanConfigData, patch: PlanConfigInput, defaultCurrency: string): PlanConfigData {
   873→  const curUpper = defaultCurrency.toUpperCase();
   874→
   875→  // pricesByCurrency — clone (never mutate the cache's object in place).
   876→  const sourcePrices = patch.pricesByCurrency ?? current.pricesByCurrency ?? {};
   877→  const pricesByCurrency: PricesByCurrency = { ...sourcePrices };
   878→
   879→  // Base price precedence:
   880→  //   1. Explicit patch values (the modal always sends these).
   881→  //   2. Legacy mirror: an explicit pricesByCurrency map with an entry
   882→  //      for the final default currency (backward-compat callers).
   883→  //   3. Current snapshot.
   884→  const mirrored =
   885→    patch.priceMonthly === undefined && patch.pricesByCurrency !== undefined
   886→      ? pricesByCurrency[curUpper]
   887→      : undefined;
   888→  const priceMonthly =
   889→    patch.priceMonthly !== undefined ? patch.priceMonthly : mirrored?.monthly ?? current.priceMonthly;
   890→  const priceYearly =
   891→    patch.priceYearly !== undefined ? patch.priceYearly : mirrored?.yearly ?? current.priceYearly;
   892→
   893→  // The base price configuration was touched (or a legacy map was sent)
   894→  // → re-derive the DEFAULT currency's entry from the base price. All
   895→  // OTHER currency entries (platform regional / legacy prices) are
   896→  // preserved untouched.
   897→  const baseTouched =
   898→    patch.priceMonthly !== undefined ||
   899→    patch.priceYearly !== undefined ||
   900→    patch.currency !== undefined ||
   901→    patch.pricesByCurrency !== undefined;
   902→  if (baseTouched) {
   903→    pricesByCurrency[curUpper] = { monthly: priceMonthly, yearly: priceYearly };
   904→  }
   905→
   906→  const stripePriceIdsByCurrency: StripePriceIdsByCurrency = {
   907→    ...(patch.stripePriceIdsByCurrency ?? current.stripePriceIdsByCurrency ?? {}),
   908→  };
   909→  // Default-currency snapshot — keep the legacy fields in sync so legacy
   910→  // callers that read stripePriceIdMonthly/Yearly see the right values.
   911→  const defIds = stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
   912→  const stripePriceIdMonthly =
   913→    patch.stripePriceIdMonthly !== undefined
   914→      ? patch.stripePriceIdMonthly
   915→      : patch.stripePriceIdsByCurrency !== undefined
   916→        ? defIds.monthly
   917→        : current.stripePriceIdMonthly;
   918→  const stripePriceIdYearly =
   919→    patch.stripePriceIdYearly !== undefined
   920→      ? patch.stripePriceIdYearly
   921→      : patch.stripePriceIdsByCurrency !== undefined
   922→        ? defIds.yearly
   923→        : current.stripePriceIdYearly;
   924→  const isFree = patch.isFree ?? current.isFree;
   925→  // ---- Billing periods ----
   926→  const billingMonthly = patch.billingMonthly ?? current.billingMonthly;
   927→  const billingYearly = patch.billingYearly ?? current.billingYearly;
   928→  // The default cadence must be an ENABLED period (single-period plans
   929→  // are pinned to their only period; both-period plans keep the
   930→  // admin's requested / current cadence).
   931→  const interval = normalizeInterval(
   932→    patch.interval ?? current.interval,
   933→    billingMonthly,
   934→    billingYearly,
   935→  );
   936→  return {
   937→    planId: current.planId,
   938→    name: patch.name ?? current.name,
   939→    priceMonthly,
   940→    priceYearly,
   941→    currency: curUpper,
   942→    autoCurrency: patch.autoCurrency ?? current.autoCurrency,
   943→    pricesByCurrency,
   944→    stripePriceIdsByCurrency,
   945→    billingMonthly,
   946→    billingYearly,
   947→    interval,
   948→    isFree,
   949→    freePlanDurationDays: patch.freePlanDurationDays ?? current.freePlanDurationDays,
   950→    stripePriceIdMonthly,
   951→    stripePriceIdYearly,
   952→    active: patch.active ?? current.active,
   953→    features: patch.features ?? current.features,
   954→    // Entitlements: the patch's editor-managed keys replace the current
   955→    // ones, while NON-editor keys (e.g. 'audit_log') are PRESERVED from
   956→    // the current plan — the Create/Edit Plan modal only manages the 9
   957→    // Feature Access keys. Then normalize (legacy 'ai_content' →
   958→    // 'ai_platform'; the removed 'api_access' key is stripped; the two
   959→    // AI keys are independent).
   960→    entitlements: normalizeEntitlementKeys(
   961→      patch.entitlements !== undefined
   962→        ? [
   963→            ...patch.entitlements,
   964→            ...current.entitlements.filter(
   965→              (k) =>
   966→                !patch.entitlements!.includes(k) &&
   967→                k !== 'ai_content' &&
   968→                // keys below are editor-managed (replaced by the patch):
   969→                !PLAN_EDITOR_KEY_SET.has(k),
   970→            ),
   971→          ]
   972→        : current.entitlements,
   973→    ),
   974→    // Limits: merge; the AI usage limits are only part of the saved
   975→    // configuration while the plan includes Platform AI (zeroed
   976→    // otherwise — Client's Own AI API usage is never limited).
   977→    limits: zeroAiLimitsIfNotPlatform(
   978→      pickLimits({ ...current.limits, ...(patch.limits ?? {}) }),
   979→      patch.entitlements !== undefined
   980→        ? normalizeEntitlementKeys(patch.entitlements)
   981→        : current.entitlements,
   982→    ),
   983→    badgeVariant: patch.badgeVariant ?? current.badgeVariant,
   984→    sortOrder: patch.sortOrder ?? current.sortOrder,
   985→  };
   986→}

   988→/** Update an existing plan config. Writes DB, refreshes cache.
   989→ *
   990→ *  AUTO-SYNC TO STRIPE: when Stripe is configured AND the plan is
   991→ *  PAID (any currency has a positive price) AND the admin did NOT
   992→ *  explicitly set the Stripe Price IDs in this patch, the backend calls
   993→ *  syncPlanToStripe (multi-currency) to ensure a Stripe Product +
   994→ *  per-currency monthly+yearly Prices exist and writes the resolved IDs
   995→ *  into stripePriceIdsByCurrency + the default-currency snapshot fields.
   996→ */
   997→export async function savePlanConfig(planId: string, patch: PlanConfigInput): Promise<PlanConfigData | null> {
   998→  const existing = await db.planConfig.findUnique({ where: { planId } });
   999→  if (!existing) return null;
  1000→  const current = rowToData(existing);
  1001→
  1002→  // When the admin omitted currency, resolve the platform default so the
  1003→  // legacy priceMonthly/priceYearly snapshot stays in the right currency.
  1004→  if (!patch.currency && !current.currency) {
  1005→    patch = { ...patch, currency: await resolvePlatformDefaultCurrency() };
  1006→  }
  1007→  const finalCurrency = (patch.currency ?? current.currency).toUpperCase();
  1008→
  1009→  // Distinguish "field omitted" from "field set". When the admin omits
  1010→  // both stripePriceId* AND stripePriceIdsByCurrency, they want the
  1011→  // backend to auto-sync. When either is explicitly provided, manual.
  1012→  const adminTouchedStripePriceIds =
  1013→    'stripePriceIdMonthly' in patch ||
  1014→    'stripePriceIdYearly' in patch ||
  1015→    'stripePriceIdsByCurrency' in patch;

  1022→  // Best-effort — never throws. Only ENABLED billing periods get
  1023→  // Stripe Prices (a disabled period never creates one).
  1024→  const hasAnyPaidCurrency = Object.values(next.pricesByCurrency).some(
  1025→    (p) =>
  1026→      (next.billingMonthly && p.monthly > 0) ||
  1027→      (next.billingYearly && p.yearly > 0),
  1028→  );
  1029→  if (!adminTouchedStripePriceIds && !next.isFree && hasAnyPaidCurrency) {

  1036→          name: next.name,

  1039→          stripePriceIdsByCurrency: next.stripePriceIdsByCurrency,

  1046→          JSON.stringify(syncResult.stripePriceIdsByCurrency) !==
  1047→            JSON.stringify(next.stripePriceIdsByCurrency) ||

  1049→          snapshot.yearly !== next.stripePriceIdYearly;

  1055→              stripePriceIdMonthly: snapshot.monthly,
  1056→              stripePriceIdYearly: snapshot.yearly,
  1057→            },
  1058→          });
  1059→          await hydrate();
  1060→        }
  1061→      }
  1062→    } catch {
  1063→      // best-effort — never block the save on a Stripe error
  1064→    }
  1065→  }
  1066→
  1067→  return getPlanConfigSync(planId);

  1073→ *  calls syncPlanToStripeMulti (multi-currency) to create a Stripe

  1186→/**
  1187→ * Delete a plan config. Refuses if:
  1188→ *  - it's the last remaining plan, OR
  1189→ *  - any Subscription still references this planId (would orphan the sub).
  1190→ *
  1191→ * Returns `{ ok: boolean, reason?: string }` describing the outcome.
  1192→ */
  1193→export async function deletePlanConfig(
  1194→  planId: string,
  1195→): Promise<{ ok: boolean; reason?: string }> {
  1196→  const count = await db.planConfig.count();
  1197→  if (count <= 1) return { ok: false, reason: 'Cannot delete the last plan.' };
  1198→  const subCount = await db.subscription.count({ where: { planId } });
  1199→  if (subCount > 0) {
  1200→    return {
  1201→      ok: false,
  1202→      reason: `Cannot delete plan "${planId}" — ${subCount} active subscription(s) reference it. Migrate them first.`,
  1203→    };
  1204→  }
  1205→  try {
  1206→    await db.planConfig.delete({ where: { planId } });
  1207→    await hydrate();
  1208→    return { ok: true };
  1209→  } catch (err) {
  1210→    return { ok: false, reason: (err as Error).message };
  1211→  }
  1212→}
