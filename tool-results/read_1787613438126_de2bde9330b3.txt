   140→  providerId: '',
   141→  modelId: '',

   150→// -------------------- Component --------------------
   151→
   152→export function PromptsPage() {
   153→  const queryClient = useQueryClient();
   154→  const [page, setPage] = useState(1);
   155→  const [pageSize] = useState(25);
   156→  const [search, setSearch] = useState('');
   157→  const [categoryFilter, setCategoryFilter] = useState<string>('all');
   158→  const [favFilter, setFavFilter] = useState<FavFilter>('all');
   159→  const [viewMode, setViewMode] = useState<ViewMode>('table');
   160→  const [dialogOpen, setDialogOpen] = useState(false);
   161→  const [editingPrompt, setEditingPrompt] = useState<AiPrompt | null>(null);
   162→  const [formData, setFormData] = useState<PromptFormData>(emptyForm);
   163→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   164→  const [deletingPrompt, setDeletingPrompt] = useState<AiPrompt | null>(null);
   165→  const [versionsDialogOpen, setVersionsDialogOpen] = useState(false);
   166→  const [versionsPrompt, setVersionsPrompt] = useState<AiPrompt | null>(null);
   167→
   168→  // Fetch prompts
   169→  const { data, isLoading, isError } = useQuery({
   170→    queryKey: queryKeys.aiPrompts.list({ page, pageSize, search, category: categoryFilter, isFavorite: favFilter }),
   171→    queryFn: () => getApi<PaginatedResponse<AiPrompt>>('/api/ai/prompts', {
   172→      page, pageSize,
   173→      search: search || undefined,
   174→      category: categoryFilter !== 'all' ? categoryFilter : undefined,
   175→      isFavorite: favFilter === 'favorites' ? true : undefined,
   176→    }),
   177→  });
   178→
   179→  const prompts = data?.data ?? [];
   180→  const pagination = data?.pagination;
   181→
   182→  // Fetch active providers for the form
   183→  const { data: providersData } = useQuery({
   184→    queryKey: queryKeys.aiProviders.list({ isActive: true }),
   185→    queryFn: () => getApi<PaginatedResponse<AiProvider>>('/api/ai/providers', { isActive: true, pageSize: 100 }),
   186→  });
   187→  const activeProviders = providersData?.data ?? [];
   188→
   189→  // Fetch models filtered by selected provider
   190→  const { data: modelsData } = useQuery({
   191→    queryKey: queryKeys.aiModels.list({ providerId: formData.providerId || undefined }),
   192→    queryFn: () => getApi<PaginatedResponse<AiModel>>('/api/ai/models', {
   193→      providerId: formData.providerId || undefined,
   194→      pageSize: 100,
   195→    }),
   196→    enabled: !!formData.providerId,
   197→  });
   198→  const models = modelsData?.data ?? [];
   199→
   200→  // Fetch versions
   201→  const { data: versionsData } = useQuery({
   202→    queryKey: queryKeys.aiPrompts.nested('versions').list(versionsPrompt?.id ?? ''),
   203→    queryFn: () => getApi<PromptVersion[]>(`/api/ai/prompts/${versionsPrompt!.id}/versions`),
   204→    enabled: !!versionsPrompt,
   205→  });
   206→  const versions = versionsData ?? [];
   207→
   208→  // Create / Update mutation
   209→  const saveMutation = useMutation({
   210→    mutationFn: (body: PromptFormData) => {
   211→      const payload = {
   212→        ...body,
   213→        tags: body.tags.split(',').map((t) => t.trim()).filter(Boolean),
   214→        variables: (() => { try { return JSON.parse(body.variables); } catch { return {}; } })(),
   215→        providerId: body.providerId || null,
   216→        modelId: body.modelId || null,
   217→      };
   218→      if (editingPrompt) {
   219→        return patchApi<AiPrompt>(`/api/ai/prompts/${editingPrompt.id}`, payload);
   220→      }
   221→      return postApi<AiPrompt>('/api/ai/prompts', payload);
   222→    },
   223→    onSuccess: () => {
   224→      queryClient.invalidateQueries({ queryKey: queryKeys.aiPrompts.all });
   225→      toast.success(editingPrompt ? 'Prompt updated' : 'Prompt created');
   226→      handleCloseDialog();
   227→    },
   228→    onError: (err: Error) => toast.error(err.message || 'Failed to save prompt'),
   229→  });
   230→
   231→  // Delete mutation
   232→  const deleteMutation = useMutation({
   233→    mutationFn: (id: string) => deleteApi(`/api/ai/prompts/${id}`),
   234→    onSuccess: () => {
   235→      queryClient.invalidateQueries({ queryKey: queryKeys.aiPrompts.all });
   236→      toast.success('Prompt deleted');
   237→      setDeleteDialogOpen(false);
   238→      setDeletingPrompt(null);
   239→    },
   240→    onError: (err: Error) => toast.error(err.message || 'Failed to delete prompt'),
   241→  });
   242→
   243→  // Favorite toggle
   244→  const favMutation = useMutation({
   245→    mutationFn: (id: string) => postApi(`/api/ai/prompts/${id}/favorite`),
   246→    onSuccess: () => {
   247→      queryClient.invalidateQueries({ queryKey: queryKeys.aiPrompts.all });
   248→    },
   249→  });
   250→
   251→  // Duplicate mutation
   252→  const duplicateMutation = useMutation({
   253→    mutationFn: (id: string) => postApi(`/api/ai/prompts/${id}/duplicate`),
   254→    onSuccess: () => {
   255→      queryClient.invalidateQueries({ queryKey: queryKeys.aiPrompts.all });
   256→      toast.success('Prompt duplicated');
   257→    },
   258→    onError: (err: Error) => toast.error(err.message || 'Failed to duplicate'),
   259→  });
   260→
   261→  const handleOpenCreate = () => {
   262→    setEditingPrompt(null);
   263→    setFormData(emptyForm);
   264→    setDialogOpen(true);
   265→  };
   266→
   267→  const handleOpenEdit = (prompt: AiPrompt) => {
   268→    setEditingPrompt(prompt);
   269→    setFormData({
   270→      name: prompt.name,
   271→      category: prompt.category,
   272→      description: prompt.description ?? '',
   273→      tags: prompt.tags?.join(', ') ?? '',
   274→      variables: JSON.stringify(prompt.variables ?? {}, null, 2),
   275→      systemPrompt: prompt.systemPrompt,
   276→      userPrompt: prompt.userPrompt,
   277→      providerId: prompt.providerId ?? '',
   278→      modelId: prompt.modelId ?? '',
   279→      temperature: prompt.temperature,
   280→      maxTokens: prompt.maxTokens,
   281→      isActive: prompt.isActive,
   282→    });
   283→    setDialogOpen(true);
   284→  };
   285→
   286→  const handleCloseDialog = () => {
   287→    setDialogOpen(false);
   288→    setEditingPrompt(null);
   289→    setFormData(emptyForm);
   290→  };
   291→
   292→  const handleSave = () => {
   293→    if (!formData.name.trim()) {
   294→      toast.error('Name is required');
   295→      return;
   296→    }
   297→    saveMutation.mutate(formData);
   298→  };
   299→
   300→  return (
   301→    <div className="space-y-6">
   302→      {/* Filters & Actions */}
   303→      <Card>
   304→        <CardContent className="p-4">
   305→          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   306→            <h2 className="text-lg font-semibold">Prompt Library</h2>
   307→            <div className="flex items-center gap-2">
   308→              <div className="flex border rounded-md">
   309→                <Button
   310→                  variant={viewMode === 'table' ? 'secondary' : 'ghost'}
   311→                  size="sm" className="rounded-r-none"
   312→                  onClick={() => setViewMode('table')}
   313→                >
   314→                  <List className="h-4 w-4" />
   315→                </Button>
   316→                <Button
   317→                  variant={viewMode === 'grid' ? 'secondary' : 'ghost'}
   318→                  size="sm" className="rounded-l-none"
   319→                  onClick={() => setViewMode('grid')}
   320→                >
   321→                  <LayoutGrid className="h-4 w-4" />
   322→                </Button>
   323→              </div>
   324→              <Button onClick={handleOpenCreate} size="sm">
   325→                <Plus className="h-4 w-4 mr-2" />
   326→                Add Prompt
   327→              </Button>
   328→            </div>
   329→          </div>
   330→          <div className="flex flex-col sm:flex-row gap-3 mt-4">
   331→            <div className="relative flex-1">
   332→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-zinc-400" />
   333→              <Input
   334→                placeholder="Search prompts..."
   335→                value={search}
   336→                onChange={(e) => { setSearch(e.target.value); setPage(1); }}
   337→                className="pl-9"
   338→              />
   339→            </div>
   340→            <Select value={categoryFilter} onValueChange={(v) => { setCategoryFilter(v); setPage(1); }}>
   341→              <SelectTrigger className="w-full sm:w-[200px]">
   342→                <SelectValue placeholder="Category" />
   343→              </SelectTrigger>
   344→              <SelectContent>
   345→                <SelectItem value="all">All Categories</SelectItem>
   346→                {PROMPT_CATEGORIES.map((c) => (
   347→                  <SelectItem key={c} value={c}>{CATEGORY_LABELS[c]}</SelectItem>
   348→                ))}
   349→              </SelectContent>
   350→            </Select>
   351→            <Select value={favFilter} onValueChange={(v) => { setFavFilter(v as FavFilter); setPage(1); }}>
   352→              <SelectTrigger className="w-full sm:w-[140px]">
   353→                <SelectValue />
   354→              </SelectTrigger>
   355→              <SelectContent>
   356→                <SelectItem value="all">All</SelectItem>
   357→                <SelectItem value="favorites">Favorites</SelectItem>
   358→              </SelectContent>
   359→            </Select>
   360→          </div>
   361→        </CardContent>
   362→      </Card>
   363→
   364→      {/* Table View */}
   365→      {viewMode === 'table' && (
   366→        <Card>
   367→          <CardContent className="p-0">
   368→            <ScrollArea className="max-h-[500px]">
   369→              <Table>
   370→                <TableHeader>
   371→                  <TableRow>
   372→                    <TableHead>Name</TableHead>
   373→                    <TableHead>Category</TableHead>
   374→                    <TableHead className="hidden md:table-cell">Tags</TableHead>
   375→                    <TableHead className="hidden lg:table-cell">Vars</TableHead>
   376→                    <TableHead className="hidden lg:table-cell">Usage</TableHead>
   377→                    <TableHead>Fav</TableHead>
   378→                    <TableHead>Status</TableHead>
   379→                    <TableHead className="text-right">Actions</TableHead>
   380→                  </TableRow>
   381→                </TableHeader>
   382→                <TableBody>
   383→                  {isLoading ? (
   384→                    Array.from({ length: 5 }).map((_, i) => (
   385→                      <TableRow key={i}>{Array.from({ length: 8 }).map((_, j) => (
   386→                        <TableCell key={j}><Skeleton className="h-5 w-20" /></TableCell>
   387→                      ))}</TableRow>
   388→                    ))
   389→                  ) : isError ? (
   390→                    <TableRow><TableCell colSpan={8} className="text-center py-8 text-zinc-500">Failed to load prompts</TableCell></TableRow>
   391→                  ) : prompts.length === 0 ? (
   392→                    <TableRow><TableCell colSpan={8} className="text-center py-8 text-zinc-500">
   393→                      <MessageSquare className="h-8 w-8 mx-auto mb-2 text-zinc-300" />
   394→                      No prompts found. Create your first prompt to get started.
   395→                    </TableCell></TableRow>
   396→                  ) : prompts.map((prompt) => (
   397→                    <TableRow key={prompt.id}>
   398→                      <TableCell className="font-medium">{prompt.name}</TableCell>
   399→                      <TableCell>
   400→                        <Badge variant="secondary" className={CATEGORY_COLORS[prompt.category]}>
   401→                          {CATEGORY_LABELS[prompt.category]}
   402→                        </Badge>
   403→                      </TableCell>
   404→                      <TableCell className="hidden md:table-cell">
   405→                        <div className="flex flex-wrap gap-1">
   406→                          {prompt.tags?.slice(0, 3).map((tag) => (
   407→                            <Badge key={tag} variant="outline" className="text-xs">{tag}</Badge>
   408→                          ))}
   409→                          {(prompt.tags?.length ?? 0) > 3 && (
   410→                            <Badge variant="outline" className="text-xs">+{prompt.tags!.length - 3}</Badge>
   411→                          )}
   412→                        </div>
   413→                      </TableCell>
   414→                      <TableCell className="hidden lg:table-cell">
   415→                        {prompt.variables ? Object.keys(prompt.variables).length : 0}
   416→                      </TableCell>
   417→                      <TableCell className="hidden lg:table-cell">{prompt.usageCount}</TableCell>
   418→                      <TableCell>
   419→                        <Button variant="ghost" size="sm" className="h-7 w-7 p-0" onClick={() => favMutation.mutate(prompt.id)}>
   420→                          <Heart className={`h-4 w-4 ${prompt.isFavorite ? 'fill-red-500 text-red-500' : 'text-zinc-400'}`} />
   421→                        </Button>
   422→                      </TableCell>
   423→                      <TableCell>
   424→                        <Badge variant={prompt.isActive ? 'default' : 'secondary'} className={prompt.isActive ? 'bg-green-100 text-green-700' : ''}>
   425→                          {prompt.isActive ? 'Active' : 'Inactive'}
   426→                        </Badge>
   427→                      </TableCell>
   428→                      <TableCell className="text-right">
   429→                        <DropdownMenu>
   430→                          <DropdownMenuTrigger asChild>
   431→                            <Button variant="ghost" size="sm"><MoreHorizontal className="h-4 w-4" /></Button>
   432→                          </DropdownMenuTrigger>
   433→                          <DropdownMenuContent align="end">
   434→                            <DropdownMenuItem onClick={() => handleOpenEdit(prompt)}><Pencil className="h-4 w-4 mr-2" />Edit</DropdownMenuItem>
   435→                            <DropdownMenuItem onClick={() => duplicateMutation.mutate(prompt.id)}><Copy className="h-4 w-4 mr-2" />Duplicate</DropdownMenuItem>
   436→                            <DropdownMenuItem onClick={() => { setVersionsPrompt(prompt); setVersionsDialogOpen(true); }}><History className="h-4 w-4 mr-2" />Version History</DropdownMenuItem>
   437→                            <DropdownMenuSeparator />
   438→                            <DropdownMenuItem className="text-red-600" onClick={() => { setDeletingPrompt(prompt); setDeleteDialogOpen(true); }}><Trash2 className="h-4 w-4 mr-2" />Delete</DropdownMenuItem>
   439→                          </DropdownMenuContent>
   440→                        </DropdownMenu>
   441→                      </TableCell>
   442→                    </TableRow>
   443→                  ))}
   444→                </TableBody>
   445→              </Table>
   446→              <ScrollBar />
   447→            </ScrollArea>
   448→
   449→            {pagination && pagination.totalPages > 1 && (
   450→              <div className="flex items-center justify-between px-4 py-3 border-t">
   451→                <p className="text-sm text-zinc-500">
   452→                  {(pagination.page - 1) * pagination.pageSize + 1}–{Math.min(pagination.page * pagination.pageSize, pagination.total)} of {pagination.total}
   453→                </p>
   454→                <div className="flex items-center gap-2">
   455→                  <Button variant="outline" size="sm" disabled={pagination.page <= 1} onClick={() => setPage((p) => p - 1)}><ChevronLeft className="h-4 w-4" /></Button>
   456→                  <span className="text-sm">{pagination.page} / {pagination.totalPages}</span>
   457→                  <Button variant="outline" size="sm" disabled={pagination.page >= pagination.totalPages} onClick={() => setPage((p) => p + 1)}><ChevronRight className="h-4 w-4" /></Button>
   458→                </div>
   459→              </div>
   460→            )}
   461→          </CardContent>
   462→        </Card>
   463→      )}
   464→
   465→      {/* Grid View */}
   466→      {viewMode === 'grid' && (
   467→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   468→          {isLoading ? (
   469→            Array.from({ length: 6 }).map((_, i) => (
   470→              <Card key={i}><CardContent className="p-4 space-y-3"><Skeleton className="h-5 w-3/4" /><Skeleton className="h-4 w-1/2" /><Skeleton className="h-16 w-full" /></CardContent></Card>
   471→            ))
   472→          ) : prompts.length === 0 ? (
   473→            <div className="col-span-full text-center py-12 text-zinc-500">
   474→              <MessageSquare className="h-10 w-10 mx-auto mb-3 text-zinc-300" />
   475→              No prompts found. Create your first prompt.
   476→            </div>
   477→          ) : prompts.map((prompt) => (
   478→            <Card key={prompt.id} className="hover:shadow-md transition-shadow">
   479→              <CardContent className="p-4 space-y-3">
   480→                <div className="flex items-start justify-between">
   481→                  <div className="flex-1 min-w-0">
   482→                    <h3 className="font-semibold truncate">{prompt.name}</h3>
   483→                    <Badge variant="secondary" className={`${CATEGORY_COLORS[prompt.category]} mt-1`}>
   484→                      {CATEGORY_LABELS[prompt.category]}
   485→                    </Badge>
   486→                  </div>
   487→                  <Button variant="ghost" size="sm" className="h-7 w-7 p-0 shrink-0" onClick={() => favMutation.mutate(prompt.id)}>
   488→                    <Heart className={`h-4 w-4 ${prompt.isFavorite ? 'fill-red-500 text-red-500' : 'text-zinc-400'}`} />
   489→                  </Button>
   490→                </div>
   491→                {prompt.description && <p className="text-sm text-zinc-500 line-clamp-2">{prompt.description}</p>}
   492→                <div className="flex flex-wrap gap-1">
   493→                  {prompt.tags?.slice(0, 3).map((tag) => (
   494→                    <Badge key={tag} variant="outline" className="text-xs">{tag}</Badge>
   495→                  ))}
   496→                </div>
   497→                <div className="flex items-center justify-between text-xs text-zinc-500">
   498→                  <span>{prompt.usageCount} uses</span>
   499→                  <Badge variant={prompt.isActive ? 'default' : 'secondary'} className={prompt.isActive ? 'bg-green-100 text-green-700' : ''}>
   500→                    {prompt.isActive ? 'Active' : 'Inactive'}
   501→                  </Badge>
   502→                </div>
   503→                <div className="flex gap-2 pt-1">
   504→                  <Button variant="outline" size="sm" className="flex-1" onClick={() => handleOpenEdit(prompt)}><Pencil className="h-3.5 w-3.5 mr-1" />Edit</Button>
   505→                  <Button variant="outline" size="sm" onClick={() => duplicateMutation.mutate(prompt.id)}><Copy className="h-3.5 w-3.5" /></Button>
   506→                  <Button variant="outline" size="sm" className="text-red-600" onClick={() => { setDeletingPrompt(prompt); setDeleteDialogOpen(true); }}><Trash2 className="h-3.5 w-3.5" /></Button>
   507→                </div>
   508→              </CardContent>
   509→            </Card>
   510→          ))}
   511→        </div>
   512→      )}
   513→
   514→      {/* Grid pagination */}
   515→      {viewMode === 'grid' && pagination && pagination.totalPages > 1 && (
   516→        <div className="flex items-center justify-center gap-2">
   517→          <Button variant="outline" size="sm" disabled={pagination.page <= 1} onClick={() => setPage((p) => p - 1)}><ChevronLeft className="h-4 w-4" /></Button>
   518→          <span className="text-sm">Page {pagination.page} of {pagination.totalPages}</span>
   519→          <Button variant="outline" size="sm" disabled={pagination.page >= pagination.totalPages} onClick={() => setPage((p) => p + 1)}><ChevronRight className="h-4 w-4" /></Button>
   520→        </div>
   521→      )}
   522→
   523→      {/* Add/Edit Dialog */}
   524→      <Dialog open={dialogOpen} onOpenChange={(open) => { if (!open) handleCloseDialog(); }}>
   525→        <DialogContent className="sm:max-w-[600px] max-h-[90vh] overflow-y-auto">
   526→          <DialogHeader>
   527→            <DialogTitle>{editingPrompt ? 'Edit Prompt' : 'Create Prompt'}</DialogTitle>
   528→          </DialogHeader>
   529→          <div className="grid gap-4 py-4">
   530→            <div className="grid gap-2">
   531→              <Label htmlFor="prompt-name">Name</Label>
   532→              <Input id="prompt-name" placeholder="e.g. Blog Post Writer" value={formData.name} onChange={(e) => setFormData((p) => ({ ...p, name: e.target.value }))} />
   533→            </div>
   534→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   535→              <div className="grid gap-2">
   536→                <Label>Category</Label>
   537→                <Select value={formData.category} onValueChange={(v) => setFormData((p) => ({ ...p, category: v as PromptCategoryNew }))}>
   538→                  <SelectTrigger><SelectValue /></SelectTrigger>
   539→                  <SelectContent>
   540→                    {PROMPT_CATEGORIES.map((c) => (
   541→                      <SelectItem key={c} value={c}>{CATEGORY_LABELS[c]}</SelectItem>
   542→                    ))}
   543→                  </SelectContent>
   544→                </Select>
   545→              </div>
   546→              <div className="grid gap-2">
   547→                <Label htmlFor="prompt-tags">Tags (comma-separated)</Label>
   548→                <Input id="prompt-tags" placeholder="blog, seo, content" value={formData.tags} onChange={(e) => setFormData((p) => ({ ...p, tags: e.target.value }))} />
   549→              </div>
   550→            </div>
   551→            <div className="grid gap-2">
   552→              <Label htmlFor="prompt-desc">Description</Label>
   553→              <Input id="prompt-desc" placeholder="Brief description..." value={formData.description} onChange={(e) => setFormData((p) => ({ ...p, description: e.target.value }))} />
   554→            </div>
   555→            <div className="grid gap-2">
   556→              <Label htmlFor="prompt-vars">Variables (JSON)</Label>
   557→              <Textarea id="prompt-vars" rows={3} placeholder='{"topic": "", "tone": "", "length": 500}' value={formData.variables} onChange={(e) => setFormData((p) => ({ ...p, variables: e.target.value }))} />
   558→              <p className="text-xs text-zinc-400">Define variables as JSON. Use {"{{variable_name}}"} in prompts.</p>
   559→            </div>
   560→            <div className="grid gap-2">
   561→              <Label htmlFor="prompt-system">System Prompt</Label>
   562→              <Textarea id="prompt-system" rows={4} placeholder="You are a helpful assistant..." value={formData.systemPrompt} onChange={(e) => setFormData((p) => ({ ...p, systemPrompt: e.target.value }))} />
   563→            </div>
   564→            <div className="grid gap-2">
   565→              <Label htmlFor="prompt-user">User Prompt</Label>
   566→              <Textarea id="prompt-user" rows={4} placeholder="Write a {{topic}} article in a {{tone}} tone..." value={formData.userPrompt} onChange={(e) => setFormData((p) => ({ ...p, userPrompt: e.target.value }))} />
   567→              <p className="text-xs text-zinc-400">Use {"{{variable_name}}"} syntax for dynamic content.</p>
   568→            </div>
   569→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   570→              <div className="grid gap-2">
   571→                <Label>Provider</Label>
   572→                <Select value={formData.providerId} onValueChange={(v) => setFormData((p) => ({ ...p, providerId: v, modelId: '' }))}>
   573→                  <SelectTrigger><SelectValue placeholder="Select provider" /></SelectTrigger>
   574→                  <SelectContent>
   575→                    {activeProviders.map((p) => (
   576→                      <SelectItem key={p.id} value={p.id}>{p.name}</SelectItem>
   577→                    ))}
   578→                  </SelectContent>
   579→                </Select>
   580→              </div>
   581→              <div className="grid gap-2">
   582→                <Label>Model</Label>
   583→                <Select
   584→                  value={formData.modelId}
   585→                  onValueChange={(v) => setFormData((p) => ({ ...p, modelId: v }))}
   586→                  disabled={!formData.providerId}
   587→                >
   588→                  <SelectTrigger>
   589→                    <SelectValue placeholder={formData.providerId ? 'Select model' : 'Select provider first'} />
   590→                  </SelectTrigger>
   591→                  <SelectContent>
   592→                    {models.map((m) => (
   593→                      <SelectItem key={m.id} value={m.id}>{m.name}</SelectItem>
   594→                    ))}
   595→                  </SelectContent>
   596→                </Select>
   597→                {formData.providerId && models.length === 0 && (
   598→                  <p className="text-xs text-muted-foreground">No active models for this provider. Add models in the Models tab.</p>
   599→                )}
   600→              </div>
   601→            </div>
   602→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   603→              <div className="grid gap-2">
   604→                <Label>Temperature: {formData.temperature}</Label>
   605→                <Slider min={0} max={2} step={0.1} value={[formData.temperature]} onValueChange={([v]) => setFormData((p) => ({ ...p, temperature: v }))} />
   606→              </div>
   607→              <div className="grid gap-2">
   608→                <Label htmlFor="prompt-tokens">Max Tokens</Label>
   609→                <Input id="prompt-tokens" type="number" min={1} max={128000} value={formData.maxTokens} onChange={(e) => setFormData((p) => ({ ...p, maxTokens: parseInt(e.target.value) || 2048 }))} />
   610→              </div>
   611→            </div>
   612→            <div className="flex items-center justify-between">
   613→              <Label htmlFor="prompt-active">Active</Label>
   614→              <Switch id="prompt-active" checked={formData.isActive} onCheckedChange={(checked) => setFormData((p) => ({ ...p, isActive: checked }))} />
   615→            </div>
   616→          </div>
   617→          <DialogFooter>
   618→            <Button variant="outline" onClick={handleCloseDialog}>Cancel</Button>
   619→            <Button onClick={handleSave} disabled={saveMutation.isPending}>
   620→              {saveMutation.isPending && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   621→              {editingPrompt ? 'Update' : 'Create'}
   622→            </Button>
   623→          </DialogFooter>
   624→        </DialogContent>
   625→      </Dialog>
   626→
   627→      {/* Version History Dialog */}
   628→      <Dialog open={versionsDialogOpen} onOpenChange={setVersionsDialogOpen}>
   629→        <DialogContent className="sm:max-w-[600px] max-h-[70vh]">
   630→          <DialogHeader>
   631→            <DialogTitle>Version History — {versionsPrompt?.name}</DialogTitle>
   632→          </DialogHeader>
   633→          <ScrollArea className="max-h-[500px]">
   634→            {versions.length === 0 ? (
   635→              <p className="text-center py-8 text-zinc-500">No versions found.</p>
   636→            ) : (
   637→              <div className="space-y-4 p-1">
   638→                {versions.map((v) => (
   639→                  <Card key={v.id}>
   640→                    <CardContent className="p-4 space-y-2">
   641→                      <div className="flex items-center justify-between">
   642→                        <Badge variant="secondary">v{v.version}</Badge>
   643→                        <span className="text-xs text-zinc-500">{new Date(v.createdAt).toLocaleString()}</span>
   644→                      </div>
   645→                      {v.systemPrompt && (
   646→                        <div>
   647→                          <p className="text-xs font-medium text-zinc-500">System:</p>
   648→                          <p className="text-sm text-zinc-700 line-clamp-2">{v.systemPrompt}</p>
   649→                        </div>
   650→                      )}
   651→                      {v.userPrompt && (
   652→                        <div>
   653→                          <p className="text-xs font-medium text-zinc-500">User:</p>
   654→                          <p className="text-sm text-zinc-700 line-clamp-2">{v.userPrompt}</p>
   655→                        </div>
   656→                      )}
   657→                    </CardContent>
   658→                  </Card>
   659→                ))}
   660→              </div>
   661→            )}
   662→          </ScrollArea>
   663→        </DialogContent>
   664→      </Dialog>
   665→
   666→      {/* Delete Confirmation */}
   667→      <AlertDialog open={deleteDialogOpen} onOpenChange={setDeleteDialogOpen}>
   668→        <AlertDialogContent>
   669→          <AlertDialogHeader>
   670→            <AlertDialogTitle>Delete Prompt</AlertDialogTitle>
   671→            <AlertDialogDescription>Are you sure you want to delete &quot;{deletingPrompt?.name}&quot;?</AlertDialogDescription>
   672→          </AlertDialogHeader>
   673→          <AlertDialogFooter>
   674→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   675→            <AlertDialogAction className="bg-red-600 hover:bg-red-700" onClick={() => deletingPrompt && deleteMutation.mutate(deletingPrompt.id)} disabled={deleteMutation.isPending}>
   676→              {deleteMutation.isPending && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}Delete
   677→            </AlertDialogAction>
   678→          </AlertDialogFooter>
   679→        </AlertDialogContent>
   680→      </AlertDialog>
   681→    </div>
   682→  );
   683→}
