   107→// ==================== Validation ====================
   108→
   109→function validateRedirectForm(form: RedirectFormData, isEdit: boolean): string | null {
   110→  if (!form.fromPath.trim()) return 'From URL is required.';
   111→  if (!form.fromPath.startsWith('/')) return 'From URL must start with "/".';
   112→  if (!form.toPath.trim()) return 'To URL is required.';
   113→  if (!form.toPath.startsWith('/')) return 'To URL must start with "/".';
   114→  if (!isEdit && form.fromPath.trim() === form.toPath.trim()) {
   115→    return 'From URL and To URL cannot be the same (self-redirect).';
   116→  }
   117→  if (form.fromPath.trim().toLowerCase() === form.toPath.trim().toLowerCase()) {
   118→    return 'From URL and To URL cannot be the same (self-redirect).';
   119→  }
   120→  return null;
   121→}

   257→  invalidRows: number;

   267→function CsvImportDialog({ open, onOpenChange }: { open: boolean; onOpenChange: (v: boolean) => void }) {
   268→  const queryClient = useQueryClient();
   269→  const [csvContent, setCsvContent] = useState('');
   270→  const [file, setFile] = useState<File | null>(null);
   271→  const [validation, setValidation] = useState<CsvValidationResult | null>(null);
   272→  const [importResult, setImportResult] = useState<CsvImportResult | null>(null);
   273→  const [step, setStep] = useState<'upload' | 'preview' | 'done'>('upload');
   274→
   275→  const validateMutation = useMutation({
   276→    mutationFn: (csv: string) => postApi<CsvValidationResult>('/api/redirects/bulk?action=import', { csvContent: csv }),
   277→    onSuccess: (data) => { setValidation(data as unknown as CsvValidationResult); setStep('preview'); },
   278→    onError: () => toast.error('Failed to validate CSV'),
   279→  });
   280→
   281→  const importMutation = useMutation({
   282→    mutationFn: () => postApi<CsvImportResult>('/api/redirects/bulk?action=import&confirm=true', { csvContent }),
   283→    onSuccess: (data) => { setImportResult(data as unknown as CsvImportResult); setStep('done'); queryClient.invalidateQueries({ queryKey: queryKeys.redirects.all }); },
   284→    onError: () => toast.error('Import failed'),
   285→  });
   286→
   287→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   288→    const f = e.target.files?.[0];
   289→    if (!f) return;
   290→    setFile(f);
   291→    const reader = new FileReader();
   292→    reader.onload = (ev) => { const text = ev.target?.result as string; setCsvContent(text); setValidation(null); setImportResult(null); setStep('upload'); };
   293→    reader.readAsText(f);
   294→  };
   295→
   296→  const handleDrop = (e: React.DragEvent) => {
   297→    e.preventDefault();
   298→    const f = e.dataTransfer.files[0];
   299→    if (!f) return;
   300→    setFile(f);
   301→    const reader = new FileReader();
   302→    reader.onload = (ev) => { const text = ev.target?.result as string; setCsvContent(text); setValidation(null); setImportResult(null); setStep('upload'); };
   303→    reader.readAsText(f);
   304→  };
   305→
   306→  const handleValidate = () => { if (csvContent.trim()) validateMutation.mutate(csvContent); };
   307→  const handleImport = () => { importMutation.mutate(); };
   308→  const handleClose = () => { onOpenChange(false); setTimeout(() => { setCsvContent(''); setFile(null); setValidation(null); setImportResult(null); setStep('upload'); }, 200); };
   309→
   310→  return (
   311→    <Dialog open={open} onOpenChange={handleClose}>
   312→      <DialogContent className="sm:max-w-[600px] max-h-[80vh] overflow-y-auto">
   313→        <DialogHeader><DialogTitle>Import Redirects CSV</DialogTitle><DialogDescription>Upload a CSV file with columns: from, to, type (301, 302, 307, 308)</DialogDescription></DialogHeader>
   314→
   315→        {step === 'upload' && (
   316→          <div className="space-y-4 py-2">
   317→            <div className="border-2 border-dashed rounded-lg p-8 text-center cursor-pointer hover:border-primary/50 transition-colors" onDragOver={(e) => e.preventDefault()} onDrop={handleDrop} onClick={() => document.getElementById('csv-file-input')?.click()}>
   318→              <input id="csv-file-input" type="file" accept=".csv" className="hidden" onChange={handleFileChange} />
   319→              <Upload className="h-8 w-8 mx-auto text-muted-foreground/50 mb-2" />
   320→              <p className="text-sm font-medium">Drop CSV file here or click to browse</p>
   321→              {file && <p className="text-xs text-muted-foreground mt-1">{file.name}</p>}
   322→            </div>
   323→            <div className="space-y-2">
   324→              <Label>Or paste CSV content:</Label>
   325→              <textarea className="w-full min-h-[120px] rounded-md border border-border bg-transparent px-3 py-2 text-xs font-mono shadow-sm placeholder:text-muted-foreground focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring resize-y" placeholder="from,to,type
   326→/old-page,/new-page,301" value={csvContent} onChange={(e) => setCsvContent(e.target.value)} />
   327→            </div>
   328→            <DialogFooter>
   329→              <Button variant="outline" onClick={handleClose}>Cancel</Button>
   330→              <Button onClick={handleValidate} disabled={!csvContent.trim() || validateMutation.isPending}>{validateMutation.isPending && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}Validate & Preview</Button>
   331→            </DialogFooter>
   332→          </div>
   333→        )}
   334→
   335→        {step === 'preview' && validation && (
   336→          <div className="space-y-4 py-2">
   337→            <div className="grid grid-cols-3 gap-3">
   338→              <div className="rounded-lg border p-3 text-center"><p className="text-lg font-bold tabular-nums text-green-600 dark:text-green-400">{validation.validRows}</p><p className="text-xs text-muted-foreground">Valid</p></div>
   339→              <div className="rounded-lg border p-3 text-center"><p className="text-lg font-bold tabular-nums text-red-600 dark:text-red-400">{validation.invalidRows}</p><p className="text-xs text-muted-foreground">Invalid</p></div>
   340→              <div className="rounded-lg border p-3 text-center"><p className="text-lg font-bold tabular-nums">{validation.errors.length}</p><p className="text-xs text-muted-foreground">Errors</p></div>
   341→            </div>
   342→            {validation.errors.length > 0 && (
   343→              <div className="max-h-40 overflow-y-auto rounded-lg border p-3 space-y-1">
   344→                {validation.errors.map((err, i) => (<div key={i} className="flex items-start gap-2 text-xs"><XCircle className="h-3 w-3 mt-0.5 text-red-500 shrink-0" /><span>Row {err.row}: {err.message}</span></div>))}
   345→              </div>
   346→            )}
   347→            <DialogFooter>
   348→              <Button variant="outline" onClick={() => setStep('upload')}>Back</Button>
   349→              <Button onClick={handleImport} disabled={validation.validRows === 0 || importMutation.isPending}>{importMutation.isPending && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}{importMutation.isPending ? 'Importing...' : `Import ${validation.validRows} Redirects`}</Button>
   350→            </DialogFooter>
   351→          </div>
   352→        )}
   353→
   354→        {step === 'done' && importResult && (
   355→          <div className="space-y-4 py-2 text-center">
   356→            <CheckCircle2 className="h-10 w-10 mx-auto text-green-500" />
   357→            <p className="font-semibold">Import Complete</p>
   358→            <div className="grid grid-cols-3 gap-3 max-w-xs mx-auto">
   359→              <div><p className="text-lg font-bold tabular-nums text-green-600">{importResult.imported}</p><p className="text-xs text-muted-foreground">Imported</p></div>
   360→              <div><p className="text-lg font-bold tabular-nums text-amber-600">{importResult.skipped}</p><p className="text-xs text-muted-foreground">Skipped</p></div>
   361→              <div><p className="text-lg font-bold tabular-nums text-red-600">{importResult.errorsDuringImport}</p><p className="text-xs text-muted-foreground">Errors</p></div>
   362→            </div>
   363→            <DialogFooter className="justify-center"><Button onClick={handleClose}>Done</Button></DialogFooter>
   364→          </div>
   365→        )}
   366→      </DialogContent>
   367→    </Dialog>
   368→  );
   369→}
