     1→// ============================================================
     2→// PLAN CONFIG — DB-backed, cached single source of truth for plans,
     3→// pricing, entitlements and usage limits.
     4→// ============================================================
     5→// The Client Billing API and the Platform Admin both read from (and
     6→// the admin mutates) this same service. `platform-data.ts` delegates
     7→// its `PLANS` / `getPlan` / `monthlyPrice` to the synchronous cache
     8→// exposed here, so an owner edit (savePlanConfig) propagates to the
     9→// client billing experience and to MRR on the next read.
    10→//
    11→// MULTI-CURRENCY: the plan carries ONE base price configuration —
    12→// priceMonthly / priceYearly denominated in the plan's default /
    13→// fallback `currency` (both edited directly in the Edit Plan modal).
    14→// The customer's currency is resolved SERVER-SIDE from their IP
    15→// geolocation (see country-pricing.ts:resolveCustomerPricing) when
    16→// the plan's `autoCurrency` flag is on; otherwise every customer is
    17→// billed in the plan default currency.
    18→//
    19→// pricesByCurrency is a DERIVED store (NOT edited per-plan in the
    20→// modal): the default currency's entry mirrors the base price on
    21→// every save, and the other entries hold the platform-level
    22→// regional prices (CountryPricing.regionalPrices + legacy entries)
    23→// — preserved across saves so customers in other currencies keep
    24→// seeing their configured price (e.g. "90 MAD / month").
    25→//
    26→// STRIPE: for each paid plan × supported currency × interval, a real
    27→// Stripe Price is created on sync and its ID persisted in
    28→// stripePriceIdsByCurrency. The default currency's IDs are mirrored
    29→// in stripePriceIdMonthly/Yearly for legacy callers. Checkout always
    30→// resolves the per-currency Stripe Price ID server-side — the
    31→// customer's currency determines which Price is charged.
    32→//
    33→// The cache is hydrated from the PlanConfig table on startup and
    34→// after every mutation. If the table is empty on first hydrate, the
    35→// DEFAULT configs are seeded.
    36→// ============================================================

    93→// -------------------- Entitlement normalization --------------------
    94→
    95→/** Normalize a raw entitlement key list (DB row or API input):
    96→ *  - legacy 'ai_content' → 'ai_platform' (it predates the two-mode
    97→ *    split and always meant the platform-provided AI).
    98→ *  - mutual exclusion: when BOTH AI modes are present (invalid data),
    99→ *    Platform AI wins and 'ai_client' is dropped.
   100→ *  - dedupes while preserving order.
   101→ *  Non-editor keys (e.g. 'audit_log') pass through untouched. */
   102→export function normalizeEntitlementKeys(keys: readonly string[]): string[] {
   103→  const out: string[] = [];
   104→  const seen = new Set<string>();
   105→  const hasPlatform = keys.includes('ai_platform') || keys.includes('ai_content');
   106→  for (const k of keys) {
   107→    if (typeof k !== 'string' || !k.trim()) continue;
   108→    if (k === 'ai_content') continue; // legacy → replaced by ai_platform below
   109→    if (k === 'ai_client' && hasPlatform) continue; // mutually exclusive → platform wins
   110→    if (k === 'ai_platform' && hasPlatform) {
   111→      if (!seen.has('ai_platform')) out.push('ai_platform');
   112→      seen.add('ai_platform');
   113→      continue;
   114→    }
   115→    if (!seen.has(k)) {
   116→      out.push(k);
   117→      seen.add(k);
   118→    }
   119→  }
   120→  // Insert the platform AI key at the position where the legacy key
   121→  // (or an existing 'ai_platform') first appeared — append at the end
   122→  // when neither was present in the ordered loop above (defensive).
   123→  if (hasPlatform && !seen.has('ai_platform')) out.push('ai_platform');
   124→  return out;
   125→}

   146→/** Pick only the known limit fields from an untyped/partial input —
   147→ *  unknown keys (stale `aiWords` / `automationRuns` from older rows)

   149→function pickLimits(parsed: unknown): PlanLimits {
   150→  const p = (parsed ?? {}) as Partial<PlanLimits>;
   151→  const num = (v: unknown): number => (typeof v === 'number' && !Number.isNaN(v) ? v : 0);
   152→  return {
   153→    maxSites: num(p.maxSites),
   154→    storageBytes: num(p.storageBytes),
   155→    aiArticlesPerMonth: num(p.aiArticlesPerMonth),
   156→    aiWordsPerMonth: num(p.aiWordsPerMonth),

   158→  };

   161→export type BillingInterval = 'monthly' | 'yearly';

   163→/** The billing periods enabled on a plan (at least one, enforced).
   164→ *  A disabled period: not shown on the Client Billing page, rejected
   165→ *  at checkout (400 BILLING_PERIOD_NOT_ENABLED), and never gets a
   166→ *  Stripe Price created or selected. */
   167→export function enabledIntervalsOf(plan: {
   168→  billingMonthly: boolean;
   169→  billingYearly: boolean;
   170→}): BillingInterval[] {
   171→  const out: BillingInterval[] = [];
   172→  if (plan.billingMonthly) out.push('monthly');
   173→  if (plan.billingYearly) out.push('yearly');
   174→  return out;
   175→}

   177→/** Clamp a requested default cadence to an ENABLED period:
   178→ *  only monthly on → 'monthly'; only yearly on → 'yearly'; both on
   179→ *  → the requested value. (Neither enabled is rejected upstream —
   180→ *  here it falls back to the requested value.) */
   181→function normalizeInterval(
   182→  requested: BillingInterval,
   183→  billingMonthly: boolean,
   184→  billingYearly: boolean,
   185→): BillingInterval {
   186→  if (billingMonthly && !billingYearly) return 'monthly';
   187→  if (!billingMonthly && billingYearly) return 'yearly';
   188→  return requested;
   189→}

   435→function notify() {
   436→  for (const cb of _subscribers) {
   437→    try {
   438→      cb();
   439→    } catch {
   440→      // ignore subscriber errors
   441→    }
   442→  }
   443→}

   445→/** Register a callback fired after every cache refresh (hydrate / mutation). */
   446→export function subscribe(cb: () => void): () => void {
   447→  _subscribers.add(cb);
   448→  return () => _subscribers.delete(cb);
   449→}

   518→function rowToData(row: PlanConfigRow): PlanConfigData {
   519→  let features: string[] = [];
   520→  let entitlements: string[] = [];
   521→  try {
   522→    features = JSON.parse(row.features || '[]');
   523→  } catch {
   524→    features = [];
   525→  }
   526→  try {
   527→    entitlements = JSON.parse(row.entitlements || '[]');
   528→  } catch {
   529→    entitlements = [];
   530→  }
   531→  // Explicitly pick only the known limit fields. Older DB rows may still
   532→  // carry stale `aiWords` / `aiArticles` / `automationRuns` keys in the
   533→  // limits JSON (those were removed because no real enforcement exists).
   534→  // Picking only the known fields prevents the stale keys from leaking
   535→  // back into the cache and being re-serialized on the next save.
   536→  let limits: PlanLimits = { ...DEFAULT_LIMITS };
   537→  try {
   538→    limits = pickLimits(JSON.parse(row.limits || '{}'));
   539→  } catch {
   540→    // keep defaults
   541→  }
   542→  // AI usage limits only exist for Platform AI plans — normalize the
   543→  // stored configuration to match the entitlement mode.
   544→  limits = zeroAiLimitsIfNotPlatform(limits, entitlements);
   545→  let pricesByCurrency: PricesByCurrency = {};
   546→  try {
   547→    const parsed = JSON.parse(row.pricesByCurrency || '{}') as Partial<PricesByCurrency>;
   548→    if (parsed && typeof parsed === 'object') {
   549→      for (const [cur, val] of Object.entries(parsed)) {
   550→        if (val && typeof val === 'object') {
   551→          const m = typeof (val as CurrencyPrice).monthly === 'number' ? (val as CurrencyPrice).monthly : 0;
   552→          const y = typeof (val as CurrencyPrice).yearly === 'number' ? (val as CurrencyPrice).yearly : 0;
   553→          pricesByCurrency[cur.toUpperCase()] = { monthly: m, yearly: y };
   554→        }
   555→      }
   556→    }
   557→  } catch {
   558→    pricesByCurrency = {};
   559→  }
   560→  let stripePriceIdsByCurrency: StripePriceIdsByCurrency = {};
   561→  try {
   562→    const parsed = JSON.parse(row.stripePriceIdsByCurrency || '{}') as Partial<StripePriceIdsByCurrency>;
   563→    if (parsed && typeof parsed === 'object') {
   564→      for (const [cur, val] of Object.entries(parsed)) {
   565→        if (val && typeof val === 'object') {
   566→          const m = typeof (val as CurrencyStripeIds).monthly === 'string' ? (val as CurrencyStripeIds).monthly : null;
   567→          const y = typeof (val as CurrencyStripeIds).yearly === 'string' ? (val as CurrencyStripeIds).yearly : null;
   568→          stripePriceIdsByCurrency[cur.toUpperCase()] = { monthly: m, yearly: y };
   569→        }
   570→      }
   571→    }
   572→  } catch {
   573→    stripePriceIdsByCurrency = {};
   574→  }
   575→  return {
   576→    planId: row.planId,
   577→    name: row.name,
   578→    priceMonthly: row.priceMonthly,
   579→    priceYearly: row.priceYearly,
   580→    currency: row.currency,
   581→    autoCurrency: row.autoCurrency ?? true,
   582→    pricesByCurrency,
   583→    stripePriceIdsByCurrency,
   584→    billingMonthly: row.billingMonthly ?? true,
   585→    billingYearly: row.billingYearly ?? true,
   586→    // Normalize the default cadence to an ENABLED period (defensive:
   587→    // legacy rows / stale values never advertise a disabled period).
   588→    interval: normalizeInterval(
   589→      (row.interval === 'yearly' ? 'yearly' : 'monthly') as BillingInterval,
   590→      row.billingMonthly ?? true,
   591→      row.billingYearly ?? true,
   592→    ),
   593→    isFree: row.isFree,
   594→    freePlanDurationDays: row.freePlanDurationDays,
   595→    stripePriceIdMonthly: row.stripePriceIdMonthly,
   596→    stripePriceIdYearly: row.stripePriceIdYearly,
   597→    active: row.active,
   598→    features,
   599→    // Normalize: legacy 'ai_content' → 'ai_platform'; the two AI modes
   600→    // are mutually exclusive (invalid both-present data → platform wins).
   601→    entitlements: normalizeEntitlementKeys(entitlements),
   602→    limits,
   603→    badgeVariant: row.badgeVariant,
   604→    sortOrder: row.sortOrder,
   605→  };
   606→}

   634→/**
   635→ * Migrate the legacy catalog to the new Free/Plus/Pro/Max shape. Idempotent.
   636→ *
   637→ * Legacy catalog (created by older bootstrap):
   638→ *   beta (free, active=false), pro, max, enterprise (stub)
   639→ *
   640→ * Migration steps (each guarded, can be re-run safely):
   641→ *  1. If `enterprise` exists AND no active subscription references it → delete it.
   642→ *     (Active subs that reference enterprise are preserved — the migration will
   643→ *      not break them; the plan row stays if needed. In practice, the demo
   644→ *      dataset has no DB subscriptions yet, so this is a clean delete.)
   645→ *  2. If `beta` exists → rename to `plus` (planId, name, badgeVariant) and
   646→ *     re-enable it (legacy `beta` was active=false in the demo DB). Also
   647→ *     shift sortOrder to 1 (Free takes 0).
   648→ *  3. If `free` does NOT exist → insert the Free plan at sortOrder 0.
   649→ *  4. Renumber sortOrder to match the canonical order: free=0, plus=1, pro=2, max=3.
   650→ *
   651→ * Returns a short report string for the bootstrap log.
   652→ */
   653→export async function migrateLegacyPlans(): Promise<string> {
   654→  const notes: string[] = [];
   655→
   656→  try {
   657→    // Step 1: delete Enterprise if safe
   658→    const enterprise = await db.planConfig.findUnique({ where: { planId: 'enterprise' } });
   659→    if (enterprise) {
   660→      const subCount = await db.subscription.count({ where: { planId: 'enterprise' } });
   661→      if (subCount === 0) {
   662→        await db.planConfig.delete({ where: { planId: 'enterprise' } });
   663→        notes.push('deleted enterprise plan (0 active subscriptions)');
   664→      } else {
   665→        notes.push(`kept enterprise plan (${subCount} active subscriptions reference it)`);
   666→      }
   667→    }
   668→
   669→    // Step 2: rename beta → plus (and refresh to a real paid tier)
   670→    const beta = await db.planConfig.findUnique({ where: { planId: 'beta' } });
   671→    if (beta) {
   672→      // Don't overwrite an existing plus row — if both exist, prefer the
   673→      // plus row and just delete the beta stub.
   674→      const plus = await db.planConfig.findUnique({ where: { planId: 'plus' } });
   675→      if (!plus) {
   676→        // Rename beta → plus AND refresh the data to the real DEFAULT
   677→        // Plus tier (paid plan with real entitlements + limits, not the
   678→        // legacy free-tier Beta stub). The old Beta was active=false and
   679→        // had no real customers, so this is a safe refresh.
   680→        const plusDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'plus')!;
   681→        await db.planConfig.update({
   682→          where: { planId: 'beta' },
   683→          data: {
   684→            ...dataToRow(plusDefault),
   685→            // Keep the original createdAt for continuity.
   686→            // (updatedAt is auto.)
   687→          },
   688→        });
   689→        notes.push('renamed + refreshed beta → plus (real paid tier)');
   690→      } else {
   691→        // Both beta and plus exist — delete the beta stub (migrate any
   692→        // legacy subscriptions on beta to plus first).
   693→        const betaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   694→        if (betaSubs > 0) {
   695→          await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   696→          notes.push(`migrated ${betaSubs} beta subscription(s) → plus`);
   697→        }
   698→        await db.planConfig.delete({ where: { planId: 'beta' } });
   699→        notes.push('deleted legacy beta stub (plus already exists)');
   700→      }
   701→    }
   702→
   703→    // Step 3: insert Free if missing
   704→    const free = await db.planConfig.findUnique({ where: { planId: 'free' } });
   705→    if (!free) {
   706→      const freeDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'free')!;
   707→      await db.planConfig.create({ data: dataToRow(freeDefault) });
   708→      notes.push('inserted free plan');
   709→    } else {
   710→      // Ensure free plan is properly flagged
   711→      if (!free.isFree || free.badgeVariant !== 'free' || free.sortOrder !== 0) {
   712→        await db.planConfig.update({
   713→          where: { planId: 'free' },
   714→          data: { isFree: true, badgeVariant: 'free', sortOrder: 0 },
   715→        });
   716→        notes.push('normalized free plan flags');
   717→      }
   718→    }
   719→
   720→    // Step 4: normalize sortOrder for all canonical plans
   721→    const canonicalOrder: Record<string, number> = { free: 0, plus: 1, pro: 2, max: 3 };
   722→    for (const [pid, order] of Object.entries(canonicalOrder)) {
   723→      const row = await db.planConfig.findUnique({ where: { planId: pid } });
   724→      if (row && row.sortOrder !== order) {
   725→        await db.planConfig.update({ where: { planId: pid }, data: { sortOrder: order } });
   726→      }
   727→    }
   728→
   729→    // Also migrate legacy subscriptions on 'beta' or 'enterprise' planId
   730→    // to the new ids (beta → plus, enterprise → max) so historical subs
   731→    // don't reference a missing plan.
   732→    const legacyBetaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   733→    if (legacyBetaSubs > 0) {
   734→      await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   735→      notes.push(`migrated ${legacyBetaSubs} legacy beta subscription(s) → plus`);
   736→    }
   737→    const legacyEnterpriseSubs = await db.subscription.count({ where: { planId: 'enterprise' } });
   738→    if (legacyEnterpriseSubs > 0) {
   739→      await db.subscription.updateMany({ where: { planId: 'enterprise' }, data: { planId: 'max' } });
   740→      notes.push(`migrated ${legacyEnterpriseSubs} legacy enterprise subscription(s) → max`);
   741→    }
   742→  } catch (err) {
   743→    notes.push(`migration error (non-fatal): ${(err as Error).message}`);
   744→  }
   745→
   746→  return notes.length > 0 ? notes.join('; ') : 'no migration needed';
   747→}

   749→/** Load all plan configs from the DB into the cache. Self-seeds the
   750→ *  DEFAULT_PLAN_CONFIGS when the table is empty. Runs the legacy
   751→ *  migration when the catalog still contains 'beta' or 'enterprise'.
   752→ *  Safe to call repeatedly. */
   753→export async function hydrate(): Promise<void> {
   754→  if (_hydrating) return _hydrating;
   755→  _hydrating = (async () => {
   756→    try {
   757→      const rows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   758→      if (rows.length === 0) {
   759→        // Self-seed so the platform is never without plans.
   760→        await db.planConfig.createMany({ data: DEFAULT_PLAN_CONFIGS.map(dataToRow) });
   761→        _cache = DEFAULT_PLAN_CONFIGS.map((p) => ({
   762→          ...p,
   763→          features: [...p.features],
   764→          entitlements: [...p.entitlements],
   765→          limits: { ...p.limits },
   766→        }));
   767→      } else {
   768→        // Check for legacy catalog (beta / enterprise) and migrate.
   769→        const hasLegacy =
   770→          rows.some((r) => r.planId === 'beta' || r.planId === 'enterprise') ||
   771→          !rows.some((r) => r.planId === 'free');
   772→        if (hasLegacy) {
   773→          await migrateLegacyPlans();
   774→          const migratedRows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   775→          _cache = migratedRows.map(rowToData);
   776→        } else {
   777→          _cache = rows.map(rowToData);
   778→        }
   779→      }
   780→      _hydrated = true;
   781→      notify();
   782→    } catch (err) {
   783→      // DB unavailable — keep defaults (still functional, just not persisted).
   784→      _hydrated = true;
   785→      notify();
   786→    } finally {
   787→      _hydrating = null;
   788→    }
   789→  })();
   790→  return _hydrating;
   791→}

   832→/** Merge a patch (Partial<PlanConfigData>) into the current data.
   833→ *  The Edit Plan modal sends the ONE base price configuration
   834→ *  (priceMonthly + priceYearly + currency + autoCurrency) — NOT a
   835→ *  per-currency matrix. Derivation rules:
   836→ *  - pricesByCurrency: starts from the current map (or the explicit
   837→ *    patch map when provided); whenever the base price / currency is
   838→ *    in the patch, the DEFAULT currency's entry is re-derived as
   839→ *    { monthly: priceMonthly, yearly: priceYearly }. Other currency
   840→ *    entries (platform regional / legacy prices) are PRESERVED.
   841→ *  - stripePriceIdsByCurrency: kept as-is when omitted (the Stripe
   842→ *    sync maintains it); the default-currency snapshots mirror the
   843→ *    default entry.
   844→ */
   845→function mergePlanPatch(current: PlanConfigData, patch: PlanConfigInput, defaultCurrency: string): PlanConfigData {
   846→  const curUpper = defaultCurrency.toUpperCase();
   847→
   848→  // pricesByCurrency — clone (never mutate the cache's object in place).
   849→  const sourcePrices = patch.pricesByCurrency ?? current.pricesByCurrency ?? {};
   850→  const pricesByCurrency: PricesByCurrency = { ...sourcePrices };
   851→
   852→  // Base price precedence:
   853→  //   1. Explicit patch values (the modal always sends these).
   854→  //   2. Legacy mirror: an explicit pricesByCurrency map with an entry
   855→  //      for the final default currency (backward-compat callers).
   856→  //   3. Current snapshot.
   857→  const mirrored =
   858→    patch.priceMonthly === undefined && patch.pricesByCurrency !== undefined
   859→      ? pricesByCurrency[curUpper]
   860→      : undefined;
   861→  const priceMonthly =
   862→    patch.priceMonthly !== undefined ? patch.priceMonthly : mirrored?.monthly ?? current.priceMonthly;
   863→  const priceYearly =
   864→    patch.priceYearly !== undefined ? patch.priceYearly : mirrored?.yearly ?? current.priceYearly;
   865→
   866→  // The base price configuration was touched (or a legacy map was sent)
   867→  // → re-derive the DEFAULT currency's entry from the base price. All
   868→  // OTHER currency entries (platform regional / legacy prices) are
   869→  // preserved untouched.
   870→  const baseTouched =
   871→    patch.priceMonthly !== undefined ||
   872→    patch.priceYearly !== undefined ||
   873→    patch.currency !== undefined ||
   874→    patch.pricesByCurrency !== undefined;
   875→  if (baseTouched) {
   876→    pricesByCurrency[curUpper] = { monthly: priceMonthly, yearly: priceYearly };
   877→  }
   878→
   879→  const stripePriceIdsByCurrency: StripePriceIdsByCurrency = {
   880→    ...(patch.stripePriceIdsByCurrency ?? current.stripePriceIdsByCurrency ?? {}),
   881→  };
   882→  // Default-currency snapshot — keep the legacy fields in sync so legacy
   883→  // callers that read stripePriceIdMonthly/Yearly see the right values.
   884→  const defIds = stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
   885→  const stripePriceIdMonthly =
   886→    patch.stripePriceIdMonthly !== undefined
   887→      ? patch.stripePriceIdMonthly
   888→      : patch.stripePriceIdsByCurrency !== undefined
   889→        ? defIds.monthly
   890→        : current.stripePriceIdMonthly;
   891→  const stripePriceIdYearly =
   892→    patch.stripePriceIdYearly !== undefined
   893→      ? patch.stripePriceIdYearly
   894→      : patch.stripePriceIdsByCurrency !== undefined
   895→        ? defIds.yearly
   896→        : current.stripePriceIdYearly;
   897→  const isFree = patch.isFree ?? current.isFree;
   898→  // ---- Billing periods ----
   899→  const billingMonthly = patch.billingMonthly ?? current.billingMonthly;
   900→  const billingYearly = patch.billingYearly ?? current.billingYearly;
   901→  // The default cadence must be an ENABLED period (single-period plans
   902→  // are pinned to their only period; both-period plans keep the
   903→  // admin's requested / current cadence).
   904→  const interval = normalizeInterval(
   905→    patch.interval ?? current.interval,
   906→    billingMonthly,
   907→    billingYearly,
   908→  );
   909→  return {
   910→    planId: current.planId,
   911→    name: patch.name ?? current.name,
   912→    priceMonthly,
   913→    priceYearly,
   914→    currency: curUpper,
   915→    autoCurrency: patch.autoCurrency ?? current.autoCurrency,
   916→    pricesByCurrency,
   917→    stripePriceIdsByCurrency,
   918→    billingMonthly,
   919→    billingYearly,
   920→    interval,
   921→    isFree,
   922→    freePlanDurationDays: patch.freePlanDurationDays ?? current.freePlanDurationDays,
   923→    stripePriceIdMonthly,
   924→    stripePriceIdYearly,
   925→    active: patch.active ?? current.active,
   926→    features: patch.features ?? current.features,
   927→    // Entitlements: the patch's editor-managed keys replace the current
   928→    // ones, while NON-editor keys (e.g. 'audit_log') are PRESERVED from
   929→    // the current plan — the Create/Edit Plan modal only manages the 10
   930→    // Feature Access keys. Then normalize (legacy 'ai_content' →
   931→    // 'ai_platform'; the two AI modes are mutually exclusive).
   932→    entitlements: normalizeEntitlementKeys(
   933→      patch.entitlements !== undefined
   934→        ? [
   935→            ...patch.entitlements,
   936→            ...current.entitlements.filter(
   937→              (k) =>
   938→                !patch.entitlements!.includes(k) &&
   939→                k !== 'ai_content' &&
   940→                // keys below are editor-managed (replaced by the patch):
   941→                !PLAN_EDITOR_KEY_SET.has(k),
   942→            ),
   943→          ]
   944→        : current.entitlements,
   945→    ),
   946→    // Limits: merge; the AI usage limits are only part of the saved
   947→    // configuration while the plan uses Platform AI (zeroed otherwise).
   948→    limits: zeroAiLimitsIfNotPlatform(
   949→      pickLimits({ ...current.limits, ...(patch.limits ?? {}) }),
   950→      patch.entitlements !== undefined
   951→        ? normalizeEntitlementKeys(patch.entitlements)
   952→        : current.entitlements,
   953→    ),
   954→    badgeVariant: patch.badgeVariant ?? current.badgeVariant,
   955→    sortOrder: patch.sortOrder ?? current.sortOrder,
   956→  };
   957→}

   959→/** Update an existing plan config. Writes DB, refreshes cache.
   960→ *
   961→ *  AUTO-SYNC TO STRIPE: when Stripe is configured AND the plan is
   962→ *  PAID (any currency has a positive price) AND the admin did NOT
   963→ *  explicitly set the Stripe Price IDs in this patch, the backend calls
   964→ *  syncPlanToStripe (multi-currency) to ensure a Stripe Product +
   965→ *  per-currency monthly+yearly Prices exist and writes the resolved IDs
   966→ *  into stripePriceIdsByCurrency + the default-currency snapshot fields.
   967→ */
   968→export async function savePlanConfig(planId: string, patch: PlanConfigInput): Promise<PlanConfigData | null> {
   969→  const existing = await db.planConfig.findUnique({ where: { planId } });
   970→  if (!existing) return null;
   971→  const current = rowToData(existing);
   972→
   973→  // When the admin omitted currency, resolve the platform default so the
   974→  // legacy priceMonthly/priceYearly snapshot stays in the right currency.
   975→  if (!patch.currency && !current.currency) {
   976→    patch = { ...patch, currency: await resolvePlatformDefaultCurrency() };
   977→  }
   978→  const finalCurrency = (patch.currency ?? current.currency).toUpperCase();
   979→
   980→  // Distinguish "field omitted" from "field set". When the admin omits
   981→  // both stripePriceId* AND stripePriceIdsByCurrency, they want the
   982→  // backend to auto-sync. When either is explicitly provided, manual.
   983→  const adminTouchedStripePriceIds =
   984→    'stripePriceIdMonthly' in patch ||
   985→    'stripePriceIdYearly' in patch ||
   986→    'stripePriceIdsByCurrency' in patch;
   987→
   988→  const next = mergePlanPatch(current, patch, finalCurrency);
   989→  await db.planConfig.update({ where: { planId }, data: dataToRow(next) });
   990→  await hydrate();
   991→
   992→  // ---- AUTO-SYNC TO STRIPE (multi-currency) ----
   993→  // Best-effort — never throws. Only ENABLED billing periods get
   994→  // Stripe Prices (a disabled period never creates one).
   995→  const hasAnyPaidCurrency = Object.values(next.pricesByCurrency).some(
   996→    (p) =>
   997→      (next.billingMonthly && p.monthly > 0) ||
   998→      (next.billingYearly && p.yearly > 0),
   999→  );
  1000→  if (!adminTouchedStripePriceIds && !next.isFree && hasAnyPaidCurrency) {
  1001→    try {
  1002→      const { isStripeConfiguredAsync, getStripeClient, syncPlanToStripeMulti } = await import('@/lib/stripe');
  1003→      if (await isStripeConfiguredAsync()) {
  1004→        const stripe = await getStripeClient();
  1005→        const syncResult = await syncPlanToStripeMulti(stripe, {
  1006→          planId: next.planId,
  1007→          name: next.name,
  1008→          defaultCurrency: finalCurrency,
  1009→          pricesByCurrency: next.pricesByCurrency,
  1010→          stripePriceIdsByCurrency: next.stripePriceIdsByCurrency,
  1011→          // Only the ENABLED billing periods get Stripe Prices.
  1012→          enabledIntervals: enabledIntervalsOf(next),
  1013→        });
  1014→        // Persist the resolved Stripe Price IDs back onto the row when they changed.
  1015→        const snapshot = syncResult.stripePriceIdsByCurrency[finalCurrency] ?? { monthly: null, yearly: null };
  1016→        const changed =
  1017→          JSON.stringify(syncResult.stripePriceIdsByCurrency) !==
  1018→            JSON.stringify(next.stripePriceIdsByCurrency) ||
  1019→          snapshot.monthly !== next.stripePriceIdMonthly ||
  1020→          snapshot.yearly !== next.stripePriceIdYearly;
  1021→        if (changed) {
  1022→          await db.planConfig.update({
  1023→            where: { planId },
  1024→            data: {
  1025→              stripePriceIdsByCurrency: JSON.stringify(syncResult.stripePriceIdsByCurrency),
  1026→              stripePriceIdMonthly: snapshot.monthly,
  1027→              stripePriceIdYearly: snapshot.yearly,
  1028→            },
  1029→          });
  1030→          await hydrate();
  1031→        }
  1032→      }
  1033→    } catch {
  1034→      // best-effort — never block the save on a Stripe error
  1035→    }
  1036→  }
  1037→
  1038→  return getPlanConfigSync(planId);
  1039→}

  1041→/** Create a new plan config. AUTO-SYNC TO STRIPE: when Stripe is
  1042→ *  configured AND the new plan is PAID AND the admin did NOT
  1043→ *  explicitly provide Stripe Price IDs in the input, the backend
  1044→ *  calls syncPlanToStripeMulti (multi-currency) to create a Stripe
  1045→ *  Product + per-currency monthly+yearly Prices and writes the
  1046→ *  resolved IDs back onto the new plan row.
  1047→ *
  1048→ *  The Create Plan modal sends the ONE base price configuration
  1049→ *  (priceMonthly + priceYearly + currency + autoCurrency); the default
  1050→ *  currency's pricesByCurrency entry is derived from the base price. */
  1051→export async function createPlanConfig(
  1052→  input: PlanConfigInput & { planId: string; name: string },
  1053→): Promise<PlanConfigData> {
  1054→  const maxOrder = await db.planConfig.count();
  1055→  const currency =
  1056→    input.currency && input.currency.trim().length > 0
  1057→      ? input.currency
  1058→      : await resolvePlatformDefaultCurrency();
  1059→  const curUpper = currency.toUpperCase();
  1060→  const pricesByCurrency: PricesByCurrency = { ...(input.pricesByCurrency ?? {}) };
  1061→  // Base price precedence: explicit input → legacy mirror from an
  1062→  // explicit pricesByCurrency map → 0.
  1063→  const mirrored = input.pricesByCurrency ? pricesByCurrency[curUpper] : undefined;
  1064→  const priceMonthly = input.priceMonthly ?? mirrored?.monthly ?? 0;
  1065→  const priceYearly = input.priceYearly ?? mirrored?.yearly ?? 0;
  1066→  // Derive the default currency entry from the base price.
  1067→  pricesByCurrency[curUpper] = { monthly: priceMonthly, yearly: priceYearly };
  1068→  const stripePriceIdsByCurrency: StripePriceIdsByCurrency = input.stripePriceIdsByCurrency ?? {};
  1069→  const defIds = stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
  1070→  const billingMonthly = input.billingMonthly ?? true;
  1071→  const billingYearly = input.billingYearly ?? true;
  1072→  const data: PlanConfigData = {
  1073→    planId: input.planId,
  1074→    name: input.name,
  1075→    priceMonthly,
  1076→    priceYearly,
  1077→    currency: curUpper,
  1078→    autoCurrency: input.autoCurrency ?? true,
  1079→    pricesByCurrency,
  1080→    stripePriceIdsByCurrency,
  1081→    billingMonthly,
  1082→    billingYearly,
  1083→    interval: normalizeInterval(input.interval ?? 'monthly', billingMonthly, billingYearly),
  1084→    isFree: input.isFree ?? false,
  1085→    freePlanDurationDays: input.freePlanDurationDays ?? null,
  1086→    stripePriceIdMonthly: input.stripePriceIdMonthly !== undefined ? input.stripePriceIdMonthly : defIds.monthly,
  1087→    stripePriceIdYearly: input.stripePriceIdYearly !== undefined ? input.stripePriceIdYearly : defIds.yearly,
  1088→    active: input.active ?? true,
  1089→    features: input.features ?? [],
  1090→    entitlements: normalizeEntitlementKeys(input.entitlements ?? []),
  1091→    limits: zeroAiLimitsIfNotPlatform(
  1092→      pickLimits({ ...DEFAULT_LIMITS, ...(input.limits ?? {}) }),
  1093→      normalizeEntitlementKeys(input.entitlements ?? []),
  1094→    ),
  1095→    badgeVariant: input.badgeVariant ?? input.planId,
  1096→    sortOrder: input.sortOrder ?? maxOrder,
  1097→  };
  1098→  await db.planConfig.create({ data: dataToRow(data) });
  1099→  await hydrate();
  1100→
  1101→  // ---- AUTO-SYNC TO STRIPE (multi-currency) ----
  1102→  // Best-effort — never throws. The plan row is already created; if
  1103→  // the Stripe side fails, the admin can use the explicit "Sync to
  1104→  // Stripe" route to surface errors.
  1105→  const adminTouchedStripePriceIds =
  1106→    'stripePriceIdMonthly' in input ||
  1107→    'stripePriceIdYearly' in input ||
  1108→    'stripePriceIdsByCurrency' in input;
  1109→  const hasAnyPaidCurrency = Object.values(data.pricesByCurrency).some(
  1110→    (p) =>
  1111→      (data.billingMonthly && p.monthly > 0) ||
  1112→      (data.billingYearly && p.yearly > 0),
  1113→  );
  1114→  if (!adminTouchedStripePriceIds && !data.isFree && hasAnyPaidCurrency) {
  1115→    try {
  1116→      const { isStripeConfiguredAsync, getStripeClient, syncPlanToStripeMulti } = await import('@/lib/stripe');
  1117→      if (await isStripeConfiguredAsync()) {
  1118→        const stripe = await getStripeClient();
  1119→        const syncResult = await syncPlanToStripeMulti(stripe, {
  1120→          planId: data.planId,
  1121→          name: data.name,
  1122→          defaultCurrency: curUpper,
  1123→          pricesByCurrency: data.pricesByCurrency,
  1124→          stripePriceIdsByCurrency: data.stripePriceIdsByCurrency,
  1125→          // Only the ENABLED billing periods get Stripe Prices.
  1126→          enabledIntervals: enabledIntervalsOf(data),
  1127→        });
  1128→        const snapshot = syncResult.stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
  1129→        const changed =
  1130→          JSON.stringify(syncResult.stripePriceIdsByCurrency) !==
  1131→            JSON.stringify(data.stripePriceIdsByCurrency) ||
  1132→          snapshot.monthly !== data.stripePriceIdMonthly ||
  1133→          snapshot.yearly !== data.stripePriceIdYearly;
  1134→        if (changed) {
  1135→          await db.planConfig.update({
  1136→            where: { planId: data.planId },
  1137→            data: {
  1138→              stripePriceIdsByCurrency: JSON.stringify(syncResult.stripePriceIdsByCurrency),
  1139→              stripePriceIdMonthly: snapshot.monthly,
  1140→              stripePriceIdYearly: snapshot.yearly,
  1141→            },
  1142→          });
  1143→          await hydrate();
  1144→          data.stripePriceIdsByCurrency = syncResult.stripePriceIdsByCurrency;
  1145→          data.stripePriceIdMonthly = snapshot.monthly;
  1146→          data.stripePriceIdYearly = snapshot.yearly;
  1147→        }
  1148→      }
  1149→    } catch {
  1150→      // best-effort — never block the create on a Stripe error
  1151→    }
  1152→  }
  1153→
  1154→  return data;
  1155→}

  1157→/**
  1158→ * Delete a plan config. Refuses if:
  1159→ *  - it's the last remaining plan, OR
  1160→ *  - any Subscription still references this planId (would orphan the sub).
  1161→ *
  1162→ * Returns `{ ok: boolean, reason?: string }` describing the outcome.
  1163→ */
  1164→export async function deletePlanConfig(
  1165→  planId: string,
  1166→): Promise<{ ok: boolean; reason?: string }> {
  1167→  const count = await db.planConfig.count();
  1168→  if (count <= 1) return { ok: false, reason: 'Cannot delete the last plan.' };
  1169→  const subCount = await db.subscription.count({ where: { planId } });
  1170→  if (subCount > 0) {
  1171→    return {
  1172→      ok: false,
  1173→      reason: `Cannot delete plan "${planId}" — ${subCount} active subscription(s) reference it. Migrate them first.`,
  1174→    };
  1175→  }
  1176→  try {
  1177→    await db.planConfig.delete({ where: { planId } });
  1178→    await hydrate();
  1179→    return { ok: true };
  1180→  } catch (err) {
  1181→    return { ok: false, reason: (err as Error).message };
  1182→  }
  1183→}
