   213→function CreateTemplateDialog({
   214→  open,
   215→  onOpenChange,
   216→}: {
   217→  open: boolean;
   218→  onOpenChange: (v: boolean) => void;
   219→}) {
   220→  const queryClient = useQueryClient();
   221→  const [form, setForm] = useState<CreateForm>({
   222→    name: '',
   223→    slug: '',
   224→    subject: '',
   225→    category: 'SYSTEM',
   226→    status: 'DRAFT',
   227→    language: 'en',
   228→  });
   229→
   230→  const reset = useCallback(() => {
   231→    setForm({
   232→      name: '',
   233→      slug: '',
   234→      subject: '',
   235→      category: 'SYSTEM',
   236→      status: 'DRAFT',
   237→      language: 'en',
   238→    });
   239→  }, []);
   240→
   241→  const createMutation = useMutation({
   242→    mutationFn: () =>
   243→      postApi('/api/email-templates', {
   244→        scope: 'platform',
   245→        name: form.name.trim(),
   246→        slug: form.slug.trim() || undefined,
   247→        subject: form.subject.trim(),
   248→        htmlBody: '',
   249→        language: form.language.trim() || 'en',
   250→        category: form.category,
   251→        status: form.status,
   252→      } as Record<string, unknown>),
   253→    onSuccess: () => {
   254→      toast.success('Platform template created');
   255→      queryClient.invalidateQueries({ queryKey: ['platform-email-templates'] });
   256→      reset();
   257→      onOpenChange(false);
   258→    },
   259→    onError: (err: Error) => {
   260→      toast.error(err.message || 'Failed to create template');
   261→    },
   262→  });
   263→
   264→  const handleClose = useCallback(
   265→    (v: boolean) => {
   266→      if (!v) reset();
   267→      onOpenChange(v);
   268→    },
   269→    [onOpenChange, reset],
   270→  );
   271→
   272→  const canSubmit = form.name.trim().length > 0;
   273→
   274→  return (
   275→    <Dialog open={open} onOpenChange={handleClose}>
   276→      <DialogContent className="sm:max-w-lg">
   277→        <DialogHeader>
   278→          <DialogTitle>Create Platform Template</DialogTitle>
   279→          <DialogDescription>
   280→            Create a new system-level email template. It will be visible to all sites.
   281→          </DialogDescription>
   282→        </DialogHeader>
   283→        <div className="grid gap-4 py-2">
   284→          <div className="grid gap-2">
   285→            <Label htmlFor="tpl-name">Name</Label>
   286→            <Input
   287→              id="tpl-name"
   288→              placeholder="e.g. Welcome Email"
   289→              value={form.name}
   290→              onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
   291→            />
   292→          </div>
   293→          <div className="grid gap-2">
   294→            <Label htmlFor="tpl-slug">Slug (optional)</Label>
   295→            <Input
   296→              id="tpl-slug"
   297→              placeholder="auto-generated from name"
   298→              value={form.slug}
   299→              onChange={(e) => setForm((f) => ({ ...f, slug: e.target.value }))}
   300→            />
   301→          </div>
   302→          <div className="grid gap-2">
   303→            <Label htmlFor="tpl-subject">Subject</Label>
   304→            <Input
   305→              id="tpl-subject"
   306→              placeholder="Welcome to {{site.name}}"
   307→              value={form.subject}
   308→              onChange={(e) => setForm((f) => ({ ...f, subject: e.target.value }))}
   309→            />
   310→          </div>
   311→          <div className="grid grid-cols-2 gap-3">
   312→            <div className="grid gap-2">
   313→              <Label>Category</Label>
   314→              <Select
   315→                value={form.category}
   316→                onValueChange={(v) =>
   317→                  setForm((f) => ({ ...f, category: v as EmailTemplateCategory }))
   318→                }
   319→              >
   320→                <SelectTrigger className="w-full">
   321→                  <SelectValue />
   322→                </SelectTrigger>
   323→                <SelectContent>
   324→                  {CATEGORY_CREATE_OPTIONS.map((c) => (
   325→                    <SelectItem key={c.value} value={c.value}>
   326→                      {c.label}
   327→                    </SelectItem>
   328→                  ))}
   329→                </SelectContent>
   330→              </Select>
   331→            </div>

   333→              <Label>Status</Label>

   336→                onValueChange={(v) =>

   388→interface EditForm {

   390→  subject: string;
   391→  htmlBody: string;

   395→
   396→function EditTemplateDialog({
   397→  open,
   398→  onOpenChange,
   399→  templateId,
   400→}: {
   401→  open: boolean;
   402→  onOpenChange: (v: boolean) => void;
   403→  templateId: string | null;
   404→}) {
   405→  const queryClient = useQueryClient();
   406→  const [form, setForm] = useState<EditForm>({
   407→    name: '',
   408→    subject: '',
   409→    htmlBody: '',
   410→    category: 'SYSTEM',
   411→    status: 'DRAFT',
   412→  });
   413→  const [loadedId, setLoadedId] = useState<string | null>(null);
   414→
   415→  // Fetch full template (incl. htmlBody) when an edit is opened.
   416→  const { data, isFetching } = useQuery({
   417→    queryKey: ['platform-email-templates', 'detail', templateId],
   418→    queryFn: () =>
   419→      getApi<{ data: TemplateDetail }>(
   420→        `/api/email-templates/${templateId}`,
   421→        undefined,
   422→        { raw: true },
   423→      ).then((r) => r?.data ?? null),
   424→    enabled: !!templateId && open,
   425→    staleTime: 0,
   426→  });
   427→
   428→  // Populate the form once the requested template arrives.
   429→  React.useEffect(() => {
   430→    if (!open || !templateId) return;
   431→    if (loadedId === templateId) return;
   432→    if (!data) return;
   433→    setForm({
   434→      name: data.name ?? '',
   435→      subject: data.subject ?? '',
   436→      htmlBody: data.htmlBody ?? '',
   437→      category: data.category ?? 'SYSTEM',
   438→      status: data.status ?? 'DRAFT',
   439→    });
   440→    setLoadedId(templateId);
   441→  }, [open, templateId, loadedId, data]);
   442→
   443→  const updateMutation = useMutation({
   444→    mutationFn: () =>
   445→      patchApi(`/api/email-templates/${templateId}`, {
   446→        name: form.name.trim(),
   447→        subject: form.subject.trim(),
   448→        htmlBody: form.htmlBody,
   449→        category: form.category,
   450→        status: form.status,
   451→      }),
   452→    onSuccess: () => {
   453→      toast.success('Template updated');
   454→      queryClient.invalidateQueries({ queryKey: ['platform-email-templates'] });
   455→      onOpenChange(false);
   456→    },
   457→    onError: (err: Error) => {
   458→      toast.error(err.message || 'Failed to update template');
   459→    },
   460→  });
   461→
   462→  const handleClose = useCallback(
   463→    (v: boolean) => {
   464→      if (!v) {
   465→        setLoadedId(null);
   466→      }
   467→      onOpenChange(v);
   468→    },
   469→    [onOpenChange],
   470→  );
   471→
   472→  const canSubmit = form.name.trim().length > 0;
   473→
   474→  return (
   475→    <Dialog open={open} onOpenChange={handleClose}>
   476→      <DialogContent className="sm:max-w-2xl">
   477→        <DialogHeader>
   478→          <DialogTitle>Edit Template</DialogTitle>
   479→          <DialogDescription>
   480→            Update the template&apos;s content and metadata. Changes are saved as a version snapshot.
   481→          </DialogDescription>
   482→        </DialogHeader>
   483→        {isFetching && !loadedId ? (
   484→          <div className="py-6 space-y-3">
   485→            <Skeleton className="h-9 w-full" />
   486→            <Skeleton className="h-9 w-full" />
   487→            <Skeleton className="h-40 w-full" />
   488→          </div>
   489→        ) : (
   490→          <>
   491→            <div className="grid gap-4 py-2 max-h-[60vh] overflow-y-auto pr-1">
   492→              <div className="grid gap-2">
   493→                <Label htmlFor="edit-name">Name</Label>
   494→                <Input
   495→                  id="edit-name"
   496→                  value={form.name}
   497→                  onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
   498→                />
   499→              </div>
   500→              <div className="grid gap-2">
   501→                <Label htmlFor="edit-subject">Subject</Label>
   502→                <Input
   503→                  id="edit-subject"
   504→                  value={form.subject}
   505→                  onChange={(e) => setForm((f) => ({ ...f, subject: e.target.value }))}
   506→                />
   507→              </div>
   508→              <div className="grid gap-2">
   509→                <Label htmlFor="edit-html">HTML Body</Label>
   510→                <Textarea
   511→                  id="edit-html"
   512→                  rows={10}
   513→                  className="font-mono text-xs"
   514→                  value={form.htmlBody}
   515→                  onChange={(e) => setForm((f) => ({ ...f, htmlBody: e.target.value }))}
   516→                />
   517→              </div>
   518→              <div className="grid grid-cols-2 gap-3">
   519→                <div className="grid gap-2">
   520→                  <Label>Category</Label>
   521→                  <Select
   522→                    value={form.category}
   523→                    onValueChange={(v) =>
   524→                      setForm((f) => ({ ...f, category: v as EmailTemplateCategory }))
   525→                    }
   526→                  >
   527→                    <SelectTrigger className="w-full">
   528→                      <SelectValue />
   529→                    </SelectTrigger>
   530→                    <SelectContent>
   531→                      {CATEGORY_CREATE_OPTIONS.map((c) => (
   532→                        <SelectItem key={c.value} value={c.value}>
   533→                          {c.label}
   534→                        </SelectItem>
   535→                      ))}
   536→                    </SelectContent>
   537→                  </Select>
   538→                </div>
   539→                <div className="grid gap-2">
   540→                  <Label>Status</Label>
   541→                  <Select
   542→                    value={form.status}
   543→                    onValueChange={(v) =>
   544→                      setForm((f) => ({ ...f, status: v as EmailTemplateStatus }))
   545→                    }
   546→                  >
   547→                    <SelectTrigger className="w-full">
   548→                      <SelectValue />
   549→                    </SelectTrigger>
   550→                    <SelectContent>
   551→                      {STATUS_CREATE_OPTIONS.map((s) => (
   552→                        <SelectItem key={s.value} value={s.value}>
   553→                          {s.label}
   554→                        </SelectItem>
   555→                      ))}
   556→                    </SelectContent>
   557→                  </Select>
   558→                </div>
   559→              </div>
   560→            </div>
   561→            <DialogFooter>
   562→              <Button
   563→                variant="outline"
   564→                onClick={() => handleClose(false)}
   565→                disabled={updateMutation.isPending}
   566→              >
   567→                Cancel
   568→              </Button>
   569→              <Button
   570→                onClick={() => updateMutation.mutate()}
   571→                disabled={!canSubmit || updateMutation.isPending}
   572→              >
   573→                {updateMutation.isPending && (
   574→                  <Loader2 className="h-4 w-4 mr-2 animate-spin" />

   576→                Save Changes
   577→              </Button>
   578→            </DialogFooter>
   579→          </>
   580→        )}
   581→      </DialogContent>
   582→    </Dialog>
   583→  );
   584→}
