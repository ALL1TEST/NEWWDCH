     1→// ============================================================
     2→// PLAN CONFIG — DB-backed, cached single source of truth for plans,
     3→// pricing, entitlements and usage limits.
     4→// ============================================================
     5→// The Client Billing API and the Platform Admin both read from (and
     6→// the admin mutates) this same service. `platform-data.ts` delegates
     7→// its `PLANS` / `getPlan` / `monthlyPrice` to the synchronous cache
     8→// exposed here, so an owner edit (savePlanConfig) propagates to the
     9→// client billing experience and to MRR on the next read.
    10→//
    11→// MULTI-CURRENCY: the plan carries ONE base price configuration —
    12→// priceMonthly / priceYearly denominated in the plan's default /
    13→// fallback `currency` (both edited directly in the Edit Plan modal).
    14→// The customer's currency is resolved SERVER-SIDE from their IP
    15→// geolocation (see country-pricing.ts:resolveCustomerPricing) when
    16→// the plan's `autoCurrency` flag is on; otherwise every customer is
    17→// billed in the plan default currency.
    18→//
    19→// pricesByCurrency is a DERIVED store (NOT edited per-plan in the
    20→// modal): the default currency's entry mirrors the base price on
    21→// every save, and the other entries hold the platform-level
    22→// regional prices (CountryPricing.regionalPrices + legacy entries)
    23→// — preserved across saves so customers in other currencies keep
    24→// seeing their configured price (e.g. "90 MAD / month").
    25→//
    26→// STRIPE: for each paid plan × supported currency × interval, a real
    27→// Stripe Price is created on sync and its ID persisted in
    28→// stripePriceIdsByCurrency. The default currency's IDs are mirrored
    29→// in stripePriceIdMonthly/Yearly for legacy callers. Checkout always
    30→// resolves the per-currency Stripe Price ID server-side — the
    31→// customer's currency determines which Price is charged.
    32→//
    33→// The cache is hydrated from the PlanConfig table on startup and
    34→// after every mutation. If the table is empty on first hydrate, the
    35→// DEFAULT configs are seeded.
    36→// ============================================================

    93→// -------------------- Entitlement normalization --------------------
    94→
    95→/** Normalize a raw entitlement key list (DB row or API input):
    96→ *  - legacy 'ai_content' → 'ai_platform' (it predates the two-mode
    97→ *    split and always meant the platform-provided AI).
    98→ *  - mutual exclusion: when BOTH AI modes are present (invalid data),
    99→ *    Platform AI wins and 'ai_client' is dropped.
   100→ *  - legacy 'custom_domains' / 'white_label' are STRIPPED: site
   101→ *    domains and site branding are client-owned in this architecture
   102→ *    (every site carries its own domain + identity), so they are not
   103→ *    plan entitlements. Old DB rows self-clean on load + next save.
   104→ *  - dedupes while preserving order.
   105→ *  Non-editor keys (e.g. 'audit_log') pass through untouched. */
   106→export function normalizeEntitlementKeys(keys: readonly string[]): string[] {
   107→  // Legacy keys that are no longer plan entitlements — stripped on
   108→  // load/save so existing rows + stale API input clean themselves.
   109→  const LEGACY_REMOVED_KEYS = new Set(['custom_domains', 'white_label']);
   110→  const out: string[] = [];
   111→  const seen = new Set<string>();
   112→  const hasPlatform = keys.includes('ai_platform') || keys.includes('ai_content');
   113→  for (const k of keys) {
   114→    if (typeof k !== 'string' || !k.trim()) continue;
   115→    if (k === 'ai_content') continue; // legacy → replaced by ai_platform below
   116→    if (LEGACY_REMOVED_KEYS.has(k)) continue; // not plan entitlements anymore
   117→    if (k === 'ai_client' && hasPlatform) continue; // mutually exclusive → platform wins
   118→    if (k === 'ai_platform' && hasPlatform) {
   119→      if (!seen.has('ai_platform')) out.push('ai_platform');
   120→      seen.add('ai_platform');
   121→      continue;
   122→    }
   123→    if (!seen.has(k)) {
   124→      out.push(k);
   125→      seen.add(k);
   126→    }
   127→  }
   128→  // Insert the platform AI key at the position where the legacy key
   129→  // (or an existing 'ai_platform') first appeared — append at the end
   130→  // when neither was present in the ordered loop above (defensive).
   131→  if (hasPlatform && !seen.has('ai_platform')) out.push('ai_platform');
   132→  return out;
   133→}

   169→export type BillingInterval = 'monthly' | 'yearly';

   171→/** The billing periods enabled on a plan (at least one, enforced).
   172→ *  A disabled period: not shown on the Client Billing page, rejected
   173→ *  at checkout (400 BILLING_PERIOD_NOT_ENABLED), and never gets a
   174→ *  Stripe Price created or selected. */
   175→export function enabledIntervalsOf(plan: {
   176→  billingMonthly: boolean;
   177→  billingYearly: boolean;
   178→}): BillingInterval[] {
   179→  const out: BillingInterval[] = [];
   180→  if (plan.billingMonthly) out.push('monthly');
   181→  if (plan.billingYearly) out.push('yearly');
   182→  return out;
   183→}

   185→/** Clamp a requested default cadence to an ENABLED period:
   186→ *  only monthly on → 'monthly'; only yearly on → 'yearly'; both on
   187→ *  → the requested value. (Neither enabled is rejected upstream —
   188→ *  here it falls back to the requested value.) */
   189→function normalizeInterval(
   190→  requested: BillingInterval,
   191→  billingMonthly: boolean,
   192→  billingYearly: boolean,
   193→): BillingInterval {
   194→  if (billingMonthly && !billingYearly) return 'monthly';
   195→  if (!billingMonthly && billingYearly) return 'yearly';
   196→  return requested;
   197→}

   446→function notify() {
   447→  for (const cb of _subscribers) {
   448→    try {
   449→      cb();
   450→    } catch {
   451→      // ignore subscriber errors
   452→    }
   453→  }
   454→}

   456→/** Register a callback fired after every cache refresh (hydrate / mutation). */
   457→export function subscribe(cb: () => void): () => void {
   458→  _subscribers.add(cb);
   459→  return () => _subscribers.delete(cb);
   460→}

   468→  }
   469→  return _cache;

   533→    features = JSON.parse(row.features || '[]');
   534→  } catch {
   535→    features = [];
   536→  }
   537→  try {
   538→    entitlements = JSON.parse(row.entitlements || '[]');
   539→  } catch {
   540→    entitlements = [];
   541→  }
   542→  // Explicitly pick only the known limit fields. Older DB rows may still
   543→  // carry stale `aiWords` / `aiArticles` / `automationRuns` keys in the
   544→  // limits JSON (those were removed because no real enforcement exists).
   545→  // Picking only the known fields prevents the stale keys from leaking
   546→  // back into the cache and being re-serialized on the next save.
   547→  let limits: PlanLimits = { ...DEFAULT_LIMITS };
   548→  try {
   549→    limits = pickLimits(JSON.parse(row.limits || '{}'));
   550→  } catch {
   551→    // keep defaults
   552→  }
   553→  // AI usage limits only exist for Platform AI plans — normalize the
   554→  // stored configuration to match the entitlement mode.
   555→  limits = zeroAiLimitsIfNotPlatform(limits, entitlements);
   556→  let pricesByCurrency: PricesByCurrency = {};
   557→  try {
   558→    const parsed = JSON.parse(row.pricesByCurrency || '{}') as Partial<PricesByCurrency>;
   559→    if (parsed && typeof parsed === 'object') {
   560→      for (const [cur, val] of Object.entries(parsed)) {
   561→        if (val && typeof val === 'object') {
   562→          const m = typeof (val as CurrencyPrice).monthly === 'number' ? (val as CurrencyPrice).monthly : 0;
   563→          const y = typeof (val as CurrencyPrice).yearly === 'number' ? (val as CurrencyPrice).yearly : 0;
   564→          pricesByCurrency[cur.toUpperCase()] = { monthly: m, yearly: y };
   565→        }
   566→      }
   567→    }
   568→  } catch {
   569→    pricesByCurrency = {};
   570→  }
   571→  let stripePriceIdsByCurrency: StripePriceIdsByCurrency = {};
   572→  try {
   573→    const parsed = JSON.parse(row.stripePriceIdsByCurrency || '{}') as Partial<StripePriceIdsByCurrency>;
   574→    if (parsed && typeof parsed === 'object') {
   575→      for (const [cur, val] of Object.entries(parsed)) {
   576→        if (val && typeof val === 'object') {
   577→          const m = typeof (val as CurrencyStripeIds).monthly === 'string' ? (val as CurrencyStripeIds).monthly : null;
   578→          const y = typeof (val as CurrencyStripeIds).yearly === 'string' ? (val as CurrencyStripeIds).yearly : null;
   579→          stripePriceIdsByCurrency[cur.toUpperCase()] = { monthly: m, yearly: y };
   580→        }
   581→      }
   582→    }
   583→  } catch {
   584→    stripePriceIdsByCurrency = {};
   585→  }
   586→  return {
   587→    planId: row.planId,
   588→    name: row.name,
   589→    priceMonthly: row.priceMonthly,
   590→    priceYearly: row.priceYearly,
   591→    currency: row.currency,
   592→    autoCurrency: row.autoCurrency ?? true,
   593→    pricesByCurrency,
   594→    stripePriceIdsByCurrency,
   595→    billingMonthly: row.billingMonthly ?? true,
   596→    billingYearly: row.billingYearly ?? true,
   597→    // Normalize the default cadence to an ENABLED period (defensive:
   598→    // legacy rows / stale values never advertise a disabled period).
   599→    interval: normalizeInterval(
   600→      (row.interval === 'yearly' ? 'yearly' : 'monthly') as BillingInterval,
   601→      row.billingMonthly ?? true,
   602→      row.billingYearly ?? true,
   603→    ),
   604→    isFree: row.isFree,
   605→    freePlanDurationDays: row.freePlanDurationDays,
   606→    stripePriceIdMonthly: row.stripePriceIdMonthly,
   607→    stripePriceIdYearly: row.stripePriceIdYearly,
   608→    active: row.active,
   609→    features,
   610→    // Normalize: legacy 'ai_content' → 'ai_platform'; the two AI modes
   611→    // are mutually exclusive (invalid both-present data → platform wins).
   612→    entitlements: normalizeEntitlementKeys(entitlements),
   613→    limits,
   614→    badgeVariant: row.badgeVariant,
   615→    sortOrder: row.sortOrder,
   616→  };
   617→}

   645→/**
   646→ * Migrate the legacy catalog to the new Free/Plus/Pro/Max shape. Idempotent.
   647→ *
   648→ * Legacy catalog (created by older bootstrap):
   649→ *   beta (free, active=false), pro, max, enterprise (stub)
   650→ *
   651→ * Migration steps (each guarded, can be re-run safely):
   652→ *  1. If `enterprise` exists AND no active subscription references it → delete it.
   653→ *     (Active subs that reference enterprise are preserved — the migration will
   654→ *      not break them; the plan row stays if needed. In practice, the demo
   655→ *      dataset has no DB subscriptions yet, so this is a clean delete.)
   656→ *  2. If `beta` exists → rename to `plus` (planId, name, badgeVariant) and
   657→ *     re-enable it (legacy `beta` was active=false in the demo DB). Also
   658→ *     shift sortOrder to 1 (Free takes 0).
   659→ *  3. If `free` does NOT exist → insert the Free plan at sortOrder 0.
   660→ *  4. Renumber sortOrder to match the canonical order: free=0, plus=1, pro=2, max=3.
   661→ *
   662→ * Returns a short report string for the bootstrap log.
   663→ */
   664→export async function migrateLegacyPlans(): Promise<string> {
   665→  const notes: string[] = [];
   666→
   667→  try {
   668→    // Step 1: delete Enterprise if safe
   669→    const enterprise = await db.planConfig.findUnique({ where: { planId: 'enterprise' } });
   670→    if (enterprise) {
   671→      const subCount = await db.subscription.count({ where: { planId: 'enterprise' } });
   672→      if (subCount === 0) {
   673→        await db.planConfig.delete({ where: { planId: 'enterprise' } });
   674→        notes.push('deleted enterprise plan (0 active subscriptions)');
   675→      } else {
   676→        notes.push(`kept enterprise plan (${subCount} active subscriptions reference it)`);
   677→      }
   678→    }
   679→
   680→    // Step 2: rename beta → plus (and refresh to a real paid tier)
   681→    const beta = await db.planConfig.findUnique({ where: { planId: 'beta' } });
   682→    if (beta) {
   683→      // Don't overwrite an existing plus row — if both exist, prefer the
   684→      // plus row and just delete the beta stub.
   685→      const plus = await db.planConfig.findUnique({ where: { planId: 'plus' } });
   686→      if (!plus) {
   687→        // Rename beta → plus AND refresh the data to the real DEFAULT
   688→        // Plus tier (paid plan with real entitlements + limits, not the
   689→        // legacy free-tier Beta stub). The old Beta was active=false and
   690→        // had no real customers, so this is a safe refresh.
   691→        const plusDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'plus')!;
   692→        await db.planConfig.update({
   693→          where: { planId: 'beta' },
   694→          data: {
   695→            ...dataToRow(plusDefault),
   696→            // Keep the original createdAt for continuity.
   697→            // (updatedAt is auto.)
   698→          },
   699→        });
   700→        notes.push('renamed + refreshed beta → plus (real paid tier)');
   701→      } else {
   702→        // Both beta and plus exist — delete the beta stub (migrate any
   703→        // legacy subscriptions on beta to plus first).
   704→        const betaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   705→        if (betaSubs > 0) {
   706→          await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   707→          notes.push(`migrated ${betaSubs} beta subscription(s) → plus`);
   708→        }
   709→        await db.planConfig.delete({ where: { planId: 'beta' } });
   710→        notes.push('deleted legacy beta stub (plus already exists)');
   711→      }
   712→    }
   713→
   714→    // Step 3: insert Free if missing
   715→    const free = await db.planConfig.findUnique({ where: { planId: 'free' } });
   716→    if (!free) {
   717→      const freeDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'free')!;
   718→      await db.planConfig.create({ data: dataToRow(freeDefault) });
   719→      notes.push('inserted free plan');
   720→    } else {
   721→      // Ensure free plan is properly flagged
   722→      if (!free.isFree || free.badgeVariant !== 'free' || free.sortOrder !== 0) {
   723→        await db.planConfig.update({
   724→          where: { planId: 'free' },
   725→          data: { isFree: true, badgeVariant: 'free', sortOrder: 0 },
   726→        });
   727→        notes.push('normalized free plan flags');
   728→      }
   729→    }
   730→
   731→    // Step 4: normalize sortOrder for all canonical plans
   732→    const canonicalOrder: Record<string, number> = { free: 0, plus: 1, pro: 2, max: 3 };
   733→    for (const [pid, order] of Object.entries(canonicalOrder)) {
   734→      const row = await db.planConfig.findUnique({ where: { planId: pid } });
   735→      if (row && row.sortOrder !== order) {
   736→        await db.planConfig.update({ where: { planId: pid }, data: { sortOrder: order } });
   737→      }
   738→    }
   739→
   740→    // Also migrate legacy subscriptions on 'beta' or 'enterprise' planId
   741→    // to the new ids (beta → plus, enterprise → max) so historical subs
   742→    // don't reference a missing plan.
   743→    const legacyBetaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   744→    if (legacyBetaSubs > 0) {
   745→      await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   746→      notes.push(`migrated ${legacyBetaSubs} legacy beta subscription(s) → plus`);
   747→    }
   748→    const legacyEnterpriseSubs = await db.subscription.count({ where: { planId: 'enterprise' } });
   749→    if (legacyEnterpriseSubs > 0) {
   750→      await db.subscription.updateMany({ where: { planId: 'enterprise' }, data: { planId: 'max' } });
   751→      notes.push(`migrated ${legacyEnterpriseSubs} legacy enterprise subscription(s) → max`);
   752→    }
   753→  } catch (err) {
   754→    notes.push(`migration error (non-fatal): ${(err as Error).message}`);
   755→  }
   756→
   757→  return notes.length > 0 ? notes.join('; ') : 'no migration needed';
   758→}

   760→/** Load all plan configs from the DB into the cache. Self-seeds the
   761→ *  DEFAULT_PLAN_CONFIGS when the table is empty. Runs the legacy
   762→ *  migration when the catalog still contains 'beta' or 'enterprise'.
   763→ *  Safe to call repeatedly. */
   764→export async function hydrate(): Promise<void> {
   765→  if (_hydrating) return _hydrating;
   766→  _hydrating = (async () => {
   767→    try {
   768→      const rows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   769→      if (rows.length === 0) {
   770→        // Self-seed so the platform is never without plans.
   771→        await db.planConfig.createMany({ data: DEFAULT_PLAN_CONFIGS.map(dataToRow) });
   772→        _cache = DEFAULT_PLAN_CONFIGS.map((p) => ({
   773→          ...p,
   774→          features: [...p.features],
   775→          entitlements: [...p.entitlements],
   776→          limits: { ...p.limits },
   777→        }));
   778→      } else {
   779→        // Check for legacy catalog (beta / enterprise) and migrate.
   780→        const hasLegacy =
   781→          rows.some((r) => r.planId === 'beta' || r.planId === 'enterprise') ||
   782→          !rows.some((r) => r.planId === 'free');
   783→        if (hasLegacy) {
   784→          await migrateLegacyPlans();
   785→          const migratedRows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   786→          _cache = migratedRows.map(rowToData);
   787→        } else {
   788→          _cache = rows.map(rowToData);
   789→        }
   790→      }
   791→      _hydrated = true;
   792→      notify();
   793→    } catch (err) {
   794→      // DB unavailable — keep defaults (still functional, just not persisted).
   795→      _hydrated = true;
   796→      notify();
   797→    } finally {
   798→      _hydrating = null;
   799→    }
   800→  })();
   801→  return _hydrating;
   802→}

   843→/** Merge a patch (Partial<PlanConfigData>) into the current data.
   844→ *  The Edit Plan modal sends the ONE base price configuration
   845→ *  (priceMonthly + priceYearly + currency + autoCurrency) — NOT a
   846→ *  per-currency matrix. Derivation rules:
   847→ *  - pricesByCurrency: starts from the current map (or the explicit
   848→ *    patch map when provided); whenever the base price / currency is
   849→ *    in the patch, the DEFAULT currency's entry is re-derived as
   850→ *    { monthly: priceMonthly, yearly: priceYearly }. Other currency
   851→ *    entries (platform regional / legacy prices) are PRESERVED.
   852→ *  - stripePriceIdsByCurrency: kept as-is when omitted (the Stripe
   853→ *    sync maintains it); the default-currency snapshots mirror the
   854→ *    default entry.
   855→ */
   856→function mergePlanPatch(current: PlanConfigData, patch: PlanConfigInput, defaultCurrency: string): PlanConfigData {
   857→  const curUpper = defaultCurrency.toUpperCase();
   858→
   859→  // pricesByCurrency — clone (never mutate the cache's object in place).
   860→  const sourcePrices = patch.pricesByCurrency ?? current.pricesByCurrency ?? {};
   861→  const pricesByCurrency: PricesByCurrency = { ...sourcePrices };
   862→
   863→  // Base price precedence:
   864→  //   1. Explicit patch values (the modal always sends these).
   865→  //   2. Legacy mirror: an explicit pricesByCurrency map with an entry
   866→  //      for the final default currency (backward-compat callers).
   867→  //   3. Current snapshot.
   868→  const mirrored =
   869→    patch.priceMonthly === undefined && patch.pricesByCurrency !== undefined
   870→      ? pricesByCurrency[curUpper]
   871→      : undefined;
   872→  const priceMonthly =
   873→    patch.priceMonthly !== undefined ? patch.priceMonthly : mirrored?.monthly ?? current.priceMonthly;
   874→  const priceYearly =
   875→    patch.priceYearly !== undefined ? patch.priceYearly : mirrored?.yearly ?? current.priceYearly;
   876→
   877→  // The base price configuration was touched (or a legacy map was sent)
   878→  // → re-derive the DEFAULT currency's entry from the base price. All
   879→  // OTHER currency entries (platform regional / legacy prices) are
   880→  // preserved untouched.
   881→  const baseTouched =
   882→    patch.priceMonthly !== undefined ||
   883→    patch.priceYearly !== undefined ||
   884→    patch.currency !== undefined ||
   885→    patch.pricesByCurrency !== undefined;
   886→  if (baseTouched) {
   887→    pricesByCurrency[curUpper] = { monthly: priceMonthly, yearly: priceYearly };
   888→  }
   889→
   890→  const stripePriceIdsByCurrency: StripePriceIdsByCurrency = {
   891→    ...(patch.stripePriceIdsByCurrency ?? current.stripePriceIdsByCurrency ?? {}),
   892→  };
   893→  // Default-currency snapshot — keep the legacy fields in sync so legacy
   894→  // callers that read stripePriceIdMonthly/Yearly see the right values.
   895→  const defIds = stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
   896→  const stripePriceIdMonthly =
   897→    patch.stripePriceIdMonthly !== undefined
   898→      ? patch.stripePriceIdMonthly
   899→      : patch.stripePriceIdsByCurrency !== undefined
   900→        ? defIds.monthly
   901→        : current.stripePriceIdMonthly;
   902→  const stripePriceIdYearly =
   903→    patch.stripePriceIdYearly !== undefined
   904→      ? patch.stripePriceIdYearly
   905→      : patch.stripePriceIdsByCurrency !== undefined
   906→        ? defIds.yearly
   907→        : current.stripePriceIdYearly;
   908→  const isFree = patch.isFree ?? current.isFree;
   909→  // ---- Billing periods ----
   910→  const billingMonthly = patch.billingMonthly ?? current.billingMonthly;
   911→  const billingYearly = patch.billingYearly ?? current.billingYearly;
   912→  // The default cadence must be an ENABLED period (single-period plans
   913→  // are pinned to their only period; both-period plans keep the
   914→  // admin's requested / current cadence).
   915→  const interval = normalizeInterval(
   916→    patch.interval ?? current.interval,
   917→    billingMonthly,
   918→    billingYearly,
   919→  );
   920→  return {
   921→    planId: current.planId,
   922→    name: patch.name ?? current.name,
   923→    priceMonthly,
   924→    priceYearly,
   925→    currency: curUpper,
   926→    autoCurrency: patch.autoCurrency ?? current.autoCurrency,
   927→    pricesByCurrency,
   928→    stripePriceIdsByCurrency,
   929→    billingMonthly,
   930→    billingYearly,
   931→    interval,
   932→    isFree,
   933→    freePlanDurationDays: patch.freePlanDurationDays ?? current.freePlanDurationDays,
   934→    stripePriceIdMonthly,
   935→    stripePriceIdYearly,
   936→    active: patch.active ?? current.active,
   937→    features: patch.features ?? current.features,
   938→    // Entitlements: the patch's editor-managed keys replace the current
   939→    // ones, while NON-editor keys (e.g. 'audit_log') are PRESERVED from
   940→    // the current plan — the Create/Edit Plan modal only manages the 10
   941→    // Feature Access keys. Then normalize (legacy 'ai_content' →
   942→    // 'ai_platform'; the two AI modes are mutually exclusive).
   943→    entitlements: normalizeEntitlementKeys(
   944→      patch.entitlements !== undefined
   945→        ? [
   946→            ...patch.entitlements,
   947→            ...current.entitlements.filter(
   948→              (k) =>
   949→                !patch.entitlements!.includes(k) &&
   950→                k !== 'ai_content' &&
   951→                // keys below are editor-managed (replaced by the patch):
   952→                !PLAN_EDITOR_KEY_SET.has(k),
   953→            ),
   954→          ]
   955→        : current.entitlements,
   956→    ),
   957→    // Limits: merge; the AI usage limits are only part of the saved
   958→    // configuration while the plan uses Platform AI (zeroed otherwise).
   959→    limits: zeroAiLimitsIfNotPlatform(
   960→      pickLimits({ ...current.limits, ...(patch.limits ?? {}) }),
   961→      patch.entitlements !== undefined
   962→        ? normalizeEntitlementKeys(patch.entitlements)
   963→        : current.entitlements,
   964→    ),
   965→    badgeVariant: patch.badgeVariant ?? current.badgeVariant,
   966→    sortOrder: patch.sortOrder ?? current.sortOrder,
   967→  };
   968→}

   970→/** Update an existing plan config. Writes DB, refreshes cache.
   971→ *
   972→ *  AUTO-SYNC TO STRIPE: when Stripe is configured AND the plan is
   973→ *  PAID (any currency has a positive price) AND the admin did NOT
   974→ *  explicitly set the Stripe Price IDs in this patch, the backend calls
   975→ *  syncPlanToStripe (multi-currency) to ensure a Stripe Product +
   976→ *  per-currency monthly+yearly Prices exist and writes the resolved IDs
   977→ *  into stripePriceIdsByCurrency + the default-currency snapshot fields.
   978→ */
   979→export async function savePlanConfig(planId: string, patch: PlanConfigInput): Promise<PlanConfigData | null> {
   980→  const existing = await db.planConfig.findUnique({ where: { planId } });
   981→  if (!existing) return null;
   982→  const current = rowToData(existing);
   983→
   984→  // When the admin omitted currency, resolve the platform default so the
   985→  // legacy priceMonthly/priceYearly snapshot stays in the right currency.
   986→  if (!patch.currency && !current.currency) {
   987→    patch = { ...patch, currency: await resolvePlatformDefaultCurrency() };
   988→  }
   989→  const finalCurrency = (patch.currency ?? current.currency).toUpperCase();
   990→
   991→  // Distinguish "field omitted" from "field set". When the admin omits
   992→  // both stripePriceId* AND stripePriceIdsByCurrency, they want the
   993→  // backend to auto-sync. When either is explicitly provided, manual.
   994→  const adminTouchedStripePriceIds =
   995→    'stripePriceIdMonthly' in patch ||
   996→    'stripePriceIdYearly' in patch ||
   997→    'stripePriceIdsByCurrency' in patch;
   998→
   999→  const next = mergePlanPatch(current, patch, finalCurrency);
  1000→  await db.planConfig.update({ where: { planId }, data: dataToRow(next) });
  1001→  await hydrate();
  1002→
  1003→  // ---- AUTO-SYNC TO STRIPE (multi-currency) ----
  1004→  // Best-effort — never throws. Only ENABLED billing periods get
  1005→  // Stripe Prices (a disabled period never creates one).
  1006→  const hasAnyPaidCurrency = Object.values(next.pricesByCurrency).some(
  1007→    (p) =>
  1008→      (next.billingMonthly && p.monthly > 0) ||
  1009→      (next.billingYearly && p.yearly > 0),
  1010→  );
  1011→  if (!adminTouchedStripePriceIds && !next.isFree && hasAnyPaidCurrency) {
  1012→    try {
  1013→      const { isStripeConfiguredAsync, getStripeClient, syncPlanToStripeMulti } = await import('@/lib/stripe');
  1014→      if (await isStripeConfiguredAsync()) {
  1015→        const stripe = await getStripeClient();
  1016→        const syncResult = await syncPlanToStripeMulti(stripe, {
  1017→          planId: next.planId,
  1018→          name: next.name,
  1019→          defaultCurrency: finalCurrency,
  1020→          pricesByCurrency: next.pricesByCurrency,
  1021→          stripePriceIdsByCurrency: next.stripePriceIdsByCurrency,
  1022→          // Only the ENABLED billing periods get Stripe Prices.
  1023→          enabledIntervals: enabledIntervalsOf(next),
  1024→        });
  1025→        // Persist the resolved Stripe Price IDs back onto the row when they changed.
  1026→        const snapshot = syncResult.stripePriceIdsByCurrency[finalCurrency] ?? { monthly: null, yearly: null };
  1027→        const changed =
  1028→          JSON.stringify(syncResult.stripePriceIdsByCurrency) !==
  1029→            JSON.stringify(next.stripePriceIdsByCurrency) ||
  1030→          snapshot.monthly !== next.stripePriceIdMonthly ||
  1031→          snapshot.yearly !== next.stripePriceIdYearly;

  1034→            where: { planId },
  1035→            data: {
  1036→              stripePriceIdsByCurrency: JSON.stringify(syncResult.stripePriceIdsByCurrency),
  1037→              stripePriceIdMonthly: snapshot.monthly,
  1038→              stripePriceIdYearly: snapshot.yearly,
  1039→            },
  1040→          });
  1041→          await hydrate();
  1042→        }
  1043→      }
  1044→    } catch {
  1045→      // best-effort — never block the save on a Stripe error
  1046→    }
  1047→  }
  1048→
  1049→  return getPlanConfigSync(planId);
  1050→}

  1052→/** Create a new plan config. AUTO-SYNC TO STRIPE: when Stripe is
  1053→ *  configured AND the new plan is PAID AND the admin did NOT
  1054→ *  explicitly provide Stripe Price IDs in the input, the backend
  1055→ *  calls syncPlanToStripeMulti (multi-currency) to create a Stripe
  1056→ *  Product + per-currency monthly+yearly Prices and writes the
  1057→ *  resolved IDs back onto the new plan row.
  1058→ *
  1059→ *  The Create Plan modal sends the ONE base price configuration
  1060→ *  (priceMonthly + priceYearly + currency + autoCurrency); the default
  1061→ *  currency's pricesByCurrency entry is derived from the base price. */
  1062→export async function createPlanConfig(
  1063→  input: PlanConfigInput & { planId: string; name: string },
  1064→): Promise<PlanConfigData> {
  1065→  const maxOrder = await db.planConfig.count();
  1066→  const currency =
  1067→    input.currency && input.currency.trim().length > 0
  1068→      ? input.currency
  1069→      : await resolvePlatformDefaultCurrency();
  1070→  const curUpper = currency.toUpperCase();
  1071→  const pricesByCurrency: PricesByCurrency = { ...(input.pricesByCurrency ?? {}) };
  1072→  // Base price precedence: explicit input → legacy mirror from an
  1073→  // explicit pricesByCurrency map → 0.
  1074→  const mirrored = input.pricesByCurrency ? pricesByCurrency[curUpper] : undefined;
  1075→  const priceMonthly = input.priceMonthly ?? mirrored?.monthly ?? 0;
  1076→  const priceYearly = input.priceYearly ?? mirrored?.yearly ?? 0;
  1077→  // Derive the default currency entry from the base price.
  1078→  pricesByCurrency[curUpper] = { monthly: priceMonthly, yearly: priceYearly };
  1079→  const stripePriceIdsByCurrency: StripePriceIdsByCurrency = input.stripePriceIdsByCurrency ?? {};
  1080→  const defIds = stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
  1081→  const billingMonthly = input.billingMonthly ?? true;
  1082→  const billingYearly = input.billingYearly ?? true;
  1083→  const data: PlanConfigData = {
  1084→    planId: input.planId,
  1085→    name: input.name,
  1086→    priceMonthly,
  1087→    priceYearly,
  1088→    currency: curUpper,
  1089→    autoCurrency: input.autoCurrency ?? true,
  1090→    pricesByCurrency,
  1091→    stripePriceIdsByCurrency,
  1092→    billingMonthly,
  1093→    billingYearly,
  1094→    interval: normalizeInterval(input.interval ?? 'monthly', billingMonthly, billingYearly),
  1095→    isFree: input.isFree ?? false,
  1096→    freePlanDurationDays: input.freePlanDurationDays ?? null,
  1097→    stripePriceIdMonthly: input.stripePriceIdMonthly !== undefined ? input.stripePriceIdMonthly : defIds.monthly,
  1098→    stripePriceIdYearly: input.stripePriceIdYearly !== undefined ? input.stripePriceIdYearly : defIds.yearly,
  1099→    active: input.active ?? true,
  1100→    features: input.features ?? [],
  1101→    entitlements: normalizeEntitlementKeys(input.entitlements ?? []),
  1102→    limits: zeroAiLimitsIfNotPlatform(
  1103→      pickLimits({ ...DEFAULT_LIMITS, ...(input.limits ?? {}) }),
  1104→      normalizeEntitlementKeys(input.entitlements ?? []),
  1105→    ),
  1106→    badgeVariant: input.badgeVariant ?? input.planId,
  1107→    sortOrder: input.sortOrder ?? maxOrder,
  1108→  };
  1109→  await db.planConfig.create({ data: dataToRow(data) });
  1110→  await hydrate();
  1111→
  1112→  // ---- AUTO-SYNC TO STRIPE (multi-currency) ----
  1113→  // Best-effort — never throws. The plan row is already created; if
  1114→  // the Stripe side fails, the admin can use the explicit "Sync to
  1115→  // Stripe" route to surface errors.
  1116→  const adminTouchedStripePriceIds =
  1117→    'stripePriceIdMonthly' in input ||
  1118→    'stripePriceIdYearly' in input ||
  1119→    'stripePriceIdsByCurrency' in input;
  1120→  const hasAnyPaidCurrency = Object.values(data.pricesByCurrency).some(
  1121→    (p) =>
  1122→      (data.billingMonthly && p.monthly > 0) ||
  1123→      (data.billingYearly && p.yearly > 0),
  1124→  );
  1125→  if (!adminTouchedStripePriceIds && !data.isFree && hasAnyPaidCurrency) {
  1126→    try {
  1127→      const { isStripeConfiguredAsync, getStripeClient, syncPlanToStripeMulti } = await import('@/lib/stripe');
  1128→      if (await isStripeConfiguredAsync()) {
  1129→        const stripe = await getStripeClient();
  1130→        const syncResult = await syncPlanToStripeMulti(stripe, {
  1131→          planId: data.planId,
  1132→          name: data.name,
  1133→          defaultCurrency: curUpper,
  1134→          pricesByCurrency: data.pricesByCurrency,
  1135→          stripePriceIdsByCurrency: data.stripePriceIdsByCurrency,
  1136→          // Only the ENABLED billing periods get Stripe Prices.
  1137→          enabledIntervals: enabledIntervalsOf(data),
  1138→        });
  1139→        const snapshot = syncResult.stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
  1140→        const changed =
  1141→          JSON.stringify(syncResult.stripePriceIdsByCurrency) !==
  1142→            JSON.stringify(data.stripePriceIdsByCurrency) ||
  1143→          snapshot.monthly !== data.stripePriceIdMonthly ||
  1144→          snapshot.yearly !== data.stripePriceIdYearly;
  1145→        if (changed) {
  1146→          await db.planConfig.update({
  1147→            where: { planId: data.planId },
  1148→            data: {
  1149→              stripePriceIdsByCurrency: JSON.stringify(syncResult.stripePriceIdsByCurrency),
  1150→              stripePriceIdMonthly: snapshot.monthly,
  1151→              stripePriceIdYearly: snapshot.yearly,
  1152→            },
  1153→          });
  1154→          await hydrate();
  1155→          data.stripePriceIdsByCurrency = syncResult.stripePriceIdsByCurrency;
  1156→          data.stripePriceIdMonthly = snapshot.monthly;
  1157→          data.stripePriceIdYearly = snapshot.yearly;
  1158→        }
  1159→      }
  1160→    } catch {
  1161→      // best-effort — never block the create on a Stripe error
  1162→    }
  1163→  }
  1164→
  1165→  return data;
  1166→}

  1168→/**
  1169→ * Delete a plan config. Refuses if:
  1170→ *  - it's the last remaining plan, OR
  1171→ *  - any Subscription still references this planId (would orphan the sub).
  1172→ *
  1173→ * Returns `{ ok: boolean, reason?: string }` describing the outcome.
  1174→ */
  1175→export async function deletePlanConfig(
  1176→  planId: string,
  1177→): Promise<{ ok: boolean; reason?: string }> {
  1178→  const count = await db.planConfig.count();
  1179→  if (count <= 1) return { ok: false, reason: 'Cannot delete the last plan.' };
  1180→  const subCount = await db.subscription.count({ where: { planId } });
  1181→  if (subCount > 0) {
  1182→    return {
  1183→      ok: false,
  1184→      reason: `Cannot delete plan "${planId}" — ${subCount} active subscription(s) reference it. Migrate them first.`,
  1185→    };
  1186→  }
  1187→  try {
  1188→    await db.planConfig.delete({ where: { planId } });
  1189→    await hydrate();
  1190→    return { ok: true };
  1191→  } catch (err) {
  1192→    return { ok: false, reason: (err as Error).message };
  1193→  }
  1194→}
