     1→     1→// ============================================================
     2→     2→// STRIPE CLIENT + CONFIGURATION GUARD.
     3→     3→// ============================================================
     4→     4→// This is the SINGLE place that resolves Stripe credentials and
     5→     5→// constructs the Stripe SDK client. The rest of the codebase calls
     6→     6→// `isStripeConfigured()` to gate real billing; when false, checkout
     7→     7→// returns 503 with a clear "not configured" message and NEVER fakes
     8→     8→// a successful payment.
     9→     9→//
    10→    10→// CREDENTIAL RESOLUTION ORDER (per credential type):
    11→    11→//   1. The StripeSettings singleton row in the DB (admin-configured
    12→    12→//      via Platform Admin → Stripe Settings). Secret keys are stored
    13→    13→//      AES-256-GCM encrypted; only the masked form is returned to the
    14→    14→//      frontend. Publishable keys are stored as plaintext.
    15→    15→//   2. process.env.STRIPE_* (fallback — works without any DB row,
    16→    16→//      e.g. on first deploy before the admin has connected Stripe).
    17→    17→//
    18→    18→// So an admin can connect Stripe from the UI without editing .env,
    19→    19→// and existing .env-based deployments keep working.
    20→    20→//
    21→    21→// Stripe SDK is only ever imported server-side (this file is imported
    22→    22→// by API routes and webhook handlers). The publishable key is exposed
    23→    23→// to the client only via /api/billing/checkout (GET) (so the client
    24→    24→// can show the right UI without leaking the secret key).
    25→    25→// ============================================================
    26→    26→
    27→    27→import Stripe from 'stripe';
    28→    28→import { encrypt, decrypt, maskSecret } from '@/lib/encryption';
    29→    29→
    30→    30→// -------------------- DB-backed settings cache --------------------
    31→    31→// In-memory cache of the StripeSettings singleton row. Re-read on
    32→    32→// demand (invalidateStripeSettingsCache below) — call after every
    33→    33→// mutation (save / test-connection) so the next getStripeClient()
    34→    34→// picks up the new credentials without a dev-server restart.
    35→    35→interface ResolvedStripeSettings {
    36→    36→  mode: 'test' | 'live';
    37→    37→  secretKey: string | null; // resolved plaintext (decrypted) for the active mode
    38→    38→  publishableKey: string | null;
    39→    39→  webhookSecret: string | null;
    40→    40→  appUrl: string | null;
    41→    41→}
    42→    42→
    43→    43→let _resolvedCache: ResolvedStripeSettings | null = null;
    44→    44→let _resolvedCacheAt = 0;
    45→    45→const RESOLVED_TTL_MS = 5_000; // 5s — keeps reads fast but picks up edits quickly
    46→    46→
    47→    47→/** Invalidate the in-memory cache of resolved Stripe settings.
    48→    48→ *  Call after every saveStripeSettings() / mutation so the next
    49→    49→ *  getStripeClient() reads the fresh DB row. */
    50→    50→export function invalidateStripeSettingsCache(): void {
    51→    51→  _resolvedCache = null;
    52→    52→  _resolvedCacheAt = 0;
    53→    53→  _client = null; // force the SDK client to be reconstructed on next use
    54→    54→}
    55→    55→
    56→    56→/** Read the StripeSettings singleton row from the DB and decrypt
    57→    57→ *  the secret keys for the ACTIVE mode. Returns null when no row
    58→    58→ *  exists yet. Never throws — every error degrades to null. */
    59→    59→async function loadResolvedSettings(): Promise<ResolvedStripeSettings | null> {
    60→    60→  const { db } = await import('@/lib/db');
    61→    61→  try {
    62→    62→    const row = await db.stripeSettings.findUnique({ where: { id: 'singleton' } });
    63→    63→    if (!row) return null;
    64→    64→    const mode = (row.mode === 'live' ? 'live' : 'test') as 'test' | 'live';
    65→    65→    const secretEncrypted = mode === 'live' ? row.secretKeyLiveEncrypted : row.secretKeyTestEncrypted;
    66→    66→    const webhookEncrypted = mode === 'live' ? row.webhookSecretLiveEncrypted : row.webhookSecretTestEncrypted;
    67→    67→    const publishable = mode === 'live' ? row.publishableKeyLive : row.publishableKeyTest;
    68→    68→
    69→    69→    let secretKey: string | null = null;
    70→    70→    if (secretEncrypted) {
    71→    71→      try {
    72→    72→        secretKey = await decrypt(secretEncrypted);
    73→    73→      } catch {
    74→    74→        secretKey = null;
    75→    75→      }
    76→    76→    }
    77→    77→    let webhookSecret: string | null = null;
    78→    78→    if (webhookEncrypted) {
    79→    79→      try {
    80→    80→        webhookSecret = await decrypt(webhookEncrypted);
    81→    81→      } catch {
    82→    82→        webhookSecret = null;
    83→    83→      }
    84→    84→    }
    85→    85→    return {
    86→    86→      mode,
    87→    87→      secretKey,
    88→    88→      publishableKey: publishable ?? null,
    89→    89→      webhookSecret,
    90→    90→      appUrl: row.appUrl ?? null,
    91→    91→    };
    92→    92→  } catch {
    93→    93→    // Table not yet created, db not reachable, etc. → fall back to env.
    94→    94→    return null;
    95→    95→  }
    96→    96→}
    97→    97→
    98→    98→/** Get the resolved Stripe settings (DB → env fallback), with a
    99→    99→ *  5s in-memory cache so it's cheap to call repeatedly. Public so
   100→   100→ *  the admin settings page can read the masked view + the test
   101→   101→ *  connection route can show what it tested with. */
   102→   102→export async function getResolvedStripeSettings(): Promise<ResolvedStripeSettings & {
   103→   103→  source: 'db' | 'env' | 'none';
   104→   104→}> {
   105→   105→  // 1. DB row (admin-configured).
   106→   106→  const now = Date.now();
   107→   107→  if (_resolvedCache && now - _resolvedCacheAt < RESOLVED_TTL_MS) {
   108→   108→    return { ..._resolvedCache, source: 'db' };
   109→   109→  }
   110→   110→  const fromDb = await loadResolvedSettings();
   111→   111→  if (fromDb && (fromDb.secretKey || fromDb.publishableKey || fromDb.webhookSecret)) {
   112→   112→    _resolvedCache = fromDb;
   113→   113→    _resolvedCacheAt = now;
   114→   114→    return { ...fromDb, source: 'db' };
   115→   115→  }
   116→   116→  // 2. Env fallback.
   117→   117→  const envSettings: ResolvedStripeSettings = {
   118→   118→    mode: process.env.STRIPE_SECRET_KEY?.startsWith('sk_live_') ? 'live' : 'test',
   119→   119→    secretKey: process.env.STRIPE_SECRET_KEY && process.env.STRIPE_SECRET_KEY.trim().length > 0
   120→   120→      ? process.env.STRIPE_SECRET_KEY
   121→   121→      : null,
   122→   122→    publishableKey: process.env.STRIPE_PUBLISHABLE_KEY && process.env.STRIPE_PUBLISHABLE_KEY.trim().length > 0
   123→   123→      ? process.env.STRIPE_PUBLISHABLE_KEY
   124→   124→      : null,
   125→   125→    webhookSecret: process.env.STRIPE_WEBHOOK_SECRET && process.env.STRIPE_WEBHOOK_SECRET.trim().length > 0
   126→   126→      ? process.env.STRIPE_WEBHOOK_SECRET
   127→   127→      : null,
   128→   128→    appUrl: process.env.STRIPE_APP_URL ?? null,
   129→   129→  };
   130→   130→  // If env has no usable secret key either, source is 'none' (the
   131→   131→  // admin UI shows "Stripe is not connected — configure below").
   132→   132→  const hasEnvCreds = Boolean(envSettings.secretKey) || Boolean(envSettings.publishableKey) || Boolean(envSettings.webhookSecret);
   133→   133→  return { ...envSettings, source: hasEnvCreds ? 'env' : 'none' };
   134→   134→}

   136→   136→/** True when Stripe secret key is configured (DB or env). */
   137→   137→export async function isStripeConfiguredAsync(): Promise<boolean> {
   138→   138→  const s = await getResolvedStripeSettings();
   139→   139→  return Boolean(s.secretKey && s.secretKey.trim().length > 0);
   140→   140→}

   142→   142→/** Sync version — checks env vars only. Kept for backwards compat
   143→   143→ *  with callers that haven't been migrated to the async pattern
   144→   144→ *  (e.g. the webhook handler's pre-flight guard). Prefer
   145→   145→ *  isStripeConfiguredAsync() everywhere new. */
   146→   146→export function isStripeConfigured(): boolean {
   147→   147→  return Boolean(process.env.STRIPE_SECRET_KEY && process.env.STRIPE_SECRET_KEY.trim().length > 0);
   148→   148→}

   150→   150→let _client: Stripe | null = null;
   151→   151→
   152→   152→/** Get the shared Stripe SDK client. Reads the secret key from DB
   153→   153→ *  first, then falls back to env. Throws if neither is configured —
   154→   154→ *  call isStripeConfiguredAsync() first. */
   155→   155→export async function getStripeClient(): Promise<Stripe> {
   156→   156→  const settings = await getResolvedStripeSettings();
   157→   157→  const key = settings.secretKey;
   158→   158→  if (!key) throw new Error('Stripe is not configured. Set credentials in Platform Admin → Stripe Settings or in .env.');
   159→   159→  // If the cache is fresh and the key matches, reuse the SDK client.
   160→   160→  if (_client && _cachedClientKey === key) return _client;
   161→   161→  _client = new Stripe(key, {
   162→   162→    // Pin to a stable API version. Update together with the SDK.
   163→   163→    apiVersion: '2026-08-26.dahlia' as Stripe.LatestApiVersion,
   164→   164→    typescript: true,
   165→   165→    // Surface errors as detailed objects (don't hide fields).
   166→   166→    maxNetworkRetries: 2,
   167→   167→  });
   168→   168→  _cachedClientKey = key;
   169→   169→  return _client;
   170→   170→}
   171→   171→let _cachedClientKey: string | null = null;

   198→   198→export async function resolveStripePriceId(

   209→   209→/** Verify the Stripe webhook signature. Reads the webhook secret from
   210→   210→ *  DB first, then falls back to STRIPE_WEBHOOK_SECRET. Returns the
   211→   211→ *  verified event or null if verification fails (or Stripe isn't
   212→   212→ *  configured). */
   213→   213→export async function verifyStripeWebhook(
   214→   214→  rawBody: Buffer | string,
   215→   215→  signature: string,
   216→   216→): Promise<Stripe.Event | null> {
   217→   217→  const settings = await getResolvedStripeSettings();
   218→   218→  const secret = settings.webhookSecret;
   219→   219→  if (!secret) return null;
   220→   220→  try {
   221→   221→    const stripe = await getStripeClient();
   222→   222→    return await stripe.webhooks.constructEventAsync(
   223→   223→      rawBody instanceof Buffer ? rawBody : Buffer.from(rawBody),
   224→   224→      signature,
   225→   225→      secret,
   226→   226→    );
   227→   227→  } catch {
   228→   228→    return null;
   229→   229→  }
   230→   230→}

   233→   233→// SERVER-SIDE STRIPE SDK HELPERS.

   235→   235→// These helpers wrap raw Stripe SDK calls so callers (webhook

   237→   237→// don't have to repeat idempotency / best-effort-recovery /

   243→   243→/**
   244→   244→ * Return the Stripe Customer id for `user`, reusing an existing
   245→   245→ * Customer when `existingCustomerId` resolves to a live (non-deleted)
   246→   246→ * Stripe Customer, otherwise creating a new Customer tagged with
   247→   247→ * `metadata.userId`. On retrieve failure of an existing id, falls
   248→   248→ * through to create.
   249→   249→ */
   250→   250→export async function getOrCreateStripeCustomer(
   251→   251→  stripe: Stripe,
   252→   252→  user: { id: string; email: string; name?: string | null },
   253→   253→  existingCustomerId?: string | null,
   254→   254→): Promise<string> {
   255→   255→  if (existingCustomerId) {
   256→   256→    try {
   257→   257→      const customer = await stripe.customers.retrieve(existingCustomerId);
   258→   258→      if (customer && !(customer as { deleted?: boolean }).deleted) {
   259→   259→        return customer.id;
   260→   260→      }
   261→   261→    } catch {
   262→   262→      // fall through to create a new Customer
   263→   263→    }
   264→   264→  }
   265→   265→  const created = await stripe.customers.create({
   266→   266→    email: user.email,
   267→   267→    name: user.name ?? undefined,
   268→   268→    metadata: { userId: user.id },
   269→   269→  });
   270→   270→  return created.id;
   271→   271→}

   273→   273→/**
   274→   274→ * Mirror a local Coupon into a Stripe Coupon + Promotion Code so
   275→   275→ * Stripe Checkout can apply it as a `discounts[]` entry. Reuses
   276→   276→ * existing Stripe objects when present (idempotent). Returns the
   277→   277→ * Stripe Coupon id + Promotion Code id so the caller can cache
   278→   278→ * them on the local Coupon row.
   279→   279→ */
   280→   280→export async function createStripeCouponMirror(
   281→   281→  stripe: Stripe,
   282→   282→  coupon: {
   283→   283→    code: string;
   284→   284→    type: 'percent' | 'fixed';
   285→   285→    value: number;
   286→   286→    currency: string;
   287→   287→    maxRedemptions?: number | null;
   288→   288→    perCustomerLimit?: number | null;
   289→   289→    expiresAt?: string | null;
   290→   290→    startsAt?: string | null;
   291→   291→  },
   292→   292→): Promise<{ stripeCouponId: string; stripePromotionCodeId: string }> {
   293→   293→  // Derive a safe Stripe Coupon id: local_<code_lowered_alphanum>, truncate to 40 chars.
   294→   294→  const derivedId = `local_${coupon.code.toLowerCase().replace(/[^a-z0-9]/g, '')}`.slice(0, 40);
   295→   295→
   296→   296→  // Reuse an existing Stripe Coupon with that id, else create one.
   297→   297→  let stripeCouponId: string;
   298→   298→  try {
   299→   299→    const existing = await stripe.coupons.retrieve(derivedId);
   300→   300→    stripeCouponId = existing.id;
   301→   301→  } catch {
   302→   302→    const created = await stripe.coupons.create({
   303→   303→      id: derivedId,
   304→   304→      percent_off: coupon.type === 'percent' ? coupon.value : undefined,
   305→   305→      amount_off: coupon.type === 'fixed' ? coupon.value * 100 : undefined,
   306→   306→      currency: coupon.type === 'fixed' ? coupon.currency.toLowerCase() : undefined,
   307→   307→      duration: 'once',
   308→   308→      max_redemptions: coupon.maxRedemptions ?? undefined,
   309→   309→      redeem_by: coupon.expiresAt ? Math.floor(new Date(coupon.expiresAt).getTime() / 1000) : undefined,
   310→   310→      metadata: { localCode: coupon.code },
   311→   311→    });
   312→   312→    stripeCouponId = created.id;
   313→   313→  }
   314→   314→
   315→   315→  // Create-or-reuse a Stripe Promotion Code with code: coupon.code.
   316→   316→  let stripePromotionCodeId: string;
   317→   317→  const existingPromos = await stripe.promotionCodes.list({ code: coupon.code, limit: 1 });
   318→   318→  if (existingPromos.data.length > 0) {
   319→   319→    stripePromotionCodeId = existingPromos.data[0].id;
   320→   320→  } else {
   321→   321→    const promo = await stripe.promotionCodes.create({
   322→   322→      coupon: stripeCouponId,
   323→   323→      code: coupon.code,
   324→   324→      max_redemptions: coupon.maxRedemptions ?? undefined,
   325→   325→      expires_at: coupon.expiresAt ? Math.floor(new Date(coupon.expiresAt).getTime() / 1000) : undefined,
   326→   326→      metadata: { localCode: coupon.code },
   327→   327→    });
   328→   328→    stripePromotionCodeId = promo.id;
   329→   329→  }
   330→   330→
   331→   331→  return { stripeCouponId, stripePromotionCodeId };
   332→   332→}
   333→   333→
   334→   334→/**
   335→   335→ * Best-effort teardown of a local Coupon's Stripe mirror: deactivate
   336→   336→ * the Promotion Code (so it can no longer be redeemed) then delete the
   337→   337→ * Stripe Coupon. Never throws — all errors are swallowed. Safe to call
   338→   338→ * with null/undefined ids (no-op).
   339→   339→ */
   340→   340→export async function clearStripeCouponMirror(
   341→   341→  stripe: Stripe,
   342→   342→  stripeCouponId?: string | null,
   343→   343→  stripePromotionCodeId?: string | null,
   344→   344→): Promise<void> {
   345→   345→  if (stripePromotionCodeId) {
   346→   346→    try {
   347→   347→      await stripe.promotionCodes.update(stripePromotionCodeId, { active: false });
   348→   348→    } catch {
   349→   349→      // best-effort — never throw
   350→   350→    }
   351→   351→  }
   352→   352→  if (stripeCouponId) {
   353→   353→    try {
   354→   354→      await stripe.coupons.del(stripeCouponId);
   355→   355→    } catch {
   356→   356→      // best-effort — never throw
   357→   357→    }
   358→   358→  }
   359→   359→}

   361→   361→/**
   362→   362→ * Sync a local PlanConfig to Stripe by ensuring both a monthly and a
   363→   363→ * yearly Stripe Price exist (with `metadata.planId` + `metadata.interval`
   364→   364→ * set so they can be discovered later). Reuses existing Stripe Prices
   365→   365→ * / Products where possible. Returns the resolved { monthly, yearly }
   366→   366→ * Stripe Price ids (existing + any newly created).
   367→   367→ */
   368→   368→export async function syncPlanToStripe(
   369→   369→  stripe: Stripe,
   370→   370→  plan: {
   371→   371→    planId: string;
   372→   372→    name: string;
   373→   373→    priceMonthly: number;
   374→   374→    priceYearly: number;
   375→   375→    currency: string;
   376→   376→    stripePriceIdMonthly?: string | null;
   377→   377→    stripePriceIdYearly?: string | null;
   378→   378→  },
   379→   379→): Promise<{ stripePriceIdMonthly: string; stripePriceIdYearly: string }> {
   380→   380→  const resolveInterval = async (interval: 'monthly' | 'yearly'): Promise<string> => {
   381→   381→    const existing = interval === 'monthly' ? plan.stripePriceIdMonthly : plan.stripePriceIdYearly;
   382→   382→    if (existing && existing.trim().length > 0) return existing;
   383→   383→
   384→   384→    // Try to find an existing active Stripe Price with matching metadata.
   385→   385→    const list = await stripe.prices.list({ active: true, limit: 100 });
   386→   386→    const match = list.data.find(
   387→   387→      (p) => p.metadata?.planId === plan.planId && p.metadata?.interval === interval,
   388→   388→    );
   389→   389→    if (match) return match.id;
   390→   390→
   391→   391→    // Create-or-reuse a Stripe Product tagged with metadata.planId.
   392→   392→    let productId: string | undefined;
   393→   393→    try {
   394→   394→      const existingProducts = await stripe.products.list({ limit: 100 });
   395→   395→      const existingProduct = existingProducts.data.find(
   396→   396→        (p) => p.metadata?.planId === plan.planId,
   397→   397→      );
   398→   398→      if (existingProduct) productId = existingProduct.id;
   399→   399→    } catch {
   400→   400→      // fall through to create
   401→   401→    }
   402→   402→    if (!productId) {
   403→   403→      const product = await stripe.products.create({
   404→   404→        name: plan.name,
   405→   405→        metadata: { planId: plan.planId },
   406→   406→      });
   407→   407→      productId = product.id;
   408→   408→    }
   409→   409→
   410→   410→    // Stripe's recurring.interval enum is 'day' | 'week' | 'month' | 'year'
   411→   411→    // — map our local 'monthly'/'yearly' terms onto it.
   412→   412→    const stripeInterval: 'month' | 'year' = interval === 'monthly' ? 'month' : 'year';
   413→   413→    const price = await stripe.prices.create({
   414→   414→      unit_amount: (interval === 'monthly' ? plan.priceMonthly : plan.priceYearly) * 100,
   415→   415→      currency: plan.currency.toLowerCase(),
   416→   416→      recurring: { interval: stripeInterval },
   417→   417→      product: productId,
   418→   418→      metadata: { planId: plan.planId, interval },
   419→   419→    });
   420→   420→    return price.id;
   421→   421→  };
   422→   422→
   423→   423→  const stripePriceIdMonthly = await resolveInterval('monthly');
   424→   424→  const stripePriceIdYearly = await resolveInterval('yearly');
   425→   425→  return { stripePriceIdMonthly, stripePriceIdYearly };
   426→   426→}

   428→   428→/**
   429→   429→ * Cancel a Stripe Subscription either at period end (subscriber keeps
   430→   430→ * access until the period ends) or immediately. On failure, throws a
   431→   431→ * typed Error with the Stripe error code/message included.
   432→   432→ */
   433→   433→export async function cancelStripeSubscription(
   434→   434→  stripe: Stripe,
   435→   435→  stripeSubscriptionId: string,
   436→   436→  atPeriodEnd: boolean,
   437→   437→): Promise<void> {
   438→   438→  try {
   439→   439→    if (atPeriodEnd) {
   440→   440→      await stripe.subscriptions.update(stripeSubscriptionId, {
   441→   441→        cancel_at_period_end: true,

   710→   710→/** Save (encrypt + upsert) Stripe credentials. Empty-string values
   711→   711→ *  for secret keys mean "leave unchanged" (the existing encrypted
   712→   712→ *  value is preserved); null/undefined means "clear". Publishable
   713→   713→ *  keys are written as plaintext (non-secret). Returns the masked
   714→   714→ *  view. Never returns raw secret values. */
   715→   715→export async function saveStripeSettings(
   716→   716→  input: StripeSettingsInput,
   717→   717→  editorUserId: string,
   718→   718→): Promise<StripeSettingsView> {
   719→   719→  const { db } = await import('@/lib/db');
   720→   720→  const existing = await readStripeSettingsRow();
   721→   721→
   722→   722→  // ---- Secret keys: encrypt when provided; preserve when empty string ----
   723→   723→  const encryptOrPreserve = async (
   724→   724→    plaintext: string | undefined,
   725→   725→    existingEncrypted: string | null | undefined,
   726→   726→  ): Promise<string | null> => {
   727→   727→    if (plaintext === undefined || plaintext === null) return null; // clear
   728→   728→    if (plaintext === '') return existingEncrypted ?? null; // preserve
   729→   729→    return await encrypt(plaintext);
   730→   730→  };
   731→   731→
   732→   732→  const secretKeyTestEncrypted = await encryptOrPreserve(
   733→   733→    input.secretKeyTest,
   734→   734→    existing?.secretKeyTestEncrypted,
   735→   735→  );
   736→   736→  const secretKeyLiveEncrypted = await encryptOrPreserve(
   737→   737→    input.secretKeyLive,
   738→   738→    existing?.secretKeyLiveEncrypted,
   739→   739→  );
   740→   740→  const webhookSecretTestEncrypted = await encryptOrPreserve(
   741→   741→    input.webhookSecretTest,
   742→   742→    existing?.webhookSecretTestEncrypted,
   743→   743→  );
   744→   744→  const webhookSecretLiveEncrypted = await encryptOrPreserve(
   745→   745→    input.webhookSecretLive,
   746→   746→    existing?.webhookSecretLiveEncrypted,
   747→   747→  );
   748→   748→
   749→   749→  // ---- Publishable keys: plaintext (non-secret). Empty string
   750→   750→  //      preserves existing; null clears. ----
   751→   751→  const resolvePub = (v: string | undefined, existingVal: string | null): string | null => {
   752→   752→    if (v === undefined || v === null) return null;
   753→   753→    if (v === '') return existingVal ?? null;
   754→   754→    return v;
   755→   755→  };
   756→   756→  const publishableKeyTest = resolvePub(input.publishableKeyTest, existing?.publishableKeyTest ?? null);
   757→   757→  const publishableKeyLive = resolvePub(input.publishableKeyLive, existing?.publishableKeyLive ?? null);
   758→   758→
   759→   759→  // ---- App URL: empty string = unset (no preserve semantic — caller
   760→   760→  //      should pass the full URL). ----
   761→   761→  const appUrl = input.appUrl ?? existing?.appUrl ?? null;
   762→   762→
   763→   763→  await db.stripeSettings.upsert({
   764→   764→    where: { id: 'singleton' },
   765→   765→    create: {
   766→   766→      id: 'singleton',
   767→   767→      mode: input.mode,
   768→   768→      secretKeyTestEncrypted,
   769→   769→      secretKeyLiveEncrypted,
   770→   770→      webhookSecretTestEncrypted,
   771→   771→      webhookSecretLiveEncrypted,
   772→   772→      publishableKeyTest,
   773→   773→      publishableKeyLive,
   774→   774→      appUrl,
   775→   775→      updatedBy: editorUserId,
   776→   776→    },
   777→   777→    update: {
   778→   778→      mode: input.mode,
   779→   779→      secretKeyTestEncrypted,
   780→   780→      secretKeyLiveEncrypted,
   781→   781→      webhookSecretTestEncrypted,
   782→   782→      webhookSecretLiveEncrypted,
   783→   783→      publishableKeyTest,
   784→   784→      publishableKeyLive,
   785→   785→      appUrl,
   786→   786→      updatedBy: editorUserId,
   787→   787→    },
   788→   788→  });
   789→   789→
   790→   790→  // Force the next read to pick up the new credentials.
   791→   791→  invalidateStripeSettingsCache();
   792→   792→
   793→   793→  return getStripeSettingsForAdmin();
   794→   794→}

   796→   796→/** Test a Stripe connection by pinging the read-only /v1/balance
   797→   797→ *  endpoint. Uses the currently-configured credentials when
   798→   798→ *  `input` is null/undefined; otherwise constructs a temporary
   799→   799→ *  client from `input` (so the admin can test BEFORE saving).
   800→   800→ *  Records the outcome on the singleton row.
   801→   801→ *
   802→   802→ *  Returns `{ success: true, accountInfo }` on success, or
   803→   803→ *  `{ success: false, code, message }` on failure. Never throws. */
   804→   804→export async function testStripeConnection(
   805→   805→  input?: StripeSettingsInput,
   806→   806→): Promise<
   807→   807→  | { success: true; mode: 'test' | 'live'; accountInfo: { id: string; type: string; country: string; email: string | null; displayName: string | null } }
   808→   808→  | { success: false; mode: 'test' | 'live'; code: string; message: string }
   809→   809→> {
   810→   810→  // Resolve the secret key to test with.
   811→   811→  let secretKey: string | null = null;
   812→   812→  let mode: 'test' | 'live' = 'test';
   813→   813→
   814→   814→  if (input) {
   815→   815→    mode = input.mode;
   816→   816→    const candidate = mode === 'live' ? input.secretKeyLive : input.secretKeyTest;
   817→   817→    if (candidate && candidate.trim().length > 0) {
   818→   818→      secretKey = candidate;
   819→   819→    } else {
   820→   820→      // Fall back to the stored value for the chosen mode.
   821→   821→      const row = await readStripeSettingsRow();
   822→   822→      const enc = mode === 'live' ? row?.secretKeyLiveEncrypted : row?.secretKeyTestEncrypted;
   823→   823→      secretKey = await tryDecrypt(enc);
   824→   824→    }
   825→   825→  } else {
   826→   826→    const resolved = await getResolvedStripeSettings();
   827→   827→    secretKey = resolved.secretKey;
   828→   828→    mode = resolved.mode;
   829→   829→  }
   830→   830→
   831→   831→  if (!secretKey || secretKey.trim().length === 0) {
   832→   832→    const msg = 'No Stripe secret key is configured for the selected mode. Set the secret key first.';
   833→   833→    await recordTestOutcome(mode, 'error', msg);
   834→   834→    return { success: false, mode, code: 'NOT_CONFIGURED', message: msg };
   835→   835→  }
   836→   836→
   837→   837→  // Basic key-shape validation — sk_test_ / sk_live_ / rk_test_ /
   838→   838→  // rk_live_ (restricted keys also work for read-only calls).
   839→   839→  const keyShapeOk =
   840→   840→    secretKey.startsWith('sk_test_') ||
   841→   841→    secretKey.startsWith('sk_live_') ||
   842→   842→    secretKey.startsWith('rk_test_') ||
   843→   843→    secretKey.startsWith('rk_live_');
   844→   844→  if (!keyShapeOk) {
   845→   845→    const msg = 'The secret key does not look like a Stripe secret key (expected sk_test_… or sk_live_…).';
   846→   846→    await recordTestOutcome(mode, 'error', msg);
   847→   847→    return { success: false, mode, code: 'INVALID_KEY_SHAPE', message: msg };
   848→   848→  }
   849→   849→
   850→   850→  try {
   851→   851→    // Construct a TEMPORARY client (not the cached singleton) so we
   852→   852→    // can test credentials that haven't been saved yet.
   853→   853→    const tempClient = new Stripe(secretKey, {
   854→   854→      apiVersion: '2026-08-26.dahlia' as Stripe.LatestApiVersion,
   855→   855→      typescript: true,
   856→   856→      maxNetworkRetries: 1,
   857→   857→    });
   858→   858→    // /v1/balance is read-only and free of side effects — perfect for
   859→   859→    // testing credentials. It also works with restricted keys that
   860→   860→    // have the "Balance" read permission.
   861→   861→    const balance = await tempClient.balance.retrieve();
   862→   862→    // Resolve the account for richer info (id, country, display name).
   863→   863→    // This call requires the "Settings" read permission, which secret
   864→   864→    // keys always have. Fall back gracefully when restricted keys
   865→   865→    // can't reach it.
   866→   866→    let accountInfo: { id: string; type: string; country: string; email: string | null; displayName: string | null };
   867→   867→    try {
   868→   868→      const acct = await tempClient.accounts.retrieve();
   869→   869→      accountInfo = {
   870→   870→        id: acct.id,
   871→   871→        type: (acct as unknown as { type?: string }).type ?? 'standard',
   872→   872→        country: (acct as unknown as { country?: string }).country ?? '',
   873→   873→        email: (acct as unknown as { email?: string | null }).email ?? null,
   874→   874→        displayName: (acct as unknown as { display_name?: string | null }).display_name ?? null,
   875→   875→      };
   876→   876→    } catch {
   877→   877→      // Restricted key with no Settings read — still a successful
   878→   878→      // test (the secret key authenticates to Stripe).
   879→   879→      accountInfo = {
   880→   880→        id: '(restricted key)',
   881→   881→        type: 'restricted',
   882→   882→        country: '',
   883→   883→        email: null,
   884→   884→        displayName: null,
   885→   885→      };
   886→   886→    }
   887→   887→    const balanceAvailable = balance.available?.[0]?.amount;
   888→   888→    await recordTestOutcome(
   889→   889→      mode,
   890→   890→      'success',
   891→   891→      null,
   892→   892→      `account_id=${accountInfo.id}, available=${balanceAvailable ?? 'n/a'}`,
   893→   893→    );
   894→   894→    return { success: true, mode, accountInfo };
   895→   895→  } catch (err: unknown) {
   896→   896→    const e = (err ?? {}) as { code?: string; message?: string; type?: string };
   897→   897→    const code = e.code ?? e.type ?? 'stripe_error';
   898→   898→    const message = e.message ?? 'Unknown Stripe error';
   899→   899→    await recordTestOutcome(mode, 'error', `${code}: ${message}`);
   900→   900→    return { success: false, mode, code, message };
   901→   901→  }
   902→   902→}
