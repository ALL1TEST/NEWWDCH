   229→/**
   230→ * Derive the top-level section ID from the current browser hash.
   231→ * Uses EXPLICIT prefix matching — #settings/seo always returns 'Settings', never 'SEO'.
   232→ */
   233→function getSectionFromHash(hash: string): string | null {
   234→  const path = hash.replace(/^#\/?,?/, '').trim();
   235→  if (!path || path === 'dashboard') return null;
   236→
   237→  const firstSegment = path.split('/')[0];
   238→  return ROUTE_PREFIX_TO_SECTION[firstSegment] ?? null;
   239→}

   375→/*
   376→ * STATE RULE (decoupled by design):
   377→ *   • Inline submenu visibility  → controlled ONLY by openSection/onToggle
   378→ *     in AppSidebar (route-derived or user-toggled while expanded).
   379→ *   • Floating popover           → local state per item, ONLY used when the

   383→function useCollapsedRail(): boolean {
   384→  const { state, isMobile } = useSidebar();
   385→  return !isMobile && state === 'collapsed';

   388→/**
   389→ * Settings parent item rendered on the COLLAPSED 48px rail.
   390→ * Clicking it opens a portal-based popover to the RIGHT of the icon —
   391→ * the sidebar itself NEVER expands, and the popover anchors to this
   392→ * exact button so the "star/sparkle alignment" grid stays intact.
   393→ */
   394→function CollapsedParentNavItem({
   395→  item,
   396→  currentModule,
   397→  currentSubPage,
   398→}: {
   399→  item: NavItem;
   400→  currentModule: string;
   401→  currentSubPage: string | null;
   402→}) {
   403→  const [floatOpen, setFloatOpen] = useState(false);
   404→  const mod = hrefToModule(item.href);
   405→  const isActive = currentModule === mod;
   406→
   407→  const handleChildNavigate = (
   408→    e: React.MouseEvent<HTMLAnchorElement>,
   409→    child: NavItem,
   410→  ) => {
   411→    e.preventDefault();
   412→    e.stopPropagation();
   413→    setFloatOpen(false); // close the floating submenu after navigation…
   414→    // …but do NOT touch sidebar expansion state (rule #7).
   415→    const hash = child.href.replace(/^#/, '');
   416→    const parts = hash.split('/');
   417→    const childSubPage = parts[1] || null;
   418→    useNavigationStore.getState().navigate(parts[0], null, childSubPage);
   419→  };
   420→
   421→  return (
   422→    <SidebarMenuItem>
   423→      <Popover open={floatOpen} onOpenChange={setFloatOpen}>
   424→        <PopoverTrigger asChild>
   425→          <SidebarMenuButton
   426→            isActive={isActive || floatOpen}
   427→            aria-expanded={floatOpen}
   428→            aria-haspopup="menu"
   429→            tooltip={floatOpen ? undefined : item.label}
   430→            onClick={(e: React.MouseEvent) => {
   431→              e.preventDefault();
   432→              setFloatOpen((o) => !o);
   433→            }}
   434→          >
   435→            <NavIcon name={item.icon} />
   436→            <span>{item.label}</span>
   437→          </SidebarMenuButton>
   438→        </PopoverTrigger>
   439→        {/* Portal → document.body: immune to sidebar overflow clipping.
   440→            Radix handles Esc + outside-click closing and re-anchors at any
   441→            viewport size/zoom via avoidCollisions. */}
   442→        <PopoverContent
   443→          side="right"
   444→          align="start"
   445→          sideOffset={6}
   446→          collisionPadding={12}
   447→          className="w-56 rounded-md border bg-popover p-1.5 shadow-md"
   448→          onEscapeKeyDown={() => setFloatOpen(false)}
   449→        >
   450→          <ul
   451→            role="menu"
   452→            aria-label={`${item.label} submenu`}
   453→            className="flex min-w-0 list-none flex-col gap-0.5 p-0 m-0"
   454→          >
   455→            {item.children!.map((child) => {
   456→              const hash = child.href.replace(/^#/, '');
   457→              const parts = hash.split('/');
   458→              const childSubPage = parts[1] || null;
   459→              const isChildActive =
   460→                currentModule === parts[0] &&
   461→                (!childSubPage || currentSubPage === childSubPage);
   462→              return (
   463→                <SidebarMenuSubItem key={child.label} className="list-none p-0">
   464→                  <SidebarMenuSubButton asChild isActive={isChildActive}>
   465→                    <a
   466→                      href={child.href}
   467→                      role="menuitem"
   468→                      onClick={(e) => handleChildNavigate(e, child)}
   469→                    >
   470→                      <NavIcon name={child.icon} />
   471→                      <span>{child.label}</span>
   472→                    </a>
   473→                  </SidebarMenuSubButton>
   474→                </SidebarMenuSubItem>
   475→              );
   476→            })}
   477→          </ul>
   478→        </PopoverContent>
   479→      </Popover>
   480→    </SidebarMenuItem>
   481→  );
   482→}

   484→function ExpandableNavItem({
   485→  item,
   486→  currentModule,
   487→  currentSubPage,
   488→  isExpanded,
   489→  onToggle,
   490→}: {
   491→  item: NavItem;
   492→  currentModule: string;
   493→  currentSubPage: string | null;
   494→  isExpanded: boolean;
   495→  onToggle: () => void;
   496→}) {
   497→  const mod = hrefToModule(item.href);
   498→  const isActive = currentModule === mod;
   499→  const sectionId = `submenu-${item.label.toLowerCase().replace(/\s+/g, '-')}`;
   500→
   501→  return (
   502→    <SidebarMenuItem>
   503→      <SidebarMenuButton
   504→        tooltip={item.label}
   505→        isActive={isActive}
   506→        onClick={(e: React.MouseEvent) => {
   507→          e.preventDefault();
   508→          onToggle();
   509→        }}
   510→        aria-expanded={isExpanded}
   511→        aria-controls={sectionId}
   512→      >
   513→        <NavIcon name={item.icon} />
   514→        <span>{item.label}</span>
   515→        <ChevronRight
   516→          className={cn(
   517→            'ml-auto h-4 w-4 shrink-0 transition-transform duration-200',
   518→            isExpanded && 'rotate-90',
   519→          )}
   520→        />
   521→      </SidebarMenuButton>
   522→
   523→      {item.badge != null && (
   524→        <SidebarMenuBadge>
   525→          {item.badge}
   526→        </SidebarMenuBadge>
   527→      )}
   528→
   529→      <AccordionSubmenu isOpen={isExpanded} sectionLabel={item.label}>
   530→        {item.children!.map((child) => {
   531→          const hash = child.href.replace(/^#/, '');
   532→          const parts = hash.split('/');
   533→          const childMod = parts[0];
   534→          const childSubPage = parts[1] || null;
   535→          const isChildActive =
   536→            currentModule === childMod &&
   537→            (!childSubPage || currentSubPage === childSubPage);
   538→
   539→          return (
   540→            <SidebarMenuSubItem key={child.label}>
   541→              <SidebarMenuSubButton
   542→                asChild
   543→                isActive={isChildActive}
   544→              >
   545→                <a
   546→                  href={child.href}
   547→                  onClick={(e) => {
   548→                    e.preventDefault();
   549→                    e.stopPropagation();
   550→                    useNavigationStore
   551→                      .getState()
   552→                      .navigate(childMod, null, childSubPage);
   553→                  }}
   554→                >
   555→                  <NavIcon name={child.icon} />
   556→                  <span>{child.label}</span>
   557→                </a>
   558→              </SidebarMenuSubButton>
   559→            </SidebarMenuSubItem>
   560→          );
   561→        })}
   562→      </AccordionSubmenu>
   563→    </SidebarMenuItem>
   564→  );
   565→}

   607→function NavGroupSection({
   608→  group,
   609→  currentModule,
   610→  currentSubPage,
   611→  openSection,
   612→  onToggleSection,
   613→}: {
   614→  group: NavGroup;
   615→  currentModule: string;
   616→  currentSubPage: string | null;
   617→  openSection: string | null;
   618→  onToggleSection: (label: string) => void;
   619→}) {
   620→  const isCollapsedRail = useCollapsedRail();
   621→
   622→  return (
   623→    <SidebarGroup>
   624→      <SidebarGroupContent>
   625→        <SidebarMenu>
   626→          {group.items.map((item) => {
   627→            const hasChildren = item.children && item.children.length > 0;
   628→
   629→            if (hasChildren && isCollapsedRail) {
   630→              /* Collapsed rail → parents with children become floating
   631→                 popovers anchored to their own icon. The rail NEVER
   632→                 auto-expands (#4/#5/#6). */
   633→              return (
   634→                <CollapsedParentNavItem
   635→                  key={item.label}
   636→                  item={item}
   637→                  currentModule={currentModule}
   638→                  currentSubPage={currentSubPage}
   639→                />
   640→              );
   641→            }
   642→
   643→            if (hasChildren) {
   644→              return (
   645→                <ExpandableNavItem
   646→                  key={item.label}
   647→                  item={item}
   648→                  currentModule={currentModule}
   649→                  currentSubPage={currentSubPage}
   650→                  isExpanded={openSection === item.label}
   651→                  onToggle={() => onToggleSection(item.label)}
   652→                />
   653→              );
   654→            }
   655→
   656→            return (
   657→              <SimpleNavItem
   658→                key={item.label}
   659→                item={item}
   660→                currentModule={currentModule}
   661→              />
   662→            );
   663→          })}
   664→        </SidebarMenu>
   665→      </SidebarGroupContent>
   666→    </SidebarGroup>
   667→  );

   670→// -------------------- Main Sidebar Component --------------------

   672→export function AppSidebar() {
   673→  const user = useAuthStore((s) => s.user);
   674→  const logout = useAuthStore((s) => s.logout);
   675→  const currentModule = useNavigationStore((s) => s.currentModule);
   676→  const currentSubPage = useNavigationStore((s) => s.currentSubPage);
   677→  const openCommandPalette = useCommandPaletteStore((s) => s.open);
   678→  // Sidebar context hook — MUST run before any early return.
   679→  const sidebarCtx = useSidebar();
   680→
   681→  const userRole = user?.role;
   682→  const pagePermissions = user?.pagePermissions ?? null;
   683→  const visibleItems = useMemo(() => {
   684→    if (!userRole) return [];
   685→    return getVisibleNavItems(userRole, NAV_ITEMS, pagePermissions);
   686→  }, [userRole, pagePermissions]);
   687→
   688→  /*
   689→   * SINGLE SOURCE OF TRUTH for the expanded top-level section.
   690→   *
   691→   * manualOverride is a tri-state:
   692→   *   null  → not overridden; use route-derived value
   693→   *   ""   → explicitly closed by user
   694→   *   "SEO" → explicitly opened by user
   695→   *
   696→   * When the module changes (navigation event), we reset manualOverride
   697→   * to null so the route-derived value takes over.
   698→   */
   699→  const [manualOverride, setManualOverride] = useState<string | null>(null);
   700→  // Track previous module to detect navigation changes synchronously during render.
   701→  // This is the React-approved pattern for "derived state from props".
   702→  const [prevModule, setPrevModule] = useState(currentModule);
   703→  if (currentModule !== prevModule) {
   704→    setPrevModule(currentModule);
   705→    setManualOverride(null);
   706→  }
   707→
   708→  // Derive the section from the current route hash (EXPLICIT prefix matching)
   709→  const routeDerivedSection = useMemo(() => {
   710→    return getSectionFromHash(window.location.hash);
   711→  }, [currentModule, currentSubPage]);
   712→
   713→  // Compute the actual open section
   714→  const openSection = useMemo(() => {
   715→    if (manualOverride !== null) {
   716→      // User manually toggled: empty string = closed, non-empty = that section
   717→      return manualOverride || null;
   718→    }
   719→    return routeDerivedSection;
   720→  }, [manualOverride, routeDerivedSection]);
   721→
   722→  // Toggle a section open/closed
   723→  const handleToggleSection = useCallback(
   724→    (label: string) => {
   725→      setManualOverride((prev) => {
   726→        // Determine what's currently open (considering route-derived if no override)
   727→        const currentlyOpen =
   728→          prev !== null ? (prev || null) : routeDerivedSection;
   729→        // If clicking the same section that's open, close it
   730→        if (currentlyOpen === label) {
   731→          return ''; // explicitly closed
   732→        }
   733→        // Otherwise, open the clicked section (closing any other)
   734→        return label;
   735→      });
   736→    },
   737→    [routeDerivedSection],
   738→  );
   739→
   740→  if (!user) return null;
   741→
   742→  const groups = buildNavGroups(visibleItems);
   743→
   744→  return (
   745→    <Sidebar collapsible="icon">
   746→      {/* ---- Header: one 32px icon cell per row, all centered at x=24px ---- */}
   747→      <SidebarHeader className="px-2 py-3 shrink-0">
   748→        {/* Expanded: [logo][title][Search][…spacer…][Collapse toggle].
   749→            The Search icon sits directly next to the "CMS Admin" name per
   750→            reference; the collapse toggle keeps its original spot at the
   751→            far right of the sidebar header. */}
   752→        <div className="flex h-8 items-center gap-2 group-data-[collapsible=icon]:hidden">
   753→          <LogoMark />
   754→          <span className="truncate font-semibold text-sm tracking-tight whitespace-nowrap text-text-primary">
   755→            CMS Admin
   756→          </span>
   757→          <Tooltip>
   758→            <TooltipTrigger asChild>
   759→              <Button
   760→                variant="ghost"
   761→                size="icon"
   762→                className="h-8 w-8 shrink-0 ml-1 rounded-md"
   763→                onClick={openCommandPalette}
   764→                aria-label="Search"
   765→              >
   766→                <Search className="h-4 w-4" />
   767→                <span className="sr-only">Search</span>
   768→              </Button>
   769→            </TooltipTrigger>
   770→            <TooltipContent side="bottom">Search</TooltipContent>
   771→          </Tooltip>
   772→          <div className="ml-auto shrink-0">
   773→            <CollapseToggle side="bottom" />
   774→          </div>
   775→        </div>
   776→        {/* Collapsed rail: ONLY the "C" logo. The previously stacked
   777→            collapse-toggle icon was removed on purpose — no extra icon may
   778→            appear under/next to the logo.
   779→            Expand affordance: the logo cell itself toggles the sidebar
   780→            (same 32px grid cell, same x=24 center-line), and the invisible
   781→            SidebarRail edge strip keeps its native toggle behavior too. */}
   782→        <div className="hidden flex-col items-center group-data-[collapsible=icon]:flex">
   783→          <CollapsedLogoButton />
   784→        </div>
   785→      </SidebarHeader>
   786→
   787→      <SidebarSeparator className="mx-0" />
   788→
   789→      <SidebarContent className="sidebar-thin-scroll">
   790→        {groups.map((group) => (
   791→          <NavGroupSection
   792→            key={group.label}
   793→            group={group}
   794→            currentModule={currentModule}
   795→            currentSubPage={currentSubPage}
   796→            openSection={openSection}
   797→            onToggleSection={handleToggleSection}
   798→          />
   799→        ))}
   800→      </SidebarContent>
   801→
   802→      {/* Bottom separator shows in BOTH states: the collapsed footer now
   803→          carries the Search / Bell / avatar utility cluster, so there is
   804→          real content to separate below it in both modes. */}
   805→      <SidebarSeparator className="mx-0" />
   806→
   807→      <SidebarFooter className="shrink-0">
   808→        {/* Expanded: unchanged row layout [avatar][name/role][logout] */}
   809→        <div className="flex items-center gap-3 px-2 py-2 group-data-[collapsible=icon]:hidden">
   810→          <Avatar className="h-8 w-8 shrink-0">
   811→            <AvatarImage src={user.avatarUrl ?? undefined} alt={user.name} />
   812→            <AvatarFallback className="text-xs">
   813→              {getInitials(user.name)}
   814→            </AvatarFallback>
   815→          </Avatar>
   816→          <div className="flex flex-1 flex-col overflow-hidden">
   817→            <span className="truncate text-sm font-medium leading-tight">
   818→              {user.name}
   819→            </span>
   820→            <Badge variant="secondary" className="mt-0.5 h-4 w-fit text-[10px] px-1.5">
   821→              {user.role.replace(/_/g, ' ')}
   822→            </Badge>
   823→          </div>
   824→          <Tooltip>
   825→            <TooltipTrigger asChild>
   826→              <Button
   827→                variant="ghost"
   828→                size="icon"
   829→                className="h-8 w-8 shrink-0 rounded-md"
   830→                onClick={() => void logout()}
   831→                aria-label="Log out"
   832→              >
   833→                <LogOut className="h-4 w-4" />
   834→                <span className="sr-only">Log out</span>

   836→            </TooltipTrigger>
   837→            <TooltipContent side="top">Log out</TooltipContent>

   841→        {/* Collapsed rail: icon-only utility cluster + bare avatar.
