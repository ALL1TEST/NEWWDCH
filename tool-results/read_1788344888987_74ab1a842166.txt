   343→// -------------------- Categories & Tags Management Dialog --------------------
   344→
   345→function CategoriesTagsDialog({
   346→  open,
   347→  onOpenChange,
   348→}: {
   349→  open: boolean;
   350→  onOpenChange: (open: boolean) => void;
   351→}) {
   352→  const queryClient = useQueryClient();
   353→  const [newCategory, setNewCategory] = useState('');
   354→  const [newTag, setNewTag] = useState('');
   355→  const [activeTab, setActiveTab] = useState<'categories' | 'tags'>('categories');
   356→
   357→  const { data: categoriesData, isLoading: catLoading } = useQuery({
   358→    queryKey: queryKeys.categories.all,
   359→    queryFn: () => getApi<{ data: { id: string; name: string; slug?: string }[] } | { id: string; name: string; slug?: string }[]>('/api/categories?pageSize=200'),
   360→    enabled: open,
   361→    staleTime: 30_000,
   362→  });
   363→  const { data: tagsData, isLoading: tagsLoading } = useQuery({
   364→    queryKey: queryKeys.tags.all,
   365→    queryFn: () => getApi<{ data: { id: string; name: string; slug?: string; color?: string }[] } | { id: string; name: string; slug?: string; color?: string }[]>('/api/tags?pageSize=200'),
   366→    enabled: open,
   367→    staleTime: 30_000,
   368→  });
   369→
   370→  // Normalize responses — API may return either an array or { data: [...] }
   371→  const categories = useMemo(() => {
   372→    const d = categoriesData as unknown;
   373→    if (Array.isArray(d)) return d as { id: string; name: string; slug?: string }[];
   374→    if (d && typeof d === 'object' && 'data' in (d as Record<string, unknown>)) {
   375→      return (d as { data: { id: string; name: string; slug?: string }[] }).data;
   376→    }
   377→    return [];
   378→  }, [categoriesData]);
   379→  const tags = useMemo(() => {
   380→    const d = tagsData as unknown;
   381→    if (Array.isArray(d)) return d as { id: string; name: string; slug?: string; color?: string }[];
   382→    if (d && typeof d === 'object' && 'data' in (d as Record<string, unknown>)) {
   383→      return (d as { data: { id: string; name: string; slug?: string; color?: string }[] }).data;
   384→    }
   385→    return [];
   386→  }, [tagsData]);
   387→
   388→  const createCategoryMutation = useMutation({
   389→    mutationFn: (name: string) => postApi('/api/categories', { name }),
   390→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: queryKeys.categories.all }); toast.success('Category created'); },
   391→    onError: (err: Error) => toast.error(err.message || 'Failed to create category'),
   392→  });
   393→  const createTagMutation = useMutation({
   394→    mutationFn: (name: string) => postApi('/api/tags', { name }),
   395→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: queryKeys.tags.all }); toast.success('Tag created'); },
   396→    onError: (err: Error) => toast.error(err.message || 'Failed to create tag'),
   397→  });
   398→  const deleteCategoryMutation = useMutation({
   399→    mutationFn: (id: string) => deleteApi(`/api/categories/${id}`),
   400→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: queryKeys.categories.all }); toast.success('Category deleted'); },
   401→    onError: (err: Error) => toast.error(err.message || 'Failed to delete category'),
   402→  });
   403→  const deleteTagMutation = useMutation({
   404→    mutationFn: (id: string) => deleteApi(`/api/tags/${id}`),
   405→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: queryKeys.tags.all }); toast.success('Tag deleted'); },
   406→    onError: (err: Error) => toast.error(err.message || 'Failed to delete tag'),
   407→  });
   408→
   409→  const handleAddCategory = useCallback(() => {
   410→    const v = newCategory.trim();
   411→    if (!v) return;
   412→    createCategoryMutation.mutate(v);
   413→    setNewCategory('');
   414→  }, [newCategory, createCategoryMutation]);
   415→  const handleAddTag = useCallback(() => {
   416→    const v = newTag.trim();
   417→    if (!v) return;
   418→    createTagMutation.mutate(v);
   419→    setNewTag('');
   420→  }, [newTag, createTagMutation]);
   421→
   422→  return (
   423→    <Dialog open={open} onOpenChange={onOpenChange}>
   424→      <DialogContent className="sm:max-w-lg">
   425→        <DialogHeader>
   426→          <DialogTitle className="flex items-center gap-2">
   427→            <FolderOpen className="h-5 w-5" />
   428→            Categories &amp; Tags
   429→          </DialogTitle>
   430→          <DialogDescription>
   431→            Organize your articles with categories and tags. These are managed here instead of having separate pages.
   432→          </DialogDescription>
   433→        </DialogHeader>
   434→
   435→        <Tabs value={activeTab} onValueChange={(v) => setActiveTab(v as 'categories' | 'tags')}>
   436→          <TabsList className="grid w-full grid-cols-2">
   437→            <TabsTrigger value="categories">
   438→              <FolderOpen className="h-3.5 w-3.5 mr-1.5" />
   439→              Categories ({categories.length})
   440→            </TabsTrigger>
   441→            <TabsTrigger value="tags">
   442→              <Tag className="h-3.5 w-3.5 mr-1.5" />
   443→              Tags ({tags.length})
   444→            </TabsTrigger>
   445→          </TabsList>
   446→
   447→          {/* Categories Tab */}
   448→          <TabsContent value="categories" className="mt-4 space-y-3">
   449→            <div className="flex gap-2">
   450→              <Input
   451→                value={newCategory}
   452→                onChange={(e) => setNewCategory(e.target.value)}
   453→                onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); handleAddCategory(); } }}
   454→                placeholder="New category name..."
   455→                className="h-9 text-sm"
   456→              />
   457→              <Button
   458→                size="sm"
   459→                className="h-9 gap-1.5"
   460→                onClick={handleAddCategory}
   461→                disabled={createCategoryMutation.isPending || !newCategory.trim()}
   462→              >
   463→                {createCategoryMutation.isPending ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Plus className="h-3.5 w-3.5" />}
   464→                Add
   465→              </Button>
   466→            </div>
   467→            <div className="max-h-72 overflow-y-auto rounded-md border">
   468→              {catLoading ? (
   469→                <div className="p-4 space-y-2">
   470→                  {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-8 w-full" />)}
   471→                </div>
   472→              ) : categories.length === 0 ? (
   473→                <div className="flex flex-col items-center justify-center py-8 text-center">
   474→                  <FolderOpen className="h-8 w-8 text-muted-foreground/40 mb-2" />
   475→                  <p className="text-sm text-muted-foreground">No categories yet</p>
   476→                </div>
   477→              ) : (
   478→                <ul className="divide-y">
   479→                  {categories.map((cat) => (
   480→                    <li key={cat.id} className="flex items-center justify-between px-3 py-2 hover:bg-muted/30 transition-colors">
   481→                      <div className="min-w-0">
   482→                        <p className="text-sm font-medium truncate">{cat.name}</p>
   483→                        {cat.slug && <p className="text-[10px] text-muted-foreground font-mono truncate">{cat.slug}</p>}
   484→                      </div>
   485→                      <Button
   486→                        variant="ghost"
   487→                        size="icon"
   488→                        className="h-7 w-7 text-destructive hover:text-destructive shrink-0"
   489→                        onClick={() => deleteCategoryMutation.mutate(cat.id)}
   490→                        disabled={deleteCategoryMutation.isPending}
   491→                        title="Delete"
   492→                      >
   493→                        <Trash2 className="h-3.5 w-3.5" />
   494→                      </Button>
   495→                    </li>
   496→                  ))}
   497→                </ul>
   498→              )}
   499→            </div>
   500→          </TabsContent>
   501→
   502→          {/* Tags Tab */}
   503→          <TabsContent value="tags" className="mt-4 space-y-3">
   504→            <div className="flex gap-2">
   505→              <Input
   506→                value={newTag}
   507→                onChange={(e) => setNewTag(e.target.value)}
   508→                onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); handleAddTag(); } }}
   509→                placeholder="New tag name..."
   510→                className="h-9 text-sm"
   511→              />
   512→              <Button
   513→                size="sm"
   514→                className="h-9 gap-1.5"
