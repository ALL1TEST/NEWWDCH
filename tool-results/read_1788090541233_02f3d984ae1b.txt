   183→  addCommands() {

   187→        const { selection } = state;
   188→        let pos = -1;
   189→        state.doc.nodesBetween(selection.from, selection.to, (node, p) => {
   190→          if (node.type.name === 'table' && pos < 0) pos = p;
   191→        });
   192→        // Also resolve from $from ancestry
   193→        if (pos < 0) {
   194→          const $from = state.doc.resolve(selection.from);
   195→          for (let d = $from.depth; d > 0; d--) {
   196→            if ($from.node(d).type.name === 'table') { pos = $from.before(d); break; }
   197→          }
   198→        }
   199→        if (pos >= 0) {
   200→          const node = state.doc.nodeAt(pos);
   201→          if (node) {
   202→            const tr2 = tr.setNodeMarkup(pos, undefined, { ...node.attrs, borders: value });
   203→            if (dispatch) dispatch(tr2);
   204→            return true;
   205→          }
   206→        }
   207→        return false;
   208→      },
   209→      moveTableUp: () => ({ tr, state, dispatch }) => {
   210→        return moveTableImpl(state, tr, dispatch, 'up');
   211→      },
   212→      moveTableDown: () => ({ tr, state, dispatch }) => {
   213→        return moveTableImpl(state, tr, dispatch, 'down');
   214→      },
   215→    } as any;

   219→function moveTableImpl(state: any, tr: any, dispatch: any, direction: 'up' | 'down'): boolean {
   220→  const { selection } = state;
   221→  let tablePos = -1;
   222→  // Find table containing the selection
   223→  const $from = state.doc.resolve(selection.from);
   224→  for (let d = $from.depth; d > 0; d--) {
   225→    if ($from.node(d).type.name === 'table') { tablePos = $from.before(d); break; }
   226→  }
   227→  if (tablePos < 0) {
   228→    // Maybe selection is exactly the table — scan
   229→    state.doc.nodesBetween(selection.from, selection.to, (node: any, pos: number) => {
   230→      if (node.type.name === 'table' && tablePos < 0) tablePos = pos;
   231→    });
   232→  }
   233→  if (tablePos < 0) return false;
   234→  const tableNode = state.doc.nodeAt(tablePos);
   235→  if (!tableNode) return false;
   236→  const tableSize = tableNode.nodeSize;
   237→
   238→  if (direction === 'up') {
   239→    // Find previous top-level block
   240→    const $before = state.doc.resolve(tablePos);
   241→    const topLevelBefore = $before.before(1);
   242→    if (topLevelBefore < 0) return false;
   243→    const prevNode = state.doc.nodeAt(topLevelBefore);
   244→    if (!prevNode) return false;
   245→    const prevSize = prevNode.nodeSize;
   246→    const newTr = tr.delete(topLevelBefore, tablePos + tableSize);
   247→    newTr.insert(topLevelBefore, tableNode);
   248→    newTr.insert(topLevelBefore + tableNode.nodeSize, prevNode);
   249→    if (dispatch) dispatch(newTr);
   250→    return true;
   251→  } else {
   252→    const tableEnd = tablePos + tableSize;
   253→    const nextNode = state.doc.nodeAt(tableEnd);
   254→    if (!nextNode) return false;
   255→    const nextSize = nextNode.nodeSize;
   256→    const newTr = tr.delete(tablePos, tableEnd + nextSize);
   257→    newTr.insert(tablePos, nextNode);
   258→    newTr.insert(tablePos + nextNode.nodeSize, tableNode);
   259→    if (dispatch) dispatch(newTr);
   260→    return true;
   261→  }
   262→}

   282→    return ['span', { ...HTMLAttributes, class: 'editor-comment' }, 0];

   285→    return {
   286→      addComment: (data: { commentId: string; comment: string; author?: string }) => ({ tr, state, dispatch }) => {
   287→        const { from, to, empty } = state.selection;
   288→        if (empty) return false;
   289→        const text = state.doc.textBetween(from, to, '\n');
   290→        if (!text) return false;
   291→        // Remove any existing comment marks in the range first
   292→        const tr2 = tr.removeMark(from, to, state.schema.marks.comment);
   293→        // Add new comment mark
   294→        const mark = state.schema.marks.comment.create({
   295→          commentId: data.commentId,
   296→          comment: data.comment,
   297→          author: data.author || 'You',
   298→        });
   299→        tr2.addMark(from, to, mark);
   300→        if (dispatch) dispatch(tr2);
   301→        return true;
   302→      },
   303→      removeComment: () => ({ tr, state, dispatch }) => {
   304→        const { from, to } = state.selection;
   305→        const tr2 = tr.removeMark(from, to, state.schema.marks.comment);
   306→        if (dispatch) dispatch(tr2);
   307→        return true;
   308→      },
   309→    } as any;
