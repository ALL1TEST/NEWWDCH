   172→   172→// -------------------- Data mapping --------------------
   173→   173→
   174→   174→function mapArticles(rows: ArticleRow[] | undefined): CalendarEvent[] {
   175→   175→  if (!rows) return [];
   176→   176→  const events: CalendarEvent[] = [];
   177→   177→  for (const a of rows) {
   178→   178→    const iso = a.scheduledAt ?? a.publishedAt;
   179→   179→    if (!iso) continue;
   180→   180→    const date = parseISO(iso);
   181→   181→    if (Number.isNaN(date.getTime())) continue;
   182→   182→    events.push({
   183→   183→      id: `article-${a.id}`,
   184→   184→      title: a.title,
   185→   185→      type: 'article',
   186→   186→      status: a.status,
   187→   187→      date,
   188→   188→      raw: a,
   189→   189→    });
   190→   190→  }
   191→   191→  return events;
   192→   192→}
   193→   193→
   194→   194→function mapCampaigns(rows: CampaignRow[] | undefined): CalendarEvent[] {
   195→   195→  if (!rows) return [];
   196→   196→  const events: CalendarEvent[] = [];
   197→   197→  for (const c of rows) {
   198→   198→    if (!c.scheduledAt) continue;
   199→   199→    const date = parseISO(c.scheduledAt);
   200→   200→    if (Number.isNaN(date.getTime())) continue;
   201→   201→    events.push({
   202→   202→      id: `campaign-${c.id}`,
   203→   203→      title: c.name,
   204→   204→      type: 'campaign',
   205→   205→      status: c.status,
   206→   206→      date,
   207→   207→      raw: c,
   208→   208→    });
   209→   209→  }
   210→   210→  return events;
   211→   211→}
   212→   212→
   213→   213→// -------------------- Filter logic --------------------
   214→   214→
   215→   215→function eventMatchesFilter(ev: CalendarEvent, filter: FilterKey): boolean {
   216→   216→  switch (filter) {
   217→   217→    case 'all':
   218→   218→      return true;
   219→   219→    case 'articles':
   220→   220→      return ev.type === 'article';
   221→   221→    case 'campaigns':
   222→   222→      return ev.type === 'campaign';
   223→   223→    case 'drafts':
   224→   224→      return ev.status === 'DRAFT';
   225→   225→    case 'scheduled':
   226→   226→      // Campaigns with SCHEDULED status, or articles whose date came from scheduledAt
   227→   227→      if (ev.type === 'campaign') return ev.status === 'SCHEDULED';
   228→   228→      return Boolean(ev.raw?.scheduledAt);
   229→   229→    case 'published':
   230→   230→      return ev.status === 'PUBLISHED' || ev.status === 'SENT';
   231→   231→    case 'cancelled':
   232→   232→      return ev.status === 'CANCELLED' || ev.status === 'ARCHIVED';
   233→   233→    default:
   234→   234→      return true;
   235→   235→  }
   236→   236→}

   241→   241→
   242→   242→export function CalendarPage() {
   243→   243→  const navigate = useNavigationStore((s) => s.navigate);
   244→   244→
   245→   245→  const [view, setView] = useState<CalendarView>('month');
   246→   246→  const [referenceDate, setReferenceDate] = useState<Date>(new Date());
   247→   247→  const [filter, setFilter] = useState<FilterKey>('all');
   248→   248→  const [selectedEvent, setSelectedEvent] = useState<CalendarEvent | null>(null);
   249→   249→
   250→   250→  // -------- Data fetching --------
   251→   251→
   252→   252→  const { data: articlesData, isLoading: articlesLoading } = useQuery({
   253→   253→    queryKey: ['calendar', 'articles'],
   254→   254→    queryFn: () =>
   255→   255→      getApi<PaginatedResponse<ArticleRow>>('/api/content', {
   256→   256→        pageSize: 100,
   257→   257→        sort: 'publishedAt',
   258→   258→        order: 'asc',
   259→   259→      }),
   260→   260→    staleTime: 30_000,
   261→   261→  });
   262→   262→
   263→   263→  const { data: campaignsData, isLoading: campaignsLoading } = useQuery({
   264→   264→    queryKey: ['calendar', 'campaigns'],
   265→   265→    queryFn: () =>
   266→   266→      getApi<CampaignRow[]>('/api/campaigns', { pageSize: 100 }),
   267→   267→    staleTime: 30_000,
   268→   268→  });
   269→   269→
   270→   270→  const articles = useMemo(
   271→   271→    () => articlesData?.data ?? [],
   272→   272→    [articlesData],
   273→   273→  );
   274→   274→  const campaigns = useMemo(
   275→   275→    () => (Array.isArray(campaignsData) ? campaignsData : []),
   276→   276→    [campaignsData],
   277→   277→  );
   278→   278→
   279→   279→  const allEvents = useMemo<CalendarEvent[]>(() => {
   280→   280→    return [...mapArticles(articles), ...mapCampaigns(campaigns)].sort(
   281→   281→      (a, b) => a.date.getTime() - b.date.getTime(),
   282→   282→    );
   283→   283→  }, [articles, campaigns]);
   284→   284→
   285→   285→  const filteredEvents = useMemo(
   286→   286→    () => allEvents.filter((ev) => eventMatchesFilter(ev, filter)),
   287→   287→    [allEvents, filter],
   288→   288→  );
   289→   289→
   290→   290→  const isLoading = articlesLoading || campaignsLoading;
   291→   291→
   292→   292→  // -------- Navigation --------
   293→   293→
   294→   294→  const goPrev = useCallback(() => {
   295→   295→    setReferenceDate((d) => {
   296→   296→      if (view === 'month') return addMonths(d, -1);
   297→   297→      if (view === 'week') return addWeeks(d, -1);
   298→   298→      return addDays(d, -1); // day
   299→   299→    });
   300→   300→  }, [view]);
   301→   301→
   302→   302→  const goNext = useCallback(() => {
   303→   303→    setReferenceDate((d) => {
   304→   304→      if (view === 'month') return addMonths(d, 1);
   305→   305→      if (view === 'week') return addWeeks(d, 1);
   306→   306→      return addDays(d, 1); // day
   307→   307→    });
   308→   308→  }, [view]);
   309→   309→
   310→   310→  const goToday = useCallback(() => setReferenceDate(new Date()), []);
   311→   311→
   312→   312→  // -------- Header label --------
   313→   313→
   314→   314→  const periodLabel = useMemo(() => {
   315→   315→    if (view === 'month') return format(referenceDate, 'MMMM yyyy');
   316→   316→    if (view === 'week') {
   317→   317→      const ws = startOfWeek(referenceDate, { weekStartsOn: 0 });
   318→   318→      const we = endOfWeek(referenceDate, { weekStartsOn: 0 });
   319→   319→      if (isSameMonth(ws, we)) {
   320→   320→        return `${format(ws, 'MMM d')} – ${format(we, 'd, yyyy')}`;
   321→   321→      }
   322→   322→      return `${format(ws, 'MMM d')} – ${format(we, 'MMM d, yyyy')}`;
   323→   323→    }
   324→   324→    if (view === 'day') return format(referenceDate, 'EEEE, MMMM d, yyyy');
   325→   325→    return format(referenceDate, 'MMMM yyyy');
   326→   326→  }, [view, referenceDate]);
   327→   327→
   328→   328→  // -------- Schedule Content actions --------
   329→   329→
   330→   330→  const handleNewArticle = useCallback(() => {
   331→   331→    navigate('content', null, 'create');
   332→   332→  }, [navigate]);
   333→   333→
   334→   334→  const handleNewCampaign = useCallback(() => {
   335→   335→    navigate('newsletter', null, 'campaigns');
   336→   336→  }, [navigate]);
   337→   337→
   338→   338→  // -------- Event selection --------
   339→   339→
   340→   340→  const handleSelectEvent = useCallback((ev: CalendarEvent) => {
   341→   341→    setSelectedEvent(ev);
   342→   342→  }, []);
   343→   343→
   344→   344→  const handleCloseModal = useCallback(() => setSelectedEvent(null), []);
   345→   345→
   346→   346→  // -------- Render --------
   347→   347→
   348→   348→  return (
   349→   349→    <div className="space-y-5">
   350→   350→      {/* Header */}
   351→   351→      <CalendarHeader
   352→   352→        periodLabel={periodLabel}
   353→   353→        view={view}
   354→   354→        onPrev={goPrev}
   355→   355→        onNext={goNext}
   356→   356→        onToday={goToday}
   357→   357→        onViewChange={setView}
   358→   358→        onNewArticle={handleNewArticle}
   359→   359→        onNewCampaign={handleNewCampaign}
   360→   360→      />
   361→   361→
   362→   362→      {/* Filter bar */}
   363→   363→      <FilterBar value={filter} onChange={setFilter} counts={allEvents} />
   364→   364→
   365→   365→      {/* Calendar body */}
   366→   366→      <div className="rounded-xl border border-border bg-card shadow-sm">
   367→   367→        {isLoading ? (
   368→   368→          <CalendarSkeleton view={view} />
   369→   369→        ) : filteredEvents.length === 0 ? (
   370→   370→          <EmptyState
   371→   371→            icon={CalendarIcon}
   372→   372→            title="No scheduled content"
   373→   373→            description="Schedule an article or newsletter campaign to see it here."
   374→   374→            className="py-20"
   375→   375→          />
   376→   376→        ) : view === 'month' ? (
   377→   377→          <MonthView
   378→   378→            referenceDate={referenceDate}
   379→   379→            events={filteredEvents}
   380→   380→            onSelectEvent={handleSelectEvent}
   381→   381→          />
   382→   382→        ) : view === 'week' ? (
   383→   383→          <WeekView
   384→   384→            referenceDate={referenceDate}
   385→   385→            events={filteredEvents}
   386→   386→            onSelectEvent={handleSelectEvent}
   387→   387→          />
   388→   388→        ) : view === 'day' ? (
   389→   389→          <DayView
   390→   390→            referenceDate={referenceDate}
   391→   391→            events={filteredEvents}
   392→   392→            onSelectEvent={handleSelectEvent}
   393→   393→          />
   394→   394→        ) : null}
   395→   395→      </div>
   396→   396→
   397→   397→      {/* Event details modal */}
   398→   398→      <EventDetailsModal
   399→   399→        event={selectedEvent}
   400→   400→        onClose={handleCloseModal}
   401→   401→        onNavigate={navigate}
   402→   402→      />
   403→   403→    </div>
   404→   404→  );
   405→   405→}
