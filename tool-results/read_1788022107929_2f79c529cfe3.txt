    87→/** Resolve the badge id for the shared PlanBadge component. Falls
    88→ *  back to 'free' for any unknown plan id so future plans still
    89→ *  render with a sensible neutral badge. */
    90→function getPlanBadgeId(planId: string): 'free' | 'plus' | 'pro' | 'max' {
    91→  if (planId === 'free' || planId === 'plus' || planId === 'pro' || planId === 'max') return planId;
    92→  // Legacy 'beta' plan id (pre-migration) → render as 'plus' badge.
    93→  if (planId === 'beta') return 'plus';
    94→  return 'free';
    95→}

    97→// -------------------- Price formatting helpers --------------------

    99→/** Display symbol for a currency code. Plans & Pricing shows a
   100→ *  symbol (not the ISO text code) per the visual spec — e.g. "$7.50"
   101→ *  not "USD 7.50" / "CHF 7.50". The stored currency code in the DB is
   102→ *  unaffected; this is a display-only mapping. Unknown codes default
   103→ *  to '$' so the price always renders as a compact symbol + number. */
   104→function currencySymbol(code: string): string {
   105→  const map: Record<string, string> = {
   106→    USD: '$',
   107→    EUR: '€',
   108→    GBP: '£',
   109→    JPY: '¥',
   110→    CNY: '¥',
   111→  };
   112→  return map[(code ?? '').toUpperCase()] ?? '$';
   113→}
   114→
   115→/** Format an integer amount with the display symbol, no decimals.
   116→ *  e.g. (90, 'USD') → "$90"; (0, 'CHF') → "$0". */
   117→function formatPriceSymbol(amount: number, currency: string): string {
   118→  return `${currencySymbol(currency)}${amount.toLocaleString(undefined, {
   119→    minimumFractionDigits: 0,
   120→    maximumFractionDigits: 0,
   121→  })}`;
   122→}
   123→
   124→/** Format with 2 decimals — used for the monthly-equivalent of a
   125→ *  yearly price (yearly 90 → "$7.50"). Always two decimals so the
   126→ *  large primary price reads "$7.50" / "$40.83" / "$82.50". */
   127→function formatPriceSymbolMonthlyEquiv(amount: number, currency: string): string {
   128→  return `${currencySymbol(currency)}${amount.toLocaleString(undefined, {
   129→    minimumFractionDigits: 2,
   130→    maximumFractionDigits: 2,
   131→  })}`;
   132→}

   356→function EditPlanDialog({

   358→  open,
   359→  onOpenChange,

   370→  const [interval, setInterval] = useState<'monthly' | 'yearly'>(plan.interval);
   371→  const [active, setActive] = useState(plan.active);

   380→  const [stripePriceIdYearly, setStripePriceIdYearly] = useState(plan.stripePriceIdYearly ?? '');
   381→  const [clientDisplayOpen, setClientDisplayOpen] = useState(false);
   382→  const [stripeOpen, setStripeOpen] = useState(false);

   387→    queryKey: ['platform-currencies'],

   390→      const json = await res.json();

   392→      const list = (rows as { currency?: string }[])
   393→        .map((r) => r.currency)
   394→        .filter((c): c is string => Boolean(c) && typeof c === 'string');
   395→      const unique = Array.from(new Set(list));
   396→      return unique.length > 0 ? unique : ['CHF', 'USD', 'EUR', 'MAD'];
   397→    },
   398→    staleTime: 5 * 60 * 1000,
   399→  });
   400→  const currencies = currenciesQuery.data ?? ['CHF'];
   401→
   402→  // EditPlanDialog is conditionally rendered by the parent (mounted
   403→  // fresh each time the user opens it), so the useState initializers
   404→  // above already seed local state from the latest server snapshot.

   406→  // renders (and the React Compiler correctly flags it).

   410→    setPriceMonthly(String(plan.priceMonthly));
   411→    setPriceYearly(String(plan.priceYearly));

   415→    setFeatures(plan.features.join('\n'));

   417→    setLimits(plan.limits);
   418→    setFreePlanDurationDays(plan.freePlanDurationDays == null ? '' : String(plan.freePlanDurationDays));

   420→    setStripePriceIdYearly(plan.stripePriceIdYearly ?? '');

   424→    mutationFn: (patch: PlanPatch) =>
   425→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, patch),
   426→    onSuccess: (data) => {
   427→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   428→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   429→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   430→      toast.success(`${data.name} updated — changes are now live for clients.`);
   431→      onOpenChange(false);
   432→    },
   433→    onError: (err: unknown) => {
   434→      const e = err as { error?: { message?: string }; message?: string };
   435→      toast.error(
   436→        e?.error?.message ?? e?.message ?? 'Unable to save plan. You may need OWNER access.',
   437→      );
   438→    },
   439→  });

   441→  const buildPatch = (): PlanPatch => {
   442→    const monthly = Number(priceMonthly) || 0;

   454→      stripePriceIdYearly: stripePriceIdYearly.trim() === '' ? null : stripePriceIdYearly.trim(),

   456→      features: features
   457→        .split('\n')
   458→        .map((f) => f.trim())
   459→        .filter(Boolean),
   460→      entitlements,
   461→      limits: { ...limits, storageBytes: Number(limits.storageBytes) || 0 },
   462→    };
   463→  };
   464→
   465→  return (
   466→    <Dialog open={open} onOpenChange={onOpenChange}>
   467→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   468→        <DialogHeader>
   469→          <div className="flex items-center gap-2">
   470→            <DialogTitle className="text-base">Edit {plan.name}</DialogTitle>
   471→            <PlanBadge planId={getPlanBadgeId(plan.planId)} />
   472→          </div>
   473→          <DialogDescription className="text-xs">
   474→            Single source of truth — changes propagate to the Client Billing page, MRR and
   475→            server-side entitlement enforcement.
   476→          </DialogDescription>
   477→        </DialogHeader>
   478→
   479→        <div className="space-y-5">
   480→          {/* -------------------- Basic Information -------------------- */}
   481→          <section className="space-y-3">
   482→            <h4 className="text-sm font-semibold">Basic Information</h4>
   483→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   484→              <div className="space-y-1">
   485→                <Label className="text-xs">Name</Label>
   486→                <Input
   487→                  value={name}
   488→                  onChange={(e) => setName(e.target.value)}
   489→                  className="h-9"
   490→                />
   491→              </div>
   492→              <div className="space-y-1">
   493→                <Label className="text-xs">Monthly Price</Label>
   494→                <Input
   495→                  type="number"
   496→                  value={priceMonthly}
   497→                  onChange={(e) => setPriceMonthly(e.target.value)}
   498→                  className="h-9"
   499→                />
   500→              </div>
   501→              <div className="space-y-1">
   502→                <Label className="text-xs">Yearly Price</Label>
