    87→/** Resolve the badge id for the shared PlanBadge component. Falls
    88→ *  back to 'free' for any unknown plan id so future plans still
    89→ *  render with a sensible neutral badge. */
    90→function getPlanBadgeId(planId: string): 'free' | 'plus' | 'pro' | 'max' {
    91→  if (planId === 'free' || planId === 'plus' || planId === 'pro' || planId === 'max') return planId;
    92→  // Legacy 'beta' plan id (pre-migration) → render as 'plus' badge.
    93→  if (planId === 'beta') return 'plus';
    94→  return 'free';
    95→}

    97→// -------------------- Price formatting helpers --------------------
    98→
    99→/** Display symbol for a currency code. Plans & Pricing shows a
   100→ *  symbol (not the ISO text code) per the visual spec — e.g. "$7.50"
   101→ *  not "USD 7.50" / "CHF 7.50". The stored currency code in the DB is
   102→ *  unaffected; this is a display-only mapping. Unknown codes default
   103→ *  to '$' so the price always renders as a compact symbol + number. */
   104→function currencySymbol(code: string): string {
   105→  const map: Record<string, string> = {
   106→    USD: '$',
   107→    EUR: '€',
   108→    GBP: '£',
   109→    JPY: '¥',
   110→    CNY: '¥',
   111→  };
   112→  return map[(code ?? '').toUpperCase()] ?? '$';
   113→}
   114→
   115→/** Format an integer amount with the display symbol, no decimals.
   116→ *  e.g. (90, 'USD') → "$90"; (0, 'CHF') → "$0". */
   117→function formatPriceSymbol(amount: number, currency: string): string {
   118→  return `${currencySymbol(currency)}${amount.toLocaleString(undefined, {
   119→    minimumFractionDigits: 0,
   120→    maximumFractionDigits: 0,
   121→  })}`;
   122→}
   123→
   124→/** Format with 2 decimals — used for the monthly-equivalent of a
   125→ *  yearly price (yearly 90 → "$7.50"). Always two decimals so the
   126→ *  large primary price reads "$7.50" / "$40.83" / "$82.50". */
   127→function formatPriceSymbolMonthlyEquiv(amount: number, currency: string): string {
   128→  return `${currencySymbol(currency)}${amount.toLocaleString(undefined, {
   129→    minimumFractionDigits: 2,
   130→    maximumFractionDigits: 2,
   131→  })}`;
   132→}

   362→function EditPlanDialog({

   387→  const [clientDisplayOpen, setClientDisplayOpen] = useState(false);

   393→    queryKey: ['platform-currencies'],

   396→      const json = await res.json();
   397→      const rows = Array.isArray(json) ? json : (json.data ?? []);
   398→      const list = (rows as { currency?: string }[])
   399→        .map((r) => r.currency)
   400→        .filter((c): c is string => Boolean(c) && typeof c === 'string');
   401→      const unique = Array.from(new Set(list));
   402→      return unique.length > 0 ? unique : ['CHF', 'USD', 'EUR', 'MAD'];
   403→    },
   404→    staleTime: 5 * 60 * 1000,
   405→  });
   406→  const currencies = currenciesQuery.data ?? ['CHF'];

   408→  // EditPlanDialog is conditionally rendered by the parent (mounted
   409→  // fresh each time the user opens it), so the useState initializers

   431→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, patch),
   432→    onSuccess: (data) => {

   434→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   435→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   436→      toast.success(`${data.name} updated — changes are now live for clients.`);
   437→      onOpenChange(false);
   438→    },
   439→    onError: (err: unknown) => {
   440→      const e = err as { error?: { message?: string }; message?: string };

   442→        e?.error?.message ?? e?.message ?? 'Unable to save plan. You may need OWNER access.',

   460→      stripePriceIdYearly: stripePriceIdYearly.trim() === '' ? null : stripePriceIdYearly.trim(),

   462→      features: features
   463→        .split('\n')
   464→        .map((f) => f.trim())
   465→        .filter(Boolean),
   466→      entitlements,
   467→      limits: { ...limits, storageBytes: Number(limits.storageBytes) || 0 },
   468→    };
   469→  };
   470→
   471→  return (
   472→    <Dialog open={open} onOpenChange={onOpenChange}>
   473→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   474→        <DialogHeader>
   475→          <div className="flex items-center gap-2">
   476→            <DialogTitle className="text-base">Edit {plan.name}</DialogTitle>
   477→            <PlanBadge planId={getPlanBadgeId(plan.planId)} />
   478→          </div>
   479→          <DialogDescription className="text-xs">
   480→            Single source of truth — changes propagate to the Client Billing page, MRR and
   481→            server-side entitlement enforcement.
   482→          </DialogDescription>
   483→        </DialogHeader>
   484→
   485→        <div className="space-y-5">
   486→          {/* -------------------- Basic Information -------------------- */}
   487→          <section className="space-y-3">
   488→            <h4 className="text-sm font-semibold">Basic Information</h4>
   489→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   490→              <div className="space-y-1">
   491→                <Label className="text-xs">Name</Label>
   492→                <Input
   493→                  value={name}
   494→                  onChange={(e) => setName(e.target.value)}
   495→                  className="h-9"
   496→                />
   497→              </div>
