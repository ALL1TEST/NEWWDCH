   134→function dateGroupKey(iso: string): DateGroupKey {
   135→  const now = new Date();
   136→  const d = new Date(iso);
   137→  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
   138→  const ts = d.getTime();
   139→  if (ts >= startOfToday) return 'today';
   140→  if (ts >= startOfToday - 86_400_000) return 'yesterday';
   141→  if (ts >= startOfToday - 7 * 86_400_000) return 'thisWeek';
   142→  return 'older';
   143→}

   145→// ==================== Notification Message (Read More / Read Less) ====================
   146→// Admin User behavior preserved: uses ResizeObserver to detect whether
   147→// the message text is actually truncated by line-clamp-2. "Read More"
   148→// only appears when the text genuinely overflows the 2-line preview.
   149→// Base message styling follows the Platform Admin row design
   150→// (text-xs text-muted-foreground line-clamp-2).
   151→
   152→function NotificationMessage({
   153→  message,
   154→  isUnread,
   155→}: {
   156→  message: string;
   157→  isUnread: boolean;
   158→}) {
   159→  const paraRef = useRef<HTMLParagraphElement>(null);
   160→  const [isExpanded, setIsExpanded] = useState(false);
   161→  const [isTruncated, setIsTruncated] = useState(false);
   162→
   163→  useEffect(() => {
   164→    const el = paraRef.current;
   165→    if (!el) return;
   166→
   167→    const check = () => {
   168→      if (isExpanded) {
   169→        // When expanded, always show "Read Less"
   170→        setIsTruncated(true);
   171→        return;
   172→      }
   173→      // scrollHeight > clientHeight means line-clamp-2 is clipping content
   174→      const isClipped = el.scrollHeight > el.clientHeight + 2;
   175→      setIsTruncated(isClipped);
   176→    };
   177→
   178→    const raf = requestAnimationFrame(check);
   179→    const observer = new ResizeObserver(() => requestAnimationFrame(check));
   180→    observer.observe(el);
   181→
   182→    return () => {
   183→      cancelAnimationFrame(raf);
   184→      observer.disconnect();
   185→    };
   186→  }, [message, isExpanded]);
   187→
   188→  return (
   189→    <>
   190→      <p
   191→        ref={paraRef}
   192→        className={cn(
   193→          'text-xs text-muted-foreground mt-0.5',
   194→          !isExpanded && 'line-clamp-2',
   195→          isUnread && 'text-foreground/70',
   196→        )}
   197→      >
   198→        {message}
   199→      </p>
   200→      {isTruncated && (
   201→        <button
   202→          type="button"
   203→          className="text-xs text-primary hover:underline mt-0.5"
   204→          onClick={(e) => {
   205→            e.stopPropagation();
   206→            setIsExpanded((prev) => !prev);
   207→          }}
   208→        >
   209→          {isExpanded ? 'Read Less' : 'Read More'}
   210→        </button>
   211→      )}
   212→    </>
   213→  );
   214→}

   218→interface NotificationRowProps {

   225→function NotificationRow({

   228→  onMarkUnread,
   229→  onDelete,

   235→    if (isUnread) onMarkRead(notification.id);
   236→  }, [isUnread, notification.id, onMarkRead]);
   237→
   238→  return (
   239→    <div
   240→      role="button"
   241→      tabIndex={0}
   242→      onClick={handleClick}
   243→      onKeyDown={(e) => {
   244→        if (e.key === 'Enter' || e.key === ' ') {
   245→          e.preventDefault();
   246→          handleClick();
   247→        }
   248→      }}
   249→      className={cn(
   250→        'group relative flex items-start gap-3 px-4 py-3 transition-colors cursor-pointer',
   251→        'hover:bg-accent/40',
   252→        isUnread ? 'bg-primary/[0.03]' : 'bg-card',

   256→      {isUnread && (
   257→        <span className="absolute left-0 top-0 bottom-0 w-0.5 bg-primary" aria-hidden />
   258→      )}
   259→
   260→      {/* Type icon avatar */}

   263→          'flex h-8 w-8 shrink-0 items-center justify-center rounded-full mt-0.5',
   264→          config.color,
   265→        )}
   266→      >
   267→        {config.icon}
   268→      </div>
   269→
   270→      {/* Content */}
   271→      <div className="flex-1 min-w-0 pr-2">
   272→        <div className="flex items-center gap-2">
   273→          <h4
   274→            className={cn(
   275→              'text-sm truncate',
   276→              isUnread ? 'font-semibold' : 'font-medium',
   277→            )}
   278→          >
   279→            {notification.title}
   280→          </h4>
   281→        </div>
   282→        <NotificationMessage
   283→          message={notification.message}
   284→          isUnread={isUnread}
   285→        />
   286→      </div>
   287→
   288→      {/* Timestamp + unread dot */}
   289→      <div className="shrink-0 flex flex-col items-end gap-1.5 pt-0.5">
   290→        <span className="text-[11px] text-muted-foreground whitespace-nowrap">
   291→          {formatRelativeTime(notification.createdAt)}
   292→        </span>
   293→        {isUnread && (
   294→          <span
   295→            className="h-1.5 w-1.5 rounded-full bg-primary"
   296→            aria-label="Unread"
   297→          />
   298→        )}
   299→      </div>
   300→
   301→      {/* Hover actions */}
   302→      <div className="absolute right-2 bottom-2 flex gap-0.5 opacity-0 group-hover:opacity-100 focus-within:opacity-100 transition-opacity">
   303→        <button
   304→          type="button"
   305→          title={isUnread ? 'Mark as read' : 'Mark as unread'}
   306→          onClick={(e) => {
   307→            e.stopPropagation();
   308→            if (isUnread) onMarkRead(notification.id);
   309→            else onMarkUnread(notification.id);
   310→          }}
   311→          className="h-7 w-7 inline-flex items-center justify-center rounded-md text-muted-foreground hover:bg-accent hover:text-foreground"
   312→        >
   313→          {isUnread ? <Check className="h-3.5 w-3.5" /> : <Undo2 className="h-3.5 w-3.5" />}
   314→        </button>
   315→        <button
   316→          type="button"
   317→          title="Delete"
   318→          onClick={(e) => {
   319→            e.stopPropagation();
   320→            onDelete(notification.id);
   321→          }}
   322→          className="h-7 w-7 inline-flex items-center justify-center rounded-md text-muted-foreground hover:bg-rose-50 hover:text-rose-600 dark:hover:bg-rose-950/40 dark:hover:text-rose-400"
   323→        >
   324→          <Trash2 className="h-3.5 w-3.5" />
   325→        </button>
   326→      </div>
   327→    </div>
   328→  );
   329→}

   575→    const buckets: Record<DateGroupKey, NotificationItem[]> = {
   576→      today: [],
   577→      yesterday: [],
   578→      thisWeek: [],
   579→      older: [],
   580→    };
   581→    for (const n of displayNotifications) {
   582→      buckets[dateGroupKey(n.createdAt)].push(n);
   583→    }
   584→    return DATE_GROUP_ORDER.filter((k) => buckets[k].length > 0).map((k) => ({
   585→      key: k,
   586→      label: DATE_GROUP_LABEL[k],
   587→      items: buckets[k],
   588→    }));
   589→  }, [displayNotifications]);
   590→
   591→  // Intersection observer for infinite scroll
   592→  const observerRef = useCallback(
   593→    (node: HTMLDivElement | null) => {
   594→      if (isFetchingNextPage || !hasNextPage) return;
   595→      if (!node) return;
   596→      const observer = new IntersectionObserver(
   597→        (entries) => {
   598→          if (entries[0].isIntersecting) {
   599→            fetchNextPage();
   600→          }
   601→        },
   602→        { threshold: 0.1 },
   603→      );
   604→      observer.observe(node);
   605→      return () => observer.disconnect();
   606→    },
   607→    [isFetchingNextPage, hasNextPage, fetchNextPage],
   608→  );
   609→
   610→  const subtitle =
   611→    unreadCount > 0
   612→      ? `You have ${unreadCount} unread notification${unreadCount > 1 ? 's' : ''}`
   613→      : 'No unread notifications';
   614→
   615→  return (
   616→    <div className="space-y-6">
   617→      {/* ==================== Header (Platform Admin styling) ==================== */}
   618→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3 pb-1">
   619→        <div className="flex items-start gap-3">
   620→          <div>
   621→            <h1 className="text-xl font-bold tracking-tight text-foreground">Notifications</h1>
   622→            <p className="text-sm text-muted-foreground mt-1">{subtitle}</p>
   623→          </div>
   624→        </div>
   625→        {(
   626→          <div className="flex flex-wrap items-center gap-2">
   627→            {/* Comment Notifications toggle (Admin User setting — preserved) */}
   628→            <div className="flex items-center gap-2">
   629→              <Switch
   630→                id="comment-notif-toggle"
   631→                checked={commentNotifsOn}
   632→                onCheckedChange={(v) => toggleCommentNotifs.mutate(v)}
   633→                disabled={toggleCommentNotifs.isPending}
   634→              />
   635→              <Label htmlFor="comment-notif-toggle" className="text-xs text-muted-foreground whitespace-nowrap cursor-pointer hidden sm:inline">
   636→                Comment Notifications
   637→              </Label>
   638→            </div>
   639→            {/* Mark All Read */}
   640→            <Button
   641→              variant="outline"
   642→              size="sm"
   643→              onClick={() => markAllReadMutation.mutate()}
   644→              disabled={markAllReadMutation.isPending || unreadCount === 0}
   645→            >
   646→              {markAllReadMutation.isPending ? (
   647→                <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   648→              ) : (
   649→                <CheckCheck className="h-4 w-4 mr-2" />
   650→              )}
   651→              Mark All Read
   652→            </Button>
   653→            {/* Delete All */}
   654→            <Button
   655→              variant="outline"
   656→              size="sm"
   657→              className="text-rose-600 border-rose-300 hover:bg-rose-50 hover:text-rose-700 dark:text-rose-400 dark:border-rose-800 dark:hover:bg-rose-950/40"
   658→              onClick={() => setDeleteAllDialogOpen(true)}
   659→              disabled={deleteAllMutation.isPending || allNotifications.length === 0}
   660→            >
   661→              {deleteAllMutation.isPending ? (
   662→                <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   663→              ) : (
   664→                <Trash2 className="h-4 w-4 mr-2" />
   665→              )}
   666→              Delete All
   667→            </Button>
   668→          </div>
   669→        )}
   670→      </div>
   671→
   672→      {/* ==================== Filter + search toolbar (Platform Admin design) ==================== */}
   673→      <div className="flex flex-col sm:flex-row sm:items-center gap-3">
   674→        {/* Admin User filter tabs — All / Unread / Info / Success / Warning / Error */}
   675→        <div className="flex flex-wrap items-center gap-1">
   676→          {FILTER_TABS.map((tab) => {
   677→            const active = activeFilter === tab.value;
   678→            return (
   679→              <button
   680→                key={tab.value}
   681→                type="button"
   682→                onClick={() => setActiveFilter(tab.value)}
   683→                className={cn(
   684→                  'text-xs px-3 py-1.5 rounded-md font-medium transition-colors',
   685→                  active
   686→                    ? 'bg-primary text-primary-foreground'
   687→                    : 'bg-muted text-muted-foreground hover:bg-muted/80 hover:text-foreground',
   688→                )}
   689→              >
   690→                {tab.label}
   691→              </button>
   692→            );
   693→          })}
   694→        </div>
   695→        {/* Search (client-side) */}
   696→        <div className="relative sm:ml-auto sm:w-64">
   697→          <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   698→          <Input
   699→            value={search}
   700→            onChange={(e) => setSearch(e.target.value)}
   701→            placeholder="Search notifications…"
   702→            className="h-9 pl-8 pr-8"
   703→          />
   704→          {search && (
   705→            <button
   706→              type="button"
   707→              onClick={() => setSearch('')}
   708→              className="absolute right-2 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   709→              aria-label="Clear search"
   710→            >
   711→              <X className="h-3.5 w-3.5" />
   712→            </button>
   713→          )}
   714→        </div>
   715→      </div>
   716→
   717→      {/* ==================== Notification list (Platform Admin card design) ==================== */}
   718→      <Card>
   719→        <CardContent className="p-0">
   720→          {isLoading && displayNotifications.length === 0 ? (
   721→            // Loading state
   722→            <div className="flex items-center justify-center py-16">
   723→              <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   724→            </div>
   725→          ) : displayNotifications.length === 0 ? (
   726→            // Empty state
   727→            <div className="flex flex-col items-center justify-center py-16 text-center">
   728→              <div className="h-12 w-12 rounded-full bg-muted flex items-center justify-center mb-3">
   729→                <BellOff className="h-6 w-6 text-muted-foreground" />
   730→              </div>
   731→              <h3 className="text-sm font-semibold">No notifications</h3>
   732→              <p className="text-xs text-muted-foreground mt-1">
   733→                {search
   734→                  ? 'No notifications match your search.'
   735→                  : activeFilter !== 'all'
   736→                    ? `No ${activeFilter.toLowerCase()} notifications found`
   737→                    : "You're all caught up!"}
   738→              </p>
   739→            </div>
   740→          ) : (
   741→            // Grouped list
   742→            <div className="divide-y">
   743→              {grouped.map((group) => (
   744→                <div key={group.key}>
   745→                  {/* Group header */}
   746→                  <div className="px-4 py-2 bg-muted/30 flex items-center gap-2">
   747→                    <h3 className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">
   748→                      {group.label}
   749→                    </h3>
   750→                    <span className="text-[11px] text-muted-foreground">
   751→                      · {group.items.length}
   752→                    </span>
   753→                  </div>
   754→                  {/* Group items */}
   755→                  <div className="divide-y">
   756→                    {group.items.map((notification) => (
   757→                      <NotificationRow
   758→                        key={notification.id}
   759→                        notification={notification}
   760→                        onMarkRead={handleMarkRead}
   761→                        onMarkUnread={handleMarkUnread}
   762→                        onDelete={handleDeleteSingle}
   763→                      />
   764→                    ))}
   765→                  </div>
   766→                </div>
   767→              ))}
   768→
   769→              {/* Infinite scroll sentinel + summary */}
   770→              <div ref={observerRef} className="flex items-center justify-center py-4">
   771→                {isFetchingNextPage && (
   772→                  <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" />
   773→                )}
   774→                {!hasNextPage && displayNotifications.length > 0 && (
   775→                  <p className="text-xs text-muted-foreground">
   776→                    {totalItems} notification{totalItems !== 1 ? 's' : ''} total
   777→                  </p>
   778→                )}
   779→              </div>
   780→            </div>
   781→          )}
   782→        </CardContent>
   783→      </Card>
   784→
   785→      {/* Delete All Confirmation Dialog (Admin User text — unchanged) */}
   786→      <ConfirmDialog
   787→        open={deleteAllDialogOpen}
   788→        onOpenChange={setDeleteAllDialogOpen}
   789→        title="Delete All Notifications"
   790→        description="Are you sure you want to delete ALL notifications? This action cannot be undone and will permanently remove every notification."
   791→        confirmLabel="Delete All"
   792→        variant="destructive"
   793→        onConfirm={() => deleteAllMutation.mutate()}
   794→        isLoading={deleteAllMutation.isPending}
   795→      />
   796→    </div>
   797→  );
   798→}
