     1→     1→// ============================================================
     2→     2→// STRIPE CLIENT + CONFIGURATION GUARD.
     3→     3→// ============================================================
     4→     4→// This is the SINGLE place that resolves Stripe credentials and
     5→     5→// constructs the Stripe SDK client. The rest of the codebase calls
     6→     6→// `isStripeConfigured()` to gate real billing; when false, checkout
     7→     7→// returns 503 with a clear "not configured" message and NEVER fakes
     8→     8→// a successful payment.
     9→     9→//
    10→    10→// CREDENTIAL RESOLUTION ORDER (per credential type):
    11→    11→//   1. The StripeSettings singleton row in the DB (admin-configured
    12→    12→//      via Platform Admin → Stripe Settings). Secret keys are stored
    13→    13→//      AES-256-GCM encrypted; only the masked form is returned to the
    14→    14→//      frontend. Publishable keys are stored as plaintext.
    15→    15→//   2. process.env.STRIPE_* (fallback — works without any DB row,
    16→    16→//      e.g. on first deploy before the admin has connected Stripe).
    17→    17→//
    18→    18→// So an admin can connect Stripe from the UI without editing .env,
    19→    19→// and existing .env-based deployments keep working.
    20→    20→//
    21→    21→// Stripe SDK is only ever imported server-side (this file is imported
    22→    22→// by API routes and webhook handlers). The publishable key is exposed
    23→    23→// to the client only via /api/billing/checkout (GET) (so the client
    24→    24→// can show the right UI without leaking the secret key).
    25→    25→// ============================================================
    26→    26→
    27→    27→import Stripe from 'stripe';
    28→    28→import { encrypt, decrypt, maskSecret } from '@/lib/encryption';
    29→    29→
    30→    30→/** The Unicode bullet character (U+2022) used by `maskSecret` to mask
    31→    31→ *  the middle of secret keys. Used to detect when the admin has
    32→    32→ *  copy-pasted the MASKED form of a secret back into the input —
    33→    33→ *  storing that would silently break Stripe API calls (Stripe would
    34→    34→ *  reject the masked string as an "Invalid API Key"). */
    35→    35→const MASK_BULLET = '\u2022'; // = '•'
    36→    36→
    37→    37→/** True when the value looks like a masked secret (contains the mask
    38→    38→ *  bullet character). Used to reject masked form values at the API
    39→    39→ *  boundary so an admin can never accidentally persist the masked
    40→    40→ *  form as the real secret. */
    41→    41→export function isMaskedSecretValue(value: string): boolean {
    42→    42→  return typeof value === 'string' && value.includes(MASK_BULLET);
    43→    43→}

    46→    46→// -------------------- DB-backed settings cache --------------------
    47→    47→// In-memory cache of the StripeSettings singleton row. Re-read on
    48→    48→// demand (invalidateStripeSettingsCache below) — call after every
    49→    49→// mutation (save / test-connection) so the next getStripeClient()
    50→    50→// picks up the new credentials without a dev-server restart.
    51→    51→interface ResolvedStripeSettings {
    52→    52→  mode: 'test' | 'live';
    53→    53→  secretKey: string | null; // resolved plaintext (decrypted) for the active mode
    54→    54→  publishableKey: string | null;
    55→    55→  webhookSecret: string | null;
    56→    56→  appUrl: string | null;
    57→    57→}
    58→    58→
    59→    59→let _resolvedCache: ResolvedStripeSettings | null = null;
    60→    60→let _resolvedCacheAt = 0;
    61→    61→const RESOLVED_TTL_MS = 5_000; // 5s — keeps reads fast but picks up edits quickly
    62→    62→
    63→    63→/** Invalidate the in-memory cache of resolved Stripe settings.
    64→    64→ *  Call after every saveStripeSettings() / mutation so the next
    65→    65→ *  getStripeClient() reads the fresh DB row. */
    66→    66→export function invalidateStripeSettingsCache(): void {
    67→    67→  _resolvedCache = null;
    68→    68→  _resolvedCacheAt = 0;
    69→    69→  _client = null; // force the SDK client to be reconstructed on next use
    70→    70→}
    71→    71→
    72→    72→/** Read the StripeSettings singleton row from the DB and decrypt
    73→    73→ *  the secret keys for the ACTIVE mode. Returns null when no row
    74→    74→ *  exists yet. Never throws — every error degrades to null. */
    75→    75→async function loadResolvedSettings(): Promise<ResolvedStripeSettings | null> {
    76→    76→  const { db } = await import('@/lib/db');
    77→    77→  try {
    78→    78→    const row = await db.stripeSettings.findUnique({ where: { id: 'singleton' } });
    79→    79→    if (!row) return null;
    80→    80→    const mode = (row.mode === 'live' ? 'live' : 'test') as 'test' | 'live';
    81→    81→    const secretEncrypted = mode === 'live' ? row.secretKeyLiveEncrypted : row.secretKeyTestEncrypted;
    82→    82→    const webhookEncrypted = mode === 'live' ? row.webhookSecretLiveEncrypted : row.webhookSecretTestEncrypted;
    83→    83→    const publishable = mode === 'live' ? row.publishableKeyLive : row.publishableKeyTest;
    84→    84→
    85→    85→    let secretKey: string | null = null;
    86→    86→    if (secretEncrypted) {
    87→    87→      try {
    88→    88→        secretKey = await decrypt(secretEncrypted);
    89→    89→      } catch {
    90→    90→        secretKey = null;
    91→    91→      }
    92→    92→    }
    93→    93→    let webhookSecret: string | null = null;
    94→    94→    if (webhookEncrypted) {
    95→    95→      try {
    96→    96→        webhookSecret = await decrypt(webhookEncrypted);
    97→    97→      } catch {
    98→    98→        webhookSecret = null;
    99→    99→      }
   100→   100→    }
   101→   101→    return {
   102→   102→      mode,
   103→   103→      secretKey,
   104→   104→      publishableKey: publishable ?? null,
   105→   105→      webhookSecret,
   106→   106→      appUrl: row.appUrl ?? null,
   107→   107→    };
   108→   108→  } catch {
   109→   109→    // Table not yet created, db not reachable, etc. → fall back to env.
   110→   110→    return null;
   111→   111→  }
   112→   112→}
   113→   113→
   114→   114→/** Get the resolved Stripe settings (DB → env fallback), with a
   115→   115→ *  5s in-memory cache so it's cheap to call repeatedly. Public so
   116→   116→ *  the admin settings page can read the masked view + the test
   117→   117→ *  connection route can show what it tested with. */
   118→   118→export async function getResolvedStripeSettings(): Promise<ResolvedStripeSettings & {
   119→   119→  source: 'db' | 'env' | 'none';
   120→   120→}> {
   121→   121→  // 1. DB row (admin-configured).
   122→   122→  const now = Date.now();
   123→   123→  if (_resolvedCache && now - _resolvedCacheAt < RESOLVED_TTL_MS) {
   124→   124→    return { ..._resolvedCache, source: 'db' };
   125→   125→  }
   126→   126→  const fromDb = await loadResolvedSettings();
   127→   127→  if (fromDb && (fromDb.secretKey || fromDb.publishableKey || fromDb.webhookSecret)) {
   128→   128→    _resolvedCache = fromDb;
   129→   129→    _resolvedCacheAt = now;
   130→   130→    return { ...fromDb, source: 'db' };
   131→   131→  }
   132→   132→  // 2. Env fallback.
   133→   133→  const envSettings: ResolvedStripeSettings = {
   134→   134→    mode: process.env.STRIPE_SECRET_KEY?.startsWith('sk_live_') ? 'live' : 'test',
   135→   135→    secretKey: process.env.STRIPE_SECRET_KEY && process.env.STRIPE_SECRET_KEY.trim().length > 0
   136→   136→      ? process.env.STRIPE_SECRET_KEY
   137→   137→      : null,
   138→   138→    publishableKey: process.env.STRIPE_PUBLISHABLE_KEY && process.env.STRIPE_PUBLISHABLE_KEY.trim().length > 0
   139→   139→      ? process.env.STRIPE_PUBLISHABLE_KEY
   140→   140→      : null,
   141→   141→    webhookSecret: process.env.STRIPE_WEBHOOK_SECRET && process.env.STRIPE_WEBHOOK_SECRET.trim().length > 0
   142→   142→      ? process.env.STRIPE_WEBHOOK_SECRET
   143→   143→      : null,
   144→   144→    appUrl: process.env.STRIPE_APP_URL ?? null,
   145→   145→  };
   146→   146→  // If env has no usable secret key either, source is 'none' (the
   147→   147→  // admin UI shows "Stripe is not connected — configure below").
   148→   148→  const hasEnvCreds = Boolean(envSettings.secretKey) || Boolean(envSettings.publishableKey) || Boolean(envSettings.webhookSecret);
   149→   149→  return { ...envSettings, source: hasEnvCreds ? 'env' : 'none' };
   150→   150→}

   152→   152→/** True when Stripe secret key is configured (DB or env). */
   153→   153→export async function isStripeConfiguredAsync(): Promise<boolean> {
   154→   154→  const s = await getResolvedStripeSettings();
   155→   155→  return Boolean(s.secretKey && s.secretKey.trim().length > 0);
   156→   156→}

   158→   158→/** Sync version — checks env vars only. Kept for backwards compat
   159→   159→ *  with callers that haven't been migrated to the async pattern
   160→   160→ *  (e.g. the webhook handler's pre-flight guard). Prefer
   161→   161→ *  isStripeConfiguredAsync() everywhere new. */
   162→   162→export function isStripeConfigured(): boolean {
   163→   163→  return Boolean(process.env.STRIPE_SECRET_KEY && process.env.STRIPE_SECRET_KEY.trim().length > 0);
   164→   164→}

   166→   166→let _client: Stripe | null = null;
   167→   167→
   168→   168→/** Get the shared Stripe SDK client. Reads the secret key from DB
   169→   169→ *  first, then falls back to env. Throws if neither is configured —
   170→   170→ *  call isStripeConfiguredAsync() first. */
   171→   171→export async function getStripeClient(): Promise<Stripe> {
   172→   172→  const settings = await getResolvedStripeSettings();
   173→   173→  const key = settings.secretKey;
   174→   174→  if (!key) throw new Error('Stripe is not configured. Set credentials in Platform Admin → Stripe Settings or in .env.');
   175→   175→  // If the cache is fresh and the key matches, reuse the SDK client.
   176→   176→  if (_client && _cachedClientKey === key) return _client;
   177→   177→  _client = new Stripe(key, {
   178→   178→    // Pin to a stable API version. Update together with the SDK.
   179→   179→    apiVersion: '2026-08-26.dahlia' as Stripe.LatestApiVersion,
   180→   180→    typescript: true,
   181→   181→    // Surface errors as detailed objects (don't hide fields).
   182→   182→    maxNetworkRetries: 2,
   183→   183→  });
   184→   184→  _cachedClientKey = key;
   185→   185→  return _client;
   186→   186→}
   187→   187→let _cachedClientKey: string | null = null;

   214→   214→export async function resolveStripePriceId(

   225→   225→/** Verify the Stripe webhook signature. Reads the webhook secret from
   226→   226→ *  DB first, then falls back to STRIPE_WEBHOOK_SECRET. Returns the
   227→   227→ *  verified event or null if verification fails (or Stripe isn't
   228→   228→ *  configured). */
   229→   229→export async function verifyStripeWebhook(
   230→   230→  rawBody: Buffer | string,
   231→   231→  signature: string,
   232→   232→): Promise<Stripe.Event | null> {
   233→   233→  const settings = await getResolvedStripeSettings();
   234→   234→  const secret = settings.webhookSecret;
   235→   235→  if (!secret) return null;
   236→   236→  try {
   237→   237→    const stripe = await getStripeClient();
   238→   238→    return await stripe.webhooks.constructEventAsync(
   239→   239→      rawBody instanceof Buffer ? rawBody : Buffer.from(rawBody),
   240→   240→      signature,
   241→   241→      secret,
   242→   242→    );
   243→   243→  } catch {
   244→   244→    return null;
   245→   245→  }
   246→   246→}

   259→   259→/**
   260→   260→ * Return the Stripe Customer id for `user`, reusing an existing
   261→   261→ * Customer when `existingCustomerId` resolves to a live (non-deleted)
   262→   262→ * Stripe Customer, otherwise creating a new Customer tagged with
   263→   263→ * `metadata.userId`. On retrieve failure of an existing id, falls
   264→   264→ * through to create.
   265→   265→ */
   266→   266→export async function getOrCreateStripeCustomer(
   267→   267→  stripe: Stripe,
   268→   268→  user: { id: string; email: string; name?: string | null },
   269→   269→  existingCustomerId?: string | null,
   270→   270→): Promise<string> {
   271→   271→  if (existingCustomerId) {
   272→   272→    try {
   273→   273→      const customer = await stripe.customers.retrieve(existingCustomerId);
   274→   274→      if (customer && !(customer as { deleted?: boolean }).deleted) {
   275→   275→        return customer.id;
   276→   276→      }
   277→   277→    } catch {
   278→   278→      // fall through to create a new Customer
   279→   279→    }
   280→   280→  }
   281→   281→  const created = await stripe.customers.create({
   282→   282→    email: user.email,
   283→   283→    name: user.name ?? undefined,
   284→   284→    metadata: { userId: user.id },
   285→   285→  });
   286→   286→  return created.id;
   287→   287→}

   289→   289→/**
   290→   290→ * Mirror a local Coupon into a Stripe Coupon + Promotion Code so
   291→   291→ * Stripe Checkout can apply it as a `discounts[]` entry. Reuses
   292→   292→ * existing Stripe objects when present (idempotent). Returns the
   293→   293→ * Stripe Coupon id + Promotion Code id so the caller can cache
   294→   294→ * them on the local Coupon row.
   295→   295→ */
   296→   296→export async function createStripeCouponMirror(
   297→   297→  stripe: Stripe,
   298→   298→  coupon: {
   299→   299→    code: string;
   300→   300→    type: 'percent' | 'fixed';
   301→   301→    value: number;
   302→   302→    currency: string;
   303→   303→    maxRedemptions?: number | null;
   304→   304→    perCustomerLimit?: number | null;
   305→   305→    expiresAt?: string | null;
   306→   306→    startsAt?: string | null;
   307→   307→  },
   308→   308→): Promise<{ stripeCouponId: string; stripePromotionCodeId: string }> {
   309→   309→  // Derive a safe Stripe Coupon id: local_<code_lowered_alphanum>, truncate to 40 chars.
   310→   310→  const derivedId = `local_${coupon.code.toLowerCase().replace(/[^a-z0-9]/g, '')}`.slice(0, 40);
   311→   311→
   312→   312→  // Reuse an existing Stripe Coupon with that id, else create one.
   313→   313→  let stripeCouponId: string;
   314→   314→  try {
   315→   315→    const existing = await stripe.coupons.retrieve(derivedId);
   316→   316→    stripeCouponId = existing.id;
   317→   317→  } catch {
   318→   318→    const created = await stripe.coupons.create({
   319→   319→      id: derivedId,
   320→   320→      percent_off: coupon.type === 'percent' ? coupon.value : undefined,
   321→   321→      amount_off: coupon.type === 'fixed' ? coupon.value * 100 : undefined,
   322→   322→      currency: coupon.type === 'fixed' ? coupon.currency.toLowerCase() : undefined,
   323→   323→      duration: 'once',
   324→   324→      max_redemptions: coupon.maxRedemptions ?? undefined,
   325→   325→      redeem_by: coupon.expiresAt ? Math.floor(new Date(coupon.expiresAt).getTime() / 1000) : undefined,
   326→   326→      metadata: { localCode: coupon.code },
   327→   327→    });
   328→   328→    stripeCouponId = created.id;
   329→   329→  }
   330→   330→
   331→   331→  // Create-or-reuse a Stripe Promotion Code with code: coupon.code.
   332→   332→  let stripePromotionCodeId: string;
   333→   333→  const existingPromos = await stripe.promotionCodes.list({ code: coupon.code, limit: 1 });
   334→   334→  if (existingPromos.data.length > 0) {
   335→   335→    stripePromotionCodeId = existingPromos.data[0].id;
   336→   336→  } else {
   337→   337→    const promo = await stripe.promotionCodes.create({
   338→   338→      coupon: stripeCouponId,
   339→   339→      code: coupon.code,
   340→   340→      max_redemptions: coupon.maxRedemptions ?? undefined,
   341→   341→      expires_at: coupon.expiresAt ? Math.floor(new Date(coupon.expiresAt).getTime() / 1000) : undefined,
   342→   342→      metadata: { localCode: coupon.code },
   343→   343→    });
   344→   344→    stripePromotionCodeId = promo.id;
   345→   345→  }
   346→   346→
   347→   347→  return { stripeCouponId, stripePromotionCodeId };
   348→   348→}
   349→   349→
   350→   350→/**
   351→   351→ * Best-effort teardown of a local Coupon's Stripe mirror: deactivate
   352→   352→ * the Promotion Code (so it can no longer be redeemed) then delete the
   353→   353→ * Stripe Coupon. Never throws — all errors are swallowed. Safe to call
   354→   354→ * with null/undefined ids (no-op).
   355→   355→ */
   356→   356→export async function clearStripeCouponMirror(
   357→   357→  stripe: Stripe,
   358→   358→  stripeCouponId?: string | null,
   359→   359→  stripePromotionCodeId?: string | null,
   360→   360→): Promise<void> {
   361→   361→  if (stripePromotionCodeId) {
   362→   362→    try {
   363→   363→      await stripe.promotionCodes.update(stripePromotionCodeId, { active: false });
   364→   364→    } catch {
   365→   365→      // best-effort — never throw
   366→   366→    }
   367→   367→  }
   368→   368→  if (stripeCouponId) {
   369→   369→    try {
   370→   370→      await stripe.coupons.del(stripeCouponId);
   371→   371→    } catch {
   372→   372→      // best-effort — never throw
   373→   373→    }
   374→   374→  }
   375→   375→}

   377→   377→/**
   378→   378→ * Sync a local PlanConfig to Stripe by ensuring both a monthly and a
   379→   379→ * yearly Stripe Price exist (with `metadata.planId` + `metadata.interval`
   380→   380→ * set so they can be discovered later). Reuses existing Stripe Prices
   381→   381→ * / Products where possible. Returns the resolved { monthly, yearly }
   382→   382→ * Stripe Price ids (existing + any newly created).
   383→   383→ *
   384→   384→ * PRICE-CHANGE DETECTION: when an existing Stripe Price ID is provided
   385→   385→ * AND the corresponding local price (priceMonthly / priceYearly)
   386→   386→ * changed since the last sync, this function retrieves the Stripe
   387→   387→ * Price, compares its `unit_amount` against the locally-configured
   388→   388→ * amount, and — on mismatch — creates a NEW Stripe Price with the
   389→   389→ * updated amount (the OLD Stripe Price is NOT archived — existing
   390→   390→ * subscriptions may still be on it; Stripe keeps them on the original
   391→   391→ * price until they upgrade / cancel / prorate). The local plan row
   392→   392→ * is then updated with the new Stripe Price ID by the caller.
   393→   393→ */
   394→   394→export async function syncPlanToStripe(
   395→   395→  stripe: Stripe,
   396→   396→  plan: {
   397→   397→    planId: string;
   398→   398→    name: string;
   399→   399→    priceMonthly: number;
   400→   400→    priceYearly: number;
   401→   401→    currency: string;
   402→   402→    stripePriceIdMonthly?: string | null;
   403→   403→    stripePriceIdYearly?: string | null;
   404→   404→  },
   405→   405→): Promise<{ stripePriceIdMonthly: string; stripePriceIdYearly: string }> {
   406→   406→  const resolveInterval = async (interval: 'monthly' | 'yearly'): Promise<string> => {
   407→   407→    const existing = interval === 'monthly' ? plan.stripePriceIdMonthly : plan.stripePriceIdYearly;
   408→   408→    const expectedAmount = (interval === 'monthly' ? plan.priceMonthly : plan.priceYearly) * 100;
   409→   409→
   410→   410→    if (existing && existing.trim().length > 0) {
   411→   411→      // PRICE-CHANGE CHECK: retrieve the existing Stripe Price and
   412→   412→      // compare its unit_amount against the locally-configured amount.
   413→   413→      // On mismatch (the admin edited the price), create a NEW Stripe
   414→   414→      // Price with the updated amount. The OLD Stripe Price is kept
   415→   415→      // active — existing Stripe subscriptions on it stay on the
   416→   416→      // original price until they renew / prorate / cancel. New
   417→   417→      // checkouts will use the new Price ID (the local plan row is
   418→   418→      // updated by the caller with the resolved ID).
   419→   419→      try {
   420→   420→        const existingPrice = await stripe.prices.retrieve(existing);
   421→   421→        const existingAmount = existingPrice.unit_amount ?? 0;
   422→   422→        if (existingAmount !== expectedAmount) {
   423→   423→          // Price changed → create a new Stripe Price on the same
   424→   424→          // Product. Reuse the existing Price's `product` so we don't
   425→   425→          // create a duplicate Product.
   426→   426→          const productId =
   427→   427→            typeof existingPrice.product === 'string'
   428→   428→              ? existingPrice.product
   429→   429→              : existingPrice.product && 'id' in existingPrice.product
   430→   430→              ? existingPrice.product.id
   431→   431→              : undefined;
   432→   432→          if (!productId) {
   433→   433→            // Can't resolve product → keep the existing Price ID.
   434→   434→            return existing;
   435→   435→          }
   436→   436→          const stripeInterval: 'month' | 'year' = interval === 'monthly' ? 'month' : 'year';
   437→   437→          const newPrice = await stripe.prices.create({
   438→   438→            unit_amount: expectedAmount,
   439→   439→            currency: plan.currency.toLowerCase(),
   440→   440→            recurring: { interval: stripeInterval },
   441→   441→            product: productId,
   442→   442→            metadata: { planId: plan.planId, interval },
   443→   443→          });
   444→   444→          return newPrice.id;
   445→   445→        }
   446→   446→        // Amount matches → keep the existing Stripe Price ID.
   447→   447→        return existing;
   448→   448→      } catch {
   449→   449→        // Retrieval failed (deleted? permission?) → fall through to
   450→   450→        // the lookup-by-metadata + create branch. The admin can use
   451→   451→        // the explicit "Sync to Stripe" route to surface real errors.
   452→   452→      }
   453→   453→    }
   454→   454→
   455→   455→    // Try to find an existing active Stripe Price with matching metadata.
   456→   456→    const list = await stripe.prices.list({ active: true, limit: 100 });
   457→   457→    const match = list.data.find(
   458→   458→      (p) => p.metadata?.planId === plan.planId && p.metadata?.interval === interval,
   459→   459→    );
   460→   460→    if (match) {
   461→   461→      // Verify the matched Price's amount matches the locally-configured
   462→   462→      // amount — if not, create a fresh Price (same logic as above).
   463→   463→      const matchAmount = match.unit_amount ?? 0;
   464→   464→      if (matchAmount === expectedAmount) return match.id;
   465→   465→      // Amount mismatch → create new on the same Product.
   466→   466→      const productId =
   467→   467→        typeof match.product === 'string'
   468→   468→          ? match.product
   469→   469→          : match.product && 'id' in match.product
   470→   470→          ? match.product.id
   471→   471→          : undefined;
   472→   472→      if (productId) {
   473→   473→        const stripeInterval: 'month' | 'year' = interval === 'monthly' ? 'month' : 'year';
   474→   474→        const newPrice = await stripe.prices.create({
   475→   475→          unit_amount: expectedAmount,
   476→   476→          currency: plan.currency.toLowerCase(),
   477→   477→          recurring: { interval: stripeInterval },
   478→   478→          product: productId,
   479→   479→          metadata: { planId: plan.planId, interval },
   480→   480→        });
   481→   481→        return newPrice.id;
   482→   482→      }
   483→   483→    }
   484→   484→
   485→   485→    // Create-or-reuse a Stripe Product tagged with metadata.planId.
   486→   486→    let productId: string | undefined;
   487→   487→    try {
   488→   488→      const existingProducts = await stripe.products.list({ limit: 100 });
   489→   489→      const existingProduct = existingProducts.data.find(
   490→   490→        (p) => p.metadata?.planId === plan.planId,
   491→   491→      );
   492→   492→      if (existingProduct) productId = existingProduct.id;
   493→   493→    } catch {
   494→   494→      // fall through to create
   495→   495→    }
   496→   496→    if (!productId) {
   497→   497→      const product = await stripe.products.create({
   498→   498→        name: plan.name,
   499→   499→        metadata: { planId: plan.planId },
   500→   500→      });
   501→   501→      productId = product.id;
   502→   502→    }
   503→   503→
   504→   504→    // Stripe's recurring.interval enum is 'day' | 'week' | 'month' | 'year'
   505→   505→    // — map our local 'monthly'/'yearly' terms onto it.
   506→   506→    const stripeInterval: 'month' | 'year' = interval === 'monthly' ? 'month' : 'year';
   507→   507→    const price = await stripe.prices.create({
   508→   508→      unit_amount: expectedAmount,
   509→   509→      currency: plan.currency.toLowerCase(),
   510→   510→      recurring: { interval: stripeInterval },
   511→   511→      product: productId,
   512→   512→      metadata: { planId: plan.planId, interval },
   513→   513→    });
   514→   514→    return price.id;
   515→   515→  };
   516→   516→
   517→   517→  const stripePriceIdMonthly = await resolveInterval('monthly');
   518→   518→  const stripePriceIdYearly = await resolveInterval('yearly');
   519→   519→  return { stripePriceIdMonthly, stripePriceIdYearly };
   520→   520→}

   643→   643→/**
   644→   644→ * IDEMPOTENCY LEDGER — claim-for-processing pattern.
   645→   645→ *
   646→   646→ * Stripe retries webhook events up to 3 days. We MUST NOT process the
   647→   647→ * same `evt_...` twice. The flow is:
   648→   648→ *
   649→   649→ *   1. claimWebhookEventForProcessing({ stripeEventId, eventType, ... })
   650→   650→ *      Tries to insert a row with outcome='processing'. Three outcomes:
   651→   651→ *        - No existing row → insert succeeds → returns 'new' (process now).
   652→   652→ *        - Existing row with outcome='processed' → returns 'processed' (skip).
   653→   653→ *        - Existing row with outcome='error' OR stale 'processing'
   654→   654→ *          (older than 10 min — likely a crashed worker) → re-claim it
   655→   655→ *          (update to 'processing') → returns 'retry' (process again).
   656→   656→ *
   657→   657→ *   2. After the handler dispatch:
   658→   658→ *      markWebhookEventOutcome(stripeEventId, 'processed')  // success
   659→   659→ *      markWebhookEventOutcome(stripeEventId, 'error', '...') // failure
   660→   660→ *
   661→   661→ * This pattern (vs the old "insert with outcome='processed' BEFORE
   662→   662→ * dispatch") lets Stripe retries recover from handler failures —
   663→   663→ * previously, a row was marked 'processed' before the handler ran,
   664→   664→ * so a crash left the row 'processed' and the retry was silently
   665→   665→ * dropped, leaving the DB in a partial state.
   666→   666→ *
   667→   667→ * Lazy-imports `@/lib/db` so this file stays safe to import from
   668→   668→ * server-only contexts without forcing Prisma to load eagerly.
   669→   669→ */
   670→   670→const STALE_PROCESSING_MS = 10 * 60_000; // 10 min — a worker that crashed mid-process
   671→   671→
   672→   672→export type WebhookEventClaim = 'new' | 'retry' | 'processed';
   673→   673→
   674→   674→export async function claimWebhookEventForProcessing(params: {
   675→   675→  stripeEventId: string;
   676→   676→  eventType: string;
   677→   677→  apiVersion?: string;
   678→   678→  objectId?: string;
   679→   679→}): Promise<WebhookEventClaim> {
   680→   680→  const { db } = await import('@/lib/db');
   681→   681→  try {
   682→   682→    // Try to insert a fresh 'processing' row. If the unique constraint
   683→   683→    // on stripeEventId rejects the insert, fall through to the existing-row
   684→   684→    // branch.
   685→   685→    await db.webhookEvent.create({
   686→   686→      data: {
   687→   687→        stripeEventId: params.stripeEventId,
   688→   688→        eventType: params.eventType,
   689→   689→        apiVersion: params.apiVersion ?? null,
   690→   690→        objectId: params.objectId ?? null,
   691→   691→        outcome: 'processing',
   692→   692→      },
   693→   693→    });
   694→   694→    return 'new';
   695→   695→  } catch (err: unknown) {
   696→   696→    const e = (err ?? {}) as { code?: string; message?: string };
   697→   697→    if (e.code !== 'P2002' && !e.message?.includes('Unique constraint')) {
   698→   698→      // Any other error → conservative: treat as 'processed' so the
   699→   699→      // webhook skips (request still returns 200 to Stripe).
   700→   700→      return 'processed';
   701→   701→    }
   702→   702→    // Unique constraint violation — a row already exists. Look it up.
   703→   703→    try {
   704→   704→      const existing = await db.webhookEvent.findUnique({
   705→   705→        where: { stripeEventId: params.stripeEventId },
   706→   706→        select: { outcome: true, processedAt: true },
   707→   707→      });
   708→   708→      if (!existing) return 'processed'; // race: row vanished → skip
   709→   709→      if (existing.outcome === 'processed') return 'processed'; // already done → skip
   710→   710→      if (existing.outcome === 'error') {
   711→   711→        // Re-claim → re-process on retry.
   712→   712→        await db.webhookEvent.update({
   713→   713→          where: { stripeEventId: params.stripeEventId },
   714→   714→          data: {
   715→   715→            outcome: 'processing',
   716→   716→            errorMessage: null,
   717→   717→            processedAt: new Date(),
   718→   718→          },
   719→   719→        });
   720→   720→        return 'retry';
   721→   721→      }
   722→   722→      if (existing.outcome === 'processing') {
   723→   723→        // Stale in-progress row (worker crashed mid-process). Re-claim
   724→   724→        // only when stale; otherwise skip to avoid double-processing.
   725→   725→        const age = Date.now() - existing.processedAt.getTime();
   726→   726→        if (age > STALE_PROCESSING_MS) {
   727→   727→          await db.webhookEvent.update({
   728→   728→            where: { stripeEventId: params.stripeEventId },
   729→   729→            data: {
   730→   730→              outcome: 'processing',
   731→   731→              errorMessage: null,
   732→   732→              processedAt: new Date(),
   733→   733→            },
   734→   734→          });
   735→   735→          return 'retry';
   736→   736→        }
   737→   737→        return 'processed'; // still being processed by another worker → skip
   738→   738→      }
   739→   739→      return 'processed';
   740→   740→    } catch {
   741→   741→      // Lookup failed → conservative: skip.
   742→   742→      return 'processed';
   743→   743→    }
   744→   744→  }
   745→   745→}
   746→   746→
   747→   747→/** Update the outcome of a webhook event row after the handler runs.
   748→   748→ *  Best-effort — never throws. Called with 'processed' on success or
   749→   749→ *  'error' on failure (Stripe will retry). */
   750→   750→export async function markWebhookEventOutcome(
   751→   751→  stripeEventId: string,
   752→   752→  outcome: 'processed' | 'error',
   753→   753→  errorMessage?: string | null,
   754→   754→): Promise<void> {
   755→   755→  const { db } = await import('@/lib/db');
   756→   756→  try {
   757→   757→    await db.webhookEvent.update({
   758→   758→      where: { stripeEventId },
   759→   759→      data: {
   760→   760→        outcome,
   761→   761→        errorMessage: errorMessage ? errorMessage.slice(0, 1000) : null,
   762→   762→        processedAt: new Date(),
   763→   763→      },
   764→   764→    });
   765→   765→    if (outcome === 'processed' || outcome === 'error') {
   766→   766→      // Invalidate the resolved-settings cache when a webhook succeeds
   767→   767→      // so subsequent reads pick up any DB changes (e.g., a Payment
   768→   768→      // row created by the webhook). Best-effort.
   769→   769→      invalidateStripeSettingsCache();
   770→   770→    }
   771→   771→  } catch {
   772→   772→    // best-effort — never throw from an outcome update
   773→   773→  }
   774→   774→}
   775→   775→
   776→   776→/**
   777→   777→ * @deprecated Use `claimWebhookEventForProcessing` + `markWebhookEventOutcome`
   778→   778→ *   instead. The new 2-step pattern allows Stripe retries to recover from
   779→   779→ *   handler failures (the old single-step insert marked the row 'processed'
   780→   780→ *   BEFORE dispatch, so retries were silently dropped).
   781→   781→ *
   782→   782→ *   Kept as a backwards-compatible wrapper so callers that haven't migrated
   783→   783→ *   keep working. Calls `claimWebhookEventForProcessing`; returns true only
   784→   784→ *   when the claim is 'new' or 'retry' (i.e. the caller should process the
   785→   785→ *   event). Returns false when the event was already 'processed'.
   786→   786→ */
   787→   787→export async function markWebhookEventProcessed(params: {
   788→   788→  stripeEventId: string;
   789→   789→  eventType: string;
   790→   790→  apiVersion?: string;
   791→   791→  objectId?: string;
   792→   792→  outcome: 'processed' | 'skipped_duplicate' | 'error';
   793→   793→  errorMessage?: string;
   794→   794→}): Promise<boolean> {
   795→   795→  const claim = await claimWebhookEventForProcessing({
   796→   796→    stripeEventId: params.stripeEventId,
   797→   797→    eventType: params.eventType,
   798→   798→    apiVersion: params.apiVersion,
   799→   799→    objectId: params.objectId,
   800→   800→  });
   801→   801→  if (claim === 'processed') return false;
   802→   802→  // 'new' or 'retry' — caller will process. If the caller also passed an
   803→   803→  // outcome, apply it now (for backwards compat with callers that used
   804→   804→  // the old single-step API to record an outcome at the start).
   805→   805→  if (params.outcome && params.outcome !== 'processed') {
   806→   806→    await markWebhookEventOutcome(params.stripeEventId, params.outcome === 'error' ? 'error' : 'processed', params.errorMessage);
   807→   807→  }
   808→   808→  return true;
   809→   809→}

   888→   888→/** Build the safe masked view for the admin UI. Never returns raw

   927→   927→/** Save (encrypt + upsert) Stripe credentials. Empty-string values
   928→   928→ *  for secret keys mean "leave unchanged" (the existing encrypted
   929→   929→ *  value is preserved); null means "clear". Omitted (undefined)
   930→   930→ *  fields also mean "preserve" — this matters because the frontend
   931→   931→ *  may omit a field entirely from the JSON body when the admin didn't
   932→   932→ *  touch it. Publishable keys are written as plaintext (non-secret).
   933→   933→ *  Returns the masked view. Never returns raw secret values.
   934→   934→ *
   935→   935→ *  MASKED-VALUE REJECTION: any non-empty secret/publishable/webhook
   936→   936→ *  value containing the mask bullet character (•) is rejected with a
   937→   937→ *  thrown Error. This catches the common mistake of copy-pasting the
   938→   938→ *  masked display form (e.g. `sk_test_••••abcd`) back into the input
   939→   939→ *  — which would otherwise be encrypted as the masked string and
   940→   940→ *  silently break every subsequent Stripe API call with
   941→   941→ *  "Invalid API Key provided". */
   942→   942→export async function saveStripeSettings(
   943→   943→  input: StripeSettingsInput,
   944→   944→  editorUserId: string,
   945→   945→): Promise<StripeSettingsView> {
   946→   946→  const { db } = await import('@/lib/db');
   947→   947→  const existing = await readStripeSettingsRow();
   948→   948→
   949→   949→  // ---- Masked-value rejection (before any encryption / storage). ----
   950→   950→  // The admin may have copy-pasted the masked form back into the
   951→   951→  // input — reject it with a clear, actionable error.
   952→   952→  const maskedChecks: Array<{ value: string | undefined; label: string }> = [
   953→   953→    { value: input.secretKeyTest, label: 'Secret Key (Test)' },
   954→   954→    { value: input.secretKeyLive, label: 'Secret Key (Live)' },
   955→   955→    { value: input.publishableKeyTest, label: 'Publishable Key (Test)' },
   956→   956→    { value: input.publishableKeyLive, label: 'Publishable Key (Live)' },
   957→   957→    { value: input.webhookSecretTest, label: 'Webhook Secret (Test)' },
   958→   958→    { value: input.webhookSecretLive, label: 'Webhook Secret (Live)' },
   959→   959→  ];
   960→   960→  for (const c of maskedChecks) {
   961→   961→    if (c.value && isMaskedSecretValue(c.value)) {
   962→   962→      throw new Error(
   963→   963→        `${c.label} looks like a masked value (contains the • character). ` +
   964→   964→          `Re-enter the full, real Stripe key — do not copy the masked form.`,
   965→   965→      );
   966→   966→    }
   967→   967→  }
   968→   968→
   969→   969→  // ---- Secret keys: encrypt when provided; preserve when empty ----
   970→   970→  // Semantics (NEW, fixes the "Invalid API Key" bug):
   971→   971→  //   - undefined  → preserve (admin omitted the field entirely)
   972→   972→  //   - null       → clear (admin explicitly wants to remove it)
   973→   973→  //   - ''         → preserve (admin left the input empty — "leave unchanged")
   974→   974→  //   - non-empty  → encrypt + store
   975→   975→  // Previously, undefined was treated as "clear", which silently wiped
   976→   976→  // the saved secret every time the admin saved without retyping the
   977→   977→  // key. That made the next Test Connection fall back to env (often
   978→   978→  // empty or wrong) → Stripe returned "Invalid API Key".
   979→   979→  const encryptOrPreserve = async (
   980→   980→    plaintext: string | undefined | null,
   981→   981→    existingEncrypted: string | null | undefined,
   982→   982→  ): Promise<string | null> => {
   983→   983→    if (plaintext === undefined) return existingEncrypted ?? null; // preserve (omitted)
   984→   984→    if (plaintext === null) return null; // clear
   985→   985→    if (plaintext === '') return existingEncrypted ?? null; // preserve (empty input)
   986→   986→    return await encrypt(plaintext);
   987→   987→  };
   988→   988→
   989→   989→  const secretKeyTestEncrypted = await encryptOrPreserve(
   990→   990→    input.secretKeyTest,
   991→   991→    existing?.secretKeyTestEncrypted,
   992→   992→  );
   993→   993→  const secretKeyLiveEncrypted = await encryptOrPreserve(
   994→   994→    input.secretKeyLive,
   995→   995→    existing?.secretKeyLiveEncrypted,
   996→   996→  );
   997→   997→  const webhookSecretTestEncrypted = await encryptOrPreserve(
   998→   998→    input.webhookSecretTest,
   999→   999→    existing?.webhookSecretTestEncrypted,
  1000→  1000→  );
  1001→  1001→  const webhookSecretLiveEncrypted = await encryptOrPreserve(
  1002→  1002→    input.webhookSecretLive,
  1003→  1003→    existing?.webhookSecretLiveEncrypted,
  1004→  1004→  );
  1005→  1005→
  1006→  1006→  // ---- Publishable keys: plaintext (non-secret). Empty string
  1007→  1007→  //      preserves existing; null clears; undefined (omitted) preserves. ----
  1008→  1008→  const resolvePub = (
  1009→  1009→    v: string | undefined | null,
  1010→  1010→    existingVal: string | null,
  1011→  1011→  ): string | null => {
  1012→  1012→    if (v === undefined) return existingVal ?? null; // preserve (omitted)
  1013→  1013→    if (v === null) return null; // clear
  1014→  1014→    if (v === '') return existingVal ?? null; // preserve (empty input)
  1015→  1015→    return v;
  1016→  1016→  };
  1017→  1017→  const publishableKeyTest = resolvePub(input.publishableKeyTest, existing?.publishableKeyTest ?? null);
  1018→  1018→  const publishableKeyLive = resolvePub(input.publishableKeyLive, existing?.publishableKeyLive ?? null);
  1019→  1019→
  1020→  1020→  // ---- App URL: empty string = unset (no preserve semantic — caller
  1021→  1021→  //      should pass the full URL). ----
  1022→  1022→  const appUrl = input.appUrl ?? existing?.appUrl ?? null;
  1023→  1023→
  1024→  1024→  await db.stripeSettings.upsert({
  1025→  1025→    where: { id: 'singleton' },
  1026→  1026→    create: {
  1027→  1027→      id: 'singleton',
  1028→  1028→      mode: input.mode,
  1029→  1029→      secretKeyTestEncrypted,
  1030→  1030→      secretKeyLiveEncrypted,
  1031→  1031→      webhookSecretTestEncrypted,
  1032→  1032→      webhookSecretLiveEncrypted,
  1033→  1033→      publishableKeyTest,
  1034→  1034→      publishableKeyLive,
  1035→  1035→      appUrl,
  1036→  1036→      updatedBy: editorUserId,
  1037→  1037→    },
  1038→  1038→    update: {
  1039→  1039→      mode: input.mode,
  1040→  1040→      secretKeyTestEncrypted,
  1041→  1041→      secretKeyLiveEncrypted,
  1042→  1042→      webhookSecretTestEncrypted,
  1043→  1043→      webhookSecretLiveEncrypted,
  1044→  1044→      publishableKeyTest,
  1045→  1045→      publishableKeyLive,
  1046→  1046→      appUrl,
  1047→  1047→      updatedBy: editorUserId,
  1048→  1048→    },
  1049→  1049→  });
  1050→  1050→
  1051→  1051→  // Force the next read to pick up the new credentials.
  1052→  1052→  invalidateStripeSettingsCache();
  1053→  1053→
  1054→  1054→  return getStripeSettingsForAdmin();
  1055→  1055→}
  1056→  1056→
  1057→  1057→/** Test a Stripe connection by pinging the read-only /v1/balance
  1058→  1058→ *  endpoint. Uses the currently-configured credentials when
  1059→  1059→ *  `input` is null/undefined; otherwise constructs a temporary
  1060→  1060→ *  client from `input` (so the admin can test BEFORE saving).
  1061→  1061→ *  Records the outcome on the singleton row.
  1062→  1062→ *
  1063→  1063→ *  Returns `{ success: true, accountInfo }` on success, or
  1064→  1064→ *  `{ success: false, code, message }` on failure. Never throws. */
  1065→  1065→export async function testStripeConnection(
  1066→  1066→  input?: StripeSettingsInput,
  1067→  1067→): Promise<
  1068→  1068→  | { success: true; mode: 'test' | 'live'; accountInfo: { id: string; type: string; country: string; email: string | null; displayName: string | null } }
  1069→  1069→  | { success: false; mode: 'test' | 'live'; code: string; message: string }
  1070→  1070→> {
  1071→  1071→  // Resolve the secret key to test with.
  1072→  1072→  let secretKey: string | null = null;
  1073→  1073→  let mode: 'test' | 'live' = 'test';
  1074→  1074→
  1075→  1075→  if (input) {
  1076→  1076→    mode = input.mode;
  1077→  1077→    const candidate = mode === 'live' ? input.secretKeyLive : input.secretKeyTest;
  1078→  1078→    if (candidate && candidate.trim().length > 0) {
  1079→  1079→      secretKey = candidate;
  1080→  1080→    } else {
  1081→  1081→      // Fall back to the stored value for the chosen mode.
  1082→  1082→      const row = await readStripeSettingsRow();
  1083→  1083→      const enc = mode === 'live' ? row?.secretKeyLiveEncrypted : row?.secretKeyTestEncrypted;
  1084→  1084→      secretKey = await tryDecrypt(enc);
  1085→  1085→    }
  1086→  1086→  } else {
  1087→  1087→    const resolved = await getResolvedStripeSettings();
  1088→  1088→    secretKey = resolved.secretKey;
  1089→  1089→    mode = resolved.mode;
  1090→  1090→  }
  1091→  1091→
  1092→  1092→  if (!secretKey || secretKey.trim().length === 0) {
  1093→  1093→    const msg = 'No Stripe secret key is configured for the selected mode. Set the secret key first.';
  1094→  1094→    await recordTestOutcome(mode, 'error', msg);
  1095→  1095→    return { success: false, mode, code: 'NOT_CONFIGURED', message: msg };
  1096→  1096→  }
  1097→  1097→
  1098→  1098→  // Reject masked values outright — the admin copy-pasted the masked
  1099→  1099→  // display form (sk_test_••••abcd) instead of the real key. Stripe
  1100→  1100→  // would reject the masked string as an "Invalid API Key" — better
  1101→  1101→  // to surface the real cause here with a clear, actionable message.
  1102→  1102→  if (isMaskedSecretValue(secretKey)) {
  1103→  1103→    const msg =
  1104→  1104→      'The provided secret key looks like a masked value (contains the • character). ' +
  1105→  1105→      'Re-enter the full, real Stripe secret key — do not copy the masked form.';
  1106→  1106→    await recordTestOutcome(mode, 'error', msg);
  1107→  1107→    return { success: false, mode, code: 'MASKED_VALUE', message: msg };
  1108→  1108→  }
  1109→  1109→
  1110→  1110→  // Basic key-shape validation — sk_test_ / sk_live_ / rk_test_ /
  1111→  1111→  // rk_live_ (restricted keys also work for read-only calls).
  1112→  1112→  const keyShapeOk =
  1113→  1113→    secretKey.startsWith('sk_test_') ||
  1114→  1114→    secretKey.startsWith('sk_live_') ||
  1115→  1115→    secretKey.startsWith('rk_test_') ||
  1116→  1116→    secretKey.startsWith('rk_live_');
  1117→  1117→  if (!keyShapeOk) {
  1118→  1118→    const msg = 'The secret key does not look like a Stripe secret key (expected sk_test_… or sk_live_…).';
  1119→  1119→    await recordTestOutcome(mode, 'error', msg);
  1120→  1120→    return { success: false, mode, code: 'INVALID_KEY_SHAPE', message: msg };
  1121→  1121→  }
  1122→  1122→
  1123→  1123→  // MODE-KEY MISMATCH VALIDATION: verify the key prefix matches the
  1124→  1124→  // selected Test/Live mode. Without this, the admin could select
  1125→  1125→  // Live mode but provide a sk_test_ key — the Stripe client would
  1126→  1126→  // authenticate against the TEST account (since the key is a test
  1127→  1127→  // key), silently bypassing the live-account check the admin meant
  1128→  1128→  // to perform. Stripe would happily return account info for the
  1129→  1129→  // test account, leading the admin to believe the LIVE credentials
  1130→  1130→  // were validated when they weren't.
  1131→  1131→  const isTestKey = secretKey.startsWith('sk_test_') || secretKey.startsWith('rk_test_');
  1132→  1132→  const isLiveKey = secretKey.startsWith('sk_live_') || secretKey.startsWith('rk_live_');
  1133→  1133→  if (mode === 'test' && !isTestKey) {
  1134→  1134→    const msg =
  1135→  1135→      'Mode/key mismatch: Test mode is selected but the secret key is a LIVE key ' +
  1136→  1136→      `(starts with ${secretKey.slice(0, 8)}…). Switch to Live mode or paste a Test key (sk_test_…).`;
  1137→  1137→    await recordTestOutcome(mode, 'error', msg);
  1138→  1138→    return { success: false, mode, code: 'MODE_KEY_MISMATCH', message: msg };
  1139→  1139→  }
  1140→  1140→  if (mode === 'live' && !isLiveKey) {
  1141→  1141→    const msg =
  1142→  1142→      'Mode/key mismatch: Live mode is selected but the secret key is a TEST key ' +
  1143→  1143→      `(starts with ${secretKey.slice(0, 8)}…). Switch to Test mode or paste a Live key (sk_live_…).`;
  1144→  1144→    await recordTestOutcome(mode, 'error', msg);
  1145→  1145→    return { success: false, mode, code: 'MODE_KEY_MISMATCH', message: msg };
  1146→  1146→  }
  1147→  1147→
  1148→  1148→  try {
  1149→  1149→    // Construct a TEMPORARY client (not the cached singleton) so we
  1150→  1150→    // can test credentials that haven't been saved yet.
  1151→  1151→    const tempClient = new Stripe(secretKey, {
  1152→  1152→      apiVersion: '2026-08-26.dahlia' as Stripe.LatestApiVersion,
  1153→  1153→      typescript: true,
  1154→  1154→      maxNetworkRetries: 1,
  1155→  1155→    });
  1156→  1156→    // /v1/balance is read-only and free of side effects — perfect for
  1157→  1157→    // testing credentials. It also works with restricted keys that
  1158→  1158→    // have the "Balance" read permission.
  1159→  1159→    const balance = await tempClient.balance.retrieve();
  1160→  1160→    // Resolve the account for richer info (id, country, display name).
  1161→  1161→    // This call requires the "Settings" read permission, which secret
  1162→  1162→    // keys always have. Fall back gracefully when restricted keys
  1163→  1163→    // can't reach it.
  1164→  1164→    let accountInfo: { id: string; type: string; country: string; email: string | null; displayName: string | null };
  1165→  1165→    try {
  1166→  1166→      const acct = await tempClient.accounts.retrieve();
  1167→  1167→      accountInfo = {
  1168→  1168→        id: acct.id,
  1169→  1169→        type: (acct as unknown as { type?: string }).type ?? 'standard',
  1170→  1170→        country: (acct as unknown as { country?: string }).country ?? '',
  1171→  1171→        email: (acct as unknown as { email?: string | null }).email ?? null,
  1172→  1172→        displayName: (acct as unknown as { display_name?: string | null }).display_name ?? null,
  1173→  1173→      };
  1174→  1174→    } catch {
  1175→  1175→      // Restricted key with no Settings read — still a successful
  1176→  1176→      // test (the secret key authenticates to Stripe).
  1177→  1177→      accountInfo = {
  1178→  1178→        id: '(restricted key)',
  1179→  1179→        type: 'restricted',
  1180→  1180→        country: '',
  1181→  1181→        email: null,
  1182→  1182→        displayName: null,
  1183→  1183→      };
  1184→  1184→    }
  1185→  1185→    const balanceAvailable = balance.available?.[0]?.amount;
  1186→  1186→    await recordTestOutcome(
  1187→  1187→      mode,
  1188→  1188→      'success',
  1189→  1189→      null,
  1190→  1190→      `account_id=${accountInfo.id}, available=${balanceAvailable ?? 'n/a'}`,
  1191→  1191→    );
  1192→  1192→    return { success: true, mode, accountInfo };
  1193→  1193→  } catch (err: unknown) {
  1194→  1194→    const e = (err ?? {}) as { code?: string; message?: string; type?: string };
  1195→  1195→    const code = e.code ?? e.type ?? 'stripe_error';
  1196→  1196→    const message = e.message ?? 'Unknown Stripe error';
  1197→  1197→    await recordTestOutcome(mode, 'error', `${code}: ${message}`);
  1198→  1198→    return { success: false, mode, code, message };
  1199→  1199→  }
  1200→  1200→}

  1206→  1206→  status: 'success' | 'error',
  1207→  1207→  errorMessage: string | null,

  1209→  1209→): Promise<void> {
  1210→  1210→  try {

  1212→  1212→    await db.stripeSettings.upsert({
  1213→  1213→      where: { id: 'singleton' },
  1214→  1214→      create: {
  1215→  1215→        id: 'singleton',
  1216→  1216→        mode,
  1217→  1217→        lastTestStatus: status,
  1218→  1218→        lastTestedAt: new Date(),
  1219→  1219→        lastTestErrorMessage: errorMessage,
  1220→  1220→      },
  1221→  1221→      update: {
  1222→  1222→        mode,
  1223→  1223→        lastTestStatus: status,
  1224→  1224→        lastTestedAt: new Date(),
  1225→  1225→        lastTestErrorMessage: errorMessage,
  1226→  1226→      },
  1227→  1227→    });
  1228→  1228→    invalidateStripeSettingsCache();
  1229→  1229→  } catch {
  1230→  1230→    // best-effort
  1231→  1231→  }
  1232→  1232→}
