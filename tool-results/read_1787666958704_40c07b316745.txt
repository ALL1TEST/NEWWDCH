   105→   105→function formatNumber(n: number | null | undefined): string {
   106→   106→  const num = n ?? 0;
   107→   107→  if (num >= 1000000) return `${(num / 1000000).toFixed(1)}M`;
   108→   108→  if (num >= 1000) return `${(num / 1000).toFixed(1)}K`;
   109→   109→  return num.toLocaleString();
   110→   110→}
   111→   111→
   112→   112→function formatPercent(n: number | null | undefined): string {
   113→   113→  return `${((n ?? 0) * 100).toFixed(2)}%`;
   114→   114→}
   115→   115→
   116→   116→function formatPosition(n: number | null | undefined): string {
   117→   117→  return (n ?? 0).toFixed(1);
   118→   118→}
   119→   119→
   120→   120→function formatDate(dateStr: string | null): string {
   121→   121→  if (!dateStr) return 'Never';
   122→   122→  try {
   123→   123→    return new Date(dateStr).toLocaleString(undefined, {
   124→   124→      month: 'short',
   125→   125→      day: 'numeric',
   126→   126→      year: 'numeric',
   127→   127→      hour: '2-digit',
   128→   128→      minute: '2-digit',
   129→   129→    });
   130→   130→  } catch {
   131→   131→    return 'Unknown';
   132→   132→  }
   133→   133→}
   134→   134→
   135→   135→function formatShortDate(dateStr: string): string {
   136→   136→  try {
   137→   137→    return new Date(dateStr).toLocaleDateString(undefined, {
   138→   138→      month: 'short',

   140→   140→    });
   141→   141→  } catch {
   142→   142→    return dateStr;
   143→   143→  }
   144→   144→}
