     5→// ============================================================

     8→// data from /api/content (articles with scheduledAt or publishedAt)

   172→// -------------------- Data mapping --------------------
   173→
   174→function mapArticles(rows: ArticleRow[] | undefined): CalendarEvent[] {
   175→  if (!rows) return [];
   176→  const events: CalendarEvent[] = [];
   177→  for (const a of rows) {
   178→    const iso = a.scheduledAt ?? a.publishedAt;
   179→    if (!iso) continue;
   180→    const date = parseISO(iso);
   181→    if (Number.isNaN(date.getTime())) continue;
   182→    events.push({
   183→      id: `article-${a.id}`,
   184→      title: a.title,
   185→      type: 'article',
   186→      status: a.status,
   187→      date,
   188→      raw: a,
   189→    });
   190→  }
   191→  return events;
   192→}
   193→
   194→function mapCampaigns(rows: CampaignRow[] | undefined): CalendarEvent[] {
   195→  if (!rows) return [];
   196→  const events: CalendarEvent[] = [];
   197→  for (const c of rows) {
   198→    if (!c.scheduledAt) continue;
   199→    const date = parseISO(c.scheduledAt);
   200→    if (Number.isNaN(date.getTime())) continue;
   201→    events.push({
   202→      id: `campaign-${c.id}`,
   203→      title: c.name,
   204→      type: 'campaign',
   205→      status: c.status,
   206→      date,
   207→      raw: c,
   208→    });
   209→  }
   210→  return events;
   211→}
   212→
   213→// -------------------- Filter logic --------------------
   214→
   215→function eventMatchesFilter(ev: CalendarEvent, filter: FilterKey): boolean {
   216→  switch (filter) {
   217→    case 'all':
   218→      return true;
   219→    case 'articles':
   220→      return ev.type === 'article';
   221→    case 'campaigns':
   222→      return ev.type === 'campaign';
   223→    case 'drafts':
   224→      return ev.status === 'DRAFT';
   225→    case 'scheduled':
   226→      // Campaigns with SCHEDULED status, or articles whose date came from scheduledAt
   227→      if (ev.type === 'campaign') return ev.status === 'SCHEDULED';
   228→      return Boolean(ev.raw?.scheduledAt);
   229→    case 'published':
   230→      return ev.status === 'PUBLISHED' || ev.status === 'SENT';
   231→    case 'cancelled':
   232→      return ev.status === 'CANCELLED' || ev.status === 'ARCHIVED';
   233→    default:
   234→      return true;
   235→  }
   236→}

   239→// Calendar Page

   241→
   242→export function CalendarPage() {
   243→  const navigate = useNavigationStore((s) => s.navigate);
   244→
   245→  const [view, setView] = useState<CalendarView>('month');
   246→  const [referenceDate, setReferenceDate] = useState<Date>(new Date());
   247→  const [filter, setFilter] = useState<FilterKey>('all');
   248→  const [selectedEvent, setSelectedEvent] = useState<CalendarEvent | null>(null);
   249→
   250→  // -------- Data fetching --------
   251→
   252→  const { data: articlesData, isLoading: articlesLoading } = useQuery({
   253→    queryKey: ['calendar', 'articles'],
   254→    queryFn: () =>
   255→      getApi<PaginatedResponse<ArticleRow>>('/api/content', {
   256→        pageSize: 100,
   257→        sort: 'publishedAt',
   258→        order: 'asc',
   259→      }),
   260→    staleTime: 30_000,
   261→  });
   262→
   263→  const { data: campaignsData, isLoading: campaignsLoading } = useQuery({
   264→    queryKey: ['calendar', 'campaigns'],
   265→    queryFn: () =>
   266→      getApi<CampaignRow[]>('/api/campaigns', { pageSize: 100 }),
   267→    staleTime: 30_000,
   268→  });
   269→
   270→  const articles = useMemo(
   271→    () => articlesData?.data ?? [],
   272→    [articlesData],
   273→  );
   274→  const campaigns = useMemo(
   275→    () => (Array.isArray(campaignsData) ? campaignsData : []),
   276→    [campaignsData],
   277→  );
   278→
   279→  const allEvents = useMemo<CalendarEvent[]>(() => {
   280→    return [...mapArticles(articles), ...mapCampaigns(campaigns)].sort(
   281→      (a, b) => a.date.getTime() - b.date.getTime(),
   282→    );
   283→  }, [articles, campaigns]);
   284→
   285→  const filteredEvents = useMemo(
   286→    () => allEvents.filter((ev) => eventMatchesFilter(ev, filter)),
   287→    [allEvents, filter],
   288→  );
   289→
   290→  const isLoading = articlesLoading || campaignsLoading;
   291→
   292→  // -------- Navigation --------
   293→
   294→  const goPrev = useCallback(() => {
   295→    setReferenceDate((d) => {
   296→      if (view === 'month') return addMonths(d, -1);
   297→      if (view === 'week') return addWeeks(d, -1);
   298→      return addDays(d, -1); // day
   299→    });
   300→  }, [view]);
   301→
   302→  const goNext = useCallback(() => {
   303→    setReferenceDate((d) => {
   304→      if (view === 'month') return addMonths(d, 1);
   305→      if (view === 'week') return addWeeks(d, 1);
   306→      return addDays(d, 1); // day
   307→    });
   308→  }, [view]);
   309→
   310→  const goToday = useCallback(() => setReferenceDate(new Date()), []);
   311→
   312→  // -------- Header label --------
   313→
   314→  const periodLabel = useMemo(() => {
   315→    if (view === 'month') return format(referenceDate, 'MMMM yyyy');
   316→    if (view === 'week') {
   317→      const ws = startOfWeek(referenceDate, { weekStartsOn: 0 });
   318→      const we = endOfWeek(referenceDate, { weekStartsOn: 0 });
   319→      if (isSameMonth(ws, we)) {
   320→        return `${format(ws, 'MMM d')} – ${format(we, 'd, yyyy')}`;
   321→      }
   322→      return `${format(ws, 'MMM d')} – ${format(we, 'MMM d, yyyy')}`;
   323→    }
   324→    if (view === 'day') return format(referenceDate, 'EEEE, MMMM d, yyyy');
   325→    return format(referenceDate, 'MMMM yyyy');
   326→  }, [view, referenceDate]);
   327→
   328→  // -------- Schedule Content actions --------
   329→
   330→  const handleNewArticle = useCallback(() => {
   331→    navigate('content', null, 'create');
   332→  }, [navigate]);
   333→
   334→  const handleNewCampaign = useCallback(() => {
   335→    navigate('newsletter', null, 'campaigns');
   336→  }, [navigate]);
   337→
   338→  // -------- Event selection --------
   339→
   340→  const handleSelectEvent = useCallback((ev: CalendarEvent) => {
   341→    setSelectedEvent(ev);
   342→  }, []);
   343→
   344→  const handleCloseModal = useCallback(() => setSelectedEvent(null), []);
   345→
   346→  // -------- Render --------
   347→
   348→  return (
   349→    <div className="space-y-5">
   350→      {/* Header */}
   351→      <CalendarHeader
   352→        periodLabel={periodLabel}
   353→        view={view}
   354→        onPrev={goPrev}
   355→        onNext={goNext}
   356→        onToday={goToday}
   357→        onViewChange={setView}
   358→        onNewArticle={handleNewArticle}
   359→        onNewCampaign={handleNewCampaign}
   360→      />
   361→
   362→      {/* Filter bar */}
   363→      <FilterBar value={filter} onChange={setFilter} counts={allEvents} />
   364→
   365→      {/* Calendar body */}
   366→      <div className="rounded-xl border border-border bg-card shadow-sm">
   367→        {isLoading ? (
   368→          <CalendarSkeleton view={view} />
   369→        ) : filteredEvents.length === 0 ? (
   370→          <EmptyState
   371→            icon={CalendarIcon}
   372→            title="No scheduled content"
   373→            description="Schedule an article or newsletter campaign to see it here."
   374→            className="py-20"
   375→          />
   376→        ) : view === 'month' ? (
   377→          <MonthView
   378→            referenceDate={referenceDate}
   379→            events={filteredEvents}
   380→            onSelectEvent={handleSelectEvent}
   381→          />
   382→        ) : view === 'week' ? (
   383→          <WeekView
   384→            referenceDate={referenceDate}
   385→            events={filteredEvents}
   386→            onSelectEvent={handleSelectEvent}
   387→          />
   388→        ) : view === 'day' ? (
   389→          <DayView
   390→            referenceDate={referenceDate}
   391→            events={filteredEvents}
   392→            onSelectEvent={handleSelectEvent}
   393→          />
   394→        ) : null}
   395→      </div>
   396→
   397→      {/* Event details modal */}
   398→      <EventDetailsModal
   399→        event={selectedEvent}
   400→        onClose={handleCloseModal}
   401→        onNavigate={navigate}
   402→      />
   403→    </div>
   404→  );
   405→}

   608→
   609→interface MonthViewProps {
   610→  referenceDate: Date;
   611→  events: CalendarEvent[];
   612→  onSelectEvent: (ev: CalendarEvent) => void;
   613→}
   614→
   615→function MonthView({ referenceDate, events, onSelectEvent }: MonthViewProps) {
   616→  const monthStart = startOfMonth(referenceDate);
   617→  const monthEnd = endOfMonth(referenceDate);
   618→  const gridStart = startOfWeek(monthStart, { weekStartsOn: 0 });
   619→  const gridEnd = endOfWeek(monthEnd, { weekStartsOn: 0 });
   620→  const days = eachDayOfInterval({ start: gridStart, end: gridEnd });
   621→
   622→  const eventsByDay = useMemo(() => {
   623→    const map = new Map<string, CalendarEvent[]>();
   624→    for (const ev of events) {
   625→      const key = format(ev.date, 'yyyy-MM-dd');
   626→      const arr = map.get(key) ?? [];
   627→      arr.push(ev);
   628→      map.set(key, arr);
   629→    }
   630→    return map;
   631→  }, [events]);
   632→
   633→  return (
   634→    <div className="flex flex-col">
