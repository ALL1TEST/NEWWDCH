     1→// ============================================================
     2→// PLAN CONFIG — DB-backed, cached single source of truth for plans,
     3→// pricing, entitlements and usage limits.
     4→// ============================================================
     5→// The Client Billing API and the Platform Admin both read from (and
     6→// the admin mutates) this same service. `platform-data.ts` delegates
     7→// its `PLANS` / `getPlan` / `monthlyPrice` to the synchronous cache
     8→// exposed here, so an owner edit (savePlanConfig) propagates to the
     9→// client billing experience and to MRR on the next read.
    10→//
    11→// MULTI-CURRENCY: the plan carries ONE base price configuration —
    12→// priceMonthly / priceYearly denominated in the plan's default /
    13→// fallback `currency` (both edited directly in the Edit Plan modal).
    14→// The customer's currency is resolved SERVER-SIDE from their IP
    15→// geolocation (see country-pricing.ts:resolveCustomerPricing) when
    16→// the plan's `autoCurrency` flag is on; otherwise every customer is
    17→// billed in the plan default currency.
    18→//
    19→// pricesByCurrency is a DERIVED store (NOT edited per-plan in the
    20→// modal): the default currency's entry mirrors the base price on
    21→// every save, and the other entries hold the platform-level
    22→// regional prices (CountryPricing.regionalPrices + legacy entries)
    23→// — preserved across saves so customers in other currencies keep
    24→// seeing their configured price (e.g. "90 MAD / month").
    25→//
    26→// STRIPE: for each paid plan × supported currency × interval, a real
    27→// Stripe Price is created on sync and its ID persisted in
    28→// stripePriceIdsByCurrency. The default currency's IDs are mirrored
    29→// in stripePriceIdMonthly/Yearly for legacy callers. Checkout always
    30→// resolves the per-currency Stripe Price ID server-side — the
    31→// customer's currency determines which Price is charged.
    32→//
    33→// The cache is hydrated from the PlanConfig table on startup and
    34→// after every mutation. If the table is empty on first hydrate, the
    35→// DEFAULT configs are seeded.
    36→// ============================================================

    52→export type BillingInterval = 'monthly' | 'yearly';
    53→
    54→/** The billing periods enabled on a plan (at least one, enforced).
    55→ *  A disabled period: not shown on the Client Billing page, rejected
    56→ *  at checkout (400 BILLING_PERIOD_NOT_ENABLED), and never gets a
    57→ *  Stripe Price created or selected. */
    58→export function enabledIntervalsOf(plan: {
    59→  billingMonthly: boolean;
    60→  billingYearly: boolean;
    61→}): BillingInterval[] {
    62→  const out: BillingInterval[] = [];
    63→  if (plan.billingMonthly) out.push('monthly');
    64→  if (plan.billingYearly) out.push('yearly');
    65→  return out;
    66→}
    67→
    68→/** Clamp a requested default cadence to an ENABLED period:
    69→ *  only monthly on → 'monthly'; only yearly on → 'yearly'; both on
    70→ *  → the requested value. (Neither enabled is rejected upstream —
    71→ *  here it falls back to the requested value.) */
    72→function normalizeInterval(
    73→  requested: BillingInterval,
    74→  billingMonthly: boolean,
    75→  billingYearly: boolean,
    76→): BillingInterval {
    77→  if (billingMonthly && !billingYearly) return 'monthly';
    78→  if (!billingMonthly && billingYearly) return 'yearly';
    79→  return requested;
    80→}

   101→export interface PlanConfigData {
   102→  planId: string;
   103→  name: string;
   104→  /** Base monthly price (in the plan default currency) — edited in the
   105→   *  Edit Plan modal. Authoritative for the default currency. */
   106→  priceMonthly: number;
   107→  /** Base yearly price (in the plan default currency). */
   108→  priceYearly: number;
   109→  /** Plan DEFAULT / FALLBACK currency (admin-selected via the
   110→   *  country/currency selector in the Edit Plan modal). */
   111→  currency: string;
   112→  /** When true the customer's currency is auto-detected server-side from
   113→   *  their IP (country → currency → supported check → fallback to
   114→   *  `currency`). When false every customer is billed in `currency`. */
   115→  autoCurrency: boolean;
   116→  /** Derived per-currency price map — default entry mirrors the base
   117→   *  price; other entries are platform regional prices (preserved). */
   118→  pricesByCurrency: PricesByCurrency;
   119→  /** Per-currency Stripe Price IDs (maintained by the Stripe sync). */
   120→  stripePriceIdsByCurrency: StripePriceIdsByCurrency;
   121→  /** ENABLED BILLING PERIODS — which checkout options exist for this
   122→   *  plan. At least one must be true (enforced on create/update).
   123→   *  Controls the Client Billing page, checkout validation, and the
   124→   *  Stripe sync (a disabled period never creates/selects a Price). */
   125→  billingMonthly: boolean;
   126→  billingYearly: boolean;
   127→  /** DERIVED default cadence — always an ENABLED period: 'monthly'
   128→   *  when only monthly is enabled, 'yearly' when only yearly is
   129→   *  enabled, and the admin's pick when both are enabled. Clients
   130→   *  with both periods can switch between them. */
   131→  interval: BillingInterval;
   132→  isFree: boolean;
   133→  /** null = unlimited free access; positive N = trial duration in days for free plans. */
   134→  freePlanDurationDays: number | null;
   135→  /** Snapshot Stripe Price IDs for the platform DEFAULT currency.
   136→   *  Authoritative IDs live in `stripePriceIdsByCurrency`. */
   137→  stripePriceIdMonthly: string | null;
   138→  stripePriceIdYearly: string | null;
   139→  active: boolean;
   140→  /** Marketing copy — auto-derived from entitlements on the client side.
   141→   *  Kept on the model for backward-compat but NOT a separate config. */
   142→  features: string[];
   143→  /** Authoritative feature keys granted by this plan (checked by hasFeature). */
   144→  entitlements: string[];
   145→  limits: PlanLimits;
   146→  badgeVariant: string;
   147→  sortOrder: number;
   148→}

   292→/** Resolve the platform's default billing currency via a LAZY import of
   293→ *  country-pricing (which itself imports `getPlanConfigSync` from here).
   294→ *  The lazy `await import()` breaks the static circular dependency —
   295→ *  this module never statically imports country-pricing, so both load.
   296→ *  Falls back to 'CHF' (the schema default) if anything fails. */
   297→async function resolvePlatformDefaultCurrency(): Promise<string> {
   298→  try {
   299→    const { getPlatformDefaultCurrency } = await import('./country-pricing');
   300→    return await getPlatformDefaultCurrency();
   301→  } catch {
   302→    return 'CHF';
   303→  }
   304→}

   316→function notify() {
   317→  for (const cb of _subscribers) {
   318→    try {
   319→      cb();
   320→    } catch {
   321→      // ignore subscriber errors
   322→    }
   323→  }
   324→}
   325→
   326→/** Register a callback fired after every cache refresh (hydrate / mutation). */
   327→export function subscribe(cb: () => void): () => void {
   328→  _subscribers.add(cb);
   329→  return () => _subscribers.delete(cb);
   330→}

   388→function rowToData(row: PlanConfigRow): PlanConfigData {
   389→  let features: string[] = [];
   390→  let entitlements: string[] = [];
   391→  try {
   392→    features = JSON.parse(row.features || '[]');
   393→  } catch {
   394→    features = [];
   395→  }
   396→  try {
   397→    entitlements = JSON.parse(row.entitlements || '[]');
   398→  } catch {
   399→    entitlements = [];
   400→  }
   401→  // Explicitly pick only the known limit fields. Older DB rows may still
   402→  // carry stale `aiWords` / `aiArticles` / `automationRuns` keys in the
   403→  // limits JSON (those were removed because no real enforcement exists).
   404→  // Picking only the known fields prevents the stale keys from leaking
   405→  // back into the cache and being re-serialized on the next save.
   406→  let limits: PlanLimits = { maxSites: 0, storageBytes: 0 };
   407→  try {
   408→    const parsed = JSON.parse(row.limits || '{}') as Partial<PlanLimits>;
   409→    limits = {
   410→      maxSites: typeof parsed.maxSites === 'number' ? parsed.maxSites : 0,
   411→      storageBytes: typeof parsed.storageBytes === 'number' ? parsed.storageBytes : 0,
   412→    };
   413→  } catch {
   414→    // keep defaults
   415→  }
   416→  let pricesByCurrency: PricesByCurrency = {};
   417→  try {
   418→    const parsed = JSON.parse(row.pricesByCurrency || '{}') as Partial<PricesByCurrency>;
   419→    if (parsed && typeof parsed === 'object') {
   420→      for (const [cur, val] of Object.entries(parsed)) {
   421→        if (val && typeof val === 'object') {
   422→          const m = typeof (val as CurrencyPrice).monthly === 'number' ? (val as CurrencyPrice).monthly : 0;
   423→          const y = typeof (val as CurrencyPrice).yearly === 'number' ? (val as CurrencyPrice).yearly : 0;
   424→          pricesByCurrency[cur.toUpperCase()] = { monthly: m, yearly: y };
   425→        }
   426→      }
   427→    }
   428→  } catch {
   429→    pricesByCurrency = {};
   430→  }
   431→  let stripePriceIdsByCurrency: StripePriceIdsByCurrency = {};
   432→  try {
   433→    const parsed = JSON.parse(row.stripePriceIdsByCurrency || '{}') as Partial<StripePriceIdsByCurrency>;
   434→    if (parsed && typeof parsed === 'object') {
   435→      for (const [cur, val] of Object.entries(parsed)) {
   436→        if (val && typeof val === 'object') {
   437→          const m = typeof (val as CurrencyStripeIds).monthly === 'string' ? (val as CurrencyStripeIds).monthly : null;
   438→          const y = typeof (val as CurrencyStripeIds).yearly === 'string' ? (val as CurrencyStripeIds).yearly : null;
   439→          stripePriceIdsByCurrency[cur.toUpperCase()] = { monthly: m, yearly: y };
   440→        }
   441→      }
   442→    }
   443→  } catch {
   444→    stripePriceIdsByCurrency = {};
   445→  }
   446→  return {
   447→    planId: row.planId,
   448→    name: row.name,
   449→    priceMonthly: row.priceMonthly,
   450→    priceYearly: row.priceYearly,
   451→    currency: row.currency,
   452→    autoCurrency: row.autoCurrency ?? true,
   453→    pricesByCurrency,
   454→    stripePriceIdsByCurrency,
   455→    billingMonthly: row.billingMonthly ?? true,
   456→    billingYearly: row.billingYearly ?? true,
   457→    // Normalize the default cadence to an ENABLED period (defensive:
   458→    // legacy rows / stale values never advertise a disabled period).
   459→    interval: normalizeInterval(
   460→      (row.interval === 'yearly' ? 'yearly' : 'monthly') as BillingInterval,
   461→      row.billingMonthly ?? true,
   462→      row.billingYearly ?? true,
   463→    ),
   464→    isFree: row.isFree,
   465→    freePlanDurationDays: row.freePlanDurationDays,
   466→    stripePriceIdMonthly: row.stripePriceIdMonthly,
   467→    stripePriceIdYearly: row.stripePriceIdYearly,
   468→    active: row.active,
   469→    features,
   470→    entitlements,
   471→    limits,
   472→    badgeVariant: row.badgeVariant,
   473→    sortOrder: row.sortOrder,
   474→  };
   475→}

   503→/**
   504→ * Migrate the legacy catalog to the new Free/Plus/Pro/Max shape. Idempotent.
   505→ *
   506→ * Legacy catalog (created by older bootstrap):
   507→ *   beta (free, active=false), pro, max, enterprise (stub)
   508→ *
   509→ * Migration steps (each guarded, can be re-run safely):
   510→ *  1. If `enterprise` exists AND no active subscription references it → delete it.
   511→ *     (Active subs that reference enterprise are preserved — the migration will
   512→ *      not break them; the plan row stays if needed. In practice, the demo
   513→ *      dataset has no DB subscriptions yet, so this is a clean delete.)
   514→ *  2. If `beta` exists → rename to `plus` (planId, name, badgeVariant) and
   515→ *     re-enable it (legacy `beta` was active=false in the demo DB). Also
   516→ *     shift sortOrder to 1 (Free takes 0).
   517→ *  3. If `free` does NOT exist → insert the Free plan at sortOrder 0.
   518→ *  4. Renumber sortOrder to match the canonical order: free=0, plus=1, pro=2, max=3.
   519→ *
   520→ * Returns a short report string for the bootstrap log.
   521→ */
   522→export async function migrateLegacyPlans(): Promise<string> {
   523→  const notes: string[] = [];
   524→
   525→  try {
   526→    // Step 1: delete Enterprise if safe
   527→    const enterprise = await db.planConfig.findUnique({ where: { planId: 'enterprise' } });
   528→    if (enterprise) {
   529→      const subCount = await db.subscription.count({ where: { planId: 'enterprise' } });
   530→      if (subCount === 0) {
   531→        await db.planConfig.delete({ where: { planId: 'enterprise' } });
   532→        notes.push('deleted enterprise plan (0 active subscriptions)');
   533→      } else {
   534→        notes.push(`kept enterprise plan (${subCount} active subscriptions reference it)`);
   535→      }
   536→    }
   537→
   538→    // Step 2: rename beta → plus (and refresh to a real paid tier)
   539→    const beta = await db.planConfig.findUnique({ where: { planId: 'beta' } });
   540→    if (beta) {
   541→      // Don't overwrite an existing plus row — if both exist, prefer the
   542→      // plus row and just delete the beta stub.
   543→      const plus = await db.planConfig.findUnique({ where: { planId: 'plus' } });
   544→      if (!plus) {
   545→        // Rename beta → plus AND refresh the data to the real DEFAULT
   546→        // Plus tier (paid plan with real entitlements + limits, not the
   547→        // legacy free-tier Beta stub). The old Beta was active=false and
   548→        // had no real customers, so this is a safe refresh.
   549→        const plusDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'plus')!;
   550→        await db.planConfig.update({
   551→          where: { planId: 'beta' },
   552→          data: {
   553→            ...dataToRow(plusDefault),
   554→            // Keep the original createdAt for continuity.
   555→            // (updatedAt is auto.)
   556→          },
   557→        });
   558→        notes.push('renamed + refreshed beta → plus (real paid tier)');
   559→      } else {
   560→        // Both beta and plus exist — delete the beta stub (migrate any
   561→        // legacy subscriptions on beta to plus first).
   562→        const betaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   563→        if (betaSubs > 0) {
   564→          await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   565→          notes.push(`migrated ${betaSubs} beta subscription(s) → plus`);
   566→        }
   567→        await db.planConfig.delete({ where: { planId: 'beta' } });
   568→        notes.push('deleted legacy beta stub (plus already exists)');
   569→      }
   570→    }
   571→
   572→    // Step 3: insert Free if missing
   573→    const free = await db.planConfig.findUnique({ where: { planId: 'free' } });
   574→    if (!free) {
   575→      const freeDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'free')!;
   576→      await db.planConfig.create({ data: dataToRow(freeDefault) });
   577→      notes.push('inserted free plan');
   578→    } else {
   579→      // Ensure free plan is properly flagged
   580→      if (!free.isFree || free.badgeVariant !== 'free' || free.sortOrder !== 0) {
   581→        await db.planConfig.update({
   582→          where: { planId: 'free' },
   583→          data: { isFree: true, badgeVariant: 'free', sortOrder: 0 },
   584→        });
   585→        notes.push('normalized free plan flags');
   586→      }
   587→    }
   588→
   589→    // Step 4: normalize sortOrder for all canonical plans
   590→    const canonicalOrder: Record<string, number> = { free: 0, plus: 1, pro: 2, max: 3 };
   591→    for (const [pid, order] of Object.entries(canonicalOrder)) {
   592→      const row = await db.planConfig.findUnique({ where: { planId: pid } });
   593→      if (row && row.sortOrder !== order) {
   594→        await db.planConfig.update({ where: { planId: pid }, data: { sortOrder: order } });
   595→      }
   596→    }
   597→
   598→    // Also migrate legacy subscriptions on 'beta' or 'enterprise' planId
   599→    // to the new ids (beta → plus, enterprise → max) so historical subs
   600→    // don't reference a missing plan.
   601→    const legacyBetaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   602→    if (legacyBetaSubs > 0) {
   603→      await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   604→      notes.push(`migrated ${legacyBetaSubs} legacy beta subscription(s) → plus`);
   605→    }
   606→    const legacyEnterpriseSubs = await db.subscription.count({ where: { planId: 'enterprise' } });
   607→    if (legacyEnterpriseSubs > 0) {
   608→      await db.subscription.updateMany({ where: { planId: 'enterprise' }, data: { planId: 'max' } });
   609→      notes.push(`migrated ${legacyEnterpriseSubs} legacy enterprise subscription(s) → max`);
   610→    }
   611→  } catch (err) {
   612→    notes.push(`migration error (non-fatal): ${(err as Error).message}`);
   613→  }
   614→
   615→  return notes.length > 0 ? notes.join('; ') : 'no migration needed';
   616→}

   618→/** Load all plan configs from the DB into the cache. Self-seeds the
   619→ *  DEFAULT_PLAN_CONFIGS when the table is empty. Runs the legacy
   620→ *  migration when the catalog still contains 'beta' or 'enterprise'.
   621→ *  Safe to call repeatedly. */
   622→export async function hydrate(): Promise<void> {
   623→  if (_hydrating) return _hydrating;
   624→  _hydrating = (async () => {
   625→    try {
   626→      const rows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   627→      if (rows.length === 0) {
   628→        // Self-seed so the platform is never without plans.
   629→        await db.planConfig.createMany({ data: DEFAULT_PLAN_CONFIGS.map(dataToRow) });
   630→        _cache = DEFAULT_PLAN_CONFIGS.map((p) => ({
   631→          ...p,
   632→          features: [...p.features],
   633→          entitlements: [...p.entitlements],
   634→          limits: { ...p.limits },
   635→        }));
   636→      } else {
   637→        // Check for legacy catalog (beta / enterprise) and migrate.
   638→        const hasLegacy =
   639→          rows.some((r) => r.planId === 'beta' || r.planId === 'enterprise') ||
   640→          !rows.some((r) => r.planId === 'free');
   641→        if (hasLegacy) {
   642→          await migrateLegacyPlans();
   643→          const migratedRows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   644→          _cache = migratedRows.map(rowToData);
   645→        } else {
   646→          _cache = rows.map(rowToData);
   647→        }
   648→      }
   649→      _hydrated = true;
   650→      notify();
   651→    } catch (err) {
   652→      // DB unavailable — keep defaults (still functional, just not persisted).
   653→      _hydrated = true;
   654→      notify();
   655→    } finally {
   656→      _hydrating = null;
   657→    }
   658→  })();
   659→  return _hydrating;
   660→}

   701→/** Merge a patch (Partial<PlanConfigData>) into the current data.
   702→ *  The Edit Plan modal sends the ONE base price configuration
   703→ *  (priceMonthly + priceYearly + currency + autoCurrency) — NOT a
   704→ *  per-currency matrix. Derivation rules:
   705→ *  - pricesByCurrency: starts from the current map (or the explicit
   706→ *    patch map when provided); whenever the base price / currency is
   707→ *    in the patch, the DEFAULT currency's entry is re-derived as
   708→ *    { monthly: priceMonthly, yearly: priceYearly }. Other currency
   709→ *    entries (platform regional / legacy prices) are PRESERVED.
   710→ *  - stripePriceIdsByCurrency: kept as-is when omitted (the Stripe
   711→ *    sync maintains it); the default-currency snapshots mirror the
   712→ *    default entry.
   713→ */
   714→function mergePlanPatch(current: PlanConfigData, patch: PlanConfigInput, defaultCurrency: string): PlanConfigData {
   715→  const curUpper = defaultCurrency.toUpperCase();
   716→
   717→  // pricesByCurrency — clone (never mutate the cache's object in place).
   718→  const sourcePrices = patch.pricesByCurrency ?? current.pricesByCurrency ?? {};
   719→  const pricesByCurrency: PricesByCurrency = { ...sourcePrices };
   720→
   721→  // Base price precedence:
   722→  //   1. Explicit patch values (the modal always sends these).
   723→  //   2. Legacy mirror: an explicit pricesByCurrency map with an entry
   724→  //      for the final default currency (backward-compat callers).
   725→  //   3. Current snapshot.
   726→  const mirrored =
   727→    patch.priceMonthly === undefined && patch.pricesByCurrency !== undefined
   728→      ? pricesByCurrency[curUpper]
   729→      : undefined;
   730→  const priceMonthly =
   731→    patch.priceMonthly !== undefined ? patch.priceMonthly : mirrored?.monthly ?? current.priceMonthly;
   732→  const priceYearly =
   733→    patch.priceYearly !== undefined ? patch.priceYearly : mirrored?.yearly ?? current.priceYearly;
   734→
   735→  // The base price configuration was touched (or a legacy map was sent)
   736→  // → re-derive the DEFAULT currency's entry from the base price. All
   737→  // OTHER currency entries (platform regional / legacy prices) are
   738→  // preserved untouched.
   739→  const baseTouched =
   740→    patch.priceMonthly !== undefined ||
   741→    patch.priceYearly !== undefined ||
   742→    patch.currency !== undefined ||
   743→    patch.pricesByCurrency !== undefined;
   744→  if (baseTouched) {
   745→    pricesByCurrency[curUpper] = { monthly: priceMonthly, yearly: priceYearly };
   746→  }
   747→
   748→  const stripePriceIdsByCurrency: StripePriceIdsByCurrency = {
   749→    ...(patch.stripePriceIdsByCurrency ?? current.stripePriceIdsByCurrency ?? {}),
   750→  };
   751→  // Default-currency snapshot — keep the legacy fields in sync so legacy
   752→  // callers that read stripePriceIdMonthly/Yearly see the right values.
   753→  const defIds = stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
   754→  const stripePriceIdMonthly =
   755→    patch.stripePriceIdMonthly !== undefined
   756→      ? patch.stripePriceIdMonthly
   757→      : patch.stripePriceIdsByCurrency !== undefined
   758→        ? defIds.monthly
   759→        : current.stripePriceIdMonthly;
   760→  const stripePriceIdYearly =
   761→    patch.stripePriceIdYearly !== undefined
   762→      ? patch.stripePriceIdYearly
   763→      : patch.stripePriceIdsByCurrency !== undefined
   764→        ? defIds.yearly
   765→        : current.stripePriceIdYearly;
   766→  const isFree = patch.isFree ?? current.isFree;
   767→  // ---- Billing periods ----
   768→  const billingMonthly = patch.billingMonthly ?? current.billingMonthly;
   769→  const billingYearly = patch.billingYearly ?? current.billingYearly;
   770→  // The default cadence must be an ENABLED period (single-period plans
   771→  // are pinned to their only period; both-period plans keep the
   772→  // admin's requested / current cadence).
   773→  const interval = normalizeInterval(
   774→    patch.interval ?? current.interval,
   775→    billingMonthly,
   776→    billingYearly,
   777→  );
   778→  return {
   779→    planId: current.planId,
   780→    name: patch.name ?? current.name,
   781→    priceMonthly,
   782→    priceYearly,
   783→    currency: curUpper,
   784→    autoCurrency: patch.autoCurrency ?? current.autoCurrency,
   785→    pricesByCurrency,
   786→    stripePriceIdsByCurrency,
   787→    billingMonthly,
   788→    billingYearly,
   789→    interval,
   790→    isFree,
   791→    freePlanDurationDays: patch.freePlanDurationDays ?? current.freePlanDurationDays,
   792→    stripePriceIdMonthly,
   793→    stripePriceIdYearly,
   794→    active: patch.active ?? current.active,
   795→    features: patch.features ?? current.features,
   796→    entitlements: patch.entitlements ?? current.entitlements,
   797→    limits: { ...current.limits, ...(patch.limits ?? {}) },
   798→    badgeVariant: patch.badgeVariant ?? current.badgeVariant,
   799→    sortOrder: patch.sortOrder ?? current.sortOrder,
   800→  };
   801→}

   803→/** Update an existing plan config. Writes DB, refreshes cache.
   804→ *
   805→ *  AUTO-SYNC TO STRIPE: when Stripe is configured AND the plan is
   806→ *  PAID (any currency has a positive price) AND the admin did NOT
   807→ *  explicitly set the Stripe Price IDs in this patch, the backend calls
   808→ *  syncPlanToStripe (multi-currency) to ensure a Stripe Product +
   809→ *  per-currency monthly+yearly Prices exist and writes the resolved IDs
   810→ *  into stripePriceIdsByCurrency + the default-currency snapshot fields.
   811→ */
   812→export async function savePlanConfig(planId: string, patch: PlanConfigInput): Promise<PlanConfigData | null> {
   813→  const existing = await db.planConfig.findUnique({ where: { planId } });
   814→  if (!existing) return null;
   815→  const current = rowToData(existing);
   816→
   817→  // When the admin omitted currency, resolve the platform default so the
   818→  // legacy priceMonthly/priceYearly snapshot stays in the right currency.
   819→  if (!patch.currency && !current.currency) {
   820→    patch = { ...patch, currency: await resolvePlatformDefaultCurrency() };
   821→  }
   822→  const finalCurrency = (patch.currency ?? current.currency).toUpperCase();
   823→
   824→  // Distinguish "field omitted" from "field set". When the admin omits
   825→  // both stripePriceId* AND stripePriceIdsByCurrency, they want the
   826→  // backend to auto-sync. When either is explicitly provided, manual.
   827→  const adminTouchedStripePriceIds =
   828→    'stripePriceIdMonthly' in patch ||
   829→    'stripePriceIdYearly' in patch ||
   830→    'stripePriceIdsByCurrency' in patch;
   831→
   832→  const next = mergePlanPatch(current, patch, finalCurrency);
   833→  await db.planConfig.update({ where: { planId }, data: dataToRow(next) });
   834→  await hydrate();
   835→
   836→  // ---- AUTO-SYNC TO STRIPE (multi-currency) ----
   837→  // Best-effort — never throws. Only ENABLED billing periods get
   838→  // Stripe Prices (a disabled period never creates one).
   839→  const hasAnyPaidCurrency = Object.values(next.pricesByCurrency).some(
   840→    (p) =>
   841→      (next.billingMonthly && p.monthly > 0) ||
   842→      (next.billingYearly && p.yearly > 0),
   843→  );
   844→  if (!adminTouchedStripePriceIds && !next.isFree && hasAnyPaidCurrency) {
   845→    try {
   846→      const { isStripeConfiguredAsync, getStripeClient, syncPlanToStripeMulti } = await import('@/lib/stripe');
   847→      if (await isStripeConfiguredAsync()) {
   848→        const stripe = await getStripeClient();
   849→        const syncResult = await syncPlanToStripeMulti(stripe, {
   850→          planId: next.planId,
   851→          name: next.name,
   852→          defaultCurrency: finalCurrency,
   853→          pricesByCurrency: next.pricesByCurrency,
   854→          stripePriceIdsByCurrency: next.stripePriceIdsByCurrency,
   855→          // Only the ENABLED billing periods get Stripe Prices.
   856→          enabledIntervals: enabledIntervalsOf(next),
   857→        });
   858→        // Persist the resolved Stripe Price IDs back onto the row when they changed.
   859→        const snapshot = syncResult.stripePriceIdsByCurrency[finalCurrency] ?? { monthly: null, yearly: null };
   860→        const changed =
   861→          JSON.stringify(syncResult.stripePriceIdsByCurrency) !==
   862→            JSON.stringify(next.stripePriceIdsByCurrency) ||
   863→          snapshot.monthly !== next.stripePriceIdMonthly ||
   864→          snapshot.yearly !== next.stripePriceIdYearly;
   865→        if (changed) {
   866→          await db.planConfig.update({
   867→            where: { planId },
   868→            data: {
   869→              stripePriceIdsByCurrency: JSON.stringify(syncResult.stripePriceIdsByCurrency),
   870→              stripePriceIdMonthly: snapshot.monthly,
   871→              stripePriceIdYearly: snapshot.yearly,
   872→            },
   873→          });
   874→          await hydrate();
   875→        }
   876→      }
   877→    } catch {
   878→      // best-effort — never block the save on a Stripe error
   879→    }
   880→  }
   881→
   882→  return getPlanConfigSync(planId);
   883→}

   885→/** Create a new plan config. AUTO-SYNC TO STRIPE: when Stripe is
   886→ *  configured AND the new plan is PAID AND the admin did NOT
   887→ *  explicitly provide Stripe Price IDs in the input, the backend
   888→ *  calls syncPlanToStripeMulti (multi-currency) to create a Stripe
   889→ *  Product + per-currency monthly+yearly Prices and writes the
   890→ *  resolved IDs back onto the new plan row.
   891→ *
   892→ *  The Create Plan modal sends the ONE base price configuration
   893→ *  (priceMonthly + priceYearly + currency + autoCurrency); the default

   931→    stripePriceIdYearly: input.stripePriceIdYearly !== undefined ? input.stripePriceIdYearly : defIds.yearly,

   937→      storageBytes: 0,
   938→      ...(input.limits ?? {}),
   939→    },
   940→    badgeVariant: input.badgeVariant ?? input.planId,

   943→  await db.planConfig.create({ data: dataToRow(data) });

   948→  // the Stripe side fails, the admin can use the explicit "Sync to
   949→  // Stripe" route to surface errors.
   950→  const adminTouchedStripePriceIds =
   951→    'stripePriceIdMonthly' in input ||
   952→    'stripePriceIdYearly' in input ||
   953→    'stripePriceIdsByCurrency' in input;
   954→  const hasAnyPaidCurrency = Object.values(data.pricesByCurrency).some(
   955→    (p) =>
   956→      (data.billingMonthly && p.monthly > 0) ||

   960→    try {
   961→      const { isStripeConfiguredAsync, getStripeClient, syncPlanToStripeMulti } = await import('@/lib/stripe');
   962→      if (await isStripeConfiguredAsync()) {
   963→        const stripe = await getStripeClient();
   964→        const syncResult = await syncPlanToStripeMulti(stripe, {
   965→          planId: data.planId,
   966→          name: data.name,
   967→          defaultCurrency: curUpper,
   968→          pricesByCurrency: data.pricesByCurrency,
   969→          stripePriceIdsByCurrency: data.stripePriceIdsByCurrency,
   970→          // Only the ENABLED billing periods get Stripe Prices.
   971→          enabledIntervals: enabledIntervalsOf(data),
   972→        });
   973→        const snapshot = syncResult.stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
   974→        const changed =
   975→          JSON.stringify(syncResult.stripePriceIdsByCurrency) !==
   976→            JSON.stringify(data.stripePriceIdsByCurrency) ||
   977→          snapshot.monthly !== data.stripePriceIdMonthly ||
   978→          snapshot.yearly !== data.stripePriceIdYearly;
   979→        if (changed) {
   980→          await db.planConfig.update({
   981→            where: { planId: data.planId },
   982→            data: {
   983→              stripePriceIdsByCurrency: JSON.stringify(syncResult.stripePriceIdsByCurrency),
   984→              stripePriceIdMonthly: snapshot.monthly,
   985→              stripePriceIdYearly: snapshot.yearly,
   986→            },
   987→          });
   988→          await hydrate();
   989→          data.stripePriceIdsByCurrency = syncResult.stripePriceIdsByCurrency;
   990→          data.stripePriceIdMonthly = snapshot.monthly;
   991→          data.stripePriceIdYearly = snapshot.yearly;
   992→        }
   993→      }
   994→    } catch {
   995→      // best-effort — never block the create on a Stripe error
   996→    }
   997→  }
   998→
   999→  return data;

  1002→/**
  1003→ * Delete a plan config. Refuses if:
  1004→ *  - it's the last remaining plan, OR
  1005→ *  - any Subscription still references this planId (would orphan the sub).
  1006→ *
  1007→ * Returns `{ ok: boolean, reason?: string }` describing the outcome.
  1008→ */
  1009→export async function deletePlanConfig(
  1010→  planId: string,
  1011→): Promise<{ ok: boolean; reason?: string }> {
  1012→  const count = await db.planConfig.count();
  1013→  if (count <= 1) return { ok: false, reason: 'Cannot delete the last plan.' };
  1014→  const subCount = await db.subscription.count({ where: { planId } });
  1015→  if (subCount > 0) {
  1016→    return {
  1017→      ok: false,
  1018→      reason: `Cannot delete plan "${planId}" — ${subCount} active subscription(s) reference it. Migrate them first.`,
  1019→    };
  1020→  }
  1021→  try {
  1022→    await db.planConfig.delete({ where: { planId } });
  1023→    await hydrate();
  1024→    return { ok: true };
  1025→  } catch (err) {
  1026→    return { ok: false, reason: (err as Error).message };
  1027→  }
