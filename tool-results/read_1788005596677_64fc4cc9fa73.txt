     4→// PLATFORM PLANS & PRICING — simplified admin UI.
     5→// ============================================================
     6→// Owner sees compact plan cards (Beta / Pro / Max …) with a quick
     7→// Active toggle, a compact price + features + limits summary,
     8→// and an "Edit Plan" button. The Edit Plan button opens a Dialog
     9→// with three compact sections: Basic Information, Feature Access,
    10→// Usage Limits. An optional collapsed "Client Display" section
    11→// holds the marketing feature list shown to clients on the
    12→// Client Billing page. A "+ Create Plan" button at the top-right
    13→// opens a Create Plan dialog (same sections, blank defaults) that
    14→// POSTs to /api/platform/admin/plans.
    15→//
    16→// All writes go through /api/platform/admin/plans (GET list,
    17→// POST create) + /api/platform/admin/plans/[planId] (PUT update)
    18→// — owner-only — the same shared plan-config cache the Client
    19→// Billing page and MRR read. Server-side entitlement enforcement
    20→// (hasFeature) and usage-limit checks (checkLimit) consume the
    21→// same data, so toggling a feature off here denies it for clients
    22→// on the next request. The Client Billing page is NOT modified
    23→// and continues to read from /api/platform/billing/me.
    24→// ============================================================

    88→/** Resolve the badge id for the shared PlanBadge component. Falls
    89→ *  back to 'beta' for any unknown plan id so future plans still
    90→ *  render with a sensible neutral badge. */
    91→function getPlanBadgeId(planId: string): 'beta' | 'pro' | 'max' {
    92→  if (planId === 'beta' || planId === 'pro' || planId === 'max') return planId;
    93→  return 'beta';

   109→function PlanSummaryCard({
   110→  plan,
   111→  canEdit,
   112→}: {
   113→  plan: PlanConfigData;
   114→  canEdit: boolean;
   115→}) {
   116→  const queryClient = useQueryClient();
   117→  const [editing, setEditing] = useState(false);
   118→
   119→  // Quick active toggle — auto-saves immediately (owner-only).
   120→  // Reuses the same PUT endpoint and the same invalidation chain
   121→  // as the full Edit modal so clients see the new status on the
   122→  // next read.
   123→  const activeMutation = useMutation({
   124→    mutationFn: (active: boolean) =>
   125→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, { active }),
   126→    onSuccess: (data) => {
   127→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   128→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   129→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   130→      toast.success(
   131→        `${data.name} ${data.active ? 'activated' : 'deactivated'} — changes are live for clients.`,
   132→      );
   133→    },
   134→    onError: (err: unknown) => {
   135→      const e = err as { error?: { message?: string }; message?: string };
   136→      toast.error(
   137→        e?.error?.message ?? e?.message ?? 'Unable to change plan status. OWNER access required.',
   138→      );
   139→    },
   140→  });
   141→
   142→  // Feature list shown on the card. Prefer the client-facing marketing
   143→  // copy (plan.features); fall back to the structured entitlement labels
   144→  // so a brand-new plan with no marketing copy still surfaces a readable
   145→  // feature list. Either way the data shown is real plan data — no mock.
   146→  const featureItems =
   147→    plan.features.length > 0
   148→      ? plan.features
   149→      : plan.entitlements.map((k) => ENTITLEMENT_LABELS[k as EntitlementKey] ?? k);
   150→
   151→  return (
   152→    <>
   153→      <div
   154→        className={`group relative flex h-full flex-col overflow-hidden rounded-3xl border bg-card p-8 text-card-foreground transition-colors duration-300 hover:border-foreground/20 ${
   155→          plan.active ? '' : 'opacity-60'
   156→        }`}
   157→      >
   158→        {/* Header: plan name (clean text label) + quick active toggle */}
   159→        <div className="flex items-start justify-between gap-3">
   160→          <div className="min-w-0">
   161→            <h3 className="text-2xl font-semibold tracking-tight text-foreground">{plan.name}</h3>
   162→          </div>
   163→          <div className="flex shrink-0 items-center gap-2">
   164→            <Label
   165→              htmlFor={`active-${plan.planId}`}
   166→              className="cursor-pointer text-xs text-muted-foreground"
   167→            >
   168→              {plan.active ? 'Active' : 'Inactive'}
   169→            </Label>
   170→            <Switch
   171→              id={`active-${plan.planId}`}
   172→              checked={plan.active}
   173→              disabled={!canEdit || activeMutation.isPending}
   174→              onCheckedChange={(v) => activeMutation.mutate(v)}
   175→            />
   176→          </div>
   177→        </div>
   178→
   179→        {/* Price — main monthly price only (yearly price intentionally omitted on the card) */}
   180→        <div className="mt-8">
   181→          {plan.isFree ? (
   182→            <div className="flex items-baseline gap-2">
   183→              <span className="text-5xl font-semibold tracking-tight text-foreground">Free</span>
   184→            </div>
   185→          ) : (
   186→            <div className="flex items-baseline gap-2">
   187→              <span className="text-5xl font-semibold tracking-tight text-foreground">
   188→                {formatCurrency(plan.priceMonthly, plan.currency)}
   189→              </span>
   190→              <span className="text-sm text-muted-foreground">/ month</span>
   191→            </div>
   192→          )}
   193→        </div>
   194→
   195→        {/* Thin horizontal divider between the price section and the features section */}
   196→        <div className="mt-8 h-px bg-border" aria-hidden />
   197→
   198→        {/* Feature items (section label omitted) */}
   199→        <div className="mt-6">
   200→          <ul className="space-y-2.5">
   201→            {featureItems.length === 0 ? (
   202→              <li className="text-sm text-muted-foreground">No features configured</li>
   203→            ) : (
   204→              featureItems.map((f, i) => (
   205→                <li key={`${f}-${i}`} className="flex items-start gap-2.5 text-sm text-card-foreground">
   206→                  <Check className="mt-0.5 h-4 w-4 shrink-0 text-muted-foreground" />
   207→                  <span className="leading-snug">{f}</span>
   208→                </li>
   209→              ))
   210→            )}
   211→          </ul>
   212→        </div>
   213→
   214→        {/* Edit button — pinned to card bottom */}
   215→        <div className="mt-auto pt-8">
   216→          <Button
   217→            variant="outline"
   218→            size="sm"
   219→            className="w-full rounded-full"
   220→            disabled={!canEdit}
   221→            onClick={() => setEditing(true)}
   222→          >
   223→            <Pencil className="h-4 w-4 mr-2" />
   224→            Edit Plan
   225→          </Button>
   226→          {!canEdit && (
   227→            <p className="mt-2 text-center text-[10px] text-muted-foreground">
   228→              Editing is restricted to the OWNER role.
   229→            </p>
   230→          )}
   231→        </div>
   232→      </div>
   233→
   234→      {editing && (
   235→        <EditPlanDialog plan={plan} open={editing} onOpenChange={setEditing} />
   236→      )}
   237→    </>
   238→  );
   239→}

   547→/** Derive a planId from a name: lowercase, hyphenated, ASCII-only.
   548→ *  "Enterprise Pro" → "enterprise-pro". Used as a sensible default
   549→ *  the owner can override. */
   550→function derivePlanId(name: string): string {
   551→  return name
   552→    .toLowerCase()
   553→    .trim()
   554→    .replace(/[^a-z0-9]+/g, '-')
   555→    .replace(/^-+|-+$/g, '');
   556→}
