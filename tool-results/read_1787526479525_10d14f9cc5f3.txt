    56→// -------------------- In-Memory Cache --------------------
    57→
    58→class SettingsCache {
    59→  private cache: Map<string, string> = new Map();
    60→  private timestamp: number = 0;
    61→  private readonly TTL_MS = 60_000; // 1 minute
    62→
    63→  get(key: string): string | undefined {
    64→    if (Date.now() - this.timestamp > this.TTL_MS) {
    65→      this.cache.clear();
    66→      return undefined;
    67→    }
    68→    return this.cache.get(key);
    69→  }
    70→
    71→  set(key: string, value: string): void {
    72→    this.cache.set(key, value);
    73→    this.timestamp = Date.now();
    74→  }
    75→
    76→  invalidate(): void {
    77→    this.cache.clear();
    78→    this.timestamp = 0;
    79→  }
    80→
    81→  invalidateKey(key: string): void {
    82→    this.cache.delete(key);
    83→  }
    84→}
    85→

    88→// -------------------- Audit Logging --------------------
    89→
    90→export interface AuditLogEntry {
    91→  settingId?: string;
    92→  key: string;
    93→  category: string;
    94→  oldValue?: string;
    95→  newValue?: string;
    96→  changedBy?: string;
    97→  changedByName?: string;
    98→  siteId?: string;
    99→  ipAddress?: string;
   100→  action?: string;
   101→}
   102→
   103→export async function logSettingChange(entry: AuditLogEntry): Promise<void> {
   104→  try {
   105→    await db.settingsAuditLog.create({
   106→      data: {
   107→        settingId: entry.settingId,
   108→        key: entry.key,
   109→        category: entry.category as any,
   110→        oldValue: entry.oldValue ?? null,
   111→        newValue: entry.newValue ?? null,
   112→        changedBy: entry.changedBy ?? null,
   113→        changedByName: entry.changedByName ?? null,
   114→        siteId: entry.siteId ?? null,
   115→        ipAddress: entry.ipAddress ?? null,
   116→        action: entry.action ?? 'UPDATE',
   117→      },
   118→    });
   119→  } catch (err) {
   120→    console.error('[SETTINGS:AUDIT] Failed to log setting change:', err);
   121→  }
   122→}

   680→/** Validate a single setting value */
   681→export function validateSettingValue(key: string, value: string): { valid: boolean; error?: string } {
   682→  const def = SETTING_DEFS[key];
   683→  if (!def) return { valid: true }; // Unknown keys are allowed (custom settings)
   684→
   685→  // Check required
   686→  if (!value && value !== '0' && value !== 'false') {
   687→    // Allow empty for optional fields
   688→  }
   689→
   690→  // Type validation
   691→  if (def.type === 'NUMBER') {
   692→    const num = Number(value);
   693→    if (isNaN(num)) return { valid: false, error: 'Must be a valid number' };
   694→    if (def.min !== undefined && num < def.min) return { valid: false, error: `Must be at least ${def.min}` };
   695→    if (def.max !== undefined && num > def.max) return { valid: false, error: `Must be at most ${def.max}` };
   696→  }
   697→
   698→  if (def.type === 'BOOLEAN') {
   699→    if (value !== 'true' && value !== 'false') return { valid: false, error: 'Must be true or false' };
   700→  }
   701→
   702→  if (def.type === 'JSON') {
   703→    try { JSON.parse(value); } catch { return { valid: false, error: 'Must be valid JSON' }; }
   704→  }
   705→
   706→  // Validation rules
   707→  if (def.validation === 'email' && value) {
   708→    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
   709→    if (!emailRegex.test(value)) return { valid: false, error: 'Must be a valid email address' };
   710→  }
   711→
   712→  if (def.validation === 'url' && value) {
   713→    try { new URL(value); } catch { return { valid: false, error: 'Must be a valid URL' }; }
   714→  }
   715→
   716→  return { valid: true };
   717→}

   721→/** Get all settings, merging with defaults */
   722→export async function getAllSettings(options?: { category?: string; scope?: string; siteId?: string }) {
   723→  const where: Record<string, unknown> = {};
   724→  if (options?.category) where.category = options.category;
   725→  if (options?.scope) where.scope = options.scope;
   726→  if (options?.siteId) where.siteId = options.siteId;
   727→
   728→  const dbSettings = await db.setting.findMany({ where, orderBy: [{ key: 'asc' }, { updatedAt: 'desc' }] });
   729→
   730→  // Merge with defaults
   731→  const result: Record<string, string> = {};
   732→  for (const cat of SETTINGS_CATEGORIES) {
   733→    if (options?.category && cat.key !== options.category) continue;
   734→    for (const field of cat.fields) {
   735→      const dbSetting = dbSettings.find((s: { key: string }) => s.key === field.key);
   736→      result[field.key] = dbSetting?.value ?? field.defaultValue;
   737→    }
   738→  }
   739→
   740→  // Include any custom settings not in definitions
   741→  for (const s of dbSettings) {
   742→    if (!(s.key in result)) {
   743→      result[s.key] = s.isEncrypted ? '[ENCRYPTED]' : s.value;
   744→    }
   745→  }
   746→
   747→  return result;
   748→}

   750→/** Get a single setting value */
   751→export async function getSettingValue(key: string, scope: string = 'GLOBAL'): Promise<string> {
   752→  const cacheKey = `${scope}:${key}`;
   753→  const cached = settingsCache.get(cacheKey);
   754→  if (cached !== undefined) return cached;
   755→
   756→  const setting = await db.setting.findFirst({ where: { key, scope } });
   757→  const def = getSettingDef(key);
   758→  const value = setting ? (setting.isEncrypted ? '[ENCRYPTED]' : setting.value) : (def?.defaultValue ?? '');
   759→
   760→  settingsCache.set(cacheKey, value);
   761→  return value;
   762→}

   764→/** Batch upsert settings with audit logging */
   765→export async function batchUpsertSettings(
   766→  settings: Array<{ key: string; value: string; type?: string; category?: string }>,
   767→  meta?: { changedBy?: string; changedByName?: string; siteId?: string; ipAddress?: string }
   768→) {
   769→  const results = [];
   770→  const errors: string[] = [];
   771→
   772→  for (const s of settings) {
   773→    // Validate
   774→    const validation = validateSettingValue(s.key, s.value);
   775→    if (!validation.valid) {
   776→      errors.push(`${s.key}: ${validation.error}`);
   777→      continue;
   778→    }
   779→
   780→    const def = getSettingDef(s.key);
   781→    const category = s.category ?? getCategoryForKey(s.key) ?? 'GENERAL';
   782→
   783→    try {
   784→      // Find existing — use findFirst (not upsert's unique key) because
   785→      // userId and siteId are null, and Prisma unique constraints treat
   786→      // NULL as distinct, so upsert.where can't match null-valued keys.
   787→      const existing = await db.setting.findFirst({
   788→        where: { key: s.key, scope: 'GLOBAL', userId: null, siteId: null },
   789→      });
   790→
   791→      const oldValue = existing?.isEncrypted ? undefined : existing?.value;
   792→      const newValue = (s.type === 'ENCRYPTED' || s.type === 'SECRET') ? '[ENCRYPTED]' : s.value;
   793→
   794→      const data = {
   795→        value: s.value,
   796→        type: (s.type as any) ?? def?.type ?? 'STRING',
   797→        category: category as any,
   798→        description: def?.description ?? null,
   799→        isEncrypted: s.type === 'ENCRYPTED' || s.type === 'SECRET' || def?.type === 'ENCRYPTED' || def?.type === 'SECRET',
   800→        isPublic: def?.isPublic ?? false,
   801→      };
   802→
   803→      let item;
   804→      if (existing) {
   805→        item = await db.setting.update({
   806→          where: { id: existing.id },
   807→          data,
   808→        });
   809→      } else {
   810→        item = await db.setting.create({
   811→          data: {
   812→            ...data,
   813→            key: s.key,
   814→            scope: 'GLOBAL',
   815→          },
   816→        });
   817→      }
   818→
   819→      // Audit log
   820→      if (oldValue !== s.value) {
   821→        await logSettingChange({
   822→          settingId: item.id,
   823→          key: s.key,
   824→          category,
   825→          oldValue,
   826→          newValue: s.value,
   827→          ...meta,
   828→        });
   829→      }
   830→
   831→      // Update cache
   832→      settingsCache.invalidateKey(`GLOBAL:${s.key}`);
   833→      results.push(item);
   834→    } catch (err) {
   835→      errors.push(`${s.key}: ${(err as Error).message}`);
   836→    }
   837→  }
   838→
   839→  return { results, errors, updated: results.length, failed: errors.length };
   840→}

   842→/** Search settings by name, key, or description */
   843→export function searchSettingDefs(query: string, options?: { visibleOnly?: boolean }) {
   844→  const q = query.toLowerCase().trim();
   845→  if (!q) return [];
   846→
   847→  const cats = options?.visibleOnly !== false
   848→    ? SETTINGS_CATEGORIES.filter((cat) => UI_VISIBLE_CATEGORY_KEYS.has(cat.key))
   849→    : SETTINGS_CATEGORIES;
   850→
   851→  const matches: Array<{ category: string; categoryLabel: string; field: SettingFieldDef }> = [];
   852→
   853→  for (const cat of cats) {
   854→    for (const field of cat.fields) {
   855→      const searchStr = `${field.key} ${field.label} ${field.description} ${field.group ?? ''}`.toLowerCase();
   856→      if (searchStr.includes(q)) {
   857→        matches.push({ category: cat.key, categoryLabel: cat.label, field });
   858→      }
   859→    }
   860→  }
   861→
   862→  return matches;
   863→}

   865→/** Export all settings */
   866→export async function exportSettings(options?: { includeSensitive?: boolean; category?: string }) {
   867→  const all = await getAllSettings({ category: options?.category });
   868→  const exportData: Record<string, { value: string; type?: string; category?: string; encrypted: boolean }> = {};
   869→
   870→  for (const [key, value] of Object.entries(all)) {
   871→    const def = getSettingDef(key);
   872→    if (def?.isSensitive && !options?.includeSensitive) continue;
   873→
   874→    exportData[key] = {
   875→      value,
   876→      type: def?.type,
   877→      category: getCategoryForKey(key),
   878→      encrypted: def?.type === 'ENCRYPTED' || def?.type === 'SECRET' || false,
   879→    };
   880→  }
   881→
   882→  return {
   883→    exportedAt: new Date().toISOString(),
   884→    version: '1.0',
   885→    settings: exportData,
   886→    totalSettings: Object.keys(exportData).length,
   887→  };
   888→}

   927→/** Get audit log entries */
   928→export async function getAuditLog(options?: {
   929→  category?: string;
   930→  key?: string;
   931→  page?: number;
   932→  pageSize?: number;
   933→  siteId?: string;
   934→}) {
   935→  const where: Record<string, unknown> = {};
   936→  if (options?.category) where.category = options.category;
   937→  if (options?.key) where.key = { contains: options.key };
   938→  if (options?.siteId) where.siteId = options.siteId;
   939→
   940→  const page = options?.page ?? 1;
   941→  const pageSize = Math.min(options?.pageSize ?? 25, 100);
   942→
   943→  const [items, total] = await Promise.all([
   944→    db.settingsAuditLog.findMany({
   945→      where,
   946→      orderBy: { createdAt: 'desc' },
   947→      skip: (page - 1) * pageSize,
   948→      take: pageSize,
   949→    }),
   950→    db.settingsAuditLog.count({ where }),
   951→  ]);
   952→
   953→  return {
   954→    items,
   955→    pagination: {
   956→      page,
   957→      pageSize,
   958→      total,
   959→      totalPages: Math.ceil(total / pageSize),
   960→    },
   961→  };
   962→}
