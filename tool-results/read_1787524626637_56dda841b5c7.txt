   100→// -------------------- Sentiment Detection --------------------
   101→
   102→const POSITIVE_WORDS = new Set([
   103→  'great', 'love', 'excellent', 'amazing', 'good', 'nice', 'best',
   104→  'wonderful', 'perfect', 'awesome',
   105→]);
   106→
   107→const NEGATIVE_WORDS = new Set([
   108→  'bad', 'terrible', 'horrible', 'worst', 'hate', 'awful', 'poor',
   109→  'disgusting', 'waste', 'disappointed',
   110→]);
   111→
   112→function detectSentiment(text: string): SentimentType {
   113→  const lower = text.toLowerCase();
   114→  const words = lower.split(/\W+/);
   115→  let posCount = 0;
   116→  let negCount = 0;
   117→  for (const w of words) {
   118→    if (POSITIVE_WORDS.has(w)) posCount++;
   119→    if (NEGATIVE_WORDS.has(w)) negCount++;
   120→  }
   121→  if (posCount > negCount) return 'positive';
   122→  if (negCount > posCount) return 'negative';
   123→  return 'neutral';
   124→}

   264→  const saveMutation = useMutation({
   265→    mutationFn: () =>
   266→      postApi('/api/settings', {
   267→        settings: [
   268→          { key: 'enable_comments', value: String(commentsEnabled), type: 'BOOLEAN', category: 'DISCUSSION' },
   269→          { key: 'comment_auto_spam_detection', value: String(spamDetection), type: 'BOOLEAN', category: 'DISCUSSION' },
   270→          { key: 'comment_spam_provider', value: spamProvider, type: 'STRING', category: 'DISCUSSION' },
   271→        ],
   272→      }),
   273→    onSuccess: () => {
   274→      setDraft({});
   275→      queryClient.invalidateQueries({ queryKey: ['settings'] });

   277→    },
   278→    onError: (err: Error) => toast.error(err.message || 'Failed to save settings'),
   279→  });

   281→  if (isLoading) {
   282→    return (
   283→      <Card className="p-4">
   284→        <div className="flex items-center gap-3">
   285→          <Skeleton className="h-6 w-32" />

   300→            checked={commentsEnabled}
   301→            onCheckedChange={(v) => setDraft((prev) => ({ ...prev, commentsEnabled: v }))}

   345→        {/* Save button — only visible when there are unsaved changes */}

   347→          size="sm"
   348→          className={cn('ml-auto', !isDirty && 'invisible')}

   354→          ) : null}

   356→        </Button>
   357→      </div>
   358→    </Card>
   359→  );
   360→}
