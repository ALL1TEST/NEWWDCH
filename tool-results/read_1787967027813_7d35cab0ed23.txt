   532→// =====================================================================
   533→
   534→export async function POST(request: NextRequest) {
   535→  const id = reqId();
   536→
   537→  try {
   538→    // Parse body to detect scope=platform (platform-admin-only). When the
   539→    // body is empty or not JSON, fall back to legacy client behavior.
   540→    let bodyScope: string | undefined;
   541→    try {
   542→      const body = await request.json();
   543→      bodyScope = typeof body === 'object' && body !== null
   544→        ? (body as { scope?: unknown }).scope as string | undefined
   545→        : undefined;
   546→    } catch {
   547→      // No body — legacy client seed. Default to client behavior.
   548→    }
   549→    const isPlatformScope = bodyScope === 'platform';
   550→
   551→    // -------- scope=platform: platform admin can seed system templates
   552→    // (siteId = null). Falls through to legacy client behavior otherwise.
   553→    let platformCreatedById: string | null = null;
   554→    if (isPlatformScope) {
   555→      const auth = await requirePlatformAdmin(request);
   556→      if ('response' in auth) return auth.response;
   557→      platformCreatedById = auth.user.id;
   558→    }
   559→
   560→    // Find a known user to use as createdById
   561→    const firstUser = await db.user.findFirst({ select: { id: true } });
   562→    if (!firstUser) {
   563→      return NextResponse.json(
   564→        { error: { code: 'NO_USER', message: 'No users found in the database. Create a user first.' }, meta: { requestId: id } },
   565→        { status: 400 },
   566→      );
   567→    }
   568→
   569→    const createdById = isPlatformScope && platformCreatedById ? platformCreatedById : firstUser.id;
   570→    let seeded = 0;
   571→    let skipped = 0;
   572→
   573→    for (const tpl of DEFAULT_TEMPLATES) {
   574→      const existing = await db.emailTemplate.findUnique({ where: { slug: tpl.slug } });
   575→      if (existing) {
   576→        skipped++;
   577→        continue;
   578→      }
   579→
   580→      await db.emailTemplate.create({
   581→        data: {
   582→          // Platform-scope templates are system-level (siteId = null);
   583→          // client-scope templates use the legacy behavior (also null here
   584→          // since the seed route has always created system templates).
   585→          siteId: null,
   586→          name: tpl.name,
   587→          slug: tpl.slug,
   588→          subject: tpl.subject,
   589→          previewText: tpl.previewText,
   590→          htmlBody: tpl.htmlBody,
   591→          defaultBody: tpl.htmlBody,
   592→          category: tpl.category as 'CUSTOMER_EMAILS' | 'AUTHENTICATION' | 'NEWSLETTER' | 'MARKETING' | 'TRANSACTIONAL' | 'NOTIFICATIONS' | 'BILLING' | 'SYSTEM',
   593→          status: 'ENABLED',
   594→          provider: 'SMTP',
   595→          isSystem: true,
   596→          createdById,
   597→        },
   598→      });
   599→      seeded++;
   600→    }
   601→
   602→    return NextResponse.json({
   603→      data: { seeded, skipped, total: DEFAULT_TEMPLATES.length },
   604→      meta: { requestId: id },
   605→    });
   606→  } catch (error) {
   607→    console.error(`[EMAIL_TEMPLATES:SEED] ${id} —`, error);
   608→    return NextResponse.json(
   609→      { error: { code: 'INTERNAL_ERROR', message: 'Failed to seed email templates' }, meta: { requestId: id } },
   610→      { status: 500 },
   611→    );
   612→  }
   613→}
