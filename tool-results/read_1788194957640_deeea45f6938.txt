     2→     2→// PLAN CONFIG — DB-backed, cached single source of truth for plans,
     3→     3→// pricing, entitlements and usage limits.
     4→     4→// ============================================================
     5→     5→// The Client Billing API and the Platform Admin both read from (and
     6→     6→// the admin mutates) this same service. `platform-data.ts` delegates
     7→     7→// its `PLANS` / `getPlan` / `monthlyPrice` to the synchronous cache
     8→     8→// exposed here, so an owner edit (savePlanConfig) propagates to the
     9→     9→// client billing experience and to MRR on the next read.
    10→    10→//
    11→    11→// MULTI-CURRENCY: the plan carries ONE base price configuration —
    12→    12→// priceMonthly / priceYearly denominated in the plan's default /
    13→    13→// fallback `currency` (both edited directly in the Edit Plan modal).
    14→    14→// The customer's currency is resolved SERVER-SIDE from their IP
    15→    15→// geolocation (see country-pricing.ts:resolveCustomerPricing) when
    16→    16→// the plan's `autoCurrency` flag is on; otherwise every customer is
    17→    17→// billed in the plan default currency.
    18→    18→//
    19→    19→// pricesByCurrency is a DERIVED store (NOT edited per-plan in the
    20→    20→// modal): the default currency's entry mirrors the base price on
    21→    21→// every save, and the other entries hold the platform-level
    22→    22→// regional prices (CountryPricing.regionalPrices + legacy entries)
    23→    23→// — preserved across saves so customers in other currencies keep
    24→    24→// seeing their configured price (e.g. "90 MAD / month").
    25→    25→//
    26→    26→// STRIPE: for each paid plan × supported currency × interval, a real
    27→    27→// Stripe Price is created on sync and its ID persisted in
    28→    28→// stripePriceIdsByCurrency. The default currency's IDs are mirrored
    29→    29→// in stripePriceIdMonthly/Yearly for legacy callers. Checkout always
    30→    30→// resolves the per-currency Stripe Price ID server-side — the
    31→    31→// customer's currency determines which Price is charged.
    32→    32→//
    33→    33→// The cache is hydrated from the PlanConfig table on startup and
    34→    34→// after every mutation. If the table is empty on first hydrate, the
    35→    35→// DEFAULT configs are seeded.

    93→    93→// -------------------- Entitlement normalization --------------------
    94→    94→
    95→    95→/** Normalize a raw entitlement key list (DB row or API input):
    96→    96→ *  - legacy 'ai_content' → 'ai_platform' (it predates the two-mode
    97→    97→ *    split and always meant the platform-provided AI).
    98→    98→ *  - mutual exclusion: when BOTH AI modes are present (invalid data),
    99→    99→ *    Platform AI wins and 'ai_client' is dropped.
   100→   100→ *  - legacy 'custom_domains' / 'white_label' are STRIPPED: site
   101→   101→ *    domains and site branding are client-owned in this architecture
   102→   102→ *    (every site carries its own domain + identity), so they are not
   103→   103→ *    plan entitlements. Old DB rows self-clean on load + next save.
   104→   104→ *  - dedupes while preserving order.
   105→   105→ *  Non-editor keys (e.g. 'audit_log') pass through untouched. */
   106→   106→export function normalizeEntitlementKeys(keys: readonly string[]): string[] {
   107→   107→  // Legacy keys that are no longer plan entitlements — stripped on
   108→   108→  // load/save so existing rows + stale API input clean themselves.
   109→   109→  const LEGACY_REMOVED_KEYS = new Set(['custom_domains', 'white_label']);
   110→   110→  const out: string[] = [];
   111→   111→  const seen = new Set<string>();
   112→   112→  const hasPlatform = keys.includes('ai_platform') || keys.includes('ai_content');
   113→   113→  for (const k of keys) {
   114→   114→    if (typeof k !== 'string' || !k.trim()) continue;
   115→   115→    if (k === 'ai_content') continue; // legacy → replaced by ai_platform below
   116→   116→    if (LEGACY_REMOVED_KEYS.has(k)) continue; // not plan entitlements anymore
   117→   117→    if (k === 'ai_client' && hasPlatform) continue; // mutually exclusive → platform wins
   118→   118→    if (k === 'ai_platform' && hasPlatform) {
   119→   119→      if (!seen.has('ai_platform')) out.push('ai_platform');
   120→   120→      seen.add('ai_platform');
   121→   121→      continue;
   122→   122→    }
   123→   123→    if (!seen.has(k)) {
   124→   124→      out.push(k);
   125→   125→      seen.add(k);
   126→   126→    }
   127→   127→  }
   128→   128→  // Insert the platform AI key at the position where the legacy key
   129→   129→  // (or an existing 'ai_platform') first appeared — append at the end
   130→   130→  // when neither was present in the ordered loop above (defensive).
   131→   131→  if (hasPlatform && !seen.has('ai_platform')) out.push('ai_platform');
   132→   132→  return out;
   133→   133→}

   169→   169→export type BillingInterval = 'monthly' | 'yearly';

   171→   171→/** The billing periods enabled on a plan (at least one, enforced).
   172→   172→ *  A disabled period: not shown on the Client Billing page, rejected
   173→   173→ *  at checkout (400 BILLING_PERIOD_NOT_ENABLED), and never gets a
   174→   174→ *  Stripe Price created or selected. */
   175→   175→export function enabledIntervalsOf(plan: {
   176→   176→  billingMonthly: boolean;
   177→   177→  billingYearly: boolean;
   178→   178→}): BillingInterval[] {
   179→   179→  const out: BillingInterval[] = [];
   180→   180→  if (plan.billingMonthly) out.push('monthly');
   181→   181→  if (plan.billingYearly) out.push('yearly');
   182→   182→  return out;
   183→   183→}

   185→   185→/** Clamp a requested default cadence to an ENABLED period:
   186→   186→ *  only monthly on → 'monthly'; only yearly on → 'yearly'; both on
   187→   187→ *  → the requested value. (Neither enabled is rejected upstream —
   188→   188→ *  here it falls back to the requested value.) */
   189→   189→function normalizeInterval(
   190→   190→  requested: BillingInterval,
   191→   191→  billingMonthly: boolean,
   192→   192→  billingYearly: boolean,
   193→   193→): BillingInterval {
   194→   194→  if (billingMonthly && !billingYearly) return 'monthly';
   195→   195→  if (!billingMonthly && billingYearly) return 'yearly';
   196→   196→  return requested;
   197→   197→}

   645→   645→/**
   646→   646→ * Migrate the legacy catalog to the new Free/Plus/Pro/Max shape. Idempotent.
   647→   647→ *
   648→   648→ * Legacy catalog (created by older bootstrap):
   649→   649→ *   beta (free, active=false), pro, max, enterprise (stub)
   650→   650→ *
   651→   651→ * Migration steps (each guarded, can be re-run safely):
   652→   652→ *  1. If `enterprise` exists AND no active subscription references it → delete it.
   653→   653→ *     (Active subs that reference enterprise are preserved — the migration will
   654→   654→ *      not break them; the plan row stays if needed. In practice, the demo
   655→   655→ *      dataset has no DB subscriptions yet, so this is a clean delete.)
   656→   656→ *  2. If `beta` exists → rename to `plus` (planId, name, badgeVariant) and
   657→   657→ *     re-enable it (legacy `beta` was active=false in the demo DB). Also
   658→   658→ *     shift sortOrder to 1 (Free takes 0).
   659→   659→ *  3. If `free` does NOT exist → insert the Free plan at sortOrder 0.
   660→   660→ *  4. Renumber sortOrder to match the canonical order: free=0, plus=1, pro=2, max=3.
   661→   661→ *
   662→   662→ * Returns a short report string for the bootstrap log.
   663→   663→ */
   664→   664→export async function migrateLegacyPlans(): Promise<string> {
   665→   665→  const notes: string[] = [];
   666→   666→
   667→   667→  try {
   668→   668→    // Step 1: delete Enterprise if safe
   669→   669→    const enterprise = await db.planConfig.findUnique({ where: { planId: 'enterprise' } });
   670→   670→    if (enterprise) {
   671→   671→      const subCount = await db.subscription.count({ where: { planId: 'enterprise' } });
   672→   672→      if (subCount === 0) {
   673→   673→        await db.planConfig.delete({ where: { planId: 'enterprise' } });
   674→   674→        notes.push('deleted enterprise plan (0 active subscriptions)');
   675→   675→      } else {
   676→   676→        notes.push(`kept enterprise plan (${subCount} active subscriptions reference it)`);
   677→   677→      }
   678→   678→    }
   679→   679→
   680→   680→    // Step 2: rename beta → plus (and refresh to a real paid tier)
   681→   681→    const beta = await db.planConfig.findUnique({ where: { planId: 'beta' } });
   682→   682→    if (beta) {
   683→   683→      // Don't overwrite an existing plus row — if both exist, prefer the
   684→   684→      // plus row and just delete the beta stub.
   685→   685→      const plus = await db.planConfig.findUnique({ where: { planId: 'plus' } });
   686→   686→      if (!plus) {
   687→   687→        // Rename beta → plus AND refresh the data to the real DEFAULT
   688→   688→        // Plus tier (paid plan with real entitlements + limits, not the
   689→   689→        // legacy free-tier Beta stub). The old Beta was active=false and
   690→   690→        // had no real customers, so this is a safe refresh.
   691→   691→        const plusDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'plus')!;
   692→   692→        await db.planConfig.update({
   693→   693→          where: { planId: 'beta' },
   694→   694→          data: {
   695→   695→            ...dataToRow(plusDefault),
   696→   696→            // Keep the original createdAt for continuity.
   697→   697→            // (updatedAt is auto.)
   698→   698→          },
   699→   699→        });
   700→   700→        notes.push('renamed + refreshed beta → plus (real paid tier)');
   701→   701→      } else {
   702→   702→        // Both beta and plus exist — delete the beta stub (migrate any
   703→   703→        // legacy subscriptions on beta to plus first).
   704→   704→        const betaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   705→   705→        if (betaSubs > 0) {
   706→   706→          await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   707→   707→          notes.push(`migrated ${betaSubs} beta subscription(s) → plus`);
   708→   708→        }
   709→   709→        await db.planConfig.delete({ where: { planId: 'beta' } });
   710→   710→        notes.push('deleted legacy beta stub (plus already exists)');
   711→   711→      }
   712→   712→    }
   713→   713→
   714→   714→    // Step 3: insert Free if missing
   715→   715→    const free = await db.planConfig.findUnique({ where: { planId: 'free' } });
   716→   716→    if (!free) {
   717→   717→      const freeDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'free')!;
   718→   718→      await db.planConfig.create({ data: dataToRow(freeDefault) });
   719→   719→      notes.push('inserted free plan');
   720→   720→    } else {
   721→   721→      // Ensure free plan is properly flagged
   722→   722→      if (!free.isFree || free.badgeVariant !== 'free' || free.sortOrder !== 0) {
   723→   723→        await db.planConfig.update({
   724→   724→          where: { planId: 'free' },
   725→   725→          data: { isFree: true, badgeVariant: 'free', sortOrder: 0 },
   726→   726→        });
   727→   727→        notes.push('normalized free plan flags');
   728→   728→      }
   729→   729→    }
   730→   730→
   731→   731→    // Step 4: normalize sortOrder for all canonical plans
   732→   732→    const canonicalOrder: Record<string, number> = { free: 0, plus: 1, pro: 2, max: 3 };
   733→   733→    for (const [pid, order] of Object.entries(canonicalOrder)) {
   734→   734→      const row = await db.planConfig.findUnique({ where: { planId: pid } });
   735→   735→      if (row && row.sortOrder !== order) {
   736→   736→        await db.planConfig.update({ where: { planId: pid }, data: { sortOrder: order } });
   737→   737→      }
   738→   738→    }
   739→   739→
   740→   740→    // Also migrate legacy subscriptions on 'beta' or 'enterprise' planId
   741→   741→    // to the new ids (beta → plus, enterprise → max) so historical subs
   742→   742→    // don't reference a missing plan.
   743→   743→    const legacyBetaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   744→   744→    if (legacyBetaSubs > 0) {
   745→   745→      await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   746→   746→      notes.push(`migrated ${legacyBetaSubs} legacy beta subscription(s) → plus`);
   747→   747→    }
   748→   748→    const legacyEnterpriseSubs = await db.subscription.count({ where: { planId: 'enterprise' } });
   749→   749→    if (legacyEnterpriseSubs > 0) {
   750→   750→      await db.subscription.updateMany({ where: { planId: 'enterprise' }, data: { planId: 'max' } });
   751→   751→      notes.push(`migrated ${legacyEnterpriseSubs} legacy enterprise subscription(s) → max`);
   752→   752→    }
   753→   753→  } catch (err) {
   754→   754→    notes.push(`migration error (non-fatal): ${(err as Error).message}`);
   755→   755→  }
   756→   756→
   757→   757→  return notes.length > 0 ? notes.join('; ') : 'no migration needed';
   758→   758→}

   760→   760→/** Load all plan configs from the DB into the cache. Self-seeds the
   761→   761→ *  DEFAULT_PLAN_CONFIGS when the table is empty. Runs the legacy
   762→   762→ *  migration when the catalog still contains 'beta' or 'enterprise'.
   763→   763→ *  Safe to call repeatedly. */
   764→   764→export async function hydrate(): Promise<void> {
   765→   765→  if (_hydrating) return _hydrating;
   766→   766→  _hydrating = (async () => {
   767→   767→    try {
   768→   768→      const rows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   769→   769→      if (rows.length === 0) {
   770→   770→        // Self-seed so the platform is never without plans.
   771→   771→        await db.planConfig.createMany({ data: DEFAULT_PLAN_CONFIGS.map(dataToRow) });
   772→   772→        _cache = DEFAULT_PLAN_CONFIGS.map((p) => ({
   773→   773→          ...p,
   774→   774→          features: [...p.features],
   775→   775→          entitlements: [...p.entitlements],
   776→   776→          limits: { ...p.limits },
   777→   777→        }));
   778→   778→      } else {
   779→   779→        // Check for legacy catalog (beta / enterprise) and migrate.
   780→   780→        const hasLegacy =
   781→   781→          rows.some((r) => r.planId === 'beta' || r.planId === 'enterprise') ||
   782→   782→          !rows.some((r) => r.planId === 'free');
   783→   783→        if (hasLegacy) {
   784→   784→          await migrateLegacyPlans();
   785→   785→          const migratedRows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   786→   786→          _cache = migratedRows.map(rowToData);
   787→   787→        } else {
   788→   788→          _cache = rows.map(rowToData);
   789→   789→        }
   790→   790→      }
   791→   791→      _hydrated = true;
   792→   792→      notify();
   793→   793→    } catch (err) {
   794→   794→      // DB unavailable — keep defaults (still functional, just not persisted).
   795→   795→      _hydrated = true;
   796→   796→      notify();
   797→   797→    } finally {
   798→   798→      _hydrating = null;
   799→   799→    }
   800→   800→  })();
   801→   801→  return _hydrating;
   802→   802→}

   843→   843→/** Merge a patch (Partial<PlanConfigData>) into the current data.
   844→   844→ *  The Edit Plan modal sends the ONE base price configuration
   845→   845→ *  (priceMonthly + priceYearly + currency + autoCurrency) — NOT a
   846→   846→ *  per-currency matrix. Derivation rules:
   847→   847→ *  - pricesByCurrency: starts from the current map (or the explicit
   848→   848→ *    patch map when provided); whenever the base price / currency is
   849→   849→ *    in the patch, the DEFAULT currency's entry is re-derived as
   850→   850→ *    { monthly: priceMonthly, yearly: priceYearly }. Other currency
   851→   851→ *    entries (platform regional / legacy prices) are PRESERVED.
   852→   852→ *  - stripePriceIdsByCurrency: kept as-is when omitted (the Stripe
   853→   853→ *    sync maintains it); the default-currency snapshots mirror the
   854→   854→ *    default entry.
   855→   855→ */
   856→   856→function mergePlanPatch(current: PlanConfigData, patch: PlanConfigInput, defaultCurrency: string): PlanConfigData {
   857→   857→  const curUpper = defaultCurrency.toUpperCase();
   858→   858→
   859→   859→  // pricesByCurrency — clone (never mutate the cache's object in place).
   860→   860→  const sourcePrices = patch.pricesByCurrency ?? current.pricesByCurrency ?? {};
   861→   861→  const pricesByCurrency: PricesByCurrency = { ...sourcePrices };
   862→   862→
   863→   863→  // Base price precedence:
   864→   864→  //   1. Explicit patch values (the modal always sends these).
   865→   865→  //   2. Legacy mirror: an explicit pricesByCurrency map with an entry
   866→   866→  //      for the final default currency (backward-compat callers).
   867→   867→  //   3. Current snapshot.
   868→   868→  const mirrored =
   869→   869→    patch.priceMonthly === undefined && patch.pricesByCurrency !== undefined
   870→   870→      ? pricesByCurrency[curUpper]
   871→   871→      : undefined;
   872→   872→  const priceMonthly =
   873→   873→    patch.priceMonthly !== undefined ? patch.priceMonthly : mirrored?.monthly ?? current.priceMonthly;
   874→   874→  const priceYearly =
   875→   875→    patch.priceYearly !== undefined ? patch.priceYearly : mirrored?.yearly ?? current.priceYearly;
   876→   876→
   877→   877→  // The base price configuration was touched (or a legacy map was sent)
   878→   878→  // → re-derive the DEFAULT currency's entry from the base price. All
   879→   879→  // OTHER currency entries (platform regional / legacy prices) are
   880→   880→  // preserved untouched.
   881→   881→  const baseTouched =
   882→   882→    patch.priceMonthly !== undefined ||
   883→   883→    patch.priceYearly !== undefined ||
   884→   884→    patch.currency !== undefined ||
   885→   885→    patch.pricesByCurrency !== undefined;
   886→   886→  if (baseTouched) {
   887→   887→    pricesByCurrency[curUpper] = { monthly: priceMonthly, yearly: priceYearly };
   888→   888→  }
   889→   889→
   890→   890→  const stripePriceIdsByCurrency: StripePriceIdsByCurrency = {
   891→   891→    ...(patch.stripePriceIdsByCurrency ?? current.stripePriceIdsByCurrency ?? {}),
   892→   892→  };
   893→   893→  // Default-currency snapshot — keep the legacy fields in sync so legacy
   894→   894→  // callers that read stripePriceIdMonthly/Yearly see the right values.
   895→   895→  const defIds = stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
   896→   896→  const stripePriceIdMonthly =
   897→   897→    patch.stripePriceIdMonthly !== undefined
   898→   898→      ? patch.stripePriceIdMonthly
   899→   899→      : patch.stripePriceIdsByCurrency !== undefined
   900→   900→        ? defIds.monthly
   901→   901→        : current.stripePriceIdMonthly;
   902→   902→  const stripePriceIdYearly =
   903→   903→    patch.stripePriceIdYearly !== undefined
   904→   904→      ? patch.stripePriceIdYearly
   905→   905→      : patch.stripePriceIdsByCurrency !== undefined
   906→   906→        ? defIds.yearly
   907→   907→        : current.stripePriceIdYearly;
   908→   908→  const isFree = patch.isFree ?? current.isFree;
   909→   909→  // ---- Billing periods ----
   910→   910→  const billingMonthly = patch.billingMonthly ?? current.billingMonthly;
   911→   911→  const billingYearly = patch.billingYearly ?? current.billingYearly;
   912→   912→  // The default cadence must be an ENABLED period (single-period plans
   913→   913→  // are pinned to their only period; both-period plans keep the
   914→   914→  // admin's requested / current cadence).
   915→   915→  const interval = normalizeInterval(
   916→   916→    patch.interval ?? current.interval,
   917→   917→    billingMonthly,
   918→   918→    billingYearly,
   919→   919→  );
   920→   920→  return {
   921→   921→    planId: current.planId,
   922→   922→    name: patch.name ?? current.name,
   923→   923→    priceMonthly,
   924→   924→    priceYearly,
   925→   925→    currency: curUpper,
   926→   926→    autoCurrency: patch.autoCurrency ?? current.autoCurrency,
   927→   927→    pricesByCurrency,
   928→   928→    stripePriceIdsByCurrency,
   929→   929→    billingMonthly,
   930→   930→    billingYearly,
   931→   931→    interval,
   932→   932→    isFree,
   933→   933→    freePlanDurationDays: patch.freePlanDurationDays ?? current.freePlanDurationDays,
   934→   934→    stripePriceIdMonthly,
   935→   935→    stripePriceIdYearly,
   936→   936→    active: patch.active ?? current.active,
   937→   937→    features: patch.features ?? current.features,
   938→   938→    // Entitlements: the patch's editor-managed keys replace the current
   939→   939→    // ones, while NON-editor keys (e.g. 'audit_log') are PRESERVED from
   940→   940→    // the current plan — the Create/Edit Plan modal only manages the 10
   941→   941→    // Feature Access keys. Then normalize (legacy 'ai_content' →
   942→   942→    // 'ai_platform'; the two AI modes are mutually exclusive).
   943→   943→    entitlements: normalizeEntitlementKeys(
   944→   944→      patch.entitlements !== undefined
   945→   945→        ? [
   946→   946→            ...patch.entitlements,
   947→   947→            ...current.entitlements.filter(
   948→   948→              (k) =>
   949→   949→                !patch.entitlements!.includes(k) &&
   950→   950→                k !== 'ai_content' &&
   951→   951→                // keys below are editor-managed (replaced by the patch):
   952→   952→                !PLAN_EDITOR_KEY_SET.has(k),
   953→   953→            ),
   954→   954→          ]
   955→   955→        : current.entitlements,
   956→   956→    ),
   957→   957→    // Limits: merge; the AI usage limits are only part of the saved
   958→   958→    // configuration while the plan uses Platform AI (zeroed otherwise).
   959→   959→    limits: zeroAiLimitsIfNotPlatform(
   960→   960→      pickLimits({ ...current.limits, ...(patch.limits ?? {}) }),
   961→   961→      patch.entitlements !== undefined
   962→   962→        ? normalizeEntitlementKeys(patch.entitlements)
   963→   963→        : current.entitlements,
   964→   964→    ),
   965→   965→    badgeVariant: patch.badgeVariant ?? current.badgeVariant,
   966→   966→    sortOrder: patch.sortOrder ?? current.sortOrder,
   967→   967→  };
   968→   968→}

   970→   970→/** Update an existing plan config. Writes DB, refreshes cache.
   971→   971→ *
   972→   972→ *  AUTO-SYNC TO STRIPE: when Stripe is configured AND the plan is
   973→   973→ *  PAID (any currency has a positive price) AND the admin did NOT
   974→   974→ *  explicitly set the Stripe Price IDs in this patch, the backend calls
   975→   975→ *  syncPlanToStripe (multi-currency) to ensure a Stripe Product +
   976→   976→ *  per-currency monthly+yearly Prices exist and writes the resolved IDs
   977→   977→ *  into stripePriceIdsByCurrency + the default-currency snapshot fields.

   979→   979→export async function savePlanConfig(planId: string, patch: PlanConfigInput): Promise<PlanConfigData | null> {
   980→   980→  const existing = await db.planConfig.findUnique({ where: { planId } });
   981→   981→  if (!existing) return null;
   982→   982→  const current = rowToData(existing);
   983→   983→
   984→   984→  // When the admin omitted currency, resolve the platform default so the
   985→   985→  // legacy priceMonthly/priceYearly snapshot stays in the right currency.
   986→   986→  if (!patch.currency && !current.currency) {
   987→   987→    patch = { ...patch, currency: await resolvePlatformDefaultCurrency() };
   988→   988→  }
   989→   989→  const finalCurrency = (patch.currency ?? current.currency).toUpperCase();
   990→   990→
   991→   991→  // Distinguish "field omitted" from "field set". When the admin omits
   992→   992→  // both stripePriceId* AND stripePriceIdsByCurrency, they want the
   993→   993→  // backend to auto-sync. When either is explicitly provided, manual.
   994→   994→  const adminTouchedStripePriceIds =
   995→   995→    'stripePriceIdMonthly' in patch ||
   996→   996→    'stripePriceIdYearly' in patch ||
   997→   997→    'stripePriceIdsByCurrency' in patch;

  1007→  1007→    (p) =>

  1009→  1009→      (next.billingYearly && p.yearly > 0),

  1017→  1017→          planId: next.planId,

  1028→  1028→          JSON.stringify(syncResult.stripePriceIdsByCurrency) !==
  1029→  1029→            JSON.stringify(next.stripePriceIdsByCurrency) ||
