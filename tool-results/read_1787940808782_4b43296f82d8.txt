   116→
   117→/** Inline empty state rendered inside the table body when an active search
   118→ * yields zero results. Distinct from the standalone "No schedules configured"

   123→function NoSchedulesSearchResultsEmpty({ onClear }: { onClear: () => void }) {

   131→      </Button>

   133→  );

   136→// -------------------- Schedules Page --------------------
   137→
   138→export function SchedulesPage() {
   139→  const queryClient = useQueryClient();
   140→  const [dialogOpen, setDialogOpen] = useState(false);
   141→  const [editingId, setEditingId] = useState<string | null>(null);
   142→  const [form, setForm] = useState<ScheduleForm>(initialForm);
   143→  const [deleteTarget, setDeleteTarget] = useState<ScheduleRow | null>(null);
   144→
   145→  const table = useDataTable({ initialSortField: 'createdAt', initialSortOrder: 'desc' });
   146→
   147→  const { data, isLoading } = useQuery({
   148→    queryKey: queryKeys.backupSchedules.list({
   149→      page: table.currentPage,
   150→      pageSize: table.pageSize,
   151→      sort: table.sortField,
   152→      order: table.sortOrder,
   153→      search: table.searchValue || undefined,
   154→    }),
   155→    queryFn: () => getApi<ApiResponse<ScheduleRow[]>>('/api/backups/schedules', {
   156→      page: table.currentPage,
   157→      pageSize: table.pageSize,
   158→      sort: table.sortField,
   159→      order: table.sortOrder,
   160→      search: table.searchValue || undefined,
   161→    }, { raw: true }),
   162→    staleTime: 10_000,
   163→  });
   164→
   165→  // Fetch configured storage destinations for the dropdown
   166→  const { data: storageData } = useQuery({
   167→    queryKey: ['backup-storage-destinations'],
   168→    queryFn: () => getApi<{ id: string; name: string; provider: string; isActive: boolean }[]>('/api/backups/storage?pageSize=100'),
   169→    staleTime: 30_000,
   170→  });
   171→  const storageDestinations = (storageData as unknown as { id: string; name: string; provider: string; isActive: boolean }[] | undefined)?.filter(s => s.isActive) ?? [];
   172→
   173→  const schedules = data?.data ?? [];
   174→  const pagination = data?.meta?.pagination;
   175→
   176→  // Dual empty-state logic (mirrors the Backups table):
   177→  // - isInitialEmpty: system genuinely has zero schedules AND no search is
   178→  //   active → show the full-page "No schedules configured" state.
   179→  // - isSearchEmpty: a search IS active but returns zero results → keep the
   180→  //   table card/headers/pagination visible and render the inline
   181→  //   "No schedules found" empty state inside the table body.
   182→  // These two states must NEVER be confused.
   183→  const hasSearch = !!table.searchValue?.trim();
   184→  const isInitialEmpty = !isLoading && schedules.length === 0 && !hasSearch;
   185→  const isSearchEmpty = !isLoading && schedules.length === 0 && hasSearch;
   186→
   187→  const createMutation = useMutation({
   188→    mutationFn: (body: ScheduleForm) => postApi('/api/backups/schedules', body),
   189→    onSuccess: () => {
   190→      queryClient.invalidateQueries({ queryKey: queryKeys.backupSchedules.all });
   191→      toast.success('Schedule created');
   192→      closeDialog();
   193→    },
   194→    onError: (err: Error) => toast.error(err.message || 'Failed to create schedule'),
   195→  });
   196→
   197→  const updateMutation = useMutation({
   198→    mutationFn: ({ id, body }: { id: string; body: Partial<ScheduleForm> }) =>
   199→      patchApi(`/api/backups/schedules/${id}`, body),
   200→    onSuccess: () => {
   201→      queryClient.invalidateQueries({ queryKey: queryKeys.backupSchedules.all });
   202→      toast.success('Schedule updated');
   203→      closeDialog();
   204→    },
   205→    onError: (err: Error) => toast.error(err.message || 'Failed to update schedule'),
   206→  });
   207→
   208→  const deleteMutation = useMutation({
   209→    mutationFn: (id: string) => deleteApi(`/api/backups/schedules/${id}`),
   210→    onSuccess: () => {
   211→      queryClient.invalidateQueries({ queryKey: queryKeys.backupSchedules.all });
   212→      setDeleteTarget(null);
   213→      toast.success('Schedule deleted');
   214→    },
   215→    onError: (err: Error) => {
   216→      setDeleteTarget(null);
   217→      toast.error(err.message || 'Failed to delete schedule');
   218→    },
   219→  });
   220→
   221→  const toggleActiveMutation = useMutation({
   222→    mutationFn: ({ id, isActive }: { id: string; isActive: boolean }) =>
   223→      patchApi(`/api/backups/schedules/${id}`, { isActive }),
   224→    onSuccess: () => {
   225→      queryClient.invalidateQueries({ queryKey: queryKeys.backupSchedules.all });
   226→    },
   227→  });
   228→
   229→  const openCreate = () => {
   230→    setEditingId(null);
   231→    setForm(initialForm);
   232→    setDialogOpen(true);
   233→  };
   234→
   235→  const openEdit = (row: ScheduleRow) => {
   236→    setEditingId(row.id);
   237→    setForm({
   238→      name: row.name,
   239→      description: row.description || '',
   240→      frequency: row.frequency,
   241→      cronExpression: row.cronExpression || '',
   242→      scope: row.scope,
   243→      storageId: row.storageId || '',
   244→      storageProvider: row.storageProvider,
   245→      encryptionEnabled: row.encryptionEnabled,
   246→      verificationEnabled: row.verificationEnabled,
   247→      retentionCount: row.retentionCount,
   248→    });
   249→    setDialogOpen(true);
   250→  };
   251→
   252→  const closeDialog = () => {
   253→    setDialogOpen(false);
   254→    setEditingId(null);
   255→    setForm(initialForm);
   256→  };
   257→
   258→  const handleSubmit = () => {
   259→    if (!form.name.trim()) return;
   260→    // Submit storageProvider (the API uses this); storageId is UI-only.
   261→    const { storageId: _omitStorageId, ...body } = form;
   262→    void _omitStorageId;
   263→    if (editingId) {
   264→      updateMutation.mutate({ id: editingId, body });
   265→    } else {
   266→      createMutation.mutate(body);
   267→    }
   268→  };
   269→
   270→  const updateForm = <K extends keyof ScheduleForm>(key: K, value: ScheduleForm[K]) => {
   271→    setForm((prev) => ({ ...prev, [key]: value }));
   272→  };
   273→
   274→  const isSaving = createMutation.isPending || updateMutation.isPending;
   275→
   276→  // Column min-widths are intentionally generous and the table uses
   277→  // `table-layout: auto` (NOT table-fixed) so cells NEVER truncate or clip —
   278→  // every value (Name, Frequency, Scope, Storage, Encrypt, Verify, Retention,
   279→  // Active, Next Run, Last Run, Actions) renders in full. Long schedule names
   280→  // simply make the table wider and trigger horizontal scroll inside the
   281→  // card. Mirrors the Backups table's reusable overflow/scroll pattern.
   282→  // Sum of min-widths: 220+110+130+130+90+90+110+80+150+150+60 = 1320px.
   283→  const columns: ColumnDef<ScheduleRow>[] = [
   284→      {
   285→        id: 'name',
   286→        header: 'Name',
   287→        accessorKey: 'name',
   288→        size: 220,
   289→        // No `truncate`/`overflow-hidden` — full schedule name always visible.
   290→        // `title` provides a hover tooltip for very long names without ever
   291→        // clipping the cell text.
   292→        cell: ({ getValue }) => {
   293→          const value = (getValue() as string) ?? '';
   294→          if (!value) return <span className="text-muted-foreground">—</span>;
   295→          return <span className="font-medium" title={value}>{value}</span>;
   296→        },
   297→      },
   298→      {
   299→        id: 'frequency',
   300→        header: 'Frequency',
   301→        accessorKey: 'frequency',
   302→        enableSorting: false,
   303→        size: 110,
   304→        cell: ({ row }) => {
   305→          const freq = row.original.frequency;
   306→          const cron = row.original.cronExpression;
   307→          return (
   308→            <span className="text-sm">
   309→              {labelize(freq)}
   310→              {freq === 'CUSTOM_CRON' && cron && (
   311→                <span className="block text-xs text-muted-foreground font-mono">{cron}</span>
   312→              )}
   313→            </span>
   314→          );
   315→        },
   316→      },
   317→      {
   318→        id: 'scope',
   319→        header: 'Scope',
   320→        accessorKey: 'scope',
   321→        enableSorting: false,
   322→        size: 130,
   323→        cell: ({ getValue }) => (
   324→          <Badge variant="outline" className="border-transparent bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400 font-medium">
   325→            {labelize(getValue() as string)}
   326→          </Badge>
   327→        ),
   328→      },
   329→      {
   330→        id: 'storage',
   331→        header: 'Storage',
   332→        accessorKey: 'storageProvider',
   333→        enableSorting: false,
   334→        size: 130,
   335→        // Full storage label visible (e.g. "Cloudflare R2", "Google Drive");
   336→        // never truncated.
   337→        cell: ({ getValue }) => (
   338→          <span className="text-xs text-muted-foreground" title={labelize(getValue() as string)}>
   339→            {labelize(getValue() as string)}
   340→          </span>
   341→        ),
   342→      },
   343→      {
   344→        id: 'encryptionEnabled',
   345→        header: 'Encrypt',
   346→        accessorKey: 'encryptionEnabled',
   347→        enableSorting: false,
   348→        size: 90,
   349→        cell: ({ getValue }) => (
   350→          <span className={cn(
   351→            'text-xs font-medium',
   352→            getValue() ? 'text-green-600 dark:text-green-400' : 'text-muted-foreground'
   353→          )}>
   354→            {getValue() ? 'Yes' : 'No'}
   355→          </span>
   356→        ),
   357→      },
   358→      {
   359→        id: 'verificationEnabled',
   360→        header: 'Verify',
   361→        accessorKey: 'verificationEnabled',
   362→        enableSorting: false,
   363→        size: 90,
   364→        cell: ({ getValue }) => (
   365→          <span className={cn(
   366→            'text-xs font-medium',
   367→            getValue() ? 'text-green-600 dark:text-green-400' : 'text-muted-foreground'
   368→          )}>
   369→            {getValue() ? 'Yes' : 'No'}
   370→          </span>
   371→        ),
   372→      },
   373→      {
   374→        id: 'retentionCount',
   375→        header: 'Retention',
   376→        accessorKey: 'retentionCount',
   377→        enableSorting: false,
   378→        size: 110,
   379→        cell: ({ getValue }) => (
   380→          <span className="tabular-nums text-sm">{getValue() as number} days</span>
   381→        ),
   382→      },
   383→      {
   384→        id: 'isActive',
   385→        header: 'Active',
   386→        accessorKey: 'isActive',
   387→        enableSorting: false,
   388→        size: 80,
   389→        cell: ({ row }) => (
   390→          <Switch
   391→            checked={row.original.isActive}
   392→            onCheckedChange={(checked) =>
   393→              toggleActiveMutation.mutate({ id: row.original.id, isActive: checked })
   394→            }
   395→          />
   396→        ),
   397→      },
   398→      {
   399→        id: 'nextRunAt',
   400→        header: 'Next Run',
   401→        accessorKey: 'nextRunAt',
   402→        size: 150,
   403→        // Full relative timestamp visible (e.g. "Tomorrow at 2:00 AM"); never
   404→        // truncated. The 150px min-width + auto layout guarantees realistic
   405→        // values fit; if a value is somehow wider the table scrolls.
   406→        cell: ({ getValue }) => {
   407→          const val = getValue() as string | null;
   408→          return (
   409→            <span className="text-xs text-muted-foreground" title={val ? formatRelativeTime(val) : undefined}>
   410→              {val ? formatRelativeTime(val) : '—'}
   411→            </span>
   412→          );
   413→        },
   414→      },
   415→      {
   416→        id: 'lastRunAt',
   417→        header: 'Last Run',
   418→        accessorKey: 'lastRunAt',
   419→        size: 150,
   420→        cell: ({ getValue }) => {
   421→          const val = getValue() as string | null;
   422→          return (
   423→            <span className="text-xs text-muted-foreground" title={val ? formatRelativeTime(val) : undefined}>
   424→              {val ? formatRelativeTime(val) : '—'}
   425→            </span>
   426→          );
   427→        },
   428→      },
   429→      ColumnDefHelper.actionColumn<ScheduleRow>({
   430→        id: 'actions',
   431→        size: 60,
   432→        render: (row) => (
   433→          <DropdownMenu>
   434→            <DropdownMenuTrigger asChild>
   435→              <Button variant="ghost" size="icon" className="h-8 w-8" onClick={(e) => e.stopPropagation()}>
   436→                <MoreHorizontal className="h-4 w-4" />
   437→              </Button>
   438→            </DropdownMenuTrigger>
   439→            <DropdownMenuContent align="end">
   440→              <DropdownMenuItem onClick={() => openEdit(row)}>
   441→                <Pencil className="h-4 w-4 mr-2" />
   442→                Edit
   443→              </DropdownMenuItem>
   444→              <DropdownMenuSeparator />
   445→              <DropdownMenuItem variant="destructive" onClick={() => setDeleteTarget(row)}>
   446→                <Trash2 className="h-4 w-4 mr-2" />
   447→                Delete
   448→              </DropdownMenuItem>
   449→            </DropdownMenuContent>
   450→          </DropdownMenu>
   451→        ),
   452→      }),
   453→    ];
   454→
   455→  return (
   456→    <div className="space-y-4">
   457→      <PageHeader
   458→        breadcrumbs={false}
   459→        title="Backup Schedules"
   460→        description="Configure automated backup schedules"
   461→        action={
   462→          <Button size="sm" onClick={openCreate}>
   463→            <Plus className="h-4 w-4 mr-2" />
   464→            Create Schedule
   465→          </Button>
   466→        }
   467→      />
   468→
   469→      {isInitialEmpty ? (
   470→        <EmptyState
   471→          icon={CalendarClock}
   472→          title="No schedules configured"
   473→          description="Create a schedule to automate your backups."
   474→          action={{ label: 'Create Schedule', onClick: openCreate }}
   475→        />
   476→      ) : (
   477→        <DataTable
   478→          columns={columns}
   479→          data={schedules}
   480→          isLoading={isLoading}
   481→          totalItems={pagination?.total ?? 0}
   482→          pageSize={table.pageSize}
   483→          currentPage={table.currentPage}
   484→          onPageChange={(p) => table.setCurrentPage(p)}
   485→          onSortChange={(f, o) => table.setSortField(f, o)}
   486→          sortField={table.sortField}
   487→          sortOrder={table.sortOrder}
   488→          searchPlaceholder="Search schedules..."
   489→          searchValue={table.searchValue}
   490→          onSearch={(v) => {
   491→            table.setSearchValue(v);
   492→            table.setCurrentPage(1);
   493→          }}
   494→          getRowId={(row) => row.id}
   495→          // Auto layout (NOT table-fixed): cells never truncate/clip — every
   496→          // value renders in full. The table's min-width (1320px = sum of
   497→          // column min-widths: Name 220 + Frequency 110 + Scope 130 +
   498→          // Storage 130 + Encrypt 90 + Verify 90 + Retention 110 + Active 80
   499→          // + Next Run 150 + Last Run 150 + Actions 60) guarantees the table
   500→          // is only as narrow as its content needs; on viewports narrower
   501→          // than that, the table-container's overflow-x-auto provides a thin,
   502→          // subtle horizontal scrollbar that stays inside the card. The
   503→          // pagination/footer lives outside the scroll area and always
   504→          // aligns with the card edges. Same reusable pattern as the Backups
   505→          // table for consistency.
   506→          tableMinWidth={1320}
   507→          emptyMessage="No schedules found."
   508→          emptyState={
   509→            isSearchEmpty ? (
   510→              <NoSchedulesSearchResultsEmpty
   511→                onClear={() => {
   512→                  table.setSearchValue('');
   513→                  table.setCurrentPage(1);
   514→                }}
   515→              />
   516→            ) : undefined
   517→          }
   518→        />
   519→      )}
   520→
   521→      {/* Create/Edit Dialog */}
   522→      <Dialog open={dialogOpen} onOpenChange={(v) => !v && closeDialog()}>
   523→        <DialogContent className="sm:max-w-[520px]">
   524→          <DialogHeader>
   525→            <DialogTitle>{editingId ? 'Edit Schedule' : 'Create Schedule'}</DialogTitle>
   526→            <DialogDescription>
   527→              {editingId
   528→                ? 'Update backup schedule configuration.'
   529→                : 'Configure a new automated backup schedule.'}
   530→            </DialogDescription>
   531→          </DialogHeader>
   532→          <div className="space-y-4 py-2 max-h-[60vh] overflow-y-auto pr-1">
   533→            <div className="space-y-2">
   534→              <Label htmlFor="sched-name">Name</Label>
   535→              <Input
   536→                id="sched-name"
   537→                placeholder="e.g., Daily Database Backup"
   538→                value={form.name}
   539→                onChange={(e) => updateForm('name', e.target.value)}
   540→              />
   541→            </div>
   542→            <div className="space-y-2">
   543→              <Label htmlFor="sched-desc">Description</Label>
   544→              <Textarea
   545→                id="sched-desc"
   546→                placeholder="Optional description..."
   547→                value={form.description}
   548→                onChange={(e) => updateForm('description', e.target.value)}
   549→                rows={2}
   550→              />
   551→            </div>
   552→            <div className="space-y-2">
   553→              <Label htmlFor="sched-freq">Frequency</Label>
   554→              <Select
   555→                value={form.frequency}
   556→                onValueChange={(v) => updateForm('frequency', v as BackupScheduleFrequency)}
   557→              >
   558→                <SelectTrigger id="sched-freq">
   559→                  <SelectValue />
   560→                </SelectTrigger>
   561→                <SelectContent>
   562→                  {FREQUENCIES.map((f) => (
   563→                    <SelectItem key={f.value} value={f.value}>
   564→                      {f.label}
   565→                    </SelectItem>
   566→                  ))}
   567→                </SelectContent>
   568→              </Select>
   569→            </div>
   570→            {form.frequency === 'CUSTOM_CRON' && (
   571→              <div className="space-y-2">
   572→                <Label htmlFor="sched-cron">Custom Cron Expression</Label>
   573→                <Input
   574→                  id="sched-cron"
   575→                  placeholder="0 2 * * *"
   576→                  value={form.cronExpression}
   577→                  onChange={(e) => updateForm('cronExpression', e.target.value)}
   578→                  className="font-mono"
   579→                />
   580→                <p className="text-xs text-muted-foreground">
   581→                  Use standard 5-field cron format (minute hour day month weekday)
   582→                </p>
   583→              </div>
   584→            )}
   585→            <div className="grid grid-cols-2 gap-4">
   586→              <div className="space-y-2">
   587→                <Label htmlFor="sched-scope">Scope</Label>
   588→                <Select
   589→                  value={form.scope}
   590→                  onValueChange={(v) => updateForm('scope', v as BackupScope)}
   591→                >

   593→                    <SelectValue />

   597→                      <SelectItem key={opt.value} value={opt.value}>{opt.label}</SelectItem>

   599→                  </SelectContent>

   601→              </div>

   605→                  value={form.storageId}

   609→                    if (dest) updateForm('storageProvider', dest.provider as BackupStorageProvider);

   613→                    <SelectValue placeholder="Select a storage destination" />

   624→                  </SelectContent>

   626→                {storageDestinations.length === 0 && (

   635→                type="number"
   636→                min={1}
   637→                max={365}
   638→                value={form.retentionCount}
   639→                onChange={(e) => updateForm('retentionCount', Number(e.target.value))}
   640→              />
   641→              <p className="text-xs text-muted-foreground">Number of backups to keep</p>

   643→            <div className="flex items-center justify-between">

   647→              </div>
   648→              <Switch
   649→                id="sched-encrypt"
   650→                checked={form.encryptionEnabled}
   651→                onCheckedChange={(v) => updateForm('encryptionEnabled', v)}
   652→              />
   653→            </div>
   654→            <div className="flex items-center justify-between">
   655→              <div>

   659→              <Switch
   660→                id="sched-verify"
   661→                checked={form.verificationEnabled}
   662→                onCheckedChange={(v) => updateForm('verificationEnabled', v)}
   663→              />
   664→            </div>
   665→          </div>
   666→          <DialogFooter>
   667→            <Button variant="outline" onClick={closeDialog}>
   668→              Cancel
   669→            </Button>
   670→            <Button
   671→              onClick={handleSubmit}
   672→              disabled={isSaving || !form.name.trim()}
   673→            >
   674→              {isSaving && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   675→              {editingId ? 'Update' : 'Create'}
   676→            </Button>
   677→          </DialogFooter>
   678→        </DialogContent>
   679→      </Dialog>
   680→
   681→      {/* Delete Confirmation */}
   682→      <ConfirmDialog
   683→        open={!!deleteTarget}
   684→        onOpenChange={(v) => !v && setDeleteTarget(null)}
   685→        title="Delete Schedule"
   686→        description={
   687→          deleteTarget
   688→            ? `Are you sure you want to delete "${deleteTarget.name}"? Automated backups associated with this schedule will no longer run.`
   689→            : undefined
   690→        }
   691→        confirmLabel="Delete"
   692→        variant="destructive"
   693→        onConfirm={() => {
   694→          if (deleteTarget) deleteMutation.mutate(deleteTarget.id);
   695→        }}
   696→        isLoading={deleteMutation.isPending}
   697→      />
   698→    </div>
   699→  );
