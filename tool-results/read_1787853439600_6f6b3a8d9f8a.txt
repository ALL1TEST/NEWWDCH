   109→export interface UseDataTableReturn {

   118→  setSortField: (field: string, order: 'asc' | 'desc') => void;
   119→  setSelectedIds: (ids: string[]) => void;
   120→  setSearchValue: (value: string) => void;
   121→  toggleSort: (field: string) => void;
   122→  toggleRowSelection: (id: string) => void;
   123→  toggleAllSelection: (allIds: string[], currentDataIds: string[]) => void;
   124→  clearSelection: () => void;
   125→}
   126→
   127→export function useDataTable(options: UseDataTableOptions = {}): UseDataTableReturn {
   128→  const {
   129→    initialPage = 1,
   130→    initialPageSize = DEFAULT_PAGE_SIZE,
   131→    initialSortField = 'createdAt',
   132→    initialSortOrder = 'desc',
   133→  } = options;
   134→
   135→  const [currentPage, setCurrentPage] = useState(initialPage);
   136→  const [pageSize, setPageSize] = useState(initialPageSize);
   137→  const [sortField, setSortFieldRaw] = useState(initialSortField);
   138→  const [sortOrder, setSortOrderRaw] = useState<'asc' | 'desc'>(initialSortOrder);
   139→  const [selectedIds, setSelectedIds] = useState<string[]>([]);
   140→  const [searchValue, setSearchValue] = useState('');
   141→
   142→  const setSortField = useCallback((field: string, order: 'asc' | 'desc') => {
   143→    setSortFieldRaw(field);
   144→    setSortOrderRaw(order);
   145→    setCurrentPage(1);
   146→  }, []);
   147→
   148→  const toggleSort = useCallback(
   149→    (field: string) => {
   150→      if (sortField === field) {
   151→        const next = sortOrder === 'asc' ? 'desc' : 'asc';
   152→        setSortFieldRaw(field);
   153→        setSortOrderRaw(next);
   154→      } else {
   155→        setSortFieldRaw(field);
   156→        setSortOrderRaw('asc');
   157→      }
   158→      setCurrentPage(1);
   159→    },
   160→    [sortField, sortOrder],
   161→  );
   162→
   163→  const toggleRowSelection = useCallback((id: string) => {
   164→    setSelectedIds((prev) =>
   165→      prev.includes(id) ? prev.filter((i) => i !== id) : [...prev, id],
   166→    );
   167→  }, []);
   168→
   169→  const toggleAllSelection = useCallback(
   170→    (allIds: string[], currentDataIds: string[]) => {
   171→      const allSelected = currentDataIds.every((id) => selectedIds.includes(id));
   172→      if (allSelected) {
   173→        setSelectedIds((prev) =>
   174→          prev.filter((id) => !currentDataIds.includes(id)),
   175→        );
   176→      } else {
   177→        const newIds = currentDataIds.filter((id) => !selectedIds.includes(id));
   178→        setSelectedIds((prev) => [...prev, ...newIds]);
   179→      }
   180→    },
   181→    [selectedIds],
   182→  );
   183→
   184→  const clearSelection = useCallback(() => {
   185→    setSelectedIds([]);
   186→  }, []);
   187→
   188→  return {
   189→    currentPage,
   190→    pageSize,
   191→    sortField,
   192→    sortOrder,
   193→    selectedIds,
   194→    searchValue,
   195→    setCurrentPage,
   196→    setPageSize,
   197→    setSortField,
   198→    setSelectedIds,
   199→    setSearchValue,
   200→    toggleSort,
   201→    toggleRowSelection,
   202→    toggleAllSelection,
   203→    clearSelection,
   204→  };
   205→}

   245→export class ColumnDefHelper {
   246→  static textColumn<TData>(options: TextColumnOptions<TData>): ColumnDef<TData> {
   247→    return {
   248→      id: options.id,
   249→      accessorKey: options.accessorKey,
   250→      accessorFn: options.accessorFn,
   251→      header: ({ column }) => {
   252→        const canSort = options.enableSorting !== false;
   253→        if (!canSort) {
   254→          return <span className="font-medium">{options.header}</span>;
   255→        }
   256→        return (
   257→          <button
   258→            className="flex items-center gap-1 hover:text-foreground transition-colors"
   259→            onClick={() => column.toggleSorting(column.getIsSorted() === 'asc')}
   260→          >
   261→            <span className="font-medium">{options.header}</span>
   262→            <SortIcon sorted={column.getIsSorted()} />
   263→          </button>
   264→        );
   265→      },
   266→      size: options.size,
   267→      enableSorting: options.enableSorting !== false,
   268→      enableHiding: options.enableHiding,
   269→      cell: ({ getValue }) => {
   270→        const value = getValue() as string;
   271→        if (!value) return <span className="text-muted-foreground">—</span>;
   272→        if (options.truncate && value.length > options.truncate) {
   273→          return (
   274→            <span className={cn('truncate block max-w-full', options.className)} title={value}>
   275→              {value.slice(0, options.truncate)}...
   276→            </span>
   277→          );
   278→        }
   279→        return <span className={options.className}>{value}</span>;
   280→      },
   281→    };
   282→  }

   361→  static avatarColumn<TData>(options: AvatarColumnOptions<TData>): ColumnDef<TData> {
   362→    return {
   363→      id: options.id,
   364→      accessorKey: options.accessorKey,
   365→      accessorFn: options.accessorFn as any,
   366→      header: ({ column }) => {
   367→        const canSort = options.enableSorting !== false;
   368→        if (!canSort) {
   369→          return <span className="font-medium">{options.header}</span>;
   370→        }
   371→        return (
   372→          <button
   373→            className="flex items-center gap-1 hover:text-foreground transition-colors"
   374→            onClick={() => column.toggleSorting(column.getIsSorted() === 'asc')}
   375→          >
   376→            <span className="font-medium">{options.header}</span>
   377→            <SortIcon sorted={column.getIsSorted()} />
   378→          </button>
   379→        );
   380→      },
   381→      size: (options.size ?? 200) as number,
   382→      enableSorting: options.enableSorting !== false,
   383→      enableHiding: options.enableHiding,
   384→      cell: ({ row }) => {
   385→        const rowData = row.original;
   386→        const src = options.getSrc?.(rowData);
   387→        const name = options.getName(rowData);
   388→        const subtitle = options.getSubtitle?.(rowData);
   389→        const sizeClasses = {
   390→          sm: 'h-8 w-8 text-xs',
   391→          md: 'h-10 w-10 text-sm',
   392→          lg: 'h-12 w-12 text-base',
   393→        };
   394→        const avatarSize = sizeClasses[options.size ?? 'md'];
   395→        const initials = name
   396→          .split(/\s+/)
   397→          .map((w) => w[0])
   398→          .slice(0, 2)
   399→          .join('')
   400→          .toUpperCase();
   401→
   402→        return (
   403→          <div className="flex items-center gap-3">
   404→            {src ? (
   405→              <img
   406→                src={src}
   407→                alt={name}
   408→                className={cn('rounded-full object-cover shrink-0', avatarSize)}
   409→              />
   410→            ) : (
   411→              <div
   412→                className={cn(
   413→                  'rounded-full bg-muted flex items-center justify-center shrink-0 font-medium text-muted-foreground',
   414→                  avatarSize,
   415→                )}
   416→              >
   417→                {initials}
   418→              </div>
   419→            )}
   420→            <div className="min-w-0 flex-1">
   421→              <div className="truncate font-medium text-sm">{name}</div>
   422→              {subtitle && (
   423→                <div className="truncate text-xs text-muted-foreground">{subtitle}</div>
   424→              )}
   425→            </div>
   426→          </div>
   427→        );
   428→      },
   429→    };
   430→  }

   507→// -------------------- Pagination --------------------
   508→
   509→function DataTablePagination({
   510→  currentPage,
   511→  pageSize,
   512→  totalItems,
   513→  onPageChange,
   514→  onPageSizeChange,
   515→}: {
   516→  currentPage: number;
   517→  pageSize: number;
   518→  totalItems: number;
   519→  onPageChange: (page: number) => void;
   520→  onPageSizeChange: (size: number) => void;
   521→}) {
   522→  const totalPages = Math.max(1, Math.ceil(totalItems / pageSize));
   523→  const rangeStart = totalItems === 0 ? 0 : (currentPage - 1) * pageSize + 1;
   524→  const rangeEnd = Math.min(currentPage * pageSize, totalItems);
   525→
   526→  return (
   527→    <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between px-4 py-3 border-t">
   528→      <div className="text-xs text-muted-foreground">
   529→        Showing {rangeStart}–{rangeEnd} of {totalItems} items
   530→      </div>
   531→      <div className="flex items-center gap-3">
   532→        <div className="flex items-center gap-1.5">
   533→          <span className="text-xs text-muted-foreground whitespace-nowrap">Rows</span>
   534→          <Select
   535→            value={String(pageSize)}
   536→            onValueChange={(v) => onPageSizeChange(Number(v))}
   537→          >
   538→            <SelectTrigger size="sm" className="w-[70px] h-8">
   539→              <SelectValue />
   540→            </SelectTrigger>
   541→            <SelectContent>
   542→              {PAGE_SIZES.map((size) => (
   543→                <SelectItem key={size} value={String(size)}>
   544→                  {size}
   545→                </SelectItem>
   546→              ))}
   547→            </SelectContent>
   548→          </Select>
   549→        </div>
   550→        <div className="flex items-center gap-1">
   551→          <Button
   552→            variant="outline"
   553→            size="icon"
   554→            className="h-8 w-8"
   555→            onClick={() => onPageChange(1)}
   556→            disabled={currentPage <= 1}
   557→          >
   558→            <ChevronsLeft className="h-4 w-4" />
   559→            <span className="sr-only">First page</span>
   560→          </Button>
   561→          <Button
   562→            variant="outline"
   563→            size="icon"
   564→            className="h-8 w-8"
   565→            onClick={() => onPageChange(currentPage - 1)}
   566→            disabled={currentPage <= 1}
   567→          >
   568→            <ChevronLeft className="h-4 w-4" />
   569→            <span className="sr-only">Previous page</span>
   570→          </Button>
   571→          <span className="text-xs text-muted-foreground min-w-[60px] text-center">
   572→            {currentPage} / {totalPages}
   573→          </span>
   574→          <Button
   575→            variant="outline"
   576→            size="icon"
   577→            className="h-8 w-8"
   578→            onClick={() => onPageChange(currentPage + 1)}
   579→            disabled={currentPage >= totalPages}
   580→          >
   581→            <ChevronRight className="h-4 w-4" />
   582→            <span className="sr-only">Next page</span>
   583→          </Button>
   584→          <Button
   585→            variant="outline"
   586→            size="icon"
   587→            className="h-8 w-8"
   588→            onClick={() => onPageChange(totalPages)}
   589→            disabled={currentPage >= totalPages}
   590→          >
   591→            <ChevronsRight className="h-4 w-4" />
   592→            <span className="sr-only">Last page</span>
   593→          </Button>
   594→        </div>
   595→      </div>
   596→    </div>
   597→  );
   598→}

   616→  bulkActions?: BulkAction[];
   617→  onBulkAction?: (action: string, ids: string[]) => void;
   618→  onSelectionChange?: (ids: string[]) => void;
   619→  filterContent?: React.ReactNode;
   620→}) {
   621→  const hasSelection = selectedIds && selectedIds.length > 0;
   622→
   623→  return (
   624→    <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between p-4 pb-2">
   625→      <div className="flex items-center gap-2 flex-1">
   626→        {(onSearch || searchValue !== undefined) && (
   627→          <div className="relative max-w-sm w-full">
   628→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   629→            <Input
   630→              placeholder={searchPlaceholder ?? 'Search...'}
   631→              value={searchValue ?? ''}
   632→              onChange={(e) => onSearch?.(e.target.value)}
   633→              className="pl-9 h-9"
   634→            />
   635→          </div>
   636→        )}
   637→        {filterContent}
   638→      </div>
   639→      {hasSelection && bulkActions && bulkActions.length > 0 && (
   640→        <div className="flex items-center gap-2">
   641→          <span className="text-xs text-muted-foreground">
   642→            {selectedIds.length} selected
   643→          </span>
   644→          <DropdownMenu>
   645→            <DropdownMenuTrigger asChild>
   646→              <Button variant="outline" size="sm">
   647→                <MoreHorizontal className="h-4 w-4 mr-1" />
   648→                Bulk Actions
   649→              </Button>
   650→            </DropdownMenuTrigger>
   651→            <DropdownMenuContent align="end">
   652→              {bulkActions.map((action) => (
   653→                <DropdownMenuItem
   654→                  key={action.label}
   655→                  variant={action.variant}
   656→                  disabled={action.disabled}
   657→                  onClick={() => onBulkAction?.(action.label, selectedIds)}
   658→                >
   659→                  {action.icon}
   660→                  {action.label}
   661→                </DropdownMenuItem>
   662→              ))}
   663→            </DropdownMenuContent>
   664→          </DropdownMenu>
   665→          <Button
   666→            variant="ghost"
   667→            size="sm"
   668→            onClick={() => onSelectionChange?.([])}
   669→          >
   670→            Clear
   671→          </Button>
   672→        </div>
   673→      )}
   674→    </div>
   675→  );
   676→}

   678→// -------------------- Main DataTable Component --------------------
   679→
   680→export function DataTable<TData, TValue>({
   681→  columns,
   682→  data,
   683→  isLoading = false,
   684→  totalItems,
   685→  pageSize,
   686→  currentPage,
   687→  onPageChange,
   688→  onSortChange,
   689→  sortField,
   690→  sortOrder,
   691→  onRowClick,
   692→  selectedIds = [],
   693→  onSelectionChange,
   694→  onBulkAction,
   695→  bulkActions,
   696→  emptyMessage = 'No data found.',
   697→  emptyIcon,
   698→  emptyState,
   699→  searchPlaceholder,
   700→  onSearch,
   701→  searchValue,
   702→  filterContent,
   703→  getRowId,
   704→  onPageSizeChange,
   705→  tableFixed = false,
   706→  tableMinWidth,
   707→}: DataTableProps<TData, TValue>) {
   708→  const hasRowSelection = !!onSelectionChange;
   709→  const columnCount = columns.length + (hasRowSelection ? 1 : 0);
   710→
   711→  const enhancedColumns = useMemo<ColumnDef<TData, TValue>[]>(() => {
   712→    if (!hasRowSelection) return columns;
   713→
   714→    const selectColumn: ColumnDef<TData, TValue> = {
   715→      id: '__select__',
   716→      size: 40,
   717→      enableSorting: false,
   718→      enableHiding: false,
   719→      header: ({ table }) => {
   720→        const allRowIds = table.getRowModel().rows.map((r) => r.id);
   721→        const allChecked =
   722→          allRowIds.length > 0 && allRowIds.every((id) => selectedIds.includes(id));
   723→        const someChecked = allRowIds.some((id) => selectedIds.includes(id));
   724→
   725→        return (
   726→          <Checkbox
   727→            checked={allChecked ? true : someChecked ? 'indeterminate' : false}
   728→            onCheckedChange={(checked) => {
   729→              if (checked) {
   730→                const allIds = data
   731→                  .map((row, idx) => getRowId?.(row) ?? String(idx))
   732→                  .filter((id) => !selectedIds.includes(id));
   733→                onSelectionChange?.([...selectedIds, ...allIds]);
   734→              } else {
   735→                const currentDataIds = data.map((row, idx) =>
   736→                  getRowId?.(row) ?? String(idx),
   737→                );
   738→                onSelectionChange?.(
   739→                  selectedIds.filter((id) => !currentDataIds.includes(id)),
   740→                );
   741→              }
   742→            }}
   743→            aria-label="Select all rows"
   744→          />
   745→        );
   746→      },
   747→      cell: ({ row }) => {
   748→        const rowId = row.id;
   749→        const isChecked = selectedIds.includes(rowId);
   750→
   751→        return (
   752→          <Checkbox
   753→            checked={isChecked}
   754→            onCheckedChange={(checked) => {
   755→              if (checked) {
   756→                onSelectionChange?.([...selectedIds, rowId]);
   757→              } else {
   758→                onSelectionChange?.(selectedIds.filter((id) => id !== rowId));
   759→              }
   760→            }}
   761→            onClick={(e) => e.stopPropagation()}
   762→            aria-label={`Select row ${rowId}`}
   763→          />
   764→        );
   765→      },
   766→    };
   767→
   768→    return [selectColumn, ...columns];
   769→  }, [columns, hasRowSelection, selectedIds, onSelectionChange, data, getRowId]);
   770→
   771→  const sorting: ColumnSort[] = useMemo(() => {
   772→    if (!sortField) return [];
   773→    return [{ id: sortField, desc: sortOrder === 'desc' }];
   774→  }, [sortField, sortOrder]);
   775→
   776→  const table = useReactTable({
   777→    data,
   778→    columns: enhancedColumns,
   779→    getCoreRowModel: getCoreRowModel(),
   780→    getSortedRowModel: getSortedRowModel(),
   781→    state: {
   782→      sorting,
   783→    },
   784→    onSortingChange: (updater) => {
   785→      const newSorting =
   786→        typeof updater === 'function' ? updater(sorting) : updater;
   787→      if (newSorting.length > 0) {
   788→        onSortChange(newSorting[0].id, newSorting[0].desc ? 'desc' : 'asc');
   789→      }
   790→    },
   791→    manualPagination: true,
   792→    manualSorting: true,
   793→    getRowId: getRowId
   794→      ? (row, index) => getRowId(row) ?? String(index)
   795→      : (_, index) => String(index),
   796→  });
   797→
   798→  const handlePageSizeChange = useCallback(
   799→    (newSize: number) => {
   800→      onPageSizeChange?.(newSize);
   801→      onPageChange(1);
   802→    },
   803→    [onPageChange, onPageSizeChange],
   804→  );
   805→
   806→  const showToolbar =
   807→    (onSearch || searchValue !== undefined) ||
   808→    (filterContent != null) ||
   809→    (selectedIds.length > 0 && bulkActions && bulkActions.length > 0);
   810→
   811→  // Only enforce a min-width when the table actually has rows. When empty
   812→  // (no data / search-empty / initial-empty), we let the table collapse to
   813→  // the card width so the empty state renders cleanly without forcing a
   814→  // horizontal scrollbar.
   815→  const tableStyle =
   816→    tableMinWidth && data.length > 0
   817→      ? { minWidth: `${tableMinWidth}px` }
   818→      : undefined;
   819→  const tableClassName = cn(tableFixed && 'table-fixed');
   820→
   821→  return (
   822→    <div className="rounded-lg border overflow-hidden">
   823→      {showToolbar && (
   824→        <TableToolbar
   825→          searchPlaceholder={searchPlaceholder}
   826→          searchValue={searchValue}
   827→          onSearch={onSearch}
   828→          selectedIds={selectedIds}
   829→          bulkActions={bulkActions}
   830→          onBulkAction={onBulkAction}
   831→          onSelectionChange={onSelectionChange}
   832→          filterContent={filterContent}
   833→        />
   834→      )}
   835→      <Table className={tableClassName} style={tableStyle}>
   836→        <TableHeader>
   837→          {table.getHeaderGroups().map((headerGroup) => (
   838→            <TableRow key={headerGroup.id}>
   839→              {headerGroup.headers.map((header) => (
   840→                <TableHead
   841→                  key={header.id}
   842→                  style={{ width: header.getSize() !== 150 ? header.getSize() : undefined }}
   843→                >
   844→                  {flexRender(
   845→                    header.column.columnDef.header,
   846→                    header.getContext(),
   847→                  )}
   848→                </TableHead>
   849→              ))}
   850→            </TableRow>
   851→          ))}
   852→        </TableHeader>
   853→        <TableBody>
   854→          {isLoading ? (
   855→            <SkeletonRows columnCount={columnCount} />
   856→          ) : data.length === 0 ? (
   857→            <DataTableEmpty icon={emptyIcon} message={emptyMessage} state={emptyState} />
   858→          ) : (
   859→            table.getRowModel().rows.map((row) => (
   860→              <TableRow
   861→                key={row.id}
   862→                data-state={selectedIds.includes(row.id) ? 'selected' : undefined}
   863→                className={cn(onRowClick && 'cursor-pointer')}
   864→                onClick={() => onRowClick?.(row.original)}
   865→              >
   866→                {row.getVisibleCells().map((cell) => (
   867→                  <TableCell key={cell.id}>
   868→                    {flexRender(
   869→                      cell.column.columnDef.cell,
   870→                      cell.getContext(),
   871→                    )}
   872→                  </TableCell>
   873→                ))}
   874→              </TableRow>
   875→            ))
   876→          )}
   877→        </TableBody>
   878→      </Table>
   879→      <DataTablePagination
   880→        currentPage={currentPage}
   881→        pageSize={pageSize}
   882→        totalItems={totalItems}
   883→        onPageChange={onPageChange}
   884→        onPageSizeChange={handlePageSizeChange}
   885→      />
   886→    </div>
   887→  );
   888→}
