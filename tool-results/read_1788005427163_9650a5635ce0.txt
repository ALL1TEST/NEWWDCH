    18→    18→//  - recentPayments = payments sorted by date desc

   142→   142→// -------------------- Plans (delegated to DB-backed plan-config) --------------------
   143→   143→// The editable plan definitions live in the PlanConfig table and are cached
   144→   144→// by ./plan-config — the SINGLE source of truth. The owner edits there via
   145→   145→// the Platform Admin; this module (admin overview + client billing) and the
   146→   146→// client billing API both read the same cached values, so an owner price
   147→   147→// change propagates to the client billing page and to MRR on the next read.
   148→   148→// The PLANS array below is a live view re-spliced whenever the config changes.
   149→   149→
   150→   150→import { getPlanConfigsSync, subscribe as subscribePlanConfig, type PlanConfigData } from './plan-config';
   151→   151→
   152→   152→function toPlan(d: PlanConfigData): Plan {
   153→   153→  return {
   154→   154→    id: d.planId as PlanId,
   155→   155→    name: d.name,
   156→   156→    price: d.priceMonthly,
   157→   157→    priceMonthly: d.priceMonthly,
   158→   158→    priceYearly: d.priceYearly,
   159→   159→    currency: d.currency,
   160→   160→    interval: d.interval,
   161→   161→    isFree: d.isFree,
   162→   162→    active: d.active,
   163→   163→    features: d.features,
   164→   164→    entitlements: d.entitlements,
   165→   165→    limits: d.limits,
   166→   166→  };
   167→   167→}
   168→   168→
   169→   169→/** Live view of the DB-backed plan configs. Mutated in place on every
   170→   170→ *  config refresh so existing references stay valid. */
   171→   171→export const PLANS: Plan[] = [];
   172→   172→
   173→   173→function refreshPLANS(): void {
   174→   174→  const data = getPlanConfigsSync();
   175→   175→  PLANS.length = 0;
   176→   176→  for (const d of data) PLANS.push(toPlan(d));
   177→   177→}
   178→   178→refreshPLANS();
   179→   179→subscribePlanConfig(refreshPLANS);
   180→   180→
   181→   181→export function getPlan(id: PlanId): Plan {
   182→   182→  return PLANS.find((p) => p.id === id) ?? PLANS[0];
   183→   183→}

   185→   185→/** Normalized monthly price. Yearly subscriptions contribute priceYearly/12
   186→   186→ *  so MRR is a true monthly run-rate. */
   187→   187→export function monthlyPrice(plan: Plan, interval: BillingInterval): number {
   188→   188→  if (interval === 'yearly') return Math.round(plan.priceYearly / 12);
   189→   189→  return plan.priceMonthly;
   190→   190→}

   198→   198→  const d = new Date();
   199→   199→  d.setDate(d.getDate() - days);
   200→   200→  d.setHours(9, 15, 0, 0);
   201→   201→  return d.toISOString();
   202→   202→}

   206→   206→  d.setHours(9, 15, 0, 0);
   207→   207→  return d.toISOString();
   208→   208→}
