     4→// PLATFORM PLANS & PRICING — simplified admin UI.
     5→// ============================================================
     6→// Owner sees compact plan cards (Beta / Pro / Max …) with a quick
     7→// Active toggle, a compact price + features + limits summary,
     8→// and an "Edit Plan" button. The Edit Plan button opens a Dialog
     9→// with three compact sections: Basic Information, Feature Access,
    10→// Usage Limits. An optional collapsed "Client Display" section
    11→// holds the marketing feature list shown to clients on the
    12→// Client Billing page. A "+ Create Plan" button at the top-right
    13→// opens a Create Plan dialog (same sections, blank defaults) that
    14→// POSTs to /api/platform/admin/plans.
    15→//
    16→// All writes go through /api/platform/admin/plans (GET list,
    17→// POST create) + /api/platform/admin/plans/[planId] (PUT update)
    18→// — owner-only — the same shared plan-config cache the Client
    19→// Billing page and MRR read. Server-side entitlement enforcement
    20→// (hasFeature) and usage-limit checks (checkLimit) consume the
    21→// same data, so toggling a feature off here denies it for clients
    22→// on the next request. The Client Billing page is NOT modified
    23→// and continues to read from /api/platform/billing/me.
    24→// ============================================================

    91→/** Resolve the badge id for the shared PlanBadge component. Falls
    92→ *  back to 'beta' for any unknown plan id so future plans still
    93→ *  render with a sensible neutral badge. */
    94→function getPlanBadgeId(planId: string): 'beta' | 'pro' | 'max' {
    95→  if (planId === 'beta' || planId === 'pro' || planId === 'max') return planId;
    96→  return 'beta';
    97→}
    98→
    99→/** Format a single limit value for the compact card summary. -1 →
   100→ *  "Unlimited"; storageBytes → human-readable (1 GB); other limits
   101→ *  use locale-formatted numbers. */
   102→function formatLimitValue(key: LimitKey, value: number): string {
   103→  if (value === UNLIMITED) return 'Unlimited';
   104→  if (key === 'storageBytes') return formatBytes(value);
   105→  return value.toLocaleString();
   106→}
   107→
   108→/** One-line compact limits summary used on the plan card. */
   109→function summarizeLimits(limits: PlanLimits): string {
   110→  return [
   111→    `${formatLimitValue('maxSites', limits.maxSites)} sites`,
   112→    formatLimitValue('storageBytes', limits.storageBytes),
   113→    `${formatLimitValue('aiWords', limits.aiWords)} AI words`,
   114→    `${formatLimitValue('aiArticles', limits.aiArticles)} AI articles`,
   115→    `${formatLimitValue('automationRuns', limits.automationRuns)} automation runs`,
   116→  ].join(' · ');
   117→}
   118→
   119→/** One-line compact entitlement summary used on the plan card. */
   120→function summarizeEntitlements(entitlements: string[]): string {
   121→  if (entitlements.length === 0) return 'No advanced features';
   122→  return entitlements
   123→    .map((key) => ENTITLEMENT_LABELS[key as EntitlementKey] ?? key)
   124→    .join(' · ');
   125→}

   140→function PlanSummaryCard({
   141→  plan,
   142→  canEdit,
   143→}: {
   144→  plan: PlanConfigData;
   145→  canEdit: boolean;
   146→}) {
   147→  const queryClient = useQueryClient();
   148→  const [editing, setEditing] = useState(false);
   149→
   150→  // Quick active toggle — auto-saves immediately (owner-only).
   151→  // Reuses the same PUT endpoint and the same invalidation chain
   152→  // as the full Edit modal so clients see the new status on the
   153→  // next read.
   154→  const activeMutation = useMutation({
   155→    mutationFn: (active: boolean) =>
   156→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, { active }),
   157→    onSuccess: (data) => {
   158→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   159→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   160→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   161→      toast.success(
   162→        `${data.name} ${data.active ? 'activated' : 'deactivated'} — changes are live for clients.`,
   163→      );
   164→    },
   165→    onError: (err: unknown) => {
   166→      const e = err as { error?: { message?: string }; message?: string };
   167→      toast.error(
   168→        e?.error?.message ?? e?.message ?? 'Unable to change plan status. OWNER access required.',
   169→      );
   170→    },
   171→  });
   172→
   173→  return (
   174→    <>
   175→      <Card className={plan.active ? '' : 'opacity-70'}>
   176→        <CardContent className="p-4 space-y-4">
   177→          {/* Header: name + badge + quick active toggle */}
   178→          <div className="flex items-center justify-between gap-2">
   179→            <div className="flex items-center gap-2 min-w-0">
   180→              <h3 className="text-base font-semibold tracking-tight truncate">{plan.name}</h3>
   181→              <PlanBadge planId={getPlanBadgeId(plan.planId)} />
   182→            </div>
   183→            <div className="flex items-center gap-2 shrink-0">
   184→              <Label
   185→                htmlFor={`active-${plan.planId}`}
   186→                className="text-xs text-muted-foreground cursor-pointer"
   187→              >
   188→                {plan.active ? 'Active' : 'Inactive'}
   189→              </Label>
   190→              <Switch
   191→                id={`active-${plan.planId}`}
   192→                checked={plan.active}
   193→                disabled={!canEdit || activeMutation.isPending}
   194→                onCheckedChange={(v) => activeMutation.mutate(v)}
   195→              />
   196→            </div>
   197→          </div>
   198→
   199→          {/* Price */}
   200→          <div className="space-y-0.5">
   201→            <p className="text-[10px] font-medium text-muted-foreground uppercase tracking-wider">
   202→              Price
   203→            </p>
   204→            {plan.isFree ? (
   205→              <p className="text-2xl font-bold tracking-tight">Free</p>
   206→            ) : (
   207→              <div className="flex items-baseline gap-2">
   208→                <span className="text-2xl font-bold tracking-tight">
   209→                  {formatCurrency(plan.priceMonthly, plan.currency)}
   210→                </span>
   211→                <span className="text-xs text-muted-foreground">/ month</span>
   212→              </div>
   213→            )}
   214→            {!plan.isFree && plan.priceYearly > 0 && (
   215→              <p className="text-xs text-muted-foreground">
   216→                {formatCurrency(plan.priceYearly, plan.currency)} / year
   217→              </p>
   218→            )}
   219→          </div>
   220→
   221→          {/* Billing interval */}
   222→          <div className="flex items-center justify-between text-xs">
   223→            <span className="text-muted-foreground">Billing</span>
   224→            <span className="font-medium capitalize">{plan.interval}</span>
   225→          </div>
   226→
   227→          {/* Features */}
   228→          <div className="space-y-1">
   229→            <p className="text-[10px] font-medium text-muted-foreground uppercase tracking-wider">
   230→              Features
   231→            </p>
   232→            <p className="text-sm leading-snug">{summarizeEntitlements(plan.entitlements)}</p>
   233→          </div>
   234→
   235→          {/* Limits */}
   236→          <div className="space-y-1">
   237→            <p className="text-[10px] font-medium text-muted-foreground uppercase tracking-wider">
   238→              Limits
   239→            </p>
   240→            <p className="text-sm leading-snug text-foreground/90">
   241→              {summarizeLimits(plan.limits)}
   242→            </p>
   243→          </div>
   244→
   245→          <Separator />
   246→
   247→          {/* Edit button */}
   248→          <Button
   249→            variant="outline"
   250→            size="sm"
   251→            className="w-full"
   252→            disabled={!canEdit}
   253→            onClick={() => setEditing(true)}
   254→          >
   255→            <Pencil className="h-4 w-4 mr-2" />
   256→            Edit Plan
   257→          </Button>
   258→          {!canEdit && (
   259→            <p className="text-[10px] text-muted-foreground text-center -mt-2">
   260→              Editing is restricted to the OWNER role.
   261→            </p>
   262→          )}
   263→        </CardContent>
   264→      </Card>
   265→
   266→      {editing && (
   267→        <EditPlanDialog plan={plan} open={editing} onOpenChange={setEditing} />
   268→      )}
   269→    </>
   270→  );
   271→}

   275→function EditPlanDialog({
   276→  plan,
   277→  open,
   278→  onOpenChange,
   279→}: {
   280→  plan: PlanConfigData;
   281→  open: boolean;
   282→  onOpenChange: (open: boolean) => void;
   283→}) {
   284→  const queryClient = useQueryClient();
   285→  const [name, setName] = useState(plan.name);
   286→  const [priceMonthly, setPriceMonthly] = useState(String(plan.priceMonthly));
   287→  const [priceYearly, setPriceYearly] = useState(String(plan.priceYearly));
   288→  const [currency, setCurrency] = useState(plan.currency);
   289→  const [interval, setInterval] = useState<'monthly' | 'yearly'>(plan.interval);
   290→  const [active, setActive] = useState(plan.active);
   291→  const [features, setFeatures] = useState(plan.features.join('\n'));
   292→  const [entitlements, setEntitlements] = useState<string[]>(plan.entitlements);
   293→  const [limits, setLimits] = useState<PlanLimits>(plan.limits);
   294→  const [clientDisplayOpen, setClientDisplayOpen] = useState(false);
   295→
   296→  // EditPlanDialog is conditionally rendered by the parent (mounted
   297→  // fresh each time the user opens it), so the useState initializers
   298→  // above already seed local state from the latest server snapshot.
   299→  // No useEffect sync is needed — that would just trigger cascading
   300→  // renders (and the React Compiler correctly flags it).
   301→
   302→  const reset = () => {
   303→    setName(plan.name);
   304→    setPriceMonthly(String(plan.priceMonthly));
   305→    setPriceYearly(String(plan.priceYearly));
   306→    setCurrency(plan.currency);
   307→    setInterval(plan.interval);
   308→    setActive(plan.active);
   309→    setFeatures(plan.features.join('\n'));
   310→    setEntitlements(plan.entitlements);
   311→    setLimits(plan.limits);
   312→  };
   313→
   314→  const saveMutation = useMutation({
   315→    mutationFn: (patch: PlanPatch) =>
   316→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, patch),
   317→    onSuccess: (data) => {
   318→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   319→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   320→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   321→      toast.success(`${data.name} updated — changes are now live for clients.`);
   322→      onOpenChange(false);
   323→    },
   324→    onError: (err: unknown) => {
   325→      const e = err as { error?: { message?: string }; message?: string };
   326→      toast.error(
   327→        e?.error?.message ?? e?.message ?? 'Unable to save plan. You may need OWNER access.',
   328→      );
   329→    },
   330→  });

   333→    name,
   334→    priceMonthly: Number(priceMonthly) || 0,
   335→    priceYearly: Number(priceYearly) || 0,
   336→    currency,
   337→    interval,
   338→    active,
   339→    features: features
   340→      .split('\n')
   341→      .map((f) => f.trim())
   342→      .filter(Boolean),
   343→    entitlements,
   344→    limits: { ...limits, storageBytes: Number(limits.storageBytes) || 0 },
   345→  });
   346→
   347→  return (
   348→    <Dialog open={open} onOpenChange={onOpenChange}>
   349→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   350→        <DialogHeader>
   351→          <div className="flex items-center gap-2">
   352→            <DialogTitle className="text-base">Edit {plan.name}</DialogTitle>
   353→            <PlanBadge planId={getPlanBadgeId(plan.planId)} />
   354→          </div>
   355→          <DialogDescription className="text-xs">
   356→            Single source of truth — changes propagate to the Client Billing page, MRR and
   357→            server-side entitlement enforcement.
   358→          </DialogDescription>
   359→        </DialogHeader>
   360→
   361→        <div className="space-y-5">
   362→          {/* -------------------- Basic Information -------------------- */}
   363→          <section className="space-y-3">
   364→            <h4 className="text-sm font-semibold">Basic Information</h4>
   365→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   366→              <div className="space-y-1">
   367→                <Label className="text-xs">Name</Label>
   368→                <Input
   369→                  value={name}
   370→                  onChange={(e) => setName(e.target.value)}
   371→                  className="h-9"
   372→                />
   373→              </div>
   374→              <div className="space-y-1">
   375→                <Label className="text-xs">Monthly Price</Label>
   376→                <Input
   377→                  type="number"
   378→                  value={priceMonthly}
   379→                  onChange={(e) => setPriceMonthly(e.target.value)}
   380→                  className="h-9"
   381→                />
   382→              </div>
   383→              <div className="space-y-1">
   384→                <Label className="text-xs">Yearly Price</Label>
   385→                <Input

   579→/** Derive a planId from a name: lowercase, hyphenated, ASCII-only.
   580→ *  "Enterprise Pro" → "enterprise-pro". Used as a sensible default
   581→ *  the owner can override. */
   582→function derivePlanId(name: string): string {
   583→  return name
   584→    .toLowerCase()
   585→    .trim()
   586→    .replace(/[^a-z0-9]+/g, '-')
   587→    .replace(/^-+|-+$/g, '');
   588→}
