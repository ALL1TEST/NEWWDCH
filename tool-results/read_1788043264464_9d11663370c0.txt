     1→// ============================================================
     2→// POST /api/webhooks/stripe — Stripe webhook handler.
     3→// ============================================================
     4→// Stripe is the source of truth for paid subscription state. This route
     5→// receives Stripe events, verifies the signature using
     6→// STRIPE_WEBHOOK_SECRET, and updates the Subscription + Payment tables.
     7→//
     8→// Handled events:
     9→//   - checkout.session.completed        → activate the subscription
    10→//     (mark active, set currentPeriodEnd from Stripe's subscription
    11→//     period end, record the Stripe customer + subscription IDs, write
    12→//     a FULLY-relational Payment row: Invoice ID, PaymentIntent ID,
    13→//     Charge ID, + payment-method metadata). Also best-effort copies the
    14→//     session-level discount snapshot onto the just-created Payment row
    15→//     (the invoice.paid handler will overwrite it with the authoritative
    16→//     value).
    17→//   - customer.subscription.updated      → refresh status + currentPeriodEnd.
    18→//   - customer.subscription.created      → refresh status + currentPeriodEnd.
    19→//   - customer.subscription.deleted      → cancel the subscription
    20→//     (preserve the row for audit; future access reverts to Free).
    21→//   - invoice.paid                      → record a new Payment row with
    22→//     the full Stripe relational + payment-method metadata, AND record
    23→//     the coupon applied (CouponRedemption audit row + Coupon.timesRedeemed
    24→//     increment + Payment.couponCode/discountAmount snapshot).
    25→//   - invoice.payment_failed            → mark the subscription past_due
    26→//     + record a failed Payment row with the failure reason.
    27→//   - payment_intent.succeeded           → record a one-time / orphan
    28→//     Payment (deduped by stripePaymentIntentId so it never duplicates
    29→//     an invoice.paid / checkout row for the same PI).
    30→//   - payment_intent.payment_failed      → record a failed one-time
    31→//     Payment with the failure reason.
    32→//   - charge.refunded                    → mark the existing Payment row
    33→//     as 'refunded' (full) or annotate 'Partial refund processed: ...'
    34→//     (partial). Deduped by stripeChargeId, falling back to PI id.
    35→//
    36→// Every Payment row written here carries: userId (Customer), planId
    37→// (Plan), subscriptionId (Subscription), stripeInvoiceId (Invoice),
    38→// stripePaymentIntentId + stripeChargeId (Stripe transaction), and
    39→// paymentMethodType + paymentMethodDetails (payment-method metadata) —
    40→// so the admin Payments table, the customer detail view, and the
    41→// Dashboard always reflect the real Stripe state. Payments reads these
    42→// rows; Plans & Pricing reads PlanConfig; the webhook keeps them in
    43→// sync with Stripe.
    44→//
    45→// IDEMPOTENCY: Stripe retries events up to 3 days. Before dispatching
    46→// any handler, this route inserts a row into the WebhookEvent ledger
    47→// (unique on stripeEventId). If the insert returns false (duplicate),
    48→// the event is ack'd and skipped — never re-processed. If a handler
    49→// throws after the ledger insert succeeds, the ledger row is updated
    50→// to outcome='error' + errorMessage (best-effort). Stripe will retry.
    51→//
    52→// When Stripe is NOT configured (no STRIPE_SECRET_KEY), this route
    53→// returns 503 — no events are processed. The platform's free-plan flow
    54→// does not require this webhook.
    55→// ============================================================
    56→
    57→import { NextRequest, NextResponse } from 'next/server';
    58→import { db } from '@/lib/db';
    59→import { activateSubscriptionFromStripe, cancelSubscription } from '@/lib/platform/subscription-data';
    60→import {
    61→  isStripeConfigured,
    62→  verifyStripeWebhook,
    63→  getStripeClient,
    64→  markWebhookEventProcessed,
    65→  extractInvoiceCoupon,
    66→} from '@/lib/stripe';
    67→import type Stripe from 'stripe';
    68→
    69→export async function POST(request: NextRequest) {
    70→  if (!isStripeConfigured()) {
    71→    return NextResponse.json(
    72→      { error: { code: 'PAYMENT_PROVIDER_NOT_CONFIGURED', message: 'Stripe is not configured.' } },
    73→      { status: 503 },
    74→    );
    75→  }
    76→
    77→  const signature = request.headers.get('stripe-signature') ?? '';
    78→  const rawBody = await request.text();
    79→
    80→  const event = await verifyStripeWebhook(rawBody, signature);
    81→  if (!event) {
    82→    return NextResponse.json(
    83→      { error: { code: 'INVALID_SIGNATURE', message: 'Webhook signature verification failed.' } },
    84→      { status: 400 },
    85→    );
    86→  }
    87→
    88→  // -------------------- Idempotency ledger --------------------
    89→  // Stripe retries events up to 3 days; we MUST NOT process the same
    90→  // evt_... twice. Insert a WebhookEvent row (unique on stripeEventId)
    91→  // BEFORE dispatching the handler. If the insert returns false, the
    92→  // event was already processed → ack and skip. The ledger insert
    93→  // happens BEFORE processing so a crash mid-processing doesn't cause
    94→  // a re-process on retry.
    95→  const stripeEventId = event.id;
    96→  const eventType = event.type;
    97→  const apiVersion = event.api_version ?? null;
    98→  const objectId =
    99→    (event.data.object as unknown as { id?: string | null } | null)?.id ?? null;
   100→
   101→  const isNewEvent = await markWebhookEventProcessed({
   102→    stripeEventId,
   103→    eventType,
   104→    apiVersion,
   105→    objectId,
   106→    outcome: 'processed',
   107→  });
   108→  if (!isNewEvent) {
   109→    // Duplicate event — already processed on a previous delivery.
   110→    return NextResponse.json({ received: true, type: event.type, duplicate: true });
   111→  }
   112→
   113→  try {
   114→    switch (event.type) {
   115→      case 'checkout.session.completed': {
   116→        await handleCheckoutCompleted(event);
   117→        break;
   118→      }
   119→      case 'customer.subscription.updated':
   120→      case 'customer.subscription.created': {
   121→        await handleSubscriptionUpdated(event);
   122→        break;
   123→      }
   124→      case 'customer.subscription.deleted': {
   125→        await handleSubscriptionDeleted(event);
   126→        break;
   127→      }
   128→      case 'invoice.paid': {
   129→        await handleInvoicePaid(event);
   130→        break;
   131→      }
   132→      case 'invoice.payment_failed': {
   133→        await handleInvoiceFailed(event);
   134→        break;
   135→      }
   136→      case 'payment_intent.succeeded': {
   137→        await handlePaymentIntentSucceeded(event);
   138→        break;
   139→      }
   140→      case 'payment_intent.payment_failed': {
   141→        await handlePaymentIntentFailed(event);
   142→        break;
   143→      }
   144→      case 'charge.refunded': {
   145→        await handleChargeRefunded(event);
   146→        break;
   147→      }
   148→      default:
   149→        // Unhandled event types are silently ignored (no DB changes).
   150→        break;
   151→    }
   152→    return NextResponse.json({ received: true, type: event.type });
   153→  } catch (err) {
   154→    // Don't leak the error message to the client — Stripe will retry.
   155→    console.error('Stripe webhook handler error:', err);
   156→    // Best-effort: mark the ledger row as errored so we can see it in
   157→    // the admin ledger view. Never mask the original error — if the
   158→    // update itself fails, log it and continue.
   159→    try {
   160→      const errorMessage = err instanceof Error ? err.message : String(err);
   161→      await db.webhookEvent.update({
   162→        where: { stripeEventId },
   163→        data: {
   164→          outcome: 'error',
   165→          errorMessage: errorMessage.slice(0, 1000),
   166→        },
   167→      });
   168→    } catch (ledgerErr) {
   169→      console.error('Stripe webhook: failed to mark ledger row as errored:', ledgerErr);
   170→    }
   171→    return NextResponse.json(
   172→      { error: { code: 'WEBHOOK_HANDLER_ERROR', message: 'Internal handler error.' } },
   173→      { status: 500 },
   174→    );
   175→  }
   176→}

   178→// -------------------- Payment-method metadata fetcher --------------------
   179→
   180→/** The normalized payment-method details extracted from a Stripe Charge's
   181→ *  `payment_method_details`. Stored as JSON on the Payment row so the admin
   182→ *  Payments table can show "Visa ••4242" / brand / last4 / exp / funding /
   183→ *  country — the real payment-method metadata, never fabricated. */
   184→interface PaymentMethodMeta {
   185→  stripeChargeId: string | null;
   186→  paymentMethodType: string | null;
   187→  paymentMethodDetails: string | null; // JSON
   188→  method: string | null; // display label, e.g. "Visa ••4242"
   189→}
   190→
   191→/** Fetch the latest Charge on a PaymentIntent and extract the
   192→ *  payment-method metadata. Returns null when the PI has no charge yet
   193→ *  (e.g. some pending states). Never throws — the caller treats null as
   194→ *  "no metadata available" and still records the payment. */
   195→async function fetchPaymentMethodMeta(
   196→  stripe: Stripe,
   197→  paymentIntentId: string | null | undefined,
   198→): Promise<PaymentMethodMeta | null> {
   199→  if (!paymentIntentId) return null;
   200→  try {
   201→    const pi = (await stripe.paymentIntents.retrieve(paymentIntentId, {
   202→      // Expand the latest charge so we can read payment_method_details
   203→      // without a second round-trip. Cast because the dahlia API version
   204→      // type doesn't expose `latest_charge` on PaymentIntent directly,
   205→      // but the API response includes it.
   206→      expand: ['latest_charge' as string],
   207→    })) as unknown as {
   208→      latest_charge?:
   209→        | string
   210→        | {
   211→            id: string;
   212→            payment_method_details?: {
   213→              type: string;
   214→              card?: {
   215→                brand?: string | null;
   216→                last4?: string | null;
   217→                exp_month?: number | null;
   218→                exp_year?: number | null;
   219→                funding?: string | null;
   220→                country?: string | null;
   221→              } | null;
   222→            } | null;
   223→          };
   224→    };
   225→
   226→    const charge = pi.latest_charge;
   227→    if (!charge) return null;
   228→
   229→    const stripeChargeId = typeof charge === 'string' ? charge : charge.id;
   230→    const pmd = typeof charge === 'string' ? null : charge.payment_method_details ?? null;
   231→
   232→    if (!pmd) {
   233→      return { stripeChargeId, paymentMethodType: null, paymentMethodDetails: null, method: null };
   234→    }
   235→
   236→    if (pmd.type === 'card' && pmd.card) {
   237→      const c = pmd.card;
   238→      const details = {
   239→        brand: c.brand ?? null,
   240→        last4: c.last4 ?? null,
   241→        expMonth: c.exp_month ?? null,
   242→        expYear: c.exp_year ?? null,
   243→        funding: c.funding ?? null,
   244→        country: c.country ?? null,
   245→      };
   246→      const method =
   247→        c.brand && c.last4 ? `${cap(c.brand)} ••${c.last4}` : c.brand ? cap(c.brand) : null;
   248→      return {
   249→        stripeChargeId,
   250→        paymentMethodType: 'card',
   251→        paymentMethodDetails: JSON.stringify(details),
   252→        method,
   253→      };
   254→    }
   255→
   256→    // Non-card payment methods (ach_debit, sepa_debit, link, ...): record
   257→    // the type; no brand/last4 to extract.
   258→    return {
   259→      stripeChargeId,
   260→      paymentMethodType: pmd.type,
   261→      paymentMethodDetails: null,
   262→      method: cap(pmd.type),
   263→    };
   264→  } catch {
   265→    // Retrieval failed (PI not found, network error, ...). Don't block
   266→    // the payment row creation — return null so the caller still writes
   267→    // the core fields (invoice / PI / amount).
   268→    return null;
   269→  }
   270→}
   271→
   272→function cap(s: string): string {
   273→  return s.charAt(0).toUpperCase() + s.slice(1);
   274→}

   276→// -------------------- Dedup helper --------------------
   277→
   278→/** Find an existing Payment row by Stripe Invoice ID OR PaymentIntent ID.
   279→ *  Stripe may fire `invoice.paid` + `payment_intent.succeeded` +
   280→ *  `checkout.session.completed` for the same underlying payment — dedup
   281→ *  by these IDs so a single real-world payment never creates duplicate
   282→ *  rows. Returns the existing row or null. */
   283→async function findExistingPayment(stripeInvoiceId?: string | null, stripePaymentIntentId?: string | null) {
   284→  if (stripeInvoiceId) {
   285→    const byInvoice = await db.payment.findFirst({ where: { stripeInvoiceId } });
   286→    if (byInvoice) return byInvoice;
   287→  }
   288→  if (stripePaymentIntentId) {
   289→    const byPi = await db.payment.findFirst({ where: { stripePaymentIntentId } });
   290→    if (byPi) return byPi;
   291→  }
   292→  return null;
   293→}

   295→// -------------------- Coupon helpers --------------------
   296→
   297→/** Best-effort: extract the discount applied to a Checkout Session's
   298→ *  `total_details.discount`. Returns { couponCode, discountAmount } or null
   299→ *  when the session had no discount.
   300→ *
   301→ *  The Stripe API exposes the discount as a Discount object whose `coupon`
   302→ *  carries the Stripe Coupon id (e.g. "kUW0yO9r") — NOT the customer-facing
   303→ *  code. We try to resolve the local customer-facing code from the Coupon
   304→ *  table by stripeCouponId; if not found, we fall back to the Stripe coupon
   305→ *  id. The authoritative coupon snapshot is set by the invoice.paid handler
   306→ *  (which has the richer invoice-level discount) — this is a best-effort
   307→ *  pre-population that gets overwritten. */
   308→async function extractSessionDiscount(
   309→  session: Stripe.Checkout.Session,
   310→): Promise<{ couponCode: string | null; discountAmount: number | null } | null> {
   311→  try {
   312→    const totalDetails = session.total_details as unknown as {
   313→      discount?: {
   314→        coupon?: {
   315→          id?: string;
   316→          name?: string | null;
   317→          percent_off?: number | null;
   318→          amount_off?: number | null;
   319→          currency?: string | null;
   320→        } | null;
   321→      } | null;
   322→    } | null;
   323→    const coupon = totalDetails?.discount?.coupon;
   324→    if (!coupon?.id) return null;
   325→
   326→    // Try to resolve the local customer-facing coupon code from the
   327→    // Coupon table by stripeCouponId. (stripeCouponId is not unique in
   328→    // the local schema — it's an index — so we use findFirst.)
   329→    let couponCode: string | null = coupon.id;
   330→    try {
   331→      const local = await db.coupon.findFirst({
   332→        where: { stripeCouponId: coupon.id },
   333→        select: { code: true },
   334→      });
   335→      if (local?.code) couponCode = local.code;
   336→    } catch {
   337→      // ignore — fall back to Stripe coupon id (best-effort).
   338→    }
   339→
   340→    return {
   341→      couponCode,
   342→      discountAmount: coupon.amount_off ?? null,
   343→    };
   344→  } catch {
   345→    return null;
   346→  }
   347→}

   349→/** Record the coupon applied to an invoice onto the Payment + the local
   350→ *  Coupon/CouponRedemption tables. Called from `handleInvoicePaid` AFTER
   351→ *  the Payment row is created (or, when the payment row already exists
   352→ *  from a previous event, against that existing row id).
   353→ *
   354→ *  Steps (all best-effort, never throws — the core payment-row creation
   355→ *  is already done by the time this runs):
   356→ *    1. extractInvoiceCoupon(invoice) → { couponCode, stripeCouponId, discountAmount }
   357→ *    2. db.payment.update → set couponCode + discountAmount on the Payment row.
   358→ *    3. db.coupon.findUnique({ where: { code } }) → resolve the local Coupon.
   359→ *    4. Idempotency check: only increment timesRedeemed if no CouponRedemption
   360→ *       row exists for this (couponId, userId, stripeEventId) tuple.
   361→ *    5. db.coupon.update → increment timesRedeemed (idempotent).
   362→ *    6. db.couponRedemption.create → audit row. Wrap in try/catch —
   363→ *       unique-violation on stripeEventId means a parallel event already
   364→ *       created the row; skip silently. */
   365→async function recordInvoiceCoupon(
   366→  event: Stripe.Event,
   367→  invoice: Stripe.Invoice,
   368→  paymentId: string,
   369→  userId: string,
   370→): Promise<void> {
   371→  try {
   372→    const couponInfo = extractInvoiceCoupon(invoice);
   373→    if (!couponInfo || !couponInfo.couponCode) return;
   374→
   375→    // 1. Set couponCode + discountAmount on the Payment row.
   376→    try {
   377→      await db.payment.update({
   378→        where: { id: paymentId },
   379→        data: {
   380→          couponCode: couponInfo.couponCode,
   381→          discountAmount: couponInfo.discountAmount ?? null,
   382→        },
   383→      });
   384→    } catch (e) {
   385→      console.error('Stripe webhook: failed to set coupon snapshot on Payment row:', e);
   386→    }
   387→
   388→    // 2. Resolve the local Coupon row by code.
   389→    let coupon: { id: string } | null = null;
   390→    try {
   391→      coupon = await db.coupon.findUnique({
   392→        where: { code: couponInfo.couponCode },
   393→        select: { id: true },
   394→      });
   395→    } catch (e) {
   396→      console.error('Stripe webhook: failed to look up local Coupon by code:', e);
   397→    }
   398→    if (!coupon) return; // no local Coupon to increment — best-effort, skip.
   399→
   400→    // 3. Idempotency: only increment timesRedeemed if no CouponRedemption
   401→    //    row exists for this (couponId, userId, stripeEventId) tuple.
   402→    const stripeEventId = event.id;
   403→    let alreadyRedeemed = false;
   404→    try {
   405→      const existingRedemption = await db.couponRedemption.findFirst({
   406→        where: { couponId: coupon.id, userId, stripeEventId },
   407→        select: { id: true },
   408→      });
   409→      alreadyRedeemed = !!existingRedemption;
   410→    } catch (e) {
   411→      console.error('Stripe webhook: failed to check existing CouponRedemption:', e);
   412→    }
   413→
   414→    if (!alreadyRedeemed) {
   415→      try {
   416→        await db.coupon.update({
   417→          where: { id: coupon.id },
   418→          data: { timesRedeemed: { increment: 1 } },
   419→        });
   420→      } catch (e) {
   421→        console.error('Stripe webhook: failed to increment Coupon.timesRedeemed:', e);
   422→      }
   423→    }
   424→
   425→    // 4. Create the CouponRedemption audit row. Unique-violation on
   426→    //    stripeEventId (a parallel event already created the row) is
   427→    //    swallowed — log and skip. Never block the payment row.
   428→    try {
   429→      await db.couponRedemption.create({
   430→        data: {
   431→          couponId: coupon.id,
   432→          userId,
   433→          stripeEventId,
   434→          stripeInvoiceId: invoice.id,
   435→          stripeSubscriptionId:
   436→            typeof invoice.subscription === 'string' ? invoice.subscription : null,
   437→          amountOff: couponInfo.discountAmount ?? 0,
   438→          currency: invoice.currency ?? 'usd',
   439→        },
   440→      });
   441→    } catch (e) {
   442→      console.error(
   443→        'Stripe webhook: CouponRedemption insert failed (likely duplicate event — swallowed):',
   444→        e,
   445→      );
   446→    }
   447→  } catch (e) {
   448→    // Best-effort — never let coupon recording break the webhook.
   449→    console.error('Stripe webhook: coupon recording failed (best-effort, swallowed):', e);
   450→  }
   451→}

   453→// -------------------- Event handlers --------------------
   454→
   455→async function handleCheckoutCompleted(event: Stripe.Event) {
   456→  const session = event.data.object as Stripe.Checkout.Session;
   457→  const userId = session.metadata?.userId ?? session.client_reference_id ?? null;
   458→  if (!userId) return;
   459→
   460→  const planId = session.metadata?.planId ?? null;
   461→  const interval = session.metadata?.interval === 'yearly' ? 'yearly' : 'monthly';
   462→  if (!planId) return;
   463→
   464→  // Look up the Stripe subscription to get the real period end + IDs.
   465→  const stripe = getStripeClient();
   466→  let stripeSub: Stripe.Subscription | null = null;
   467→  if (session.subscription) {
   468→    stripeSub = await stripe.subscriptions.retrieve(session.subscription as string);
   469→  }
   470→
   471→  // The Stripe SDK 2026-08-26.dahlia type doesn't expose `current_period_end`
   472→  // on Subscription directly, but the API response still includes it. We cast
   473→  // to access it safely.
   474→  const subAny = stripeSub as unknown as { current_period_end?: number; current_period_start?: number } | null;
   475→  const periodEnd = subAny?.current_period_end ? new Date(subAny.current_period_end * 1000) : new Date();
   476→  const customerId =
   477→    typeof session.customer === 'string'
   478→      ? session.customer
   479→      : session.customer && typeof session.customer === 'object' && 'id' in session.customer
   480→      ? session.customer.id
   481→      : '';
   482→
   483→  // Fetch the payment-method metadata from the session's PaymentIntent so
   484→  // the recorded Payment row carries the full relational + card details.
   485→  const piId = typeof session.payment_intent === 'string' ? session.payment_intent : null;
   486→  const meta = await fetchPaymentMethodMeta(stripe, piId);
   487→
   488→  // Build the subscription activation payload.
   489→  await activateSubscriptionFromStripe({
   490→    userId,
   491→    planId,
   492→    billingInterval: interval,
   493→    stripeCustomerId: customerId,
   494→    stripeSubscriptionId: stripeSub?.id ?? (typeof session.subscription === 'string' ? session.subscription : '') ?? '',
   495→    stripeCurrentPeriodEnd: periodEnd,
   496→    status: stripeSub?.status === 'trialing' ? 'trial' : 'active',
   497→    payment:
   498→      session.amount_total && session.amount_total > 0
   499→        ? {
   500→            amount: session.amount_total,
   501→            currency: session.currency ?? 'usd',
   502→            stripeInvoiceId: typeof session.invoice === 'string' ? session.invoice : undefined,
   503→            stripePaymentIntentId: piId ?? undefined,
   504→            stripeChargeId: meta?.stripeChargeId ?? undefined,
   505→            paymentMethodType: meta?.paymentMethodType ?? undefined,
   506→            paymentMethodDetails: meta?.paymentMethodDetails ?? undefined,
   507→            method: meta?.method ?? 'Stripe',
   508→          }
   509→        : undefined,
   510→  });
   511→
   512→  // Best-effort: pass the session-level discount snapshot along to the
   513→  // just-created Payment row. We do NOT write a CouponRedemption row
   514→  // here — the coupon snapshot lives on the invoice that fires
   515→  // `invoice.paid`, and that handler will set the authoritative value
   516→  // (which overwrites whatever we set here).
   517→  if (session.amount_total && session.amount_total > 0) {
   518→    try {
   519→      const sessionDiscount = await extractSessionDiscount(session);
   520→      if (sessionDiscount && sessionDiscount.couponCode) {
   521→        // Find the just-created Payment row by invoice / PI id.
   522→        const justCreated = await findExistingPayment(
   523→          typeof session.invoice === 'string' ? session.invoice : null,
   524→          piId,
   525→        );
   526→        if (justCreated) {
   527→          await db.payment.update({
   528→            where: { id: justCreated.id },
   529→            data: {
   530→              couponCode: sessionDiscount.couponCode,
   531→              discountAmount: sessionDiscount.discountAmount ?? null,
   532→            },
   533→          });
   534→        }
   535→      }
   536→    } catch (e) {
   537→      console.error(
   538→        'Stripe webhook: failed to set session-discount snapshot on Payment row (best-effort):',
   539→        e,
   540→      );
   541→    }
   542→  }
   543→}

   545→async function handleSubscriptionUpdated(event: Stripe.Event) {
   546→  const sub = event.data.object as Stripe.Subscription;
   547→  const userId = sub.metadata?.userId;
   548→  if (!userId) return;
   549→
   550→  // Look up the existing subscription row by userId.
   551→  const existing = await db.subscription.findUnique({ where: { userId } });
   552→  if (!existing) return;
   553→
   554→  // Compute the effective status.
   555→  const status =
   556→    sub.status === 'trialing'
   557→      ? 'trial'
   558→      : sub.status === 'active'
   559→      ? 'active'
   560→      : sub.status === 'past_due' || sub.status === 'unpaid'
   561→      ? 'past_due'
   562→      : sub.status === 'canceled'
   563→      ? 'cancelled'
   564→      : 'active';
   565→
   566→  // The Stripe SDK 2026-08-26.dahlia type doesn't expose current_period_end
   567→  // on Subscription directly; cast to access it safely.
   568→  const subAny = sub as unknown as { current_period_end?: number };
   569→  const customerId =
   570→    typeof sub.customer === 'string'
   571→      ? sub.customer
   572→      : sub.customer && typeof sub.customer === 'object' && 'id' in sub.customer
   573→      ? sub.customer.id
   574→      : existing.stripeCustomerId;
   575→
   576→  await db.subscription.update({
   577→    where: { userId },
   578→    data: {
   579→      status,
   580→      currentPeriodEnd: subAny.current_period_end ? new Date(subAny.current_period_end * 1000) : null,
   581→      stripeSubscriptionId: sub.id,
   582→      stripeCustomerId: customerId,
   583→      trialEnd: sub.trial_end ? new Date(sub.trial_end * 1000) : null,
   584→    },
   585→  });
   586→}

   595→async function handleInvoicePaid(event: Stripe.Event) {
   596→  const invoice = event.data.object as Stripe.Invoice;
   597→  const userId = invoice.metadata?.userId;
   598→  if (!userId) return;
   599→  const planId = invoice.metadata?.planId;
   600→  if (!planId) return;
   601→
   602→  // PaymentIntent id on the invoice (may be a string or null for some
   603→  // invoice types, e.g. pre-payment).
   604→  const piId =
   605→    typeof invoice.payment_intent === 'string'
   606→      ? invoice.payment_intent
   607→      : null;
   608→
   609→  // Don't duplicate the payment row if it already exists (Stripe may
   610→  // fire `invoice.paid` + `payment_intent.succeeded` for the same charge).
   611→  // Dedup by invoice id OR payment-intent id.
   612→  const existing = await findExistingPayment(invoice.id, piId);
   613→  if (existing) {
   614→    // Even when the payment row already exists (e.g. from a previous
   615→    // event for the same underlying charge), record the coupon snapshot
   616→    // + redemption audit if the invoice had a discount.
   617→    await recordInvoiceCoupon(event, invoice, existing.id, userId);
   618→    return;
   619→  }
   620→
   621→  // Fetch the payment-method metadata for the full relational record.
   622→  const stripe = getStripeClient();
   623→  const meta = await fetchPaymentMethodMeta(stripe, piId);
   624→
   625→  // Find the user's subscription row.
   626→  const sub = await db.subscription.findUnique({ where: { userId } });
   627→  const payment = await db.payment.create({
   628→    data: {
   629→      userId,
   630→      subscriptionId: sub?.id ?? null,
   631→      planId,
   632→      amount: invoice.amount_paid ?? 0,
   633→      currency: invoice.currency ?? 'usd',
   634→      status: 'paid',
   635→      method: meta?.method ?? 'Stripe',
   636→      invoiceNumber: invoice.number ?? null,
   637→      stripeInvoiceId: invoice.id,
   638→      stripePaymentIntentId: piId,
   639→      stripeChargeId: meta?.stripeChargeId ?? null,
   640→      paymentMethodType: meta?.paymentMethodType ?? null,
   641→      paymentMethodDetails: meta?.paymentMethodDetails ?? null,
   642→      paidAt: new Date(),
   643→    },
   644→  });
   645→
   646→  // Coupon recording — best-effort, never blocks the payment row creation
   647→  // (the row was already created above). Sets couponCode/discountAmount on
   648→  // the just-created Payment row, increments the local Coupon.timesRedeemed
   649→  // (idempotent), and creates a CouponRedemption audit row (idempotent).
   650→  await recordInvoiceCoupon(event, invoice, payment.id, userId);
   651→}

   653→async function handleInvoiceFailed(event: Stripe.Event) {
   654→  const invoice = event.data.object as Stripe.Invoice;
   655→  const userId = invoice.metadata?.userId;
   656→  if (!userId) return;
   657→
   658→  // Mark the subscription as past_due (Stripe handles retries; we just
   659→  // reflect the current state).
   660→  const existing = await db.subscription.findUnique({ where: { userId } });
   661→  if (existing) {
   662→    await db.subscription.update({
   663→      where: { userId },
   664→      data: { status: 'past_due' },
   665→    });
   666→  }
   667→
   668→  const planId = invoice.metadata?.planId;
   669→  if (!planId) return;
   670→
   671→  const piId = typeof invoice.payment_intent === 'string' ? invoice.payment_intent : null;
   672→
   673→  // Don't duplicate a failed row if one already exists for this invoice/PI.
   674→  const existingPayment = await findExistingPayment(invoice.id, piId);
   675→  if (existingPayment) return;
   676→
   677→  // Fetch payment-method metadata + the failure reason for the description.
   678→  const stripe = getStripeClient();
   679→  const meta = await fetchPaymentMethodMeta(stripe, piId);
   680→  let failureReason: string | null = null;
   681→  if (piId) {
   682→    try {
   683→      const pi = (await stripe.paymentIntents.retrieve(piId)) as unknown as {
   684→        last_payment_error?: { message?: string } | null;
   685→      };
   686→      failureReason = pi.last_payment_error?.message ?? null;
   687→    } catch {
   688→      // ignore — failure reason is best-effort
   689→    }
   690→  }
   691→
   692→  await db.payment.create({
   693→    data: {
   694→      userId,
   695→      subscriptionId: existing?.id ?? null,
   696→      planId,
   697→      amount: invoice.amount_due ?? 0,
   698→      currency: invoice.currency ?? 'usd',
   699→      status: 'failed',
   700→      method: meta?.method ?? 'Stripe',
   701→      invoiceNumber: invoice.number ?? null,
   702→      stripeInvoiceId: invoice.id,
   703→      stripePaymentIntentId: piId,
   704→      stripeChargeId: meta?.stripeChargeId ?? null,
   705→      paymentMethodType: meta?.paymentMethodType ?? null,
   706→      paymentMethodDetails: meta?.paymentMethodDetails ?? null,
   707→      description: failureReason,
   708→    },
   709→  });
   710→}

   712→/** payment_intent.succeeded — captures a one-time / orphan charge that
   713→ *  isn't tied to a subscription invoice (or acts as a dedup-safe mirror
   714→ *  of an invoice.paid that already recorded the same PI). Never
   715→ *  duplicates because findExistingPayment dedups by PI id. */
   716→async function handlePaymentIntentSucceeded(event: Stripe.Event) {
   717→  const pi = event.data.object as Stripe.PaymentIntent;
   718→  const userId = pi.metadata?.userId;
   719→  if (!userId) return;
   720→
   721→  // Dedup against any row already created by invoice.paid /
   722→  // checkout.session.completed for this PI.
   723→  const existing = await findExistingPayment(null, pi.id);
   724→  if (existing) return;
   725→
   726→  // Determine the planId: prefer metadata, else fall back to the user's

   734→
   735→  const sub = await db.subscription.findUnique({ where: { userId } });
   736→  const stripe = getStripeClient();
   737→  const meta = await fetchPaymentMethodMeta(stripe, pi.id);
   738→
   739→  await db.payment.create({
   740→    data: {
   741→      userId,
   742→      subscriptionId: sub?.id ?? null,
   743→      planId,

   745→      currency: pi.currency ?? 'usd',
   746→      status: 'paid',
   747→      method: meta?.method ?? 'Stripe',

   800→/** charge.refunded — updates the existing Payment row to reflect a refund.
   801→ *  Full refund → status='refunded'. Partial refund → keep current status
   802→ *  but annotate 'Partial refund processed: <amount_refunded> <currency>'.
   803→ *  Deduped by stripeChargeId, falling back to PI id. Never creates a
   804→ *  duplicate Payment row — the original row is updated in place. If no
   805→ *  Payment row exists for the charge (orphan charge that never created
   806→ *  one), returns silently. */
   807→async function handleChargeRefunded(event: Stripe.Event) {
   808→  const charge = event.data.object as Stripe.Charge;
   809→
   810→  // Find the existing Payment row by charge id, falling back to PI id.
   811→  let payment = await db.payment.findFirst({ where: { stripeChargeId: charge.id } });
   812→  if (!payment) {
   813→    const piId = typeof charge.payment_intent === 'string' ? charge.payment_intent : null;
   814→    if (piId) {
   815→      payment = await db.payment.findFirst({ where: { stripePaymentIntentId: piId } });
   816→    }
   817→  }
   818→  if (!payment) {
   819→    // Orphan charge (no Payment row ever created) — silently ignore.
   820→    return;
   821→  }
   822→
   823→  // The Stripe SDK 2026-08-26.dahlia type may not expose `refunded` /
   824→  // `amount_refunded` on Charge in all API versions; cast to access
   825→  // them safely.
   826→  const chargeAny = charge as unknown as {
   827→    refunded?: boolean;
   828→    amount_refunded?: number;
   829→    amount?: number;
   830→    currency?: string;
   831→  };
   832→  const fullyRefunded = chargeAny.refunded === true;
   833→  const amountRefunded = chargeAny.amount_refunded ?? 0;
   834→  const currency = (chargeAny.currency ?? charge.currency ?? 'usd').toUpperCase();
   835→
   836→  if (fullyRefunded) {
   837→    await db.payment.update({
   838→      where: { id: payment.id },
   839→      data: { status: 'refunded' },
   840→    });
   841→  } else {
   842→    // Partial refund — keep current status but record a memo so the
   843→    // admin Payments table can show "Partial refund processed: …".
   844→    await db.payment.update({
   845→      where: { id: payment.id },
   846→      data: {
   847→        description: `Partial refund processed: ${amountRefunded} ${currency}`,
   848→      },
   849→    });
   850→  }
   851→}
