    50→    50→// -------------------- Model Resolution Helper --------------------
    51→    51→// The frontend sends DB cuids as `modelId` (e.g. "m-openai-gpt5"), but the
    52→    52→// upstream provider APIs need the actual model string (e.g. "gpt-5").
    53→    53→// This helper resolves a DB cuid to the AiModel row, validates ownership +
    54→    54→// active status + type, and returns the model row. If no modelId is provided,
    55→    55→// it falls back to AI Settings defaults, then the provider's default model.
    56→    56→
    57→    57→interface ResolvedModel {
    58→    58→  modelId: string;        // upstream model string (e.g. "gpt-5")
    59→    59→  modelDbId: string;      // DB cuid (e.g. "m-openai-gpt5")
    60→    60→  inputCostPer1k: number | null;
    61→    61→  outputCostPer1k: number | null;
    62→    62→  type: string;           // 'TEXT' | 'IMAGE'
    63→    63→}
    64→    64→
    65→    65→async function resolveModel(
    66→    66→  providerId: string,
    67→    67→  modelId: string | undefined,
    68→    68→  expectedType: 'TEXT' | 'IMAGE',
    69→    69→  providerModels: Array<{ id: string; modelId: string; isActive: boolean; isDefault: boolean; type: string; inputCostPer1k: number | null; outputCostPer1k: number | null }>,
    70→    70→): Promise<ResolvedModel> {
    71→    71→  // If a modelId is provided, it's a DB cuid — look it up
    72→    72→  if (modelId) {
    73→    73→    const model = providerModels.find((m) => m.id === modelId);
    74→    74→    if (!model) {
    75→    75→      throw new Error('The selected model was not found for this provider. Please select a valid model.');
    76→    76→    }
    77→    77→    if (model.providerId !== undefined && model.providerId !== providerId) {
    78→    78→      throw new Error('The selected model does not belong to the selected provider.');
    79→    79→    }
    80→    80→    if (!model.isActive) {
    81→    81→      throw new Error('The selected model is inactive. Please activate it or select another model.');
    82→    82→    }
    83→    83→    if (model.type?.toUpperCase() !== expectedType) {
    84→    84→      throw new Error(`The selected model is not a ${expectedType.toLowerCase()} model. Please select a ${expectedType.toLowerCase()} model.`);
    85→    85→    }
    86→    86→    return {
    87→    87→      modelId: model.modelId,
    88→    88→      modelDbId: model.id,
    89→    89→      inputCostPer1k: model.inputCostPer1k,
    90→    90→      outputCostPer1k: model.outputCostPer1k,
    91→    91→      type: model.type,
    92→    92→    };
    93→    93→  }
    94→    94→
    95→    95→  // No modelId provided — fall back to AI Settings defaults
    96→    96→  const settings = await db.aiSettings.findUnique({ where: { scope: 'global' } });
    97→    97→  const settingsModelId = expectedType === 'TEXT' ? settings?.defaultModelId : settings?.imageModelId;
    98→    98→  if (settingsModelId) {
    99→    99→    const model = providerModels.find((m) => m.id === settingsModelId && m.isActive && m.type?.toUpperCase() === expectedType);
   100→   100→    if (model) {
   101→   101→      return {
   102→   102→        modelId: model.modelId,
   103→   103→        modelDbId: model.id,
   104→   104→        inputCostPer1k: model.inputCostPer1k,
   105→   105→        outputCostPer1k: model.outputCostPer1k,
   106→   106→        type: model.type,
   107→   107→      };
   108→   108→    }
   109→   109→  }
   110→   110→
   111→   111→  // Fall back to the provider's default model of the correct type
   112→   112→  const defaultModel = providerModels.find((m) => m.isActive && m.isDefault && m.type?.toUpperCase() === expectedType)
   113→   113→    ?? providerModels.find((m) => m.isActive && m.type?.toUpperCase() === expectedType);
   114→   114→  if (!defaultModel) {
   115→   115→    throw new Error(`No active ${expectedType.toLowerCase()} model is configured for this provider. Please add or activate a model.`);
   116→   116→  }
   117→   117→  return {
   118→   118→    modelId: defaultModel.modelId,
   119→   119→    modelDbId: defaultModel.id,
   120→   120→    inputCostPer1k: defaultModel.inputCostPer1k,
   121→   121→    outputCostPer1k: defaultModel.outputCostPer1k,
   122→   122→    type: defaultModel.type,
   123→   123→  };
   124→   124→}

   126→   126→// -------------------- Core AI Service --------------------
   127→   127→
   128→   128→export async function executeChat(req: ChatRequest): Promise<ChatResponse> {
   129→   129→  const provider = await db.aiProvider.findUnique({
   130→   130→    where: { id: req.providerId },
   131→   131→    include: { models: true },
   132→   132→  });
   133→   133→
   134→   134→  if (!provider) throw new Error('Provider not found');
   135→   135→  if (!provider.isActive) throw new Error('Provider is disabled. Please activate it first.');
   136→   136→  if (!provider.apiKeyEncrypted) throw new Error('API key not configured for this provider.');
   137→   137→
   138→   138→  // Resolve + validate the model (handles DB cuid → model string, type=TEXT, active, belongs-to-provider)
   139→   139→  const resolved = await resolveModel(req.providerId, req.modelId, 'TEXT', provider.models);
   140→   140→  const modelId = resolved.modelId;
   141→   141→
   142→   142→  // Apply AI Settings defaults for temperature/maxTokens if not provided
   143→   143→  const settings = await db.aiSettings.findUnique({ where: { scope: 'global' } });
   144→   144→  const temperature = req.temperature ?? settings?.defaultTemperature ?? 0.7;
   145→   145→  const maxTokens = req.maxTokens ?? settings?.defaultMaxTokens ?? 2048;
   146→   146→
   147→   147→  const apiKey = await decrypt(provider.apiKeyEncrypted);
   148→   148→  const config = getProviderConfig(provider.kind);
   149→   149→  const baseUrl = provider.baseUrl || config.defaultBaseUrl;
   150→   150→  // CUSTOM providers have no defaultBaseUrl — they must have one set explicitly.
   151→   151→  if (!baseUrl) {
   152→   152→    throw new Error('No Base URL configured for this custom provider. Please edit the provider and set a Base URL.');
   153→   153→  }
   154→   154→
   155→   155→  const startTime = Date.now();
   156→   156→  let inputTokens = 0;
   157→   157→  let outputTokens = 0;
   158→   158→  let content = '';
   159→   159→  let usedProvider = provider;
   160→   160→  let usedModelId = modelId;
   161→   161→
   162→   162→  try {
   163→   163→    if (provider.kind === 'ANTHROPIC') {
   164→   164→      const result = await callAnthropic(baseUrl, apiKey, modelId, req.messages, {
   165→   165→        temperature, maxTokens, jsonMode: req.jsonMode,
   166→   166→      });
   167→   167→      inputTokens = result.inputTokens;
   168→   168→      outputTokens = result.outputTokens;
   169→   169→      content = result.content;
   170→   170→    } else if (provider.kind === 'GEMINI') {
   171→   171→      const result = await callGemini(baseUrl, apiKey, modelId, req.messages, {
   172→   172→        temperature, maxTokens, jsonMode: req.jsonMode,
   173→   173→      });
   174→   174→      inputTokens = result.inputTokens;
   175→   175→      outputTokens = result.outputTokens;
   176→   176→      content = result.content;
   177→   177→    } else {
   178→   178→      // OpenAI-compatible (OpenAI, Groq, DeepSeek, Custom)
   179→   179→      const result = await callOpenAI(baseUrl, apiKey, modelId, req.messages, {
   180→   180→        temperature, maxTokens,
   181→   181→        topP: req.topP,
   182→   182→        frequencyPenalty: req.frequencyPenalty,
   183→   183→        presencePenalty: req.presencePenalty,
   184→   184→        jsonMode: req.jsonMode,
   185→   185→      });
   186→   186→      inputTokens = result.inputTokens;
   187→   187→      outputTokens = result.outputTokens;
   188→   188→      content = result.content;
   189→   189→    }
   190→   190→  } catch (err) {
   191→   191→    // Try fallback providers
   192→   192→    const fallbacks = await db.aiProviderFallback.findMany({
   193→   193→      where: { providerId: provider.id },
   194→   194→      include: { fallback: { include: { models: true } } },
   195→   195→      orderBy: { priority: 'asc' },
   196→   196→    });
   197→   197→
   198→   198→    let lastError = err instanceof Error ? err : new Error('Unknown error');
   199→   199→
   200→   200→    for (const fb of fallbacks) {
   201→   201→      if (!fb.fallback.isActive || !fb.fallback.apiKeyEncrypted) continue;
   202→   202→      try {
   203→   203→        const fbResolved = await resolveModel(fb.fallback.id, undefined, 'TEXT', fb.fallback.models);
   204→   204→        const fbApiKey = await decrypt(fb.fallback.apiKeyEncrypted);
   205→   205→        const fbConfig = getProviderConfig(fb.fallback.kind);
   206→   206→        const fbBaseUrl = fb.fallback.baseUrl || fbConfig.defaultBaseUrl;
   207→   207→        // CUSTOM providers have no defaultBaseUrl — skip if the admin hasn't set one.
   208→   208→        if (!fbBaseUrl) continue;
   209→   209→
   210→   210→        let fbResult: { content: string; inputTokens: number; outputTokens: number };
   211→   211→        if (fb.fallback.kind === 'ANTHROPIC') {
   212→   212→          fbResult = await callAnthropic(fbBaseUrl, fbApiKey, fbResolved.modelId, req.messages, { temperature, maxTokens });
   213→   213→        } else if (fb.fallback.kind === 'GEMINI') {
   214→   214→          fbResult = await callGemini(fbBaseUrl, fbApiKey, fbResolved.modelId, req.messages, { temperature, maxTokens });
   215→   215→        } else {
   216→   216→          // OpenAI-compatible (OpenAI, Groq, DeepSeek, Custom)
   217→   217→          fbResult = await callOpenAI(fbBaseUrl, fbApiKey, fbResolved.modelId, req.messages, { temperature, maxTokens });
   218→   218→        }
   219→   219→        inputTokens = fbResult.inputTokens;
   220→   220→        outputTokens = fbResult.outputTokens;
   221→   221→        content = fbResult.content;
   222→   222→        usedProvider = fb.fallback;
   223→   223→        usedModelId = fbResolved.modelId;
   224→   224→        lastError = null as unknown as Error;
   225→   225→        break;
   226→   226→      } catch (fbErr) {
   227→   227→        lastError = fbErr instanceof Error ? fbErr : new Error('Fallback failed');
   228→   228→        continue;
   229→   229→      }
   230→   230→    }
   231→   231→
   232→   232→    if (!content) {
   233→   233→      // Log the failed request
   234→   234→      const durationMs = Date.now() - startTime;
   235→   235→      await db.aiLog.create({
   236→   236→        data: {
   237→   237→          providerId: provider.id,
   238→   238→          providerName: provider.name,
   239→   239→          modelId,
   240→   240→          question: req.messages.map((m) => m.content).join('\n'),
   241→   241→          response: null,
   242→   242→          inputTokens: 0,
   243→   243→          outputTokens: 0,
   244→   244→          totalTokens: 0,
   245→   245→          costUsd: 0,
   246→   246→          durationMs,
   247→   247→          status: 'error',
   248→   248→          errorMessage: lastError?.message ?? 'Unknown error',
   249→   249→          siteId: req.siteId,
   250→   250→          userId: req.userId,
   251→   251→        },
   252→   252→      }).catch(() => { /* logging failure shouldn't mask the original error */ });
   253→   253→      throw lastError ?? new Error('Chat request failed');
   254→   254→    }
   255→   255→  }
   256→   256→
   257→   257→  const durationMs = Date.now() - startTime;
   258→   258→  const totalTokens = inputTokens + outputTokens;
   259→   259→  const costUsd = (inputTokens / 1000) * (resolved.inputCostPer1k || 0)
   260→   260→    + (outputTokens / 1000) * (resolved.outputCostPer1k || 0);
   261→   261→
   262→   262→  // Update provider
   263→   263→  await db.aiProvider.update({
   264→   264→    where: { id: usedProvider.id },
   265→   265→    data: {
   266→   266→      lastUsedAt: new Date(),
   267→   267→      latencyMs: durationMs,
   268→   268→      connectionStatus: 'CONNECTED',
   269→   269→      lastError: null,
   270→   270→    },
   271→   271→  });
   272→   272→
   273→   273→  // Log the request
   274→   274→  await db.aiLog.create({
   275→   275→    data: {
   276→   276→      providerId: usedProvider.id,
   277→   277→      providerName: usedProvider.name,
   278→   278→      modelId: usedModelId,
   279→   279→      question: req.messages.map((m) => m.content).join('\n'),
   280→   280→      response: content,
   281→   281→      inputTokens,
   282→   282→      outputTokens,
   283→   283→      totalTokens,
   284→   284→      costUsd,
   285→   285→      durationMs,
   286→   286→      status: 'success',
   287→   287→      siteId: req.siteId,
   288→   288→      userId: req.userId,
   289→   289→    },
   290→   290→  });
   291→   291→
   292→   292→  return {
   293→   293→    content,
   294→   294→    model: usedModelId,
   295→   295→    inputTokens,
   296→   296→    outputTokens,
   297→   297→    totalTokens,
   298→   298→    costUsd,
   299→   299→    durationMs,
   300→   300→    providerName: usedProvider.name,
   301→   301→  };
   302→   302→}

   304→   304→// -------------------- Provider-specific call implementations --------------------
   305→   305→
   306→   306→async function callOpenAI(
   307→   307→  baseUrl: string,
   308→   308→  apiKey: string,
   309→   309→  model: string,
   310→   310→  messages: ChatMessage[],
   311→   311→  opts: { temperature?: number; maxTokens?: number; topP?: number; frequencyPenalty?: number; presencePenalty?: number; jsonMode?: boolean; apiVersion?: string } = {},
   312→   312→) {
   313→   313→  const url = opts.apiVersion
   314→   314→    ? `${baseUrl}?api-version=${opts.apiVersion}`
   315→   315→    : `${baseUrl}/chat/completions`;
   316→   316→
   317→   317→  const body: Record<string, unknown> = {
   318→   318→    model,
   319→   319→    messages,
   320→   320→    temperature: opts.temperature ?? 0.7,
   321→   321→    max_tokens: opts.maxTokens ?? 2048,
   322→   322→  };
   323→   323→  if (opts.topP !== undefined) body.top_p = opts.topP;
   324→   324→  if (opts.frequencyPenalty !== undefined) body.frequency_penalty = opts.frequencyPenalty;
   325→   325→  if (opts.presencePenalty !== undefined) body.presence_penalty = opts.presencePenalty;
   326→   326→  if (opts.jsonMode) {
   327→   327→    body.response_format = { type: 'json_object' };
   328→   328→  }
   329→   329→
   330→   330→  const res = await fetch(url, {
   331→   331→    method: 'POST',
   332→   332→    headers: {
   333→   333→      'Content-Type': 'application/json',
   334→   334→      'Authorization': `Bearer ${apiKey}`,
   335→   335→    },
   336→   336→    body: JSON.stringify(body),
   337→   337→  });
   338→   338→
   339→   339→  if (!res.ok) {
   340→   340→    const err = await res.text();
   341→   341→    throw new Error(`OpenAI API error: ${res.status} — ${err}`);
   342→   342→  }
   343→   343→
   344→   344→  const data = await res.json();
   345→   345→  return {
   346→   346→    content: data.choices?.[0]?.message?.content || '',
   347→   347→    inputTokens: data.usage?.prompt_tokens || 0,
   348→   348→    outputTokens: data.usage?.completion_tokens || 0,
   349→   349→  };
   350→   350→}
   351→   351→
   352→   352→async function callAnthropic(
   353→   353→  baseUrl: string,
   354→   354→  apiKey: string,
   355→   355→  model: string,
   356→   356→  messages: ChatMessage[],
   357→   357→  opts: { temperature?: number; maxTokens?: number; jsonMode?: boolean } = {},
   358→   358→) {
   359→   359→  const systemMsg = messages.find(m => m.role === 'system');
   360→   360→  const chatMsgs = messages.filter(m => m.role !== 'system');
   361→   361→
   362→   362→  const body: Record<string, unknown> = {
   363→   363→    model,
   364→   364→    messages: chatMsgs.map(m => ({ role: m.role, content: m.content })),
   365→   365→    max_tokens: opts.maxTokens ?? 2048,
   366→   366→    temperature: opts.temperature ?? 0.7,
   367→   367→  };
   368→   368→  if (systemMsg) body.system = systemMsg.content;
   369→   369→
   370→   370→  const res = await fetch(`${baseUrl}/messages`, {
   371→   371→    method: 'POST',
   372→   372→    headers: {
   373→   373→      'Content-Type': 'application/json',
   374→   374→      'x-api-key': apiKey,
   375→   375→      'anthropic-version': '2023-06-01',
   376→   376→    },
   377→   377→    body: JSON.stringify(body),
   378→   378→  });
   379→   379→
   380→   380→  if (!res.ok) {
   381→   381→    const err = await res.text();
   382→   382→    throw new Error(`Anthropic API error: ${res.status} — ${err}`);
   383→   383→  }
   384→   384→
   385→   385→  const data = await res.json();
   386→   386→  return {
   387→   387→    content: data.content?.[0]?.text || '',
   388→   388→    inputTokens: data.usage?.input_tokens || 0,
   389→   389→    outputTokens: data.usage?.output_tokens || 0,
   390→   390→  };
   391→   391→}
   392→   392→
   393→   393→async function callGemini(
   394→   394→  baseUrl: string,
   395→   395→  apiKey: string,
   396→   396→  model: string,
   397→   397→  messages: ChatMessage[],
   398→   398→  opts: { temperature?: number; maxTokens?: number; jsonMode?: boolean } = {},
   399→   399→) {
   400→   400→  const systemMsg = messages.find(m => m.role === 'system');
   401→   401→  const chatMsgs = messages.filter(m => m.role !== 'system');
   402→   402→
   403→   403→  const body: Record<string, unknown> = {
   404→   404→    contents: chatMsgs.map(m => ({
   405→   405→      role: m.role === 'assistant' ? 'model' : 'user',
   406→   406→      parts: [{ text: m.content }],
   407→   407→    })),
   408→   408→    generationConfig: {
   409→   409→      temperature: opts.temperature ?? 0.7,
   410→   410→      maxOutputTokens: opts.maxTokens ?? 2048,
   411→   411→    },
   412→   412→  };
   413→   413→
   414→   414→  const res = await fetch(
   415→   415→    `${baseUrl}/models/${model}:generateContent?key=${apiKey}`,
   416→   416→    {
   417→   417→      method: 'POST',
   418→   418→      headers: { 'Content-Type': 'application/json' },
   419→   419→      body: JSON.stringify(body),
   420→   420→    },
   421→   421→  );
   422→   422→
   423→   423→  if (!res.ok) {
   424→   424→    const err = await res.text();
   425→   425→    throw new Error(`Gemini API error: ${res.status} — ${err}`);
   426→   426→  }
   427→   427→
   428→   428→  const data = await res.json();
   429→   429→  return {
   430→   430→    content: data.candidates?.[0]?.content?.parts?.[0]?.text || '',
   431→   431→    inputTokens: data.usageMetadata?.promptTokenCount || 0,
   432→   432→    outputTokens: data.usageMetadata?.candidatesTokenCount || 0,
   433→   433→  };
   434→   434→}

   436→   436→// -------------------- Health Check --------------------
   437→   437→
   438→   438→export async function healthCheck(providerId: string): Promise<HealthCheckResult> {
   439→   439→  const provider = await db.aiProvider.findUnique({
   440→   440→    where: { id: providerId },
   441→   441→    include: { models: { where: { isActive: true } } },
   442→   442→  });
   443→   443→  if (!provider) throw new Error('Provider not found');
   444→   444→  if (!provider.apiKeyEncrypted) return { status: 'DISCONNECTED', latencyMs: 0, error: 'No API key configured' };
   445→   445→
   446→   446→  const apiKey = await decrypt(provider.apiKeyEncrypted);
   447→   447→  const config = getProviderConfig(provider.kind);
   448→   448→  const baseUrl = provider.baseUrl || config.defaultBaseUrl;
   449→   449→  // CUSTOM providers have no defaultBaseUrl — they must have one set explicitly.
   450→   450→  if (!baseUrl) {
   451→   451→    return { status: 'DISCONNECTED', latencyMs: 0, error: 'No Base URL configured for this custom provider. Please edit the provider and set a Base URL.' };
   452→   452→  }
   453→   453→
   454→   454→  const start = Date.now();
   455→   455→  try {
   456→   456→    let models: ProviderModel[] = [];
   457→   457→
   458→   458→    if (provider.kind === 'ANTHROPIC') {
   459→   459→      // Anthropic has no /models endpoint — send a minimal chat request to verify the API key.
   460→   460→      // Use the provider's first active TEXT model, or fall back to a known-good default.
   461→   461→      const testModel = provider.models.find((m) => m.type?.toUpperCase() === 'TEXT')?.modelId
   462→   462→        ?? config.defaultModels.find((m) => !isImageModelId(m.modelId))?.modelId
   463→   463→        ?? 'claude-3-5-haiku-20241022';
   464→   464→      const res = await fetch(`${baseUrl}/messages`, {
   465→   465→        method: 'POST',

   654→   654→export async function getUsageAnalytics(siteId?: string, period: 'day' | 'week' | 'month' = 'month') {
   655→   655→  const now = new Date();
   656→   656→  const periodStart = new Date();
   657→   657→  if (period === 'day') periodStart.setHours(now.getHours() - 24);
   658→   658→  else if (period === 'week') periodStart.setDate(now.getDate() - 7);
   659→   659→  else periodStart.setMonth(now.getMonth() - 1);
   660→   660→
   661→   661→  const where: Prisma.AiLogWhereInput = {
   662→   662→    createdAt: { gte: periodStart },
   663→   663→    ...(siteId ? { siteId } : {}),
   664→   664→  };
   665→   665→
   666→   666→  const [totalCount, totalTokens, totalCost, errorCount, avgDuration] = await Promise.all([
   667→   667→    db.aiLog.count({ where }),
   668→   668→    db.aiLog.aggregate({ where, _sum: { totalTokens: true, inputTokens: true, outputTokens: true } }),
   669→   669→    db.aiLog.aggregate({ where, _sum: { costUsd: true } }),
   670→   670→    db.aiLog.count({ where: { ...where, status: 'error' } }),
   671→   671→    db.aiLog.aggregate({ where: { ...where, status: 'success' }, _avg: { durationMs: true } }),
   672→   672→  ]);
   673→   673→
   674→   674→  // Provider breakdown
   675→   675→  const providerBreakdown = await db.aiLog.groupBy({
   676→   676→    by: ['providerName'],
   677→   677→    where,
   678→   678→    _sum: { totalTokens: true, costUsd: true, inputTokens: true, outputTokens: true },
   679→   679→    _count: true,
   680→   680→    orderBy: { _sum: { costUsd: 'desc' } },
   681→   681→  });
   682→   682→
   683→   683→  // Model breakdown
   684→   684→  const modelBreakdown = await db.aiLog.groupBy({
   685→   685→    by: ['modelId'],
   686→   686→    where,
   687→   687→    _sum: { totalTokens: true, costUsd: true },
   688→   688→    _count: true,
   689→   689→    orderBy: { _count: { id: 'desc' } },
   690→   690→    take: 10,
   691→   691→  });
   692→   692→
   693→   693→  // Daily usage chart
   694→   694→  const dailyUsage = await db.$queryRawUnsafe<Array<{ date: string; count: number; tokens: number; cost: number }>>(`
   695→   695→    SELECT DATE(createdAt) as date, COUNT(*) as count, COALESCE(SUM(totalTokens), 0) as tokens, COALESCE(SUM(costUsd), 0) as cost
   696→   696→    FROM AiLog
   697→   697→    WHERE createdAt >= ?
   698→   698→    ${siteId ? 'AND siteId = ?' : ''}
   699→   699→    GROUP BY DATE(createdAt)
   700→   700→    ORDER BY date ASC
   701→   701→  `, periodStart.toISOString(), ...(siteId ? [siteId] : []));
   702→   702→
   703→   703→  return {
   704→   704→    totalRequests: totalCount,
   705→   705→    totalTokens: totalTokens._sum.totalTokens || 0,
   706→   706→    inputTokens: totalTokens._sum.inputTokens || 0,
   707→   707→    outputTokens: totalTokens._sum.outputTokens || 0,
   708→   708→    totalCost: totalCost._sum.costUsd || 0,
   709→   709→    errorCount,
   710→   710→    successRate: totalCount > 0 ? ((totalCount - errorCount) / totalCount) * 100 : 100,
   711→   711→    avgDurationMs: Math.round(avgDuration._avg.durationMs || 0),
   712→   712→    providerBreakdown,
   713→   713→    modelBreakdown,
   714→   714→    dailyUsage,
   715→   715→  };
   716→   716→}

   821→   821→async function callGeminiImageGeneration(
   822→   822→  baseUrl: string,
   823→   823→  apiKey: string,
   824→   824→  model: string,
   825→   825→  prompt: string,
   826→   826→  opts: {

   838→   838→  if (Math.abs(width - height) < 100) {
   839→   839→    ratio = '1:1'; // square

   841→   841→    ratio = width >= height ? '16:9' : '9:16';
   842→   842→  }

   853→   853→        generationConfig: {
   854→   854→          responseModalities: ['TEXT', 'IMAGE'],
   855→   855→          // @ts-expect-error -- Gemini-specific field
   856→   856→          imageGenerationConfig: {
   857→   857→            numberOfImages: opts.n ?? 1,
   858→   858→            aspectRatio: ratio,
   859→   859→          },
   860→   860→        },
   861→   861→      }),
   862→   862→    },
   863→   863→  );
   864→   864→
   865→   865→  if (!res.ok) {
   866→   866→    const err = await res.text();
   867→   867→    throw new Error(`Gemini image generation error: ${res.status} — ${err}`);
   868→   868→  }

   870→   870→  const data = await res.json();
   871→   871→  const images: GeneratedImage[] = [];

   876→   876→      images.push({
   877→   877→        url: null,
   878→   878→        base64: part.inlineData.data ?? null,
   879→   879→        revisedPrompt: prompt,
   880→   880→      });
   881→   881→    }
   882→   882→  }
   883→   883→
   884→   884→  // Fallback: if no image parts returned, throw
   885→   885→  if (images.length === 0) {
   886→   886→    throw new Error('No images were generated by Gemini');
   887→   887→  }
   888→   888→
   889→   889→  return { images };

   921→   921→      const result = await callGeminiImageGeneration(baseUrl, apiKey, modelId, req.prompt, {
   922→   922→        negativePrompt: req.negativePrompt,
   923→   923→        size: req.size,
   924→   924→        n: req.n,
   925→   925→        responseFormat: req.responseFormat,
   926→   926→      });
   927→   927→      images = result.images;
   928→   928→    } else if (provider.kind === 'OPENAI' || provider.kind === 'CUSTOM') {
   929→   929→      // OpenAI and Custom (OpenAI-compatible) providers support image generation
   930→   930→      const result = await callOpenAIImageGeneration(baseUrl, apiKey, modelId, req.prompt, {
   931→   931→        negativePrompt: req.negativePrompt,
   932→   932→        size: req.size,
   933→   933→        quality: req.quality,
   934→   934→        style: req.style,
   935→   935→        n: req.n,
   936→   936→        responseFormat: req.responseFormat,
   937→   937→      });
   938→   938→      images = result.images;
   939→   939→    } else {
   940→   940→      // GROQ and DEEPSEEK do not support image generation
   941→   941→      throw new Error(`${config.name} does not support image generation. Please use OpenAI, Gemini, or a Custom OpenAI-compatible provider.`);
   942→   942→    }
   943→   943→  } catch (err) {
   944→   944→    // Try fallback providers for image generation
   945→   945→    const fallbacks = await db.aiProviderFallback.findMany({
   946→   946→      where: { providerId: provider.id },
   947→   947→      include: { fallback: { include: { models: true } } },
   948→   948→      orderBy: { priority: 'asc' },
   949→   949→    });
   950→   950→
   951→   951→    let lastError = err instanceof Error ? err : new Error('Unknown error');
   952→   952→
   953→   953→    for (const fb of fallbacks) {
   954→   954→      if (!fb.fallback.isActive || !fb.fallback.apiKeyEncrypted) continue;
   955→   955→      // Only try fallbacks that support image generation (OpenAI, Gemini, or Custom OpenAI-compatible)
   956→   956→      if (fb.fallback.kind !== 'OPENAI' && fb.fallback.kind !== 'GEMINI' && fb.fallback.kind !== 'CUSTOM') continue;
   957→   957→      try {
   958→   958→        const fbResolved = await resolveModel(fb.fallback.id, undefined, 'IMAGE', fb.fallback.models);
   959→   959→        const fbApiKey = await decrypt(fb.fallback.apiKeyEncrypted);
   960→   960→        const fbConfig = getProviderConfig(fb.fallback.kind);
   961→   961→        const fbBaseUrl = fb.fallback.baseUrl || fbConfig.defaultBaseUrl;
   962→   962→        // CUSTOM providers have no defaultBaseUrl — skip if not set.
   963→   963→        if (!fbBaseUrl) continue;
   964→   964→
   965→   965→        let fbResult: { images: GeneratedImage[] };
   966→   966→        if (fb.fallback.kind === 'GEMINI') {
   967→   967→          fbResult = await callGeminiImageGeneration(fbBaseUrl, fbApiKey, fbResolved.modelId, req.prompt, {
   968→   968→            negativePrompt: req.negativePrompt, size: req.size, n: req.n, responseFormat: req.responseFormat,
   969→   969→          });
   970→   970→        } else {
   971→   971→          // OpenAI or Custom (OpenAI-compatible)
   972→   972→          fbResult = await callOpenAIImageGeneration(fbBaseUrl, fbApiKey, fbResolved.modelId, req.prompt, {
   973→   973→            size: req.size, quality: req.quality, style: req.style, n: req.n, responseFormat: req.responseFormat,
   974→   974→          });
   975→   975→        }
   976→   976→        images = fbResult.images;
   977→   977→        usedProvider = fb.fallback;
   978→   978→        usedModelId = fbResolved.modelId;
   979→   979→        lastError = null as unknown as Error;
   980→   980→        break;
   981→   981→      } catch (fbErr) {
   982→   982→        lastError = fbErr instanceof Error ? fbErr : new Error('Fallback failed');
   983→   983→        continue;
   984→   984→      }
   985→   985→    }
   986→   986→
   987→   987→    if (images.length === 0) {
   988→   988→      // Log the failed request
   989→   989→      const durationMs = Date.now() - startTime;
   990→   990→      await db.aiLog.create({
   991→   991→        data: {
   992→   992→          providerId: provider.id,
   993→   993→          providerName: provider.name,
   994→   994→          modelId,
   995→   995→          question: `[IMAGE] ${req.prompt}`,
   996→   996→          response: null,
   997→   997→          inputTokens: 0,
   998→   998→          outputTokens: 0,
   999→   999→          totalTokens: 0,
  1000→  1000→          costUsd: 0,
  1001→  1001→          durationMs,
  1002→  1002→          status: 'error',
  1003→  1003→          errorMessage: lastError?.message ?? 'Unknown error',
  1004→  1004→          siteId: req.siteId,
  1005→  1005→          userId: req.userId,
  1006→  1006→        },
  1007→  1007→      }).catch(() => { /* logging failure shouldn't mask the original error */ });
  1008→  1008→      throw lastError ?? new Error('Image generation failed');
  1009→  1009→    }
  1010→  1010→  }
  1011→  1011→
  1012→  1012→  const durationMs = Date.now() - startTime;
  1013→  1013→  const size = req.size ?? '1024x1024';
  1014→  1014→  const n = images.length;
  1015→  1015→  const costUsd = n * getImageCost(usedModelId, size);
  1016→  1016→
  1017→  1017→  // Update provider stats
  1018→  1018→  await db.aiProvider.update({
  1019→  1019→    where: { id: usedProvider.id },
  1020→  1020→    data: {
  1021→  1021→      lastUsedAt: new Date(),
  1022→  1022→      latencyMs: durationMs,
  1023→  1023→      connectionStatus: 'CONNECTED',
  1024→  1024→      lastError: null,
  1025→  1025→    },
  1026→  1026→  });
  1027→  1027→
  1028→  1028→  // Log the request
  1029→  1029→  await db.aiLog.create({
  1030→  1030→    data: {
  1031→  1031→      providerId: usedProvider.id,
  1032→  1032→      providerName: usedProvider.name,
  1033→  1033→      modelId: usedModelId,
  1034→  1034→      question: `[IMAGE] ${req.prompt}`,
  1035→  1035→      response: JSON.stringify({
  1036→  1036→        imagesGenerated: n,
  1037→  1037→        size,
  1038→  1038→        format: req.responseFormat ?? 'url',
  1039→  1039→        model: usedModelId,
  1040→  1040→      }),
  1041→  1041→      inputTokens: 0,
  1042→  1042→      outputTokens: 0,
  1043→  1043→      totalTokens: 0,
  1044→  1044→      costUsd,
  1045→  1045→      durationMs,
  1046→  1046→      status: 'success',
  1047→  1047→      siteId: req.siteId,
  1048→  1048→      userId: req.userId,
  1049→  1049→    },
  1050→  1050→  });
  1051→  1051→
  1052→  1052→  return {
  1053→  1053→    images,
  1054→  1054→    model: usedModelId,
  1055→  1055→    costUsd,
  1056→  1056→    durationMs,
  1057→  1057→    providerName: usedProvider.name,
  1058→  1058→    providerKind: usedProvider.kind,
  1059→  1059→  };
  1060→  1060→}
