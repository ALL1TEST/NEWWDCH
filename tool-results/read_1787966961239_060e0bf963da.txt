   279→// -------------------- Category Count Query --------------------
   280→
   281→/**
   282→ * Per-scope category counts. `scope=platform` requests system templates
   283→ * (siteId IS NULL) via the scope=platform query param; `scope=client` uses
   284→ * the default client behavior. The query key is scoped so the two caches
   285→ * never collide.
   286→ */
   287→function useCategoryCounts(scope: 'client' | 'platform' = 'client') {
   288→  return useQuery({
   289→    queryKey: ['email-templates', 'category-counts', scope],
   290→    queryFn: async () => {
   291→      const allCategories = CATEGORIES.filter((c) => c.value !== 'ALL');
   292→      const results = await Promise.allSettled(
   293→        allCategories.map(async (cat) => {
   294→          const res = await getApi<{ data: EmailTemplate[]; meta: { pagination: { total: number } } }>(
   295→            '/api/email-templates',
   296→            {
   297→              category: cat.value,
   298→              page: 1,
   299→              pageSize: 1,
   300→              ...(scope === 'platform' ? { scope: 'platform' } : {}),
   301→            },
   302→            { raw: true },
   303→          );
   304→          return { category: cat.value, count: res?.meta?.pagination?.total ?? 0 };
   305→        }),
   306→      );
   307→      const counts: Record<string, number> = {};
   308→      for (const r of results) {
   309→        if (r.status === 'fulfilled') {
   310→          counts[r.value.category] = r.value.count;
   311→        }
   312→      }
   313→      return counts;
   314→    },
   315→    staleTime: 30_000,
   316→  });
   317→}

   321→export function TemplateList({ onEdit, onPreview, scope = 'client' }: TemplateListProps) {
   322→  const isPlatform = scope === 'platform';
   323→  const moduleName = isPlatform ? 'platform-email-templates' : 'email-templates';
   324→  const queryClient = useQueryClient();
   325→  const navigate = useNavigationStore((s) => s.navigate);
   326→
   327→  // -------------------- Filter State --------------------
   328→  const [category, setCategory] = useState<EmailTemplateCategory | 'ALL'>('ALL');
   329→  const [statusFilter, setStatusFilter] = useState('ALL');
   330→  const [search, setSearch] = useState('');
   331→  const [searchInput, setSearchInput] = useState('');
   332→  const [sortValue, setSortValue] = useState('createdAt:desc');
   333→  const [page, setPage] = useState(1);
   334→  const [pageSize, setPageSize] = useState(DEFAULT_PAGE_SIZE);
   335→
   336→  // -------------------- Dialog State --------------------
   337→  const [deleteTarget, setDeleteTarget] = useState<EmailTemplate | null>(null);
   338→  const [revertTarget, setRevertTarget] = useState<EmailTemplate | null>(null);
   339→  const [sendTestTarget, setSendTestTarget] = useState<string | null>(null);
   340→  const [seedDialogOpen, setSeedDialogOpen] = useState(false);
   341→
   342→  // -------------------- Derived Sort Values --------------------
   343→  const [sortField, sortOrder] = useMemo(() => {
   344→    const [f, o] = sortValue.split(':');
   345→    return [f, o as 'asc' | 'desc'];
   346→  }, [sortValue]);
   347→
   348→  // -------------------- Search Debounce --------------------
   349→  const debounceRef = useRef<ReturnType<typeof setTimeout>>();
   350→
   351→  const updateSearch = useCallback(
   352→    (value: string) => {
   353→      setSearchInput(value);
   354→      if (debounceRef.current) clearTimeout(debounceRef.current);
   355→      debounceRef.current = setTimeout(() => {
   356→        setSearch(value);
   357→        setPage(1);
   358→      }, 300);
   359→    },
   360→    [],
   361→  );
   362→
   363→  // -------------------- Filter Change Handlers --------------------
   364→  const handleCategoryChange = useCallback((value: string) => {
   365→    setCategory(value as EmailTemplateCategory | 'ALL');
   366→    setPage(1);
   367→  }, []);
   368→
   369→  const handleStatusChange = useCallback((value: string) => {
   370→    setStatusFilter(value);
   371→    setPage(1);
   372→  }, []);
   373→
   374→  const handleSortChange = useCallback((value: string) => {
   375→    setSortValue(value);
   376→    setPage(1);
   377→  }, []);
   378→
   379→  // -------------------- Build Query Params --------------------
   380→  const queryParams = useMemo(
   381→    () => ({
   382→      page,
   383→      pageSize,
   384→      sort: sortField,
   385→      order: sortOrder,
   386→      ...(category !== 'ALL' ? { category } : {}),
   387→      ...(statusFilter !== 'ALL' ? { status: statusFilter } : {}),
   388→      ...(search ? { search } : {}),
   389→      // Platform scope requests system templates (siteId IS NULL) and is
   390→      // guarded by requirePlatformAdmin on the server. Client scope omits
   391→      // the param so the default client behavior is preserved.
   392→      ...(isPlatform ? { scope: 'platform' } : {}),
   393→    }),
   394→    [page, pageSize, sortField, sortOrder, category, statusFilter, search, isPlatform],
   395→  );
   396→
   397→  // -------------------- Data Fetching --------------------
   398→  // The query key is scoped so client and platform caches never collide —
   399→  // platform queries are prefixed with the scope to keep them isolated.
   400→  const { data: raw, isLoading } = useQuery({
   401→    queryKey: isPlatform
   402→      ? ['email-templates', 'list', 'platform', queryParams]
   403→      : queryKeys.emailTemplates.list(queryParams),
   404→    queryFn: () =>
   405→      getApi<{ data: EmailTemplate[]; meta: { pagination: { page: number; pageSize: number; total: number; totalPages: number } } }>('/api/email-templates', queryParams, { raw: true }),
   406→    staleTime: 10_000,
   407→  });
   408→
   409→  const templates = (Array.isArray(raw?.data) ? raw.data : []) as EmailTemplate[];
   410→  const pagination = raw?.meta?.pagination;
   411→
   412→  // -------------------- Category Counts --------------------
   413→  const { data: categoryCounts } = useCategoryCounts(scope);
   414→  const totalCount = categoryCounts
   415→    ? Object.values(categoryCounts).reduce((a, b) => a + b, 0)
   416→    : 0;
   417→
   418→  // -------------------- Mutations --------------------
   419→  // Scope-aware invalidation predicate. Platform invalidates the platform
   420→  // cache keys (prefixed with 'platform'); client invalidates the legacy
   421→  // keys. Both scopes also invalidate the detail cache (scope-agnostic,
   422→  // keyed by id) so an edited template's detail refetches.
   423→  const invalidateAll = useCallback(() => {
   424→    if (isPlatform) {
   425→      queryClient.invalidateQueries({ queryKey: ['email-templates', 'list', 'platform'] });
   426→      queryClient.invalidateQueries({ queryKey: ['email-templates', 'category-counts', 'platform'] });
   427→    } else {
   428→      queryClient.invalidateQueries({ queryKey: queryKeys.emailTemplates.all });
   429→      queryClient.invalidateQueries({ queryKey: ['email-templates', 'category-counts', 'client'] });
   430→    }
   431→  }, [queryClient, isPlatform]);
   432→
   433→  const duplicateMutation = useMutation({
   434→    mutationFn: (id: string) => postApi(`/api/email-templates/${id}/duplicate`),
   435→    onSuccess: () => {
   436→      toast.success('Template duplicated successfully');
   437→      invalidateAll();
   438→    },
   439→    onError: (err: Error) => {
   440→      toast.error(err.message || 'Failed to duplicate template');
   441→    },
   442→  });
   443→
   444→  const toggleStatusMutation = useMutation({
   445→    mutationFn: ({ id, status }: { id: string; status: EmailTemplateStatus }) =>
   446→      patchApi(`/api/email-templates/${id}`, { status }),
   447→    onSuccess: () => {
   448→      invalidateAll();
   449→    },
   450→    onError: (err: Error) => {
   451→      toast.error(err.message || 'Failed to update template status');
   452→    },
   453→  });
   454→
   455→  const deleteMutation = useMutation({
   456→    mutationFn: (id: string) => deleteApi(`/api/email-templates/${id}`),
   457→    onSuccess: () => {
   458→      toast.success('Template deleted');
   459→      invalidateAll();
   460→      setDeleteTarget(null);
   461→    },
   462→    onError: (err: Error) => {
   463→      toast.error(err.message || 'Failed to delete template');
   464→    },
   465→  });
   466→
   467→  const revertMutation = useMutation({
   468→    mutationFn: (id: string) => postApi(`/api/email-templates/${id}/revert`),
   469→    onSuccess: () => {
   470→      toast.success('Template reverted to default');
   471→      invalidateAll();
   472→      setRevertTarget(null);
   473→    },
   474→    onError: (err: Error) => {
   475→      toast.error(err.message || 'Failed to revert template');
   476→    },
   477→  });
   478→
   479→  const seedMutation = useMutation({
   480→    mutationFn: () => postApi<{ seeded: number; skipped: number; total: number }>(
   481→      '/api/email-templates/seed',
   482→      // Platform scope is guarded by requirePlatformAdmin on the server.
   483→      isPlatform ? { scope: 'platform' } : {},
   484→    ),
   485→    onSuccess: (res: any) => {
   486→      const data = res?.data ?? res ?? { seeded: 0, skipped: 0, total: 0 };
   487→      const { seeded, skipped } = data;
   488→      invalidateAll();
   489→      if (seeded > 0) {
   490→        toast.success(`${seeded} default template${seeded > 1 ? 's' : ''} created${skipped > 0 ? `, ${skipped} already existed` : ''}`);
   491→      } else if (skipped > 0) {
   492→        toast.info('All default email templates are already available.');
   493→      } else {
   494→        toast.info('No default templates to seed.');
   495→      }
   496→      setSeedDialogOpen(false);
   497→    },
   498→    onError: (err: Error) => {
   499→      toast.error(err.message || 'Failed to seed templates');
   500→    },
   501→  });
   502→
   503→  // -------------------- Navigation Handlers --------------------
   504→  // Platform scope navigates under 'platform-email-templates' so the URL
   505→  // hash is #platform-email-templates/new and the platform module router
   506→  // picks it up. Client scope uses the legacy 'email-templates' module.
   507→  const handleCreate = useCallback(() => {
   508→    navigate(moduleName, 'new');
   509→  }, [navigate, moduleName]);
   510→
   511→  // -------------------- Render --------------------
   512→  // Scope-aware page header. Platform scope uses PlatformPageHeader (keeps
   513→  // the PLATFORM badge) with platform-level copy; client scope uses the
   514→  // legacy PageHeader with the original copy.
   515→  //
   516→  // Platform scope intentionally hides the "Seed Defaults" manual-seeding
   517→  // action — platform system templates are created through the application
   518→  // bootstrap / migration logic, not via an admin-facing button. The client
   519→  // scope keeps the button (it seeds CMS article/comment templates).
   520→  const headerAction = (
   521→    <div className="flex flex-col items-end gap-2">
   522→      <Button size="sm" onClick={handleCreate}>
   523→        <Plus className="h-4 w-4 mr-2" />
   524→        Create Template
   525→      </Button>
   526→      {!isPlatform && (
   527→        <Button
   528→          size="sm"
   529→          variant="outline"
   530→          onClick={() => setSeedDialogOpen(true)}
   531→          disabled={seedMutation.isPending}
   532→          className="border-amber-300 text-amber-700 hover:bg-amber-50 hover:text-amber-800 dark:border-amber-700 dark:text-amber-400 dark:hover:bg-amber-950/40"
   533→        >
   534→          {seedMutation.isPending ? (
   535→            <Loader2 className="h-3.5 w-3.5 mr-1.5 animate-spin" />
   536→          ) : (
   537→            <Cpu className="h-3.5 w-3.5 mr-1.5" />
   538→          )}
   539→          Seed Defaults
   540→        </Button>
   541→      )}
   542→    </div>
   543→  );
   544→
   545→  return (
   546→    <div className="space-y-6">
   547→      {/* Page Header */}
   548→      {isPlatform ? (
   549→        <PlatformPageHeader
   550→          title="Email Templates"
   551→          subtitle="Platform-wide system email templates: welcome, payment, subscription, trial, invoice, and account lifecycle. Visible to all customers and sites."
   552→          actions={headerAction}
   553→        />
   554→      ) : (
   555→        <PageHeader
   556→          title="Email Templates"
   557→          description="Manage email templates for all system notifications, newsletters, and transactional emails."
   558→          action={headerAction}
   559→        />
   560→      )}
   561→
   562→      {/* Category Tabs */}
   563→      <div className="border-b">
   564→        <ScrollArea className="w-full">
   565→          <div className="flex items-center gap-1 pb-px overflow-x-auto">
   566→            {CATEGORIES.map((cat) => {
   567→              const isActive = category === cat.value;
   568→              const count =
   569→                cat.value === 'ALL'
   570→                  ? totalCount
   571→                  : (categoryCounts?.[cat.value] ?? 0);
   572→
   573→              return (
   574→                <button
   575→                  key={cat.value}
   576→                  onClick={() => handleCategoryChange(cat.value)}
   577→                  className={cn(
   578→                    'relative flex items-center gap-1.5 whitespace-nowrap px-3 py-2.5 text-sm font-medium transition-colors border-b-2 -mb-px',
   579→                    isActive
   580→                      ? 'border-primary text-foreground'
   581→                      : 'border-transparent text-muted-foreground hover:text-foreground hover:border-border',
   582→                  )}
   583→                >
   584→                  {cat.label}
   585→                  <span
   586→                    className={cn(
   587→                      'inline-flex h-5 min-w-5 items-center justify-center rounded-full px-1.5 text-[11px] font-medium leading-none',
   588→                      isActive
   589→                        ? 'bg-primary/10 text-primary'
   590→                        : 'bg-muted text-muted-foreground',
   591→                    )}
   592→                  >
   593→                    {count}
   594→                  </span>
   595→                </button>
   596→              );
   597→            })}
   598→          </div>
   599→        </ScrollArea>
   600→      </div>
   601→
   602→      {/* Search & Filter Bar */}
   603→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   604→        <div className="relative flex-1 max-w-sm">
   605→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   606→          <Input
   607→            placeholder="Search templates..."
   608→            value={searchInput}
   609→            onChange={(e) => updateSearch(e.target.value)}
   610→            className="pl-9 h-9"
   611→          />
   612→        </div>
   613→        <div className="flex items-center gap-2">
   614→          <Select value={statusFilter} onValueChange={handleStatusChange}>
   615→            <SelectTrigger size="sm" className="w-[150px]">
   616→              <SelectValue placeholder="All Statuses" />
   617→            </SelectTrigger>
   618→            <SelectContent>
   619→              {STATUS_OPTIONS.map((s) => (
   620→                <SelectItem key={s.value} value={s.value}>
   621→                  {s.label}
   622→                </SelectItem>
   623→              ))}
   624→            </SelectContent>
   625→          </Select>
   626→          <Select value={sortValue} onValueChange={handleSortChange}>
   627→            <SelectTrigger size="sm" className="w-[130px]">
   628→              <SelectValue placeholder="Newest" />
   629→            </SelectTrigger>
   630→            <SelectContent>
   631→              {SORT_OPTIONS.map((s) => (
   632→                <SelectItem key={s.value} value={s.value}>
   633→                  {s.label}
   634→                </SelectItem>
   635→              ))}
   636→            </SelectContent>
   637→          </Select>
   638→        </div>
   639→      </div>
   640→
   641→      {/* Table */}
   642→      <div className="rounded-lg border bg-card">
   643→        {isLoading ? (
   644→          <div className="p-6 space-y-3">
   645→            {Array.from({ length: 5 }).map((_, i) => (
   646→              <div key={i} className="flex items-center gap-4">
   647→                <Skeleton className="h-5 w-48" />
   648→                <Skeleton className="h-5 w-28" />
   649→                <Skeleton className="h-5 w-20" />
   650→                <Skeleton className="h-5 w-20" />
   651→                <Skeleton className="h-5 w-32" />
   652→                <Skeleton className="h-8 w-8 ml-auto" />
   653→              </div>
   654→            ))}
   655→          </div>
   656→        ) : templates.length === 0 ? (
   657→          <EmptyState
   658→            icon={Mail}
   659→            title="No templates found"
   660→            description={
   661→              search || statusFilter !== 'ALL' || category !== 'ALL'
   662→                ? 'Try adjusting your filters to find what you\'re looking for.'
   663→                : isPlatform
   664→                  ? 'Create your first platform template to get started.'
   665→                  : 'Create your first email template or seed the defaults to get started.'
   666→            }
   667→            action={
   668→              !search && statusFilter === 'ALL' && category === 'ALL' && !isPlatform
   669→                ? {
   670→                    label: 'Seed Default Templates',
   671→                    onClick: () => seedMutation.mutate(),
   672→                  }
   673→                : undefined
   674→            }
   675→          />
   676→        ) : (
   677→          <>
   678→            <Table>
   679→              <TableHeader>
   680→                <TableRow>
   681→                  <TableHead className="pl-4">Template Name</TableHead>
   682→                  <TableHead>Category</TableHead>
   683→                  <TableHead>Status</TableHead>
   684→                  <TableHead>Provider</TableHead>
   685→                  <TableHead>Last Updated</TableHead>
   686→                  <TableHead className="pr-4 w-12"></TableHead>
   687→                </TableRow>
   688→              </TableHeader>
   689→              <TableBody>
   690→                {templates.map((template) => (
   691→                  <TableRow
   692→                    key={template.id}
   693→                    className="cursor-pointer group"
   694→                    onClick={() => onEdit(template.id)}
   695→                  >
   696→                    {/* Template Name */}
   697→                    <TableCell className="pl-4">
   698→                      <div className="flex items-center gap-2 min-w-0">
   699→                        <span className="truncate font-medium text-foreground">
   700→                          {template.name}
   701→                        </span>
   702→                        {template.isSystem && (
   703→                          <Badge
   704→                            variant="outline"
   705→                            className="shrink-0 px-1.5 py-0 text-[10px] font-medium text-muted-foreground border-dashed"
   706→                          >
   707→                            System
   708→                          </Badge>
   709→                        )}
   710→                      </div>
   711→                    </TableCell>
   712→
   713→                    {/* Category */}
   714→                    <TableCell>
   715→                      <StatusBadge
   716→                        status={template.category}
   717→                        size="sm"
   718→                      />
   719→                    </TableCell>
   720→
   721→                    {/* Status */}
   722→                    <TableCell>
   723→                      <Badge
   724→                        variant="outline"
   725→                        className={cn(
   726→                          'border-transparent font-medium shrink-0 px-2 py-0.5 text-xs',
   727→                          getStatusColor(template.status),
   728→                        )}
   729→                      >
   730→                        {getStatusLabel(template.status)}
   731→                      </Badge>
   732→                    </TableCell>
   733→
   734→                    {/* Provider */}
   735→                    <TableCell>
   736→                      <Badge
   737→                        variant="outline"
   738→                        className={cn(
   739→                          'border-transparent shrink-0 px-2 py-0.5 text-xs font-medium',
   740→                          PROVIDER_COLORS[template.provider] ??
   741→                            'bg-zinc-100 text-zinc-700 dark:bg-zinc-800 dark:text-zinc-300',
   742→                        )}
   743→                      >
   744→                        {template.provider}
   745→                      </Badge>
   746→                    </TableCell>
   747→
   748→                    {/* Last Updated */}
   749→                    <TableCell className="text-muted-foreground text-sm">
   750→                      {formatRelativeTime(template.updatedAt)}
   751→                    </TableCell>
   752→
   753→                    {/* Actions */}
   754→                    <TableCell className="pr-4" onClick={(e) => e.stopPropagation()}>
   755→                      <DropdownMenu>
   756→                        <DropdownMenuTrigger asChild>
   757→                          <Button
   758→                            variant="ghost"
   759→                            size="icon"
   760→                            className="h-8 w-8 opacity-0 group-hover:opacity-100 transition-opacity"
   761→                          >
   762→                            <MoreHorizontal className="h-4 w-4" />
   763→                          </Button>
   764→                        </DropdownMenuTrigger>
   765→                        <DropdownMenuContent align="end" className="w-48">
   766→                          <DropdownMenuItem onClick={() => onEdit(template.id)}>
   767→                            <Pencil className="h-4 w-4 mr-2" />
   768→                            Edit
   769→                          </DropdownMenuItem>
   770→                          <DropdownMenuItem onClick={() => onPreview(template.id)}>
   771→                            <Eye className="h-4 w-4 mr-2" />
   772→                            Preview
   773→                          </DropdownMenuItem>
   774→                          <DropdownMenuItem
   775→                            onClick={() => duplicateMutation.mutate(template.id)}
   776→                            disabled={duplicateMutation.isPending}
   777→                          >
   778→                            <Copy className="h-4 w-4 mr-2" />
   779→                            Duplicate
   780→                          </DropdownMenuItem>
   781→                          <DropdownMenuItem
   782→                            onClick={() => setSendTestTarget(template.id)}
   783→                          >
   784→                            <Send className="h-4 w-4 mr-2" />
   785→                            Send Test
   786→                          </DropdownMenuItem>
   787→                          <DropdownMenuSeparator />
   788→                          <DropdownMenuItem
   789→                            onClick={() =>
   790→                              toggleStatusMutation.mutate({
   791→                                id: template.id,
   792→                                status:
   793→                                  template.status === 'ENABLED'
   794→                                    ? 'DISABLED'
   795→                                    : 'ENABLED',
   796→                              })
   797→                            }
   798→                            disabled={toggleStatusMutation.isPending}
   799→                          >
   800→                            {template.status === 'ENABLED' ? (
   801→                              <>
   802→                                <span className="h-4 w-4 mr-2 inline-flex items-center justify-center rounded-full bg-zinc-200 dark:bg-zinc-700">
   803→                                  <span className="h-2 w-2 rounded-full" />
   804→                                </span>
   805→                                Disable
   806→                              </>
   807→                            ) : (
   808→                              <>
   809→                                <span className="h-4 w-4 mr-2 inline-flex items-center justify-center rounded-full bg-emerald-200 dark:bg-emerald-800">
   810→                                  <span className="h-2 w-2 rounded-full" />
   811→                                </span>
   812→                                Enable
   813→                              </>
   814→                            )}
   815→                          </DropdownMenuItem>
   816→                          {template.isSystem && (
   817→                            <DropdownMenuItem
   818→                              onClick={() => setRevertTarget(template)}
   819→                              className="text-amber-600 dark:text-amber-400"
   820→                            >
   821→                              <RotateCcw className="h-4 w-4 mr-2" />
   822→                              Revert to Default
   823→                            </DropdownMenuItem>
   824→                          )}
   825→                          <DropdownMenuSeparator />
   826→                          <DropdownMenuItem
   827→                            variant="destructive"
   828→                            onClick={() => setDeleteTarget(template)}
   829→                          >
   830→                            <Trash2 className="h-4 w-4 mr-2" />
   831→                            Delete
   832→                          </DropdownMenuItem>
   833→                        </DropdownMenuContent>
   834→                      </DropdownMenu>
   835→                    </TableCell>
   836→                  </TableRow>
   837→                ))}
   838→              </TableBody>
   839→            </Table>
   840→
   841→            {/* Pagination */}
   842→            {pagination && pagination.totalPages > 1 && (
   843→              <div className="flex items-center justify-between border-t px-4 py-3">
   844→                <div className="flex items-center gap-2 text-sm text-muted-foreground">
   845→                  <span>
   846→                    Page {pagination.page} of {pagination.totalPages}
   847→                  </span>
   848→                  <span className="text-muted-foreground/50">·</span>
   849→                  <span>{pagination.total} total</span>
   850→                </div>
   851→                <div className="flex items-center gap-3">
   852→                  <Select
   853→                    value={String(pageSize)}
   854→                    onValueChange={(v) => {
   855→                      setPageSize(Number(v));
   856→                      setPage(1);
   857→                    }}
   858→                  >
   859→                    <SelectTrigger size="sm" className="w-[80px] h-8">
   860→                      <SelectValue />
   861→                    </SelectTrigger>
   862→                    <SelectContent>
   863→                      {PAGE_SIZES.map((s) => (
   864→                        <SelectItem key={s} value={String(s)}>
   865→                          {s}
   866→                        </SelectItem>
   867→                      ))}
   868→                    </SelectContent>
   869→                  </Select>
   870→                  <div className="flex items-center gap-1">
   871→                    <Button
   872→                      variant="outline"
   873→                      size="icon"
   874→                      className="h-8 w-8"
   875→                      disabled={pagination.page <= 1}
   876→                      onClick={() => setPage((p) => p - 1)}
   877→                    >
   878→                      <ChevronLeft className="h-4 w-4" />
   879→                    </Button>
   880→                    <Button
   881→                      variant="outline"
   882→                      size="icon"
   883→                      className="h-8 w-8"
   884→                      disabled={pagination.page >= pagination.totalPages}
   885→                      onClick={() => setPage((p) => p + 1)}
   886→                    >
   887→                      <ChevronRight className="h-4 w-4" />
   888→                    </Button>
   889→                  </div>
   890→                </div>
   891→              </div>
   892→            )}
   893→          </>
   894→        )}
   895→      </div>
   896→
   897→      {/* Send Test Dialog */}
   898→      <SendTestDialog
   899→        open={!!sendTestTarget}
   900→        onOpenChange={(v) => !v && setSendTestTarget(null)}
   901→        templateId={sendTestTarget}
   902→      />
   903→
   904→      {/* Delete Confirm Dialog */}
   905→      <ConfirmDialog
   906→        open={!!deleteTarget}
   907→        onOpenChange={(v) => !v && setDeleteTarget(null)}
   908→        title="Delete Template"
   909→        description={
   910→          deleteTarget
   911→            ? `Are you sure you want to delete "${deleteTarget.name}"? This action cannot be undone.`
   912→            : undefined
   913→        }
   914→        confirmLabel="Delete"
   915→        variant="destructive"
   916→        onConfirm={() => {
   917→          if (deleteTarget) deleteMutation.mutate(deleteTarget.id);
   918→        }}
   919→        isLoading={deleteMutation.isPending}
   920→      />
   921→
   922→      {/* Revert Confirm Dialog */}
   923→      <ConfirmDialog
   924→        open={!!revertTarget}
   925→        onOpenChange={(v) => !v && setRevertTarget(null)}
   926→        title="Revert to Default"
   927→        description={
   928→          revertTarget
   929→            ? `This will reset "${revertTarget.name}" to its original default content. Your current changes will be saved as a version before reverting.`
   930→            : undefined
   931→        }
   932→        confirmLabel="Revert"
   933→        onConfirm={() => {
   934→          if (revertTarget) revertMutation.mutate(revertTarget.id);
   935→        }}
   936→        isLoading={revertMutation.isPending}
   937→      />
   938→
   939→      {/* Seed Defaults Confirm Dialog — platform scope hides this since the
   940→          platform page no longer exposes the Seed Defaults action (system
   941→          templates are seeded via bootstrap/migration). The dialog remains
   942→          available for client scope. */}
   943→      {!isPlatform && (
   944→        <ConfirmDialog
   945→          open={seedDialogOpen}
   946→          onOpenChange={setSeedDialogOpen}
   947→          title="Seed Default Templates"
   948→          description="This will add any missing default system email templates. Existing templates (including your custom ones) will not be modified."
   949→          confirmLabel="Seed Defaults"
   950→          onConfirm={() => seedMutation.mutate()}
   951→          isLoading={seedMutation.isPending}
   952→        />
   953→      )}
   954→    </div>
   955→  );
   956→}
