    51→  AlertDialogDescription,
    52→  AlertDialogFooter,
    53→  AlertDialogHeader,
    54→  AlertDialogTitle,
    55→} from '@/components/ui/alert-dialog';
    56→import {
    57→  Tooltip,
    58→  TooltipContent,
    59→  TooltipProvider,
    60→  TooltipTrigger,
    61→} from '@/components/ui/tooltip';

    83→// -------------------- Types --------------------
    84→
    85→interface AiProvider {
    86→  id: string;
    87→  name: string;
    88→  kind: AiProviderKind;
    89→  baseUrl: string;
    90→  apiKey: string | null;
    91→  apiVersion: string | null;
    92→  isActive: boolean;
    93→  isDefault: boolean;
    94→  connectionStatus: AiConnectionStatus;
    95→  latency: number | null;
    96→  lastSyncAt: string | null;
    97→  createdAt: string;
    98→  updatedAt: string;
    99→}
   100→
   101→interface ProviderFormData {
   102→  name: string;
   103→  kind: AiProviderKind;
   104→  baseUrl: string;
   105→  apiKey: string;
   106→  apiVersion: string;
   107→  isActive: boolean;
   108→}
   109→
   110→// -------------------- Constants --------------------
   111→
   112→const PROVIDER_KINDS: AiProviderKind[] = [
   113→  'OPENAI', 'ANTHROPIC', 'GEMINI', 'OPENROUTER', 'GROQ', 'DEEPSEEK', 'OLLAMA', 'AZURE_OPENAI',
   114→];
   115→
   116→const PROVIDER_CONFIGS: Record<AiProviderKind, { label: string; defaultUrl: string; color: string }> = {
   117→  OPENAI: { label: 'OpenAI', defaultUrl: 'https://api.openai.com/v1', color: 'bg-emerald-100 text-emerald-700' },
   118→  ANTHROPIC: { label: 'Anthropic', defaultUrl: 'https://api.anthropic.com/v1', color: 'bg-orange-100 text-orange-700' },
   119→  GEMINI: { label: 'Gemini', defaultUrl: 'https://generativelanguage.googleapis.com/v1', color: 'bg-sky-100 text-sky-700' },
   120→  OPENROUTER: { label: 'OpenRouter', defaultUrl: 'https://openrouter.ai/api/v1', color: 'bg-violet-100 text-violet-700' },
   121→  GROQ: { label: 'Groq', defaultUrl: 'https://api.groq.com/openai/v1', color: 'bg-rose-100 text-rose-700' },
   122→  DEEPSEEK: { label: 'DeepSeek', defaultUrl: 'https://api.deepseek.com/v1', color: 'bg-cyan-100 text-cyan-700' },
   123→  OLLAMA: { label: 'Ollama', defaultUrl: 'http://localhost:11434/v1', color: 'bg-zinc-100 text-zinc-700' },
   124→  AZURE_OPENAI: { label: 'Azure OpenAI', defaultUrl: '', color: 'bg-teal-100 text-teal-700' },
   125→};
   126→
   127→const CONNECTION_STATUS_CONFIG: Record<AiConnectionStatus, { color: string; label: string }> = {
   128→  CONNECTED: { color: 'bg-green-500', label: 'Connected' },
   129→  DISCONNECTED: { color: 'bg-zinc-400', label: 'Disconnected' },
   130→  ERROR: { color: 'bg-red-500', label: 'Error' },
   131→};
   132→
   133→const emptyForm: ProviderFormData = {
   134→  name: '',
   135→  kind: 'OPENAI',
   136→  baseUrl: PROVIDER_CONFIGS.OPENAI.defaultUrl,
   137→  apiKey: '',
   138→  apiVersion: '',
   139→  isActive: true,
   140→};
   141→
   142→// -------------------- Component --------------------
   143→
   144→export function ProvidersPage() {
   145→  const queryClient = useQueryClient();
   146→  const [page, setPage] = useState(1);
   147→  const [pageSize] = useState(25);
   148→  const [search, setSearch] = useState('');
   149→  const [kindFilter, setKindFilter] = useState<string>('all');
   150→  const [statusFilter, setStatusFilter] = useState<string>('all');
   151→  const [dialogOpen, setDialogOpen] = useState(false);
   152→  const [editingProvider, setEditingProvider] = useState<AiProvider | null>(null);
   153→  const [formData, setFormData] = useState<ProviderFormData>(emptyForm);
   154→  const [showApiKey, setShowApiKey] = useState(false);
   155→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   156→  const [deletingProvider, setDeletingProvider] = useState<AiProvider | null>(null);
   157→
   158→  // Fetch providers
   159→  const { data, isLoading, isError } = useQuery({
   160→    queryKey: queryKeys.aiProviders.list({ page, pageSize, search, kind: kindFilter, connectionStatus: statusFilter }),
   161→    queryFn: () => getApi<PaginatedResponse<AiProvider>>('/api/ai/providers', {
   162→      page,
   163→      pageSize,
   164→      search: search || undefined,
   165→      kind: kindFilter !== 'all' ? kindFilter : undefined,
   166→      connectionStatus: statusFilter !== 'all' ? statusFilter : undefined,
   167→    }),
   168→  });
   169→
   170→  const providers = data?.data ?? [];
   171→  const pagination = data?.pagination;
   172→
   173→  // Create / Update mutation
   174→  const saveMutation = useMutation({
   175→    mutationFn: (body: ProviderFormData) => {
   176→      if (editingProvider) {
   177→        const patchBody: Record<string, unknown> = { ...body };
   178→        if (!patchBody.apiKey) delete patchBody.apiKey;
   179→        return patchApi<AiProvider>(`/api/ai/providers/${editingProvider.id}`, patchBody);
   180→      }
   181→      return postApi<AiProvider>('/api/ai/providers', body);
   182→    },
   183→    onSuccess: () => {
   184→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   185→      toast.success(editingProvider ? 'Provider updated' : 'Provider created');
   186→      handleCloseDialog();
   187→    },
   188→    onError: (err: Error) => {
   189→      toast.error(err.message || 'Failed to save provider');
   190→    },
   191→  });
   192→
   193→  // Delete mutation
   194→  const deleteMutation = useMutation({
   195→    mutationFn: (id: string) => deleteApi(`/api/ai/providers/${id}`),
   196→    onSuccess: () => {
   197→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   198→      toast.success('Provider deleted');
   199→      setDeleteDialogOpen(false);
   200→      setDeletingProvider(null);
   201→    },
   202→    onError: (err: Error) => {
   203→      toast.error(err.message || 'Failed to delete provider');
   204→    },
   205→  });
   206→
   207→  // Test connection mutation
   208→  const testMutation = useMutation({
   209→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/test`),
   210→    onSuccess: (result: unknown) => {
   211→      const res = result as { success?: boolean; latency?: number; message?: string };
   212→      if (res.success) {
   213→        toast.success(`Connection successful (${res.latency ?? 0}ms)`);
   214→      } else {
   215→        toast.error(res.message || 'Connection failed');
   216→      }
   217→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   218→    },
   219→    onError: (err: Error) => {
   220→      toast.error(err.message || 'Connection test failed');
   221→    },
   222→  });
   223→
   224→  // Sync models mutation
   225→  const syncMutation = useMutation({
   226→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/sync-models`),
   227→    onSuccess: (result: unknown) => {
   228→      const res = result as { count?: number };
   229→      toast.success(`Synced ${res.count ?? 0} models`);
   230→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   231→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   232→    },
   233→    onError: (err: Error) => {
   234→      toast.error(err.message || 'Failed to sync models');
   235→    },
   236→  });
   237→
   238→  // Set default mutation
   239→  const setDefaultMutation = useMutation({
   240→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/set-default`),
   241→    onSuccess: () => {
   242→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   243→      toast.success('Default provider updated');
   244→    },
   245→    onError: (err: Error) => {
   246→      toast.error(err.message || 'Failed to set default');
   247→    },
   248→  });
   249→
   250→  // Toggle active mutation
   251→  const toggleActiveMutation = useMutation({
   252→    mutationFn: ({ id, isActive }: { id: string; isActive: boolean }) =>
   253→      patchApi(`/api/ai/providers/${id}`, { isActive }),
   254→    onSuccess: () => {
   255→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   256→    },
   257→    onError: (err: Error) => {
   258→      toast.error(err.message || 'Failed to update provider');
   259→    },
   260→  });
   261→
   262→  // Handlers
   263→  const handleOpenCreate = () => {
   264→    setEditingProvider(null);
   265→    setFormData(emptyForm);
   266→    setShowApiKey(false);
   267→    setDialogOpen(true);
   268→  };
   269→
   270→  const handleOpenEdit = (provider: AiProvider) => {
   271→    setEditingProvider(provider);
   272→    setFormData({
   273→      name: provider.name,
   274→      kind: provider.kind,
   275→      baseUrl: provider.baseUrl,
   276→      apiKey: '', // masked in edit
   277→      apiVersion: provider.apiVersion ?? '',
   278→      isActive: provider.isActive,
   279→    });
   280→    setShowApiKey(false);
   281→    setDialogOpen(true);
   282→  };
   283→
   284→  const handleCloseDialog = () => {
   285→    setDialogOpen(false);
   286→    setEditingProvider(null);
   287→    setFormData(emptyForm);
   288→    setShowApiKey(false);
   289→  };
   290→
   291→  const handleSave = () => {
   292→    if (!formData.name.trim()) {
   293→      toast.error('Name is required');
   294→      return;
   295→    }
   296→    saveMutation.mutate(formData);
   297→  };
   298→
   299→  const handleKindChange = (kind: string) => {
   300→    const providerKind = kind as AiProviderKind;
   301→    setFormData((prev) => ({
   302→      ...prev,
   303→      kind: providerKind,
   304→      baseUrl: PROVIDER_CONFIGS[providerKind].defaultUrl,
   305→    }));
   306→  };
   307→
   308→  // KPI calculations
   309→  const totalProviders = pagination?.total ?? 0;
   310→  const connectedCount = providers.filter((p) => p.connectionStatus === 'CONNECTED').length;
   311→  const defaultProvider = providers.find((p) => p.isDefault);
   312→
   313→  return (
   314→    <div className="space-y-6">
   315→      {/* KPI Cards */}
   316→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   317→        <Card>
   318→          <CardContent className="p-4 flex items-center gap-4">
   319→            <div className="p-2 rounded-lg bg-zinc-100">
   320→              <Server className="h-5 w-5 text-zinc-600" />
   321→            </div>
   322→            <div>
   323→              <p className="text-sm text-zinc-500">Total Providers</p>
   324→              <div className="text-2xl font-bold">{isLoading ? <Skeleton className="h-8 w-12 inline-block" /> : totalProviders}</div>
   325→            </div>
   326→          </CardContent>
   327→        </Card>
   328→        <Card>
   329→          <CardContent className="p-4 flex items-center gap-4">
   330→            <div className="p-2 rounded-lg bg-green-100">
   331→              <Wifi className="h-5 w-5 text-green-600" />
   332→            </div>
   333→            <div>
   334→              <p className="text-sm text-zinc-500">Connected</p>
   335→              <div className="text-2xl font-bold">{isLoading ? <Skeleton className="h-8 w-12 inline-block" /> : connectedCount}</div>
   336→            </div>
   337→          </CardContent>
   338→        </Card>
   339→        <Card>
   340→          <CardContent className="p-4 flex items-center gap-4">
   341→            <div className="p-2 rounded-lg bg-amber-100">
   342→              <Star className="h-5 w-5 text-amber-600" />
   343→            </div>
   344→            <div>
   345→              <p className="text-sm text-zinc-500">Default Provider</p>
   346→              <div className="text-lg font-bold truncate max-w-[200px]">{isLoading ? <Skeleton className="h-6 w-32 inline-block" /> : defaultProvider?.name ?? 'None'}</div>
   347→            </div>
   348→          </CardContent>
   349→        </Card>
   350→      </div>
   351→
   352→      {/* Filters & Actions */}
   353→      <Card>
   354→        <CardHeader className="pb-4">
   355→          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   356→            <CardTitle className="text-lg">AI Providers</CardTitle>
   357→            <Button onClick={handleOpenCreate} size="sm">
   358→              <Plus className="h-4 w-4 mr-2" />
   359→              Add Provider
   360→            </Button>
   361→          </div>
   362→          <div className="flex flex-col sm:flex-row gap-3 mt-2">
   363→            <div className="relative flex-1">
   364→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-zinc-400" />
   365→              <Input
   366→                placeholder="Search providers..."
   367→                value={search}
   368→                onChange={(e) => { setSearch(e.target.value); setPage(1); }}
   369→                className="pl-9"
   370→              />
   371→            </div>
   372→            <Select value={kindFilter} onValueChange={(v) => { setKindFilter(v); setPage(1); }}>
   373→              <SelectTrigger className="w-full sm:w-[180px]">
   374→                <SelectValue placeholder="Provider Kind" />
   375→              </SelectTrigger>
   376→              <SelectContent>
   377→                <SelectItem value="all">All Kinds</SelectItem>
   378→                {PROVIDER_KINDS.map((k) => (
   379→                  <SelectItem key={k} value={k}>{PROVIDER_CONFIGS[k].label}</SelectItem>
   380→                ))}
   381→              </SelectContent>
   382→            </Select>
   383→            <Select value={statusFilter} onValueChange={(v) => { setStatusFilter(v); setPage(1); }}>
   384→              <SelectTrigger className="w-full sm:w-[180px]">
   385→                <SelectValue placeholder="Status" />
   386→              </SelectTrigger>
   387→              <SelectContent>
   388→                <SelectItem value="all">All Status</SelectItem>
   389→                <SelectItem value="CONNECTED">Connected</SelectItem>
   390→                <SelectItem value="DISCONNECTED">Disconnected</SelectItem>
   391→                <SelectItem value="ERROR">Error</SelectItem>
   392→              </SelectContent>
   393→            </Select>
   394→          </div>
   395→        </CardHeader>
   396→        <CardContent className="p-0">
   397→          <ScrollArea className="max-h-[500px]">
   398→            <Table>
   399→              <TableHeader>
   400→                <TableRow>
   401→                  <TableHead>Name</TableHead>
   402→                  <TableHead>Kind</TableHead>
   403→                  <TableHead>Status</TableHead>
   404→                  <TableHead className="hidden md:table-cell">Latency</TableHead>
   405→                  <TableHead className="hidden lg:table-cell">Last Sync</TableHead>
   406→                  <TableHead>Default</TableHead>
   407→                  <TableHead>Active</TableHead>
   408→                  <TableHead className="text-right">Actions</TableHead>
   409→                </TableRow>
   410→              </TableHeader>
   411→              <TableBody>
   412→                {isLoading ? (
   413→                  Array.from({ length: 5 }).map((_, i) => (
   414→                    <TableRow key={i}>
   415→                      {Array.from({ length: 8 }).map((_, j) => (
   416→                        <TableCell key={j}><Skeleton className="h-5 w-20" /></TableCell>
   417→                      ))}
   418→                    </TableRow>
   419→                  ))
   420→                ) : isError ? (
   421→                  <TableRow>
   422→                    <TableCell colSpan={8} className="text-center py-8 text-zinc-500">
   423→                      <AlertCircle className="h-8 w-8 mx-auto mb-2 text-red-400" />
   424→                      Failed to load providers
   425→                    </TableCell>
   426→                  </TableRow>
   427→                ) : providers.length === 0 ? (
   428→                  <TableRow>
   429→                    <TableCell colSpan={8} className="text-center py-8 text-zinc-500">
   430→                      <Server className="h-8 w-8 mx-auto mb-2 text-zinc-300" />
   431→                      No providers found. Click &quot;Add Provider&quot; to get started.
   432→                    </TableCell>
   433→                  </TableRow>
   434→                ) : (
   435→                  providers.map((provider) => {
   436→                    const kindConfig = PROVIDER_CONFIGS[provider.kind];
   437→                    const statusConfig = CONNECTION_STATUS_CONFIG[provider.connectionStatus];
   438→                    return (
   439→                      <TableRow key={provider.id}>
   440→                        <TableCell className="font-medium">{provider.name}</TableCell>
   441→                        <TableCell>
   442→                          <Badge variant="secondary" className={kindConfig.color}>
   443→                            {kindConfig.label}
   444→                          </Badge>
   445→                        </TableCell>
   446→                        <TableCell>
   447→                          <div className="flex items-center gap-2">
   448→                            <span className={`h-2.5 w-2.5 rounded-full ${statusConfig.color}`} />
   449→                            <span className="text-sm">{statusConfig.label}</span>
   450→                          </div>
   451→                        </TableCell>
   452→                        <TableCell className="hidden md:table-cell">
   453→                          {provider.latency != null ? `${provider.latency}ms` : '—'}
   454→                        </TableCell>
   455→                        <TableCell className="hidden lg:table-cell text-sm text-zinc-500">
   456→                          {provider.lastSyncAt
   457→                            ? new Date(provider.lastSyncAt).toLocaleDateString()
   458→                            : '—'}
   459→                        </TableCell>
   460→                        <TableCell>
   461→                          {provider.isDefault && (
   462→                            <Badge variant="secondary" className="bg-amber-100 text-amber-700">
   463→                              Default
   464→                            </Badge>
   465→                          )}
   466→                        </TableCell>
   467→                        <TableCell>
   468→                          <Switch
   469→                            checked={provider.isActive}
   470→                            onCheckedChange={(checked) =>
   471→                              toggleActiveMutation.mutate({ id: provider.id, isActive: checked })
   472→                            }
   473→                          />
   474→                        </TableCell>
   475→                        <TableCell className="text-right">
   476→                          <DropdownMenu>
   477→                            <DropdownMenuTrigger asChild>
   478→                              <Button variant="ghost" size="sm">
   479→                                <MoreHorizontal className="h-4 w-4" />
   480→                              </Button>
   481→                            </DropdownMenuTrigger>
   482→                            <DropdownMenuContent align="end">
   483→                              <DropdownMenuItem onClick={() => testMutation.mutate(provider.id)} disabled={testMutation.isPending}>
   484→                                <Zap className="h-4 w-4 mr-2" />
   485→                                Test Connection
   486→                              </DropdownMenuItem>
   487→                              <DropdownMenuItem onClick={() => syncMutation.mutate(provider.id)} disabled={syncMutation.isPending}>
   488→                                <RefreshCw className="h-4 w-4 mr-2" />
   489→                                Sync Models
   490→                              </DropdownMenuItem>
   491→                              {!provider.isDefault && (
   492→                                <DropdownMenuItem onClick={() => setDefaultMutation.mutate(provider.id)}>
   493→                                  <Star className="h-4 w-4 mr-2" />
   494→                                  Set Default
   495→                                </DropdownMenuItem>
   496→                              )}
   497→                              <DropdownMenuSeparator />
   498→                              <DropdownMenuItem onClick={() => handleOpenEdit(provider)}>
   499→                                <Pencil className="h-4 w-4 mr-2" />
   500→                                Edit
   501→                              </DropdownMenuItem>
   502→                              <DropdownMenuItem
   503→                                className="text-red-600"
   504→                                onClick={() => { setDeletingProvider(provider); setDeleteDialogOpen(true); }}
   505→                              >
   506→                                <Trash2 className="h-4 w-4 mr-2" />
   507→                                Delete
   508→                              </DropdownMenuItem>
   509→                            </DropdownMenuContent>
   510→                          </DropdownMenu>
   511→                        </TableCell>
   512→                      </TableRow>
   513→                    );
   514→                  })
   515→                )}
   516→              </TableBody>
   517→            </Table>
   518→            <ScrollBar />
   519→          </ScrollArea>
   520→
   521→          {/* Pagination */}
   522→          {pagination && pagination.totalPages > 1 && (
   523→            <div className="flex items-center justify-between px-4 py-3 border-t">
   524→              <p className="text-sm text-zinc-500">
   525→                Showing {(pagination.page - 1) * pagination.pageSize + 1}–
   526→                {Math.min(pagination.page * pagination.pageSize, pagination.total)} of {pagination.total}
   527→              </p>
   528→              <div className="flex items-center gap-2">
   529→                <Button
   530→                  variant="outline" size="sm"
   531→                  disabled={pagination.page <= 1}
   532→                  onClick={() => setPage((p) => p - 1)}
   533→                >
   534→                  <ChevronLeft className="h-4 w-4" />
   535→                </Button>
   536→                <span className="text-sm">Page {pagination.page} of {pagination.totalPages}</span>
   537→                <Button
   538→                  variant="outline" size="sm"
   539→                  disabled={pagination.page >= pagination.totalPages}
   540→                  onClick={() => setPage((p) => p + 1)}
   541→                >
   542→                  <ChevronRight className="h-4 w-4" />
   543→                </Button>
   544→              </div>
   545→            </div>
   546→          )}
   547→        </CardContent>
   548→      </Card>
   549→
   550→      {/* Add/Edit Dialog */}
   551→      <Dialog open={dialogOpen} onOpenChange={(open) => { if (!open) handleCloseDialog(); }}>
   552→        <DialogContent className="sm:max-w-[500px] max-h-[90vh] overflow-y-auto">
   553→          <DialogHeader>
   554→            <DialogTitle>{editingProvider ? 'Edit Provider' : 'Add Provider'}</DialogTitle>
   555→          </DialogHeader>
   556→          <div className="grid gap-4 py-4">
   557→            <div className="grid gap-2">
   558→              <Label htmlFor="provider-name">Name</Label>
   559→              <Input
   560→                id="provider-name"
   561→                placeholder="e.g. My OpenAI"
   562→                value={formData.name}
   563→                onChange={(e) => setFormData((p) => ({ ...p, name: e.target.value }))}
   564→              />
   565→            </div>
   566→            <div className="grid gap-2">
   567→              <Label htmlFor="provider-kind">Provider Kind</Label>
   568→              <Select value={formData.kind} onValueChange={handleKindChange}>
   569→                <SelectTrigger id="provider-kind">
   570→                  <SelectValue />
   571→                </SelectTrigger>
   572→                <SelectContent>
   573→                  {PROVIDER_KINDS.map((k) => (
   574→                    <SelectItem key={k} value={k}>{PROVIDER_CONFIGS[k].label}</SelectItem>
   575→                  ))}
   576→                </SelectContent>
   577→              </Select>
   578→            </div>
   579→            <div className="grid gap-2">
   580→              <Label htmlFor="provider-url">Base URL</Label>
   581→              <Input
   582→                id="provider-url"
   583→                placeholder="https://api.openai.com/v1"
   584→                value={formData.baseUrl}
   585→                onChange={(e) => setFormData((p) => ({ ...p, baseUrl: e.target.value }))}
   586→              />
   587→            </div>
   588→            <div className="grid gap-2">
   589→              <Label htmlFor="provider-key">API Key</Label>
   590→              <div className="relative">
   591→                <Input
   592→                  id="provider-key"
   593→                  type={showApiKey ? 'text' : 'password'}
   594→                  placeholder={editingProvider ? 'Leave blank to keep existing key' : 'sk-...'}
   595→                  value={formData.apiKey}
   596→                  onChange={(e) => setFormData((p) => ({ ...p, apiKey: e.target.value }))}
   597→                />
   598→                <Button
   599→                  type="button"
   600→                  variant="ghost"
   601→                  size="sm"
   602→                  className="absolute right-1 top-1/2 -translate-y-1/2 h-7 w-7 p-0"
   603→                  onClick={() => setShowApiKey(!showApiKey)}
   604→                >
   605→                  {showApiKey ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   606→                </Button>
   607→              </div>
   608→            </div>
   609→            {formData.kind === 'AZURE_OPENAI' && (
   610→              <div className="grid gap-2">
   611→                <Label htmlFor="provider-version">API Version</Label>
   612→                <Input
   613→                  id="provider-version"
   614→                  placeholder="2024-02-01"
   615→                  value={formData.apiVersion}
   616→                  onChange={(e) => setFormData((p) => ({ ...p, apiVersion: e.target.value }))}
   617→                />
   618→              </div>
   619→            )}
   620→            <div className="flex items-center justify-between">
   621→              <Label htmlFor="provider-active">Active</Label>
   622→              <Switch
   623→                id="provider-active"
   624→                checked={formData.isActive}
   625→                onCheckedChange={(checked) => setFormData((p) => ({ ...p, isActive: checked }))}
   626→              />
   627→            </div>
   628→          </div>
   629→          <DialogFooter>
   630→            <Button variant="outline" onClick={handleCloseDialog}>Cancel</Button>
   631→            <Button onClick={handleSave} disabled={saveMutation.isPending}>
   632→              {saveMutation.isPending && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   633→              {editingProvider ? 'Update' : 'Create'}
   634→            </Button>
   635→          </DialogFooter>
   636→        </DialogContent>
   637→      </Dialog>
   638→
   639→      {/* Delete Confirmation */}
   640→      <AlertDialog open={deleteDialogOpen} onOpenChange={setDeleteDialogOpen}>
   641→        <AlertDialogContent>
   642→          <AlertDialogHeader>
   643→            <AlertDialogTitle>Delete Provider</AlertDialogTitle>
   644→            <AlertDialogDescription>
   645→              Are you sure you want to delete &quot;{deletingProvider?.name}&quot;? This action cannot be undone.
   646→            </AlertDialogDescription>
   647→          </AlertDialogHeader>
   648→          <AlertDialogFooter>
   649→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   650→            <AlertDialogAction
   651→              className="bg-red-600 hover:bg-red-700"
   652→              onClick={() => deletingProvider && deleteMutation.mutate(deletingProvider.id)}
   653→              disabled={deleteMutation.isPending}
   654→            >
   655→              {deleteMutation.isPending && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   656→              Delete
   657→            </AlertDialogAction>
   658→          </AlertDialogFooter>
   659→        </AlertDialogContent>
   660→      </AlertDialog>
   661→    </div>
   662→  );
