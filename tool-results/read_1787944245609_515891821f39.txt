   272→// -------------------- Category Count Query --------------------
   273→
   274→function useCategoryCounts() {
   275→  return useQuery({
   276→    queryKey: ['email-templates', 'category-counts'],
   277→    queryFn: async () => {
   278→      const allCategories = CATEGORIES.filter((c) => c.value !== 'ALL');
   279→      const results = await Promise.allSettled(
   280→        allCategories.map(async (cat) => {
   281→          const res = await getApi<{ data: EmailTemplate[]; meta: { pagination: { total: number } } }>(
   282→            '/api/email-templates',
   283→            { category: cat.value, page: 1, pageSize: 1 },
   284→            { raw: true },
   285→          );
   286→          return { category: cat.value, count: res?.meta?.pagination?.total ?? 0 };
   287→        }),
   288→      );
   289→      const counts: Record<string, number> = {};
   290→      for (const r of results) {
   291→        if (r.status === 'fulfilled') {
   292→          counts[r.value.category] = r.value.count;
   293→        }
   294→      }
   295→      return counts;
   296→    },
   297→    staleTime: 30_000,
   298→  });
   299→}

   336→        setSearch(value);

   345→    setCategory(value as EmailTemplateCategory | 'ALL');

   347→  }, []);
   348→

   377→      getApi<{ data: EmailTemplate[]; meta: { pagination: { page: number; pageSize: number; total: number; totalPages: number } } }>('/api/email-templates', queryParams, { raw: true }),
   378→    staleTime: 10_000,
   379→  });
   380→
   381→  const templates = (Array.isArray(raw?.data) ? raw.data : []) as EmailTemplate[];
   382→  const pagination = raw?.meta?.pagination;
   383→
   384→  // -------------------- Category Counts --------------------
   385→  const { data: categoryCounts } = useCategoryCounts();
   386→  const totalCount = categoryCounts
   387→    ? Object.values(categoryCounts).reduce((a, b) => a + b, 0)
   388→    : 0;
   389→
   390→  // -------------------- Mutations --------------------
   391→  const duplicateMutation = useMutation({
   392→    mutationFn: (id: string) => postApi(`/api/email-templates/${id}/duplicate`),
   393→    onSuccess: () => {
   394→      toast.success('Template duplicated successfully');
   395→      queryClient.invalidateQueries({ queryKey: queryKeys.emailTemplates.all });
   396→    },
   397→    onError: (err: Error) => {
   398→      toast.error(err.message || 'Failed to duplicate template');
   399→    },
   400→  });
   401→
   402→  const toggleStatusMutation = useMutation({
   403→    mutationFn: ({ id, status }: { id: string; status: EmailTemplateStatus }) =>
   404→      patchApi(`/api/email-templates/${id}`, { status }),
   405→    onSuccess: () => {
   406→      queryClient.invalidateQueries({ queryKey: queryKeys.emailTemplates.all });
   407→    },
   408→    onError: (err: Error) => {
   409→      toast.error(err.message || 'Failed to update template status');
   410→    },
   411→  });
   412→
   413→  const deleteMutation = useMutation({
   414→    mutationFn: (id: string) => deleteApi(`/api/email-templates/${id}`),
   415→    onSuccess: () => {
   416→      toast.success('Template deleted');
   417→      queryClient.invalidateQueries({ queryKey: queryKeys.emailTemplates.all });
   418→      setDeleteTarget(null);
   419→    },
   420→    onError: (err: Error) => {
   421→      toast.error(err.message || 'Failed to delete template');
   422→    },
   423→  });
   424→
   425→  const revertMutation = useMutation({
   426→    mutationFn: (id: string) => postApi(`/api/email-templates/${id}/revert`),
   427→    onSuccess: () => {
   428→      toast.success('Template reverted to default');
   429→      queryClient.invalidateQueries({ queryKey: queryKeys.emailTemplates.all });
   430→      setRevertTarget(null);
   431→    },
   432→    onError: (err: Error) => {
   433→      toast.error(err.message || 'Failed to revert template');
   434→    },

   542→          />

   585→              </div>
   586→            ))}
   587→          </div>
   588→        ) : templates.length === 0 ? (
   589→          <EmptyState
   590→            icon={Mail}
   591→            title="No templates found"
   592→            description={
   593→              search || statusFilter !== 'ALL' || category !== 'ALL'
   594→                ? 'Try adjusting your filters to find what you\'re looking for.'
   595→                : 'Create your first email template or seed the defaults to get started.'
   596→            }
   597→            action={
   598→              !search && statusFilter === 'ALL' && category === 'ALL'
   599→                ? {
   600→                    label: 'Seed Default Templates',
   601→                    onClick: () => seedMutation.mutate(),
   602→                  }
   603→                : undefined
   604→            }
   605→          />
   606→        ) : (
   607→          <>
   608→            <Table>
   609→              <TableHeader>
   610→                <TableRow>
   611→                  <TableHead className="pl-4">Template Name</TableHead>
   612→                  <TableHead>Category</TableHead>
   613→                  <TableHead>Status</TableHead>
   614→                  <TableHead>Provider</TableHead>
   615→                  <TableHead>Last Updated</TableHead>
   616→                  <TableHead>Language</TableHead>
   617→                  <TableHead className="pr-4 w-12"></TableHead>
   618→                </TableRow>
   619→              </TableHeader>
   620→              <TableBody>
   621→                {templates.map((template) => (
   622→                  <TableRow
   623→                    key={template.id}
   624→                    className="cursor-pointer group"
   625→                    onClick={() => onEdit(template.id)}
   626→                  >
   627→                    {/* Template Name */}
   628→                    <TableCell className="pl-4">
   629→                      <div className="flex items-center gap-2 min-w-0">
   630→                        <span className="truncate font-medium text-foreground">
   631→                          {template.name}
   632→                        </span>
   633→                        {template.isSystem && (
   634→                          <Badge
   635→                            variant="outline"
   636→                            className="shrink-0 px-1.5 py-0 text-[10px] font-medium text-muted-foreground border-dashed"
   637→                          >
   638→                            System
   639→                          </Badge>
   640→                        )}
   641→                      </div>
   642→                    </TableCell>
   643→
   644→                    {/* Category */}
   645→                    <TableCell>
   646→                      <StatusBadge
   647→                        status={template.category}
   648→                        size="sm"
   649→                      />
   650→                    </TableCell>
   651→
   652→                    {/* Status */}
   653→                    <TableCell>
   654→                      <Badge
   655→                        variant="outline"
   656→                        className={cn(
   657→                          'border-transparent font-medium shrink-0 px-2 py-0.5 text-xs',
   658→                          getStatusColor(template.status),
   659→                        )}
   660→                      >
   661→                        {getStatusLabel(template.status)}
   662→                      </Badge>
   663→                    </TableCell>
   664→
   665→                    {/* Provider */}
   666→                    <TableCell>
   667→                      <Badge
   668→                        variant="outline"
   669→                        className={cn(
   670→                          'border-transparent shrink-0 px-2 py-0.5 text-xs font-medium',
   671→                          PROVIDER_COLORS[template.provider] ??
   672→                            'bg-zinc-100 text-zinc-700 dark:bg-zinc-800 dark:text-zinc-300',
   673→                        )}
   674→                      >
   675→                        {template.provider}
   676→                      </Badge>
   677→                    </TableCell>
   678→
   679→                    {/* Last Updated */}
   680→                    <TableCell className="text-muted-foreground text-sm">
   681→                      {formatRelativeTime(template.updatedAt)}
   682→                    </TableCell>
   683→
   684→                    {/* Language */}
   685→                    <TableCell>
   686→                      <Badge
   687→                        variant="outline"
   688→                        className="shrink-0 px-1.5 py-0 text-[10px] font-bold uppercase"
   689→                      >
   690→                        {template.language || 'EN'}
   691→                      </Badge>
   692→                    </TableCell>
   693→
   694→                    {/* Actions */}
   695→                    <TableCell className="pr-4" onClick={(e) => e.stopPropagation()}>
   696→                      <DropdownMenu>
   697→                        <DropdownMenuTrigger asChild>
   698→                          <Button
   699→                            variant="ghost"
   700→                            size="icon"
   701→                            className="h-8 w-8 opacity-0 group-hover:opacity-100 transition-opacity"
   702→                          >
   703→                            <MoreHorizontal className="h-4 w-4" />
   704→                          </Button>
   705→                        </DropdownMenuTrigger>
   706→                        <DropdownMenuContent align="end" className="w-48">
   707→                          <DropdownMenuItem onClick={() => onEdit(template.id)}>
   708→                            <Pencil className="h-4 w-4 mr-2" />
   709→                            Edit
   710→                          </DropdownMenuItem>
   711→                          <DropdownMenuItem onClick={() => onPreview(template.id)}>
   712→                            <Eye className="h-4 w-4 mr-2" />
   713→                            Preview
   714→                          </DropdownMenuItem>
   715→                          <DropdownMenuItem
   716→                            onClick={() => duplicateMutation.mutate(template.id)}
   717→                            disabled={duplicateMutation.isPending}
   718→                          >
   719→                            <Copy className="h-4 w-4 mr-2" />
   720→                            Duplicate
   721→                          </DropdownMenuItem>
   722→                          <DropdownMenuItem
   723→                            onClick={() => setSendTestTarget(template.id)}
   724→                          >
   725→                            <Send className="h-4 w-4 mr-2" />
   726→                            Send Test
   727→                          </DropdownMenuItem>
   728→                          <DropdownMenuSeparator />
   729→                          <DropdownMenuItem
   730→                            onClick={() =>
   731→                              toggleStatusMutation.mutate({
   732→                                id: template.id,
   733→                                status:
   734→                                  template.status === 'ENABLED'
   735→                                    ? 'DISABLED'
   736→                                    : 'ENABLED',
   737→                              })
   738→                            }
   739→                            disabled={toggleStatusMutation.isPending}
   740→                          >
   741→                            {template.status === 'ENABLED' ? (
   742→                              <>
   743→                                <span className="h-4 w-4 mr-2 inline-flex items-center justify-center rounded-full bg-zinc-200 dark:bg-zinc-700">
   744→                                  <span className="h-2 w-2 rounded-full" />
   745→                                </span>
   746→                                Disable
   747→                              </>
   748→                            ) : (
   749→                              <>
   750→                                <span className="h-4 w-4 mr-2 inline-flex items-center justify-center rounded-full bg-emerald-200 dark:bg-emerald-800">
   751→                                  <span className="h-2 w-2 rounded-full" />
   752→                                </span>
   753→                                Enable
   754→                              </>
   755→                            )}
   756→                          </DropdownMenuItem>
   757→                          {template.isSystem && (
   758→                            <DropdownMenuItem
   759→                              onClick={() => setRevertTarget(template)}
   760→                              className="text-amber-600 dark:text-amber-400"
   761→                            >
   762→                              <RotateCcw className="h-4 w-4 mr-2" />
   763→                              Revert to Default
   764→                            </DropdownMenuItem>
   765→                          )}
   766→                          <DropdownMenuSeparator />
   767→                          <DropdownMenuItem
   768→                            variant="destructive"
   769→                            onClick={() => setDeleteTarget(template)}
   770→                          >
   771→                            <Trash2 className="h-4 w-4 mr-2" />
   772→                            Delete
   773→                          </DropdownMenuItem>
   774→                        </DropdownMenuContent>
   775→                      </DropdownMenu>
   776→                    </TableCell>
   777→                  </TableRow>
   778→                ))}
   779→              </TableBody>
   780→            </Table>
   781→
   782→            {/* Pagination */}
   783→            {pagination && pagination.totalPages > 1 && (
   784→              <div className="flex items-center justify-between border-t px-4 py-3">
   785→                <div className="flex items-center gap-2 text-sm text-muted-foreground">
   786→                  <span>
   787→                    Page {pagination.page} of {pagination.totalPages}
   788→                  </span>
   789→                  <span className="text-muted-foreground/50">·</span>
   790→                  <span>{pagination.total} total</span>
   791→                </div>
   792→                <div className="flex items-center gap-3">
   793→                  <Select
   794→                    value={String(pageSize)}
   795→                    onValueChange={(v) => {
   796→                      setPageSize(Number(v));
   797→                      setPage(1);
   798→                    }}
   799→                  >
   800→                    <SelectTrigger size="sm" className="w-[80px] h-8">
   801→                      <SelectValue />
   802→                    </SelectTrigger>
   803→                    <SelectContent>
   804→                      {PAGE_SIZES.map((s) => (
   805→                        <SelectItem key={s} value={String(s)}>
   806→                          {s}
   807→                        </SelectItem>
   808→                      ))}
   809→                    </SelectContent>
   810→                  </Select>
   811→                  <div className="flex items-center gap-1">
   812→                    <Button
   813→                      variant="outline"
   814→                      size="icon"
   815→                      className="h-8 w-8"
   816→                      disabled={pagination.page <= 1}
   817→                      onClick={() => setPage((p) => p - 1)}
   818→                    >
   819→                      <ChevronLeft className="h-4 w-4" />
   820→                    </Button>
   821→                    <Button
   822→                      variant="outline"
   823→                      size="icon"
   824→                      className="h-8 w-8"
   825→                      disabled={pagination.page >= pagination.totalPages}
   826→                      onClick={() => setPage((p) => p + 1)}
   827→                    >
   828→                      <ChevronRight className="h-4 w-4" />
   829→                    </Button>
   830→                  </div>
   831→                </div>
   832→              </div>
   833→            )}
   834→          </>
   835→        )}
   836→      </div>
   837→
   838→      {/* Send Test Dialog */}
   839→      <SendTestDialog
   840→        open={!!sendTestTarget}
   841→        onOpenChange={(v) => !v && setSendTestTarget(null)}
   842→        templateId={sendTestTarget}
   843→      />
   844→
   845→      {/* Delete Confirm Dialog */}
   846→      <ConfirmDialog
   847→        open={!!deleteTarget}
   848→        onOpenChange={(v) => !v && setDeleteTarget(null)}
   849→        title="Delete Template"
   850→        description={
   851→          deleteTarget
   852→            ? `Are you sure you want to delete "${deleteTarget.name}"? This action cannot be undone.`
   853→            : undefined
   854→        }
   855→        confirmLabel="Delete"
   856→        variant="destructive"
   857→        onConfirm={() => {
   858→          if (deleteTarget) deleteMutation.mutate(deleteTarget.id);
   859→        }}
   860→        isLoading={deleteMutation.isPending}
   861→      />
   862→
   863→      {/* Revert Confirm Dialog */}
   864→      <ConfirmDialog
   865→        open={!!revertTarget}
   866→        onOpenChange={(v) => !v && setRevertTarget(null)}
   867→        title="Revert to Default"
   868→        description={
   869→          revertTarget
   870→            ? `This will reset "${revertTarget.name}" to its original default content. Your current changes will be saved as a version before reverting.`
   871→            : undefined
   872→        }
   873→        confirmLabel="Revert"
   874→        onConfirm={() => {
   875→          if (revertTarget) revertMutation.mutate(revertTarget.id);
   876→        }}
   877→        isLoading={revertMutation.isPending}
   878→      />
   879→
   880→      {/* Seed Defaults Confirm Dialog */}
   881→      <ConfirmDialog
   882→        open={seedDialogOpen}
   883→        onOpenChange={setSeedDialogOpen}
   884→        title="Seed Default Templates"
   885→        description="This will add any missing default system email templates. Existing templates (including your custom ones) will not be modified."
   886→        confirmLabel="Seed Defaults"
   887→        onConfirm={() => seedMutation.mutate()}
   888→        isLoading={seedMutation.isPending}
   889→      />
   890→    </div>
   891→  );
   892→}
