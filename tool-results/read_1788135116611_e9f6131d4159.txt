     1→// ============================================================
     2→// PLAN CONFIG — DB-backed, cached single source of truth for plans,
     3→// pricing, entitlements and usage limits.
     4→// ============================================================
     5→// The Client Billing API and the Platform Admin both read from (and
     6→// the admin mutates) this same service. `platform-data.ts` delegates
     7→// its `PLANS` / `getPlan` / `monthlyPrice` to the synchronous cache
     8→// exposed here, so an owner edit (savePlanConfig) propagates to the
     9→// client billing experience and to MRR on the next read.
    10→//
    11→// MULTI-CURRENCY: each plan stores pricesByCurrency (JSON map
    12→// { USD: { monthly, yearly }, EUR: {...}, MAD: {...}, CHF: {...} })
    13→// so the same plan can be sold in multiple currencies at independent
    14→// prices. The customer's currency is resolved server-side from their
    15→// IP geolocation (see country-pricing.ts:resolveCustomerCurrency).
    16→// The plan's `currency` field is the platform default currency
    17→// (auto-resolved from CountryPricing.isDefault) — it's a snapshot
    18→// only, used by legacy callers. `priceMonthly` / `priceYearly` mirror
    19→// the default currency's price for the same reason.
    20→//
    21→// STRIPE: for each paid plan × supported currency × interval, a real
    22→// Stripe Price is created on sync and its ID persisted in
    23→// stripePriceIdsByCurrency. The default currency's IDs are mirrored
    24→// in stripePriceIdMonthly/Yearly for legacy callers. Checkout always
    25→// resolves the per-currency Stripe Price ID server-side — the
    26→// customer's currency determines which Price is charged.
    27→//
    28→// The cache is hydrated from the PlanConfig table on startup and
    29→// after every mutation. If the table is empty on first hydrate, the
    30→// DEFAULT configs are seeded.
    31→// ============================================================

   232→/** Resolve the platform's default billing currency via a LAZY import of
   233→ *  country-pricing (which itself imports `getPlanConfigSync` from here).
   234→ *  The lazy `await import()` breaks the static circular dependency —
   235→ *  this module never statically imports country-pricing, so both load.
   236→ *  Falls back to 'CHF' (the schema default) if anything fails. */
   237→async function resolvePlatformDefaultCurrency(): Promise<string> {
   238→  try {
   239→    const { getPlatformDefaultCurrency } = await import('./country-pricing');
   240→    return await getPlatformDefaultCurrency();
   241→  } catch {
   242→    return 'CHF';
   243→  }
   244→}

   252→let _hydrated = false;
   253→let _hydrating: Promise<void> | null = null;
   254→const _subscribers = new Set<() => void>();
   255→
   256→function notify() {
   257→  for (const cb of _subscribers) {
   258→    try {
   259→      cb();
   260→    } catch {
   261→      // ignore subscriber errors
   262→    }
   263→  }
   264→}
   265→
   266→/** Register a callback fired after every cache refresh (hydrate / mutation). */
   267→export function subscribe(cb: () => void): () => void {
   268→  _subscribers.add(cb);
   269→  return () => _subscribers.delete(cb);
   270→}

   272→/** Synchronous access to the cached plan configs. Returns defaults before
   273→ *  the first hydrate completes (the defaults match the seeded DB rows, so
   274→ *  numbers are correct from the very first render). */
   275→export function getPlanConfigsSync(): PlanConfigData[] {
   276→  if (!_hydrated && !_hydrating) {
   277→    void hydrate();
   278→  }
   279→  return _cache;
   280→}

   303→type PlanConfigRow = {
   304→  id: string;
   305→  planId: string;
   306→  name: string;
   307→  priceMonthly: number;
   308→  priceYearly: number;
   309→  currency: string;
   310→  pricesByCurrency: string;
   311→  stripePriceIdsByCurrency: string;
   312→  interval: string;
   313→  isFree: boolean;
   314→  freePlanDurationDays: number | null;
   315→  stripePriceIdMonthly: string | null;
   316→  stripePriceIdYearly: string | null;
   317→  active: boolean;
   318→  features: string;
   319→  entitlements: string;
   320→  limits: string;
   321→  badgeVariant: string;
   322→  sortOrder: number;
   323→};
   324→
   325→function rowToData(row: PlanConfigRow): PlanConfigData {
   326→  let features: string[] = [];
   327→  let entitlements: string[] = [];
   328→  try {
   329→    features = JSON.parse(row.features || '[]');
   330→  } catch {
   331→    features = [];
   332→  }
   333→  try {
   334→    entitlements = JSON.parse(row.entitlements || '[]');
   335→  } catch {
   336→    entitlements = [];
   337→  }
   338→  // Explicitly pick only the known limit fields. Older DB rows may still
   339→  // carry stale `aiWords` / `aiArticles` / `automationRuns` keys in the
   340→  // limits JSON (those were removed because no real enforcement exists).
   341→  // Picking only the known fields prevents the stale keys from leaking
   342→  // back into the cache and being re-serialized on the next save.
   343→  let limits: PlanLimits = { maxSites: 0, storageBytes: 0 };
   344→  try {
   345→    const parsed = JSON.parse(row.limits || '{}') as Partial<PlanLimits>;
   346→    limits = {
   347→      maxSites: typeof parsed.maxSites === 'number' ? parsed.maxSites : 0,
   348→      storageBytes: typeof parsed.storageBytes === 'number' ? parsed.storageBytes : 0,
   349→    };
   350→  } catch {
   351→    // keep defaults
   352→  }
   353→  let pricesByCurrency: PricesByCurrency = {};
   354→  try {
   355→    const parsed = JSON.parse(row.pricesByCurrency || '{}') as Partial<PricesByCurrency>;
   356→    if (parsed && typeof parsed === 'object') {
   357→      for (const [cur, val] of Object.entries(parsed)) {
   358→        if (val && typeof val === 'object') {
   359→          const m = typeof (val as CurrencyPrice).monthly === 'number' ? (val as CurrencyPrice).monthly : 0;
   360→          const y = typeof (val as CurrencyPrice).yearly === 'number' ? (val as CurrencyPrice).yearly : 0;
   361→          pricesByCurrency[cur.toUpperCase()] = { monthly: m, yearly: y };
   362→        }
   363→      }
   364→    }
   365→  } catch {
   366→    pricesByCurrency = {};
   367→  }
   368→  let stripePriceIdsByCurrency: StripePriceIdsByCurrency = {};
   369→  try {
   370→    const parsed = JSON.parse(row.stripePriceIdsByCurrency || '{}') as Partial<StripePriceIdsByCurrency>;
   371→    if (parsed && typeof parsed === 'object') {
   372→      for (const [cur, val] of Object.entries(parsed)) {
   373→        if (val && typeof val === 'object') {
   374→          const m = typeof (val as CurrencyStripeIds).monthly === 'string' ? (val as CurrencyStripeIds).monthly : null;
   375→          const y = typeof (val as CurrencyStripeIds).yearly === 'string' ? (val as CurrencyStripeIds).yearly : null;
   376→          stripePriceIdsByCurrency[cur.toUpperCase()] = { monthly: m, yearly: y };
   377→        }
   378→      }
   379→    }
   380→  } catch {
   381→    stripePriceIdsByCurrency = {};
   382→  }
   383→  return {
   384→    planId: row.planId,
   385→    name: row.name,
   386→    priceMonthly: row.priceMonthly,
   387→    priceYearly: row.priceYearly,
   388→    currency: row.currency,
   389→    pricesByCurrency,
   390→    stripePriceIdsByCurrency,
   391→    interval: (row.interval === 'yearly' ? 'yearly' : 'monthly') as BillingInterval,
   392→    isFree: row.isFree,
   393→    freePlanDurationDays: row.freePlanDurationDays,
   394→    stripePriceIdMonthly: row.stripePriceIdMonthly,
   395→    stripePriceIdYearly: row.stripePriceIdYearly,
   396→    active: row.active,
   397→    features,
   398→    entitlements,
   399→    limits,
   400→    badgeVariant: row.badgeVariant,
   401→    sortOrder: row.sortOrder,
   402→  };
   403→}

   428→/**
   429→ * Migrate the legacy catalog to the new Free/Plus/Pro/Max shape. Idempotent.
   430→ *
   431→ * Legacy catalog (created by older bootstrap):
   432→ *   beta (free, active=false), pro, max, enterprise (stub)
   433→ *
   434→ * Migration steps (each guarded, can be re-run safely):
   435→ *  1. If `enterprise` exists AND no active subscription references it → delete it.
   436→ *     (Active subs that reference enterprise are preserved — the migration will
   437→ *      not break them; the plan row stays if needed. In practice, the demo
   438→ *      dataset has no DB subscriptions yet, so this is a clean delete.)
   439→ *  2. If `beta` exists → rename to `plus` (planId, name, badgeVariant) and
   440→ *     re-enable it (legacy `beta` was active=false in the demo DB). Also
   441→ *     shift sortOrder to 1 (Free takes 0).
   442→ *  3. If `free` does NOT exist → insert the Free plan at sortOrder 0.
   443→ *  4. Renumber sortOrder to match the canonical order: free=0, plus=1, pro=2, max=3.
   444→ *
   445→ * Returns a short report string for the bootstrap log.
   446→ */
   447→export async function migrateLegacyPlans(): Promise<string> {
   448→  const notes: string[] = [];
   449→
   450→  try {
   451→    // Step 1: delete Enterprise if safe
   452→    const enterprise = await db.planConfig.findUnique({ where: { planId: 'enterprise' } });
   453→    if (enterprise) {
   454→      const subCount = await db.subscription.count({ where: { planId: 'enterprise' } });
   455→      if (subCount === 0) {
   456→        await db.planConfig.delete({ where: { planId: 'enterprise' } });
   457→        notes.push('deleted enterprise plan (0 active subscriptions)');
   458→      } else {
   459→        notes.push(`kept enterprise plan (${subCount} active subscriptions reference it)`);
   460→      }
   461→    }
   462→
   463→    // Step 2: rename beta → plus (and refresh to a real paid tier)
   464→    const beta = await db.planConfig.findUnique({ where: { planId: 'beta' } });
   465→    if (beta) {
   466→      // Don't overwrite an existing plus row — if both exist, prefer the
   467→      // plus row and just delete the beta stub.
   468→      const plus = await db.planConfig.findUnique({ where: { planId: 'plus' } });
   469→      if (!plus) {
   470→        // Rename beta → plus AND refresh the data to the real DEFAULT
   471→        // Plus tier (paid plan with real entitlements + limits, not the
   472→        // legacy free-tier Beta stub). The old Beta was active=false and
   473→        // had no real customers, so this is a safe refresh.
   474→        const plusDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'plus')!;
   475→        await db.planConfig.update({
   476→          where: { planId: 'beta' },
   477→          data: {
   478→            ...dataToRow(plusDefault),
   479→            // Keep the original createdAt for continuity.
   480→            // (updatedAt is auto.)
   481→          },
   482→        });
   483→        notes.push('renamed + refreshed beta → plus (real paid tier)');
   484→      } else {
   485→        // Both beta and plus exist — delete the beta stub (migrate any
   486→        // legacy subscriptions on beta to plus first).
   487→        const betaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   488→        if (betaSubs > 0) {
   489→          await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   490→          notes.push(`migrated ${betaSubs} beta subscription(s) → plus`);
   491→        }
   492→        await db.planConfig.delete({ where: { planId: 'beta' } });
   493→        notes.push('deleted legacy beta stub (plus already exists)');
   494→      }
   495→    }
   496→
   497→    // Step 3: insert Free if missing
   498→    const free = await db.planConfig.findUnique({ where: { planId: 'free' } });
   499→    if (!free) {
   500→      const freeDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'free')!;
   501→      await db.planConfig.create({ data: dataToRow(freeDefault) });
   502→      notes.push('inserted free plan');
   503→    } else {
   504→      // Ensure free plan is properly flagged
   505→      if (!free.isFree || free.badgeVariant !== 'free' || free.sortOrder !== 0) {
   506→        await db.planConfig.update({
   507→          where: { planId: 'free' },
   508→          data: { isFree: true, badgeVariant: 'free', sortOrder: 0 },
   509→        });
   510→        notes.push('normalized free plan flags');
   511→      }
   512→    }
   513→
   514→    // Step 4: normalize sortOrder for all canonical plans
   515→    const canonicalOrder: Record<string, number> = { free: 0, plus: 1, pro: 2, max: 3 };
   516→    for (const [pid, order] of Object.entries(canonicalOrder)) {
   517→      const row = await db.planConfig.findUnique({ where: { planId: pid } });
   518→      if (row && row.sortOrder !== order) {
   519→        await db.planConfig.update({ where: { planId: pid }, data: { sortOrder: order } });
   520→      }
   521→    }
   522→
   523→    // Also migrate legacy subscriptions on 'beta' or 'enterprise' planId
   524→    // to the new ids (beta → plus, enterprise → max) so historical subs
   525→    // don't reference a missing plan.
   526→    const legacyBetaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   527→    if (legacyBetaSubs > 0) {
   528→      await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   529→      notes.push(`migrated ${legacyBetaSubs} legacy beta subscription(s) → plus`);
   530→    }
   531→    const legacyEnterpriseSubs = await db.subscription.count({ where: { planId: 'enterprise' } });
   532→    if (legacyEnterpriseSubs > 0) {
   533→      await db.subscription.updateMany({ where: { planId: 'enterprise' }, data: { planId: 'max' } });
   534→      notes.push(`migrated ${legacyEnterpriseSubs} legacy enterprise subscription(s) → max`);
   535→    }
   536→  } catch (err) {
   537→    notes.push(`migration error (non-fatal): ${(err as Error).message}`);
   538→  }
   539→
   540→  return notes.length > 0 ? notes.join('; ') : 'no migration needed';
   541→}

   543→/** Load all plan configs from the DB into the cache. Self-seeds the
   544→ *  DEFAULT_PLAN_CONFIGS when the table is empty. Runs the legacy
   545→ *  migration when the catalog still contains 'beta' or 'enterprise'.
   546→ *  Safe to call repeatedly. */
   547→export async function hydrate(): Promise<void> {
   548→  if (_hydrating) return _hydrating;
   549→  _hydrating = (async () => {
   550→    try {
   551→      const rows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   552→      if (rows.length === 0) {
   553→        // Self-seed so the platform is never without plans.
   554→        await db.planConfig.createMany({ data: DEFAULT_PLAN_CONFIGS.map(dataToRow) });
   555→        _cache = DEFAULT_PLAN_CONFIGS.map((p) => ({
   556→          ...p,
   557→          features: [...p.features],
   558→          entitlements: [...p.entitlements],
   559→          limits: { ...p.limits },
   560→        }));
   561→      } else {
   562→        // Check for legacy catalog (beta / enterprise) and migrate.
   563→        const hasLegacy =
   564→          rows.some((r) => r.planId === 'beta' || r.planId === 'enterprise') ||
   565→          !rows.some((r) => r.planId === 'free');
   566→        if (hasLegacy) {
   567→          await migrateLegacyPlans();
   568→          const migratedRows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   569→          _cache = migratedRows.map(rowToData);
   570→        } else {
   571→          _cache = rows.map(rowToData);
   572→        }
   573→      }
   574→      _hydrated = true;
   575→      notify();
   576→    } catch (err) {
   577→      // DB unavailable — keep defaults (still functional, just not persisted).
   578→      _hydrated = true;
   579→      notify();
   580→    } finally {
   581→      _hydrating = null;
   582→    }
   583→  })();
   584→  return _hydrating;
   585→}

   589→export interface PlanConfigInput {
   590→  name?: string;
   591→  /** Snapshot of the platform DEFAULT currency's monthly price. */
   592→  priceMonthly?: number;
   593→  /** Snapshot of the platform DEFAULT currency's yearly price. */
   594→  priceYearly?: number;
   595→  /** Platform default currency. Auto-resolved when omitted. */
   596→  currency?: string;
   597→  /** Authoritative per-currency prices: { USD: { monthly, yearly }, ... }. */
   598→  pricesByCurrency?: PricesByCurrency;
   599→  /** Stripe Price ID snapshots for the default currency. */
   600→  stripePriceIdMonthly?: string | null;
   601→  stripePriceIdYearly?: string | null;
   602→  /** Authoritative per-currency Stripe Price IDs: { USD: { monthly, yearly }, ... }. */
   603→  stripePriceIdsByCurrency?: StripePriceIdsByCurrency;
   604→  interval?: BillingInterval;
   605→  isFree?: boolean;
   606→  freePlanDurationDays?: number | null;
   607→  active?: boolean;
   608→  /** Marketing copy. Auto-derived from entitlements when omitted. */
   609→  features?: string[];
   610→  entitlements?: string[];
   611→  limits?: Partial<PlanLimits>;
   612→  badgeVariant?: string;
   613→  sortOrder?: number;

   616→/** Merge a patch (Partial<PlanConfigData>) into the current data,
   617→ *  applying defaults for the multi-currency fields when omitted.
   618→ *  - pricesByCurrency: keep current when omitted.
   619→ *  - stripePriceIdsByCurrency: keep current when omitted.
   620→ *  - When pricesByCurrency is provided but priceMonthly/priceYearly
   621→ *    are NOT, snapshot the default currency's price into the legacy
   622→ *    fields for backward-compat with legacy callers.
   623→ *  - When stripePriceIdsByCurrency is provided but stripePriceIdMonthly/
   624→ *    Yearly are NOT, snapshot the default currency's IDs.
   625→ */
   626→function mergePlanPatch(current: PlanConfigData, patch: PlanConfigInput, defaultCurrency: string): PlanConfigData {
   627→  const pricesByCurrency: PricesByCurrency = patch.pricesByCurrency ?? current.pricesByCurrency ?? {};
   628→  const stripePriceIdsByCurrency: StripePriceIdsByCurrency =
   629→    patch.stripePriceIdsByCurrency ?? current.stripePriceIdsByCurrency ?? {};
   630→  // Default-currency snapshot — keep the legacy fields in sync so legacy
   631→  // callers that read priceMonthly/priceYearly/stripePriceIdMonthly/Yearly
   632→  // see the right values.
   633→  const curUpper = defaultCurrency.toUpperCase();
   634→  const defPrice = pricesByCurrency[curUpper] ?? { monthly: 0, yearly: 0 };
   635→  const defIds = stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
   636→  const priceMonthly =
   637→    patch.priceMonthly !== undefined
   638→      ? patch.priceMonthly
   639→      : patch.pricesByCurrency !== undefined
   640→        ? defPrice.monthly
   641→        : current.priceMonthly;
   642→  const priceYearly =
   643→    patch.priceYearly !== undefined
   644→      ? patch.priceYearly
   645→      : patch.pricesByCurrency !== undefined
   646→        ? defPrice.yearly
   647→        : current.priceYearly;
   648→  const stripePriceIdMonthly =
   649→    patch.stripePriceIdMonthly !== undefined
   650→      ? patch.stripePriceIdMonthly
   651→      : patch.stripePriceIdsByCurrency !== undefined
   652→        ? defIds.monthly
   653→        : current.stripePriceIdMonthly;
   654→  const stripePriceIdYearly =
   655→    patch.stripePriceIdYearly !== undefined
   656→      ? patch.stripePriceIdYearly
   657→      : patch.stripePriceIdsByCurrency !== undefined
   658→        ? defIds.yearly
   659→        : current.stripePriceIdYearly;
   660→  const isFree = patch.isFree ?? current.isFree;
   661→  return {
   662→    planId: current.planId,
   663→    name: patch.name ?? current.name,
   664→    priceMonthly,
   665→    priceYearly,
   666→    currency: patch.currency ?? current.currency,
   667→    pricesByCurrency,
   668→    stripePriceIdsByCurrency,
   669→    interval: patch.interval ?? current.interval,
   670→    isFree,
   671→    freePlanDurationDays: patch.freePlanDurationDays ?? current.freePlanDurationDays,
   672→    stripePriceIdMonthly,
   673→    stripePriceIdYearly,
   674→    active: patch.active ?? current.active,
   675→    features: patch.features ?? current.features,
   676→    entitlements: patch.entitlements ?? current.entitlements,
   677→    limits: { ...current.limits, ...(patch.limits ?? {}) },
   678→    badgeVariant: patch.badgeVariant ?? current.badgeVariant,
   679→    sortOrder: patch.sortOrder ?? current.sortOrder,
   680→  };
   681→}

   683→/** Update an existing plan config. Writes DB, refreshes cache.
   684→ *
   685→ *  AUTO-SYNC TO STRIPE: when Stripe is configured AND the plan is
   686→ *  PAID (any currency has a positive price) AND the admin did NOT
   687→ *  explicitly set the Stripe Price IDs in this patch, the backend calls
   688→ *  syncPlanToStripe (multi-currency) to ensure a Stripe Product +
   689→ *  per-currency monthly+yearly Prices exist and writes the resolved IDs
   690→ *  into stripePriceIdsByCurrency + the default-currency snapshot fields.
   691→ */
   692→export async function savePlanConfig(planId: string, patch: PlanConfigInput): Promise<PlanConfigData | null> {
   693→  const existing = await db.planConfig.findUnique({ where: { planId } });
   694→  if (!existing) return null;
   695→  const current = rowToData(existing);
   696→
   697→  // When the admin omitted currency, resolve the platform default so the
   698→  // legacy priceMonthly/priceYearly snapshot stays in the right currency.
   699→  if (!patch.currency && !current.currency) {
   700→    patch = { ...patch, currency: await resolvePlatformDefaultCurrency() };
   701→  }
   702→  const finalCurrency = (patch.currency ?? current.currency).toUpperCase();
   703→
   704→  // Distinguish "field omitted" from "field set". When the admin omits
   705→  // both stripePriceId* AND stripePriceIdsByCurrency, they want the
   706→  // backend to auto-sync. When either is explicitly provided, manual.
   707→  const adminTouchedStripePriceIds =
   708→    'stripePriceIdMonthly' in patch ||
   709→    'stripePriceIdYearly' in patch ||
   710→    'stripePriceIdsByCurrency' in patch;
   711→
   712→  const next = mergePlanPatch(current, patch, finalCurrency);
   713→  await db.planConfig.update({ where: { planId }, data: dataToRow(next) });
   714→  await hydrate();
   715→
   716→  // ---- AUTO-SYNC TO STRIPE (multi-currency) ----
   717→  // Best-effort — never throws.
   718→  const hasAnyPaidCurrency = Object.values(next.pricesByCurrency).some(
   719→    (p) => p.monthly > 0 || p.yearly > 0,
   720→  );
   721→  if (!adminTouchedStripePriceIds && !next.isFree && hasAnyPaidCurrency) {
   722→    try {
   723→      const { isStripeConfiguredAsync, getStripeClient, syncPlanToStripeMulti } = await import('@/lib/stripe');
   724→      if (await isStripeConfiguredAsync()) {
   725→        const stripe = await getStripeClient();
   726→        const syncResult = await syncPlanToStripeMulti(stripe, {
   727→          planId: next.planId,
   728→          name: next.name,
   729→          defaultCurrency: finalCurrency,
   730→          pricesByCurrency: next.pricesByCurrency,
   731→          stripePriceIdsByCurrency: next.stripePriceIdsByCurrency,
   732→        });
   733→        // Persist the resolved Stripe Price IDs back onto the row when they changed.
   734→        const snapshot = syncResult.stripePriceIdsByCurrency[finalCurrency] ?? { monthly: null, yearly: null };
   735→        const changed =
   736→          JSON.stringify(syncResult.stripePriceIdsByCurrency) !==
   737→            JSON.stringify(next.stripePriceIdsByCurrency) ||
   738→          snapshot.monthly !== next.stripePriceIdMonthly ||
   739→          snapshot.yearly !== next.stripePriceIdYearly;
   740→        if (changed) {
   741→          await db.planConfig.update({
   742→            where: { planId },
   743→            data: {
   744→              stripePriceIdsByCurrency: JSON.stringify(syncResult.stripePriceIdsByCurrency),
   745→              stripePriceIdMonthly: snapshot.monthly,
   746→              stripePriceIdYearly: snapshot.yearly,
   747→            },
   748→          });
   749→          await hydrate();
   750→        }
   751→      }
   752→    } catch {
   753→      // best-effort — never block the save on a Stripe error
   754→    }
   755→  }
   756→
   757→  return getPlanConfigSync(planId);
   758→}

   760→/** Create a new plan config. AUTO-SYNC TO STRIPE: when Stripe is
   761→ *  configured AND the new plan is PAID AND the admin did NOT
   762→ *  explicitly provide Stripe Price IDs in the input, the backend
   763→ *  calls syncPlanToStripeMulti (multi-currency) to create a Stripe
   764→ *  Product + per-currency monthly+yearly Prices and writes the
   765→ *  resolved IDs back onto the new plan row.
   766→ *
   767→ *  When pricesByCurrency is omitted, an empty map is used (the admin
   768→ *  will need to add prices per currency via the editor). When provided,
   769→ *  the default currency's price is mirrored into priceMonthly/Yearly. */
   770→export async function createPlanConfig(
   771→  input: PlanConfigInput & { planId: string; name: string },
   772→): Promise<PlanConfigData> {
   773→  const maxOrder = await db.planConfig.count();
   774→  const currency =
   775→    input.currency && input.currency.trim().length > 0
   776→      ? input.currency
   777→      : await resolvePlatformDefaultCurrency();
   778→  const curUpper = currency.toUpperCase();
   779→  const pricesByCurrency: PricesByCurrency = input.pricesByCurrency ?? {};
   780→  const defPrice = pricesByCurrency[curUpper] ?? { monthly: 0, yearly: 0 };
   781→  const stripePriceIdsByCurrency: StripePriceIdsByCurrency = input.stripePriceIdsByCurrency ?? {};
   782→  const defIds = stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
   783→  const data: PlanConfigData = {
   784→    planId: input.planId,
   785→    name: input.name,
   786→    priceMonthly: input.priceMonthly !== undefined ? input.priceMonthly : defPrice.monthly,
   787→    priceYearly: input.priceYearly !== undefined ? input.priceYearly : defPrice.yearly,
   788→    currency: curUpper,
   789→    pricesByCurrency,
   790→    stripePriceIdsByCurrency,
   791→    interval: input.interval ?? 'monthly',
   792→    isFree: input.isFree ?? false,
   793→    freePlanDurationDays: input.freePlanDurationDays ?? null,
   794→    stripePriceIdMonthly: input.stripePriceIdMonthly !== undefined ? input.stripePriceIdMonthly : defIds.monthly,
   795→    stripePriceIdYearly: input.stripePriceIdYearly !== undefined ? input.stripePriceIdYearly : defIds.yearly,
   796→    active: input.active ?? true,
   797→    features: input.features ?? [],
   798→    entitlements: input.entitlements ?? [],
   799→    limits: {
   800→      maxSites: 0,
   801→      storageBytes: 0,
   802→      ...(input.limits ?? {}),
   803→    },
   804→    badgeVariant: input.badgeVariant ?? input.planId,
   805→    sortOrder: input.sortOrder ?? maxOrder,
   806→  };
   807→  await db.planConfig.create({ data: dataToRow(data) });
   808→  await hydrate();
   809→
   810→  // ---- AUTO-SYNC TO STRIPE (multi-currency) ----
   811→  // Best-effort — never throws. The plan row is already created; if
   812→  // the Stripe side fails, the admin can use the explicit "Sync to
   813→  // Stripe" route to surface errors.
   814→  const adminTouchedStripePriceIds =
   815→    'stripePriceIdMonthly' in input ||
   816→    'stripePriceIdYearly' in input ||
   817→    'stripePriceIdsByCurrency' in input;
   818→  const hasAnyPaidCurrency = Object.values(data.pricesByCurrency).some(
   819→    (p) => p.monthly > 0 || p.yearly > 0,
   820→  );
   821→  if (!adminTouchedStripePriceIds && !data.isFree && hasAnyPaidCurrency) {
   822→    try {
   823→      const { isStripeConfiguredAsync, getStripeClient, syncPlanToStripeMulti } = await import('@/lib/stripe');
   824→      if (await isStripeConfiguredAsync()) {
   825→        const stripe = await getStripeClient();
   826→        const syncResult = await syncPlanToStripeMulti(stripe, {
   827→          planId: data.planId,
   828→          name: data.name,
   829→          defaultCurrency: curUpper,
   830→          pricesByCurrency: data.pricesByCurrency,
   831→          stripePriceIdsByCurrency: data.stripePriceIdsByCurrency,
   832→        });
   833→        const snapshot = syncResult.stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
   834→        const changed =
   835→          JSON.stringify(syncResult.stripePriceIdsByCurrency) !==
   836→            JSON.stringify(data.stripePriceIdsByCurrency) ||
   837→          snapshot.monthly !== data.stripePriceIdMonthly ||
   838→          snapshot.yearly !== data.stripePriceIdYearly;
   839→        if (changed) {
   840→          await db.planConfig.update({
   841→            where: { planId: data.planId },
   842→            data: {
   843→              stripePriceIdsByCurrency: JSON.stringify(syncResult.stripePriceIdsByCurrency),
   844→              stripePriceIdMonthly: snapshot.monthly,
   845→              stripePriceIdYearly: snapshot.yearly,
   846→            },
   847→          });
   848→          await hydrate();
   849→          data.stripePriceIdsByCurrency = syncResult.stripePriceIdsByCurrency;
   850→          data.stripePriceIdMonthly = snapshot.monthly;
   851→          data.stripePriceIdYearly = snapshot.yearly;
   852→        }
   853→      }
   854→    } catch {
   855→      // best-effort — never block the create on a Stripe error
   856→    }
   857→  }
   858→
   859→  return data;
   860→}

   862→/**
   863→ * Delete a plan config. Refuses if:
   864→ *  - it's the last remaining plan, OR
   865→ *  - any Subscription still references this planId (would orphan the sub).
   866→ *
   867→ * Returns `{ ok: boolean, reason?: string }` describing the outcome.
   868→ */
   869→export async function deletePlanConfig(
   870→  planId: string,
   871→): Promise<{ ok: boolean; reason?: string }> {
   872→  const count = await db.planConfig.count();
   873→  if (count <= 1) return { ok: false, reason: 'Cannot delete the last plan.' };
   874→  const subCount = await db.subscription.count({ where: { planId } });
   875→  if (subCount > 0) {
   876→    return {
   877→      ok: false,
   878→      reason: `Cannot delete plan "${planId}" — ${subCount} active subscription(s) reference it. Migrate them first.`,
   879→    };
   880→  }
   881→  try {
   882→    await db.planConfig.delete({ where: { planId } });
   883→    await hydrate();
   884→    return { ok: true };
   885→  } catch (err) {
   886→    return { ok: false, reason: (err as Error).message };
   887→  }
   888→}
