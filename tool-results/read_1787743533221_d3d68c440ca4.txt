   123→// ==================== Helpers ====================
   124→
   125→function formatNumber(n: number | null | undefined): string {
   126→  const num = n ?? 0;
   127→  if (num >= 1000000) return `${(num / 1000000).toFixed(1)}M`;
   128→  if (num >= 1000) return `${(num / 1000).toFixed(1)}K`;
   129→  return num.toLocaleString();
   130→}
   131→
   132→function formatPercent(n: number | null | undefined): string {
   133→  return `${((n ?? 0) * 100).toFixed(2)}%`;
   134→}
   135→
   136→function formatPosition(n: number | null | undefined): string {
   137→  return (n ?? 0).toFixed(1);
   138→}
   139→
   140→function formatDate(dateStr: string | null): string {
   141→  if (!dateStr) return 'Never';
   142→  try {
   143→    return new Date(dateStr).toLocaleString(undefined, {
   144→      month: 'short',
   145→      day: 'numeric',
   146→      year: 'numeric',
   147→      hour: '2-digit',
   148→      minute: '2-digit',
   149→    });
   150→  } catch {
   151→    return 'Unknown';
   152→  }
   153→}
   154→
   155→function formatShortDate(dateStr: string): string {
   156→  try {
   157→    return new Date(dateStr).toLocaleDateString(undefined, {
   158→      month: 'short',
   159→      day: 'numeric',
   160→    });
   161→  } catch {
   162→    return dateStr;
   163→  }
   164→}

   243→// Custom recharts tooltip — shows the exact date + per-day values.
   244→function ChartTooltip({
   245→  active,
   246→  payload,
   247→}: {
   248→  active?: boolean;
   249→  payload?: Array<{ payload: DailyStat }>;
   250→}) {
   251→  if (!active || !payload || payload.length === 0) return null;
   252→  const point = payload[0].payload;
   253→  let dateLabel = point.date;
   254→  try {
   255→    dateLabel = new Date(`${point.date}T00:00:00`).toLocaleDateString(undefined, {
   256→      year: 'numeric',
   257→      month: 'short',
   258→      day: 'numeric',
   259→    });
   260→  } catch {
   261→    /* keep raw date */
   262→  }
   263→  return (
   264→    <div className="rounded-md border bg-popover text-popover-foreground shadow-md px-3 py-2 text-xs min-w-[170px]">
   265→      <div className="font-medium mb-1.5">{dateLabel}</div>
   266→      <div className="flex items-center justify-between gap-3">
   267→        <span className="flex items-center gap-1.5">
   268→          <span className="h-2 w-2 rounded-sm bg-primary" /> Clicks
   269→        </span>
   270→        <span className="font-medium tabular-nums">{point.clicks.toLocaleString()}</span>
   271→      </div>
   272→      <div className="flex items-center justify-between gap-3 mt-0.5">
   273→        <span className="flex items-center gap-1.5">
   274→          <span className="h-2 w-2 rounded-sm bg-primary/30" /> Impressions
   275→        </span>
   276→        <span className="font-medium tabular-nums">{point.impressions.toLocaleString()}</span>
   277→      </div>
   278→      {point.ctr != null && (
   279→        <div className="flex items-center justify-between gap-3 mt-0.5 text-muted-foreground">
   280→          <span>CTR</span>
   281→          <span className="tabular-nums">{(point.ctr * 100).toFixed(2)}%</span>
   282→        </div>
   283→      )}
   284→      {point.position != null && (
   285→        <div className="flex items-center justify-between gap-3 mt-0.5 text-muted-foreground">
   286→          <span>Avg Position</span>
   287→          <span className="tabular-nums">{point.position.toFixed(1)}</span>
   288→        </div>
   289→      )}
   290→    </div>
   291→  );
   292→}

   294→function PerformanceChart({ stats, isLoading, rangeLabel, onSync, isSyncing }: PerformanceChartProps) {
   295→  if (isLoading) {
   296→    return (
   297→      <div className="space-y-3">
   298→        <Skeleton className="h-4 w-24" />
   299→        <Skeleton className="h-56 w-full" />
   300→      </div>
   301→    );
   302→  }
   303→
   304→  if (!stats || stats.length === 0) {
   305→    return (
   306→      <div className="flex flex-col items-center justify-center py-16 text-center">
   307→        <BarChart3 className="h-12 w-12 text-muted-foreground/30 mb-3" strokeWidth={1.5} />
   308→        <p className="text-sm font-medium text-muted-foreground">
   309→          No Search Console data available for this period.
   310→        </p>
   311→        <p className="text-xs text-muted-foreground mt-1">
   312→          Try a different range or sync with Search Console to see chart data.
   313→        </p>
   314→        {onSync && (
   315→          <Button
   316→            variant="outline"
   317→            size="sm"
   318→            className="mt-4"
   319→            onClick={onSync}
   320→            disabled={isSyncing}
   321→          >
   322→            {isSyncing ? (
   323→              <Loader2 className="h-3.5 w-3.5 mr-2 animate-spin" />
   324→            ) : (
   325→              <RefreshCw className="h-3.5 w-3.5 mr-2" />
   326→            )}
   327→            Sync Now
   328→          </Button>
   329→        )}
   330→      </div>
   331→    );
   332→  }
   333→
   334→  // Defensive chronological sort (oldest → newest). The API already returns
   335→  // ascending by date, but a mis-sorted payload must never scramble the X-axis.
   336→  const data = [...stats]
   337→    .map((s) => ({
   338→      date: s.date,
   339→      clicks: s.clicks,
   340→      impressions: s.impressions,
   341→      ctr: s.ctr,
   342→      position: s.position,
   343→    }))
   344→    .sort((a, b) => (a.date < b.date ? -1 : a.date > b.date ? 1 : 0));
   345→
   346→  const axisTick = { fontSize: 10, fill: 'hsl(var(--muted-foreground))' };
   347→
   348→  return (
   349→    <div className="space-y-4">
   350→      <div className="flex items-center justify-between">
   351→        <div className="flex items-center gap-4 text-xs text-muted-foreground">
   352→          <span className="flex items-center gap-1.5">
   353→            <span className="h-2.5 w-2.5 rounded-sm bg-primary" />
   354→            Clicks
   355→          </span>
   356→          <span className="flex items-center gap-1.5">
   357→            <span className="h-2.5 w-2.5 rounded-sm bg-primary/30" />
   358→            Impressions
   359→          </span>
   360→        </div>
   361→        <Badge variant="outline" className="text-[10px] font-normal bg-background/80">
   362→          {rangeLabel}
   363→        </Badge>
   364→      </div>
   365→
   366→      {/*
   367→        Dual Y-axis area chart: left axis = Impressions (large numbers),
   368→        right axis = Clicks (small numbers). Without two axes the Clicks series
   369→        would collapse to a flat line at the bottom and be unreadable. Colors
   370→        stay on the existing `primary` hue to preserve the page palette and the
   371→        manual legend above.
   372→      */}
   373→      <div className="h-56 w-full">
   374→        <ResponsiveContainer width="100%" height="100%">
   375→          <AreaChart data={data} margin={{ top: 8, right: 8, left: 0, bottom: 0 }}>
   376→            <defs>
   377→              <linearGradient id="scImpGrad" x1="0" y1="0" x2="0" y2="1">
   378→                <stop offset="0%" stopColor="hsl(var(--primary))" stopOpacity={0.3} />
   379→                <stop offset="100%" stopColor="hsl(var(--primary))" stopOpacity={0.04} />
   380→              </linearGradient>
   381→              <linearGradient id="scClkGrad" x1="0" y1="0" x2="0" y2="1">
   382→                <stop offset="0%" stopColor="hsl(var(--primary))" stopOpacity={0.9} />
   383→                <stop offset="100%" stopColor="hsl(var(--primary))" stopOpacity={0.45} />
   384→              </linearGradient>
   385→            </defs>
   386→            <CartesianGrid
   387→              strokeDasharray="3 3"
   388→              vertical={false}
   389→              stroke="hsl(var(--border))"
   390→              strokeOpacity={0.6}
   391→            />
   392→            <XAxis
   393→              dataKey="date"
   394→              tickFormatter={(value: string) => formatShortDate(value)}
   395→              minTickGap={20}
   396→              tick={axisTick}
   397→              tickLine={false}
   398→              axisLine={{ stroke: 'hsl(var(--border))' }}
   399→            />
   400→            <YAxis
   401→              yAxisId="impressions"
   402→              tickFormatter={(n: number) => formatNumber(n)}
   403→              tick={axisTick}
   404→              tickLine={false}
   405→              axisLine={false}
   406→              width={46}
   407→            />
   408→            <YAxis
   409→              yAxisId="clicks"
   410→              orientation="right"
   411→              tickFormatter={(n: number) => formatNumber(n)}
   412→              tick={axisTick}
   413→              tickLine={false}
   414→              axisLine={false}
   415→              width={40}
   416→            />
   417→            <RechartsTooltip
   418→              content={<ChartTooltip />}
   419→              cursor={{ stroke: 'hsl(var(--border))', strokeDasharray: '3 3' }}
   420→            />
   421→            <Area
   422→              yAxisId="impressions"
   423→              type="monotone"
   424→              dataKey="impressions"
   425→              name="Impressions"
   426→              stroke="hsl(var(--primary))"
   427→              strokeOpacity={0.55}
   428→              strokeWidth={1.5}
   429→              fill="url(#scImpGrad)"
   430→              fillOpacity={1}
   431→              isAnimationActive={false}
   432→            />
   433→            <Area
   434→              yAxisId="clicks"
   435→              type="monotone"
   436→              dataKey="clicks"
   437→              name="Clicks"
   438→              stroke="hsl(var(--primary))"
   439→              strokeWidth={2}
   440→              fill="url(#scClkGrad)"
   441→              fillOpacity={1}
   442→              isAnimationActive={false}
   443→            />
   444→          </AreaChart>
   445→        </ResponsiveContainer>
   446→      </div>
   447→    </div>
   448→  );
   449→}

   561→                  return (

   565→                      rel={external ? 'noopener noreferrer' : undefined}
   566→                      className="inline-flex items-center gap-1 font-mono text-xs text-primary hover:underline underline-offset-2 decoration-primary/40"
   567→                      title={item.pageUrl}
   568→                    >
   569→                      <span className="truncate max-w-[240px]">{item.pageUrl}</span>
   570→                      {external ? (
   571→                        <ExternalLink className="h-3 w-3 shrink-0 opacity-50" />
   572→                      ) : (
