    83→// -------------------- Types --------------------
    84→
    85→interface AiProvider {
    86→  id: string;
    87→  name: string;
    88→  kind: AiProviderKind;
    89→  baseUrl: string;
    90→  apiKey: string | null;
    91→  apiVersion: string | null;
    92→  isActive: boolean;
    93→  isDefault: boolean;
    94→  connectionStatus: AiConnectionStatus;
    95→  latency: number | null;
    96→  lastSyncAt: string | null;
    97→  createdAt: string;
    98→  updatedAt: string;
    99→}
   100→
   101→interface ProviderFormData {
   102→  name: string;
   103→  kind: AiProviderKind;
   104→  baseUrl: string;
   105→  apiKey: string;
   106→  apiVersion: string;
   107→  isActive: boolean;
   108→}
   109→
   110→// -------------------- Constants --------------------
   111→
   112→// Provider kinds that can be selected when creating/editing a provider.
   113→// Legacy kinds (OPENROUTER, OLLAMA, AZURE_OPENAI) are kept in PROVIDER_CONFIGS
   114→// for display of existing rows but are no longer selectable.
   115→const PROVIDER_KINDS: AiProviderKind[] = [
   116→  'OPENAI', 'ANTHROPIC', 'GEMINI', 'GROQ', 'DEEPSEEK', 'CUSTOM',
   117→];
   118→
   119→const PROVIDER_CONFIGS: Record<AiProviderKind, { label: string; defaultUrl: string; color: string }> = {
   120→  OPENAI: { label: 'OpenAI', defaultUrl: 'https://api.openai.com/v1', color: 'bg-emerald-100 text-emerald-700' },
   121→  ANTHROPIC: { label: 'Anthropic', defaultUrl: 'https://api.anthropic.com/v1', color: 'bg-orange-100 text-orange-700' },
   122→  GEMINI: { label: 'Gemini', defaultUrl: 'https://generativelanguage.googleapis.com/v1', color: 'bg-sky-100 text-sky-700' },
   123→  OPENROUTER: { label: 'OpenRouter', defaultUrl: 'https://openrouter.ai/api/v1', color: 'bg-violet-100 text-violet-700' },
   124→  GROQ: { label: 'Groq', defaultUrl: 'https://api.groq.com/openai/v1', color: 'bg-rose-100 text-rose-700' },
   125→  DEEPSEEK: { label: 'DeepSeek', defaultUrl: 'https://api.deepseek.com/v1', color: 'bg-cyan-100 text-cyan-700' },
   126→  OLLAMA: { label: 'Ollama', defaultUrl: 'http://localhost:11434/v1', color: 'bg-zinc-100 text-zinc-700' },
   127→  AZURE_OPENAI: { label: 'Azure OpenAI', defaultUrl: '', color: 'bg-teal-100 text-teal-700' },
   128→  CUSTOM: { label: 'Custom', defaultUrl: '', color: 'bg-stone-100 text-stone-700' },
   129→};
   130→
   131→function kindConfig(kind: string): { label: string; color: string } {
   132→  const cfg = (PROVIDER_CONFIGS as Record<string, { label: string; defaultUrl: string; color: string }>)[kind];
   133→  return cfg ?? { label: kind, color: 'bg-zinc-100 text-zinc-700' };
   134→}
   135→
   136→const CONNECTION_STATUS_CONFIG: Record<AiConnectionStatus, { color: string; label: string }> = {
   137→  CONNECTED: { color: 'bg-green-500', label: 'Connected' },
   138→  DISCONNECTED: { color: 'bg-zinc-400', label: 'Disconnected' },
   139→  ERROR: { color: 'bg-red-500', label: 'Error' },
   140→};
   141→
   142→const emptyForm: ProviderFormData = {
   143→  name: '',
   144→  kind: 'OPENAI',
   145→  baseUrl: PROVIDER_CONFIGS.OPENAI.defaultUrl,
   146→  apiKey: '',
   147→  apiVersion: '',
   148→  isActive: true,
   149→};

   151→// -------------------- Component --------------------
   152→
   153→export function ProvidersPage() {
   154→  const queryClient = useQueryClient();
   155→  const [page, setPage] = useState(1);
   156→  const [pageSize] = useState(25);
   157→  const [search, setSearch] = useState('');
   158→  const [kindFilter, setKindFilter] = useState<string>('all');
   159→  const [statusFilter, setStatusFilter] = useState<string>('all');
   160→  const [dialogOpen, setDialogOpen] = useState(false);
   161→  const [editingProvider, setEditingProvider] = useState<AiProvider | null>(null);
   162→  const [formData, setFormData] = useState<ProviderFormData>(emptyForm);
   163→  const [showApiKey, setShowApiKey] = useState(false);
   164→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   165→  const [deletingProvider, setDeletingProvider] = useState<AiProvider | null>(null);
   166→
   167→  // Fetch providers
   168→  const { data, isLoading, isError } = useQuery({
   169→    queryKey: queryKeys.aiProviders.list({ page, pageSize, search, kind: kindFilter, connectionStatus: statusFilter }),
   170→    queryFn: () => getApi<PaginatedResponse<AiProvider>>('/api/ai/providers', {
   171→      page,
   172→      pageSize,
   173→      search: search || undefined,
   174→      kind: kindFilter !== 'all' ? kindFilter : undefined,
   175→      connectionStatus: statusFilter !== 'all' ? statusFilter : undefined,
   176→    }),
   177→  });
   178→
   179→  const providers = data?.data ?? [];
   180→  const pagination = data?.pagination;
   181→
   182→  // Create / Update mutation
   183→  const saveMutation = useMutation({
   184→    mutationFn: (body: ProviderFormData) => {
   185→      if (editingProvider) {
   186→        const patchBody: Record<string, unknown> = { ...body };
   187→        if (!patchBody.apiKey) delete patchBody.apiKey;
   188→        return patchApi<AiProvider>(`/api/ai/providers/${editingProvider.id}`, patchBody);
   189→      }
   190→      return postApi<AiProvider>('/api/ai/providers', body);
   191→    },
   192→    onSuccess: () => {
   193→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   194→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   195→      queryClient.invalidateQueries({ queryKey: queryKeys.aiSettings.all });
   196→      queryClient.invalidateQueries({ queryKey: queryKeys.aiPrompts.all });
   197→      toast.success(editingProvider ? 'Provider updated' : 'Provider created');
   198→      handleCloseDialog();
   199→    },
   200→    onError: (err: Error) => {
   201→      toast.error(err.message || 'Failed to save provider');
   202→    },
   203→  });
   204→
   205→  // Delete mutation
   206→  const deleteMutation = useMutation({
   207→    mutationFn: (id: string) => deleteApi(`/api/ai/providers/${id}`),
   208→    onSuccess: () => {
   209→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   210→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   211→      queryClient.invalidateQueries({ queryKey: queryKeys.aiSettings.all });
   212→      queryClient.invalidateQueries({ queryKey: queryKeys.aiPrompts.all });
   213→      toast.success('Provider deleted');
   214→      setDeleteDialogOpen(false);
   215→      setDeletingProvider(null);
   216→    },
   217→    onError: (err: Error) => {
   218→      toast.error(err.message || 'Failed to delete provider');
   219→    },
   220→  });
   221→
   222→  // Test connection mutation
   223→  const testMutation = useMutation({
   224→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/test`),
   225→    onSuccess: (result: unknown) => {
   226→      const res = result as { success?: boolean; latency?: number; message?: string; status?: string };
   227→      if (res.success) {
   228→        toast.success(`Connection successful (${res.latency ?? 0}ms)`);
   229→      } else {
   230→        toast.error(res.message || `Connection ${res.status || 'failed'}`);
   231→      }
   232→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   233→    },
   234→    onError: (err: Error) => {
   235→      toast.error(err.message || 'Connection test failed');
   236→    },
   237→  });
   238→
   239→  // Sync models mutation
   240→  const syncMutation = useMutation({
   241→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/sync-models`),
   242→    onSuccess: (result: unknown) => {
   243→      const res = result as { syncedCount?: number; count?: number };
   244→      toast.success(`Synced ${res.syncedCount ?? res.count ?? 0} models`);
   245→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   246→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   247→    },
   248→    onError: (err: Error) => {
   249→      toast.error(err.message || 'Failed to sync models');
   250→    },
   251→  });
   252→
   253→  // Set default mutation
   254→  const setDefaultMutation = useMutation({
   255→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/set-default`),
   256→    onSuccess: () => {
   257→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   258→      toast.success('Default provider updated');
   259→    },
   260→    onError: (err: Error) => {
   261→      toast.error(err.message || 'Failed to set default');
   262→    },
   263→  });
   264→
   265→  // Toggle active mutation
   266→  const toggleActiveMutation = useMutation({
   267→    mutationFn: ({ id, isActive }: { id: string; isActive: boolean }) =>
   268→      patchApi(`/api/ai/providers/${id}`, { isActive }),
   269→    onSuccess: () => {
   270→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   271→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   272→      queryClient.invalidateQueries({ queryKey: queryKeys.aiSettings.all });
   273→      queryClient.invalidateQueries({ queryKey: queryKeys.aiPrompts.all });
   274→    },
   275→    onError: (err: Error) => {
   276→      toast.error(err.message || 'Failed to update provider');
   277→    },
   278→  });
   279→
   280→  // Handlers
   281→  const handleOpenCreate = () => {
   282→    setEditingProvider(null);
   283→    setFormData(emptyForm);
   284→    setShowApiKey(false);
   285→    setDialogOpen(true);
   286→  };
   287→
   288→  const handleOpenEdit = (provider: AiProvider) => {
   289→    setEditingProvider(provider);
   290→    setFormData({
   291→      name: provider.name,
   292→      kind: provider.kind,
   293→      baseUrl: provider.baseUrl,
   294→      apiKey: '', // masked in edit
   295→      apiVersion: provider.apiVersion ?? '',
   296→      isActive: provider.isActive,
   297→    });
   298→    setShowApiKey(false);
   299→    setDialogOpen(true);
   300→  };
   301→
   302→  const handleCloseDialog = () => {
   303→    setDialogOpen(false);
   304→    setEditingProvider(null);
   305→    setFormData(emptyForm);
   306→    setShowApiKey(false);
   307→  };
   308→
   309→  const handleSave = () => {
   310→    if (!formData.name.trim()) {
   311→      toast.error('Name is required');
   312→      return;
   313→    }
   314→    if (!editingProvider && !formData.apiKey.trim()) {
   315→      toast.error('API key is required');
   316→      return;
   317→    }
   318→    // CUSTOM providers require a Base URL
   319→    if (formData.kind === 'CUSTOM' && !formData.baseUrl.trim()) {
   320→      toast.error('Base URL is required for Custom providers');
   321→      return;
   322→    }
   323→    // Validate Base URL format for CUSTOM providers
   324→    if (formData.kind === 'CUSTOM' && formData.baseUrl.trim()) {
   325→      try {
   326→        const u = new URL(formData.baseUrl);
   327→        if (!['http:', 'https:'].includes(u.protocol)) {
   328→          toast.error('Base URL must use http or https protocol');
   329→          return;
   330→        }
   331→      } catch {
   332→        toast.error('Base URL is not a valid URL');
   333→        return;
   334→      }
   335→    }
   336→    saveMutation.mutate(formData);
   337→  };
   338→
   339→  const handleKindChange = (kind: string) => {
   340→    const providerKind = kind as AiProviderKind;
   341→    const prevDefaultUrl = PROVIDER_CONFIGS[formData.kind]?.defaultUrl ?? '';
   342→    setFormData((prev) => ({
   343→      ...prev,
   344→      kind: providerKind,
   345→      baseUrl:
   346→        !prev.baseUrl || prev.baseUrl === prevDefaultUrl
   347→          ? PROVIDER_CONFIGS[providerKind].defaultUrl
   348→          : prev.baseUrl,
   349→    }));
   350→  };
   351→
   352→  // KPI calculations
   353→  const totalProviders = pagination?.total ?? 0;
   354→  const connectedCount = providers.filter((p) => p.connectionStatus === 'CONNECTED').length;
   355→  const defaultProvider = providers.find((p) => p.isDefault);
   356→
   357→  return (
   358→    <div className="space-y-6">
   359→      {/* KPI Cards */}
   360→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   361→        <Card>
   362→          <CardContent className="p-4 flex items-center gap-4">
   363→            <div className="p-2 rounded-lg bg-zinc-100">
   364→              <Server className="h-5 w-5 text-zinc-600" />
   365→            </div>
   366→            <div>
   367→              <p className="text-sm text-zinc-500">Total Providers</p>
   368→              <div className="text-2xl font-bold">{isLoading ? <Skeleton className="h-8 w-12 inline-block" /> : totalProviders}</div>
   369→            </div>
   370→          </CardContent>
   371→        </Card>
   372→        <Card>
   373→          <CardContent className="p-4 flex items-center gap-4">
   374→            <div className="p-2 rounded-lg bg-green-100">
   375→              <Wifi className="h-5 w-5 text-green-600" />
   376→            </div>
   377→            <div>
   378→              <p className="text-sm text-zinc-500">Connected</p>
   379→              <div className="text-2xl font-bold">{isLoading ? <Skeleton className="h-8 w-12 inline-block" /> : connectedCount}</div>
   380→            </div>
   381→          </CardContent>
   382→        </Card>
   383→        <Card>
   384→          <CardContent className="p-4 flex items-center gap-4">
   385→            <div className="p-2 rounded-lg bg-amber-100">
   386→              <Star className="h-5 w-5 text-amber-600" />
   387→            </div>
   388→            <div>
   389→              <p className="text-sm text-zinc-500">Default Provider</p>
   390→              <div className="text-lg font-bold truncate max-w-[200px]">{isLoading ? <Skeleton className="h-6 w-32 inline-block" /> : defaultProvider?.name ?? 'None'}</div>
   391→            </div>
   392→          </CardContent>
   393→        </Card>
   394→      </div>
   395→
   396→      {/* Filters & Actions */}
   397→      <Card>
   398→        <CardHeader className="pb-4">
   399→          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   400→            <CardTitle className="text-lg">AI Providers</CardTitle>
   401→            <Button onClick={handleOpenCreate} size="sm">
   402→              <Plus className="h-4 w-4 mr-2" />
   403→              Add Provider
   404→            </Button>
   405→          </div>
   406→          <div className="flex flex-col sm:flex-row gap-3 mt-2">
   407→            <div className="relative flex-1">
   408→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-zinc-400" />
   409→              <Input
   410→                placeholder="Search providers..."
   411→                value={search}
   412→                onChange={(e) => { setSearch(e.target.value); setPage(1); }}
   413→                className="pl-9"
   414→              />
   415→            </div>
   416→            <Select value={kindFilter} onValueChange={(v) => { setKindFilter(v); setPage(1); }}>
   417→              <SelectTrigger className="w-full sm:w-[180px]">
   418→                <SelectValue placeholder="Provider Kind" />
   419→              </SelectTrigger>
   420→              <SelectContent>
   421→                <SelectItem value="all">All Kinds</SelectItem>
   422→                {PROVIDER_KINDS.map((k) => (
   423→                  <SelectItem key={k} value={k}>{PROVIDER_CONFIGS[k].label}</SelectItem>
   424→                ))}
   425→              </SelectContent>
   426→            </Select>
   427→            <Select value={statusFilter} onValueChange={(v) => { setStatusFilter(v); setPage(1); }}>
   428→              <SelectTrigger className="w-full sm:w-[180px]">
   429→                <SelectValue placeholder="Status" />
   430→              </SelectTrigger>
   431→              <SelectContent>
   432→                <SelectItem value="all">All Status</SelectItem>
   433→                <SelectItem value="CONNECTED">Connected</SelectItem>
   434→                <SelectItem value="DISCONNECTED">Disconnected</SelectItem>
   435→                <SelectItem value="ERROR">Error</SelectItem>
   436→              </SelectContent>
   437→            </Select>
   438→          </div>
   439→        </CardHeader>
   440→        <CardContent className="p-0">
   441→          <ScrollArea className="max-h-[500px]">
   442→            <Table>
   443→              <TableHeader>
   444→                <TableRow>
   445→                  <TableHead>Name</TableHead>
   446→                  <TableHead>Kind</TableHead>
   447→                  <TableHead>Status</TableHead>
   448→                  <TableHead className="hidden md:table-cell">Latency</TableHead>
   449→                  <TableHead className="hidden lg:table-cell">Last Sync</TableHead>
   450→                  <TableHead>Default</TableHead>
   451→                  <TableHead>Active</TableHead>
   452→                  <TableHead className="text-right">Actions</TableHead>
   453→                </TableRow>
   454→              </TableHeader>
   455→              <TableBody>
   456→                {isLoading ? (
   457→                  Array.from({ length: 5 }).map((_, i) => (
   458→                    <TableRow key={i}>
   459→                      {Array.from({ length: 8 }).map((_, j) => (
   460→                        <TableCell key={j}><Skeleton className="h-5 w-20" /></TableCell>
   461→                      ))}
   462→                    </TableRow>
   463→                  ))
   464→                ) : isError ? (
   465→                  <TableRow>
   466→                    <TableCell colSpan={8} className="text-center py-8 text-zinc-500">
   467→                      <AlertCircle className="h-8 w-8 mx-auto mb-2 text-red-400" />
   468→                      Failed to load providers
   469→                    </TableCell>
   470→                  </TableRow>
   471→                ) : providers.length === 0 ? (
   472→                  <TableRow>
   473→                    <TableCell colSpan={8} className="text-center py-8 text-zinc-500">
   474→                      <Server className="h-8 w-8 mx-auto mb-2 text-zinc-300" />
   475→                      No providers found. Click &quot;Add Provider&quot; to get started.
   476→                    </TableCell>
   477→                  </TableRow>
   478→                ) : (
   479→                  providers.map((provider) => {
   480→                    const kc = kindConfig(provider.kind);
   481→                    const statusConfig = CONNECTION_STATUS_CONFIG[provider.connectionStatus];
   482→                    return (
   483→                      <TableRow key={provider.id}>
   484→                        <TableCell className="font-medium">{provider.name}</TableCell>
   485→                        <TableCell>
   486→                          <Badge variant="secondary" className={kc.color}>
   487→                            {kc.label}
   488→                          </Badge>
   489→                        </TableCell>
   490→                        <TableCell>
   491→                          <div className="flex items-center gap-2">
   492→                            <span className={`h-2.5 w-2.5 rounded-full ${statusConfig.color}`} />
   493→                            <span className="text-sm">{statusConfig.label}</span>
   494→                          </div>
   495→                        </TableCell>
   496→                        <TableCell className="hidden md:table-cell">
   497→                          {provider.latency != null ? `${provider.latency}ms` : '—'}
   498→                        </TableCell>
   499→                        <TableCell className="hidden lg:table-cell text-sm text-zinc-500">
   500→                          {provider.lastSyncAt
   501→                            ? new Date(provider.lastSyncAt).toLocaleDateString()
   502→                            : '—'}
   503→                        </TableCell>
   504→                        <TableCell>
   505→                          {provider.isDefault && (
   506→                            <Badge variant="secondary" className="bg-amber-100 text-amber-700">
   507→                              Default
   508→                            </Badge>
   509→                          )}
   510→                        </TableCell>
   511→                        <TableCell>
   512→                          <Switch
   513→                            checked={provider.isActive}
   514→                            disabled={
   515→                              toggleActiveMutation.isPending &&
   516→                              toggleActiveMutation.variables?.id === provider.id
   517→                            }
   518→                            onCheckedChange={(checked) =>
   519→                              toggleActiveMutation.mutate({ id: provider.id, isActive: checked })
   520→                            }
   521→                          />
   522→                        </TableCell>
   523→                        <TableCell className="text-right">
   524→                          <DropdownMenu>
   525→                            <DropdownMenuTrigger asChild>
   526→                              <Button variant="ghost" size="sm">
   527→                                <MoreHorizontal className="h-4 w-4" />
   528→                              </Button>
   529→                            </DropdownMenuTrigger>
   530→                            <DropdownMenuContent align="end">
   531→                              <DropdownMenuItem
   532→                                onClick={() => testMutation.mutate(provider.id)}
   533→                                disabled={testMutation.variables === provider.id && testMutation.isPending}
   534→                              >
   535→                                <Zap className="h-4 w-4 mr-2" />
   536→                                Test Connection
   537→                              </DropdownMenuItem>
   538→                              <DropdownMenuItem
   539→                                onClick={() => syncMutation.mutate(provider.id)}
   540→                                disabled={syncMutation.variables === provider.id && syncMutation.isPending}
   541→                              >
   542→                                <RefreshCw className="h-4 w-4 mr-2" />
   543→                                Sync Models
   544→                              </DropdownMenuItem>
   545→                              {!provider.isDefault && (
   546→                                <DropdownMenuItem onClick={() => setDefaultMutation.mutate(provider.id)}>
   547→                                  <Star className="h-4 w-4 mr-2" />
   548→                                  Set Default
   549→                                </DropdownMenuItem>
   550→                              )}
   551→                              <DropdownMenuSeparator />
   552→                              <DropdownMenuItem onClick={() => handleOpenEdit(provider)}>
   553→                                <Pencil className="h-4 w-4 mr-2" />
   554→                                Edit
   555→                              </DropdownMenuItem>
   556→                              <DropdownMenuItem
   557→                                className="text-red-600"
   558→                                onClick={() => { setDeletingProvider(provider); setDeleteDialogOpen(true); }}
   559→                              >
   560→                                <Trash2 className="h-4 w-4 mr-2" />
   561→                                Delete
   562→                              </DropdownMenuItem>
   563→                            </DropdownMenuContent>
   564→                          </DropdownMenu>
   565→                        </TableCell>
   566→                      </TableRow>
   567→                    );
   568→                  })
   569→                )}
   570→              </TableBody>
   571→            </Table>
   572→            <ScrollBar />
   573→          </ScrollArea>
   574→
   575→          {/* Pagination */}
   576→          {pagination && pagination.totalPages > 1 && (
   577→            <div className="flex items-center justify-between px-4 py-3 border-t">
   578→              <p className="text-sm text-zinc-500">
   579→                Showing {(pagination.page - 1) * pagination.pageSize + 1}–
   580→                {Math.min(pagination.page * pagination.pageSize, pagination.total)} of {pagination.total}
   581→              </p>
   582→              <div className="flex items-center gap-2">
   583→                <Button
   584→                  variant="outline" size="sm"
   585→                  disabled={pagination.page <= 1}
   586→                  onClick={() => setPage((p) => p - 1)}
   587→                >
   588→                  <ChevronLeft className="h-4 w-4" />
   589→                </Button>
   590→                <span className="text-sm">Page {pagination.page} of {pagination.totalPages}</span>
   591→                <Button
   592→                  variant="outline" size="sm"
   593→                  disabled={pagination.page >= pagination.totalPages}
   594→                  onClick={() => setPage((p) => p + 1)}
   595→                >
   596→                  <ChevronRight className="h-4 w-4" />
   597→                </Button>
   598→              </div>
   599→            </div>
   600→          )}
   601→        </CardContent>
   602→      </Card>
   603→
   604→      {/* Add/Edit Dialog */}
   605→      <Dialog open={dialogOpen} onOpenChange={(open) => { if (!open) handleCloseDialog(); }}>
   606→        <DialogContent className="sm:max-w-[500px] max-h-[90vh] overflow-y-auto">
   607→          <DialogHeader>
   608→            <DialogTitle>{editingProvider ? 'Edit Provider' : 'Add Provider'}</DialogTitle>
   609→          </DialogHeader>
   610→          <div className="grid gap-4 py-4">
   611→            <div className="grid gap-2">
   612→              <Label htmlFor="provider-name">Name</Label>
   613→              <Input
   614→                id="provider-name"
   615→                placeholder="e.g. My OpenAI"
   616→                value={formData.name}
   617→                onChange={(e) => setFormData((p) => ({ ...p, name: e.target.value }))}
   618→              />
   619→            </div>
   620→            <div className="grid gap-2">
   621→              <Label htmlFor="provider-kind">Provider Kind</Label>
   622→              <Select value={formData.kind} onValueChange={handleKindChange}>
   623→                <SelectTrigger id="provider-kind">
   624→                  <SelectValue />
   625→                </SelectTrigger>
   626→                <SelectContent>
   627→                  {PROVIDER_KINDS.map((k) => (
   628→                    <SelectItem key={k} value={k}>{PROVIDER_CONFIGS[k].label}</SelectItem>
   629→                  ))}
   630→                </SelectContent>
   631→              </Select>
   632→            </div>
   633→            <div className="grid gap-2">
   634→              <Label htmlFor="provider-url">
   635→                Base URL{formData.kind === 'CUSTOM' ? ' *' : ''}
   636→              </Label>
   637→              <Input
   638→                id="provider-url"
   639→                placeholder={formData.kind === 'CUSTOM' ? 'https://api.example.com/v1' : 'https://api.openai.com/v1'}
   640→                value={formData.baseUrl}
   641→                onChange={(e) => setFormData((p) => ({ ...p, baseUrl: e.target.value }))}
   642→                required={formData.kind === 'CUSTOM'}
   643→              />
   644→              {formData.kind === 'CUSTOM' && (
   645→                <p className="text-xs text-muted-foreground">
   646→                  Enter the base URL of your OpenAI-compatible provider (e.g. https://api.example.com/v1). The provider must expose the OpenAI-compatible /chat/completions and /models endpoints.
   647→                </p>
   648→              )}
   649→            </div>
   650→            <div className="grid gap-2">
   651→              <Label htmlFor="provider-key">API Key</Label>
   652→              <div className="relative">
   653→                <Input
   654→                  id="provider-key"
   655→                  type={showApiKey ? 'text' : 'password'}
   656→                  placeholder={editingProvider ? 'Leave blank to keep existing key' : 'sk-...'}
   657→                  value={formData.apiKey}
   658→                  onChange={(e) => setFormData((p) => ({ ...p, apiKey: e.target.value }))}
   659→                />
   660→                <Button
   661→                  type="button"
   662→                  variant="ghost"
   663→                  size="sm"
   664→                  className="absolute right-1 top-1/2 -translate-y-1/2 h-7 w-7 p-0"
   665→                  onClick={() => setShowApiKey(!showApiKey)}
   666→                >
   667→                  {showApiKey ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   668→                </Button>
   669→              </div>
   670→              {editingProvider && (
   671→                <p className="text-xs text-muted-foreground">
   672→                  Leave blank to keep the existing API key. The key is stored encrypted and never displayed in full.
   673→                </p>
   674→              )}
   675→            </div>
   676→            {formData.kind === 'AZURE_OPENAI' && (
   677→              <p className="text-xs text-muted-foreground">
   678→                Note: Azure OpenAI is a legacy kind and no longer configurable from this form.
   679→              </p>
   680→            )}
   681→            <div className="flex items-center justify-between">
   682→              <Label htmlFor="provider-active">Active</Label>
   683→              <Switch
   684→                id="provider-active"
   685→                checked={formData.isActive}
   686→                onCheckedChange={(checked) => setFormData((p) => ({ ...p, isActive: checked }))}
   687→              />
   688→            </div>
   689→          </div>
   690→          <DialogFooter>
   691→            <Button variant="outline" onClick={handleCloseDialog}>Cancel</Button>
   692→            <Button onClick={handleSave} disabled={saveMutation.isPending}>
   693→              {saveMutation.isPending && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   694→              {editingProvider ? 'Update' : 'Create'}
   695→            </Button>
   696→          </DialogFooter>
   697→        </DialogContent>
   698→      </Dialog>
   699→
   700→      {/* Delete Confirmation */}
   701→      <AlertDialog open={deleteDialogOpen} onOpenChange={setDeleteDialogOpen}>
   702→        <AlertDialogContent>
   703→          <AlertDialogHeader>
   704→            <AlertDialogTitle>Delete Provider</AlertDialogTitle>
   705→            <AlertDialogDescription>
   706→              Are you sure you want to delete &quot;{deletingProvider?.name}&quot;? This action cannot be undone.
   707→            </AlertDialogDescription>
   708→          </AlertDialogHeader>
   709→          <AlertDialogFooter>
   710→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   711→            <AlertDialogAction
   712→              className="bg-red-600 hover:bg-red-700"
   713→              onClick={() => deletingProvider && deleteMutation.mutate(deletingProvider.id)}
   714→              disabled={deleteMutation.isPending}
   715→            >
   716→              {deleteMutation.isPending && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   717→              Delete
   718→            </AlertDialogAction>
   719→          </AlertDialogFooter>
   720→        </AlertDialogContent>
   721→      </AlertDialog>
   722→    </div>
   723→  );
   724→}
