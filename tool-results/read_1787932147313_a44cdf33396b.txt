   272→   272→// -------------------- Category Count Query --------------------
   273→   273→
   274→   274→function useCategoryCounts() {
   275→   275→  return useQuery({
   276→   276→    queryKey: ['email-templates', 'category-counts'],
   277→   277→    queryFn: async () => {
   278→   278→      const allCategories = CATEGORIES.filter((c) => c.value !== 'ALL');
   279→   279→      const results = await Promise.allSettled(
   280→   280→        allCategories.map(async (cat) => {
   281→   281→          const res = await getApi<{ data: EmailTemplate[]; meta: { pagination: { total: number } } }>(
   282→   282→            '/api/email-templates',
   283→   283→            { category: cat.value, page: 1, pageSize: 1 },
   284→   284→            { raw: true },
   285→   285→          );
   286→   286→          return { category: cat.value, count: res?.meta?.pagination?.total ?? 0 };
   287→   287→        }),
   288→   288→      );
   289→   289→      const counts: Record<string, number> = {};
   290→   290→      for (const r of results) {
   291→   291→        if (r.status === 'fulfilled') {
   292→   292→          counts[r.value.category] = r.value.count;
   293→   293→        }
   294→   294→      }
   295→   295→      return counts;
   296→   296→    },
   297→   297→    staleTime: 30_000,
   298→   298→  });
   299→   299→}
