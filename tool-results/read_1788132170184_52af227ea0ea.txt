     1→     1→// ============================================================
     2→     2→// PLAN CONFIG — DB-backed, cached single source of truth for plans,
     3→     3→// pricing, entitlements and usage limits.
     4→     4→// ============================================================
     5→     5→// The Client Billing API and the Platform Admin both read from (and
     6→     6→// the admin mutates) this same service. `platform-data.ts` delegates
     7→     7→// its `PLANS` / `getPlan` / `monthlyPrice` to the synchronous cache
     8→     8→// exposed here, so an owner edit (savePlanConfig) propagates to the
     9→     9→// client billing experience and to MRR on the next read.
    10→    10→//
    11→    11→// The cache is hydrated from the PlanConfig table on startup and after
    12→    12→// every mutation. If the table is empty on first hydrate, the DEFAULT
    13→    13→// configs are seeded so the platform is never without plans. If the
    14→    14→// table contains the LEGACY catalog (Beta/Pro/Max/Enterprise), the
    15→    15→// migration runs automatically: Enterprise is removed (only when no
    16→    16→// active subscription references it), Beta is renamed to Plus, and a
    17→    17→// Free plan is inserted at sortOrder 0.
    18→    18→//
    19→    19→// Final canonical catalog:
    20→    20→//   1. Free  (planId='free',  isFree=true,  priceMonthly=0)
    21→    21→//   2. Plus  (planId='plus',  isFree=false, priceMonthly=0 or 9)
    22→    22→//   3. Pro   (planId='pro',   priceMonthly=49)
    23→    23→//   4. Max   (planId='max',   priceMonthly=99)

   188→   188→function notify() {
   189→   189→  for (const cb of _subscribers) {
   190→   190→    try {
   191→   191→      cb();
   192→   192→    } catch {
   193→   193→      // ignore subscriber errors
   194→   194→    }
   195→   195→  }
   196→   196→}

   198→   198→/** Register a callback fired after every cache refresh (hydrate / mutation). */
   199→   199→export function subscribe(cb: () => void): () => void {
   200→   200→  _subscribers.add(cb);
   201→   201→  return () => _subscribers.delete(cb);
   202→   202→}

   315→   315→/**
   316→   316→ * Migrate the legacy catalog to the new Free/Plus/Pro/Max shape. Idempotent.
   317→   317→ *
   318→   318→ * Legacy catalog (created by older bootstrap):
   319→   319→ *   beta (free, active=false), pro, max, enterprise (stub)
   320→   320→ *
   321→   321→ * Migration steps (each guarded, can be re-run safely):
   322→   322→ *  1. If `enterprise` exists AND no active subscription references it → delete it.
   323→   323→ *     (Active subs that reference enterprise are preserved — the migration will
   324→   324→ *      not break them; the plan row stays if needed. In practice, the demo
   325→   325→ *      dataset has no DB subscriptions yet, so this is a clean delete.)
   326→   326→ *  2. If `beta` exists → rename to `plus` (planId, name, badgeVariant) and
   327→   327→ *     re-enable it (legacy `beta` was active=false in the demo DB). Also
   328→   328→ *     shift sortOrder to 1 (Free takes 0).
   329→   329→ *  3. If `free` does NOT exist → insert the Free plan at sortOrder 0.
   330→   330→ *  4. Renumber sortOrder to match the canonical order: free=0, plus=1, pro=2, max=3.
   331→   331→ *
   332→   332→ * Returns a short report string for the bootstrap log.
   333→   333→ */
   334→   334→export async function migrateLegacyPlans(): Promise<string> {
   335→   335→  const notes: string[] = [];
   336→   336→
   337→   337→  try {
   338→   338→    // Step 1: delete Enterprise if safe
   339→   339→    const enterprise = await db.planConfig.findUnique({ where: { planId: 'enterprise' } });
   340→   340→    if (enterprise) {
   341→   341→      const subCount = await db.subscription.count({ where: { planId: 'enterprise' } });
   342→   342→      if (subCount === 0) {
   343→   343→        await db.planConfig.delete({ where: { planId: 'enterprise' } });
   344→   344→        notes.push('deleted enterprise plan (0 active subscriptions)');
   345→   345→      } else {
   346→   346→        notes.push(`kept enterprise plan (${subCount} active subscriptions reference it)`);
   347→   347→      }
   348→   348→    }
   349→   349→
   350→   350→    // Step 2: rename beta → plus (and refresh to a real paid tier)
   351→   351→    const beta = await db.planConfig.findUnique({ where: { planId: 'beta' } });
   352→   352→    if (beta) {
   353→   353→      // Don't overwrite an existing plus row — if both exist, prefer the
   354→   354→      // plus row and just delete the beta stub.
   355→   355→      const plus = await db.planConfig.findUnique({ where: { planId: 'plus' } });
   356→   356→      if (!plus) {
   357→   357→        // Rename beta → plus AND refresh the data to the real DEFAULT
   358→   358→        // Plus tier (paid plan with real entitlements + limits, not the
   359→   359→        // legacy free-tier Beta stub). The old Beta was active=false and
   360→   360→        // had no real customers, so this is a safe refresh.
   361→   361→        const plusDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'plus')!;
   362→   362→        await db.planConfig.update({
   363→   363→          where: { planId: 'beta' },
   364→   364→          data: {
   365→   365→            ...dataToRow(plusDefault),
   366→   366→            // Keep the original createdAt for continuity.
   367→   367→            // (updatedAt is auto.)
   368→   368→          },
   369→   369→        });
   370→   370→        notes.push('renamed + refreshed beta → plus (real paid tier)');
   371→   371→      } else {
   372→   372→        // Both beta and plus exist — delete the beta stub (migrate any
   373→   373→        // legacy subscriptions on beta to plus first).
   374→   374→        const betaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   375→   375→        if (betaSubs > 0) {
   376→   376→          await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   377→   377→          notes.push(`migrated ${betaSubs} beta subscription(s) → plus`);
   378→   378→        }
   379→   379→        await db.planConfig.delete({ where: { planId: 'beta' } });
   380→   380→        notes.push('deleted legacy beta stub (plus already exists)');
   381→   381→      }
   382→   382→    }
   383→   383→
   384→   384→    // Step 3: insert Free if missing
   385→   385→    const free = await db.planConfig.findUnique({ where: { planId: 'free' } });
   386→   386→    if (!free) {
   387→   387→      const freeDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'free')!;
   388→   388→      await db.planConfig.create({ data: dataToRow(freeDefault) });
   389→   389→      notes.push('inserted free plan');
   390→   390→    } else {
   391→   391→      // Ensure free plan is properly flagged
   392→   392→      if (!free.isFree || free.badgeVariant !== 'free' || free.sortOrder !== 0) {
   393→   393→        await db.planConfig.update({
   394→   394→          where: { planId: 'free' },
   395→   395→          data: { isFree: true, badgeVariant: 'free', sortOrder: 0 },
   396→   396→        });
   397→   397→        notes.push('normalized free plan flags');
   398→   398→      }
   399→   399→    }
   400→   400→
   401→   401→    // Step 4: normalize sortOrder for all canonical plans
   402→   402→    const canonicalOrder: Record<string, number> = { free: 0, plus: 1, pro: 2, max: 3 };
   403→   403→    for (const [pid, order] of Object.entries(canonicalOrder)) {
   404→   404→      const row = await db.planConfig.findUnique({ where: { planId: pid } });
   405→   405→      if (row && row.sortOrder !== order) {
   406→   406→        await db.planConfig.update({ where: { planId: pid }, data: { sortOrder: order } });
   407→   407→      }
   408→   408→    }
   409→   409→
   410→   410→    // Also migrate legacy subscriptions on 'beta' or 'enterprise' planId
   411→   411→    // to the new ids (beta → plus, enterprise → max) so historical subs
   412→   412→    // don't reference a missing plan.
   413→   413→    const legacyBetaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   414→   414→    if (legacyBetaSubs > 0) {
   415→   415→      await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   416→   416→      notes.push(`migrated ${legacyBetaSubs} legacy beta subscription(s) → plus`);
   417→   417→    }
   418→   418→    const legacyEnterpriseSubs = await db.subscription.count({ where: { planId: 'enterprise' } });
   419→   419→    if (legacyEnterpriseSubs > 0) {
   420→   420→      await db.subscription.updateMany({ where: { planId: 'enterprise' }, data: { planId: 'max' } });
   421→   421→      notes.push(`migrated ${legacyEnterpriseSubs} legacy enterprise subscription(s) → max`);
   422→   422→    }
   423→   423→  } catch (err) {
   424→   424→    notes.push(`migration error (non-fatal): ${(err as Error).message}`);
   425→   425→  }
   426→   426→
   427→   427→  return notes.length > 0 ? notes.join('; ') : 'no migration needed';
   428→   428→}

   430→   430→/** Load all plan configs from the DB into the cache. Self-seeds the
   431→   431→ *  DEFAULT_PLAN_CONFIGS when the table is empty. Runs the legacy
   432→   432→ *  migration when the catalog still contains 'beta' or 'enterprise'.
   433→   433→ *  Safe to call repeatedly. */
   434→   434→export async function hydrate(): Promise<void> {
   435→   435→  if (_hydrating) return _hydrating;
   436→   436→  _hydrating = (async () => {
   437→   437→    try {
   438→   438→      const rows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   439→   439→      if (rows.length === 0) {
   440→   440→        // Self-seed so the platform is never without plans.
   441→   441→        await db.planConfig.createMany({ data: DEFAULT_PLAN_CONFIGS.map(dataToRow) });
   442→   442→        _cache = DEFAULT_PLAN_CONFIGS.map((p) => ({
   443→   443→          ...p,
   444→   444→          features: [...p.features],
   445→   445→          entitlements: [...p.entitlements],
   446→   446→          limits: { ...p.limits },
   447→   447→        }));
   448→   448→      } else {
   449→   449→        // Check for legacy catalog (beta / enterprise) and migrate.
   450→   450→        const hasLegacy =
   451→   451→          rows.some((r) => r.planId === 'beta' || r.planId === 'enterprise') ||
   452→   452→          !rows.some((r) => r.planId === 'free');
   453→   453→        if (hasLegacy) {
   454→   454→          await migrateLegacyPlans();
   455→   455→          const migratedRows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   456→   456→          _cache = migratedRows.map(rowToData);
   457→   457→        } else {
   458→   458→          _cache = rows.map(rowToData);
   459→   459→        }
   460→   460→      }
   461→   461→      _hydrated = true;
   462→   462→      notify();
   463→   463→    } catch (err) {
   464→   464→      // DB unavailable — keep defaults (still functional, just not persisted).
   465→   465→      _hydrated = true;
   466→   466→      notify();
   467→   467→    } finally {
   468→   468→      _hydrating = null;
   469→   469→    }
   470→   470→  })();
   471→   471→  return _hydrating;
   472→   472→}

   494→   494→/** Update an existing plan config. Writes DB, refreshes cache.
   495→   495→ *
   496→   496→ *  AUTO-SYNC TO STRIPE: when Stripe is configured AND the plan is
   497→   497→ *  PAID (priceMonthly > 0 OR priceYearly > 0) AND the admin did NOT
   498→   498→ *  explicitly set the Stripe Price IDs in this patch (i.e. the field
   499→   499→ *  is omitted from `patch`), the backend calls syncPlanToStripe to
   500→   500→ *  ensure a Stripe Product + monthly + yearly Prices exist and writes
   501→   501→ *  the resolved Stripe Price IDs back onto the row. This means the
   502→   502→ *  admin can edit a paid plan's name / features / limits and the
   503→   503→ *  Stripe side stays in sync without manually wiring price IDs.
   504→   504→ *
   505→   505→ *  When the admin explicitly sets `stripePriceIdMonthly` / `stripePriceIdYearly`
   506→   506→ *  (even to null = "clear"), the auto-sync is skipped — the admin is
   507→   507→ *  taking manual control of the Stripe side. They can use the dedicated
   508→   508→ *  "Sync to Stripe" route (/api/platform/admin/plans/[planId]/sync-stripe)
   509→   509→ *  to push the local plan to Stripe at any time. */
   510→   510→export async function savePlanConfig(planId: string, patch: PlanConfigInput): Promise<PlanConfigData | null> {
   511→   511→  const existing = await db.planConfig.findUnique({ where: { planId } });
   512→   512→  if (!existing) return null;
   513→   513→  const current = rowToData(existing);
   514→   514→
   515→   515→  // Distinguish "field omitted" from "field set to null". When the
   516→   516→  // admin omits both stripePriceId* fields, they want the backend to
   517→   517→  // keep them in sync with Stripe automatically (see AUTO-SYNC above).
   518→   518→  // When either is explicitly provided (including null to clear),
   519→   519→  // the admin is taking manual control.
   520→   520→  const adminTouchedStripePriceIds =
   521→   521→    'stripePriceIdMonthly' in patch || 'stripePriceIdYearly' in patch;
   522→   522→
   523→   523→  const next: PlanConfigData = {
   524→   524→    planId: current.planId,
   525→   525→    name: patch.name ?? current.name,
   526→   526→    priceMonthly: patch.priceMonthly ?? current.priceMonthly,
   527→   527→    priceYearly: patch.priceYearly ?? current.priceYearly,
   528→   528→    currency: patch.currency ?? current.currency,
   529→   529→    interval: patch.interval ?? current.interval,
   530→   530→    isFree: patch.isFree ?? current.isFree,
   531→   531→    freePlanDurationDays: patch.freePlanDurationDays ?? current.freePlanDurationDays,
   532→   532→    stripePriceIdMonthly: patch.stripePriceIdMonthly ?? current.stripePriceIdMonthly,
   533→   533→    stripePriceIdYearly: patch.stripePriceIdYearly ?? current.stripePriceIdYearly,
   534→   534→    active: patch.active ?? current.active,
   535→   535→    features: patch.features ?? current.features,
   536→   536→    entitlements: patch.entitlements ?? current.entitlements,
   537→   537→    limits: { ...current.limits, ...(patch.limits ?? {}) },
   538→   538→    badgeVariant: patch.badgeVariant ?? current.badgeVariant,
   539→   539→    sortOrder: patch.sortOrder ?? current.sortOrder,
   540→   540→  };
   541→   541→  await db.planConfig.update({ where: { planId }, data: dataToRow(next) });
   542→   542→  await hydrate();
   543→   543→
   544→   544→  // ---- AUTO-SYNC TO STRIPE ----
   545→   545→  // Best-effort — never throws. Swallows Stripe errors so an admin
   546→   546→  // edit doesn't fail just because Stripe is unreachable. The admin
   547→   547→  // can use the explicit "Sync to Stripe" route to surface errors.
   548→   548→  if (!adminTouchedStripePriceIds && !next.isFree && (next.priceMonthly > 0 || next.priceYearly > 0)) {
   549→   549→    try {
   550→   550→      const { isStripeConfiguredAsync, getStripeClient, syncPlanToStripe } = await import('@/lib/stripe');
   551→   551→      if (await isStripeConfiguredAsync()) {
   552→   552→        const stripe = await getStripeClient();
   553→   553→        const syncResult = await syncPlanToStripe(stripe, {
   554→   554→          planId: next.planId,
   555→   555→          name: next.name,
   556→   556→          priceMonthly: next.priceMonthly,
   557→   557→          priceYearly: next.priceYearly,
   558→   558→          currency: next.currency,
   559→   559→          stripePriceIdMonthly: next.stripePriceIdMonthly,
   560→   560→          stripePriceIdYearly: next.stripePriceIdYearly,
   561→   561→        });
   562→   562→        // Persist the resolved Stripe Price IDs back onto the row
   563→   563→        // (only when they changed — otherwise the update is a no-op).
   564→   564→        if (
   565→   565→          syncResult.stripePriceIdMonthly !== next.stripePriceIdMonthly ||
   566→   566→          syncResult.stripePriceIdYearly !== next.stripePriceIdYearly
   567→   567→        ) {
   568→   568→          await db.planConfig.update({
   569→   569→            where: { planId },
   570→   570→            data: {
   571→   571→              stripePriceIdMonthly: syncResult.stripePriceIdMonthly,
   572→   572→              stripePriceIdYearly: syncResult.stripePriceIdYearly,
   573→   573→            },
   574→   574→          });
   575→   575→          await hydrate();
   576→   576→        }
   577→   577→      }
   578→   578→    } catch {
   579→   579→      // best-effort — never block the save on a Stripe error
   580→   580→    }
   581→   581→  }
   582→   582→
   583→   583→  return getPlanConfigSync(planId);
   584→   584→}
   585→   585→
   586→   586→/** Create a new plan config.
   587→   587→ *
   588→   588→ *  AUTO-SYNC TO STRIPE: when Stripe is configured AND the new plan
   589→   589→ *  is PAID (priceMonthly > 0 OR priceYearly > 0) AND the admin did
   590→   590→ *  NOT explicitly provide Stripe Price IDs in the input, the backend
   591→   591→ *  calls syncPlanToStripe to create a Stripe Product + monthly +
   592→   592→ *  yearly Prices and writes the resolved Stripe Price IDs back onto
   593→   593→ *  the new plan row. So creating a new paid plan in Platform Admin
   594→   594→ *  automatically creates the corresponding Stripe Product + Prices —
   595→   595→ *  no need to manually wire Stripe Price IDs in the Stripe dashboard.
   596→   596→ *
   597→   597→ *  When the admin explicitly provides Stripe Price IDs (manually
   598→   598→ *  created in the Stripe dashboard), the auto-sync is skipped — the
   599→   599→ *  admin is asserting the Stripe side is already wired. */

   680→   680→/**
   681→   681→ * Delete a plan config. Refuses if:
   682→   682→ *  - it's the last remaining plan, OR
   683→   683→ *  - any Subscription still references this planId (would orphan the sub).
   684→   684→ *
   685→   685→ * Returns `{ ok: boolean, reason?: string }` describing the outcome.
   686→   686→ */
   687→   687→export async function deletePlanConfig(
   688→   688→  planId: string,
   689→   689→): Promise<{ ok: boolean; reason?: string }> {
   690→   690→  const count = await db.planConfig.count();
   691→   691→  if (count <= 1) return { ok: false, reason: 'Cannot delete the last plan.' };
   692→   692→  const subCount = await db.subscription.count({ where: { planId } });
   693→   693→  if (subCount > 0) {
   694→   694→    return {
   695→   695→      ok: false,
   696→   696→      reason: `Cannot delete plan "${planId}" — ${subCount} active subscription(s) reference it. Migrate them first.`,
   697→   697→    };
   698→   698→  }
   699→   699→  try {
   700→   700→    await db.planConfig.delete({ where: { planId } });
   701→   701→    await hydrate();
   702→   702→    return { ok: true };
   703→   703→  } catch (err) {
   704→   704→    return { ok: false, reason: (err as Error).message };
   705→   705→  }
   706→   706→}
