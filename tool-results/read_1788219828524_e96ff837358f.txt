   159→// -------------------- Collapsed-Rail Tooltip Positioning -------------------
   160→/*
   161→ * SINGLE source of truth for the positioning of every tooltip/label that
   162→ * appears when the sidebar is collapsed into the 48px icon rail.
   163→ *
   164→ *   side="right"         → opens to the RIGHT of the 48px collapsed rail,
   165→ *                          fully inside the main viewport (never overlaps
   166→ *                          the rail, never opens to the left/outside).
   167→ *   align="center"       → vertically centers the bubble on the icon's
   168→ *                          32×32 hover target so labels line up across
   169→ *                          every row of the rail (logo, theme, nav items,
   170→ *                          parent items with submenu).
   171→ *   sideOffset=8         → ~8px visible gap from the trigger button's right
   172→ *                          edge (button right edge ~x=40 inside the 48px
   173→ *                          rail, tooltip left edge ~x=48 = the rail's
   174→ *                          right edge — flush with the RAIL, NOT with the
   175→ *                          icon, so the bubble never touches the glyph).
   176→ *   collisionPadding=12 → 12px viewport-edge collision padding so the
   177→ *                          bubble can never be clipped at any viewport
   178→ *                          edge or scroll past the visible area.
   179→ *
   180→ * Portal-based rendering (Radix TooltipPrimitive.Portal → document.body,
   181→ * wired inside src/components/ui/tooltip.tsx) means every tooltip is
   182→ * rendered OUTSIDE the sidebar DOM tree — so the sidebar's
   183→ * overflow-hidden / overflow-x-hidden on the rail container can NEVER
   184→ * clip or hide the bubble. The tooltip floats above every other
   185→ * element at z-50.
   186→ *
   187→ * Used by:
   188→ *   • CollapsedLogoButton         (header "Expand" tooltip)
   189→ *   • SimpleNavItem              (leaf nav items: Dashboard, Articles,
   190→ *                                  Calendar, Media, Users, Comments,
   191→ *                                  Newsletter, SEO, AI, Automation)
   192→ *   • ExpandableNavItem          (parent nav items in EXPANDED state —
   193→ *                                  tooltip is hidden by SidebarMenuButton
   194→ *                                  when state!=="collapsed", but the
   195→ *                                  positioning object is still applied
   196→ *                                  so the prop stays consistent across
   197→ *                                  every nav item renderer)
   198→ *   • CollapsedParentNavItem     (parent nav items in COLLAPSED state —
   199→ *                                  label tooltip shows when the floating
   200→ *                                  popover is CLOSED; when the popover is
   201→ *                                  open, no tooltip shows to avoid visual
   202→ *                                  conflict)
   203→ *
   204→ * The ThemeToggle component (src/components/layout/theme-toggle.tsx) lives
   205→ * in a separate file and inlines the SAME four values — see the comment
   206→ * there for the rationale.
   207→ */

   428→/**
   429→ * Collapsed-rail logo cell. Since the extra collapse-toggle icon was removed
   430→ * from below the logo, THIS cell is the explicit expand control:
   431→ *   • identical 32×32 grid geometry as every other rail icon
   432→ *   • click toggles the sidebar open (SidebarRail edge strip still works)
   433→ * No new icon is introduced and no empty clickable ghost remains.
   434→ *
   435→ * HOVER BEHAVIOR (collapsed state only — this button is rendered ONLY
   436→ * inside the collapsed-rail cluster, so the expanded LogoMark is untouched):
   437→ * The `hovered` prop is LIFTED to the rail level (see AppSidebar's
   438→ * `railHovered` state + the onMouseEnter/onMouseLeave handlers on the
   439→ * <Sidebar> root). This means the swap fires when the mouse is ANYWHERE
   440→ * over the collapsed 48px rail — NOT only when the pointer is directly
   441→ * over the logo cell:
   442→ *   • rail at rest ........  shows the "C" logo mark on its bg-primary
   443→ *                            black box (NORMAL LOGO — background UNCHANGED,
   444→ *                            same as the expanded LogoMark).
   445→ *   • mouse over ANY rail   the "C" is replaced by a PanelLeftOpen icon
   446→ *     item (Dashboard,       AND the button's background becomes
   447→ *     Articles, …, Profile,  transparent so ONLY the Expand icon is
   448→ *     Theme, Settings, …)    visible with no box behind it. The icon
   449→ *                            color switches to muted-foreground (gray)
   450→ *                            so it stays visible against the
   451→ *                            transparent/page bg in both Light and Dark.
   452→ *   • mouse leaves the     restores the "C" logo mark AND its bg-primary
   453→ *     entire rail            box (onMouseLeave on the <Sidebar> root).
   454→ *   • click ............. calls toggleSidebar (existing functionality,
   455→ *                         unchanged).
   456→ *
   457→ * Each rail item keeps its OWN Radix Tooltip label (Dashboard, Articles,
   458→ * Profile, etc.) — the lifted `hovered` prop ONLY drives the logo cell's
   459→ * C↔PanelLeftOpen swap; it does NOT touch any item's tooltip. The logo's
   460→ * own "Expand" Tooltip still fires ONLY when the pointer is directly over
   461→ * the logo button (the Tooltip trigger is the button itself, independent
   462→ * of the `hovered` prop).
   463→ *
   464→ * Background is removed ONLY from the temporary Expand-icon state; the
   465→ * normal logo (at rest) keeps its bg-primary background exactly as-is.
   466→ *
   467→ * Implementation: React state lifted to AppSidebar (railHovered +
   468→ * onMouseEnter/onMouseLeave on <Sidebar>) — NOT CSS :hover / group-hover
   469→ * variants. This is deliberate: Tailwind v4 wraps its hover:/group-hover:
   470→ * utilities inside @media (hover: hover), so on any browser that reports
   471→ * (hover: none) (headless, touch, some preview iframes) those utilities
   472→ * NEVER activate and the swap + bg-transparent would silently fail. React
   473→ * mouse events fire on ANY pointer input regardless of the hover media
   474→ * query, so the behavior is identical in every environment. The
   475→ * conditional className + conditional render swap the bg/color/glyph
   476→ * atomically per state. Works in both Light and Dark.
   477→ */
   478→function CollapsedLogoButton({ hovered }: { hovered: boolean }) {
   479→  const { toggleSidebar } = useSidebar();
   480→  return (
   481→    /* Hover-only "Expand" tooltip: instant (provider delayDuration=0),
   482→       appears on mouse-enter, disappears on mouse-leave. Radix closes it
   483→       on any pointer exit — disableHoverableContent guarantees it can
   484→       never linger or trap the pointer. The Tooltip trigger is the logo
   485→       button itself, so the "Expand" TEXT fires ONLY on direct logo hover
   486→       (NOT on rail-level hover of other items) — the C↔PanelLeftOpen ICON
   487→       swap, by contrast, is driven by the lifted `hovered` prop and fires
   488→       for ANY rail item. */
   489→    <Tooltip disableHoverableContent>
   490→      <TooltipTrigger asChild>
   491→        <button
   492→          type="button"
   493→          onClick={toggleSidebar}
   494→          aria-label="Expand sidebar"
   495→          className={cn(
   496→            'flex h-8 w-8 shrink-0 cursor-pointer items-center justify-center rounded-lg font-bold text-sm outline-none transition-opacity hover:opacity-90 focus-visible:ring-2 focus-visible:ring-ring select-none',
   497→            // NORMAL LOGO STATE (at rest): bg-primary black box +
   498→            // text-primary-foreground white "C" — background UNCHANGED,
   499→            // same as the expanded LogoMark.
   500→            // EXPAND-ICON STATE (on hover): bg-transparent (NO background)
   501→            // + text-muted-foreground gray icon — ONLY this temporary
   502→            // state loses the background. The icon stays visible (gray
   503→            // on transparent/page bg) in both Light and Dark mode.
   504→            hovered
   505→              ? 'bg-transparent text-muted-foreground'
   506→              : 'bg-primary text-primary-foreground',
   507→          )}
   508→        >
   509→          {/* Conditional render: "C" logo at rest, PanelLeftOpen icon on
   510→              hover. h-4 w-4 matches the CollapseToggle's PanelLeftClose
   511→              icon size; the icon is the visual opposite of the collapse
   512→              icon so the collapsed logo reads as a clear "expand"
   513→              affordance when the pointer is over it. */}
   514→          {hovered ? (
   515→            <PanelLeftOpen className="h-4 w-4" />
   516→          ) : (
   517→            <span>C</span>
   518→          )}
   519→          <span className="sr-only">Expand sidebar</span>
   520→        </button>
   521→      </TooltipTrigger>
   522→      {/* Collapsed-rail tooltip — uses the SHARED COLLAPSED_TOOLTIP_PROPS
   523→          constant so the "Expand" label has IDENTICAL positioning (side,
   524→          align, sideOffset, collisionPadding) to every other collapsed-
   525→          rail tooltip (ThemeToggle, SimpleNavItem, ExpandableNavItem,
   526→          CollapsedParentNavItem). Portal-based rendering means the bubble
   527→          floats outside the sidebar DOM at z-50 — never clipped by the
   528→          rail's overflow-hidden. */}
   529→      <TooltipContent {...COLLAPSED_TOOLTIP_PROPS}>Expand</TooltipContent>
   530→    </Tooltip>
   531→  );
   532→}

   604→ * Clicking it opens a portal-based popover to the RIGHT of the icon —
   605→ * the sidebar itself NEVER expands, and the popover anchors to this
   606→ * exact button so the "star/sparkle alignment" grid stays intact.
   607→ */
   608→function CollapsedParentNavItem({
   609→  item,
   610→  currentModule,
   611→  currentSubPage,
   612→}: {
   613→  item: NavItem;
   614→  currentModule: string;
   615→  currentSubPage: string | null;
   616→}) {
   617→  const [floatOpen, setFloatOpen] = useState(false);
   618→  const mod = hrefToModule(item.href);
   619→  const isActive = currentModule === mod;
   620→
   621→  const handleChildNavigate = (
   622→    e: React.MouseEvent<HTMLAnchorElement>,
   623→    child: NavItem,
   624→  ) => {
   625→    e.preventDefault();
   626→    e.stopPropagation();
   627→    setFloatOpen(false); // close the floating submenu after navigation…
   628→    // …but do NOT touch sidebar expansion state (rule #7).
   629→    const hash = child.href.replace(/^#/, '');
   630→    const parts = hash.split('/');
   631→    const childSubPage = parts[1] || null;
   632→    useNavigationStore.getState().navigate(parts[0], null, childSubPage);
   633→  };
   634→
   635→  return (
   636→    <SidebarMenuItem>
   637→      <Popover open={floatOpen} onOpenChange={setFloatOpen}>
   638→        <PopoverTrigger asChild>
   639→          <SidebarMenuButton
   640→            isActive={isActive || floatOpen}
   641→            aria-expanded={floatOpen}
   642→            aria-haspopup="menu"
   643→            // Collapsed-rail tooltip — uses the SHARED COLLAPSED_TOOLTIP_PROPS
   644→            // so the parent item's label has IDENTICAL positioning (side,
   645→            // align, sideOffset, collisionPadding) to every other collapsed-
   646→            // rail tooltip (CollapsedLogoButton, ThemeToggle, SimpleNavItem,
   647→            // ExpandableNavItem). Suppressed when the popover is open so the
   648→            // floating submenu never visually conflicts with a lingering
   649→            // label bubble. Portal-based rendering means the bubble floats
   650→            // outside the sidebar DOM at z-50 — never clipped by the rail's
   651→            // overflow-hidden.
   652→            tooltip={
   653→              floatOpen
   654→                ? undefined
   655→                : { ...COLLAPSED_TOOLTIP_PROPS, children: item.label }
   656→            }
   657→            onClick={(e: React.MouseEvent) => {
   658→              e.preventDefault();
   659→              setFloatOpen((o) => !o);
   660→            }}
   661→          >
   662→            <NavIcon name={item.icon} />
   663→            <span>{item.label}</span>
   664→          </SidebarMenuButton>
   665→        </PopoverTrigger>
   666→        {/* Portal → document.body: immune to sidebar overflow clipping.
   667→            Radix handles Esc + outside-click closing and re-anchors at any
   668→            viewport size/zoom via avoidCollisions. */}
   669→        <PopoverContent
   670→          side="right"
   671→          align="start"
   672→          sideOffset={6}
   673→          collisionPadding={12}
   674→          className="w-56 rounded-md border bg-popover p-1.5 shadow-md"
   675→          onEscapeKeyDown={() => setFloatOpen(false)}
   676→        >
   677→          <ul
   678→            role="menu"
   679→            aria-label={`${item.label} submenu`}
   680→            className="flex min-w-0 list-none flex-col gap-0.5 p-0 m-0"
   681→          >
   682→            {item.children!.map((child) => {
   683→              const hash = child.href.replace(/^#/, '');
   684→              const parts = hash.split('/');
   685→              const childSubPage = parts[1] || null;
   686→              const isChildActive =
   687→                currentModule === parts[0] &&
   688→                (!childSubPage || currentSubPage === childSubPage);
   689→              return (
   690→                <SidebarMenuSubItem key={child.label} className="list-none p-0">
   691→                  <SidebarMenuSubButton asChild isActive={isChildActive}>
   692→                    <a
   693→                      href={child.href}
   694→                      role="menuitem"
   695→                      onClick={(e) => handleChildNavigate(e, child)}
   696→                    >
   697→                      <NavIcon name={child.icon} />
   698→                      <span>{child.label}</span>
   699→                    </a>
   700→                  </SidebarMenuSubButton>
   701→                </SidebarMenuSubItem>
   702→              );
   703→            })}
   704→          </ul>
   705→        </PopoverContent>
   706→      </Popover>
   707→    </SidebarMenuItem>
   708→  );
   709→}

   711→function ExpandableNavItem({
   712→  item,
   713→  currentModule,
   714→  currentSubPage,
   715→  isExpanded,
   716→  onToggle,
   717→}: {
   718→  item: NavItem;
   719→  currentModule: string;
   720→  currentSubPage: string | null;
   721→  isExpanded: boolean;
   722→  onToggle: () => void;
   723→}) {
   724→  const mod = hrefToModule(item.href);
   725→  const isActive = currentModule === mod;
   726→  const sectionId = `submenu-${item.label.toLowerCase().replace(/\s+/g, '-')}`;
   727→
   728→  return (
   729→    <SidebarMenuItem>
   730→      <SidebarMenuButton
   731→        // Expandable parent item rendered in EXPANDED state. SidebarMenuButton
   732→        // hides its tooltip whenever state!=="collapsed" (see ui/sidebar.tsx),
   733→        // so this label never shows while expanded — but passing the SAME
   734→        // COLLAPSED_TOOLTIP_PROPS object as every other nav-item renderer
   735→        // keeps the prop consistent across the codebase. If the sidebar
   736→        // toggles between expanded and collapsed while this row stays
   737→        // mounted, the tooltip that appears is identical in positioning to
   738→        // SimpleNavItem / CollapsedParentNavItem / CollapsedLogoButton.
   739→        tooltip={{ ...COLLAPSED_TOOLTIP_PROPS, children: item.label }}
   740→        isActive={isActive}
   741→        onClick={(e: React.MouseEvent) => {
   742→          e.preventDefault();
   743→          onToggle();
   744→        }}
   745→        aria-expanded={isExpanded}
   746→        aria-controls={sectionId}
   747→      >
   748→        <NavIcon name={item.icon} />
   749→        <span>{item.label}</span>
   750→        <ChevronRight
   751→          className={cn(
   752→            'ml-auto h-4 w-4 shrink-0 transition-transform duration-200',
   753→            isExpanded && 'rotate-90',
   754→          )}
   755→        />
   756→      </SidebarMenuButton>
   757→
   758→      {item.badge != null && (
   759→        <SidebarMenuBadge>
   760→          {item.badge}
   761→        </SidebarMenuBadge>
   762→      )}
   763→
   764→      <AccordionSubmenu isOpen={isExpanded} sectionLabel={item.label}>
   765→        {item.children!.map((child) => {
   766→          const hash = child.href.replace(/^#/, '');
   767→          const parts = hash.split('/');
   768→          const childMod = parts[0];
   769→          const childSubPage = parts[1] || null;
   770→          const isChildActive =
   771→            currentModule === childMod &&
   772→            (!childSubPage || currentSubPage === childSubPage);
   773→
   774→          return (
   775→            <SidebarMenuSubItem key={child.label}>
   776→              <SidebarMenuSubButton
   777→                asChild
   778→                isActive={isChildActive}
   779→              >
   780→                <a
   781→                  href={child.href}
   782→                  onClick={(e) => {
   783→                    e.preventDefault();
   784→                    e.stopPropagation();
   785→                    useNavigationStore
   786→                      .getState()
   787→                      .navigate(childMod, null, childSubPage);
   788→                  }}
   789→                >
   790→                  <NavIcon name={child.icon} />
   791→                  <span>{child.label}</span>
   792→                </a>
   793→              </SidebarMenuSubButton>
   794→            </SidebarMenuSubItem>
   795→          );
   796→        })}
   797→      </AccordionSubmenu>
   798→    </SidebarMenuItem>
   799→  );

   851→function NavGroupSection({
   852→  group,
   853→  currentModule,
   854→  currentSubPage,
   855→  openSection,
   856→  onToggleSection,
   857→}: {
   858→  group: NavGroup;
   859→  currentModule: string;
   860→  currentSubPage: string | null;
   861→  openSection: string | null;
   862→  onToggleSection: (label: string) => void;
   863→}) {
   864→  const isCollapsedRail = useCollapsedRail();
   865→
   866→  return (
   867→    <SidebarGroup>
   868→      <SidebarGroupContent>
   869→        <SidebarMenu>
   870→          {group.items.map((item) => {
   871→            const hasChildren = item.children && item.children.length > 0;
   872→
   873→            if (hasChildren && isCollapsedRail) {
   874→              /* Collapsed rail → parents with children become floating
   875→                 popovers anchored to their own icon. The rail NEVER
   876→                 auto-expands (#4/#5/#6). */
   877→              return (
   878→                <CollapsedParentNavItem
   879→                  key={item.label}
   880→                  item={item}
   881→                  currentModule={currentModule}
   882→                  currentSubPage={currentSubPage}
   883→                />
   884→              );
   885→            }
   886→
   887→            if (hasChildren) {
   888→              return (
   889→                <ExpandableNavItem
   890→                  key={item.label}
   891→                  item={item}
   892→                  currentModule={currentModule}
   893→                  currentSubPage={currentSubPage}
   894→                  isExpanded={openSection === item.label}
   895→                  onToggle={() => onToggleSection(item.label)}
   896→                />
   897→              );
   898→            }
   899→
   900→            return (
   901→              <SimpleNavItem
   902→                key={item.label}
   903→                item={item}
   904→                currentModule={currentModule}
   905→              />
   906→            );
   907→          })}
   908→        </SidebarMenu>
   909→      </SidebarGroupContent>
   910→    </SidebarGroup>
   911→  );
   912→}

   954→          : undefined,

   956→      .filter((item) => isModuleAllowedByPlan(pageKeyOf(item.href), planEntitlements));

   973→  const [prevModule, setPrevModule] = useState(currentModule);
