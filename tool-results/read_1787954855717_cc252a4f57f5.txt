    50→// -------------------- Model Resolution Helper --------------------
    51→// The frontend sends DB cuids as `modelId` (e.g. "m-openai-gpt5"), but the
    52→// upstream provider APIs need the actual model string (e.g. "gpt-5").
    53→// This helper resolves a DB cuid to the AiModel row, validates ownership +
    54→// active status + type, and returns the model row. If no modelId is provided,
    55→// it falls back to AI Settings defaults, then the provider's default model.
    56→
    57→interface ResolvedModel {
    58→  modelId: string;        // upstream model string (e.g. "gpt-5")
    59→  modelDbId: string;      // DB cuid (e.g. "m-openai-gpt5")
    60→  inputCostPer1k: number | null;
    61→  outputCostPer1k: number | null;
    62→  type: string;           // 'TEXT' | 'IMAGE'
    63→}
    64→
    65→async function resolveModel(
    66→  providerId: string,
    67→  modelId: string | undefined,
    68→  expectedType: 'TEXT' | 'IMAGE',
    69→  providerModels: Array<{ id: string; modelId: string; providerId: string; isActive: boolean; isDefault: boolean; type: string; inputCostPer1k: number | null; outputCostPer1k: number | null }>,
    70→): Promise<ResolvedModel> {
    71→  // If a modelId is provided, it's a DB cuid — look it up
    72→  if (modelId) {
    73→    const model = providerModels.find((m) => m.id === modelId);
    74→    if (!model) {
    75→      throw new Error('The selected model was not found for this provider. Please select a valid model.');
    76→    }
    77→    if (model.providerId !== providerId) {
    78→      throw new Error('The selected model does not belong to the selected provider.');
    79→    }
    80→    if (!model.isActive) {
    81→      throw new Error('The selected model is inactive. Please activate it or select another model.');
    82→    }
    83→    if (model.type?.toUpperCase() !== expectedType) {
    84→      throw new Error(`The selected model is not a ${expectedType.toLowerCase()} model. Please select a ${expectedType.toLowerCase()} model.`);
    85→    }
    86→    return {
    87→      modelId: model.modelId,
    88→      modelDbId: model.id,
    89→      inputCostPer1k: model.inputCostPer1k,
    90→      outputCostPer1k: model.outputCostPer1k,
    91→      type: model.type,
    92→    };
    93→  }
    94→
    95→  // No modelId provided — fall back to AI Settings defaults
    96→  const settings = await db.aiSettings.findUnique({ where: { scope: 'global' } });
    97→  const settingsModelId = expectedType === 'TEXT' ? settings?.defaultModelId : settings?.imageModelId;
    98→  if (settingsModelId) {
    99→    const model = providerModels.find((m) => m.id === settingsModelId && m.isActive && m.type?.toUpperCase() === expectedType);
   100→    if (model) {
   101→      return {
   102→        modelId: model.modelId,
   103→        modelDbId: model.id,
   104→        inputCostPer1k: model.inputCostPer1k,
   105→        outputCostPer1k: model.outputCostPer1k,
   106→        type: model.type,
   107→      };
   108→    }
   109→  }
   110→
   111→  // Fall back to the provider's default model of the correct type
   112→  const defaultModel = providerModels.find((m) => m.isActive && m.isDefault && m.type?.toUpperCase() === expectedType)
   113→    ?? providerModels.find((m) => m.isActive && m.type?.toUpperCase() === expectedType);
   114→  if (!defaultModel) {
   115→    throw new Error(`No active ${expectedType.toLowerCase()} model is configured for this provider. Please add or activate a model.`);
   116→  }
   117→  return {
   118→    modelId: defaultModel.modelId,
   119→    modelDbId: defaultModel.id,
   120→    inputCostPer1k: defaultModel.inputCostPer1k,
   121→    outputCostPer1k: defaultModel.outputCostPer1k,
   122→    type: defaultModel.type,
   123→  };
   124→}

   128→export async function executeChat(req: ChatRequest): Promise<ChatResponse> {
   129→  const provider = await db.aiProvider.findUnique({
   130→    where: { id: req.providerId },
   131→    include: { models: true },
   132→  });
   133→
   134→  if (!provider) throw new Error('Provider not found');
   135→  if (!provider.isActive) throw new Error('Provider is disabled. Please activate it first.');
   136→  if (!provider.apiKeyEncrypted) throw new Error('API key not configured for this provider.');
   137→
   138→  // Resolve + validate the model (handles DB cuid → model string, type=TEXT, active, belongs-to-provider)
   139→  const resolved = await resolveModel(req.providerId, req.modelId, 'TEXT', provider.models);
   140→  const modelId = resolved.modelId;
   141→
   142→  // Apply AI Settings defaults for temperature/maxTokens if not provided
   143→  const settings = await db.aiSettings.findUnique({ where: { scope: 'global' } });
   144→  const temperature = req.temperature ?? settings?.defaultTemperature ?? 0.7;
   145→  const maxTokens = req.maxTokens ?? settings?.defaultMaxTokens ?? 2048;
   146→
   147→  const apiKey = await decrypt(provider.apiKeyEncrypted);
   148→  const config = getProviderConfig(provider.kind);
   149→  const baseUrl = provider.baseUrl || config.defaultBaseUrl;
   150→  // CUSTOM providers have no defaultBaseUrl — they must have one set explicitly.
   151→  if (!baseUrl) {
   152→    throw new Error('No Base URL configured for this custom provider. Please edit the provider and set a Base URL.');
   153→  }
   154→
   155→  const startTime = Date.now();
   156→  let inputTokens = 0;
   157→  let outputTokens = 0;
   158→  let content = '';
   159→  let usedProvider = provider;
   160→  let usedModelId = modelId;
   161→  // Track the resolved model for cost calculation — updated when a fallback succeeds
   162→  // so cost is calculated using the fallback's rates, not the primary's.
   163→  let usedResolved = resolved;
   164→
   165→  try {
   166→    if (provider.kind === 'ANTHROPIC') {
   167→      const result = await callAnthropic(baseUrl, apiKey, modelId, req.messages, {
   168→        temperature, maxTokens, jsonMode: req.jsonMode,
   169→      });
   170→      inputTokens = result.inputTokens;
   171→      outputTokens = result.outputTokens;
   172→      content = result.content;
   173→    } else if (provider.kind === 'GEMINI') {
   174→      const result = await callGemini(baseUrl, apiKey, modelId, req.messages, {
   175→        temperature, maxTokens, jsonMode: req.jsonMode,
   176→      });
   177→      inputTokens = result.inputTokens;
   178→      outputTokens = result.outputTokens;
   179→      content = result.content;
   180→    } else {
   181→      // OpenAI-compatible (OpenAI, Groq, DeepSeek, Custom)
   182→      const result = await callOpenAI(baseUrl, apiKey, modelId, req.messages, {
   183→        temperature, maxTokens,
   184→        topP: req.topP,
   185→        frequencyPenalty: req.frequencyPenalty,
   186→        presencePenalty: req.presencePenalty,
   187→        jsonMode: req.jsonMode,
   188→      });
   189→      inputTokens = result.inputTokens;
   190→      outputTokens = result.outputTokens;
   191→      content = result.content;
   192→    }
   193→  } catch (err) {
   194→    // Try fallback providers
   195→    const fallbacks = await db.aiProviderFallback.findMany({
   196→      where: { providerId: provider.id },
   197→      include: { fallback: { include: { models: true } } },
   198→      orderBy: { priority: 'asc' },
   199→    });
   200→
   201→    let lastError = err instanceof Error ? err : new Error('Unknown error');
   202→
   203→    for (const fb of fallbacks) {
   204→      if (!fb.fallback.isActive || !fb.fallback.apiKeyEncrypted) continue;
   205→      try {
   206→        const fbResolved = await resolveModel(fb.fallback.id, undefined, 'TEXT', fb.fallback.models);
   207→        const fbApiKey = await decrypt(fb.fallback.apiKeyEncrypted);
   208→        const fbConfig = getProviderConfig(fb.fallback.kind);
   209→        const fbBaseUrl = fb.fallback.baseUrl || fbConfig.defaultBaseUrl;
   210→        // CUSTOM providers have no defaultBaseUrl — skip if the admin hasn't set one.
   211→        if (!fbBaseUrl) continue;
   212→
   213→        let fbResult: { content: string; inputTokens: number; outputTokens: number };
   214→        if (fb.fallback.kind === 'ANTHROPIC') {
   215→          fbResult = await callAnthropic(fbBaseUrl, fbApiKey, fbResolved.modelId, req.messages, { temperature, maxTokens });
   216→        } else if (fb.fallback.kind === 'GEMINI') {
   217→          fbResult = await callGemini(fbBaseUrl, fbApiKey, fbResolved.modelId, req.messages, { temperature, maxTokens });
   218→        } else {
   219→          // OpenAI-compatible (OpenAI, Groq, DeepSeek, Custom)
   220→          fbResult = await callOpenAI(fbBaseUrl, fbApiKey, fbResolved.modelId, req.messages, { temperature, maxTokens });
   221→        }
   222→        inputTokens = fbResult.inputTokens;
   223→        outputTokens = fbResult.outputTokens;
   224→        content = fbResult.content;
   225→        usedProvider = fb.fallback;
   226→        usedModelId = fbResolved.modelId;
   227→        usedResolved = fbResolved; // update cost rates to the fallback's
   228→        lastError = null as unknown as Error;
   229→        break;
   230→      } catch (fbErr) {
   231→        lastError = fbErr instanceof Error ? fbErr : new Error('Fallback failed');
   232→        continue;
   233→      }
   234→    }
   235→
   236→    if (!content) {
   237→      // Log the failed request
   238→      const durationMs = Date.now() - startTime;
   239→      await db.aiLog.create({
   240→        data: {
   241→          providerId: provider.id,
   242→          providerName: provider.name,
   243→          modelId,
   244→          question: req.messages.map((m) => m.content).join('\n'),
   245→          response: null,
   246→          inputTokens: 0,
   247→          outputTokens: 0,
   248→          totalTokens: 0,
   249→          costUsd: 0,
   250→          durationMs,
   251→          status: 'error',
   252→          errorMessage: lastError?.message ?? 'Unknown error',
   253→          siteId: req.siteId,
   254→          userId: req.userId,
   255→        },
   256→      }).catch(() => { /* logging failure shouldn't mask the original error */ });
   257→      throw lastError ?? new Error('Chat request failed');
   258→    }
   259→  }
   260→
   261→  const durationMs = Date.now() - startTime;
   262→  const totalTokens = inputTokens + outputTokens;
   263→  const costUsd = (inputTokens / 1000) * (usedResolved.inputCostPer1k || 0)
   264→    + (outputTokens / 1000) * (usedResolved.outputCostPer1k || 0);
   265→
   266→  // Update provider
   267→  await db.aiProvider.update({
   268→    where: { id: usedProvider.id },
   269→    data: {
   270→      lastUsedAt: new Date(),
   271→      latencyMs: durationMs,
   272→      connectionStatus: 'CONNECTED',
   273→      lastError: null,
   274→    },
   275→  });
   276→
   277→  // Log the request
   278→  await db.aiLog.create({
   279→    data: {
   280→      providerId: usedProvider.id,
   281→      providerName: usedProvider.name,
   282→      modelId: usedModelId,
   283→      question: req.messages.map((m) => m.content).join('\n'),
   284→      response: content,
   285→      inputTokens,
   286→      outputTokens,
   287→      totalTokens,
   288→      costUsd,
   289→      durationMs,
   290→      status: 'success',
   291→      siteId: req.siteId,
   292→      userId: req.userId,
   293→    },
   294→  });
   295→
   296→  return {
   297→    content,
   298→    model: usedModelId,
   299→    inputTokens,
   300→    outputTokens,
   301→    totalTokens,
   302→    costUsd,
   303→    durationMs,
   304→    providerName: usedProvider.name,
   305→  };
   306→}

   308→// -------------------- Provider-specific call implementations --------------------
   309→
   310→async function callOpenAI(
   311→  baseUrl: string,
   312→  apiKey: string,
   313→  model: string,
   314→  messages: ChatMessage[],
   315→  opts: { temperature?: number; maxTokens?: number; topP?: number; frequencyPenalty?: number; presencePenalty?: number; jsonMode?: boolean; apiVersion?: string } = {},
   316→) {
   317→  const url = opts.apiVersion
   318→    ? `${baseUrl}?api-version=${opts.apiVersion}`
   319→    : `${baseUrl}/chat/completions`;
   320→
   321→  const body: Record<string, unknown> = {
   322→    model,
   323→    messages,
   324→    temperature: opts.temperature ?? 0.7,
   325→    max_tokens: opts.maxTokens ?? 2048,
   326→  };
   327→  if (opts.topP !== undefined) body.top_p = opts.topP;
   328→  if (opts.frequencyPenalty !== undefined) body.frequency_penalty = opts.frequencyPenalty;
   329→  if (opts.presencePenalty !== undefined) body.presence_penalty = opts.presencePenalty;
   330→  if (opts.jsonMode) {
   331→    body.response_format = { type: 'json_object' };
   332→  }
   333→
   334→  const res = await fetch(url, {
   335→    method: 'POST',
   336→    headers: {
   337→      'Content-Type': 'application/json',
   338→      'Authorization': `Bearer ${apiKey}`,
   339→    },
   340→    body: JSON.stringify(body),
   341→  });
   342→
   343→  if (!res.ok) {
   344→    const err = await res.text();
   345→    throw new Error(`OpenAI API error: ${res.status} — ${err}`);
   346→  }
   347→
   348→  const data = await res.json();
   349→  return {
   350→    content: data.choices?.[0]?.message?.content || '',
   351→    inputTokens: data.usage?.prompt_tokens || 0,
   352→    outputTokens: data.usage?.completion_tokens || 0,
   353→  };
   354→}

   356→async function callAnthropic(
   357→  baseUrl: string,
   358→  apiKey: string,
   359→  model: string,
   360→  messages: ChatMessage[],
   361→  opts: { temperature?: number; maxTokens?: number; jsonMode?: boolean } = {},
   362→) {
   363→  const systemMsg = messages.find(m => m.role === 'system');
   364→  const chatMsgs = messages.filter(m => m.role !== 'system');
   365→
   366→  const body: Record<string, unknown> = {
   367→    model,
   368→    messages: chatMsgs.map(m => ({ role: m.role, content: m.content })),
   369→    max_tokens: opts.maxTokens ?? 2048,
   370→    temperature: opts.temperature ?? 0.7,
   371→  };
   372→  if (systemMsg) body.system = systemMsg.content;
   373→
   374→  const res = await fetch(`${baseUrl}/messages`, {
   375→    method: 'POST',
   376→    headers: {
   377→      'Content-Type': 'application/json',
   378→      'x-api-key': apiKey,
   379→      'anthropic-version': '2023-06-01',
   380→    },
   381→    body: JSON.stringify(body),
   382→  });
   383→
   384→  if (!res.ok) {
   385→    const err = await res.text();
   386→    throw new Error(`Anthropic API error: ${res.status} — ${err}`);
   387→  }
   388→
   389→  const data = await res.json();
   390→  return {
   391→    content: data.content?.[0]?.text || '',
   392→    inputTokens: data.usage?.input_tokens || 0,
   393→    outputTokens: data.usage?.output_tokens || 0,
   394→  };
   395→}

   397→async function callGemini(
   398→  baseUrl: string,
   399→  apiKey: string,
   400→  model: string,
   401→  messages: ChatMessage[],
   402→  opts: { temperature?: number; maxTokens?: number; jsonMode?: boolean } = {},
   403→) {
   404→  const systemMsg = messages.find(m => m.role === 'system');
   405→  const chatMsgs = messages.filter(m => m.role !== 'system');
   406→
   407→  const body: Record<string, unknown> = {
   408→    contents: chatMsgs.map(m => ({
   409→      role: m.role === 'assistant' ? 'model' : 'user',
   410→      parts: [{ text: m.content }],
   411→    })),
   412→    generationConfig: {
   413→      temperature: opts.temperature ?? 0.7,
   414→      maxOutputTokens: opts.maxTokens ?? 2048,
   415→    },
   416→  };
   417→
   418→  const res = await fetch(
   419→    `${baseUrl}/models/${model}:generateContent?key=${apiKey}`,
   420→    {
   421→      method: 'POST',
   422→      headers: { 'Content-Type': 'application/json' },
   423→      body: JSON.stringify(body),
   424→    },
   425→  );
   426→
   427→  if (!res.ok) {
   428→    const err = await res.text();
   429→    throw new Error(`Gemini API error: ${res.status} — ${err}`);
   430→  }
   431→
   432→  const data = await res.json();
   433→  return {
   434→    content: data.candidates?.[0]?.content?.parts?.[0]?.text || '',
   435→    inputTokens: data.usageMetadata?.promptTokenCount || 0,
   436→    outputTokens: data.usageMetadata?.candidatesTokenCount || 0,
   437→  };
   438→}

   442→export async function healthCheck(providerId: string): Promise<HealthCheckResult> {
   443→  const provider = await db.aiProvider.findUnique({
   444→    where: { id: providerId },
   445→    include: { models: { where: { isActive: true } } },
   446→  });
   447→  if (!provider) throw new Error('Provider not found');
   448→  if (!provider.apiKeyEncrypted) return { status: 'DISCONNECTED', latencyMs: 0, error: 'No API key configured' };
   449→
   450→  const apiKey = await decrypt(provider.apiKeyEncrypted);
   451→  const config = getProviderConfig(provider.kind);
   452→  const baseUrl = provider.baseUrl || config.defaultBaseUrl;
   453→  // CUSTOM providers have no defaultBaseUrl — they must have one set explicitly.
   454→  if (!baseUrl) {
   455→    return { status: 'DISCONNECTED', latencyMs: 0, error: 'No Base URL configured for this custom provider. Please edit the provider and set a Base URL.' };
   456→  }
   457→
   458→  const start = Date.now();
   459→  try {
   460→    let models: ProviderModel[] = [];
   461→
   462→    if (provider.kind === 'ANTHROPIC') {
   463→      // Anthropic has no /models endpoint — send a minimal chat request to verify the API key.
   464→      // Use the provider's first active TEXT model, or fall back to a known-good default.
   465→      const testModel = provider.models.find((m) => m.type?.toUpperCase() === 'TEXT')?.modelId
   466→        ?? config.defaultModels.find((m) => !isImageModelId(m.modelId))?.modelId
   467→        ?? 'claude-3-5-haiku-20241022';
   468→      const res = await fetch(`${baseUrl}/messages`, {
   469→        method: 'POST',
   470→        headers: { 'Content-Type': 'application/json', 'x-api-key': apiKey, 'anthropic-version': '2023-06-01' },
   471→        body: JSON.stringify({ model: testModel, max_tokens: 1, messages: [{ role: 'user', content: 'hi' }] }),
   472→      });
   473→      if (!res.ok) {
   474→        const errBody = await res.text().catch(() => '');
   475→        throw new Error(`HTTP ${res.status}${errBody ? `: ${errBody.slice(0, 200)}` : ''}`);
   476→      }
   477→      models = config.defaultModels;
   478→    } else if (provider.kind === 'GEMINI') {
   479→      const res = await fetch(`${baseUrl}/models?key=${apiKey}`);
   480→      if (!res.ok) {
   481→        const errBody = await res.text().catch(() => '');
   482→        throw new Error(`HTTP ${res.status}${errBody ? `: ${errBody.slice(0, 200)}` : ''}`);
   483→      }
   484→      models = config.defaultModels;
   485→    } else if (config.modelsEndpoint) {
   486→      // OpenAI-compatible (OpenAI, Groq, DeepSeek, Custom)
   487→      const res = await fetch(`${baseUrl}${config.modelsEndpoint}`, {
   488→        headers: { 'Authorization': `Bearer ${apiKey}` },
   489→      });
   490→      if (!res.ok) {
   491→        const errBody = await res.text().catch(() => '');
   492→        throw new Error(`HTTP ${res.status}${errBody ? `: ${errBody.slice(0, 200)}` : ''}`);
   493→      }
   494→      const data = await res.json();
   495→      if (data.data && Array.isArray(data.data)) {
   496→        models = data.data.map((m: { id: string }) => ({
   497→          modelId: m.id, name: m.id, contextLength: 0,
   498→          inputCostPer1k: 0, outputCostPer1k: 0,
   499→          supportsImages: false, supportsVision: false,
   500→          supportsFunctionCalling: false, supportsJsonMode: false,
   501→          supportsStreaming: true, supportsTools: false,
   502→        }));
   503→      } else {
   504→        models = config.defaultModels;
   505→      }
   506→    }
   507→
   508→    const latencyMs = Date.now() - start;
   509→
   510→    await db.aiProvider.update({
   511→      where: { id: providerId },
   512→      data: {
   513→        connectionStatus: 'CONNECTED',
   514→        latencyMs,
   515→        lastHealthCheckAt: new Date(),
   516→        lastError: null,
   517→      },
   518→    });
   519→
   520→    return { status: 'CONNECTED', latencyMs, availableModels: models };
   521→  } catch (err) {
   522→    const latencyMs = Date.now() - start;
   523→    const errorMsg = err instanceof Error ? err.message : 'Unknown error';
   524→
   525→    await db.aiProvider.update({
   526→      where: { id: providerId },
   527→      data: {
   528→        connectionStatus: 'ERROR',
   529→        latencyMs,
   530→        lastError: errorMsg,
   531→        lastHealthCheckAt: new Date(),
   532→      },
   533→    });
   534→
   535→    return { status: 'ERROR', latencyMs, error: errorMsg };
   536→  }
   537→}

   541→export async function syncModels(providerId: string): Promise<number> {
   542→  const provider = await db.aiProvider.findUnique({ where: { id: providerId } });
   543→  if (!provider) throw new Error('Provider not found');
   544→  if (!provider.apiKeyEncrypted) throw new Error('No API key configured for this provider.');
   545→
   546→  const apiKey = await decrypt(provider.apiKeyEncrypted);
   547→  const config = getProviderConfig(provider.kind);
   548→  const baseUrl = provider.baseUrl || config.defaultBaseUrl;
   549→  // CUSTOM providers have no defaultBaseUrl — they must have one set explicitly.
   550→  if (!baseUrl) {
   551→    throw new Error('No Base URL configured for this custom provider. Please edit the provider and set a Base URL.');
   552→  }
   553→
   554→  let fetchedModels: ProviderModel[] = config.defaultModels;
   555→
   556→  // Try to fetch from API if endpoint exists (Anthropic has no /models endpoint)
   557→  if (config.modelsEndpoint && provider.kind !== 'ANTHROPIC') {
   558→    try {
   559→      const res = await fetch(`${baseUrl}${config.modelsEndpoint}`, {
   560→        headers: { 'Authorization': `Bearer ${apiKey}` },
   561→      });
   562→      if (res.ok) {
   563→        const data = await res.json();
   564→        if (data.data && Array.isArray(data.data)) {
   565→          const existingIds = new Set(config.defaultModels.map((m) => m.modelId));
   566→          const newModels = data.data
   567→            .filter((m: { id: string }) => !existingIds.has(m.id))
   568→            .map((m: { id: string }) => ({
   569→              modelId: m.id, name: m.id, contextLength: 0,
   570→              inputCostPer1k: 0, outputCostPer1k: 0,
   571→              supportsImages: false, supportsVision: false,
   572→              supportsFunctionCalling: false, supportsJsonMode: false,
   573→              supportsStreaming: true, supportsTools: false,
   574→            }));
   575→          fetchedModels = [...config.defaultModels, ...newModels];
   576→        }
   577→      }
   578→    } catch { /* use defaults */ }
   579→  }
   580→
   581→  // Determine the correct type for each model
   582→  const isImage = (mid: string) => isImageModelId(mid);
   583→
   584→  // Upsert models + set type correctly.
   585→  // For existing models (update): only overwrite fields that the upstream API
   586→  // actually provides meaningful data for. Preserve admin-set cost/name overrides
   587→  // by only updating from config.defaultModels (which have real cost data) —
   588→  // API-fetched models with zero cost/context don't overwrite admin edits.
   589→  let count = 0;
   590→  for (const model of fetchedModels) {
   591→    const modelType = isImage(model.modelId) ? 'IMAGE' : 'TEXT';
   592→    const isFromDefaults = config.defaultModels.some((m) => m.modelId === model.modelId);
   593→
   594→    await db.aiModel.upsert({
   595→      where: { providerId_modelId: { providerId, modelId: model.modelId } },
   596→      update: {
   597→        // Only update cost/capability fields from the default config (which has real data).
   598→        // API-fetched models with zeros don't overwrite admin-set values.
   599→        ...(isFromDefaults ? {
   600→          name: model.name,
   601→          contextLength: model.contextLength,
   602→          inputCostPer1k: model.inputCostPer1k,
   603→          outputCostPer1k: model.outputCostPer1k,
   604→          supportsImages: model.supportsImages,
   605→          supportsVision: model.supportsVision,
   606→          supportsFunctionCalling: model.supportsFunctionCalling,
   607→          supportsJsonMode: model.supportsJsonMode,
   608→          supportsStreaming: model.supportsStreaming,
   609→          supportsTools: model.supportsTools,
   610→        } : {}),
   611→        // Always update type + lastSyncedAt
   612→        type: modelType,
   613→        lastSyncedAt: new Date(),
   614→      },
   615→      create: {
   616→        providerId,
   617→        modelId: model.modelId,
   618→        name: model.name,
   619→        contextLength: model.contextLength,
   620→        inputCostPer1k: model.inputCostPer1k,
   621→        outputCostPer1k: model.outputCostPer1k,
   622→        supportsImages: model.supportsImages,
   623→        supportsVision: model.supportsVision,
   624→        supportsFunctionCalling: model.supportsFunctionCalling,
   625→        supportsJsonMode: model.supportsJsonMode,
   626→        supportsStreaming: model.supportsStreaming,
   627→        supportsTools: model.supportsTools,
   628→        type: modelType,
   629→        isActive: true,
   630→        lastSyncedAt: new Date(),
   631→      },
   632→    });
   633→    count++;
   634→  }
   635→
   636→  // Always ensure there's at least one default TEXT model for this provider
   637→  const textDefault = await db.aiModel.findFirst({
   638→    where: { providerId, type: 'TEXT', isActive: true },
   639→    orderBy: { createdAt: 'asc' },
   640→  });
   641→  if (textDefault) {
   642→    const hasTextDefault = await db.aiModel.findFirst({
   643→      where: { providerId, type: 'TEXT', isDefault: true },
   644→    });
   645→    if (!hasTextDefault) {
   646→      await db.aiModel.update({ where: { id: textDefault.id }, data: { isDefault: true } });
   647→    }
   648→  }
   649→
   650→  await db.aiProvider.update({
   651→    where: { id: providerId },
   652→    data: { lastSyncAt: new Date() },
   653→  });
   654→
   655→  return count;
   656→}

   670→export async function getUsageAnalytics(siteId?: string, period: 'day' | 'week' | 'month' = 'month') {
   671→  const now = new Date();
   672→  const periodStart = new Date();
   673→  if (period === 'day') periodStart.setHours(now.getHours() - 24);
   674→  else if (period === 'week') periodStart.setDate(now.getDate() - 7);
   675→  else periodStart.setMonth(now.getMonth() - 1);
   676→
   677→  const where: Prisma.AiLogWhereInput = {
   678→    createdAt: { gte: periodStart },
   679→    ...(siteId ? { siteId } : {}),
   680→  };
   681→
   682→  const [totalCount, totalTokens, totalCost, errorCount, avgDuration] = await Promise.all([
   683→    db.aiLog.count({ where }),
   684→    db.aiLog.aggregate({ where, _sum: { totalTokens: true, inputTokens: true, outputTokens: true } }),
   685→    db.aiLog.aggregate({ where, _sum: { costUsd: true } }),
   686→    db.aiLog.count({ where: { ...where, status: 'error' } }),
   687→    db.aiLog.aggregate({ where: { ...where, status: 'success' }, _avg: { durationMs: true } }),
   688→  ]);
   689→
   690→  // Provider breakdown
   691→  const providerBreakdown = await db.aiLog.groupBy({
   692→    by: ['providerName'],
   693→    where,
   694→    _sum: { totalTokens: true, costUsd: true, inputTokens: true, outputTokens: true },
   695→    _count: true,
   696→    orderBy: { _sum: { costUsd: 'desc' } },
   697→  });
   698→
   699→  // Model breakdown
   700→  const modelBreakdown = await db.aiLog.groupBy({
   701→    by: ['modelId'],
   702→    where,
   703→    _sum: { totalTokens: true, costUsd: true },
   704→    _count: true,
   705→    orderBy: { _count: { id: 'desc' } },
   706→    take: 10,
   707→  });
   708→
   709→  // Daily usage chart
   710→  const dailyUsage = await db.$queryRawUnsafe<Array<{ date: string; count: number; tokens: number; cost: number }>>(`
   711→    SELECT DATE(createdAt) as date, COUNT(*) as count, COALESCE(SUM(totalTokens), 0) as tokens, COALESCE(SUM(costUsd), 0) as cost
   712→    FROM AiLog
   713→    WHERE createdAt >= ?
   714→    ${siteId ? 'AND siteId = ?' : ''}
   715→    GROUP BY DATE(createdAt)
   716→    ORDER BY date ASC
   717→  `, periodStart.toISOString(), ...(siteId ? [siteId] : []));
   718→
   719→  return {
   720→    totalRequests: totalCount,
   721→    totalTokens: totalTokens._sum.totalTokens || 0,
   722→    inputTokens: totalTokens._sum.inputTokens || 0,
   723→    outputTokens: totalTokens._sum.outputTokens || 0,
   724→    totalCost: totalCost._sum.costUsd || 0,
   725→    errorCount,
   726→    successRate: totalCount > 0 ? ((totalCount - errorCount) / totalCount) * 100 : 100,
   727→    avgDurationMs: Math.round(avgDuration._avg.durationMs || 0),
   728→    providerBreakdown,
   729→    modelBreakdown,
   730→    dailyUsage,
   731→  };
   732→}

   767→// Approximate cost per image (USD) for known models, keyed by model then size
   768→const IMAGE_MODEL_COSTS: Record<string, Record<string, number>> = {
   769→  'dall-e-2': { '256x256': 0.016, '512x512': 0.016, '1024x1024': 0.02 },
   770→  'dall-e-3': { '1024x1024': 0.040, '1792x1024': 0.080, '1024x1792': 0.080 },
   771→  'gpt-image-1': { '1024x1024': 0.040, '1536x1024': 0.080, '1024x1536': 0.080 },
   772→};
   773→
   774→function getImageCost(modelId: string, size: string): number {
   775→  const modelCosts = IMAGE_MODEL_COSTS[modelId];
   776→  if (modelCosts) {
   777→    return modelCosts[size] ?? modelCosts['1024x1024'] ?? 0.04;
   778→  }
   779→  // Fallback: use default estimate
   780→  return 0.04;
   781→}

   783→async function callOpenAIImageGeneration(
   784→  baseUrl: string,
   785→  apiKey: string,
   786→  model: string,
   787→  prompt: string,
   788→  opts: {
   789→    negativePrompt?: string;
   790→    size?: string;
   791→    quality?: string;
   792→    style?: string;
   793→    n?: number;
   794→    responseFormat?: string;
   795→    apiVersion?: string;
   796→  } = {},
   797→): Promise<{ images: GeneratedImage[] }> {
   798→  const url = opts.apiVersion
   799→    ? `${baseUrl}/images/generations?api-version=${opts.apiVersion}`
   800→    : `${baseUrl}/images/generations`;
   801→
   802→  const body: Record<string, unknown> = {
   803→    model,
   804→    prompt,
   805→    n: opts.n ?? 1,
   806→    size: opts.size ?? '1024x1024',
   807→    response_format: opts.responseFormat ?? 'url',
   808→  };
   809→
   810→  if (opts.quality) body.quality = opts.quality;
   811→  if (opts.style) body.style = opts.style;
   812→
   813→  const res = await fetch(url, {
   814→    method: 'POST',
   815→    headers: {
   816→      'Content-Type': 'application/json',
   817→      'Authorization': `Bearer ${apiKey}`,
   818→    },
   819→    body: JSON.stringify(body),
   820→  });
   821→
   822→  if (!res.ok) {
   823→    const err = await res.text();
   824→    throw new Error(`Image generation API error: ${res.status} — ${err}`);
   825→  }
   826→
   827→  const data = await res.json();
   828→  const images: GeneratedImage[] = (data.data ?? []).map((item: Record<string, unknown>) => ({
   829→    url: (item.url as string) ?? null,
   830→    base64: (item.b64_json as string) ?? null,
   831→    revisedPrompt: (item.revised_prompt as string) ?? null,
   832→  }));
   833→
   834→  return { images };
   835→}

   837→async function callGeminiImageGeneration(
   838→  baseUrl: string,
   839→  apiKey: string,
   840→  model: string,
   841→  prompt: string,
   842→  opts: {
   843→    negativePrompt?: string;
   844→    size?: string;
   845→    n?: number;
   846→    responseFormat?: string;
   847→  } = {},
   848→): Promise<{ images: GeneratedImage[] }> {
   849→  // Parse size to aspect ratio for Gemini Imagen
   850→  const sizeParts = (opts.size ?? '1024x1024').split('x');
   851→  const width = parseInt(sizeParts[0] ?? '1024', 10);
   852→  const height = parseInt(sizeParts[1] ?? '1024', 10);
   853→  let ratio: string;
   854→  if (Math.abs(width - height) < 100) {
   855→    ratio = '1:1'; // square
   856→  } else {
   857→    ratio = width >= height ? '16:9' : '9:16';
   858→  }
   859→
   860→  // Gemini uses Imagen via the generateContent endpoint with image generation config
   861→  // or the dedicated Imagen predict endpoint
   862→  const res = await fetch(
   863→    `${baseUrl}/models/${model}:generateContent?key=${apiKey}`,
   864→    {
   865→      method: 'POST',
   866→      headers: { 'Content-Type': 'application/json' },
   867→      body: JSON.stringify({
   868→        contents: [{ parts: [{ text: prompt }] }],
   869→        generationConfig: {
   870→          responseModalities: ['TEXT', 'IMAGE'],
   871→          // @ts-expect-error -- Gemini-specific field
   872→          imageGenerationConfig: {
   873→            numberOfImages: opts.n ?? 1,
   874→            aspectRatio: ratio,
   875→          },
   876→        },
   877→      }),
   878→    },
   879→  );
   880→
   881→  if (!res.ok) {
   882→    const err = await res.text();
   883→    throw new Error(`Gemini image generation error: ${res.status} — ${err}`);
   884→  }
   885→
   886→  const data = await res.json();
   887→  const images: GeneratedImage[] = [];
   888→
   889→  const parts = data.candidates?.[0]?.content?.parts ?? [];
   890→  for (const part of parts) {
   891→    if (part.inlineData) {
   892→      images.push({
   893→        url: null,
   894→        base64: part.inlineData.data ?? null,
   895→        revisedPrompt: prompt,
   896→      });
   897→    }
   898→  }
   899→
   900→  // Fallback: if no image parts returned, throw
   901→  if (images.length === 0) {
   902→    throw new Error('No images were generated by Gemini');
   903→  }
   904→
   905→  return { images };
   906→}

   908→export async function executeImageGeneration(req: ImageGenerationRequest): Promise<ImageGenerationResponse> {
   909→  const provider = await db.aiProvider.findUnique({
   910→    where: { id: req.providerId },
   911→    include: { models: true },
   912→  });
   913→
   914→  if (!provider) throw new Error('Provider not found');
   915→  if (!provider.isActive) throw new Error('Provider is disabled. Please activate it first.');
   916→  if (!provider.apiKeyEncrypted) throw new Error('API key not configured for this provider.');
   917→
   918→  // Resolve + validate the model (type must be IMAGE)
   919→  const resolved = await resolveModel(req.providerId, req.modelId, 'IMAGE', provider.models);
   920→  const modelId = resolved.modelId;
   921→
   922→  const apiKey = await decrypt(provider.apiKeyEncrypted);
   923→  const config = getProviderConfig(provider.kind);
   924→  const baseUrl = provider.baseUrl || config.defaultBaseUrl;
   925→  // CUSTOM providers have no defaultBaseUrl — they must have one set explicitly.
   926→  if (!baseUrl) {
   927→    throw new Error('No Base URL configured for this custom provider. Please edit the provider and set a Base URL.');
   928→  }
   929→
   930→  const startTime = Date.now();
   931→  let images: GeneratedImage[] = [];
   932→  let usedProvider = provider;
   933→  let usedModelId = modelId;
   934→
   935→  try {
   936→    if (provider.kind === 'GEMINI') {
   937→      const result = await callGeminiImageGeneration(baseUrl, apiKey, modelId, req.prompt, {
   938→        negativePrompt: req.negativePrompt,
   939→        size: req.size,
   940→        n: req.n,
   941→        responseFormat: req.responseFormat,
   942→      });
   943→      images = result.images;
   944→    } else if (provider.kind === 'OPENAI' || provider.kind === 'CUSTOM') {
   945→      // OpenAI and Custom (OpenAI-compatible) providers support image generation
   946→      const result = await callOpenAIImageGeneration(baseUrl, apiKey, modelId, req.prompt, {
   947→        negativePrompt: req.negativePrompt,
   948→        size: req.size,
   949→        quality: req.quality,
   950→        style: req.style,
   951→        n: req.n,
   952→        responseFormat: req.responseFormat,
   953→      });
   954→      images = result.images;
   955→    } else {
   956→      // GROQ and DEEPSEEK do not support image generation
   957→      throw new Error(`${config.name} does not support image generation. Please use OpenAI, Gemini, or a Custom OpenAI-compatible provider.`);
   958→    }
   959→  } catch (err) {
   960→    // Try fallback providers for image generation
   961→    const fallbacks = await db.aiProviderFallback.findMany({
   962→      where: { providerId: provider.id },
   963→      include: { fallback: { include: { models: true } } },
   964→      orderBy: { priority: 'asc' },
   965→    });
   966→
   967→    let lastError = err instanceof Error ? err : new Error('Unknown error');
   968→
   969→    for (const fb of fallbacks) {
   970→      if (!fb.fallback.isActive || !fb.fallback.apiKeyEncrypted) continue;
   971→      // Only try fallbacks that support image generation (OpenAI, Gemini, or Custom OpenAI-compatible)
   972→      if (fb.fallback.kind !== 'OPENAI' && fb.fallback.kind !== 'GEMINI' && fb.fallback.kind !== 'CUSTOM') continue;
   973→      try {
   974→        const fbResolved = await resolveModel(fb.fallback.id, undefined, 'IMAGE', fb.fallback.models);
   975→        const fbApiKey = await decrypt(fb.fallback.apiKeyEncrypted);
   976→        const fbConfig = getProviderConfig(fb.fallback.kind);
   977→        const fbBaseUrl = fb.fallback.baseUrl || fbConfig.defaultBaseUrl;
   978→        // CUSTOM providers have no defaultBaseUrl — skip if not set.
   979→        if (!fbBaseUrl) continue;
   980→
   981→        let fbResult: { images: GeneratedImage[] };
   982→        if (fb.fallback.kind === 'GEMINI') {
   983→          fbResult = await callGeminiImageGeneration(fbBaseUrl, fbApiKey, fbResolved.modelId, req.prompt, {
   984→            negativePrompt: req.negativePrompt, size: req.size, n: req.n, responseFormat: req.responseFormat,
   985→          });
   986→        } else {
   987→          // OpenAI or Custom (OpenAI-compatible)
   988→          fbResult = await callOpenAIImageGeneration(fbBaseUrl, fbApiKey, fbResolved.modelId, req.prompt, {
   989→            size: req.size, quality: req.quality, style: req.style, n: req.n, responseFormat: req.responseFormat,
   990→          });
   991→        }
   992→        images = fbResult.images;
   993→        usedProvider = fb.fallback;
   994→        usedModelId = fbResolved.modelId;
   995→        lastError = null as unknown as Error;
   996→        break;
   997→      } catch (fbErr) {
   998→        lastError = fbErr instanceof Error ? fbErr : new Error('Fallback failed');
   999→        continue;
  1000→      }
  1001→    }
  1002→
  1003→    if (images.length === 0) {
  1004→      // Log the failed request
  1005→      const durationMs = Date.now() - startTime;
  1006→      await db.aiLog.create({
  1007→        data: {
  1008→          providerId: provider.id,
  1009→          providerName: provider.name,
  1010→          modelId,
  1011→          question: `[IMAGE] ${req.prompt}`,
  1012→          response: null,
  1013→          inputTokens: 0,
  1014→          outputTokens: 0,
  1015→          totalTokens: 0,
  1016→          costUsd: 0,
  1017→          durationMs,
  1018→          status: 'error',
  1019→          errorMessage: lastError?.message ?? 'Unknown error',
  1020→          siteId: req.siteId,
  1021→          userId: req.userId,
  1022→        },
  1023→      }).catch(() => { /* logging failure shouldn't mask the original error */ });
  1024→      throw lastError ?? new Error('Image generation failed');
  1025→    }
  1026→  }
  1027→
  1028→  const durationMs = Date.now() - startTime;
  1029→  const size = req.size ?? '1024x1024';
  1030→  const n = images.length;
  1031→  const costUsd = n * getImageCost(usedModelId, size);
  1032→
  1033→  // Update provider stats
  1034→  await db.aiProvider.update({
  1035→    where: { id: usedProvider.id },
  1036→    data: {
  1037→      lastUsedAt: new Date(),
  1038→      latencyMs: durationMs,
  1039→      connectionStatus: 'CONNECTED',
  1040→      lastError: null,
  1041→    },
  1042→  });
  1043→
  1044→  // Log the request
  1045→  await db.aiLog.create({
  1046→    data: {
  1047→      providerId: usedProvider.id,
  1048→      providerName: usedProvider.name,
  1049→      modelId: usedModelId,
  1050→      question: `[IMAGE] ${req.prompt}`,
  1051→      response: JSON.stringify({
  1052→        imagesGenerated: n,
  1053→        size,
  1054→        format: req.responseFormat ?? 'url',
  1055→        model: usedModelId,
  1056→      }),
  1057→      inputTokens: 0,
  1058→      outputTokens: 0,
  1059→      totalTokens: 0,
  1060→      costUsd,
  1061→      durationMs,
  1062→      status: 'success',
  1063→      siteId: req.siteId,
  1064→      userId: req.userId,
  1065→    },
  1066→  });
  1067→
  1068→  return {
  1069→    images,
  1070→    model: usedModelId,
  1071→    costUsd,
  1072→    durationMs,
  1073→    providerName: usedProvider.name,
  1074→    providerKind: usedProvider.kind,
  1075→  };
  1076→}
