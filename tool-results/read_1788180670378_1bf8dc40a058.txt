    49→// -------------------- Multi-currency + derived feature helpers --------------------
    50→
    51→const GB_FACTOR = 1024 * 1024 * 1024;
    52→
    53→/** Server-side currency resolution context (from /api/platform/billing/me
    54→ *  → customerCurrencyResolution). The customer never selects a currency —
    55→ *  the server detects it from the request IP. */
    56→interface CustomerCurrencyResolution {
    57→  currency: string;
    58→  countryCode: string;
    59→  countryName: string;
    60→  source: 'ip' | 'default' | 'local';
    61→}
    62→
    63→/** Server-resolved FINAL price for one plan (from /api/platform/billing/me
    64→ *  → planPricing[planId]). Computed by the same resolveCustomerPricing
    65→ *  the checkout route uses — the price/currency displayed here is exactly
    66→ *  what Stripe charges. */
    67→interface ServerPlanPricing {
    68→  planId: string;
    69→  currency: string;
    70→  monthly: number;
    71→  yearly: number;
    72→  source: 'ip' | 'default' | 'local' | 'plan';
    73→  supported: boolean;
    74→  detectedCurrency: string;
    75→  countryCode: string;
    76→  countryName: string;
    77→  regional: boolean;
    78→}
    79→
    80→type BillingStateWithCurrency = ClientBillingState & {
    81→  customerCurrencyResolution?: CustomerCurrencyResolution;
    82→  customerCurrency?: string;
    83→  customerCountryCode?: string;
    84→  customerCountryName?: string;
    85→  currencySource?: 'ip' | 'default' | 'local';
    86→  planPricing?: Record<string, ServerPlanPricing>;
    87→};
    88→
    89→/** Resolve the price + currency to display for a plan. Prefers the
    90→ *  SERVER-RESOLVED pricing (planPricing — the same values checkout
    91→ *  charges); falls back to a local pricesByCurrency lookup for older
    92→ *  API responses. */
    93→function resolvePlanPricing(
    94→  plan: {
    95→    id: string;
    96→    pricesByCurrency?: Record<string, { monthly: number; yearly: number }>;
    97→    priceMonthly: number;
    98→    priceYearly: number;
    99→    currency: string;
   100→    interval: 'monthly' | 'yearly';
   101→  },
   102→  serverPricing: ServerPlanPricing | undefined,
   103→  customerCurrency: string,
   104→): { monthly: number; yearly: number; currency: string } {
   105→  if (serverPricing) {
   106→    return {
   107→      monthly: serverPricing.monthly,
   108→      yearly: serverPricing.yearly,
   109→      currency: serverPricing.currency,
   110→    };
   111→  }
   112→  // Legacy fallback — local lookup (display only).
   113→  const entry = plan.pricesByCurrency?.[customerCurrency];
   114→  if (entry) return { monthly: entry.monthly, yearly: entry.yearly, currency: customerCurrency };
   115→  return { monthly: plan.priceMonthly, yearly: plan.priceYearly, currency: plan.currency };
   116→}
   117→
   118→/** Derive the marketing feature list from the plan's entitlements +
   119→ *  a few standard items (max sites, storage, support tier). Uses the
   120→ *  legacy plan.features array when it has entries (legacy compat for
   121→ *  plans that still have manually-maintained marketing copy, e.g. the
   122→ *  synthetic Internal plan). The structured entitlements are the
   123→ *  single source of truth — the admin no longer maintains a separate
   124→ *  marketing list per plan. */
   125→function derivePlanFeatures(plan: {
   126→  features: string[];
   127→  entitlements: string[];
   128→  isFree: boolean;
   129→  limits: { maxSites: number; storageBytes: number };
   130→}): string[] {
   131→  if (plan.features.length > 0) return plan.features;
   132→  const items: string[] = [];
   133→  items.push(
   134→    plan.limits.maxSites === -1
   135→      ? 'Unlimited sites'
   136→      : `Up to ${plan.limits.maxSites} sites`,
   137→  );
   138→  const gb = plan.limits.storageBytes / GB_FACTOR;
   139→  if (gb >= 1) {
   140→    items.push(`${Math.floor(gb)} GB storage`);
   141→  } else if (plan.limits.storageBytes > 0) {
   142→    items.push(`${plan.limits.storageBytes} bytes storage`);
   143→  } else {
   144→    items.push('No storage');
   145→  }
   146→  items.push(plan.isFree ? 'Community support' : 'Priority support');
   147→  for (const e of plan.entitlements) {
   148→    const label = ENTITLEMENT_LABELS[e as EntitlementKey];
   149→    if (label) items.push(label);
   150→  }
   151→  return items;
   152→}

   527→                  <p className="text-[11px] text-muted-foreground ml-7">
   528→                    Priced for {customerCountryName} ({currentDisplayCurrency})
   529→                  </p>
   530→                )}
   531→            </div>
   532→            <Badge
   533→              variant={status === 'active' ? 'default' : 'outline'}
   534→              className="capitalize"
   535→            >
   536→              {status}
   537→            </Badge>
   538→          </div>
   539→
   540→          {/* Current period end / next billing date (DB Subscription source of truth) */}
   541→          {currentPeriodEnd && !isCancelled && (
   542→            <div className="flex items-center gap-2 text-sm text-muted-foreground bg-muted/50 rounded-md p-3">
   543→              <Calendar className="h-4 w-4 shrink-0" />

   549→
   550→          {trialEnd && (

   559→          <Separator />
   560→
   561→          {isCancelled ? (
   562→            <div className="flex items-center justify-end">
   563→              <p className="text-xs text-muted-foreground italic">
   564→                Your subscription is cancelled
   565→              </p>

   569→              <div className="flex-1" />
   570→              <div className="flex gap-2">
   571→                <Button variant="outline" size="sm" disabled>
   572→                  {t('billing.managePayment')}

   574→                <Button
   575→                  variant="outline"
   576→                  size="sm"
   577→                  className="text-destructive hover:text-destructive"
   578→                  disabled={cancelMutation.isPending}
   579→                  onClick={handleCancel}
   580→                >
   581→                  {cancelMutation.isPending ? (
   582→                    <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   583→                  ) : null}
   584→                  {t('billing.cancelSubscription')}
   585→                </Button>
   586→              </div>
   587→            </div>
   588→          )}
   589→        </CardContent>
   590→      </Card>
   591→
   592→      {/* Other Plans */}
   593→      <div>
   594→        <h2 className="text-base font-semibold mb-4">{t('billing.otherPlans')}</h2>
   595→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   596→          {otherPlans.map((plan) => {

   598→            // Server-resolved FINAL price — the same values checkout charges.
   599→            const planPricingResolved = resolvePlanPricing(plan, planPricing?.[plan.id], customerCurrency);
   600→            const planMonthly = planPricingResolved.monthly;
   601→            const planDisplayCurrency = planPricingResolved.currency;
   602→            const planSupported = planPricing?.[plan.id]?.supported ?? true;
   603→            const isBusy =
   604→              (changePlanMutation.isPending && changePlanMutation.variables?.planId === plan.id) ||
   605→              (checkoutMutation.isPending && checkoutMutation.variables?.planId === plan.id);

   608→                key={plan.id}
   609→                className={`relative ${getPlanCardBorderClasses(storePlan.badgeVariant)}`}

   612→                  <div className="flex items-center justify-between">

   614→                    <Badge
   615→                      variant="outline"
   616→                      className={`text-[10px] font-semibold ${getPlanBadgeClasses(storePlan.badgeVariant)}`}
   617→                    >
   618→                      {plan.name}
   619→                    </Badge>
   620→                  </div>
   621→                </CardHeader>
   622→                <CardContent className="space-y-4">
   623→                  <div>
   624→                    <span className="text-2xl font-bold">
   625→                      {planMonthly === 0 ? t('billing.free') : formatMoney(planMonthly, planDisplayCurrency)}

   628→                      <span className="text-sm text-muted-foreground ml-1">
   629→                        /{normalizeInterval(plan.interval)}
   630→                      </span>
   631→                    )}

   633→                      <p className="text-[11px] text-muted-foreground mt-1">

   635→                      </p>

   638→                      <p className="text-[11px] text-muted-foreground mt-1">
   639→                        Priced in {planDisplayCurrency} (plan default) — {planPricing[plan.id].detectedCurrency} not

   641→                      </p>
   642→                    )}
   643→                  </div>
   644→                  <Separator />
   645→                  <ul className="space-y-2">

   649→                        <span>{feature}</span>
   650→                      </li>
   651→                    ))}
   652→                  </ul>
   653→                  <Button
   654→                    className="w-full"
   655→                    variant={isHigherPlan(plan) ? 'default' : 'outline'}

   658→                  >
   659→                    {isBusy ? (
   660→                      <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   661→                    ) : null}
   662→                    {getActionLabel(plan)}
   663→                  </Button>
   664→                </CardContent>
   665→              </Card>
   666→            );
   667→          })}
   668→        </div>
   669→      </div>
   670→
   671→      {/* Payment History */}
   672→      <Card>
   673→        <CardHeader>
   674→          <CardTitle className="text-base">{t('billing.paymentHistory')}</CardTitle>
   675→        </CardHeader>
   676→        <CardContent>
   677→          {billingState.paymentHistory.length === 0 ? (
   678→            <div className="flex flex-col items-center justify-center py-12 text-center">
   679→              <div className="h-12 w-12 rounded-full bg-muted flex items-center justify-center mb-4">
   680→                <Receipt className="h-6 w-6 text-muted-foreground" />
   681→              </div>
   682→              <h3 className="font-medium text-sm">{t('billing.noPayments')}</h3>
   683→            </div>
   684→          ) : (
   685→            <div className="overflow-x-auto">
   686→              <table className="w-full text-sm">
   687→                <thead>
   688→                  <tr className="border-b text-left">
   689→                    <th className="pb-2 font-medium text-xs text-muted-foreground">{t('billing.invoice')}</th>
   690→                    <th className="pb-2 font-medium text-xs text-muted-foreground">Plan</th>
   691→                    <th className="pb-2 font-medium text-xs text-muted-foreground text-right">{t('billing.amount')}</th>
   692→                    <th className="pb-2 font-medium text-xs text-muted-foreground">{t('billing.status')}</th>
   693→                    <th className="pb-2 font-medium text-xs text-muted-foreground">Method</th>
   694→                    <th className="pb-2 font-medium text-xs text-muted-foreground text-right">{t('billing.date')}</th>
   695→                  </tr>
   696→                </thead>
   697→                <tbody className="divide-y">
   698→                  {billingState.paymentHistory.map((p: Payment) => {
   699→                    const planForPayment = billingState.allPlans.find((pl) => pl.id === p.planId);
   700→                    return (
   701→                      <tr key={p.id} className="hover:bg-accent/30 transition-colors">
   702→                        <td className="py-2.5 pr-4">
   703→                          <span className="font-mono text-xs">{p.invoiceNumber}</span>
   704→                        </td>
   705→                        <td className="py-2.5 pr-4">
   706→                          <Badge
   707→                            variant="outline"
   708→                            className={`text-[10px] font-semibold ${getPlanBadgeClasses(getStorePlan(p.planId).badgeVariant)}`}
   709→                          >
   710→                            {planForPayment?.name ?? p.planId}
   711→                          </Badge>
   712→                        </td>
   713→                        <td className="py-2.5 pr-4 text-right font-medium">
   714→                          {formatCurrency(p.amount, p.currency)}
   715→                        </td>
   716→                        <td className="py-2.5 pr-4">
   717→                          <PaymentStatusBadge status={p.status} />
   718→                        </td>
   719→                        <td className="py-2.5 pr-4 text-xs text-muted-foreground">{p.method}</td>
   720→                        <td className="py-2.5 text-right text-xs text-muted-foreground">
   721→                          {formatDate(p.date)}
   722→                        </td>
   723→                      </tr>
   724→                    );
   725→                  })}
   726→                </tbody>
   727→              </table>
   728→            </div>

   730→        </CardContent>
