     4→     4→// PLATFORM PLANS & PRICING — simplified admin UI.
     5→     5→// ============================================================
     6→     6→// Owner sees compact plan cards (Beta / Pro / Max …) with a quick
     7→     7→// Active toggle, a compact price + features + limits summary,
     8→     8→// and an "Edit Plan" button. The Edit Plan button opens a Dialog
     9→     9→// with three compact sections: Basic Information, Feature Access,
    10→    10→// Usage Limits. An optional collapsed "Client Display" section
    11→    11→// holds the marketing feature list shown to clients on the
    12→    12→// Client Billing page. A "+ Create Plan" button at the top-right
    13→    13→// opens a Create Plan dialog (same sections, blank defaults) that
    14→    14→// POSTs to /api/platform/admin/plans.
    15→    15→//
    16→    16→// All writes go through /api/platform/admin/plans (GET list,
    17→    17→// POST create) + /api/platform/admin/plans/[planId] (PUT update)
    18→    18→// — owner-only — the same shared plan-config cache the Client
    19→    19→// Billing page and MRR read. Server-side entitlement enforcement
    20→    20→// (hasFeature) and usage-limit checks (checkLimit) consume the
    21→    21→// same data, so toggling a feature off here denies it for clients
    22→    22→// on the next request. The Client Billing page is NOT modified
    23→    23→// and continues to read from /api/platform/billing/me.

    91→    91→/** Resolve the badge id for the shared PlanBadge component. Falls
    92→    92→ *  back to 'beta' for any unknown plan id so future plans still
    93→    93→ *  render with a sensible neutral badge. */
    94→    94→function getPlanBadgeId(planId: string): 'beta' | 'pro' | 'max' {
    95→    95→  if (planId === 'beta' || planId === 'pro' || planId === 'max') return planId;
    96→    96→  return 'beta';
    97→    97→}
    98→    98→
    99→    99→/** Format a single limit value for the compact card summary. -1 →
   100→   100→ *  "Unlimited"; storageBytes → human-readable (1 GB); other limits
   101→   101→ *  use locale-formatted numbers. */
   102→   102→function formatLimitValue(key: LimitKey, value: number): string {
   103→   103→  if (value === UNLIMITED) return 'Unlimited';
   104→   104→  if (key === 'storageBytes') return formatBytes(value);
   105→   105→  return value.toLocaleString();
   106→   106→}
   107→   107→
   108→   108→/** One-line compact limits summary used on the plan card. */
   109→   109→function summarizeLimits(limits: PlanLimits): string {
   110→   110→  return [
   111→   111→    `${formatLimitValue('maxSites', limits.maxSites)} sites`,
   112→   112→    formatLimitValue('storageBytes', limits.storageBytes),
   113→   113→    `${formatLimitValue('aiWords', limits.aiWords)} AI words`,
   114→   114→    `${formatLimitValue('aiArticles', limits.aiArticles)} AI articles`,
   115→   115→    `${formatLimitValue('automationRuns', limits.automationRuns)} automation runs`,
   116→   116→  ].join(' · ');
   117→   117→}
   118→   118→
   119→   119→/** One-line compact entitlement summary used on the plan card. */
   120→   120→function summarizeEntitlements(entitlements: string[]): string {
   121→   121→  if (entitlements.length === 0) return 'No advanced features';
   122→   122→  return entitlements
   123→   123→    .map((key) => ENTITLEMENT_LABELS[key as EntitlementKey] ?? key)
   124→   124→    .join(' · ');
   125→   125→}

   166→   166→      const e = err as { error?: { message?: string }; message?: string };

   168→   168→        e?.error?.message ?? e?.message ?? 'Unable to change plan status. OWNER access required.',
   169→   169→      );
   170→   170→    },

   172→   172→
   173→   173→  return (

   189→   189→              </Label>
   190→   190→              <Switch
   191→   191→                id={`active-${plan.planId}`}
   192→   192→                checked={plan.active}

   197→   197→          </div>

   208→   208→                <span className="text-2xl font-bold tracking-tight">
   209→   209→                  {formatCurrency(plan.priceMonthly, plan.currency)}
   210→   210→                </span>
   211→   211→                <span className="text-xs text-muted-foreground">/ month</span>
   212→   212→              </div>
   213→   213→            )}
   214→   214→            {!plan.isFree && plan.priceYearly > 0 && (
   215→   215→              <p className="text-xs text-muted-foreground">
   216→   216→                {formatCurrency(plan.priceYearly, plan.currency)} / year

   219→   219→          </div>

   221→   221→          {/* Billing interval */}

   225→   225→          </div>
   226→   226→
   227→   227→          {/* Features */}
   228→   228→          <div className="space-y-1">
   229→   229→            <p className="text-[10px] font-medium text-muted-foreground uppercase tracking-wider">
   230→   230→              Features
