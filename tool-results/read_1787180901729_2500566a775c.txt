   103→  excerpt: string;

   114→    year: 'numeric', month: 'long', day: 'numeric',

   117→  return (

   146→        <h1 className="text-3xl font-bold tracking-tight sm:text-4xl">{title || 'Untitled Article'}</h1>

   181→        <div
   182→          className="prose prose-gray dark:prose-invert max-w-none prose-headings:font-bold prose-h2:text-2xl prose-h3:text-xl prose-p:leading-relaxed prose-a:text-primary"
   183→          dangerouslySetInnerHTML={{ __html: content || '<p class="text-muted-foreground">No content yet.</p>' }}

   297→function ScheduleDialog({
   298→  open,
   299→  onOpenChange,
   300→  onSchedule,
   301→  isPending,
   302→}: {
   303→  open: boolean;

   418→export function ContentCreatePage() {
   419→  const navigate = useNavigationStore((s) => s.navigate);
   420→  const queryClient = useQueryClient();
   421→  const fileInputRef = useRef<HTMLInputElement>(null);
   422→  const editorRef = useRef<TiptapEditorRef>(null);
   423→
   424→  const [tagSearch, setTagSearch] = useState('');
   425→  const [slugValue, setSlugValue] = useState('');
   426→  const [aiInput, setAiInput] = useState('');
   427→  const [editorContent, setEditorContent] = useState('');
   428→  const [featuredImage, setFeaturedImage] = useState<MediaItem | null>(null);
   429→  const [customTags, setCustomTags] = useState<string[]>([]); // For newly created tags
   430→  const [selectedText, setSelectedText] = useState(''); // Tracks the currently selected text in the editor
   431→
   432→  // Dialog states
   433→  const [previewOpen, setPreviewOpen] = useState(false);
   434→  const [scheduleOpen, setScheduleOpen] = useState(false);
   435→  const [mediaLibraryOpen, setMediaLibraryOpen] = useState(false);
   436→  const [aiAssistOpen, setAiAssistOpen] = useState(false);
   437→
   438→  // Fetch dropdown data
   439→  const { data: contentTypes } = useQuery({
   440→    queryKey: queryKeys.contentTypes.all,
   441→    queryFn: () => getApi<ContentTypeOption[]>('/api/content-types?pageSize=100'),
   442→    staleTime: 60_000,
   443→  });
   444→  const { data: categories } = useQuery({
   445→    queryKey: queryKeys.categories.all,
   446→    queryFn: () => getApi<CategoryOption[]>('/api/categories?pageSize=200'),
   447→    staleTime: 60_000,
   448→  });
   449→  const { data: allTags, refetch: refetchTags } = useQuery({
   450→    queryKey: queryKeys.tags.all,
   451→    queryFn: () => getApi<TagOption[]>('/api/tags?pageSize=200'),
   452→    staleTime: 60_000,
   453→  });
   454→
   455→  const filteredTags = useMemo(() => {
   456→    if (!allTags) return [];
   457→    if (!tagSearch) return allTags;
   458→    return allTags.filter((t) => t.name.toLowerCase().includes(tagSearch.toLowerCase()));
   459→  }, [allTags, tagSearch]);
   460→
   461→  const {
   462→    register, handleSubmit, control, watch, setValue, getValues, formState: { errors },
   463→  } = useForm<ContentFormValues>({
   464→    resolver: zodResolver(contentFormSchema),
   465→    defaultValues: { title: '', excerpt: '', content: '', status: 'DRAFT', contentTypeId: '', categoryId: '', tagIds: [], seoTitle: '', seoDescription: '' },
   466→  });
   467→
   468→  const selectedTagIds = watch('tagIds');
   469→  const watchedTitle = watch('title');
   470→  const watchedExcerpt = watch('excerpt');
   471→  const watchedSeoTitle = watch('seoTitle');
   472→  const watchedSeoDescription = watch('seoDescription');
   473→
   474→  // Auto-slug
   475→  React.useEffect(() => {
   476→    setSlugValue(slugify(watchedTitle));
   477→  }, [watchedTitle]);
   478→
   479→  // Create mutation
   480→  const createMutation = useMutation({
   481→    mutationFn: (data: ContentFormValues & { scheduledAt?: string }) => {
   482→      const slug = slugify(data.title);
   483→      return postApi<CreatedContent>('/api/content', {
   484→        title: data.title,
   485→        slug,
   486→        excerpt: data.excerpt || undefined,
   487→        content: editorContent || undefined,
   488→        status: data.status,
   489→        contentTypeId: data.contentTypeId,
   490→        categoryId: data.categoryId || undefined,
   491→        tagIds: data.tagIds,
   492→        seoTitle: data.seoTitle || undefined,
   493→        seoDescription: data.seoDescription || undefined,
   494→        featuredImageId: featuredImage?.id || undefined,
   495→        scheduledAt: data.scheduledAt || undefined,
   496→      } as Record<string, unknown>);
   497→    },
   498→    onSuccess: (created) => {
   499→      queryClient.invalidateQueries({ queryKey: queryKeys.content.all });
   500→      toast.success('Content created successfully');
   501→      navigate('content', created.id);
   502→    },
   503→    onError: (err: Error) => toast.error(err.message || 'Failed to create content'),
   504→  });
   505→
   506→  // Create tag mutation
   507→  const createTagMutation = useMutation({
   508→    mutationFn: (name: string) => postApi<TagOption>('/api/tags', { name }),
   509→    onSuccess: (newTag) => {
   510→      refetchTags();
   511→      if (!selectedTagIds.includes(newTag.id)) {
   512→        setValue('tagIds', [...selectedTagIds, newTag.id], { shouldValidate: true });
   513→      }
   514→      setCustomTags((prev) => [...prev, newTag.name]);
   515→      toast.success(`Tag "${newTag.name}" created`);
   516→    },
   517→    onError: (err: Error) => toast.error(err.message || 'Failed to create tag'),
   518→  });
   519→
   520→  // Upload featured image mutation
   521→  const uploadMutation = useMutation({
   522→    mutationFn: async (file: File) => {
   523→      const formData = new FormData();
   524→      formData.append('file', file);
   525→      const res = await fetch('/api/media/upload', {
   526→        method: 'POST',
   527→        body: formData,
   528→      });
   529→      if (!res.ok) {
   530→        const err = await res.json().catch(() => ({}));
   531→        throw new Error(err?.error?.message || 'Upload failed');
   532→      }
   533→      const json = await res.json();
   534→      return json.data as MediaItem;
   535→    },
   536→    onSuccess: (media) => {
   537→      setFeaturedImage(media);
   538→      toast.success('Image uploaded');
   539→    },
   540→    onError: (err: Error) => toast.error(err.message || 'Failed to upload image'),
   541→  });
   542→
   543→  // AI content generation mutation (full article — used when no text is selected)
   544→  const aiGenerateMutation = useMutation({
   545→    mutationFn: (prompt: string) =>
   546→      postApi('/api/content/ai-generate', {
   547→        title: watchedTitle || 'Untitled',
   548→        brief: prompt,
   549→        writingStyle: 'Professional',
   550→        targetLength: 'Medium (800-1200 words)',
   551→        numberOfDrafts: 1,
   552→      }),
   553→    onSuccess: (result) => {
   554→      const draft = result.data?.drafts?.[0];
