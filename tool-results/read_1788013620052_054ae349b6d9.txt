     4→// PLATFORM PLANS & PRICING — simplified admin UI.
     5→// ============================================================
     6→// Owner sees compact plan cards (Beta / Pro / Max …) with a quick
     7→// Active toggle, a compact price + features + limits summary,
     8→// and an "Edit Plan" button. The Edit Plan button opens a Dialog
     9→// with three compact sections: Basic Information, Feature Access,
    10→// Usage Limits. An optional collapsed "Client Display" section
    11→// holds the marketing feature list shown to clients on the
    12→// Client Billing page. A "+ Create Plan" button at the top-right
    13→// opens a Create Plan dialog (same sections, blank defaults) that
    14→// POSTs to /api/platform/admin/plans.
    15→//
    16→// All writes go through /api/platform/admin/plans (GET list,
    17→// POST create) + /api/platform/admin/plans/[planId] (PUT update)
    18→// — owner-only — the same shared plan-config cache the Client
    19→// Billing page and MRR read. Server-side entitlement enforcement
    20→// (hasFeature) and usage-limit checks (checkLimit) consume the
    21→// same data, so toggling a feature off here denies it for clients
    22→// on the next request. The Client Billing page is NOT modified
    23→// and continues to read from /api/platform/billing/me.
    24→// ============================================================

    88→/** Resolve the badge id for the shared PlanBadge component. Falls
    89→ *  back to 'free' for any unknown plan id so future plans still
    90→ *  render with a sensible neutral badge. */
    91→function getPlanBadgeId(planId: string): 'free' | 'plus' | 'pro' | 'max' {
    92→  if (planId === 'free' || planId === 'plus' || planId === 'pro' || planId === 'max') return planId;
    93→  // Legacy 'beta' plan id (pre-migration) → render as 'plus' badge.
    94→  if (planId === 'beta') return 'plus';
    95→  return 'free';
    96→}

   111→function PlanSummaryCard({
   112→  plan,
   113→  canEdit,
   114→  billingInterval,
   115→}: {
   116→  plan: PlanConfigData;
   117→  canEdit: boolean;
   118→  billingInterval: 'monthly' | 'yearly';
   119→}) {
   120→  const queryClient = useQueryClient();
   121→  const [editing, setEditing] = useState(false);
   122→
   123→  // Quick active toggle — auto-saves immediately (owner-only).
   124→  // Reuses the same PUT endpoint and the same invalidation chain
   125→  // as the full Edit modal so clients see the new status on the
   126→  // next read.
   127→  const activeMutation = useMutation({
   128→    mutationFn: (active: boolean) =>
   129→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, { active }),
   130→    onSuccess: (data) => {
   131→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   132→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   133→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   134→      toast.success(
   135→        `${data.name} ${data.active ? 'activated' : 'deactivated'} — changes are live for clients.`,
   136→      );
   137→    },
   138→    onError: (err: unknown) => {
   139→      const e = err as { error?: { message?: string }; message?: string };
   140→      toast.error(
   141→        e?.error?.message ?? e?.message ?? 'Unable to change plan status. OWNER access required.',
   142→      );
   143→    },
   144→  });
   145→
   146→  // Feature list shown on the card. Prefer the client-facing marketing
   147→  // copy (plan.features); fall back to the structured entitlement labels
   148→  // so a brand-new plan with no marketing copy still surfaces a readable
   149→  // feature list. Either way the data shown is real plan data — no mock.
   150→  const featureItems =
   151→    plan.features.length > 0
   152→      ? plan.features
   153→      : plan.entitlements.map((k) => ENTITLEMENT_LABELS[k as EntitlementKey] ?? k);
   154→
   155→  return (
   156→    <>
   157→      <div
   158→        className={`group relative flex h-full flex-col overflow-hidden rounded-3xl border bg-card p-8 text-card-foreground transition-colors duration-300 hover:border-foreground/20 ${
   159→          plan.active ? '' : 'opacity-60'
   160→        }`}
   161→      >
   162→        {/* Header: plan name (clean text label) + quick active toggle */}
   163→        <div className="flex items-start justify-between gap-3">
   164→          <div className="min-w-0">
   165→            <h3 className="text-2xl font-semibold tracking-tight text-foreground">{plan.name}</h3>
   166→          </div>
   167→          <div className="flex shrink-0 items-center gap-2">
   168→            <Label
   169→              htmlFor={`active-${plan.planId}`}
   170→              className="cursor-pointer text-xs text-muted-foreground"
   171→            >
   172→              {plan.active ? 'Active' : 'Inactive'}
   173→            </Label>
   174→            <Switch
   175→              id={`active-${plan.planId}`}
   176→              checked={plan.active}
   177→              disabled={!canEdit || activeMutation.isPending}
   178→              onCheckedChange={(v) => activeMutation.mutate(v)}
   179→            />
   180→          </div>
   181→        </div>
   182→
   183→        {/* Price — reflects the global billing-interval selector (Monthly / Yearly).
   184→             Free plan always shows "Free" regardless of the selector.
   185→             The price row + divider break out with -mx-6 into the card
   186→             padding (8px from the card edge) so the Yearly layout —
   187→             large monthly-equivalent + "/ month" + small "CHF X / year"
   188→             group, all on one line — has enough width to keep the small
   189→             yearly price fully visible instead of clipped. Monthly and
   190→             Free simply get extra breathing room; their visible output
   191→             is unchanged. */}
   192→        <div className="-mx-6 mt-8">
   193→          {plan.isFree ? (
   194→            <div className="flex items-baseline gap-2">
   195→              <span className="text-5xl font-semibold tracking-tight text-foreground">Free</span>
   196→            </div>
   197→          ) : billingInterval === 'yearly' ? (
   198→            // Yearly selected — compact INLINE presentation. The
   199→            // EQUIVALENT MONTHLY price (priceYearly / 12) is the MAIN
   200→            // LARGE price; the ACTUAL yearly price is shown smaller and
   201→            // muted directly beside it on the same horizontal line (no
   202→            // separate "≈ / month" line below). "/ month" belongs to
   203→            // the large monthly price; "/ year" belongs to the small
   204→            // yearly price. The monthly equivalent is computed
   205→            // dynamically from the real yearly price stored in the DB
   206→            // (priceYearly / 12) — never hardcoded.
   207→            //   e.g. Plus →  CHF 7.50 / month   CHF 90 / year
   208→            //        Pro  →  CHF 40.83 / month  CHF 490 / year
   209→            //        Max  →  CHF 82.50 / month  CHF 990 / year
   210→            //
   211→            // The card content is only ~245px (sidebar + 3-col grid +
   212→            // p-8), which cannot hold a text-5xl "CHF 7.50" + "/ month"
   213→            // + the full "CHF X / year" group on one line (the yearly
   214→            // would be clipped off). The large monthly-equivalent uses
   215→            // text-xl — still clearly larger than the text-sm yearly
   216→            // price, satisfying the large/prominent hierarchy — and the
   217→            // row breaks out with the container's -mx-6 (above) to gain
   218→            // the width needed to keep the small yearly price fully
   219→            // visible. shrink-0 + whitespace-nowrap + leading-none on
   220→            // the large span prevent it wrapping to a 2nd line box.
   221→            // ml-2 widens the gap between the two price groups vs. the
   222→            // gap between the large number and its "/ month" label.
   223→            <div className="flex items-baseline gap-2">
   224→              <span className="shrink-0 whitespace-nowrap text-xl font-semibold leading-none tracking-tight text-foreground">
   225→                {plan.currency}{' '}
   226→                {(plan.priceYearly / 12).toLocaleString(undefined, {
   227→                  minimumFractionDigits: 2,
   228→                  maximumFractionDigits: 2,
   229→                })}
   230→              </span>
   231→              <span className="whitespace-nowrap text-sm text-muted-foreground">/ month</span>
   232→              <span className="ml-2 shrink-0 whitespace-nowrap text-sm text-muted-foreground">
   233→                {formatCurrency(plan.priceYearly, plan.currency)} / year
   234→              </span>
   235→            </div>
   236→          ) : (
   237→            <div className="flex items-baseline gap-2">
   238→              <span className="shrink-0 whitespace-nowrap text-5xl font-semibold leading-none tracking-tight text-foreground">
   239→                {formatCurrency(plan.priceMonthly, plan.currency)}
   240→              </span>
   241→              <span className="whitespace-nowrap text-sm text-muted-foreground">/ month</span>
   242→            </div>
   243→          )}
   244→        </div>
   245→
   246→        {/* Thin horizontal divider between the price section and the features section */}
   247→        <div className="-mx-6 mt-8 h-px bg-border" aria-hidden />
   248→
   249→        {/* Feature items (section label omitted) */}
   250→        <div className="mt-6">
   251→          <ul className="space-y-2.5">
   252→            {featureItems.length === 0 ? (
   253→              <li className="text-sm text-muted-foreground">No features configured</li>
   254→            ) : (
   255→              featureItems.map((f, i) => (
   256→                <li key={`${f}-${i}`} className="flex items-start gap-2.5 text-sm text-card-foreground">
   257→                  <Check className="mt-0.5 h-4 w-4 shrink-0 text-muted-foreground" />
   258→                  <span className="leading-snug">{f}</span>
   259→                </li>
   260→              ))
   261→            )}
   262→          </ul>
   263→        </div>
   264→
   265→        {/* Edit button — pinned to card bottom */}
   266→        <div className="mt-auto pt-8">
   267→          <Button
   268→            variant="outline"
   269→            size="sm"
   270→            className="w-full rounded-full"
   271→            disabled={!canEdit}
   272→            onClick={() => setEditing(true)}
   273→          >
   274→            <Pencil className="h-4 w-4 mr-2" />
   275→            Edit Plan
   276→          </Button>
   277→          {!canEdit && (
   278→            <p className="mt-2 text-center text-[10px] text-muted-foreground">
   279→              Editing is restricted to the OWNER role.
   280→            </p>
   281→          )}
   282→        </div>
   283→      </div>
   284→
   285→      {editing && (
   286→        <EditPlanDialog plan={plan} open={editing} onOpenChange={setEditing} />
   287→      )}
   288→    </>
   289→  );
   290→}

   294→function EditPlanDialog({
   295→  plan,
   296→  open,
   297→  onOpenChange,
   298→}: {
   299→  plan: PlanConfigData;
   300→  open: boolean;
   301→  onOpenChange: (open: boolean) => void;
   302→}) {
   303→  const queryClient = useQueryClient();
   304→  const [name, setName] = useState(plan.name);
   305→  const [priceMonthly, setPriceMonthly] = useState(String(plan.priceMonthly));
   306→  const [priceYearly, setPriceYearly] = useState(String(plan.priceYearly));
   307→  const [currency, setCurrency] = useState(plan.currency);
   308→  const [interval, setInterval] = useState<'monthly' | 'yearly'>(plan.interval);
   309→  const [active, setActive] = useState(plan.active);
   310→  const [features, setFeatures] = useState(plan.features.join('\n'));
   311→  const [entitlements, setEntitlements] = useState<string[]>(plan.entitlements);
   312→  const [limits, setLimits] = useState<PlanLimits>(plan.limits);
   313→  // New (Task 64): free-trial duration + Stripe Price IDs.
   314→  const [freePlanDurationDays, setFreePlanDurationDays] = useState<string>(
   315→    plan.freePlanDurationDays == null ? '' : String(plan.freePlanDurationDays),
   316→  );
   317→  const [stripePriceIdMonthly, setStripePriceIdMonthly] = useState(plan.stripePriceIdMonthly ?? '');
   318→  const [stripePriceIdYearly, setStripePriceIdYearly] = useState(plan.stripePriceIdYearly ?? '');
   319→  const [clientDisplayOpen, setClientDisplayOpen] = useState(false);
   320→  const [stripeOpen, setStripeOpen] = useState(false);
   321→
   322→  // Fetch the supported currency list from CountryPricing so the
   323→  // currency field is a Select (not a free-text input).
   324→  const currenciesQuery = useQuery({
   325→    queryKey: ['platform-currencies'],
   326→    queryFn: async () => {
   327→      const res = await fetch('/api/platform/admin/countries', { credentials: 'include' });
   328→      const json = await res.json();
   329→      const rows = Array.isArray(json) ? json : (json.data ?? []);
   330→      const list = (rows as { currency?: string }[])
   331→        .map((r) => r.currency)
   332→        .filter((c): c is string => Boolean(c) && typeof c === 'string');
   333→      const unique = Array.from(new Set(list));
   334→      return unique.length > 0 ? unique : ['CHF', 'USD', 'EUR', 'MAD'];
   335→    },
   336→    staleTime: 5 * 60 * 1000,
   337→  });
   338→  const currencies = currenciesQuery.data ?? ['CHF'];
   339→
   340→  // EditPlanDialog is conditionally rendered by the parent (mounted
   341→  // fresh each time the user opens it), so the useState initializers
   342→  // above already seed local state from the latest server snapshot.
   343→  // No useEffect sync is needed — that would just trigger cascading
   344→  // renders (and the React Compiler correctly flags it).
   345→
   346→  const reset = () => {
   347→    setName(plan.name);
   348→    setPriceMonthly(String(plan.priceMonthly));
   349→    setPriceYearly(String(plan.priceYearly));
   350→    setCurrency(plan.currency);
   351→    setInterval(plan.interval);
   352→    setActive(plan.active);
   353→    setFeatures(plan.features.join('\n'));
   354→    setEntitlements(plan.entitlements);
   355→    setLimits(plan.limits);
   356→    setFreePlanDurationDays(plan.freePlanDurationDays == null ? '' : String(plan.freePlanDurationDays));
   357→    setStripePriceIdMonthly(plan.stripePriceIdMonthly ?? '');
   358→    setStripePriceIdYearly(plan.stripePriceIdYearly ?? '');
   359→  };
   360→
   361→  const saveMutation = useMutation({
   362→    mutationFn: (patch: PlanPatch) =>
   363→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, patch),
   364→    onSuccess: (data) => {
   365→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   366→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   367→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   368→      toast.success(`${data.name} updated — changes are now live for clients.`);
   369→      onOpenChange(false);
   370→    },
   371→    onError: (err: unknown) => {
   372→      const e = err as { error?: { message?: string }; message?: string };
   373→      toast.error(
   374→        e?.error?.message ?? e?.message ?? 'Unable to save plan. You may need OWNER access.',
   375→      );
   376→    },
   377→  });
   378→
   379→  const buildPatch = (): PlanPatch => {
   380→    const monthly = Number(priceMonthly) || 0;
   381→    const yearly = Number(priceYearly) || 0;
   382→    const isFreeDerived = monthly === 0 && yearly === 0;
   383→    return {
   384→      name,
   385→      priceMonthly: monthly,
   386→      priceYearly: yearly,
   387→      currency,
   388→      interval,
   389→      isFree: isFreeDerived,
   390→      freePlanDurationDays: isFreeDerived ? (freePlanDurationDays.trim() === '' ? null : Number(freePlanDurationDays) || null) : null,
   391→      stripePriceIdMonthly: stripePriceIdMonthly.trim() === '' ? null : stripePriceIdMonthly.trim(),
   392→      stripePriceIdYearly: stripePriceIdYearly.trim() === '' ? null : stripePriceIdYearly.trim(),
   393→      active,
   394→      features: features
   395→        .split('\n')
   396→        .map((f) => f.trim())
   397→        .filter(Boolean),
   398→      entitlements,
   399→      limits: { ...limits, storageBytes: Number(limits.storageBytes) || 0 },
   400→    };
   401→  };
   402→
   403→  return (
   404→    <Dialog open={open} onOpenChange={onOpenChange}>
   405→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   406→        <DialogHeader>
   407→          <div className="flex items-center gap-2">
   408→            <DialogTitle className="text-base">Edit {plan.name}</DialogTitle>
   409→            <PlanBadge planId={getPlanBadgeId(plan.planId)} />
   410→          </div>
   411→          <DialogDescription className="text-xs">
   412→            Single source of truth — changes propagate to the Client Billing page, MRR and
   413→            server-side entitlement enforcement.
   414→          </DialogDescription>
   415→        </DialogHeader>
   416→
   417→        <div className="space-y-5">
   418→          {/* -------------------- Basic Information -------------------- */}
   419→          <section className="space-y-3">
   420→            <h4 className="text-sm font-semibold">Basic Information</h4>
   421→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   422→              <div className="space-y-1">
   423→                <Label className="text-xs">Name</Label>
   424→                <Input
   425→                  value={name}
   426→                  onChange={(e) => setName(e.target.value)}
   427→                  className="h-9"
   428→                />
   429→              </div>
   430→              <div className="space-y-1">
   431→                <Label className="text-xs">Monthly Price</Label>
   432→                <Input
   433→                  type="number"
   434→                  value={priceMonthly}
   435→                  onChange={(e) => setPriceMonthly(e.target.value)}
   436→                  className="h-9"
   437→                />
   438→              </div>
   439→              <div className="space-y-1">
   440→                <Label className="text-xs">Yearly Price</Label>
   441→                <Input
   442→                  type="number"
   443→                  value={priceYearly}
   444→                  onChange={(e) => setPriceYearly(e.target.value)}
   445→                  className="h-9"
   446→                />
   447→              </div>
   448→              <div className="space-y-1">
   449→                <Label className="text-xs">Currency</Label>
   450→                <Select value={currency} onValueChange={setCurrency}>
   451→                  <SelectTrigger className="h-9">
   452→                    <SelectValue />
   453→                  </SelectTrigger>
   454→                  <SelectContent>
   455→                    {currencies.map((c) => (
   456→                      <SelectItem key={c} value={c}>
   457→                        {c}
   458→                      </SelectItem>
   459→                    ))}
   460→                  </SelectContent>
   461→                </Select>
   462→              </div>
   463→            </div>
   464→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   465→              <div className="space-y-1">
   466→                <Label className="text-xs">Billing Interval</Label>
   467→                <Select
   468→                  value={interval}
   469→                  onValueChange={(v) => setInterval(v as 'monthly' | 'yearly')}
   470→                >
   471→                  <SelectTrigger className="h-9">
   472→                    <SelectValue />
   473→                  </SelectTrigger>
   474→                  <SelectContent>
   475→                    <SelectItem value="monthly">monthly</SelectItem>
   476→                    <SelectItem value="yearly">yearly</SelectItem>
   477→                  </SelectContent>
   478→                </Select>
   479→              </div>
   480→              <div className="flex items-center justify-between rounded-md border px-3 h-9 sm:mt-5">
   481→                <Label
   482→                  htmlFor="dialog-active"
   483→                  className="text-xs text-muted-foreground cursor-pointer"
   484→                >
   485→                  Active
   486→                </Label>
   487→                <Switch
   488→                  id="dialog-active"
   489→                  checked={active}
   490→                  onCheckedChange={setActive}
   491→                />
   492→              </div>
   493→            </div>
   494→            {/* Free plan trial duration — shown only when both prices are 0
   495→                (i.e. this is a free plan). Empty = unlimited. */}
   496→            {Number(priceMonthly) === 0 && Number(priceYearly) === 0 && (
   497→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   498→                <div className="space-y-1">
   499→                  <Label className="text-xs">Free Access Duration (days)</Label>
   500→                  <Input
   501→                    type="number"
   502→                    value={freePlanDurationDays}
   503→                    onChange={(e) => setFreePlanDurationDays(e.target.value)}
   504→                    className="h-9"
   505→                    placeholder="empty = unlimited"
   506→                  />
   507→                  <p className="text-[10px] text-muted-foreground">
   508→                    Server-side enforced: when set, the user&apos;s free trial expires N days after activation.
   509→                    Empty = unlimited free access.
   510→                  </p>
   511→                </div>
   512→              </div>
   513→            )}
   514→          </section>
   515→
   516→          {/* -------------------- Stripe Billing (optional) -------------------- */}
   517→          <Collapsible open={stripeOpen} onOpenChange={setStripeOpen}>
   518→            <CollapsibleTrigger asChild>
   519→              <button
   520→                type="button"
   521→                className="flex items-center justify-between w-full rounded-md hover:bg-accent/30 px-2 py-1.5 transition-colors"
   522→              >
   523→                <div className="flex items-center gap-2">
