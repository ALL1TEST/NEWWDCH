   289→// -------------------- Validation helpers --------------------
   290→
   291→/** True when the Name field is non-empty AND every required field for the
   292→ *  currently-selected provider has a non-empty value. For providers that
   293→ *  require a validated connection, the connection state must also be
   294→ *  'connected'. Coming-soon providers are never valid (Create disabled). */
   295→function isFormValid(form: StorageForm): boolean {
   296→  if (!form.name.trim()) return false;
   297→  const def = getProviderDefinition(form.provider);
   298→  if (!def) return false;
   299→  if (def.status === 'coming_soon') return false;
   300→
   301→  const allFieldsFilled = def.fields.every((f) => {
   302→    if (!f.required) return true;
   303→    const v = form.config[f.key];
   304→    if (v == null) return false;
   305→    return f.type === 'switch' ? true : v.trim() !== '';
   306→  });
   307→  if (!allFieldsFilled) return false;
   308→
   309→  if (def.requiresConnection && form.connection !== 'connected') return false;
   310→  return true;
   311→}
   312→
   313→/** Signature for the connection/credential-group fields of the current
   314→ *  provider. Used to detect whether the user has edited any such field
   315→ *  since the last Test/Connect — if so, the 'connected' state is stale. */
   316→function connectionSignature(form: StorageForm): string {
   317→  const def = getProviderDefinition(form.provider);
   318→  if (!def) return '';
   319→  const groups: FieldGroup[] =
   320→    def.connectionType === 'oauth' ? ['connection'] : ['credentials'];
   321→  return def.fields
   322→    .filter((f) => groups.includes(f.group))
   323→    .map((f) => `${f.key}=${form.config[f.key] ?? ''}`)
   324→    .join('|');
   325→}
   326→
   327→/** True when all required connection/credential fields are filled, so the
   328→ *  Test/Connect button can be enabled. */
   329→function canValidateConnection(form: StorageForm): boolean {
   330→  const def = getProviderDefinition(form.provider);
   331→  if (!def || def.connectionType === 'none') return false;
   332→  const groups: FieldGroup[] =
   333→    def.connectionType === 'oauth' ? ['connection'] : ['credentials'];
   334→  return def.fields
   335→    .filter((f) => groups.includes(f.group))
   336→    .every((f) => {
   337→      if (!f.required) return true;
   338→      const v = form.config[f.key];
   339→      return v != null && v.trim() !== '';
   340→    });
   341→}
   342→
   343→/** Detects password-type connection/credential fields still holding the
   344→ *  API mask placeholder ('••••••••'). When editing an existing storage,
   345→ *  secrets come back masked; re-testing with a mask sends a bogus secret.
   346→ *  We block the test and ask the user to re-enter the secret. */
   347→function maskedSecretFields(form: StorageForm): string[] {
   348→  const def = getProviderDefinition(form.provider);
   349→  if (!def) return [];
   350→  const groups: FieldGroup[] =
   351→    def.connectionType === 'oauth' ? ['connection'] : ['credentials'];
   352→  return def.fields
   353→    .filter((f) => f.type === 'password' && groups.includes(f.group))
   354→    .map((f) => f.label)
   355→    .filter((label) => label && label.length > 0)
   356→    .filter((_, i) => {
   357→      const f = def.fields
   358→        .filter((x) => x.type === 'password' && groups.includes(x.group))[i];
   359→      const v = form.config[f.key];
   360→      return typeof v === 'string' && v.includes('•');
   361→    });
   362→}
   363→
   364→/** Build the config object from the current provider's fields only. Empty
   365→ *  optional fields are omitted. Number fields → Number. Switch → boolean. */
   366→function buildConfigObject(form: StorageForm): Record<string, unknown> {
   367→  const def = getProviderDefinition(form.provider);
   368→  if (!def) return {};
   369→  const configObj: Record<string, unknown> = {};
   370→  for (const field of def.fields) {
   371→    const raw = form.config[field.key];
   372→    if (raw == null || raw.trim() === '') continue;
   373→    if (field.type === 'number') {
   374→      const n = Number(raw);
   375→      if (!Number.isNaN(n)) configObj[field.key] = n;
   376→    } else if (field.type === 'switch') {
   377→      configObj[field.key] = raw === 'true';
   378→    } else {
   379→      configObj[field.key] = raw;
   380→    }
   381→  }
   382→  return configObj;
   383→}

  1112→                    !canValidateConnection(form) ||
  1113→                    (currentDef.connectionType === 'credentials' && !form.name.trim())
  1114→                  }
  1115→                >
  1116→                  {isTesting ? (
  1117→                    <Loader2 className="h-4 w-4 mr-2 animate-spin" />
  1118→                  ) : currentDef.connectionType === 'oauth' ? (
  1119→                    <Plug className="h-4 w-4 mr-2" />
  1120→                  ) : (
  1121→                    <RefreshCw className="h-4 w-4 mr-2" />
  1122→                  )}
  1123→                  {currentDef.actionLabel}
  1124→                </Button>
  1125→              )}
  1126→            </div>
  1127→
  1128→            {/* Right: Cancel + Create */}
  1129→            <div className="flex items-center gap-2">
  1130→              <Button variant="outline" size="sm" onClick={closeDialog}>
  1131→                Cancel
  1132→              </Button>
  1133→              <Button
  1134→                size="sm"
  1135→                onClick={handleSubmit}
  1136→                disabled={isSaving || !isFormValid(form)}
  1137→              >
  1138→                {isSaving && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
  1139→                {editingId ? 'Update' : 'Create'}
  1140→              </Button>
  1141→            </div>
  1142→          </DialogFooter>
  1143→        </DialogContent>
  1144→      </Dialog>
  1145→
  1146→      {/* Delete Confirmation */}
  1147→      <ConfirmDialog
  1148→        open={!!deleteTarget}
  1149→        onOpenChange={(v) => !v && setDeleteTarget(null)}
  1150→        title="Delete Storage Configuration"
  1151→        description={
  1152→          deleteTarget
  1153→            ? `Are you sure you want to delete "${deleteTarget.name}"? Existing backups stored via this configuration will not be affected.`
  1154→            : undefined
  1155→        }
  1156→        confirmLabel="Delete"
  1157→        variant="destructive"
  1158→        onConfirm={() => {
  1159→          if (deleteTarget) deleteMutation.mutate(deleteTarget.id);
  1160→        }}
  1161→        isLoading={deleteMutation.isPending}
  1162→      />
  1163→    </div>
  1164→  );
  1165→}

  1177→// -------------------- Config Field Renderer --------------------
  1178→// Renders a single provider field generically. Password fields mask
  1179→// their value (type=password + a lock affordance). Switch fields use
  1180→// the Switch component. Multiline password fields (e.g. GCS private
  1181→// key) render as a masked Textarea. The component is fully driven by
  1182→// the ProviderField definition — it has no provider-specific logic.
  1183→
  1184→function ConfigField({
  1185→  field,
  1186→  value,
  1187→  onChange,
  1188→  disabled,
  1189→}: {
  1190→  field: ProviderField;
  1191→  value: string;
  1192→  onChange: (v: string) => void;
  1193→  disabled?: boolean;
  1194→}) {
  1195→  const inputId = `storage-config-${field.key}`;
  1196→  return (
  1197→    <div className="space-y-1.5">
  1198→      <Label htmlFor={inputId} className="text-xs">
  1199→        {field.label}
  1200→        {field.required && <span className="text-destructive ml-0.5">*</span>}
  1201→      </Label>
  1202→      {field.type === 'switch' ? (
  1203→        <div className="flex items-center gap-2 h-9">
  1204→          <Switch
  1205→            id={inputId}
  1206→            checked={value === 'true'}
  1207→            onCheckedChange={(checked) => onChange(checked ? 'true' : 'false')}
  1208→            disabled={disabled}
  1209→          />
  1210→          <span className="text-xs text-muted-foreground">
  1211→            {value === 'true' ? 'Enabled' : 'Disabled'}
  1212→          </span>
  1213→        </div>
  1214→      ) : field.type === 'password' && field.multiline ? (
  1215→        <Textarea
  1216→          id={inputId}
  1217→          placeholder={field.placeholder}
  1218→          value={value}
  1219→          onChange={(e) => onChange(e.target.value)}
  1220→          rows={4}
  1221→          disabled={disabled}
  1222→          className="font-mono text-xs"
  1223→        />
  1224→      ) : (
  1225→        <div className="relative">
  1226→          <Input
  1227→            id={inputId}
  1228→            type={field.type === 'password' ? 'password' : field.type === 'number' ? 'number' : 'text'}
  1229→            placeholder={field.placeholder}
  1230→            value={value}
  1231→            onChange={(e) => onChange(e.target.value)}
  1232→            disabled={disabled}
  1233→            autoComplete={field.type === 'password' ? 'new-password' : undefined}
  1234→            className={field.type === 'password' ? 'pr-9' : ''}
  1235→          />
  1236→          {field.type === 'password' && value && (
  1237→            <Lock className="absolute right-3 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground/60 pointer-events-none" />
  1238→          )}
  1239→        </div>
  1240→      )}
  1241→      {field.helpText && (
  1242→        <p className="text-xs text-muted-foreground leading-relaxed">{field.helpText}</p>
  1243→      )}
  1244→    </div>
  1245→  );
  1246→}
