     1→// ============================================================
     2→// PLAN CONFIG — DB-backed, cached single source of truth for plans,
     3→// pricing, entitlements and usage limits.
     4→// ============================================================
     5→// The Client Billing API and the Platform Admin both read from (and
     6→// the admin mutates) this same service. `platform-data.ts` delegates
     7→// its `PLANS` / `getPlan` / `monthlyPrice` to the synchronous cache
     8→// exposed here, so an owner edit (savePlanConfig) propagates to the
     9→// client billing experience and to MRR on the next read.
    10→//
    11→// MULTI-CURRENCY: the plan carries ONE base price configuration —
    12→// priceMonthly / priceYearly denominated in the plan's default /
    13→// fallback `currency` (both edited directly in the Edit Plan modal).
    14→// The customer's currency is resolved SERVER-SIDE from their IP
    15→// geolocation (see country-pricing.ts:resolveCustomerPricing) when
    16→// the plan's `autoCurrency` flag is on; otherwise every customer is
    17→// billed in the plan default currency.
    18→//
    19→// pricesByCurrency is a DERIVED store (NOT edited per-plan in the
    20→// modal): the default currency's entry mirrors the base price on
    21→// every save, and the other entries hold the platform-level
    22→// regional prices (CountryPricing.regionalPrices + legacy entries)
    23→// — preserved across saves so customers in other currencies keep
    24→// seeing their configured price (e.g. "90 MAD / month").
    25→//
    26→// STRIPE: for each paid plan × supported currency × interval, a real
    27→// Stripe Price is created on sync and its ID persisted in
    28→// stripePriceIdsByCurrency. The default currency's IDs are mirrored
    29→// in stripePriceIdMonthly/Yearly for legacy callers. Checkout always
    30→// resolves the per-currency Stripe Price ID server-side — the
    31→// customer's currency determines which Price is charged.
    32→//
    33→// The cache is hydrated from the PlanConfig table on startup and
    34→// after every mutation. If the table is empty on first hydrate, the
    35→// DEFAULT configs are seeded.
    36→// ============================================================

    95→// -------------------- Entitlement normalization --------------------
    96→
    97→/** Normalize a raw entitlement key list (DB row or API input):
    98→ *  - legacy 'ai_content' → 'ai_platform' (it predates the two-key
    99→ *    split and always meant the platform-provided AI).
   100→ *  - Platform AI and Client's Own AI API are INDEPENDENT — both may
   101→ *    be present (no mutual exclusion to enforce).
   102→ *  - API Access is REMOVED as a feature: 'api_access' is a legacy
   103→ *    removed key (like custom_domains / white_label) — Client's Own
   104→ *    AI API already represents the client's own-provider/API access,
   105→ *    so the duplicate key is STRIPPED on load + save. Old DB rows
   106→ *    self-clean; stale API input never reaches the saved plan.
   107→ *  - legacy 'custom_domains' / 'white_label' are STRIPPED: site
   108→ *    domains and site branding are client-owned in this architecture
   109→ *    (every site carries its own domain + identity), so they are not
   110→ *    plan entitlements. Old DB rows self-clean on load + next save.
   111→ *  - dedupes while preserving order.
   112→ *  Non-editor keys (e.g. 'audit_log') pass through untouched. */
   113→export function normalizeEntitlementKeys(keys: readonly string[]): string[] {
   114→  // Legacy keys that are no longer plan entitlements — stripped on
   115→  // load/save so existing rows + stale API input clean themselves.
   116→  // 'api_access' joined them: Client's Own AI API already covers the
   117→  // client's own-API connectivity, so a separate API Access feature
   118→  // no longer exists anywhere (editor, cards, server checks).
   119→  const LEGACY_REMOVED_KEYS = new Set(['custom_domains', 'white_label', 'api_access']);
   120→  const out: string[] = [];
   121→  const seen = new Set<string>();
   122→  const hasPlatform = keys.includes('ai_platform') || keys.includes('ai_content');
   123→  for (const k of keys) {
   124→    if (typeof k !== 'string' || !k.trim()) continue;
   125→    if (k === 'ai_content') continue; // legacy → replaced by ai_platform below
   126→    if (LEGACY_REMOVED_KEYS.has(k)) continue; // not plan entitlements anymore
   127→    if (k === 'ai_platform' && hasPlatform) {
   128→      if (!seen.has('ai_platform')) out.push('ai_platform');
   129→      seen.add('ai_platform');
   130→      continue;
   131→    }
   132→    if (!seen.has(k)) {
   133→      out.push(k);
   134→      seen.add(k);
   135→    }
   136→  }
   137→  // Insert the platform AI key at the position where the legacy key
   138→  // (or an existing 'ai_platform') first appeared — append at the end
   139→  // when neither was present in the ordered loop above (defensive).
   140→  if (hasPlatform && !seen.has('ai_platform')) out.push('ai_platform');
   141→  return out;
   142→}

   167→/** Pick only the known limit fields from an untyped/partial input —
   168→ *  unknown keys (stale `aiWords` / `automationRuns` from older rows)
   169→ *  never leak into the cache and are not re-serialized on save. */
   170→function pickLimits(parsed: unknown): PlanLimits {
   171→  const p = (parsed ?? {}) as Partial<PlanLimits>;
   172→  const num = (v: unknown): number => (typeof v === 'number' && !Number.isNaN(v) ? v : 0);
   173→  return {
   174→    maxSites: num(p.maxSites),
   175→    storageBytes: num(p.storageBytes),
   176→    aiArticlesPerMonth: num(p.aiArticlesPerMonth),
   177→    aiWordsPerMonth: num(p.aiWordsPerMonth),
   178→    aiImagesPerMonth: num(p.aiImagesPerMonth),
   179→  };
   180→}

   182→export type BillingInterval = 'monthly' | 'yearly';
   183→
   184→/** The billing periods enabled on a plan (at least one, enforced).
   185→ *  A disabled period: not shown on the Client Billing page, rejected
   186→ *  at checkout (400 BILLING_PERIOD_NOT_ENABLED), and never gets a
   187→ *  Stripe Price created or selected. */
   188→export function enabledIntervalsOf(plan: {
   189→  billingMonthly: boolean;
   190→  billingYearly: boolean;
   191→}): BillingInterval[] {
   192→  const out: BillingInterval[] = [];
   193→  if (plan.billingMonthly) out.push('monthly');
   194→  if (plan.billingYearly) out.push('yearly');
   195→  return out;
   196→}
   197→
   198→/** Clamp a requested default cadence to an ENABLED period:
   199→ *  only monthly on → 'monthly'; only yearly on → 'yearly'; both on
   200→ *  → the requested value. (Neither enabled is rejected upstream —
   201→ *  here it falls back to the requested value.) */
   202→function normalizeInterval(
   203→  requested: BillingInterval,
   204→  billingMonthly: boolean,
   205→  billingYearly: boolean,
   206→): BillingInterval {
   207→  if (billingMonthly && !billingYearly) return 'monthly';
   208→  if (!billingMonthly && billingYearly) return 'yearly';
   209→  return requested;
   210→}

   434→/** Resolve the platform's default billing currency via a LAZY import of
   435→ *  country-pricing (which itself imports `getPlanConfigSync` from here).
   436→ *  The lazy `await import()` breaks the static circular dependency —

   439→async function resolvePlatformDefaultCurrency(): Promise<string> {
   440→  try {
   441→    const { getPlatformDefaultCurrency } = await import('./country-pricing');
   442→    return await getPlatformDefaultCurrency();
   443→  } catch {
   444→    return 'CHF';
   445→  }
   446→}

   454→let _hydrated = false;

   456→const _subscribers = new Set<() => void>();

   458→function notify() {
   459→  for (const cb of _subscribers) {
   460→    try {
   461→      cb();
   462→    } catch {
   463→      // ignore subscriber errors
   464→    }
   465→  }
   466→}

   468→/** Register a callback fired after every cache refresh (hydrate / mutation). */
   469→export function subscribe(cb: () => void): () => void {
   470→  _subscribers.add(cb);
   471→  return () => _subscribers.delete(cb);
   472→}

   474→/** Synchronous access to the cached plan configs. Returns defaults before
   475→ *  the first hydrate completes (the defaults match the seeded DB rows, so
   476→ *  numbers are correct from the very first render). */
   477→export function getPlanConfigsSync(): PlanConfigData[] {
   478→  if (!_hydrated && !_hydrating) {
   479→    void hydrate();
   480→  }
   481→  return _cache;
   482→}

   497→ *  `requireFeature(request, 'ai_content')` gate keeps passing for
   498→ *  BOTH Platform AI and Client's Own AI API plans (general AI-tools
   499→ *  access) with zero route changes. Source-specific checks use

   501→export function getPlanEntitlements(planId: string): string[] {

   505→    return base.includes('ai_content') ? base : [...base, 'ai_content'];
   506→  }
   507→  return base;
   508→}

   542→function rowToData(row: PlanConfigRow): PlanConfigData {
   543→  let features: string[] = [];
   544→  let entitlements: string[] = [];
   545→  try {
   546→    features = JSON.parse(row.features || '[]');
   547→  } catch {
   548→    features = [];
   549→  }
   550→  try {
   551→    entitlements = JSON.parse(row.entitlements || '[]');
   552→  } catch {
   553→    entitlements = [];
   554→  }
   555→  // Explicitly pick only the known limit fields. Older DB rows may still
   556→  // carry stale `aiWords` / `aiArticles` / `automationRuns` keys in the
   557→  // limits JSON (those were removed because no real enforcement exists).
   558→  // Picking only the known fields prevents the stale keys from leaking
   559→  // back into the cache and being re-serialized on the next save.
   560→  let limits: PlanLimits = { ...DEFAULT_LIMITS };
   561→  try {
   562→    limits = pickLimits(JSON.parse(row.limits || '{}'));
   563→  } catch {
   564→    // keep defaults
   565→  }
   566→  // AI usage limits only exist for Platform AI plans — normalize the
   567→  // stored configuration to match the entitlement mode.
   568→  limits = zeroAiLimitsIfNotPlatform(limits, entitlements);
   569→  let pricesByCurrency: PricesByCurrency = {};
   570→  try {
   571→    const parsed = JSON.parse(row.pricesByCurrency || '{}') as Partial<PricesByCurrency>;
   572→    if (parsed && typeof parsed === 'object') {
   573→      for (const [cur, val] of Object.entries(parsed)) {
   574→        if (val && typeof val === 'object') {
   575→          const m = typeof (val as CurrencyPrice).monthly === 'number' ? (val as CurrencyPrice).monthly : 0;
   576→          const y = typeof (val as CurrencyPrice).yearly === 'number' ? (val as CurrencyPrice).yearly : 0;
   577→          pricesByCurrency[cur.toUpperCase()] = { monthly: m, yearly: y };
   578→        }
   579→      }
   580→    }
   581→  } catch {
   582→    pricesByCurrency = {};
   583→  }
   584→  let stripePriceIdsByCurrency: StripePriceIdsByCurrency = {};
   585→  try {
   586→    const parsed = JSON.parse(row.stripePriceIdsByCurrency || '{}') as Partial<StripePriceIdsByCurrency>;
   587→    if (parsed && typeof parsed === 'object') {
   588→      for (const [cur, val] of Object.entries(parsed)) {
   589→        if (val && typeof val === 'object') {
   590→          const m = typeof (val as CurrencyStripeIds).monthly === 'string' ? (val as CurrencyStripeIds).monthly : null;
   591→          const y = typeof (val as CurrencyStripeIds).yearly === 'string' ? (val as CurrencyStripeIds).yearly : null;
   592→          stripePriceIdsByCurrency[cur.toUpperCase()] = { monthly: m, yearly: y };
   593→        }
   594→      }
   595→    }
   596→  } catch {
   597→    stripePriceIdsByCurrency = {};
   598→  }
   599→  return {
   600→    planId: row.planId,
   601→    name: row.name,
   602→    priceMonthly: row.priceMonthly,
   603→    priceYearly: row.priceYearly,
   604→    currency: row.currency,
   605→    autoCurrency: row.autoCurrency ?? true,
   606→    pricesByCurrency,
   607→    stripePriceIdsByCurrency,
   608→    billingMonthly: row.billingMonthly ?? true,
   609→    billingYearly: row.billingYearly ?? true,
   610→    // Normalize the default cadence to an ENABLED period (defensive:
   611→    // legacy rows / stale values never advertise a disabled period).
   612→    interval: normalizeInterval(
   613→      (row.interval === 'yearly' ? 'yearly' : 'monthly') as BillingInterval,
   614→      row.billingMonthly ?? true,
   615→      row.billingYearly ?? true,
   616→    ),
   617→    isFree: row.isFree,
   618→    freePlanDurationDays: row.freePlanDurationDays,
   619→    stripePriceIdMonthly: row.stripePriceIdMonthly,
   620→    stripePriceIdYearly: row.stripePriceIdYearly,
   621→    active: row.active,
   622→    features,
   623→    // Normalize: legacy 'ai_content' → 'ai_platform'; the legacy
   624→    // 'api_access' key is stripped (no separate API Access feature —
   625→    // Client's Own AI API covers it); the two AI keys are independent
   626→    // (both may be present).
   627→    entitlements: normalizeEntitlementKeys(entitlements),
   628→    limits,
   629→    badgeVariant: row.badgeVariant,
   630→    sortOrder: row.sortOrder,
   631→  };
   632→}

   660→/**
   661→ * Migrate the legacy catalog to the new Free/Plus/Pro/Max shape. Idempotent.
   662→ *
   663→ * Legacy catalog (created by older bootstrap):
   664→ *   beta (free, active=false), pro, max, enterprise (stub)
   665→ *
   666→ * Migration steps (each guarded, can be re-run safely):
   667→ *  1. If `enterprise` exists AND no active subscription references it → delete it.
   668→ *     (Active subs that reference enterprise are preserved — the migration will
   669→ *      not break them; the plan row stays if needed. In practice, the demo
   670→ *      dataset has no DB subscriptions yet, so this is a clean delete.)
   671→ *  2. If `beta` exists → rename to `plus` (planId, name, badgeVariant) and
   672→ *     re-enable it (legacy `beta` was active=false in the demo DB). Also
   673→ *     shift sortOrder to 1 (Free takes 0).
   674→ *  3. If `free` does NOT exist → insert the Free plan at sortOrder 0.
   675→ *  4. Renumber sortOrder to match the canonical order: free=0, plus=1, pro=2, max=3.
   676→ *
   677→ * Returns a short report string for the bootstrap log.
   678→ */
   679→export async function migrateLegacyPlans(): Promise<string> {
   680→  const notes: string[] = [];
   681→
   682→  try {
   683→    // Step 1: delete Enterprise if safe
   684→    const enterprise = await db.planConfig.findUnique({ where: { planId: 'enterprise' } });
   685→    if (enterprise) {
   686→      const subCount = await db.subscription.count({ where: { planId: 'enterprise' } });
   687→      if (subCount === 0) {
   688→        await db.planConfig.delete({ where: { planId: 'enterprise' } });
   689→        notes.push('deleted enterprise plan (0 active subscriptions)');
   690→      } else {
   691→        notes.push(`kept enterprise plan (${subCount} active subscriptions reference it)`);
   692→      }
   693→    }
   694→
   695→    // Step 2: rename beta → plus (and refresh to a real paid tier)
   696→    const beta = await db.planConfig.findUnique({ where: { planId: 'beta' } });
   697→    if (beta) {
   698→      // Don't overwrite an existing plus row — if both exist, prefer the
   699→      // plus row and just delete the beta stub.
   700→      const plus = await db.planConfig.findUnique({ where: { planId: 'plus' } });
   701→      if (!plus) {
   702→        // Rename beta → plus AND refresh the data to the real DEFAULT
   703→        // Plus tier (paid plan with real entitlements + limits, not the
   704→        // legacy free-tier Beta stub). The old Beta was active=false and
   705→        // had no real customers, so this is a safe refresh.
   706→        const plusDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'plus')!;
   707→        await db.planConfig.update({
   708→          where: { planId: 'beta' },
   709→          data: {
   710→            ...dataToRow(plusDefault),
   711→            // Keep the original createdAt for continuity.
   712→            // (updatedAt is auto.)
   713→          },
   714→        });
   715→        notes.push('renamed + refreshed beta → plus (real paid tier)');
   716→      } else {
   717→        // Both beta and plus exist — delete the beta stub (migrate any
   718→        // legacy subscriptions on beta to plus first).
   719→        const betaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   720→        if (betaSubs > 0) {
   721→          await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   722→          notes.push(`migrated ${betaSubs} beta subscription(s) → plus`);
   723→        }
   724→        await db.planConfig.delete({ where: { planId: 'beta' } });
   725→        notes.push('deleted legacy beta stub (plus already exists)');
   726→      }
   727→    }
   728→
   729→    // Step 3: insert Free if missing
   730→    const free = await db.planConfig.findUnique({ where: { planId: 'free' } });
   731→    if (!free) {
   732→      const freeDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'free')!;
   733→      await db.planConfig.create({ data: dataToRow(freeDefault) });
   734→      notes.push('inserted free plan');
   735→    } else {
   736→      // Ensure free plan is properly flagged
   737→      if (!free.isFree || free.badgeVariant !== 'free' || free.sortOrder !== 0) {
   738→        await db.planConfig.update({
   739→          where: { planId: 'free' },
   740→          data: { isFree: true, badgeVariant: 'free', sortOrder: 0 },
   741→        });
   742→        notes.push('normalized free plan flags');
   743→      }
   744→    }
   745→
   746→    // Step 4: normalize sortOrder for all canonical plans
   747→    const canonicalOrder: Record<string, number> = { free: 0, plus: 1, pro: 2, max: 3 };
   748→    for (const [pid, order] of Object.entries(canonicalOrder)) {
   749→      const row = await db.planConfig.findUnique({ where: { planId: pid } });
   750→      if (row && row.sortOrder !== order) {
   751→        await db.planConfig.update({ where: { planId: pid }, data: { sortOrder: order } });
   752→      }
   753→    }
   754→
   755→    // Also migrate legacy subscriptions on 'beta' or 'enterprise' planId
   756→    // to the new ids (beta → plus, enterprise → max) so historical subs
   757→    // don't reference a missing plan.
   758→    const legacyBetaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   759→    if (legacyBetaSubs > 0) {
   760→      await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   761→      notes.push(`migrated ${legacyBetaSubs} legacy beta subscription(s) → plus`);
   762→    }
   763→    const legacyEnterpriseSubs = await db.subscription.count({ where: { planId: 'enterprise' } });
   764→    if (legacyEnterpriseSubs > 0) {
   765→      await db.subscription.updateMany({ where: { planId: 'enterprise' }, data: { planId: 'max' } });
   766→      notes.push(`migrated ${legacyEnterpriseSubs} legacy enterprise subscription(s) → max`);
   767→    }
   768→  } catch (err) {
   769→    notes.push(`migration error (non-fatal): ${(err as Error).message}`);
   770→  }
   771→
   772→  return notes.length > 0 ? notes.join('; ') : 'no migration needed';
   773→}

   775→/** Load all plan configs from the DB into the cache. Self-seeds the
   776→ *  DEFAULT_PLAN_CONFIGS when the table is empty. Runs the legacy
   777→ *  migration when the catalog still contains 'beta' or 'enterprise'.
   778→ *  Safe to call repeatedly. */
   779→export async function hydrate(): Promise<void> {
   780→  if (_hydrating) return _hydrating;
   781→  _hydrating = (async () => {
   782→    try {
   783→      const rows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   784→      if (rows.length === 0) {
   785→        // Self-seed so the platform is never without plans.
   786→        await db.planConfig.createMany({ data: DEFAULT_PLAN_CONFIGS.map(dataToRow) });
   787→        _cache = DEFAULT_PLAN_CONFIGS.map((p) => ({
   788→          ...p,
   789→          features: [...p.features],
   790→          entitlements: [...p.entitlements],
   791→          limits: { ...p.limits },
   792→        }));
   793→      } else {
   794→        // Check for legacy catalog (beta / enterprise) and migrate.
   795→        const hasLegacy =
   796→          rows.some((r) => r.planId === 'beta' || r.planId === 'enterprise') ||
   797→          !rows.some((r) => r.planId === 'free');
   798→        if (hasLegacy) {
   799→          await migrateLegacyPlans();
   800→          const migratedRows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   801→          _cache = migratedRows.map(rowToData);
   802→        } else {
   803→          _cache = rows.map(rowToData);
   804→        }
   805→      }
   806→      _hydrated = true;
   807→      notify();
   808→    } catch (err) {
   809→      // DB unavailable — keep defaults (still functional, just not persisted).
   810→      _hydrated = true;
   811→      notify();
   812→    } finally {
   813→      _hydrating = null;
   814→    }
   815→  })();
   816→  return _hydrating;
   817→}

   858→/** Merge a patch (Partial<PlanConfigData>) into the current data.
   859→ *  The Edit Plan modal sends the ONE base price configuration
   860→ *  (priceMonthly + priceYearly + currency + autoCurrency) — NOT a
   861→ *  per-currency matrix. Derivation rules:
   862→ *  - pricesByCurrency: starts from the current map (or the explicit
   863→ *    patch map when provided); whenever the base price / currency is
   864→ *    in the patch, the DEFAULT currency's entry is re-derived as
   865→ *    { monthly: priceMonthly, yearly: priceYearly }. Other currency
   866→ *    entries (platform regional / legacy prices) are PRESERVED.
   867→ *  - stripePriceIdsByCurrency: kept as-is when omitted (the Stripe
   868→ *    sync maintains it); the default-currency snapshots mirror the
   869→ *    default entry.
   870→ */
   871→function mergePlanPatch(current: PlanConfigData, patch: PlanConfigInput, defaultCurrency: string): PlanConfigData {
   872→  const curUpper = defaultCurrency.toUpperCase();
   873→
   874→  // pricesByCurrency — clone (never mutate the cache's object in place).
   875→  const sourcePrices = patch.pricesByCurrency ?? current.pricesByCurrency ?? {};
   876→  const pricesByCurrency: PricesByCurrency = { ...sourcePrices };
   877→
   878→  // Base price precedence:
   879→  //   1. Explicit patch values (the modal always sends these).
   880→  //   2. Legacy mirror: an explicit pricesByCurrency map with an entry
   881→  //      for the final default currency (backward-compat callers).
   882→  //   3. Current snapshot.
   883→  const mirrored =
   884→    patch.priceMonthly === undefined && patch.pricesByCurrency !== undefined
   885→      ? pricesByCurrency[curUpper]
   886→      : undefined;
   887→  const priceMonthly =
   888→    patch.priceMonthly !== undefined ? patch.priceMonthly : mirrored?.monthly ?? current.priceMonthly;
   889→  const priceYearly =
   890→    patch.priceYearly !== undefined ? patch.priceYearly : mirrored?.yearly ?? current.priceYearly;
   891→
   892→  // The base price configuration was touched (or a legacy map was sent)
   893→  // → re-derive the DEFAULT currency's entry from the base price. All
   894→  // OTHER currency entries (platform regional / legacy prices) are
   895→  // preserved untouched.
   896→  const baseTouched =
   897→    patch.priceMonthly !== undefined ||
   898→    patch.priceYearly !== undefined ||
   899→    patch.currency !== undefined ||
   900→    patch.pricesByCurrency !== undefined;
   901→  if (baseTouched) {
   902→    pricesByCurrency[curUpper] = { monthly: priceMonthly, yearly: priceYearly };
   903→  }
   904→
   905→  const stripePriceIdsByCurrency: StripePriceIdsByCurrency = {
   906→    ...(patch.stripePriceIdsByCurrency ?? current.stripePriceIdsByCurrency ?? {}),
   907→  };
   908→  // Default-currency snapshot — keep the legacy fields in sync so legacy
   909→  // callers that read stripePriceIdMonthly/Yearly see the right values.
   910→  const defIds = stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
   911→  const stripePriceIdMonthly =
   912→    patch.stripePriceIdMonthly !== undefined
   913→      ? patch.stripePriceIdMonthly
   914→      : patch.stripePriceIdsByCurrency !== undefined
   915→        ? defIds.monthly
   916→        : current.stripePriceIdMonthly;
   917→  const stripePriceIdYearly =
   918→    patch.stripePriceIdYearly !== undefined
   919→      ? patch.stripePriceIdYearly
   920→      : patch.stripePriceIdsByCurrency !== undefined
   921→        ? defIds.yearly
   922→        : current.stripePriceIdYearly;
   923→  const isFree = patch.isFree ?? current.isFree;
   924→  // ---- Billing periods ----
   925→  const billingMonthly = patch.billingMonthly ?? current.billingMonthly;
   926→  const billingYearly = patch.billingYearly ?? current.billingYearly;
   927→  // The default cadence must be an ENABLED period (single-period plans
   928→  // are pinned to their only period; both-period plans keep the
   929→  // admin's requested / current cadence).
   930→  const interval = normalizeInterval(
   931→    patch.interval ?? current.interval,
   932→    billingMonthly,
   933→    billingYearly,
   934→  );
   935→  return {
   936→    planId: current.planId,
   937→    name: patch.name ?? current.name,
   938→    priceMonthly,
   939→    priceYearly,
   940→    currency: curUpper,
   941→    autoCurrency: patch.autoCurrency ?? current.autoCurrency,
   942→    pricesByCurrency,
   943→    stripePriceIdsByCurrency,
   944→    billingMonthly,
   945→    billingYearly,
   946→    interval,
   947→    isFree,
   948→    freePlanDurationDays: patch.freePlanDurationDays ?? current.freePlanDurationDays,
   949→    stripePriceIdMonthly,
   950→    stripePriceIdYearly,
   951→    active: patch.active ?? current.active,
   952→    features: patch.features ?? current.features,
   953→    // Entitlements: the patch's editor-managed keys replace the current
   954→    // ones, while NON-editor keys (e.g. 'audit_log') are PRESERVED from
   955→    // the current plan — the Create/Edit Plan modal only manages the 9
   956→    // Feature Access keys. Then normalize (legacy 'ai_content' →
   957→    // 'ai_platform'; the removed 'api_access' key is stripped; the two
   958→    // AI keys are independent).
   959→    entitlements: normalizeEntitlementKeys(
   960→      patch.entitlements !== undefined
   961→        ? [
   962→            ...patch.entitlements,
   963→            ...current.entitlements.filter(
   964→              (k) =>
   965→                !patch.entitlements!.includes(k) &&
   966→                k !== 'ai_content' &&
   967→                // keys below are editor-managed (replaced by the patch):
   968→                !PLAN_EDITOR_KEY_SET.has(k),
   969→            ),
   970→          ]
   971→        : current.entitlements,
   972→    ),
   973→    // Limits: merge; the AI usage limits are only part of the saved
   974→    // configuration while the plan includes Platform AI (zeroed
   975→    // otherwise — Client's Own AI API usage is never limited).
   976→    limits: zeroAiLimitsIfNotPlatform(
   977→      pickLimits({ ...current.limits, ...(patch.limits ?? {}) }),
   978→      patch.entitlements !== undefined
   979→        ? normalizeEntitlementKeys(patch.entitlements)
   980→        : current.entitlements,
   981→    ),
   982→    badgeVariant: patch.badgeVariant ?? current.badgeVariant,
   983→    sortOrder: patch.sortOrder ?? current.sortOrder,
   984→  };
   985→}

   987→/** Update an existing plan config. Writes DB, refreshes cache.
   988→ *
   989→ *  AUTO-SYNC TO STRIPE: when Stripe is configured AND the plan is
   990→ *  PAID (any currency has a positive price) AND the admin did NOT
   991→ *  explicitly set the Stripe Price IDs in this patch, the backend calls
   992→ *  syncPlanToStripe (multi-currency) to ensure a Stripe Product +
   993→ *  per-currency monthly+yearly Prices exist and writes the resolved IDs
   994→ *  into stripePriceIdsByCurrency + the default-currency snapshot fields.
   995→ */
   996→export async function savePlanConfig(planId: string, patch: PlanConfigInput): Promise<PlanConfigData | null> {
   997→  const existing = await db.planConfig.findUnique({ where: { planId } });
   998→  if (!existing) return null;
   999→  const current = rowToData(existing);
  1000→
  1001→  // When the admin omitted currency, resolve the platform default so the
  1002→  // legacy priceMonthly/priceYearly snapshot stays in the right currency.
  1003→  if (!patch.currency && !current.currency) {
  1004→    patch = { ...patch, currency: await resolvePlatformDefaultCurrency() };
  1005→  }
  1006→  const finalCurrency = (patch.currency ?? current.currency).toUpperCase();
  1007→
  1008→  // Distinguish "field omitted" from "field set". When the admin omits
  1009→  // both stripePriceId* AND stripePriceIdsByCurrency, they want the
  1010→  // backend to auto-sync. When either is explicitly provided, manual.
  1011→  const adminTouchedStripePriceIds =
  1012→    'stripePriceIdMonthly' in patch ||
  1013→    'stripePriceIdYearly' in patch ||
  1014→    'stripePriceIdsByCurrency' in patch;
  1015→
  1016→  const next = mergePlanPatch(current, patch, finalCurrency);
  1017→  await db.planConfig.update({ where: { planId }, data: dataToRow(next) });
  1018→  await hydrate();
  1019→
  1020→  // ---- AUTO-SYNC TO STRIPE (multi-currency) ----
  1021→  // Best-effort — never throws. Only ENABLED billing periods get
  1022→  // Stripe Prices (a disabled period never creates one).
  1023→  const hasAnyPaidCurrency = Object.values(next.pricesByCurrency).some(
  1024→    (p) =>
  1025→      (next.billingMonthly && p.monthly > 0) ||
  1026→      (next.billingYearly && p.yearly > 0),
  1027→  );
  1028→  if (!adminTouchedStripePriceIds && !next.isFree && hasAnyPaidCurrency) {
  1029→    try {
  1030→      const { isStripeConfiguredAsync, getStripeClient, syncPlanToStripeMulti } = await import('@/lib/stripe');
  1031→      if (await isStripeConfiguredAsync()) {
  1032→        const stripe = await getStripeClient();
  1033→        const syncResult = await syncPlanToStripeMulti(stripe, {
  1034→          planId: next.planId,
  1035→          name: next.name,
  1036→          defaultCurrency: finalCurrency,
  1037→          pricesByCurrency: next.pricesByCurrency,
  1038→          stripePriceIdsByCurrency: next.stripePriceIdsByCurrency,
  1039→          // Only the ENABLED billing periods get Stripe Prices.
  1040→          enabledIntervals: enabledIntervalsOf(next),
  1041→        });
  1042→        // Persist the resolved Stripe Price IDs back onto the row when they changed.
  1043→        const snapshot = syncResult.stripePriceIdsByCurrency[finalCurrency] ?? { monthly: null, yearly: null };
  1044→        const changed =
  1045→          JSON.stringify(syncResult.stripePriceIdsByCurrency) !==
  1046→            JSON.stringify(next.stripePriceIdsByCurrency) ||
  1047→          snapshot.monthly !== next.stripePriceIdMonthly ||
  1048→          snapshot.yearly !== next.stripePriceIdYearly;
  1049→        if (changed) {
  1050→          await db.planConfig.update({
  1051→            where: { planId },
  1052→            data: {
  1053→              stripePriceIdsByCurrency: JSON.stringify(syncResult.stripePriceIdsByCurrency),
  1054→              stripePriceIdMonthly: snapshot.monthly,
  1055→              stripePriceIdYearly: snapshot.yearly,
  1056→            },
  1057→          });
  1058→          await hydrate();
  1059→        }
  1060→      }
  1061→    } catch {
  1062→      // best-effort — never block the save on a Stripe error
  1063→    }
  1064→  }
  1065→
  1066→  return getPlanConfigSync(planId);
  1067→}

  1070→ *  configured AND the new plan is PAID AND the admin did NOT
  1071→ *  explicitly provide Stripe Price IDs in the input, the backend
  1072→ *  calls syncPlanToStripeMulti (multi-currency) to create a Stripe
  1073→ *  Product + per-currency monthly+yearly Prices and writes the
  1074→ *  resolved IDs back onto the new plan row.
  1075→ *
  1076→ *  The Create Plan modal sends the ONE base price configuration
  1077→ *  (priceMonthly + priceYearly + currency + autoCurrency); the default
  1078→ *  currency's pricesByCurrency entry is derived from the base price. */
  1079→export async function createPlanConfig(
  1080→  input: PlanConfigInput & { planId: string; name: string },

  1089→  // Base price precedence: explicit input → legacy mirror from an
  1090→  // explicit pricesByCurrency map → 0.

  1092→  const priceMonthly = input.priceMonthly ?? mirrored?.monthly ?? 0;

  1096→  const stripePriceIdsByCurrency: StripePriceIdsByCurrency = input.stripePriceIdsByCurrency ?? {};

  1185→/**
  1186→ * Delete a plan config. Refuses if:
  1187→ *  - it's the last remaining plan, OR
  1188→ *  - any Subscription still references this planId (would orphan the sub).
  1189→ *
  1190→ * Returns `{ ok: boolean, reason?: string }` describing the outcome.
  1191→ */
  1192→export async function deletePlanConfig(
  1193→  planId: string,
  1194→): Promise<{ ok: boolean; reason?: string }> {
  1195→  const count = await db.planConfig.count();
  1196→  if (count <= 1) return { ok: false, reason: 'Cannot delete the last plan.' };
  1197→  const subCount = await db.subscription.count({ where: { planId } });
  1198→  if (subCount > 0) {
  1199→    return {
  1200→      ok: false,
  1201→      reason: `Cannot delete plan "${planId}" — ${subCount} active subscription(s) reference it. Migrate them first.`,
  1202→    };
  1203→  }
  1204→  try {
  1205→    await db.planConfig.delete({ where: { planId } });
  1206→    await hydrate();
  1207→    return { ok: true };
  1208→  } catch (err) {
  1209→    return { ok: false, reason: (err as Error).message };
  1210→  }
  1211→}
