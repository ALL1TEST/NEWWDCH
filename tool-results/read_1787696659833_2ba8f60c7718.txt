   106→  setSelectedIds: (ids: string[]) => void;
   107→  setSearchValue: (value: string) => void;
   108→  toggleSort: (field: string) => void;

   110→  toggleAllSelection: (allIds: string[], currentDataIds: string[]) => void;
   111→  clearSelection: () => void;
   112→}
   113→
   114→export function useDataTable(options: UseDataTableOptions = {}): UseDataTableReturn {
   115→  const {
   116→    initialPage = 1,
   117→    initialPageSize = DEFAULT_PAGE_SIZE,
   118→    initialSortField = 'createdAt',
   119→    initialSortOrder = 'desc',
   120→  } = options;
   121→
   122→  const [currentPage, setCurrentPage] = useState(initialPage);
   123→  const [pageSize, setPageSize] = useState(initialPageSize);
   124→  const [sortField, setSortFieldRaw] = useState(initialSortField);
   125→  const [sortOrder, setSortOrderRaw] = useState<'asc' | 'desc'>(initialSortOrder);
   126→  const [selectedIds, setSelectedIds] = useState<string[]>([]);
   127→  const [searchValue, setSearchValue] = useState('');
   128→
   129→  const setSortField = useCallback((field: string, order: 'asc' | 'desc') => {
   130→    setSortFieldRaw(field);
   131→    setSortOrderRaw(order);
   132→    setCurrentPage(1);
   133→  }, []);
   134→
   135→  const toggleSort = useCallback(
   136→    (field: string) => {
   137→      if (sortField === field) {
   138→        const next = sortOrder === 'asc' ? 'desc' : 'asc';
   139→        setSortFieldRaw(field);
   140→        setSortOrderRaw(next);
   141→      } else {
   142→        setSortFieldRaw(field);
   143→        setSortOrderRaw('asc');
   144→      }
   145→      setCurrentPage(1);
   146→    },
   147→    [sortField, sortOrder],
   148→  );
   149→
   150→  const toggleRowSelection = useCallback((id: string) => {
   151→    setSelectedIds((prev) =>
   152→      prev.includes(id) ? prev.filter((i) => i !== id) : [...prev, id],
   153→    );
   154→  }, []);
   155→
   156→  const toggleAllSelection = useCallback(
   157→    (allIds: string[], currentDataIds: string[]) => {
   158→      const allSelected = currentDataIds.every((id) => selectedIds.includes(id));
   159→      if (allSelected) {
   160→        setSelectedIds((prev) =>
   161→          prev.filter((id) => !currentDataIds.includes(id)),
   162→        );
   163→      } else {
   164→        const newIds = currentDataIds.filter((id) => !selectedIds.includes(id));
   165→        setSelectedIds((prev) => [...prev, ...newIds]);
   166→      }
   167→    },
   168→    [selectedIds],
   169→  );
   170→
   171→  const clearSelection = useCallback(() => {
   172→    setSelectedIds([]);
   173→  }, []);
   174→
   175→  return {
   176→    currentPage,
   177→    pageSize,
   178→    sortField,
   179→    sortOrder,
   180→    selectedIds,
   181→    searchValue,
   182→    setCurrentPage,
   183→    setPageSize,
   184→    setSortField,
   185→    setSelectedIds,
   186→    setSearchValue,
   187→    toggleSort,
   188→    toggleRowSelection,
   189→    toggleAllSelection,
   190→    clearSelection,
   191→  };
   192→}

   232→export class ColumnDefHelper {
   233→  static textColumn<TData>(options: TextColumnOptions<TData>): ColumnDef<TData> {
   234→    return {
   235→      id: options.id,
   236→      accessorKey: options.accessorKey,
   237→      accessorFn: options.accessorFn,
   238→      header: ({ column }) => {
   239→        const canSort = options.enableSorting !== false;
   240→        if (!canSort) {
   241→          return <span className="font-medium">{options.header}</span>;
   242→        }
   243→        return (
   244→          <button
   245→            className="flex items-center gap-1 hover:text-foreground transition-colors"
   246→            onClick={() => column.toggleSorting(column.getIsSorted() === 'asc')}
   247→          >
   248→            <span className="font-medium">{options.header}</span>
   249→            <SortIcon sorted={column.getIsSorted()} />
   250→          </button>
   251→        );
   252→      },
   253→      size: options.size,
   254→      enableSorting: options.enableSorting !== false,
   255→      enableHiding: options.enableHiding,
   256→      cell: ({ getValue }) => {
   257→        const value = getValue() as string;
   258→        if (!value) return <span className="text-muted-foreground">—</span>;
   259→        if (options.truncate && value.length > options.truncate) {
   260→          return (
   261→            <span className={cn('truncate block max-w-full', options.className)} title={value}>
   262→              {value.slice(0, options.truncate)}...
   263→            </span>
   264→          );
   265→        }
   266→        return <span className={options.className}>{value}</span>;
   267→      },
   268→    };
   269→  }

   348→  static avatarColumn<TData>(options: AvatarColumnOptions<TData>): ColumnDef<TData> {
   349→    return {
   350→      id: options.id,
   351→      accessorKey: options.accessorKey,
   352→      accessorFn: options.accessorFn as any,
   353→      header: ({ column }) => {
   354→        const canSort = options.enableSorting !== false;
   355→        if (!canSort) {
   356→          return <span className="font-medium">{options.header}</span>;

   368→      size: (options.size ?? 200) as number,
   369→      enableSorting: options.enableSorting !== false,
   370→      enableHiding: options.enableHiding,

   372→        const rowData = row.original;
   373→        const src = options.getSrc?.(rowData);
   374→        const name = options.getName(rowData);

   379→          lg: 'h-12 w-12 text-base',
   380→        };
   381→        const avatarSize = sizeClasses[options.size ?? 'md'];
   382→        const initials = name
   383→          .split(/\s+/)
   384→          .map((w) => w[0])
   385→          .slice(0, 2)
   386→          .join('')
   387→          .toUpperCase();

   389→        return (
   390→          <div className="flex items-center gap-3">
   391→            {src ? (
   392→              <img
   393→                src={src}
   394→                alt={name}
   395→                className={cn('rounded-full object-cover shrink-0', avatarSize)}
   396→              />
   397→            ) : (
   398→              <div
   399→                className={cn(
   400→                  'rounded-full bg-muted flex items-center justify-center shrink-0 font-medium text-muted-foreground',
   401→                  avatarSize,
   402→                )}
   403→              >
   404→                {initials}
   405→              </div>
   406→            )}
   407→            <div className="min-w-0 flex-1">
   408→              <div className="truncate font-medium text-sm">{name}</div>
   409→              {subtitle && (
   410→                <div className="truncate text-xs text-muted-foreground">{subtitle}</div>
   411→              )}
   412→            </div>
   413→          </div>
   414→        );
   415→      },
   416→    };
   417→  }

   483→// -------------------- Pagination --------------------
   484→
   485→function DataTablePagination({
   486→  currentPage,
   487→  pageSize,
   488→  totalItems,
   489→  onPageChange,
   490→  onPageSizeChange,
   491→}: {
   492→  currentPage: number;
   493→  pageSize: number;
   494→  totalItems: number;
   495→  onPageChange: (page: number) => void;
   496→  onPageSizeChange: (size: number) => void;
   497→}) {
   498→  const totalPages = Math.max(1, Math.ceil(totalItems / pageSize));
   499→  const rangeStart = totalItems === 0 ? 0 : (currentPage - 1) * pageSize + 1;
   500→  const rangeEnd = Math.min(currentPage * pageSize, totalItems);
   501→
   502→  return (
   503→    <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between px-2 py-3 border-t">
   504→      <div className="text-xs text-muted-foreground">
   505→        Showing {rangeStart}–{rangeEnd} of {totalItems} items
   506→      </div>
   507→      <div className="flex items-center gap-3">
   508→        <div className="flex items-center gap-1.5">
   509→          <span className="text-xs text-muted-foreground whitespace-nowrap">Rows</span>
   510→          <Select
   511→            value={String(pageSize)}
   512→            onValueChange={(v) => onPageSizeChange(Number(v))}
   513→          >
   514→            <SelectTrigger size="sm" className="w-[70px] h-8">
   515→              <SelectValue />
   516→            </SelectTrigger>
   517→            <SelectContent>
   518→              {PAGE_SIZES.map((size) => (
   519→                <SelectItem key={size} value={String(size)}>
   520→                  {size}
   521→                </SelectItem>
   522→              ))}
   523→            </SelectContent>
   524→          </Select>
   525→        </div>
   526→        <div className="flex items-center gap-1">
   527→          <Button
   528→            variant="outline"
   529→            size="icon"
   530→            className="h-8 w-8"
   531→            onClick={() => onPageChange(1)}
   532→            disabled={currentPage <= 1}
   533→          >
   534→            <ChevronsLeft className="h-4 w-4" />
   535→            <span className="sr-only">First page</span>
   536→          </Button>
   537→          <Button
   538→            variant="outline"
   539→            size="icon"
   540→            className="h-8 w-8"
   541→            onClick={() => onPageChange(currentPage - 1)}
   542→            disabled={currentPage <= 1}
   543→          >
   544→            <ChevronLeft className="h-4 w-4" />
   545→            <span className="sr-only">Previous page</span>
   546→          </Button>
   547→          <span className="text-xs text-muted-foreground min-w-[60px] text-center">
   548→            {currentPage} / {totalPages}
   549→          </span>
   550→          <Button
   551→            variant="outline"
   552→            size="icon"
   553→            className="h-8 w-8"
   554→            onClick={() => onPageChange(currentPage + 1)}
   555→            disabled={currentPage >= totalPages}
   556→          >
   557→            <ChevronRight className="h-4 w-4" />
   558→            <span className="sr-only">Next page</span>
   559→          </Button>
   560→          <Button
   561→            variant="outline"
   562→            size="icon"
   563→            className="h-8 w-8"
   564→            onClick={() => onPageChange(totalPages)}
   565→            disabled={currentPage >= totalPages}
   566→          >
   567→            <ChevronsRight className="h-4 w-4" />
   568→            <span className="sr-only">Last page</span>
   569→          </Button>
   570→        </div>
   571→      </div>
   572→    </div>
   573→  );
   574→}

   578→function TableToolbar<TData, TValue>({
   579→  searchPlaceholder,
   580→  searchValue,
   581→  onSearch,
   582→  selectedIds,
   583→  bulkActions,
   584→  onBulkAction,
   585→  onSelectionChange,
   586→  filterContent,
   587→}: {
   588→  searchPlaceholder?: string;
   589→  searchValue?: string;
   590→  onSearch?: (value: string) => void;
   591→  selectedIds?: string[];
   592→  bulkActions?: BulkAction[];
   593→  onBulkAction?: (action: string, ids: string[]) => void;
   594→  onSelectionChange?: (ids: string[]) => void;
   595→  filterContent?: React.ReactNode;
   596→}) {
   597→  const hasSelection = selectedIds && selectedIds.length > 0;
   598→
   599→  return (
   600→    <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between p-4 pb-2">
   601→      <div className="flex items-center gap-2 flex-1">
   602→        {(onSearch || searchValue !== undefined) && (
   603→          <div className="relative max-w-sm w-full">
   604→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   605→            <Input
   606→              placeholder={searchPlaceholder ?? 'Search...'}
   607→              value={searchValue ?? ''}
   608→              onChange={(e) => onSearch?.(e.target.value)}
   609→              className="pl-9 h-9"
   610→            />
   611→          </div>
   612→        )}
   613→        {filterContent}
   614→      </div>
   615→      {hasSelection && bulkActions && bulkActions.length > 0 && (
   616→        <div className="flex items-center gap-2">
   617→          <span className="text-xs text-muted-foreground">
   618→            {selectedIds.length} selected
   619→          </span>
   620→          <DropdownMenu>
   621→            <DropdownMenuTrigger asChild>
   622→              <Button variant="outline" size="sm">
   623→                <MoreHorizontal className="h-4 w-4 mr-1" />
   624→                Bulk Actions
   625→              </Button>
   626→            </DropdownMenuTrigger>
   627→            <DropdownMenuContent align="end">
   628→              {bulkActions.map((action) => (
   629→                <DropdownMenuItem
   630→                  key={action.label}
   631→                  variant={action.variant}
   632→                  disabled={action.disabled}
   633→                  onClick={() => onBulkAction?.(action.label, selectedIds)}
   634→                >
   635→                  {action.icon}
   636→                  {action.label}
   637→                </DropdownMenuItem>
   638→              ))}
   639→            </DropdownMenuContent>
   640→          </DropdownMenu>
   641→          <Button
   642→            variant="ghost"
   643→            size="sm"
   644→            onClick={() => onSelectionChange?.([])}
   645→          >
   646→            Clear
   647→          </Button>
   648→        </div>
   649→      )}
   650→    </div>
   651→  );
   652→}

   654→// -------------------- Main DataTable Component --------------------
   655→
   656→export function DataTable<TData, TValue>({
   657→  columns,
   658→  data,
   659→  isLoading = false,
   660→  totalItems,
   661→  pageSize,
   662→  currentPage,
   663→  onPageChange,
   664→  onSortChange,
   665→  sortField,
   666→  sortOrder,
   667→  onRowClick,
   668→  selectedIds = [],
   669→  onSelectionChange,
   670→  onBulkAction,
   671→  bulkActions,
   672→  emptyMessage = 'No data found.',
   673→  emptyIcon,
   674→  searchPlaceholder,
   675→  onSearch,
   676→  searchValue,
   677→  filterContent,
   678→  getRowId,
   679→}: DataTableProps<TData, TValue>) {
   680→  const hasRowSelection = !!onSelectionChange;
   681→  const columnCount = columns.length + (hasRowSelection ? 1 : 0);
   682→
   683→  const enhancedColumns = useMemo<ColumnDef<TData, TValue>[]>(() => {
   684→    if (!hasRowSelection) return columns;
   685→
   686→    const selectColumn: ColumnDef<TData, TValue> = {
   687→      id: '__select__',
   688→      size: 40,
   689→      enableSorting: false,
   690→      enableHiding: false,
   691→      header: ({ table }) => {
   692→        const allRowIds = table.getRowModel().rows.map((r) => r.id);
   693→        const allChecked =
   694→          allRowIds.length > 0 && allRowIds.every((id) => selectedIds.includes(id));
   695→        const someChecked = allRowIds.some((id) => selectedIds.includes(id));
   696→
   697→        return (
   698→          <Checkbox
   699→            checked={allChecked ? true : someChecked ? 'indeterminate' : false}
   700→            onCheckedChange={(checked) => {
   701→              if (checked) {
   702→                const allIds = data
   703→                  .map((row, idx) => getRowId?.(row) ?? String(idx))
   704→                  .filter((id) => !selectedIds.includes(id));
   705→                onSelectionChange?.([...selectedIds, ...allIds]);
   706→              } else {
   707→                const currentDataIds = data.map((row, idx) =>
   708→                  getRowId?.(row) ?? String(idx),
   709→                );
   710→                onSelectionChange?.(
   711→                  selectedIds.filter((id) => !currentDataIds.includes(id)),
   712→                );
   713→              }
   714→            }}
   715→            aria-label="Select all rows"
   716→          />
   717→        );
   718→      },
   719→      cell: ({ row }) => {
   720→        const rowId = row.id;
   721→        const isChecked = selectedIds.includes(rowId);
   722→
   723→        return (
   724→          <Checkbox
   725→            checked={isChecked}
   726→            onCheckedChange={(checked) => {
   727→              if (checked) {
   728→                onSelectionChange?.([...selectedIds, rowId]);
   729→              } else {
   730→                onSelectionChange?.(selectedIds.filter((id) => id !== rowId));
   731→              }
   732→            }}
   733→            onClick={(e) => e.stopPropagation()}
   734→            aria-label={`Select row ${rowId}`}
   735→          />
   736→        );
   737→      },
   738→    };
   739→
   740→    return [selectColumn, ...columns];
   741→  }, [columns, hasRowSelection, selectedIds, onSelectionChange, data, getRowId]);
   742→
   743→  const sorting: ColumnSort[] = useMemo(() => {
   744→    if (!sortField) return [];
   745→    return [{ id: sortField, desc: sortOrder === 'desc' }];
   746→  }, [sortField, sortOrder]);
   747→
   748→  const table = useReactTable({
   749→    data,
   750→    columns: enhancedColumns,
   751→    getCoreRowModel: getCoreRowModel(),
   752→    getSortedRowModel: getSortedRowModel(),
   753→    state: {
   754→      sorting,
   755→    },
   756→    onSortingChange: (updater) => {
   757→      const newSorting =
   758→        typeof updater === 'function' ? updater(sorting) : updater;
   759→      if (newSorting.length > 0) {
   760→        onSortChange(newSorting[0].id, newSorting[0].desc ? 'desc' : 'asc');
   761→      }
   762→    },
   763→    manualPagination: true,
   764→    manualSorting: true,
   765→    getRowId: getRowId
   766→      ? (row, index) => getRowId(row) ?? String(index)
   767→      : (_, index) => String(index),
   768→  });
   769→
   770→  const handlePageSizeChange = useCallback(
   771→    (newSize: number) => {
   772→      onPageChange(1);
   773→    },
   774→    [onPageChange],
   775→  );
   776→
   777→  const showToolbar =
   778→    (onSearch || searchValue !== undefined) ||
   779→    (filterContent != null) ||
   780→    (selectedIds.length > 0 && bulkActions && bulkActions.length > 0);
   781→
   782→  return (
   783→    <div className="space-y-0">
   784→      {showToolbar && (
   785→        <TableToolbar
   786→          searchPlaceholder={searchPlaceholder}
   787→          searchValue={searchValue}
   788→          onSearch={onSearch}
   789→          selectedIds={selectedIds}
   790→          bulkActions={bulkActions}
   791→          onBulkAction={onBulkAction}
   792→          onSelectionChange={onSelectionChange}
   793→          filterContent={filterContent}
   794→        />
   795→      )}
   796→      <div className="rounded-lg border">
   797→        <Table>
   798→          <TableHeader>
   799→            {table.getHeaderGroups().map((headerGroup) => (
   800→              <TableRow key={headerGroup.id}>
   801→                {headerGroup.headers.map((header) => (
   802→                  <TableHead
   803→                    key={header.id}
   804→                    style={{ width: header.getSize() !== 150 ? header.getSize() : undefined }}
   805→                  >
   806→                    {flexRender(
   807→                      header.column.columnDef.header,
   808→                      header.getContext(),
   809→                    )}
   810→                  </TableHead>
   811→                ))}
   812→              </TableRow>
   813→            ))}
   814→          </TableHeader>
   815→          <TableBody>
   816→            {isLoading ? (
   817→              <SkeletonRows columnCount={columnCount} />
   818→            ) : data.length === 0 ? (
   819→              <DataTableEmpty icon={emptyIcon} message={emptyMessage} />
   820→            ) : (
   821→              table.getRowModel().rows.map((row) => (
   822→                <TableRow
   823→                  key={row.id}
   824→                  data-state={selectedIds.includes(row.id) ? 'selected' : undefined}
   825→                  className={cn(onRowClick && 'cursor-pointer')}
   826→                  onClick={() => onRowClick?.(row.original)}
   827→                >
   828→                  {row.getVisibleCells().map((cell) => (
   829→                    <TableCell key={cell.id}>
   830→                      {flexRender(
   831→                        cell.column.columnDef.cell,
   832→                        cell.getContext(),
   833→                      )}
   834→                    </TableCell>
   835→                  ))}
   836→                </TableRow>
   837→              ))
   838→            )}
   839→          </TableBody>
   840→        </Table>
   841→      </div>
   842→      <DataTablePagination
   843→        currentPage={currentPage}
   844→        pageSize={pageSize}
   845→        totalItems={totalItems}
   846→        onPageChange={onPageChange}
   847→        onPageSizeChange={handlePageSizeChange}
   848→      />
   849→    </div>
   850→  );
   851→}
