   433→// ============================================================
   434→// CHANGE EMAIL DIALOG
   435→// ------------------------------------------------------------
   436→// Modal with:
   437→//   • Current email (read-only display)
   438→//   • New email input (validated client + server-side)
   439→//   • Current password input (server-verified)
   440→// Server uniqueness check at /api/auth/change-email.
   441→// On success: toast + update auth store + close.
   442→// ============================================================
   443→
   444→function ChangeEmailDialog({
   445→  open,
   446→  onOpenChange,
   447→  currentEmail,
   448→  onEmailChanged,
   449→}: {
   450→  open: boolean;
   451→  onOpenChange: (v: boolean) => void;
   452→  currentEmail: string;
   453→  onEmailChanged: (newEmail: string) => void;
   454→}) {
   455→  const [newEmail, setNewEmail] = useState('');
   456→  const [currentPassword, setCurrentPassword] = useState('');
   457→  const [showPw, setShowPw] = useState(false);
   458→  const [isSaving, setIsSaving] = useState(false);
   459→  const [localError, setLocalError] = useState<string | null>(null);
   460→
   461→  useEffect(() => {
   462→    if (!open) {
   463→      setNewEmail('');
   464→      setCurrentPassword('');
   465→      setShowPw(false);
   466→      setLocalError(null);
   467→    }
   468→  }, [open]);
   469→
   470→  const emailValid = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(newEmail.trim());
   471→  const isDifferent = newEmail.trim().toLowerCase() !== currentEmail.toLowerCase();
   472→
   473→  const handleSubmit = async () => {
   474→    setLocalError(null);
   475→    if (!newEmail.trim()) {
   476→      setLocalError('Please enter a new email address.');
   477→      return;
   478→    }
   479→    if (!emailValid) {
   480→      setLocalError('Please enter a valid email address.');
   481→      return;
   482→    }
   483→    if (!isDifferent) {
   484→      setLocalError('New email must be different from your current email.');
   485→      return;
   486→    }
   487→    if (!currentPassword) {
   488→      setLocalError('Please enter your current password to confirm.');
   489→      return;
   490→    }
   491→
   492→    setIsSaving(true);
   493→    try {
   494→      const res = await postApi<{ email: string }>('/api/auth/change-email', {
   495→        currentPassword,
   496→        newEmail: newEmail.trim(),
   497→      });
   498→      onEmailChanged(res.email);
   499→      toast.success('Email address updated successfully.');
   500→      onOpenChange(false);
   501→    } catch (err) {
   502→      const msg = err instanceof Error ? err.message : 'Failed to change email.';
   503→      setLocalError(msg);
   504→    } finally {
   505→      setIsSaving(false);
   506→    }
   507→  };
   508→
   509→  return (
   510→    <Dialog open={open} onOpenChange={(v) => { if (!isSaving) onOpenChange(v); }}>
   511→      <DialogContent className="sm:max-w-md">
   512→        <DialogHeader>
   513→          <DialogTitle className="flex items-center gap-2">
   514→            <Mail className="h-5 w-5" />
   515→            Change Email Address
   516→          </DialogTitle>
   517→          <DialogDescription>
   518→            Enter your new email and current password to confirm the change.
   519→          </DialogDescription>
   520→        </DialogHeader>
   521→        <div className="py-2 space-y-4">
   522→          <div className="grid gap-2">
   523→            <Label className="text-xs text-muted-foreground">Current Email</Label>
   524→            <div className="rounded-md border bg-muted/50 px-3 py-2 text-sm text-muted-foreground">
   525→              {currentEmail}
   526→            </div>
   527→          </div>
   528→          <div className="grid gap-2">
   529→            <Label htmlFor="new-email" className="text-xs">New Email Address</Label>
   530→            <div className="relative">
   531→              <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   532→              <Input
   533→                id="new-email"
   534→                type="email"
   535→                autoComplete="email"
   536→                value={newEmail}
   537→                onChange={(e) => setNewEmail(e.target.value)}
   538→                placeholder="you@example.com"
   539→                className="pl-9"
   540→              />
   541→            </div>
   542→          </div>
   543→          <div className="grid gap-2">
   544→            <Label htmlFor="ce-current-pw" className="text-xs">Current Password</Label>
   545→            <div className="relative">
   546→              <Input
   547→                id="ce-current-pw"
   548→                type={showPw ? 'text' : 'password'}
   549→                autoComplete="current-password"
   550→                value={currentPassword}
   551→                onChange={(e) => setCurrentPassword(e.target.value)}
   552→                placeholder="Enter current password"
   553→              />
   554→              <button
   555→                type="button"
   556→                className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   557→                onClick={() => setShowPw(!showPw)}
   558→              >
   559→                {showPw ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   560→              </button>
   561→            </div>
   562→          </div>
   563→          {localError && (
   564→            <p className="text-xs text-destructive flex items-start gap-1.5">
