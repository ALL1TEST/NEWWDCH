     1→'use client';
     2→

     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
     5→import { useForm, Controller } from 'react-hook-form';
     6→import { zodResolver } from '@hookform/resolvers/zod';
     7→import { z } from 'zod/v4';
     8→import {
     9→  ArrowLeft, ChevronDown, Sparkles, Send, Image as ImageIcon,
    10→  Clock, Search, X, Upload, FileText, Loader2, Trash2, Save, CircleCheck, AlertTriangle,
    11→  Lightbulb, List, FolderOpen, AlignLeft, PenTool,
    12→} from 'lucide-react';
    13→import { Button } from '@/components/ui/button';
    14→import { Input } from '@/components/ui/input';
    15→import { Textarea } from '@/components/ui/textarea';
    16→import { Label } from '@/components/ui/label';
    17→import { Badge } from '@/components/ui/badge';
    18→import { Skeleton } from '@/components/ui/skeleton';
    19→import { Progress } from '@/components/ui/progress';
    20→import {
    21→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    22→} from '@/components/ui/select';
    23→import { Collapsible, CollapsibleTrigger, CollapsibleContent } from '@/components/ui/collapsible';
    24→import { Separator } from '@/components/ui/separator';
    25→import {
    26→  Tooltip, TooltipTrigger, TooltipContent,
    27→} from '@/components/ui/tooltip';
    28→import { ConfirmDialog } from '@/components/patterns';
    29→import { TiptapEditor, type TiptapEditorRef } from '@/components/editor/tiptap-editor';
    30→import { getApi, postApi, patchApi, deleteApi } from '@/lib/api-client';
    31→import { queryKeys } from '@/lib/query-keys';
    32→import { useNavigationStore } from '@/lib/stores/navigation-store';
    33→import { slugify, cn, normalizeContentToHtml } from '@/lib/utils';
    34→import { toast } from 'sonner';

    36→
    37→// -------------------- Types --------------------
    38→
    39→interface ContentAuthor { id: string; name: string; avatar?: string; }
    40→interface ContentTypeOption { id: string; name: string; }
    41→interface CategoryOption { id: string; name: string; }
    42→interface TagOption { id: string; name: string; }
    43→
    44→interface ContentDetail {
    45→  id: string; title: string; slug: string; status: PostStatus; version: number;
    46→  content?: string; excerpt?: string; contentTypeId: string; contentType: ContentTypeOption;
    47→  categoryId?: string; category?: CategoryOption | null;
    48→  seoTitle?: string; seoDescription?: string; focusKeyword?: string;
    49→  tags?: TagOption[]; author: ContentAuthor; createdAt: string; updatedAt: string;
    50→}
    51→
    52→// -------------------- Form Schema --------------------
    53→
    54→const contentEditSchema = z.object({
    55→  title: z.string().min(1, 'Title is required').max(200, 'Title must be 200 characters or less'),
    56→  excerpt: z.string().max(300).optional().or(z.literal('')),
    57→  content: z.string().optional().or(z.literal('')),
    58→  status: z.enum(['DRAFT', 'IN_REVIEW', 'APPROVED', 'PUBLISHED', 'UNPUBLISHED', 'ARCHIVED']),
    59→  contentTypeId: z.string().min(1, 'Content type is required'),
    60→  categoryId: z.string().optional().or(z.literal('')),
    61→  tagIds: z.array(z.string()),
    62→  seoTitle: z.string().max(60).optional().or(z.literal('')),
    63→  seoDescription: z.string().max(160).optional().or(z.literal('')),
    64→});
    65→
    66→type ContentEditFormValues = z.infer<typeof contentEditSchema>;
    67→
    68→type SaveState = 'idle' | 'saving' | 'saved' | 'conflict';
    69→
    70→const AI_QUICK_ACTIONS = ['Duplicate', 'Make shorter', 'Fix grammar', 'Professional tone', 'Add conclusion', 'Generate FAQ', 'Generate outline'];
    71→
    72→// -------------------- Component --------------------
    73→
    74→export function ContentEditPage({ contentId }: { contentId: string }) {
    75→  const navigate = useNavigationStore((s) => s.navigate);
    76→  const queryClient = useQueryClient();
    77→  const editorRef = useRef<TiptapEditorRef>(null);
    78→  const [tagSearch, setTagSearch] = useState('');
    79→  const [saveState, setSaveState] = useState<SaveState>('idle');
    80→  const [showDeleteDialog, setShowDeleteDialog] = useState(false);
    81→  const [slugValue, setSlugValue] = useState('');
    82→  const [aiInput, setAiInput] = useState('');
    83→
    84→  const { data: content, isLoading: isLoadingContent } = useQuery({
    85→    queryKey: queryKeys.content.detail(contentId),
    86→    queryFn: () => getApi<ContentDetail>(`/api/content/${contentId}?include=contentType,category,tags,author`),
    87→    staleTime: 5_000, enabled: !!contentId,
    88→  });
    89→
    90→  const { data: contentTypes } = useQuery({
    91→    queryKey: queryKeys.contentTypes.all,
    92→    queryFn: () => getApi<ContentTypeOption[]>('/api/content-types?pageSize=100'), staleTime: 60_000,
    93→  });
    94→  const { data: categories } = useQuery({
    95→    queryKey: queryKeys.categories.all,
    96→    queryFn: () => getApi<CategoryOption[]>('/api/categories?pageSize=200'), staleTime: 60_000,
    97→  });
    98→  const { data: allTags } = useQuery({
    99→    queryKey: queryKeys.tags.all,
   100→    queryFn: () => getApi<TagOption[]>('/api/tags?pageSize=200'), staleTime: 60_000,
   101→  });
   102→
   103→  const { register, handleSubmit, control, watch, reset, setValue, formState: { errors, isDirty } } = useForm<ContentEditFormValues>({
   104→    resolver: zodResolver(contentEditSchema),
   105→    defaultValues: { title: '', excerpt: '', content: '', status: 'DRAFT', contentTypeId: '', categoryId: '', tagIds: [], seoTitle: '', seoDescription: '' },
   106→  });
   107→
   108→  // Normalize content to HTML (handles legacy ProseMirror JSON stored in DB)
   109→  const normalizedContent = useMemo(
   110→    () => (content?.content ? normalizeContentToHtml(content.content) : ''),
   111→    [content?.content],
   112→  );
   113→
   114→  useEffect(() => {
   115→    if (content) {
   116→      reset({
   117→        title: content.title ?? '',
   118→        excerpt: content.excerpt ?? '',
   119→        content: normalizedContent,
   120→        status: content.status,
   121→        contentTypeId: content.contentTypeId ?? '',
   122→        categoryId: content.categoryId ?? '',
   123→        tagIds: content.tags?.map((t) => t.id) ?? [],
   124→        seoTitle: content.seoTitle ?? '',
   125→        seoDescription: content.seoDescription ?? '',
   126→      });
   127→      setSlugValue(content.slug ?? '');
   128→    }
   129→  }, [content, normalizedContent, reset]);
   130→
   131→  const selectedTagIds = watch('tagIds');
   132→  const watchedTitle = watch('title');
   133→  const watchedContent = watch('content');
   134→  const watchedSeoTitle = watch('seoTitle');
   135→  const watchedSeoDescription = watch('seoDescription');
   136→
   137→  const filteredTags = useMemo(() => {
   138→    if (!allTags) return [];
   139→    if (!tagSearch) return allTags;
   140→    return allTags.filter((t) => t.name.toLowerCase().includes(tagSearch.toLowerCase()));
   141→  }, [allTags, tagSearch]);
   142→
   143→  // Word count from plain text extraction
   144→  const wordCount = useMemo(() => {
   145→    if (!watchedContent) return 0;
   146→    const tmp = document.createElement('div');
   147→    tmp.innerHTML = watchedContent;
   148→    const text = tmp.textContent || tmp.innerText || '';
   149→    return text.trim() ? text.trim().split(/\s+/).length : 0;
   150→  }, [watchedContent]);
   151→  const readingTime = useMemo(() => { const m = Math.ceil(wordCount / 200); return m < 1 ? '< 1 min' : `${m} min`; }, [wordCount]);
   152→
   153→  const seoScore = useMemo(() => {
   154→    let s = 0;
   155→    if (watchedTitle.length > 0) s += 20; if (watchedTitle.length <= 60) s += 10;
   156→    if (watchedSeoTitle.length > 0) s += 20; if (watchedSeoTitle.length <= 60) s += 10;
   157→    if (watchedSeoDescription.length > 0) s += 20; if (watchedSeoDescription.length <= 160) s += 10;
   158→    if (watchedContent.length > 0) s += 10;
   159→    return s;
   160→  }, [watchedTitle, watchedSeoTitle, watchedSeoDescription, watchedContent]);
   161→
   162→  const seoColor = seoScore >= 80 ? 'text-emerald-600' : seoScore >= 50 ? 'text-amber-600' : 'text-red-600';
   163→  const seoBar = seoScore >= 80 ? '[&>div]:bg-emerald-500' : seoScore >= 50 ? '[&>div]:bg-amber-500' : '[&>div]:bg-red-500';
   164→
   165→  const saveMutation = useMutation({
   166→    mutationFn: (data: ContentEditFormValues) => {
   167→      setSaveState('saving');
   168→      return patchApi<ContentDetail>(`/api/content/${contentId}`, {
   169→        title: data.title, excerpt: data.excerpt || undefined, content: data.content || undefined,
   170→        status: data.status, contentTypeId: data.contentTypeId, categoryId: data.categoryId || undefined,
   171→        tagIds: data.tagIds, seoTitle: data.seoTitle || undefined, seoDescription: data.seoDescription || undefined,
   172→      });
   173→    },
   174→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: queryKeys.content.all }); queryClient.invalidateQueries({ queryKey: queryKeys.content.detail(contentId) }); setSaveState('saved'); toast.success('Content saved'); setTimeout(() => setSaveState('idle'), 2000); },
   175→    onError: () => { setSaveState('conflict'); toast.error('Failed to save content'); setTimeout(() => setSaveState('idle'), 5000); },
   176→  });
   177→
   178→  const deleteMutation = useMutation({
   179→    mutationFn: () => deleteApi(`/api/content/${contentId}`),
   180→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: queryKeys.content.all }); toast.success('Content deleted'); navigate('content'); },
   181→    onError: () => toast.error('Failed to delete content'),
   182→  });
   183→
   184→  // AI edit selected text mutation
   185→  const aiEditSelectionMutation = useMutation({
   186→    mutationFn: ({ text, action }: { text: string; action: string }) =>
   187→      postApi<{ data: { editedText: string } }>('/api/content/ai-edit-selection', { text, action }),
   188→    onSuccess: (result) => {
   189→      const editedText = result.data?.editedText;
   190→      if (editedText) {
   191→        editorRef.current?.replaceSelection(editedText);
   192→        toast.success('Text updated');
   193→      }
   194→    },
   195→    onError: (err: Error) => toast.error(err.message || 'AI edit failed'),
   196→  });
   197→
   198→  // Selection-aware action handler
   199→  const handleQuickAction = useCallback((action: string) => {
   200→    const selectedText = editorRef.current?.saveSelectionForReplace() || '';
   201→
   202→    if (action === 'Duplicate') {
   203→      if (selectedText) {
   204→        editorRef.current?.insertAfterSelection(selectedText);
   205→        toast.success('Text duplicated');
   206→      } else {
   207→        toast.error('Select text to duplicate');
   208→      }
   209→      return;
   210→    }
   211→
   212→    if (selectedText) {
   213→      aiEditSelectionMutation.mutate({ text: selectedText, action });
   214→    } else {
   215→      toast.error('Select text first to apply this action');
   216→    }
   217→  }, [aiEditSelectionMutation]);
   218→
   219→  const handleAiSubmit = useCallback((prompt: string) => {
   220→    const selectedText = editorRef.current?.saveSelectionForReplace() || '';
   221→    if (selectedText) {
   222→      aiEditSelectionMutation.mutate({ text: selectedText, action: prompt });
   223→    } else {
   224→      toast.error('Select text first to edit with AI');
   225→    }
   226→  }, [aiEditSelectionMutation]);
   227→
   228→  const onSubmit = useCallback((data: ContentEditFormValues) => saveMutation.mutate(data), [saveMutation]);
   229→  const goBack = useCallback(() => navigate('content'), [navigate]);
   230→  const addTag = useCallback((tagId: string) => { if (!selectedTagIds.includes(tagId)) { setValue('tagIds', [...selectedTagIds, tagId], { shouldValidate: true }); } setTagSearch(''); }, [selectedTagIds, setValue]);
   231→  const removeTag = useCallback((tagId: string) => { setValue('tagIds', selectedTagIds.filter((id) => id !== tagId), { shouldValidate: true }); }, [selectedTagIds, setValue]);
   232→
   233→  const submitWithStatus = useCallback((status: string) => {
   234→    handleSubmit((data) => saveMutation.mutate({ ...data, status: status as ContentEditFormValues['status'] }))();
   235→  }, [handleSubmit, saveMutation]);
   236→
   237→  const handleEditorChange = useCallback((html: string) => {
   238→    setValue('content', html, { shouldValidate: false });
   239→  }, [setValue]);
   240→
   241→  const saveIndicator = useMemo(() => {
   242→    switch (saveState) {
   243→      case 'saving': return <span className="inline-flex items-center gap-1.5 text-xs text-muted-foreground"><Loader2 className="h-3 w-3 animate-spin" />Saving...</span>;
   244→      case 'saved': return <span className="inline-flex items-center gap-1.5 text-xs text-emerald-600"><CircleCheck className="h-3 w-3" />Saved</span>;
   245→      case 'conflict': return <span className="inline-flex items-center gap-1.5 text-xs text-amber-600"><AlertTriangle className="h-3 w-3" />Conflict</span>;
   246→      default: return isDirty ? <span className="inline-flex items-center gap-1.5 text-xs text-amber-500">Unsaved changes</span> : null;
   247→    }
   248→  }, [saveState, isDirty]);
   249→
   250→  // ---- Sidebar Card ----
   251→  const SidebarCard = ({
   252→    icon, title, defaultOpen = false, children,
   253→  }: { icon: React.ReactNode; title: string; defaultOpen?: boolean; children: React.ReactNode }) => (
   254→    <Collapsible defaultOpen={defaultOpen}>
   255→      <CollapsibleTrigger className="flex items-center justify-between w-full px-4 py-3 hover:bg-muted/30 transition-colors cursor-pointer group">
   256→        <div className="flex items-center gap-2.5">
   257→          <span className="text-muted-foreground">{icon}</span>
   258→          <span className="text-sm font-semibold">{title}</span>
   259→        </div>
   260→        <ChevronDown className="h-4 w-4 text-muted-foreground transition-transform [[data-state=open]>&]:rotate-180" />
   261→      </CollapsibleTrigger>
   262→      <CollapsibleContent>
   263→        <div className="px-4 pb-4 space-y-3">{children}</div>
   264→      </CollapsibleContent>
   265→    </Collapsible>
   266→  );
   267→
   268→  if (isLoadingContent) {
   269→    return (
   270→      <div className="flex flex-col h-[calc(100vh-8rem)]">
   271→        <div className="flex items-center justify-between border-b px-6 py-3"><Skeleton className="h-6 w-48" /><div className="flex gap-2"><Skeleton className="h-8 w-24" /><Skeleton className="h-8 w-20" /><Skeleton className="h-8 w-20" /></div></div>
   272→        <div className="flex flex-1 min-h-0"><div className="flex-1 p-6 space-y-4"><Skeleton className="h-10 w-3/4" /><Skeleton className="h-24 w-full" /><Skeleton className="h-96 w-full" /></div><div className="w-80 border-l p-4 space-y-4"><Skeleton className="h-40 w-full" /><Skeleton className="h-32 w-full" /></div></div>
   273→      </div>
   274→    );
   275→  }
   276→
   277→  return (
   278→    <form onSubmit={handleSubmit(onSubmit)}>
   279→      <div className="flex flex-col h-[calc(100vh-8rem)]">
   280→        {/* ========== TOP BAR ========== */}
   281→        <div className="flex items-center justify-between bg-background border-b border-border/50 px-4 py-2.5 shrink-0">
   282→          <div className="flex items-center gap-3">
   283→            <Tooltip>
   284→              <TooltipTrigger asChild>
   285→                <Button type="button" variant="ghost" size="icon" className="h-8 w-8" onClick={goBack}>
   286→                  <ArrowLeft className="h-4 w-4" />
   287→                </Button>
   288→              </TooltipTrigger>
   289→              <TooltipContent side="bottom" sideOffset={4}>Go back</TooltipContent>
   290→            </Tooltip>
   291→            <h1 className="text-base font-semibold truncate max-w-[280px] lg:max-w-[400px]">{content?.title ?? 'Edit Content'}</h1>
   292→            {content && <Badge variant="outline" className="font-mono text-[10px] px-1.5 py-0">v{content.version}</Badge>}
   293→            {saveIndicator}
   294→          </div>
   295→          <div className="flex items-center gap-2">
   296→            <Button type="button" variant="outline" size="sm" className="h-8" onClick={() => submitWithStatus('DRAFT')} disabled={saveMutation.isPending}>
   297→              {saveMutation.isPending && <Loader2 className="h-3.5 w-3.5 mr-1.5 animate-spin" />}
   298→              {!saveMutation.isPending && <Save className="h-3.5 w-3.5 mr-1.5" />}Save Draft
   299→            </Button>
   300→            <Button type="button" size="sm" className="h-8" onClick={() => submitWithStatus('PUBLISHED')} disabled={saveMutation.isPending}>
   301→              Publish
   302→            </Button>
   303→            <Tooltip>
   304→              <TooltipTrigger asChild>
   305→                <Button type="button" variant="ghost" size="icon" className="h-8 w-8 text-destructive hover:text-destructive" onClick={() => setShowDeleteDialog(true)}>
   306→                  <Trash2 className="h-4 w-4" />
   307→                </Button>
   308→              </TooltipTrigger>
   309→              <TooltipContent side="bottom" sideOffset={4}>Delete</TooltipContent>
   310→            </Tooltip>
   311→          </div>
   312→        </div>
   313→
   314→        {/* ========== MAIN AREA ========== */}
   315→        <div className="flex flex-1 min-h-0 overflow-hidden">
   316→          {/* Editor Column */}
   317→          <div className="flex-1 flex flex-col min-w-0 min-h-0">
   318→            {/* Title + Excerpt above the rich editor */}
   319→            <div className="px-6 md:px-12 lg:px-16 pt-6 pb-2 max-w-4xl mx-auto w-full shrink-0">
   320→              <input
   321→                {...register('title')}
   322→                placeholder="Untitled Article"
   323→                className="w-full bg-transparent text-3xl md:text-4xl font-bold placeholder:text-muted-foreground/40 focus:outline-none border-0 pb-2"
   324→              />
   325→              {errors.title && <p className="text-xs text-destructive -mt-1 mb-2">{errors.title.message}</p>}
   326→              <Textarea
   327→                {...register('excerpt')}
   328→                placeholder="Write a brief excerpt..."
   329→                rows={2}
   330→                className="w-full bg-transparent resize-none text-base text-muted-foreground italic placeholder:text-muted-foreground/30 focus-visible:ring-0 border-0 pb-2"
   331→              />
   332→            </div>
   333→
   334→            {/* ===== TIPTAP RICH TEXT EDITOR ===== */}
   335→            <div className="flex-1 min-h-0">
   336→              <TiptapEditor
   337→                ref={editorRef}
   338→                content={watchedContent || ''}
   339→                onChange={handleEditorChange}
   340→                placeholder=""
   341→              />
   342→            </div>
   343→          </div>
   344→
   345→          {/* ========== RIGHT SIDEBAR ========== */}
   346→          <aside className="hidden md:flex flex-col w-[340px] border-l border-border/50 bg-muted/20 shrink-0 overflow-y-auto">
   347→            <div className="space-y-0.5">
   348→              {/* Featured Image */}
   349→              <SidebarCard icon={<ImageIcon className="h-4 w-4" />} title="Featured Image" defaultOpen>
   350→                <div className="border-2 border-dashed border-border/60 rounded-lg p-8 flex flex-col items-center justify-center gap-2 cursor-pointer hover:border-primary/50 hover:bg-muted/20 transition-colors">
   351→                  <div className="h-12 w-12 rounded-full bg-muted/50 flex items-center justify-center">
   352→                    <ImageIcon className="h-6 w-6 text-muted-foreground/50" />
   353→                  </div>
   354→                  <p className="text-sm text-muted-foreground mt-1">Click to upload or drag and drop</p>
   355→                  <p className="text-[11px] text-muted-foreground/60">PNG, JPG, WebP up to 10MB</p>
   356→                </div>
   357→                <div className="flex gap-2">
   358→                  <Button type="button" variant="outline" size="sm" className="flex-1 h-8 text-xs"><Upload className="h-3.5 w-3.5 mr-1.5" />Upload</Button>
   359→                  <Button type="button" variant="outline" size="sm" className="flex-1 h-8 text-xs"><FolderOpen className="h-3.5 w-3.5 mr-1.5" />Library</Button>
   360→                  <Button type="button" variant="outline" size="sm" className="flex-1 h-8 text-xs"><Sparkles className="h-3.5 w-3.5 mr-1.5" />AI</Button>
   361→                </div>
   362→              </SidebarCard>
   363→
   364→              <Separator className="my-0.5" />
   365→
   366→              {/* Publishing */}
   367→              <SidebarCard icon={<Clock className="h-4 w-4" />} title="Publishing" defaultOpen>
   368→                <div className="space-y-1.5">
   369→                  <Label className="text-xs">Status</Label>
   370→                  <Controller control={control} name="status" render={({ field }) => (
   371→                    <Select value={field.value} onValueChange={field.onChange}>
   372→                      <SelectTrigger className="h-9"><SelectValue /></SelectTrigger>
   373→                      <SelectContent>
   374→                        <SelectItem value="DRAFT">Draft</SelectItem>
   375→                        <SelectItem value="IN_REVIEW">In Review</SelectItem>
   376→                        <SelectItem value="APPROVED">Approved</SelectItem>
   377→                        <SelectItem value="PUBLISHED">Published</SelectItem>
   378→                        <SelectItem value="UNPUBLISHED">Unpublished</SelectItem>
   379→                        <SelectItem value="ARCHIVED">Archived</SelectItem>
   380→                      </SelectContent>
   381→                    </Select>
   382→                  )} />
   383→                </div>
   384→                <div className="space-y-1.5">
   385→                  <Label className="text-xs">Content Type <span className="text-destructive">*</span></Label>
   386→                  <Controller control={control} name="contentTypeId" render={({ field }) => (
   387→                    <Select value={field.value} onValueChange={field.onChange}>
   388→                      <SelectTrigger className="h-9"><SelectValue placeholder="Select type" /></SelectTrigger>
   389→                      <SelectContent>{(contentTypes ?? []).map((ct) => <SelectItem key={ct.id} value={ct.id}>{ct.name}</SelectItem>)}</SelectContent>
   390→                    </Select>
   391→                  )} />
   392→                  {errors.contentTypeId && <p className="text-xs text-destructive">{errors.contentTypeId.message}</p>}
   393→                </div>
   394→                <div className="space-y-1.5">
   395→                  <Label className="text-xs">Category</Label>
   396→                  <Controller control={control} name="categoryId" render={({ field }) => (
   397→                    <Select value={field.value ?? ''} onValueChange={field.onChange}>
   398→                      <SelectTrigger className="h-9"><SelectValue placeholder="Select category" /></SelectTrigger>
   399→                      <SelectContent>{(categories ?? []).map((cat) => <SelectItem key={cat.id} value={cat.id}>{cat.name}</SelectItem>)}</SelectContent>
   400→                    </Select>
   401→                  )} />
   402→                </div>
   403→              </SidebarCard>
   404→
   405→              <Separator className="my-0.5" />
   406→
   407→              {/* Title & Slug */}
   408→              <SidebarCard icon={<FileText className="h-4 w-4" />} title="Title & Slug">
   409→                <div className="space-y-1.5">
   410→                  <Label className="text-xs">Title</Label>
   411→                  <Input {...register('title')} placeholder="Page title" className="h-9 text-sm" />
   412→                </div>
   413→                <div className="space-y-1.5">
   414→                  <Label className="text-xs">Slug</Label>
   415→                  <div className="flex items-center gap-1.5">
   416→                    <Input value={slugValue} onChange={(e) => setSlugValue(e.target.value)} placeholder="post-slug" className="h-9 text-sm font-mono" />
   417→                    <Tooltip>
