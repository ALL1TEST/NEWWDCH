     1→// ============================================================
     2→// PLAN CONFIG — DB-backed, cached single source of truth for plans,
     3→// pricing, entitlements and usage limits.
     4→// ============================================================
     5→// The Client Billing API and the Platform Admin both read from (and
     6→// the admin mutates) this same service. `platform-data.ts` delegates
     7→// its `PLANS` / `getPlan` / `monthlyPrice` to the synchronous cache
     8→// exposed here, so an owner edit (savePlanConfig) propagates to the
     9→// client billing experience and to MRR on the next read.
    10→//
    11→// The cache is hydrated from the PlanConfig table on startup and after
    12→// every mutation. If the table is empty on first hydrate, the DEFAULT
    13→// configs are seeded so the platform is never without plans. If the
    14→// table contains the LEGACY catalog (Beta/Pro/Max/Enterprise), the
    15→// migration runs automatically: Enterprise is removed (only when no
    16→// active subscription references it), Beta is renamed to Plus, and a
    17→// Free plan is inserted at sortOrder 0.
    18→//
    19→// Final canonical catalog:
    20→//   1. Free  (planId='free',  isFree=true,  priceMonthly=0)
    21→//   2. Plus  (planId='plus',  isFree=false, priceMonthly=0 or 9)
    22→//   3. Pro   (planId='pro',   priceMonthly=49)
    23→//   4. Max   (planId='max',   priceMonthly=99)
    24→// ============================================================

   178→/** Resolve the platform's default billing currency via a LAZY import of
   179→ *  country-pricing (which itself imports `getPlanConfigSync` from here).
   180→ *  The lazy `await import()` breaks the static circular dependency —
   181→ *  this module never statically imports country-pricing, so both load.
   182→ *  Falls back to 'CHF' (the schema default) if anything fails. */
   183→async function resolvePlatformDefaultCurrency(): Promise<string> {
   184→  try {
   185→    const { getPlatformDefaultCurrency } = await import('./country-pricing');
   186→    return await getPlatformDefaultCurrency();
   187→  } catch {
   188→    return 'CHF';
   189→  }

   198→let _hydrated = false;
   199→let _hydrating: Promise<void> | null = null;
   200→const _subscribers = new Set<() => void>();
   201→
   202→function notify() {
   203→  for (const cb of _subscribers) {
   204→    try {
   205→      cb();
   206→    } catch {
   207→      // ignore subscriber errors
   208→    }
   209→  }
   210→}
   211→
   212→/** Register a callback fired after every cache refresh (hydrate / mutation). */
   213→export function subscribe(cb: () => void): () => void {
   214→  _subscribers.add(cb);
   215→  return () => _subscribers.delete(cb);
   216→}

   269→function rowToData(row: PlanConfigRow): PlanConfigData {
   270→  let features: string[] = [];
   271→  let entitlements: string[] = [];
   272→  try {
   273→    features = JSON.parse(row.features || '[]');
   274→  } catch {
   275→    features = [];
   276→  }
   277→  try {
   278→    entitlements = JSON.parse(row.entitlements || '[]');
   279→  } catch {
   280→    entitlements = [];
   281→  }
   282→  // Explicitly pick only the known limit fields. Older DB rows may still
   283→  // carry stale `aiWords` / `aiArticles` / `automationRuns` keys in the
   284→  // limits JSON (those were removed because no real enforcement exists).
   285→  // Picking only the known fields prevents the stale keys from leaking
   286→  // back into the cache and being re-serialized on the next save.
   287→  let limits: PlanLimits = { maxSites: 0, storageBytes: 0 };
   288→  try {
   289→    const parsed = JSON.parse(row.limits || '{}') as Partial<PlanLimits>;
   290→    limits = {
   291→      maxSites: typeof parsed.maxSites === 'number' ? parsed.maxSites : 0,
   292→      storageBytes: typeof parsed.storageBytes === 'number' ? parsed.storageBytes : 0,
   293→    };
   294→  } catch {
   295→    // keep defaults
   296→  }
   297→  return {
   298→    planId: row.planId,
   299→    name: row.name,
   300→    priceMonthly: row.priceMonthly,
   301→    priceYearly: row.priceYearly,
   302→    currency: row.currency,
   303→    interval: (row.interval === 'yearly' ? 'yearly' : 'monthly') as BillingInterval,
   304→    isFree: row.isFree,
   305→    freePlanDurationDays: row.freePlanDurationDays,
   306→    stripePriceIdMonthly: row.stripePriceIdMonthly,
   307→    stripePriceIdYearly: row.stripePriceIdYearly,
   308→    active: row.active,
   309→    features,
   310→    entitlements,
   311→    limits,
   312→    badgeVariant: row.badgeVariant,
   313→    sortOrder: row.sortOrder,
   314→  };
   315→}

   338→/**
   339→ * Migrate the legacy catalog to the new Free/Plus/Pro/Max shape. Idempotent.
   340→ *
   341→ * Legacy catalog (created by older bootstrap):
   342→ *   beta (free, active=false), pro, max, enterprise (stub)
   343→ *
   344→ * Migration steps (each guarded, can be re-run safely):
   345→ *  1. If `enterprise` exists AND no active subscription references it → delete it.
   346→ *     (Active subs that reference enterprise are preserved — the migration will
   347→ *      not break them; the plan row stays if needed. In practice, the demo
   348→ *      dataset has no DB subscriptions yet, so this is a clean delete.)
   349→ *  2. If `beta` exists → rename to `plus` (planId, name, badgeVariant) and
   350→ *     re-enable it (legacy `beta` was active=false in the demo DB). Also
   351→ *     shift sortOrder to 1 (Free takes 0).
   352→ *  3. If `free` does NOT exist → insert the Free plan at sortOrder 0.
   353→ *  4. Renumber sortOrder to match the canonical order: free=0, plus=1, pro=2, max=3.
   354→ *
   355→ * Returns a short report string for the bootstrap log.
   356→ */
   357→export async function migrateLegacyPlans(): Promise<string> {
   358→  const notes: string[] = [];
   359→
   360→  try {
   361→    // Step 1: delete Enterprise if safe
   362→    const enterprise = await db.planConfig.findUnique({ where: { planId: 'enterprise' } });
   363→    if (enterprise) {
   364→      const subCount = await db.subscription.count({ where: { planId: 'enterprise' } });
   365→      if (subCount === 0) {
   366→        await db.planConfig.delete({ where: { planId: 'enterprise' } });
   367→        notes.push('deleted enterprise plan (0 active subscriptions)');
   368→      } else {
   369→        notes.push(`kept enterprise plan (${subCount} active subscriptions reference it)`);
   370→      }
   371→    }
   372→
   373→    // Step 2: rename beta → plus (and refresh to a real paid tier)
   374→    const beta = await db.planConfig.findUnique({ where: { planId: 'beta' } });
   375→    if (beta) {
   376→      // Don't overwrite an existing plus row — if both exist, prefer the
   377→      // plus row and just delete the beta stub.
   378→      const plus = await db.planConfig.findUnique({ where: { planId: 'plus' } });
   379→      if (!plus) {
   380→        // Rename beta → plus AND refresh the data to the real DEFAULT
   381→        // Plus tier (paid plan with real entitlements + limits, not the
   382→        // legacy free-tier Beta stub). The old Beta was active=false and
   383→        // had no real customers, so this is a safe refresh.
   384→        const plusDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'plus')!;
   385→        await db.planConfig.update({
   386→          where: { planId: 'beta' },
   387→          data: {
   388→            ...dataToRow(plusDefault),
   389→            // Keep the original createdAt for continuity.
   390→            // (updatedAt is auto.)
   391→          },
   392→        });
   393→        notes.push('renamed + refreshed beta → plus (real paid tier)');
   394→      } else {
   395→        // Both beta and plus exist — delete the beta stub (migrate any
   396→        // legacy subscriptions on beta to plus first).
   397→        const betaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   398→        if (betaSubs > 0) {
   399→          await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   400→          notes.push(`migrated ${betaSubs} beta subscription(s) → plus`);
   401→        }
   402→        await db.planConfig.delete({ where: { planId: 'beta' } });
   403→        notes.push('deleted legacy beta stub (plus already exists)');
   404→      }
   405→    }
   406→
   407→    // Step 3: insert Free if missing
   408→    const free = await db.planConfig.findUnique({ where: { planId: 'free' } });
   409→    if (!free) {
   410→      const freeDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'free')!;
   411→      await db.planConfig.create({ data: dataToRow(freeDefault) });
   412→      notes.push('inserted free plan');
   413→    } else {
   414→      // Ensure free plan is properly flagged
   415→      if (!free.isFree || free.badgeVariant !== 'free' || free.sortOrder !== 0) {
   416→        await db.planConfig.update({
   417→          where: { planId: 'free' },
   418→          data: { isFree: true, badgeVariant: 'free', sortOrder: 0 },
   419→        });
   420→        notes.push('normalized free plan flags');
   421→      }
   422→    }
   423→
   424→    // Step 4: normalize sortOrder for all canonical plans
   425→    const canonicalOrder: Record<string, number> = { free: 0, plus: 1, pro: 2, max: 3 };
   426→    for (const [pid, order] of Object.entries(canonicalOrder)) {
   427→      const row = await db.planConfig.findUnique({ where: { planId: pid } });
   428→      if (row && row.sortOrder !== order) {
   429→        await db.planConfig.update({ where: { planId: pid }, data: { sortOrder: order } });
   430→      }
   431→    }
   432→
   433→    // Also migrate legacy subscriptions on 'beta' or 'enterprise' planId
   434→    // to the new ids (beta → plus, enterprise → max) so historical subs
   435→    // don't reference a missing plan.
   436→    const legacyBetaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   437→    if (legacyBetaSubs > 0) {
   438→      await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   439→      notes.push(`migrated ${legacyBetaSubs} legacy beta subscription(s) → plus`);
   440→    }
   441→    const legacyEnterpriseSubs = await db.subscription.count({ where: { planId: 'enterprise' } });
   442→    if (legacyEnterpriseSubs > 0) {
   443→      await db.subscription.updateMany({ where: { planId: 'enterprise' }, data: { planId: 'max' } });
   444→      notes.push(`migrated ${legacyEnterpriseSubs} legacy enterprise subscription(s) → max`);
   445→    }
   446→  } catch (err) {
   447→    notes.push(`migration error (non-fatal): ${(err as Error).message}`);
   448→  }
   449→
   450→  return notes.length > 0 ? notes.join('; ') : 'no migration needed';
   451→}

   453→/** Load all plan configs from the DB into the cache. Self-seeds the
   454→ *  DEFAULT_PLAN_CONFIGS when the table is empty. Runs the legacy
   455→ *  migration when the catalog still contains 'beta' or 'enterprise'.
   456→ *  Safe to call repeatedly. */
   457→export async function hydrate(): Promise<void> {
   458→  if (_hydrating) return _hydrating;
   459→  _hydrating = (async () => {
   460→    try {
   461→      const rows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   462→      if (rows.length === 0) {
   463→        // Self-seed so the platform is never without plans.
   464→        await db.planConfig.createMany({ data: DEFAULT_PLAN_CONFIGS.map(dataToRow) });
   465→        _cache = DEFAULT_PLAN_CONFIGS.map((p) => ({
   466→          ...p,
   467→          features: [...p.features],
   468→          entitlements: [...p.entitlements],
   469→          limits: { ...p.limits },
   470→        }));
   471→      } else {
   472→        // Check for legacy catalog (beta / enterprise) and migrate.
   473→        const hasLegacy =
   474→          rows.some((r) => r.planId === 'beta' || r.planId === 'enterprise') ||
   475→          !rows.some((r) => r.planId === 'free');
   476→        if (hasLegacy) {
   477→          await migrateLegacyPlans();
   478→          const migratedRows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   479→          _cache = migratedRows.map(rowToData);
   480→        } else {
   481→          _cache = rows.map(rowToData);
   482→        }
   483→      }
   484→      _hydrated = true;
   485→      notify();
   486→    } catch (err) {
   487→      // DB unavailable — keep defaults (still functional, just not persisted).
   488→      _hydrated = true;
   489→      notify();
   490→    } finally {
   491→      _hydrating = null;
   492→    }
   493→  })();
   494→  return _hydrating;
   495→}

   517→/** Update an existing plan config. Writes DB, refreshes cache.
   518→ *
   519→ *  AUTO-SYNC TO STRIPE: when Stripe is configured AND the plan is
   520→ *  PAID (priceMonthly > 0 OR priceYearly > 0) AND the admin did NOT
   521→ *  explicitly set the Stripe Price IDs in this patch (i.e. the field
   522→ *  is omitted from `patch`), the backend calls syncPlanToStripe to
   523→ *  ensure a Stripe Product + monthly + yearly Prices exist and writes
   524→ *  the resolved Stripe Price IDs back onto the row. This means the
   525→ *  admin can edit a paid plan's name / features / limits and the
   526→ *  Stripe side stays in sync without manually wiring price IDs.
   527→ *
   528→ *  When the admin explicitly sets `stripePriceIdMonthly` / `stripePriceIdYearly`
   529→ *  (even to null = "clear"), the auto-sync is skipped — the admin is
   530→ *  taking manual control of the Stripe side. They can use the dedicated
   531→ *  "Sync to Stripe" route (/api/platform/admin/plans/[planId]/sync-stripe)
   532→ *  to push the local plan to Stripe at any time. */
   533→export async function savePlanConfig(planId: string, patch: PlanConfigInput): Promise<PlanConfigData | null> {
   534→  const existing = await db.planConfig.findUnique({ where: { planId } });
   535→  if (!existing) return null;
   536→  const current = rowToData(existing);
   537→
   538→  // Distinguish "field omitted" from "field set to null". When the
   539→  // admin omits both stripePriceId* fields, they want the backend to
   540→  // keep them in sync with Stripe automatically (see AUTO-SYNC above).
   541→  // When either is explicitly provided (including null to clear),
   542→  // the admin is taking manual control.
   543→  const adminTouchedStripePriceIds =
   544→    'stripePriceIdMonthly' in patch || 'stripePriceIdYearly' in patch;
   545→
   546→  const next: PlanConfigData = {
   547→    planId: current.planId,
   548→    name: patch.name ?? current.name,
   549→    priceMonthly: patch.priceMonthly ?? current.priceMonthly,
   550→    priceYearly: patch.priceYearly ?? current.priceYearly,
   551→    currency: patch.currency ?? current.currency,
   552→    interval: patch.interval ?? current.interval,
   553→    isFree: patch.isFree ?? current.isFree,
   554→    freePlanDurationDays: patch.freePlanDurationDays ?? current.freePlanDurationDays,
   555→    stripePriceIdMonthly: patch.stripePriceIdMonthly ?? current.stripePriceIdMonthly,
   556→    stripePriceIdYearly: patch.stripePriceIdYearly ?? current.stripePriceIdYearly,
   557→    active: patch.active ?? current.active,
   558→    features: patch.features ?? current.features,
   559→    entitlements: patch.entitlements ?? current.entitlements,
   560→    limits: { ...current.limits, ...(patch.limits ?? {}) },
   561→    badgeVariant: patch.badgeVariant ?? current.badgeVariant,
   562→    sortOrder: patch.sortOrder ?? current.sortOrder,
   563→  };
   564→  await db.planConfig.update({ where: { planId }, data: dataToRow(next) });
   565→  await hydrate();
   566→
   567→  // ---- AUTO-SYNC TO STRIPE ----
   568→  // Best-effort — never throws. Swallows Stripe errors so an admin
   569→  // edit doesn't fail just because Stripe is unreachable. The admin
   570→  // can use the explicit "Sync to Stripe" route to surface errors.
   571→  if (!adminTouchedStripePriceIds && !next.isFree && (next.priceMonthly > 0 || next.priceYearly > 0)) {
   572→    try {
   573→      const { isStripeConfiguredAsync, getStripeClient, syncPlanToStripe } = await import('@/lib/stripe');
   574→      if (await isStripeConfiguredAsync()) {
   575→        const stripe = await getStripeClient();
   576→        const syncResult = await syncPlanToStripe(stripe, {
   577→          planId: next.planId,
   578→          name: next.name,
   579→          priceMonthly: next.priceMonthly,
   580→          priceYearly: next.priceYearly,
   581→          currency: next.currency,
   582→          stripePriceIdMonthly: next.stripePriceIdMonthly,
   583→          stripePriceIdYearly: next.stripePriceIdYearly,
   584→        });
   585→        // Persist the resolved Stripe Price IDs back onto the row
   586→        // (only when they changed — otherwise the update is a no-op).
   587→        if (
   588→          syncResult.stripePriceIdMonthly !== next.stripePriceIdMonthly ||
   589→          syncResult.stripePriceIdYearly !== next.stripePriceIdYearly
   590→        ) {
   591→          await db.planConfig.update({
   592→            where: { planId },
   593→            data: {
   594→              stripePriceIdMonthly: syncResult.stripePriceIdMonthly,
   595→              stripePriceIdYearly: syncResult.stripePriceIdYearly,
   596→            },
   597→          });
   598→          await hydrate();
   599→        }
   600→      }
   601→    } catch {
   602→      // best-effort — never block the save on a Stripe error
   603→    }
   604→  }
   605→
   606→  return getPlanConfigSync(planId);
   607→}

   609→/** Create a new plan config.
   610→ *
   611→ *  AUTO-SYNC TO STRIPE: when Stripe is configured AND the new plan
   612→ *  is PAID (priceMonthly > 0 OR priceYearly > 0) AND the admin did
   613→ *  NOT explicitly provide Stripe Price IDs in the input, the backend
   614→ *  calls syncPlanToStripe to create a Stripe Product + monthly +
   615→ *  yearly Prices and writes the resolved Stripe Price IDs back onto
   616→ *  the new plan row. So creating a new paid plan in Platform Admin
   617→ *  automatically creates the corresponding Stripe Product + Prices —
   618→ *  no need to manually wire Stripe Price IDs in the Stripe dashboard.
   619→ *
   620→ *  When the admin explicitly provides Stripe Price IDs (manually
   621→ *  created in the Stripe dashboard), the auto-sync is skipped — the
   622→ *  admin is asserting the Stripe side is already wired. */
   623→export async function createPlanConfig(
   624→  input: PlanConfigInput & { planId: string; name: string },
   625→): Promise<PlanConfigData> {
   626→  const maxOrder = await db.planConfig.count();
   627→  // Currency is NOT chosen per-plan in the admin editor — it is derived
   628→  // from the platform's default country (CountryPricing table). When the
   629→  // caller omits `currency`, resolve it from the platform default so a
   630→  // newly-created plan immediately matches the platform's billing
   631→  // configuration. When the caller explicitly provides a currency (e.g.
   632→  // a migration script), honor it.
   633→  const currency =
   634→    input.currency && input.currency.trim().length > 0
   635→      ? input.currency
   636→      : await resolvePlatformDefaultCurrency();
   637→  const data: PlanConfigData = {
   638→    planId: input.planId,
   639→    name: input.name,
   640→    priceMonthly: input.priceMonthly ?? 0,
   641→    priceYearly: input.priceYearly ?? 0,
   642→    currency,
   643→    interval: input.interval ?? 'monthly',
   644→    isFree: input.isFree ?? false,
   645→    freePlanDurationDays: input.freePlanDurationDays ?? null,
   646→    stripePriceIdMonthly: input.stripePriceIdMonthly ?? null,
   647→    stripePriceIdYearly: input.stripePriceIdYearly ?? null,
   648→    active: input.active ?? true,
   649→    features: input.features ?? [],
   650→    entitlements: input.entitlements ?? [],
   651→    limits: {
   652→      maxSites: 0,
   653→      storageBytes: 0,
   654→      ...(input.limits ?? {}),
   655→    },
   656→    badgeVariant: input.badgeVariant ?? input.planId,
   657→    sortOrder: input.sortOrder ?? maxOrder,
   658→  };
   659→  await db.planConfig.create({ data: dataToRow(data) });
   660→  await hydrate();
   661→
   662→  // ---- AUTO-SYNC TO STRIPE ----
   663→  // Best-effort — never throws. The plan row is already created; if
   664→  // the Stripe side fails, the admin can use the explicit "Sync to
   665→  // Stripe" route to surface errors.
   666→  const adminTouchedStripePriceIds =
   667→    'stripePriceIdMonthly' in input || 'stripePriceIdYearly' in input;
   668→  if (
   669→    !adminTouchedStripePriceIds &&
   670→    !data.isFree &&
   671→    (data.priceMonthly > 0 || data.priceYearly > 0)
   672→  ) {
   673→    try {
   674→      const { isStripeConfiguredAsync, getStripeClient, syncPlanToStripe } = await import('@/lib/stripe');
   675→      if (await isStripeConfiguredAsync()) {
   676→        const stripe = await getStripeClient();
   677→        const syncResult = await syncPlanToStripe(stripe, {
   678→          planId: data.planId,
   679→          name: data.name,
   680→          priceMonthly: data.priceMonthly,
   681→          priceYearly: data.priceYearly,
   682→          currency: data.currency,
   683→          stripePriceIdMonthly: data.stripePriceIdMonthly,
   684→          stripePriceIdYearly: data.stripePriceIdYearly,
   685→        });
   686→        if (
   687→          syncResult.stripePriceIdMonthly !== data.stripePriceIdMonthly ||
   688→          syncResult.stripePriceIdYearly !== data.stripePriceIdYearly
   689→        ) {
   690→          await db.planConfig.update({
   691→            where: { planId: data.planId },
   692→            data: {
   693→              stripePriceIdMonthly: syncResult.stripePriceIdMonthly,
   694→              stripePriceIdYearly: syncResult.stripePriceIdYearly,
   695→            },
   696→          });
   697→          await hydrate();
   698→          data.stripePriceIdMonthly = syncResult.stripePriceIdMonthly;
   699→          data.stripePriceIdYearly = syncResult.stripePriceIdYearly;
   700→        }
   701→      }
   702→    } catch {
   703→      // best-effort — never block the create on a Stripe error
   704→    }
   705→  }
   706→
   707→  return data;
   708→}

   710→/**
   711→ * Delete a plan config. Refuses if:
   712→ *  - it's the last remaining plan, OR
   713→ *  - any Subscription still references this planId (would orphan the sub).
   714→ *
   715→ * Returns `{ ok: boolean, reason?: string }` describing the outcome.
   716→ */
   717→export async function deletePlanConfig(
   718→  planId: string,
   719→): Promise<{ ok: boolean; reason?: string }> {
   720→  const count = await db.planConfig.count();
   721→  if (count <= 1) return { ok: false, reason: 'Cannot delete the last plan.' };
   722→  const subCount = await db.subscription.count({ where: { planId } });
   723→  if (subCount > 0) {
   724→    return {
   725→      ok: false,
   726→      reason: `Cannot delete plan "${planId}" — ${subCount} active subscription(s) reference it. Migrate them first.`,
   727→    };
   728→  }
   729→  try {
   730→    await db.planConfig.delete({ where: { planId } });
   731→    await hydrate();
   732→    return { ok: true };
   733→  } catch (err) {
   734→    return { ok: false, reason: (err as Error).message };
   735→  }
   736→}
