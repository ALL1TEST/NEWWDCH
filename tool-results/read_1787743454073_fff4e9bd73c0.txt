   121→function formatDate(dateStr: string | null | undefined): string {
   122→  if (!dateStr) return '—';
   123→  try {
   124→    return new Date(dateStr).toLocaleDateString(undefined, {
   125→      month: 'short',
   126→      day: 'numeric',
   127→      year: 'numeric',
   128→    });
   129→  } catch {
   130→    return '—';
   131→  }
   132→}
   133→
   134→function formatRelative(dateStr: string | null | undefined): string {
   135→  if (!dateStr) return '—';
   136→  try {
   137→    const date = new Date(dateStr);
   138→    const now = Date.now();
   139→    const diffMs = now - date.getTime();
   140→    const seconds = Math.floor(diffMs / 1000);
   141→    const minutes = Math.floor(seconds / 60);
   142→    const hours = Math.floor(minutes / 60);
   143→    const days = Math.floor(hours / 24);
   144→    if (days > 30) return formatDate(dateStr);
   145→    if (days > 0) return `${days}d ago`;
   146→    if (hours > 0) return `${hours}h ago`;
   147→    if (minutes > 0) return `${minutes}m ago`;
   148→    return 'just now';
   149→  } catch {
   150→    return '—';
   151→  }
   152→}

   256→// ==================== Validation ====================
   257→
   258→function validateRedirectForm(form: RedirectFormData, isEdit: boolean): string | null {
   259→  if (!form.fromPath.trim()) return 'From path is required.';
   260→  if (!form.fromPath.startsWith('/')) return 'From path must start with "/".';
   261→  if (!form.toPath.trim()) return 'To path is required.';
   262→  if (!form.toPath.startsWith('/')) return 'To path must start with "/".';
   263→  // Self-redirect check (case-insensitive — /About and /about are the same URL).
   264→  if (form.fromPath.trim().toLowerCase() === form.toPath.trim().toLowerCase()) {
   265→    return 'From path and to path cannot be the same (self-redirect).';
   266→  }
   267→  // Basic path validity — no spaces, no illegal chars.
   268→  const pathRe = /^\/[^\s]*$/;
   269→  if (!pathRe.test(form.fromPath.trim())) return 'From path contains invalid characters.';
   270→  if (!pathRe.test(form.toPath.trim())) return 'To path contains invalid characters.';
   271→  // Edit-mode warning hint (not a hard block) — handled elsewhere.
   272→  if (!isEdit && form.toPath.trim().toLowerCase().startsWith(form.fromPath.trim().toLowerCase())) {
   273→    // Allow but caller may warn — kept silent here to avoid noise.
   274→  }
   275→  return null;
   276→}

   430→function CsvImportDialog({ open, onOpenChange }: { open: boolean; onOpenChange: (v: boolean) => void }) {
   431→  const queryClient = useQueryClient();
   432→  const [csvContent, setCsvContent] = useState('');
   433→  const [file, setFile] = useState<File | null>(null);
   434→  const [validation, setValidation] = useState<CsvValidationResult | null>(null);
   435→  const [importResult, setImportResult] = useState<CsvImportResult | null>(null);
   436→  const [step, setStep] = useState<'upload' | 'preview' | 'done'>('upload');
   437→
   438→  const validateMutation = useMutation({
   439→    mutationFn: (csv: string) =>
   440→      postApi<CsvValidationResult>('/api/redirects/bulk?action=import', { csvContent: csv }),
   441→    onSuccess: (data) => {
   442→      setValidation(data as unknown as CsvValidationResult);
   443→      setStep('preview');
   444→    },
   445→    onError: (err: Error) => toast.error(err.message || 'Failed to validate CSV'),
   446→  });
   447→
   448→  const importMutation = useMutation({
   449→    mutationFn: () =>
   450→      postApi<CsvImportResult>(
   451→        '/api/redirects/bulk?action=import&confirm=true',
   452→        { csvContent },
   453→      ),
   454→    onSuccess: (data) => {
   455→      const result = data as unknown as CsvImportResult;
   456→      setImportResult(result);
   457→      setStep('done');
   458→      queryClient.invalidateQueries({ queryKey: queryKeys.redirects.all });
   459→      if (result.imported > 0) {
   460→        toast.success(`Imported ${result.imported} redirect${result.imported === 1 ? '' : 's'}`);
   461→      }
   462→      if (result.errorsDuringImport > 0) {
   463→        toast.error(`${result.errorsDuringImport} row(s) failed to import`);
   464→      }
   465→    },
   466→    onError: (err: Error) => toast.error(err.message || 'Import failed'),
   467→  });
   468→
   469→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   470→    const f = e.target.files?.[0];
   471→    if (!f) return;
   472→    setFile(f);
   473→    const reader = new FileReader();
   474→    reader.onload = (ev) => {
   475→      const text = ev.target?.result as string;
   476→      setCsvContent(text);
   477→      setValidation(null);
   478→      setImportResult(null);
   479→      setStep('upload');
   480→    };
   481→    reader.readAsText(f);
   482→  };
   483→
   484→  const handleDrop = (e: React.DragEvent) => {
   485→    e.preventDefault();

   487→    if (!f) return;
   488→    setFile(f);
   489→    const reader = new FileReader();
   490→    reader.onload = (ev) => {
   491→      const text = ev.target?.result as string;
   492→      setCsvContent(text);
   493→      setValidation(null);
   494→      setImportResult(null);
   495→      setStep('upload');
   496→    };
   497→    reader.readAsText(f);
   498→  };
   499→
   500→  const handleValidate = () => {
   501→    if (csvContent.trim()) validateMutation.mutate(csvContent);
   502→  };
   503→  const handleImport = () => importMutation.mutate();
   504→  const handleClose = () => {
   505→    onOpenChange(false);
   506→    setTimeout(() => {
   507→      setCsvContent('');
   508→      setFile(null);
   509→      setValidation(null);
   510→      setImportResult(null);
   511→      setStep('upload');
   512→    }, 200);
   513→  };
   514→
   515→  return (
   516→    <Dialog open={open} onOpenChange={handleClose}>

   523→            <code className="text-xs bg-muted px-1 py-0.5 rounded">toPath</code>,{' '}
   524→            <code className="text-xs bg-muted px-1 py-0.5 rounded">type</code>{' '}
   525→            (301/302/307/308),{' '}
   526→            <code className="text-xs bg-muted px-1 py-0.5 rounded">status</code>{' '}
   527→            (active/inactive).
   528→          </DialogDescription>
   529→        </DialogHeader>
   530→
   531→        {step === 'upload' && (
   532→          <div className="space-y-4 py-2">
   533→            <div
   534→              className="border-2 border-dashed rounded-lg p-8 text-center cursor-pointer hover:border-primary/50 hover:bg-muted/30 transition-colors"
   535→              onDragOver={(e) => e.preventDefault()}
   536→              onDrop={handleDrop}
   537→              onClick={() => document.getElementById('csv-file-input')?.click()}
   538→            >
   539→              <input
   540→                id="csv-file-input"
   541→                type="file"
   542→                accept=".csv,text/csv"
   543→                className="hidden"
   544→                onChange={handleFileChange}
   545→              />
   546→              <Upload className="h-8 w-8 mx-auto text-muted-foreground/50 mb-2" />
   547→              <p className="text-sm font-medium">
   548→                Drop CSV file here or click to browse
   549→              </p>

   551→                <p className="text-xs text-muted-foreground mt-1">{file.name}</p>
   552→              )}
   553→            </div>
   554→            <div className="space-y-2">
   555→              <Label>Or paste CSV content:</Label>
   556→              <textarea
   557→                className="w-full min-h-[120px] rounded-md border border-border bg-transparent px-3 py-2 text-xs font-mono shadow-sm placeholder:text-muted-foreground focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring resize-y"

   564→              <Button variant="outline" onClick={handleClose}>
   565→                Cancel
   566→              </Button>
   567→              <Button
   568→                onClick={handleValidate}
   569→                disabled={!csvContent.trim() || validateMutation.isPending}
   570→              >
   571→                {validateMutation.isPending && (
   572→                  <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   573→                )}
   574→                Validate &amp; Preview
   575→              </Button>
   576→            </DialogFooter>
   577→          </div>
   578→        )}
   579→
   580→        {step === 'preview' && validation && (
   581→          <div className="space-y-4 py-2">
   582→            <div className="grid grid-cols-3 gap-3">
