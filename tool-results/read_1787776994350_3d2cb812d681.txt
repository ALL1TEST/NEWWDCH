   446→// -------------------- Validation --------------------
   447→
   448→/** Returns true when the Name field is non-empty AND every required field for
   449→ *  the currently-selected provider has a non-empty value. For providers that
   450→ *  require a validated connection (R2/FTP/OAuth), the connection state must
   451→ *  also be 'connected'. Fields belonging to OTHER providers are never
   452→ *  checked — they were cleared on provider switch. Optional fields
   453→ *  (required: false) are never checked. */
   454→function isFormValid(form: StorageForm): boolean {
   455→  if (!form.name.trim()) return false;
   456→  const config = PROVIDER_CONFIG[form.provider];
   457→  if (!config) return false;
   458→
   459→  // All required fields for this provider must be filled. For OAuth
   460→  // providers, destination fields (e.g. folder) are only reachable after a
   461→  // successful connection, so a 'connected' state implies the connection
   462→  // fields are filled — but we still verify destination required fields.
   463→  const allFieldsFilled = config.fields.every((f) => {
   464→    if (!f.required) return true;
   465→    const v = form.config[f.key];
   466→    if (v == null) return false;
   467→    return f.type === 'switch' ? true : v.trim() !== '';
   468→  });
   469→  if (!allFieldsFilled) return false;
   470→
   471→  if (config.requiresConnection && form.connection !== 'connected') {
   472→    return false;
   473→  }
   474→  return true;
   475→}
   476→
   477→/** Compute a signature for the connection-group fields of the current
   478→ *  provider. Used to detect whether the user has edited any connection
   479→ *  field since the last Test/Connect — if so, the previous 'connected'
   480→ *  state is invalidated and they must re-validate. */
   481→function connectionSignature(form: StorageForm): string {
   482→  const config = PROVIDER_CONFIG[form.provider];
   483→  if (!config) return '';
   484→  return config.fields
   485→    .filter((f) => f.group === 'connection')
   486→    .map((f) => `${f.key}=${form.config[f.key] ?? ''}`)
   487→    .join('|');
   488→}
   489→
   490→/** True when all required connection-group fields are filled, so the
   491→ *  Test/Connect button can be enabled. */
   492→function canValidateConnection(form: StorageForm): boolean {
   493→  const config = PROVIDER_CONFIG[form.provider];
   494→  if (!config || config.connectionType === 'none') return false;
   495→  return config.fields
   496→    .filter((f) => f.group === 'connection')
   497→    .every((f) => {
   498→      if (!f.required) return true;
   499→      const v = form.config[f.key];
   500→      return v != null && v.trim() !== '';
   501→    });
   502→}

  1107→              />
  1108→            ) : undefined
  1109→          }
  1110→        />
  1111→      )}
  1112→
  1113→      {/* Create/Edit Dialog */}
  1114→      <Dialog open={dialogOpen} onOpenChange={(v) => !v && closeDialog()}>
  1115→        <DialogContent className="sm:max-w-[540px]">

  1118→            <DialogDescription>
  1119→              {editingId
  1120→                ? 'Update storage destination configuration.'
  1121→                : 'Configure a new backup storage destination.'}
  1122→            </DialogDescription>
  1123→          </DialogHeader>
  1124→
  1125→          <div className="space-y-4 py-2 max-h-[62vh] overflow-y-auto pr-1">
  1126→            {/* Name */}
  1127→            <div className="space-y-2">

  1130→              </Label>

  1132→                id="storage-name"

  1135→                onChange={(e) => updateForm('name', e.target.value)}
  1136→              />
  1137→            </div>
  1138→
  1139→            {/* Provider */}
  1140→            <div className="space-y-2">
  1141→              <Label htmlFor="storage-provider">Provider</Label>
  1142→              <ProviderDropdown
  1143→                value={form.provider}
  1144→                onChange={handleProviderChange}
  1145→              />
  1146→            </div>
  1147→
  1148→            {/* Provider-specific configuration */}
  1149→            {currentConfig && (
  1150→              <div className="space-y-3 rounded-lg border border-border bg-muted/30 p-4">

  1159→                {currentConfig.connectionType !== 'none' && (
  1160→                  <div className="space-y-3">

  1168→                          variant="ghost"
  1169→                          size="sm"
  1170→                          className="h-7 text-xs"

  1182→                    form.connection === 'connected' &&
  1183→                    !connectionStale ? (
  1184→                      <div className="flex items-center gap-3 rounded-md border border-emerald-200 bg-emerald-50 dark:bg-emerald-900/20 dark:border-emerald-800 px-3 py-2.5">

  1186→                          <CheckCircle2 className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
  1187→                        </div>
  1188→                        <div className="min-w-0 flex-1">
  1189→                          <p className="text-sm font-medium text-emerald-800 dark:text-emerald-200">
  1190→                            {PROVIDERS.find((p) => p.value === form.provider)?.label} connected

  1192→                          <p className="text-xs text-emerald-600/80 dark:text-emerald-400/80 truncate">
  1193→                            {form.connectionMessage || 'Credentials configured'}
  1194→                          </p>
  1195→                        </div>
  1196→                      </div>
  1197→                    ) : (
  1198→                      <>
  1199→                        {currentConfig.fields
  1200→                          .filter((f) => f.group === 'connection')
  1201→                          .map((f) => (
  1202→                            <ConfigField
  1203→                              key={f.key}
  1204→                              field={f}
  1205→                              value={form.config[f.key] ?? ''}
  1206→                              onChange={(v) => updateConfigField(f.key, v)}
  1207→                              disabled={isTesting}
  1208→                            />
  1209→                          ))}
  1210→
  1211→                        <div className="flex items-center gap-2">
  1212→                          <Button
  1213→                            type="button"
  1214→                            variant="outline"
  1215→                            size="sm"
  1216→                            onClick={
  1217→                              currentConfig.connectionType === 'oauth'
  1218→                                ? handleOAuthConnect
  1219→                                : handleTestConnection
  1220→                            }
  1221→                            disabled={
  1222→                              isTesting ||
  1223→                              !canValidateConnection(form) ||
  1224→                              (currentConfig.connectionType === 'credentials' && !form.name.trim())
  1225→                            }
  1226→                          >
  1227→                            {isTesting ? (
  1228→                              <Loader2 className="h-4 w-4 mr-2 animate-spin" />

  1230→                              <Plug className="h-4 w-4 mr-2" />
  1231→                            ) : (
  1232→                              <RefreshCw className="h-4 w-4 mr-2" />

  1234→                            {currentConfig.actionLabel}

  1236→                          {currentConfig.connectionType === 'oauth' && (

  1238→                              <Lock className="h-3 w-3" />
  1239→                              OAuth credentials stored encrypted
  1240→                            </span>

  1242→                        </div>
  1243→
  1244→                        {form.connection !== 'idle' && form.connection !== 'testing' && (

  1247→                            message={connectionStale ? '' : form.connectionMessage}

  1250→                        {connectionStale && (

  1259→
  1260→                {/* Destination-group fields (Local path; OAuth folder after
  1261→                    a successful connection). For credential providers there
  1262→                    are no destination fields (everything was tested above). */}
  1263→                {currentConfig.connectionType === 'none' &&
  1264→                  currentConfig.fields
  1265→                    .filter((f) => f.group === 'destination')
  1266→                    .map((f) => (
  1267→                      <ConfigField
  1268→                        key={f.key}
  1269→                        field={f}
  1270→                        value={form.config[f.key] ?? ''}
  1271→                        onChange={(v) => updateConfigField(f.key, v)}
  1272→                        disabled={isTesting}
  1273→                      />
  1274→                    ))}
  1275→
  1276→                {currentConfig.connectionType === 'oauth' &&
  1277→                  form.connection === 'connected' &&
  1278→                  !connectionStale &&
  1279→                  currentConfig.fields
  1280→                    .filter((f) => f.group === 'destination')
  1281→                    .map((f) => (
  1282→                      <ConfigField
  1283→                        key={f.key}
  1284→                        field={f}
  1285→                        value={form.config[f.key] ?? ''}
  1286→                        onChange={(v) => updateConfigField(f.key, v)}
  1287→                        disabled={isTesting}
  1288→                      />
  1289→                    ))}
  1290→
  1291→                {currentConfig.connectionType === 'oauth' &&
  1292→                  (form.connection !== 'connected' || connectionStale) && (
  1293→                    <div className="flex items-center gap-2 rounded-md border border-dashed border-border bg-background/50 px-3 py-2 text-xs text-muted-foreground">
  1294→                      <FolderOpen className="h-3.5 w-3.5 shrink-0" />
  1295→                      <span>Connect the account first to choose a destination folder.</span>
  1296→                    </div>
  1297→                  )}
  1298→              </div>
  1299→            )}
  1300→          </div>
  1301→
  1302→          <DialogFooter>
  1303→            <Button variant="outline" onClick={closeDialog}>
  1304→              Cancel
  1305→            </Button>
  1306→            <Button
  1307→              onClick={handleSubmit}
  1308→              disabled={isSaving || !isFormValid(form)}
  1309→            >
  1310→              {isSaving && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
  1311→              {editingId ? 'Update' : 'Create'}
  1312→            </Button>
  1313→          </DialogFooter>
  1314→        </DialogContent>
  1315→      </Dialog>
  1316→
  1317→      {/* Delete Confirmation */}
  1318→      <ConfirmDialog
  1319→        open={!!deleteTarget}
  1320→        onOpenChange={(v) => !v && setDeleteTarget(null)}
  1321→        title="Delete Storage Configuration"
  1322→        description={
  1323→          deleteTarget
  1324→            ? `Are you sure you want to delete "${deleteTarget.name}"? Existing backups stored via this configuration will not be affected.`
  1325→            : undefined
  1326→        }
  1327→        confirmLabel="Delete"
  1328→        variant="destructive"
  1329→        onConfirm={() => {
  1330→          if (deleteTarget) deleteMutation.mutate(deleteTarget.id);
  1331→        }}
  1332→        isLoading={deleteMutation.isPending}
  1333→      />
  1334→    </div>
  1335→  );

  1338→// -------------------- Config Field Renderer --------------------
  1339→
  1340→function ConfigField({
  1341→  field,
  1342→  value,
  1343→  onChange,
  1344→  disabled,
  1345→}: {
  1346→  field: ProviderField;
  1347→  value: string;
  1348→  onChange: (v: string) => void;
  1349→  disabled?: boolean;
  1350→}) {
  1351→  const inputId = `storage-config-${field.key}`;
  1352→  return (
  1353→    <div className="space-y-1.5">
  1354→      <Label htmlFor={inputId} className="text-xs">
  1355→        {field.label}
  1356→        {field.required && <span className="text-destructive ml-0.5">*</span>}
  1357→      </Label>
  1358→      {field.type === 'switch' ? (
  1359→        <div className="flex items-center gap-2 h-9">
  1360→          <Switch
  1361→            id={inputId}
  1362→            checked={value === 'true'}
  1363→            onCheckedChange={(checked) => onChange(checked ? 'true' : 'false')}
  1364→            disabled={disabled}
  1365→          />
  1366→          <span className="text-xs text-muted-foreground">
  1367→            {value === 'true' ? 'Enabled' : 'Disabled'}
  1368→          </span>
  1369→        </div>
  1370→      ) : field.multiline ? (
  1371→        <Textarea
  1372→          id={inputId}
  1373→          placeholder={field.placeholder}
  1374→          value={value}
  1375→          onChange={(e) => onChange(e.target.value)}
  1376→          rows={4}
  1377→          disabled={disabled}
  1378→          className="font-mono text-xs"
  1379→        />
  1380→      ) : (
  1381→        <div className="relative">
  1382→          <Input
  1383→            id={inputId}
  1384→            type={field.type === 'password' ? 'password' : field.type === 'number' ? 'number' : 'text'}
  1385→            placeholder={field.placeholder}
  1386→            value={value}
  1387→            onChange={(e) => onChange(e.target.value)}
  1388→            disabled={disabled}
  1389→            autoComplete={field.type === 'password' ? 'new-password' : undefined}
  1390→            className={field.type === 'password' ? 'pr-9' : ''}
  1391→          />
  1392→          {field.type === 'password' && value && (
  1393→            <Lock className="absolute right-3 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground/60 pointer-events-none" />
  1394→          )}
  1395→        </div>
  1396→      )}
  1397→      {field.helpText && (
  1398→        <p className="text-xs text-muted-foreground leading-relaxed">{field.helpText}</p>
  1399→      )}
  1400→    </div>
  1401→  );
  1402→}
