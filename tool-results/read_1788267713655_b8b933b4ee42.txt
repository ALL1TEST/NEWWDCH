    45→const GB_FACTOR = 1024 * 1024 * 1024;
    46→
    47→/** Server-side currency resolution context (from /api/platform/billing/me
    48→ *  → customerCurrencyResolution). The customer never selects a currency —
    49→ *  the server detects it from the request IP. */
    50→interface CustomerCurrencyResolution {
    51→  currency: string;
    52→  countryCode: string;
    53→  countryName: string;
    54→  source: 'ip' | 'default' | 'local';
    55→}
    56→
    57→/** Server-resolved FINAL price for one plan (from /api/platform/billing/me
    58→ *  → planPricing[planId]). Computed by the same resolveCustomerPricing
    59→ *  the checkout route uses — the price/currency displayed here is exactly
    60→ *  what Stripe charges. */
    61→interface ServerPlanPricing {
    62→  planId: string;
    63→  currency: string;
    64→  monthly: number;
    65→  yearly: number;
    66→  source: 'ip' | 'default' | 'local' | 'plan';
    67→  supported: boolean;
    68→  detectedCurrency: string;
    69→  countryCode: string;
    70→  countryName: string;
    71→  regional: boolean;
    72→}

    74→type BillingStateWithCurrency = ClientBillingState & {

    79→  currencySource?: 'ip' | 'default' | 'local';
    80→  planPricing?: Record<string, ServerPlanPricing>;

    83→/** Resolve the price + currency to display for a plan. Prefers the
    84→ *  SERVER-RESOLVED pricing (planPricing — the same values checkout
    85→ *  charges); falls back to a local pricesByCurrency lookup for older
    86→ *  API responses. */
    87→function resolvePlanPricing(
    88→  plan: {
    89→    id: string;
    90→    pricesByCurrency?: Record<string, { monthly: number; yearly: number }>;
    91→    priceMonthly: number;
    92→    priceYearly: number;
    93→    currency: string;
    94→    interval: 'monthly' | 'yearly';
    95→  },
    96→  serverPricing: ServerPlanPricing | undefined,
    97→  customerCurrency: string,
    98→): { monthly: number; yearly: number; currency: string } {
    99→  if (serverPricing) {
   100→    return {
   101→      monthly: serverPricing.monthly,
   102→      yearly: serverPricing.yearly,
   103→      currency: serverPricing.currency,
   104→    };
   105→  }
   106→  // Legacy fallback — local lookup (display only).
   107→  const entry = plan.pricesByCurrency?.[customerCurrency];
   108→  if (entry) return { monthly: entry.monthly, yearly: entry.yearly, currency: customerCurrency };
   109→  return { monthly: plan.priceMonthly, yearly: plan.priceYearly, currency: plan.currency };
   110→}

   112→/** One customer-facing plan feature: a label line (rendered with a ✓)
   113→ *  plus optional indented sub-lines. Used for the Platform AI block —
   114→ *  its description + the plan's AI usage limits sit NEXT TO the ✓
   115→ *  Platform AI feature so the customer immediately understands what
   116→ *  is included (never "✓ Platform AI" alone when Platform AI is
   117→ *  enabled). */
   118→interface PlanFeatureItem {
   119→  label: string;
   120→  subLines?: string[];
   121→}
   122→
   123→/** Compact AI-limit number for the customer-facing plan cards:
   124→ *  200000 → "200K", 25000 → "25K", 1500 → "1.5K", 100 → "100",
   125→ *  -1 → "Unlimited". */
   126→function formatAiCount(n: number): string {
   127→  if (n === UNLIMITED) return 'Unlimited';
   128→  if (n >= 1000) {
   129→    const k = n / 1000;
   130→    return `${k % 1 === 0 ? String(k) : k.toFixed(1)}K`;
   131→  }
   132→  return String(n);
   133→}
   134→
   135→/** Derive the customer-facing feature list from the plan's entitlements +
   136→ *  a few standard items (max sites, storage, support tier). Uses the
   137→ *  legacy plan.features array when it has entries (legacy compat for
   138→ *  plans that still have manually-maintained marketing copy, e.g. the
   139→ *  synthetic Internal plan). The structured entitlements are the
   140→ *  single source of truth — the admin no longer maintains a separate
   141→ *  marketing list per plan.
   142→ *
   143→ *  PLATFORM AI: the ✓ Platform AI feature item carries its own
   144→ *  description ("AI provided by the platform") plus the plan's AI
   145→ *  usage limits (article + image generations per month — words/tokens
   146→ *  are never metered) as nested sub-lines — shown ONLY while Platform
   147→ *  AI is enabled (Client's Own AI API-only plans never show platform AI
   148→ *  limits; own-API usage is not counted against them). */
   149→function derivePlanFeatures(plan: {
   150→  features: string[];
   151→  entitlements: string[];
   152→  isFree: boolean;
   153→  limits: {
   154→    maxSites: number;
   155→    storageBytes: number;
   156→    aiArticlesPerMonth?: number;
   157→    aiImagesPerMonth?: number;
   158→  };
   159→}): PlanFeatureItem[] {
   160→  if (plan.features.length > 0) {
   161→    // Legacy marketing copy — plain items, no sub-lines.
   162→    return plan.features.map((f) => ({ label: f }));
   163→  }
   164→  const items: PlanFeatureItem[] = [];
   165→  items.push(
   166→    plan.limits.maxSites === -1
   167→      ? { label: 'Unlimited sites' }
   168→      : { label: `Up to ${plan.limits.maxSites} sites` },
   169→  );
   170→  const gb = plan.limits.storageBytes / GB_FACTOR;
   171→  if (gb >= 1) {
   172→    items.push({ label: `${Math.floor(gb)} GB storage` });
   173→  } else if (plan.limits.storageBytes > 0) {
   174→    items.push({ label: `${plan.limits.storageBytes} bytes storage` });
   175→  } else {
   176→    items.push({ label: 'No storage' });
   177→  }
   178→  items.push({ label: plan.isFree ? 'Community support' : 'Priority support' });
   179→  // Enabled features — ONLY the enabled ones are displayed (each
   180→  // entitlement key renders its label). Platform AI nests its limits
   181→  // directly next to the feature; Client's Own AI API states that
   182→  // platform AI limits do not apply to it. The legacy 'ai_content'
   183→  // alias is skipped (it duplicates the AI feature keys).
   184→  for (const e of plan.entitlements) {
   185→    if (e === 'ai_content') continue;
   186→    const label = ENTITLEMENT_LABELS[e as EntitlementKey];
   187→    if (!label) continue;
   188→    if (e === 'ai_platform') {
   189→      const articles = plan.limits.aiArticlesPerMonth ?? 0;
   190→      const images = plan.limits.aiImagesPerMonth ?? 0;
   191→      // Display-only: the monthly usage-limit logic is unchanged; the
   192→      // customer-facing lines just drop the "/ month" suffix (they are
   193→      // usage limits, not prices) and render as separate lines.
   194→      items.push({
   195→        label,
   196→        subLines: [
   197→          'AI provided by the platform',
   198→          `${formatAiCount(articles)} AI articles`,
   199→          `${formatAiCount(images)} AI images`,
   200→        ],
   201→      });
   202→      continue;
   203→    }
   204→    if (e === 'ai_client') {
   205→      items.push({
   206→        label,
   207→        subLines: ["Your own AI provider — platform AI usage limits don't apply"],
   208→      });
   209→      continue;
   210→    }
   211→    items.push({ label });
   212→  }
   213→  return items;
   214→}
   215→
   216→/** Render the derived plan feature items — one ✓ row per item with
   217→ *  optional indented sub-lines (the Platform AI limits next to the
   218→ *  ✓ Platform AI feature). */
   219→function PlanFeatureList({ items }: { items: PlanFeatureItem[] }) {
   220→  return (
   221→    <ul className="space-y-2">
   222→      {items.map((item, idx) => (
   223→        <li key={`${item.label}-${idx}`} className="flex items-start gap-2 text-sm">
   224→          <Check className="h-4 w-4 text-muted-foreground shrink-0 mt-0.5" />
   225→          <div className="min-w-0">
   226→            <span className="block">{item.label}</span>
   227→            {item.subLines?.map((s) => (
   228→              <span key={s} className="block text-xs text-muted-foreground leading-snug mt-0.5">
   229→                {s}
   230→              </span>
   231→            ))}
   232→          </div>
   233→        </li>
   234→      ))}
   235→    </ul>
   236→  );
   237→}

   305→            status?: number;

   325→          'Stripe is not configured on this platform. An admin must connect Stripe in Platform Admin → Stripe Settings to enable real checkout. Free plans can still be selected directly.',
   326→        );

   328→        toast.error(
   329→          'This plan does not have a Stripe Price ID configured. An admin must wire it via Platform Admin → Edit Plan → Stripe Price ID.',
   330→        );

   332→        toast.error(e?.message ?? 'Unable to start checkout. Please try again.');
   333→      }
   334→    },
   335→  });

   362→            <div className="flex items-start justify-between">

   525→  // any ambiguity when a card's period differs from the global one).
   526→
   527→  const isHigherPlan = (plan: { price: number }) => plan.price > currentPlan.price;
   528→  const getActionLabel = (plan: { price: number; isFree: boolean }) => {

   530→    if (isHigherPlan(plan)) return t('billing.upgrade');
   531→    if (plan.price < currentPlan.price) return t('billing.downgrade');
   532→    return t('billing.changePlan');
   533→  };
   534→
   535→  const handleCancel = () => {
   536→    if (window.confirm('Are you sure you want to cancel your subscription?')) {
   537→      cancelMutation.mutate();
   538→    }
   539→  };
   540→
   541→  // Handle a plan change / upgrade:
   542→  //   - Free plan → direct change-plan API (no Stripe needed)
   543→  //   - Paid plan → checkout API (Stripe Checkout Session) — the body
   544→  //     sends ONLY planId + interval; the currency/price is resolved
   545→  //     SERVER-SIDE from the request IP (the frontend cannot pick it).
   546→  //     The interval is the card's DISPLAYED billing period — the
   547→  //     global Monthly/Yearly selection when the plan supports it,
   548→  //     otherwise the plan's only enabled period (the backend rejects
   549→  //     a disabled interval) — so the charged amount always matches.
   550→  const handleSelectPlan = (
   551→    plan: {
   552→      id: PlanId;
   553→      name: string;
   554→      price: number;
   555→      isFree: boolean;
   556→      billingMonthly?: boolean;
   557→      billingYearly?: boolean;
   558→      interval: 'monthly' | 'yearly';
   559→    },
   560→    interval: 'monthly' | 'yearly',
   561→  ) => {
   562→    if (plan.isFree) {
   563→      changePlanMutation.mutate({
   564→        planId: plan.id,
   565→        planName: plan.name,
   566→        isUpgrade: isHigherPlan(plan),
   567→      });
   568→    } else {
   569→      checkoutMutation.mutate({
   570→        planId: plan.id,
   571→        planName: plan.name,
   572→        interval,
   573→      });
   574→    }
   575→  };
   576→
   577→  return (
   578→    <div className="max-w-6xl mx-auto space-y-8">
   579→      {/* Page Header */}
   580→      <div>
   581→        <h1 className="text-xl font-bold tracking-tight text-foreground">{t('billing.title')}</h1>
   582→        <p className="text-sm text-muted-foreground mt-1">{t('billing.description')}</p>
   583→        {/* Auto Currency indicator — shows the customer's resolved
   584→            currency + the source (IP-detected / platform default /
   585→            local development). The server resolves this from the
   586→            request IP; the client cannot change currency manually. */}
   587→        {currencySource && (
   588→          <div className="mt-3 flex flex-wrap items-center gap-2">
   589→            <Badge variant="outline" className="text-[10px] font-mono">
   590→              Currency: {customerCurrency}
   591→            </Badge>
   592→            <span className="text-[11px] text-muted-foreground">
   593→              {currencySource === 'ip'
   594→                ? `Auto-detected from your location${customerCountryName ? ` (${customerCountryName})` : ''}`
   595→                : currencySource === 'default'
   596→                  ? 'Platform default'
   597→                  : 'Local development'}
   598→            </span>
   599→          </div>
   600→        )}
   601→      </div>
   602→
   603→      {/* Free-trial-expired banner — surfaces server-side enforcement */}
   604→      {freeTrialExpired && freeTrialExpiresAt && (
   605→        <Card className="border-amber-500/40 bg-amber-50 dark:bg-amber-950/40">
   606→          <CardContent className="flex items-start gap-3 py-4">
   607→            <AlertTriangle className="mt-0.5 h-5 w-5 shrink-0 text-amber-600 dark:text-amber-400" />
   608→            <div className="text-sm">
   609→              <p className="font-semibold text-amber-700 dark:text-amber-300">
   610→                Free trial expired
   611→              </p>
   612→              <p className="text-muted-foreground">
   613→                Your free access expired on {formatDate(freeTrialExpiresAt)}. Upgrade to a paid plan to continue using gated features.
   614→              </p>
   615→            </div>
   616→          </CardContent>
   617→        </Card>
   618→      )}
   619→
   620→      {/* Current Subscription */}
   621→      <Card>
   622→        <CardHeader>
   623→          <CardTitle className="text-base">{t('billing.currentPlan')}</CardTitle>
   624→        </CardHeader>
   625→        <CardContent className="space-y-4">
   626→          <div className="flex flex-wrap items-start justify-between gap-3">
   627→            <div className="space-y-3 min-w-0">
   628→              <div className="flex flex-wrap items-center gap-2">
   629→                <CreditCard className="h-5 w-5 text-muted-foreground shrink-0" />
   630→                <span className="text-2xl font-semibold tracking-tight">{currentPlan.name}</span>
   631→                <Badge
   632→                  variant="outline"
   633→                  className={`text-[10px] font-semibold ${getPlanBadgeClasses(currentStorePlan.badgeVariant)}`}
   634→                >
   635→                  {currentPlan.name}
   636→                </Badge>
   637→              </div>
   638→              {/* Same price treatment as the redesigned plan cards —
   639→                  large amount + explicit period suffix (display only;
   640→                  the price source and suffix logic are unchanged). */}
   641→              <div className="flex items-baseline gap-1.5 ml-7">
   642→                <span className="text-3xl font-bold tracking-tight">
   643→                  {currentPrice === 0
   644→                    ? t('billing.free')
   645→                    : formatMoney(currentPrice, currentDisplayCurrency)}
   646→                </span>
   647→                {currentPrice > 0 && (
   648→                  <span className="text-sm text-muted-foreground whitespace-nowrap">
   649→                    / {currentInterval === 'yearly' ? 'year' : 'month'}
   650→                  </span>
   651→                )}
   652→              </div>
   653→              {currentPrice > 0 &&
   654→                planPricing?.[currentPlan.id] &&
   655→                planPricing[currentPlan.id].supported &&
   656→                customerCountryName && (
   657→                  <p className="text-[11px] text-muted-foreground ml-7">
   658→                    Priced for {customerCountryName} ({currentDisplayCurrency})
   659→                  </p>
   660→                )}
   661→            </div>
   662→            <Badge
   663→              variant={status === 'active' ? 'default' : 'outline'}
   664→              className="capitalize"
   665→            >
   666→              {status}
   667→            </Badge>
   668→          </div>
   669→
   670→          {/* Current period end / next billing date (DB Subscription source of truth) */}
   671→          {currentPeriodEnd && !isCancelled && (
   672→            <div className="flex items-center gap-2 text-sm text-muted-foreground bg-muted/50 rounded-md p-3">
   673→              <Calendar className="h-4 w-4 shrink-0" />
   674→              <span>
   675→                Next billing: <span className="font-medium text-foreground">{formatDate(currentPeriodEnd)}</span>
   676→              </span>
   677→            </div>
   678→          )}
   679→
   680→          {trialEnd && (
   681→            <div className="flex items-center gap-2 text-sm text-muted-foreground bg-muted/50 rounded-md p-3">

   695→          <div className="space-y-2">
   696→            <p className="text-xs font-medium text-muted-foreground uppercase tracking-wide">
   697→              Included in your plan
   698→            </p>
   699→            <PlanFeatureList items={derivePlanFeatures(currentPlan)} />

   701→
   702→          <Separator />

   706→              <p className="text-xs text-muted-foreground italic">

   714→                <Button variant="outline" size="sm" disabled>
   715→                  {t('billing.managePayment')}
   716→                </Button>
   717→                <Button
   718→                  variant="outline"
   719→                  size="sm"
   720→                  className="text-destructive hover:text-destructive"
   721→                  disabled={cancelMutation.isPending}
   722→                  onClick={handleCancel}
   723→                >
   724→                  {cancelMutation.isPending ? (
   725→                    <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   726→                  ) : null}
   727→                  {t('billing.cancelSubscription')}
   728→                </Button>
   729→              </div>
   730→            </div>
   731→          )}
   732→        </CardContent>
   733→      </Card>
   734→
   735→      {/* Other Plans — ONE global Monthly / Yearly selector above ALL
   736→          cards (never repeated inside a card). Card layout: large plan
   737→          name → large price → divider → feature list with check icons →
   738→          action button pinned to the bottom. */}
   739→      <div>
   740→        <div className="flex flex-wrap items-center justify-between gap-3 mb-5">
   741→          <h2 className="text-base font-semibold">{t('billing.otherPlans')}</h2>
   742→          {/* Global billing-period selector — shown only when at least
   743→              one available plan supports yearly billing; it then drives
   744→              the displayed billing period of every plan card below. */}
   745→          {showPeriodSelector && (
   746→            <div
   747→              role="group"
   748→              aria-label="Billing period"
   749→              className="inline-flex items-center rounded-full border bg-muted/40 p-1"
   750→            >
   751→              <button
   752→                type="button"
   753→                onClick={() => setGlobalPeriod('monthly')}
   754→                aria-pressed={effectivePeriod === 'monthly'}
   755→                className={`h-8 rounded-full px-5 text-xs font-medium transition-colors ${
   756→                  effectivePeriod === 'monthly'
   757→                    ? 'bg-foreground text-background shadow-sm'
   758→                    : 'text-muted-foreground hover:text-foreground'
   759→                }`}
   760→              >
   761→                Monthly
   762→              </button>
   763→              <button
   764→                type="button"
   765→                onClick={() => setGlobalPeriod('yearly')}
   766→                aria-pressed={effectivePeriod === 'yearly'}
   767→                className={`h-8 rounded-full px-5 text-xs font-medium transition-colors ${
   768→                  effectivePeriod === 'yearly'
   769→                    ? 'bg-foreground text-background shadow-sm'
   770→                    : 'text-muted-foreground hover:text-foreground'
   771→                }`}
   772→              >
   773→                Yearly
   774→              </button>
   775→            </div>
   776→          )}
   777→        </div>
   778→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-5">
   779→          {otherPlans.map((plan) => {
   780→            const storePlan = getStorePlan(plan.id);
   781→            // Server-resolved FINAL price — the same values checkout charges.
   782→            const planPricingResolved = resolvePlanPricing(plan, planPricing?.[plan.id], customerCurrency);
   783→            const planDisplayCurrency = planPricingResolved.currency;
   784→            const planSupported = planPricing?.[plan.id]?.supported ?? true;
   785→            // ---- Enabled billing periods (admin-configured per plan) ----
   786→            const planHasMonthly = plan.billingMonthly ?? true;
   787→            const planHasYearly = plan.billingYearly ?? true;
   788→            // The card's displayed period: the GLOBAL selection when this
   789→            // plan supports it; otherwise the plan's ONLY enabled period
   790→            // (pinned — existing billing logic: a disabled period is
   791→            // never displayed and never sent to checkout, so no
   792→            // invalid/fake price can appear; the backend rejects a
   793→            // disabled interval too).
   794→            const followsSelection =
   795→              effectivePeriod === 'monthly' ? planHasMonthly : planHasYearly;
   796→            const cardInterval: 'monthly' | 'yearly' = followsSelection
   797→              ? effectivePeriod
   798→              : planHasMonthly
   799→                ? 'monthly'
   800→                : planHasYearly
   801→                  ? 'yearly'
   802→                  : plan.interval;
   803→            const planPrice =
   804→              cardInterval === 'yearly' ? planPricingResolved.yearly : planPricingResolved.monthly;
   805→            const isBusy =
   806→              (changePlanMutation.isPending && changePlanMutation.variables?.planId === plan.id) ||
   807→              (checkoutMutation.isPending && checkoutMutation.variables?.planId === plan.id);
   808→            return (
   809→              <Card
   810→                key={plan.id}
   811→                className={`relative flex flex-col ${getPlanCardBorderClasses(storePlan.badgeVariant)}`}
   812→              >
   813→                <CardContent className="flex flex-col flex-1">
   814→                  {/* Large plan name — keeps the plan's color identity via
   815→                      the card border + its soft badge. */}
   816→                  <div className="flex flex-wrap items-start justify-between gap-2">
   817→                    <h3 className="text-2xl font-semibold tracking-tight text-foreground">
   818→                      {plan.name}
   819→                    </h3>
   820→                    <Badge
   821→                      variant="outline"
   822→                      className={`text-[10px] font-semibold shrink-0 ${getPlanBadgeClasses(storePlan.badgeVariant)}`}
   823→                    >
   824→                      {plan.name}
   825→                    </Badge>
   826→                  </div>
   827→                  {/* Large price — explicit period suffix so the amount is
   828→                      always unambiguous (also for pinned-period cards). */}
   829→                  <div className="mt-4">
   830→                    <div className="flex items-baseline gap-1.5">
   831→                      <span className="text-4xl font-bold tracking-tight text-foreground">
   832→                        {planPrice === 0 ? t('billing.free') : formatMoney(planPrice, planDisplayCurrency)}
   833→                      </span>
   834→                      {planPrice > 0 && (
   835→                        <span className="text-sm text-muted-foreground whitespace-nowrap">
   836→                          / {cardInterval === 'yearly' ? 'year' : 'month'}
   837→                        </span>
   838→                      )}
   839→                    </div>
   840→                    {/* A plan that cannot follow the global selection keeps
   841→                        its only enabled period — stated explicitly (never
   842→                        an invalid price for the selected period). */}
   843→                    {!plan.isFree && !followsSelection && planHasMonthly !== planHasYearly && (
   844→                      <p className="text-[11px] text-muted-foreground mt-1.5">
   845→                        {planHasMonthly ? 'Monthly billing only' : 'Yearly billing only'}
   846→                      </p>
   847→                    )}
   848→                    {planPrice > 0 && customerCountryName && planSupported && (
   849→                      <p className="text-[11px] text-muted-foreground mt-1.5">
   850→                        Priced for {customerCountryName} ({planDisplayCurrency})
   851→                      </p>
   852→                    )}
   853→                    {planPrice > 0 && !planSupported && planPricing?.[plan.id] && (
   854→                      <p className="text-[11px] text-muted-foreground mt-1.5">
   855→                        Priced in {planDisplayCurrency} (plan default) — {planPricing[plan.id].detectedCurrency} not
   856→                        configured for this plan
   857→                      </p>
   858→                    )}
   859→                  </div>
   860→                  <Separator className="my-5" />
   861→                  {/* Feature list with check icons */}
   862→                  <div className="flex-1">
   863→                    <PlanFeatureList items={derivePlanFeatures(plan)} />
   864→                  </div>
   865→                  {/* Action button pinned to the card bottom (label +
   866→                      behavior unchanged: Upgrade / Downgrade / Change
   867→                      Plan per the existing subscription logic). */}
   868→                  <Button
   869→                    className="mt-6 w-full"
   870→                    variant={isHigherPlan(plan) ? 'default' : 'outline'}
   871→                    disabled={isBusy}
   872→                    onClick={() => handleSelectPlan(plan, cardInterval)}
   873→                  >
   874→                    {isBusy ? (
   875→                      <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   876→                    ) : null}
   877→                    {getActionLabel(plan)}
   878→                  </Button>
   879→                </CardContent>
   880→              </Card>
   881→            );
   882→          })}
   883→        </div>
   884→      </div>
   885→
   886→      {/* Payment History */}
   887→      <Card>
   888→        <CardHeader>
   889→          <CardTitle className="text-base">{t('billing.paymentHistory')}</CardTitle>
   890→        </CardHeader>
   891→        <CardContent>
   892→          {billingState.paymentHistory.length === 0 ? (
   893→            <div className="flex flex-col items-center justify-center py-12 text-center">
   894→              <div className="h-12 w-12 rounded-full bg-muted flex items-center justify-center mb-4">
   895→                <Receipt className="h-6 w-6 text-muted-foreground" />
   896→              </div>
   897→              <h3 className="font-medium text-sm">{t('billing.noPayments')}</h3>
   898→            </div>
   899→          ) : (
   900→            <div className="overflow-x-auto">
   901→              <table className="w-full text-sm">
   902→                <thead>
   903→                  <tr className="border-b text-left">
   904→                    <th className="pb-2 font-medium text-xs text-muted-foreground">{t('billing.invoice')}</th>
   905→                    <th className="pb-2 font-medium text-xs text-muted-foreground">Plan</th>
   906→                    <th className="pb-2 font-medium text-xs text-muted-foreground text-right">{t('billing.amount')}</th>
   907→                    <th className="pb-2 font-medium text-xs text-muted-foreground">{t('billing.status')}</th>
   908→                    <th className="pb-2 font-medium text-xs text-muted-foreground">Method</th>
   909→                    <th className="pb-2 font-medium text-xs text-muted-foreground text-right">{t('billing.date')}</th>
   910→                  </tr>
   911→                </thead>
   912→                <tbody className="divide-y">
   913→                  {billingState.paymentHistory.map((p: Payment) => {
   914→                    const planForPayment = billingState.allPlans.find((pl) => pl.id === p.planId);
   915→                    return (
   916→                      <tr key={p.id} className="hover:bg-accent/30 transition-colors">
   917→                        <td className="py-2.5 pr-4">
   918→                          <span className="font-mono text-xs">{p.invoiceNumber}</span>
   919→                        </td>
   920→                        <td className="py-2.5 pr-4">
   921→                          <Badge
   922→                            variant="outline"
   923→                            className={`text-[10px] font-semibold ${getPlanBadgeClasses(getStorePlan(p.planId).badgeVariant)}`}
   924→                          >
   925→                            {planForPayment?.name ?? p.planId}
   926→                          </Badge>
   927→                        </td>
   928→                        <td className="py-2.5 pr-4 text-right font-medium">
   929→                          {formatCurrency(p.amount, p.currency)}
   930→                        </td>
   931→                        <td className="py-2.5 pr-4">
   932→                          <PaymentStatusBadge status={p.status} />

   934→                        <td className="py-2.5 pr-4 text-xs text-muted-foreground">{p.method}</td>

   939→                    );
   940→                  })}
   941→                </tbody>
   942→              </table>
   943→            </div>
