   550→export function TemplateEditor({ templateId, isNew = false, scope = 'client', onBack, onPreview, onCreated }: TemplateEditorProps) {
   551→  const isPlatform = scope === 'platform';
   552→  const queryClient = useQueryClient();
   553→  const currentUser = useAuthStore((s) => s.user);
   554→
   555→  // -------------------- State --------------------
   556→
   557→  const [subject, setSubject] = useState('');
   558→  const [htmlBody, setHtmlBody] = useState('');
   559→  const [settings, setSettings] = useState<TemplateSettings>({
   560→    category: 'SYSTEM',
   561→    status: 'DRAFT',
   562→  });
   563→  const [templateName, setTemplateName] = useState('');
   564→  const [saveState, setSaveState] = useState<SaveState>('idle');
   565→  const [isFullscreen, setIsFullscreen] = useState(false);
   566→  const [showSearch, setShowSearch] = useState(false);
   567→  const [sidebarOpen, setSidebarOpen] = useState(true);
   568→  const [originalData, setOriginalData] = useState<{ name: string; subject: string; htmlBody: string; settings: TemplateSettings } | null>(null);
   569→
   570→  // Refs
   571→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   572→  const autoSaveTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   573→
   574→  // -------------------- Track dirty --------------------
   575→
   576→  const isDirty = useMemo(() => {
   577→    if (!originalData) return false;
   578→    return (
   579→      originalData.name !== templateName ||
   580→      originalData.subject !== subject ||
   581→      originalData.htmlBody !== htmlBody ||
   582→      JSON.stringify(originalData.settings) !== JSON.stringify(settings)
   583→    );
   584→  }, [templateName, subject, htmlBody, settings, originalData]);
   585→
   586→  // Derive display save state: saving/saved take priority over dirty/idle
   587→  const displaySaveState = useMemo((): SaveState => {
   588→    if (saveState === 'saving' || saveState === 'saved') return saveState;
   589→    return isDirty ? 'dirty' : 'idle';
   590→  }, [isDirty, saveState]);
   591→
   592→  // -------------------- Data Fetching --------------------
   593→
   594→  const { data: template, isLoading: isLoadingTemplate } = useQuery<EmailTemplate>({
   595→    queryKey: queryKeys.emailTemplates.detail(templateId),
   596→    queryFn: () => getApi<EmailTemplate>(`/api/email-templates/${templateId}`),
   597→    enabled: !!templateId && !isNew,
   598→  });
   599→
   600→  // -------------------- Create Mutation (for new templates) --------------------
   601→
   602→  const createMutation = useMutation({
   603→    mutationFn: (data: {
   604→      name: string;
   605→      subject: string;
   606→      htmlBody: string;
   607→      category?: EmailTemplateCategory;
   608→      status?: EmailTemplateStatus;
   609→    }) => postApi<EmailTemplate>('/api/email-templates', {
   610→      ...data,
   611→      createdById: currentUser?.id,
   612→      // Platform scope is guarded by requirePlatformAdmin on the server and
   613→      // creates a system-level template (siteId = null). Client scope omits
   614→      // the param so the legacy client behavior is preserved.
   615→      ...(isPlatform ? { scope: 'platform' } : {}),
   616→    }),
   617→    onSuccess: (created) => {
   618→      // Invalidate the scope-appropriate list cache. Platform uses the
   619→      // platform-prefixed key; client uses the legacy key. Also invalidate
   620→      // the detail cache (scope-agnostic, keyed by id).
   621→      if (isPlatform) {
   622→        queryClient.invalidateQueries({ queryKey: ['email-templates', 'list', 'platform'] });
   623→        queryClient.invalidateQueries({ queryKey: ['email-templates', 'category-counts', 'platform'] });
   624→      } else {
   625→        queryClient.invalidateQueries({ queryKey: queryKeys.emailTemplates.all });
   626→        queryClient.invalidateQueries({ queryKey: ['email-templates', 'category-counts', 'client'] });
   627→      }
   628→      toast.success('Template created successfully');
   629→      onCreated?.(created.id);
   630→    },
   631→    onError: (err: Error) => {
   632→      setSaveState('idle');
   633→      toast.error(err.message || 'Failed to create template');
   634→    },
   635→  });
   636→
   637→  // -------------------- Sync template data to state --------------------
   638→
   639→  useEffect(() => {
   640→    if (template) {
   641→      const newSettings: TemplateSettings = {
   642→        category: template.category,
   643→        status: template.status,
   644→      };
