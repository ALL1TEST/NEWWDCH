   183→   183→          email: '',

   186→   186→          pagePermissions: DEFAULT_EDITOR_PAGES,

   194→   194→    setForm((p) => ({
   195→   195→      ...p,
   196→   196→      role: role as UserRole,

   216→   216→      setErrors(errs);
   217→   217→      return;
   218→   218→    }
   219→   219→    onSubmit({ ...form, email: form.email.trim(), name: form.name.trim() });

   222→   222→  const isEditor = form.role === 'EDITOR';

   229→   229→            <DialogTitle>{editMode ? 'Edit User' : 'Invite User'}</DialogTitle>

   231→   231→              {editMode
   232→   232→                ? 'Update user details, role, and page access.'
   233→   233→                : 'Send an invitation email to add a new team member to your organization.'}
   234→   234→            </DialogDescription>
   235→   235→          </DialogHeader>
   236→   236→          <div className="grid gap-6 py-2">
   237→   237→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   238→   238→              <div className="grid gap-2">
   239→   239→                <Label htmlFor="invite-email">Email <span className="text-destructive">*</span></Label>
   240→   240→                <Input
   241→   241→                  id="invite-email"
   242→   242→                  type="email"
   243→   243→                  value={form.email}
   244→   244→                  onChange={(e) => {
   245→   245→                    setForm((p) => ({ ...p, email: e.target.value }));
   246→   246→                    setErrors((p) => { const n = { ...p }; delete n.email; return n; });
   247→   247→                  }}
   248→   248→                  placeholder="user@example.com"
   249→   249→                  autoFocus
   250→   250→                />
   251→   251→                {errors.email && <p className="text-xs text-destructive">{errors.email}</p>}
   252→   252→              </div>
   253→   253→              <div className="grid gap-2">
   254→   254→                <Label htmlFor="invite-name">Name</Label>
   255→   255→                <Input
   256→   256→                  id="invite-name"
   257→   257→                  value={form.name}
   258→   258→                  onChange={(e) => setForm((p) => ({ ...p, name: e.target.value }))}
   259→   259→                  placeholder="Full name (optional)"
   260→   260→                />
   261→   261→              </div>
   262→   262→            </div>
   263→   263→
   264→   264→            <div className="grid gap-2">
   265→   265→              <Label htmlFor="invite-role">Role</Label>

   268→   268→                  <SelectValue placeholder="Select a role" />
   269→   269→                </SelectTrigger>
   270→   270→                <SelectContent>
   271→   271→                  {ROLE_OPTIONS.map((opt) => (
   272→   272→                    <SelectItem key={opt.value} value={opt.value}>
   273→   273→                      {opt.label}
   274→   274→                    </SelectItem>
   275→   275→                  ))}
   276→   276→                </SelectContent>
   277→   277→              </Select>
   278→   278→              <p className="text-xs text-muted-foreground">
   279→   279→                <strong>Admin</strong> has full access to all pages. <strong>Editor</strong> only sees the pages
   280→   280→                selected below.
   281→   281→              </p>
   282→   282→            </div>
   283→   283→
   284→   284→            {isEditor ? (
   285→   285→              <div className="grid gap-2">
   286→   286→                <div className="flex items-center justify-between">
   287→   287→                  <Label>Page Access</Label>
   288→   288→                  <span className="text-xs text-muted-foreground">
   289→   289→                    {form.pagePermissions.length} selected
   290→   290→                  </span>
   291→   291→                </div>
   292→   292→                <div className="border rounded-lg divide-y max-h-72 overflow-y-auto">
   293→   293→                  {BUILTIN_PAGES.map((page) => {
   294→   294→                    const checked = form.pagePermissions.includes(page.key);
   295→   295→                    const isSettings = page.key === 'settings';
   296→   296→                    return (
   297→   297→                      <div key={page.key}>
   298→   298→                        <label className="flex items-center gap-3 px-3 py-2.5 hover:bg-accent/50 cursor-pointer transition-colors">
   299→   299→                          <Checkbox
   300→   300→                            checked={checked}
   301→   301→                            onCheckedChange={() => togglePage(page.key)}
   302→   302→                          />
   303→   303→                          <span className="text-sm font-medium flex-1">{page.label}</span>
   304→   304→                          {isSettings && (
   305→   305→                            <span className="text-[10px] uppercase tracking-wide text-muted-foreground">
   306→   306→                              grants all sub-pages
   307→   307→                            </span>
   308→   308→                          )}

   312→   312→                            {SETTINGS_SUBPAGES.map((sub) => {
   313→   313→                              const subChecked = form.pagePermissions.includes(sub.key);
   314→   314→                              return (
   315→   315→                                <label
   316→   316→                                  key={sub.key}
   317→   317→                                  className="flex items-center gap-3 px-3 py-1.5 hover:bg-accent/50 cursor-pointer transition-colors"
   318→   318→                                >
   319→   319→                                  <Checkbox
   320→   320→                                    checked={subChecked}
   321→   321→                                    onCheckedChange={() => togglePage(sub.key)}
   322→   322→                                  />
   323→   323→                                  <span className="text-sm text-muted-foreground">{sub.label}</span>
   324→   324→                                </label>
   325→   325→                              );
   326→   326→                            })}
   327→   327→                          </div>
   328→   328→                        )}
   329→   329→                      </div>
   330→   330→                    );
   331→   331→                  })}
   332→   332→                </div>
   333→   333→              </div>
   334→   334→            ) : (
   335→   335→              <div className="flex items-center gap-2 p-3 rounded-md bg-orange-50 border border-orange-200 text-orange-800 dark:bg-orange-950/30 dark:border-orange-900/50 dark:text-orange-300">

   342→   342→          </div>

   344→   344→            <Button variant="outline" onClick={() => onOpenChange(false)} disabled={isLoading}>

   363→   363→  // Table state

   389→   389→    }),
   390→   390→    [
   391→   391→      table.currentPage,
   392→   392→      table.pageSize,
   393→   393→      table.sortField,
   394→   394→      table.sortOrder,

   404→   404→    queryFn: () =>
   405→   405→      getApi<{ data: UserRow[]; meta: { pagination: { page: number; pageSize: number; total: number; totalPages: number } } }>(
   406→   406→        '/api/users',

   424→   424→  });

   443→   443→        pagePermissions: data.pagePermissions,

   468→   468→        name: data.name,
   469→   469→        email: data.email,
   470→   470→        role: data.role,
   471→   471→        pagePermissions: data.pagePermissions,
   472→   472→      }),

   474→   474→      queryClient.invalidateQueries({ queryKey: queryKeys.users.all });
   475→   475→      setEditDialogOpen(false);
   476→   476→      setEditingUser(null);
   477→   477→      toast.success('User updated successfully');
   478→   478→    },
   479→   479→    onError: (err: Error) => {

   487→   487→        user.status === 'SUSPENDED' ? 'ACTIVE' : 'SUSPENDED';
   488→   488→      toggleStatusMutation.mutate({ id: user.id, status: newStatus });
   489→   489→      setSuspendTarget(null);

   491→   491→    [toggleStatusMutation],
   492→   492→  );
   493→   493→
   494→   494→  // Column definitions
   495→   495→  const columns = useMemo<ColumnDef<UserRow>[]>(
   496→   496→    () => [
