   146→// -------------------- Collapsed-Rail Tooltip Positioning -------------------
   147→/*
   148→ * SINGLE source of truth for the positioning of every tooltip/label that
   149→ * appears when the sidebar is collapsed into the 48px icon rail.
   150→ *
   151→ *   side="right"         → opens to the RIGHT of the 48px collapsed rail,
   152→ *                          fully inside the main viewport (never overlaps
   153→ *                          the rail, never opens to the left/outside).
   154→ *   align="center"       → vertically centers the bubble on the icon's
   155→ *                          32×32 hover target so labels line up across
   156→ *                          every row of the rail (logo, theme, nav items,
   157→ *                          parent items with submenu).
   158→ *   sideOffset=8         → ~8px visible gap from the trigger button's right
   159→ *                          edge (button right edge ~x=40 inside the 48px
   160→ *                          rail, tooltip left edge ~x=48 = the rail's
   161→ *                          right edge — flush with the RAIL, NOT with the
   162→ *                          icon, so the bubble never touches the glyph).
   163→ *   collisionPadding=12 → 12px viewport-edge collision padding so the
   164→ *                          bubble can never be clipped at any viewport
   165→ *                          edge or scroll past the visible area.
   166→ *
   167→ * Portal-based rendering (Radix TooltipPrimitive.Portal → document.body,
   168→ * wired inside src/components/ui/tooltip.tsx) means every tooltip is
   169→ * rendered OUTSIDE the sidebar DOM tree — so the sidebar's
   170→ * overflow-hidden / overflow-x-hidden on the rail container can NEVER
   171→ * clip or hide the bubble. The tooltip floats above every other
   172→ * element at z-50.
   173→ *
   174→ * Used by:
   175→ *   • CollapsedLogoButton         (header "Expand" tooltip)
   176→ *   • SimpleNavItem              (leaf nav items: Dashboard, Articles,
   177→ *                                  Calendar, Media, Users, Comments,
   178→ *                                  Newsletter, SEO, AI, Automation)
   179→ *   • ExpandableNavItem          (parent nav items in EXPANDED state —
   180→ *                                  tooltip is hidden by SidebarMenuButton
   181→ *                                  when state!=="collapsed", but the
   182→ *                                  positioning object is still applied
   183→ *                                  so the prop stays consistent across
   184→ *                                  every nav item renderer)
   185→ *   • CollapsedParentNavItem     (parent nav items in COLLAPSED state —
   186→ *                                  label tooltip shows when the floating
   187→ *                                  popover is CLOSED; when the popover is
   188→ *                                  open, no tooltip shows to avoid visual
   189→ *                                  conflict)
   190→ *
   191→ * The ThemeToggle component (src/components/layout/theme-toggle.tsx) lives
   192→ * in a separate file and inlines the SAME four values — see the comment
   193→ * there for the rationale.
   194→ */

   287→/**
   288→ * Derive the top-level section ID from the current browser hash.
   289→ * Uses EXPLICIT prefix matching — #settings/seo always returns 'Settings', never 'SEO'.

   291→function getSectionFromHash(hash: string): string | null {
   292→  const path = hash.replace(/^#\/?,?/, '').trim();
   293→  if (!path || path === 'dashboard') return null;
   294→
   295→  const firstSegment = path.split('/')[0];
   296→  return ROUTE_PREFIX_TO_SECTION[firstSegment] ?? null;
   297→}

   348→/**
   349→ * Collapsed-rail logo cell. Since the extra collapse-toggle icon was removed
   350→ * from below the logo, THIS cell is the explicit expand control:
   351→ *   • identical 32×32 grid geometry as every other rail icon
   352→ *   • click toggles the sidebar open (SidebarRail edge strip still works)
   353→ * No new icon is introduced and no empty clickable ghost remains.
   354→ *
   355→ * HOVER BEHAVIOR (collapsed state only — this button is rendered ONLY
   356→ * inside the collapsed-rail cluster, so the expanded LogoMark is untouched):
   357→ * The `hovered` prop is LIFTED to the rail level (see AppSidebar's
   358→ * `railHovered` state + the onMouseEnter/onMouseLeave handlers on the
   359→ * <Sidebar> root). This means the swap fires when the mouse is ANYWHERE
   360→ * over the collapsed 48px rail — NOT only when the pointer is directly
   361→ * over the logo cell:
   362→ *   • rail at rest ........  shows the "C" logo mark on its bg-primary
   363→ *                            black box (NORMAL LOGO — background UNCHANGED,
   364→ *                            same as the expanded LogoMark).
   365→ *   • mouse over ANY rail   the "C" is replaced by a PanelLeftOpen icon
   366→ *     item (Dashboard,       AND the button's background becomes
   367→ *     Articles, …, Profile,  transparent so ONLY the Expand icon is
   368→ *     Theme, Settings, …)    visible with no box behind it. The icon
   369→ *                            color switches to muted-foreground (gray)
   370→ *                            so it stays visible against the
   371→ *                            transparent/page bg in both Light and Dark.
   372→ *   • mouse leaves the     restores the "C" logo mark AND its bg-primary
   373→ *     entire rail            box (onMouseLeave on the <Sidebar> root).
   374→ *   • click ............. calls toggleSidebar (existing functionality,
   375→ *                         unchanged).
   376→ *
   377→ * Each rail item keeps its OWN Radix Tooltip label (Dashboard, Articles,
   378→ * Profile, etc.) — the lifted `hovered` prop ONLY drives the logo cell's
   379→ * C↔PanelLeftOpen swap; it does NOT touch any item's tooltip. The logo's
   380→ * own "Expand" Tooltip still fires ONLY when the pointer is directly over
   381→ * the logo button (the Tooltip trigger is the button itself, independent
   382→ * of the `hovered` prop).
   383→ *
   384→ * Background is removed ONLY from the temporary Expand-icon state; the
   385→ * normal logo (at rest) keeps its bg-primary background exactly as-is.
   386→ *
   387→ * Implementation: React state lifted to AppSidebar (railHovered +
   388→ * onMouseEnter/onMouseLeave on <Sidebar>) — NOT CSS :hover / group-hover
   389→ * variants. This is deliberate: Tailwind v4 wraps its hover:/group-hover:
   390→ * utilities inside @media (hover: hover), so on any browser that reports
   391→ * (hover: none) (headless, touch, some preview iframes) those utilities
   392→ * NEVER activate and the swap + bg-transparent would silently fail. React
   393→ * mouse events fire on ANY pointer input regardless of the hover media
   394→ * query, so the behavior is identical in every environment. The
   395→ * conditional className + conditional render swap the bg/color/glyph
   396→ * atomically per state. Works in both Light and Dark.
   397→ */
   398→function CollapsedLogoButton({ hovered }: { hovered: boolean }) {
   399→  const { toggleSidebar } = useSidebar();
   400→  return (
   401→    /* Hover-only "Expand" tooltip: instant (provider delayDuration=0),
   402→       appears on mouse-enter, disappears on mouse-leave. Radix closes it
   403→       on any pointer exit — disableHoverableContent guarantees it can
   404→       never linger or trap the pointer. The Tooltip trigger is the logo
   405→       button itself, so the "Expand" TEXT fires ONLY on direct logo hover
   406→       (NOT on rail-level hover of other items) — the C↔PanelLeftOpen ICON
   407→       swap, by contrast, is driven by the lifted `hovered` prop and fires
   408→       for ANY rail item. */
   409→    <Tooltip disableHoverableContent>
   410→      <TooltipTrigger asChild>
   411→        <button
   412→          type="button"
   413→          onClick={toggleSidebar}
   414→          aria-label="Expand sidebar"
   415→          className={cn(
   416→            'flex h-8 w-8 shrink-0 cursor-pointer items-center justify-center rounded-lg font-bold text-sm outline-none transition-opacity hover:opacity-90 focus-visible:ring-2 focus-visible:ring-ring select-none',
   417→            // NORMAL LOGO STATE (at rest): bg-primary black box +
   418→            // text-primary-foreground white "C" — background UNCHANGED,
   419→            // same as the expanded LogoMark.
   420→            // EXPAND-ICON STATE (on hover): bg-transparent (NO background)
   421→            // + text-muted-foreground gray icon — ONLY this temporary
   422→            // state loses the background. The icon stays visible (gray
   423→            // on transparent/page bg) in both Light and Dark mode.
   424→            hovered
   425→              ? 'bg-transparent text-muted-foreground'
   426→              : 'bg-primary text-primary-foreground',
   427→          )}
   428→        >
   429→          {/* Conditional render: "C" logo at rest, PanelLeftOpen icon on
   430→              hover. h-4 w-4 matches the CollapseToggle's PanelLeftClose
   431→              icon size; the icon is the visual opposite of the collapse
   432→              icon so the collapsed logo reads as a clear "expand"
   433→              affordance when the pointer is over it. */}
   434→          {hovered ? (
   435→            <PanelLeftOpen className="h-4 w-4" />
   436→          ) : (
   437→            <span>C</span>
   438→          )}
   439→          <span className="sr-only">Expand sidebar</span>
   440→        </button>
   441→      </TooltipTrigger>
   442→      {/* Collapsed-rail tooltip — uses the SHARED COLLAPSED_TOOLTIP_PROPS
   443→          constant so the "Expand" label has IDENTICAL positioning (side,
   444→          align, sideOffset, collisionPadding) to every other collapsed-
   445→          rail tooltip (ThemeToggle, SimpleNavItem, ExpandableNavItem,
   446→          CollapsedParentNavItem). Portal-based rendering means the bubble
   447→          floats outside the sidebar DOM at z-50 — never clipped by the
   448→          rail's overflow-hidden. */}
   449→      <TooltipContent {...COLLAPSED_TOOLTIP_PROPS}>Expand</TooltipContent>
   450→    </Tooltip>
   451→  );
   452→}

   522→/**
   523→ * Settings parent item rendered on the COLLAPSED 48px rail.
   524→ * Clicking it opens a portal-based popover to the RIGHT of the icon —
   525→ * the sidebar itself NEVER expands, and the popover anchors to this
   526→ * exact button so the "star/sparkle alignment" grid stays intact.
   527→ */
   528→function CollapsedParentNavItem({
   529→  item,
   530→  currentModule,
   531→  currentSubPage,
   532→}: {
   533→  item: NavItem;
   534→  currentModule: string;
   535→  currentSubPage: string | null;
   536→}) {
   537→  const [floatOpen, setFloatOpen] = useState(false);
   538→  const mod = hrefToModule(item.href);
   539→  const isActive = currentModule === mod;
   540→
   541→  const handleChildNavigate = (
   542→    e: React.MouseEvent<HTMLAnchorElement>,
   543→    child: NavItem,
   544→  ) => {
   545→    e.preventDefault();
   546→    e.stopPropagation();
   547→    setFloatOpen(false); // close the floating submenu after navigation…
   548→    // …but do NOT touch sidebar expansion state (rule #7).
   549→    const hash = child.href.replace(/^#/, '');
   550→    const parts = hash.split('/');
   551→    const childSubPage = parts[1] || null;
   552→    useNavigationStore.getState().navigate(parts[0], null, childSubPage);
   553→  };
   554→
   555→  return (
   556→    <SidebarMenuItem>
   557→      <Popover open={floatOpen} onOpenChange={setFloatOpen}>
   558→        <PopoverTrigger asChild>
   559→          <SidebarMenuButton
   560→            isActive={isActive || floatOpen}
   561→            aria-expanded={floatOpen}
   562→            aria-haspopup="menu"
   563→            // Collapsed-rail tooltip — uses the SHARED COLLAPSED_TOOLTIP_PROPS
   564→            // so the parent item's label has IDENTICAL positioning (side,
   565→            // align, sideOffset, collisionPadding) to every other collapsed-
   566→            // rail tooltip (CollapsedLogoButton, ThemeToggle, SimpleNavItem,
   567→            // ExpandableNavItem). Suppressed when the popover is open so the
   568→            // floating submenu never visually conflicts with a lingering
   569→            // label bubble. Portal-based rendering means the bubble floats
   570→            // outside the sidebar DOM at z-50 — never clipped by the rail's
   571→            // overflow-hidden.
   572→            tooltip={
   573→              floatOpen
   574→                ? undefined
   575→                : { ...COLLAPSED_TOOLTIP_PROPS, children: item.label }
   576→            }
   577→            onClick={(e: React.MouseEvent) => {
   578→              e.preventDefault();
   579→              setFloatOpen((o) => !o);
   580→            }}
   581→          >
   582→            <NavIcon name={item.icon} />
   583→            <span>{item.label}</span>
   584→          </SidebarMenuButton>
   585→        </PopoverTrigger>

   631→function ExpandableNavItem({
   632→  item,
   633→  currentModule,
   634→  currentSubPage,
   635→  isExpanded,
   636→  onToggle,
   637→}: {
   638→  item: NavItem;
   639→  currentModule: string;
   640→  currentSubPage: string | null;
   641→  isExpanded: boolean;
   642→  onToggle: () => void;
   643→}) {
   644→  const mod = hrefToModule(item.href);
   645→  const isActive = currentModule === mod;
   646→  const sectionId = `submenu-${item.label.toLowerCase().replace(/\s+/g, '-')}`;
   647→
   648→  return (
   649→    <SidebarMenuItem>
   650→      <SidebarMenuButton
   651→        // Expandable parent item rendered in EXPANDED state. SidebarMenuButton
   652→        // hides its tooltip whenever state!=="collapsed" (see ui/sidebar.tsx),
   653→        // so this label never shows while expanded — but passing the SAME
   654→        // COLLAPSED_TOOLTIP_PROPS object as every other nav-item renderer
   655→        // keeps the prop consistent across the codebase. If the sidebar
   656→        // toggles between expanded and collapsed while this row stays
   657→        // mounted, the tooltip that appears is identical in positioning to
   658→        // SimpleNavItem / CollapsedParentNavItem / CollapsedLogoButton.
   659→        tooltip={{ ...COLLAPSED_TOOLTIP_PROPS, children: item.label }}
   660→        isActive={isActive}
   661→        onClick={(e: React.MouseEvent) => {
   662→          e.preventDefault();
   663→          onToggle();
   664→        }}
   665→        aria-expanded={isExpanded}
   666→        aria-controls={sectionId}
   667→      >
   668→        <NavIcon name={item.icon} />
   669→        <span>{item.label}</span>
   670→        <ChevronRight
   671→          className={cn(
   672→            'ml-auto h-4 w-4 shrink-0 transition-transform duration-200',
   673→            isExpanded && 'rotate-90',
   674→          )}
   675→        />
   676→      </SidebarMenuButton>
   677→
   678→      {item.badge != null && (
   679→        <SidebarMenuBadge>
   680→          {item.badge}
   681→        </SidebarMenuBadge>
   682→      )}
   683→
   684→      <AccordionSubmenu isOpen={isExpanded} sectionLabel={item.label}>
   685→        {item.children!.map((child) => {
   686→          const hash = child.href.replace(/^#/, '');
   687→          const parts = hash.split('/');
   688→          const childMod = parts[0];
   689→          const childSubPage = parts[1] || null;
   690→          const isChildActive =
   691→            currentModule === childMod &&
   692→            (!childSubPage || currentSubPage === childSubPage);
   693→
   694→          return (
   695→            <SidebarMenuSubItem key={child.label}>
   696→              <SidebarMenuSubButton
   697→                asChild
   698→                isActive={isChildActive}
   699→              >
   700→                <a
   701→                  href={child.href}
   702→                  onClick={(e) => {
   703→                    e.preventDefault();
   704→                    e.stopPropagation();
   705→                    useNavigationStore
   706→                      .getState()
   707→                      .navigate(childMod, null, childSubPage);
   708→                  }}
   709→                >
   710→                  <NavIcon name={child.icon} />
   711→                  <span>{child.label}</span>
   712→                </a>
   713→              </SidebarMenuSubButton>
   714→            </SidebarMenuSubItem>
   715→          );
   716→        })}
   717→      </AccordionSubmenu>
   718→    </SidebarMenuItem>
   719→  );

   771→function NavGroupSection({
   772→  group,
   773→  currentModule,
   774→  currentSubPage,
   775→  openSection,
   776→  onToggleSection,
   777→}: {
   778→  group: NavGroup;
   779→  currentModule: string;

   794→              /* Collapsed rail → parents with children become floating
   795→                 popovers anchored to their own icon. The rail NEVER

   799→                  key={item.label}

   801→                  currentModule={currentModule}
   802→                  currentSubPage={currentSubPage}

   805→            }

   810→                  key={item.label}

   812→                  currentModule={currentModule}
   813→                  currentSubPage={currentSubPage}

   821→              <SimpleNavItem
