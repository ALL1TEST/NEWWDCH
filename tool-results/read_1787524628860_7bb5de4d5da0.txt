   100→   100→// -------------------- Sentiment Detection --------------------

   102→   102→const POSITIVE_WORDS = new Set([
   103→   103→  'great', 'love', 'excellent', 'amazing', 'good', 'nice', 'best',
   104→   104→  'wonderful', 'perfect', 'awesome',
   105→   105→]);
   106→   106→
   107→   107→const NEGATIVE_WORDS = new Set([
   108→   108→  'bad', 'terrible', 'horrible', 'worst', 'hate', 'awful', 'poor',
   109→   109→  'disgusting', 'waste', 'disappointed',
   110→   110→]);
   111→   111→
   112→   112→function detectSentiment(text: string): SentimentType {
   113→   113→  const lower = text.toLowerCase();
   114→   114→  const words = lower.split(/\W+/);
   115→   115→  let posCount = 0;
   116→   116→  let negCount = 0;
   117→   117→  for (const w of words) {
   118→   118→    if (POSITIVE_WORDS.has(w)) posCount++;
   119→   119→    if (NEGATIVE_WORDS.has(w)) negCount++;
   120→   120→  }
   121→   121→  if (posCount > negCount) return 'positive';
   122→   122→  if (negCount > posCount) return 'negative';
   123→   123→  return 'neutral';
   124→   124→}
