    52→export async function executeChat(req: ChatRequest): Promise<ChatResponse> {
    53→  const provider = await db.aiProvider.findUnique({
    54→    where: { id: req.providerId },
    55→    include: { models: true },
    56→  });
    57→
    58→  if (!provider) throw new Error('Provider not found');
    59→  if (!provider.isActive) throw new Error('Provider is disabled');
    60→  if (!provider.apiKeyEncrypted) throw new Error('API key not configured');
    61→
    62→  const apiKey = await decrypt(provider.apiKeyEncrypted);
    63→  const config = getProviderConfig(provider.kind);
    64→  const baseUrl = provider.baseUrl || config.defaultBaseUrl;
    65→  const modelId = req.modelId || provider.models.find(m => m.isDefault)?.modelId || provider.models[0]?.modelId;
    66→  if (!modelId) throw new Error('No model available');
    67→
    68→  const startTime = Date.now();
    69→  let inputTokens = 0;
    70→  let outputTokens = 0;
    71→  let content = '';
    72→
    73→  try {
    74→    if (provider.kind === 'ANTHROPIC') {
    75→      const result = await callAnthropic(baseUrl, apiKey, modelId, req.messages, {
    76→        temperature: req.temperature,
    77→        maxTokens: req.maxTokens,
    78→        jsonMode: req.jsonMode,
    79→      });
    80→      inputTokens = result.inputTokens;
    81→      outputTokens = result.outputTokens;
    82→      content = result.content;
    83→    } else if (provider.kind === 'GEMINI') {
    84→      const result = await callGemini(baseUrl, apiKey, modelId, req.messages, {
    85→        temperature: req.temperature,
    86→        maxTokens: req.maxTokens,
    87→        jsonMode: req.jsonMode,
    88→      });
    89→      inputTokens = result.inputTokens;
    90→      outputTokens = result.outputTokens;
    91→      content = result.content;
    92→    } else if (provider.kind === 'AZURE_OPENAI') {
    93→      const azureConfig = JSON.parse(provider.config || '{}');
    94→      const result = await callOpenAI(
    95→        `${baseUrl}/openai/deployments/${azureConfig.deployment || modelId}`,
    96→        apiKey,
    97→        modelId,
    98→        req.messages,
    99→        { temperature: req.temperature, maxTokens: req.maxTokens, jsonMode: req.jsonMode, apiVersion: provider.apiVersion || '2024-02-01' },
   100→      );
   101→      inputTokens = result.inputTokens;
   102→      outputTokens = result.outputTokens;
   103→      content = result.content;
   104→    } else {
   105→      // OpenAI-compatible (OpenAI, Groq, DeepSeek, OpenRouter, Ollama)
   106→      const result = await callOpenAI(baseUrl, apiKey, modelId, req.messages, {
   107→        temperature: req.temperature,
   108→        maxTokens: req.maxTokens,
   109→        topP: req.topP,
   110→        frequencyPenalty: req.frequencyPenalty,
   111→        presencePenalty: req.presencePenalty,
   112→        jsonMode: req.jsonMode,
   113→      });
   114→      inputTokens = result.inputTokens;
   115→      outputTokens = result.outputTokens;
   116→      content = result.content;
   117→    }
   118→  } catch (err) {
   119→    // Try fallback providers
   120→    const fallbacks = await db.aiProviderFallback.findMany({
   121→      where: { providerId: provider.id },
   122→      include: { fallback: { include: { models: true } } },
   123→      orderBy: { priority: 'asc' },
   124→    });
   125→
   126→    for (const fb of fallbacks) {
   127→      if (!fb.fallback.isActive || !fb.fallback.apiKeyEncrypted) continue;
   128→      try {
   129→        const fbApiKey = await decrypt(fb.fallback.apiKeyEncrypted);
   130→        const fbConfig = getProviderConfig(fb.fallback.kind);
   131→        const fbBaseUrl = fb.fallback.baseUrl || fbConfig.defaultBaseUrl;
   132→        const fbModel = req.modelId || fb.fallback.models.find(m => m.isDefault)?.modelId || fb.fallback.models[0]?.modelId;
   133→        if (!fbModel) continue;
   134→
   135→        const result = await callOpenAI(fbBaseUrl, fbApiKey, fbModel, req.messages, {
   136→          temperature: req.temperature,
   137→          maxTokens: req.maxTokens,
   138→          jsonMode: req.jsonMode,
   139→        });
   140→        inputTokens = result.inputTokens;
   141→        outputTokens = result.outputTokens;
   142→        content = result.content;
   143→        break;
   144→      } catch {
   145→        continue;
   146→      }
   147→    }
   148→
   149→    if (!content) throw err;
   150→  }
   151→
   152→  const durationMs = Date.now() - startTime;
   153→  const totalTokens = inputTokens + outputTokens;
   154→  const model = provider.models.find(m => m.modelId === modelId);
   155→  const costUsd = model
   156→    ? (inputTokens / 1000) * (model.inputCostPer1k || 0) + (outputTokens / 1000) * (model.outputCostPer1k || 0)
   157→    : 0;
   158→
   159→  // Update provider
   160→  await db.aiProvider.update({
   161→    where: { id: provider.id },
   162→    data: {
   163→      lastUsedAt: new Date(),
   164→      latencyMs: durationMs,
   165→      connectionStatus: 'CONNECTED',
   166→      lastError: null,
   167→    },
   168→  });
   169→
   170→  // Log the request
   171→  await db.aiLog.create({
   172→    data: {
   173→      providerId: provider.id,
   174→      providerName: provider.name,
   175→      modelId,
   176→      question: req.messages.map(m => m.content).join('\n'),
   177→      response: content,
   178→      inputTokens,
   179→      outputTokens,
   180→      totalTokens,
   181→      costUsd,
   182→      durationMs,
   183→      status: 'success',
   184→      siteId: req.siteId,
   185→      userId: req.userId,
   186→    },
   187→  });
   188→
   189→  return {
   190→    content,
   191→    model: modelId,
   192→    inputTokens,
   193→    outputTokens,
   194→    totalTokens,
   195→    costUsd,
   196→    durationMs,
   197→    providerName: provider.name,
   198→  };
   199→}

   201→// -------------------- Provider-specific call implementations --------------------
   202→
   203→async function callOpenAI(
   204→  baseUrl: string,
   205→  apiKey: string,
   206→  model: string,
   207→  messages: ChatMessage[],
   208→  opts: { temperature?: number; maxTokens?: number; topP?: number; frequencyPenalty?: number; presencePenalty?: number; jsonMode?: boolean; apiVersion?: string } = {},
   209→) {
   210→  const url = opts.apiVersion
   211→    ? `${baseUrl}?api-version=${opts.apiVersion}`
   212→    : `${baseUrl}/chat/completions`;
   213→
   214→  const body: Record<string, unknown> = {
   215→    model,
   216→    messages,
   217→    temperature: opts.temperature ?? 0.7,
   218→    max_tokens: opts.maxTokens ?? 2048,
   219→  };
   220→  if (opts.topP !== undefined) body.top_p = opts.topP;
   221→  if (opts.frequencyPenalty !== undefined) body.frequency_penalty = opts.frequencyPenalty;
   222→  if (opts.presencePenalty !== undefined) body.presence_penalty = opts.presencePenalty;
   223→  if (opts.jsonMode) {
   224→    body.response_format = { type: 'json_object' };
   225→  }
   226→
   227→  const res = await fetch(url, {
   228→    method: 'POST',
   229→    headers: {
   230→      'Content-Type': 'application/json',
   231→      'Authorization': `Bearer ${apiKey}`,
   232→    },
   233→    body: JSON.stringify(body),
   234→  });
   235→
   236→  if (!res.ok) {
   237→    const err = await res.text();
   238→    throw new Error(`OpenAI API error: ${res.status} — ${err}`);
   239→  }
   240→
   241→  const data = await res.json();
   242→  return {
   243→    content: data.choices?.[0]?.message?.content || '',
   244→    inputTokens: data.usage?.prompt_tokens || 0,
   245→    outputTokens: data.usage?.completion_tokens || 0,
   246→  };
   247→}

   249→async function callAnthropic(
   250→  baseUrl: string,
   251→  apiKey: string,
   252→  model: string,
   253→  messages: ChatMessage[],
   254→  opts: { temperature?: number; maxTokens?: number; jsonMode?: boolean } = {},
   255→) {
   256→  const systemMsg = messages.find(m => m.role === 'system');
   257→  const chatMsgs = messages.filter(m => m.role !== 'system');
   258→
   259→  const body: Record<string, unknown> = {
   260→    model,
   261→    messages: chatMsgs.map(m => ({ role: m.role, content: m.content })),
   262→    max_tokens: opts.maxTokens ?? 2048,
   263→    temperature: opts.temperature ?? 0.7,
   264→  };
   265→  if (systemMsg) body.system = systemMsg.content;
   266→
   267→  const res = await fetch(`${baseUrl}/messages`, {
   268→    method: 'POST',
   269→    headers: {
   270→      'Content-Type': 'application/json',
   271→      'x-api-key': apiKey,
   272→      'anthropic-version': '2023-06-01',
   273→    },
   274→    body: JSON.stringify(body),
   275→  });
   276→
   277→  if (!res.ok) {
   278→    const err = await res.text();
   279→    throw new Error(`Anthropic API error: ${res.status} — ${err}`);
   280→  }
   281→
   282→  const data = await res.json();
   283→  return {
   284→    content: data.content?.[0]?.text || '',
   285→    inputTokens: data.usage?.input_tokens || 0,
   286→    outputTokens: data.usage?.output_tokens || 0,
   287→  };
   288→}

   290→async function callGemini(
   291→  baseUrl: string,
   292→  apiKey: string,
   293→  model: string,
   294→  messages: ChatMessage[],
   295→  opts: { temperature?: number; maxTokens?: number; jsonMode?: boolean } = {},
   296→) {
   297→  const systemMsg = messages.find(m => m.role === 'system');
   298→  const chatMsgs = messages.filter(m => m.role !== 'system');
   299→
   300→  const body: Record<string, unknown> = {
   301→    contents: chatMsgs.map(m => ({
   302→      role: m.role === 'assistant' ? 'model' : 'user',
   303→      parts: [{ text: m.content }],
   304→    })),
   305→    generationConfig: {
   306→      temperature: opts.temperature ?? 0.7,
   307→      maxOutputTokens: opts.maxTokens ?? 2048,
   308→    },
   309→  };
   310→
   311→  const res = await fetch(
   312→    `${baseUrl}/models/${model}:generateContent?key=${apiKey}`,
   313→    {
   314→      method: 'POST',
   315→      headers: { 'Content-Type': 'application/json' },
   316→      body: JSON.stringify(body),
   317→    },
   318→  );
   319→
   320→  if (!res.ok) {
   321→    const err = await res.text();
   322→    throw new Error(`Gemini API error: ${res.status} — ${err}`);
   323→  }
   324→
   325→  const data = await res.json();
   326→  return {
   327→    content: data.candidates?.[0]?.content?.parts?.[0]?.text || '',
   328→    inputTokens: data.usageMetadata?.promptTokenCount || 0,
   329→    outputTokens: data.usageMetadata?.candidatesTokenCount || 0,
   330→  };
   331→}

   335→export async function healthCheck(providerId: string): Promise<HealthCheckResult> {
   336→  const provider = await db.aiProvider.findUnique({ where: { id: providerId } });
   337→  if (!provider) throw new Error('Provider not found');
   338→  if (!provider.apiKeyEncrypted) return { status: 'DISCONNECTED', latencyMs: 0, error: 'No API key configured' };
   339→
   340→  const apiKey = await decrypt(provider.apiKeyEncrypted);
   341→  const config = getProviderConfig(provider.kind);
   342→  const baseUrl = provider.baseUrl || config.defaultBaseUrl;
   343→
   344→  const start = Date.now();
   345→  try {
   346→    let models: ProviderModel[] = [];
   347→
   348→    if (provider.kind === 'ANTHROPIC') {
   349→      // Anthropic has no models endpoint, just test with a simple request
   350→      const res = await fetch(`${baseUrl}/messages`, {
   351→        method: 'POST',
   352→        headers: { 'Content-Type': 'application/json', 'x-api-key': apiKey, 'anthropic-version': '2023-06-01' },
   353→        body: JSON.stringify({ model: 'claude-3-haiku-20240307', max_tokens: 1, messages: [{ role: 'user', content: 'hi' }] }),
   354→      });
   355→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   356→      models = config.defaultModels;
   357→    } else if (provider.kind === 'GEMINI') {
   358→      const res = await fetch(`${baseUrl}/models?key=${apiKey}`);
   359→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   360→      models = config.defaultModels;
   361→    } else if (config.modelsEndpoint) {
   362→      const res = await fetch(`${baseUrl}${config.modelsEndpoint}`, {
   363→        headers: provider.kind === 'OLLAMA' ? {} : { 'Authorization': `Bearer ${apiKey}` },
   364→      });
   365→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   366→      const data = await res.json();
   367→      if (data.data) {
   368→        models = data.data.map((m: { id: string }) => ({
   369→          modelId: m.id, name: m.id, contextLength: 0,
   370→          inputCostPer1k: 0, outputCostPer1k: 0,
   371→          supportsImages: false, supportsVision: false,
   372→          supportsFunctionCalling: false, supportsJsonMode: false,
   373→          supportsStreaming: true, supportsTools: false,
   374→        }));
   375→      }
   376→    }
   377→
   378→    const latencyMs = Date.now() - start;
   379→
   380→    await db.aiProvider.update({
   381→      where: { id: providerId },
   382→      data: {
   383→        connectionStatus: 'CONNECTED',
   384→        latencyMs,
   385→        lastHealthCheckAt: new Date(),
   386→        lastError: null,
   387→      },
   388→    });
   389→
   390→    return { status: 'CONNECTED', latencyMs, availableModels: models };
   391→  } catch (err) {
   392→    const latencyMs = Date.now() - start;
   393→    const errorMsg = err instanceof Error ? err.message : 'Unknown error';
   394→
   395→    await db.aiProvider.update({
   396→      where: { id: providerId },
   397→      data: {
   398→        connectionStatus: 'ERROR',
   399→        latencyMs,
   400→        lastError: errorMsg,
   401→        lastHealthCheckAt: new Date(),
   402→      },
   403→    });
   404→
   405→    return { status: 'ERROR', latencyMs, error: errorMsg };
   406→  }
   407→}

   411→export async function syncModels(providerId: string): Promise<number> {
   412→  const provider = await db.aiProvider.findUnique({ where: { id: providerId } });
   413→  if (!provider) throw new Error('Provider not found');
   414→  if (!provider.apiKeyEncrypted) throw new Error('No API key');
   415→
   416→  const apiKey = await decrypt(provider.apiKeyEncrypted);
   417→  const config = getProviderConfig(provider.kind);
   418→  const baseUrl = provider.baseUrl || config.defaultBaseUrl;
   419→
   420→  let fetchedModels: ProviderModel[] = config.defaultModels;
   421→
   422→  // Try to fetch from API if endpoint exists
   423→  if (config.modelsEndpoint && provider.kind !== 'ANTHROPIC' && provider.kind !== 'AZURE_OPENAI') {
   424→    try {
   425→      const res = await fetch(`${baseUrl}${config.modelsEndpoint}`, {
   426→        headers: provider.kind === 'OLLAMA' ? {} : { 'Authorization': `Bearer ${apiKey}` },
   427→      });
   428→      if (res.ok) {
   429→        const data = await res.json();
   430→        if (data.data && Array.isArray(data.data)) {
   431→          const existingIds = new Set(config.defaultModels.map(m => m.modelId));
   432→          const newModels = data.data
   433→            .filter((m: { id: string }) => !existingIds.has(m.id))
   434→            .map((m: { id: string }) => ({
   435→              modelId: m.id, name: m.id, contextLength: 0,
   436→              inputCostPer1k: 0, outputCostPer1k: 0,
   437→              supportsImages: false, supportsVision: false,
   438→              supportsFunctionCalling: false, supportsJsonMode: false,
   439→              supportsStreaming: true, supportsTools: false,
   440→            }));
   441→          fetchedModels = [...config.defaultModels, ...newModels];
   442→        }
   443→      }
   444→    } catch { /* use defaults */ }
   445→  }
   446→
   447→  // Upsert models
   448→  let count = 0;
   449→  for (const model of fetchedModels) {
   450→    await db.aiModel.upsert({
   451→      where: { providerId_modelId: { providerId, modelId: model.modelId } },
   452→      update: { name: model.name, contextLength: model.contextLength, inputCostPer1k: model.inputCostPer1k, outputCostPer1k: model.outputCostPer1k, supportsImages: model.supportsImages, supportsVision: model.supportsVision, supportsFunctionCalling: model.supportsFunctionCalling, supportsJsonMode: model.supportsJsonMode, supportsStreaming: model.supportsStreaming, supportsTools: model.supportsTools, lastSyncedAt: new Date() },
   453→      create: { providerId, modelId: model.modelId, name: model.name, contextLength: model.contextLength, inputCostPer1k: model.inputCostPer1k, outputCostPer1k: model.outputCostPer1k, supportsImages: model.supportsImages, supportsVision: model.supportsVision, supportsFunctionCalling: model.supportsFunctionCalling, supportsJsonMode: model.supportsJsonMode, supportsStreaming: model.supportsStreaming, supportsTools: model.supportsTools, lastSyncedAt: new Date() },
   454→    });
   455→    count++;
   456→  }
   457→
   458→  await db.aiProvider.update({
   459→    where: { id: providerId },
   460→    data: { lastSyncAt: new Date() },
   461→  });
   462→
   463→  return count;
   464→}

   478→export async function getUsageAnalytics(siteId?: string, period: 'day' | 'week' | 'month' = 'month') {
   479→  const now = new Date();
   480→  const periodStart = new Date();
   481→  if (period === 'day') periodStart.setHours(now.getHours() - 24);
   482→  else if (period === 'week') periodStart.setDate(now.getDate() - 7);
   483→  else periodStart.setMonth(now.getMonth() - 1);
   484→
   485→  const where: Prisma.AiLogWhereInput = {
   486→    createdAt: { gte: periodStart },
   487→    ...(siteId ? { siteId } : {}),
   488→  };
   489→
   490→  const [totalCount, totalTokens, totalCost, errorCount, avgDuration] = await Promise.all([
   491→    db.aiLog.count({ where }),
   492→    db.aiLog.aggregate({ where, _sum: { totalTokens: true, inputTokens: true, outputTokens: true } }),
   493→    db.aiLog.aggregate({ where, _sum: { costUsd: true } }),
   494→    db.aiLog.count({ where: { ...where, status: 'error' } }),
   495→    db.aiLog.aggregate({ where: { ...where, status: 'success' }, _avg: { durationMs: true } }),
   496→  ]);
   497→
   498→  // Provider breakdown
   499→  const providerBreakdown = await db.aiLog.groupBy({
   500→    by: ['providerName'],
   501→    where,
   502→    _sum: { totalTokens: true, costUsd: true, inputTokens: true, outputTokens: true },
   503→    _count: true,
   504→    orderBy: { _sum: { costUsd: 'desc' } },
   505→  });
   506→
   507→  // Model breakdown
   508→  const modelBreakdown = await db.aiLog.groupBy({
   509→    by: ['modelId'],
   510→    where,
   511→    _sum: { totalTokens: true, costUsd: true },
   512→    _count: true,
   513→    orderBy: { _count: { id: 'desc' } },
   514→    take: 10,
   515→  });
   516→
   517→  // Daily usage chart
   518→  const dailyUsage = await db.$queryRawUnsafe<Array<{ date: string; count: number; tokens: number; cost: number }>>(`
   519→    SELECT DATE(createdAt) as date, COUNT(*) as count, COALESCE(SUM(totalTokens), 0) as tokens, COALESCE(SUM(costUsd), 0) as cost
   520→    FROM AiLog
   521→    WHERE createdAt >= ?
   522→    ${siteId ? 'AND siteId = ?' : ''}
   523→    GROUP BY DATE(createdAt)
   524→    ORDER BY date ASC
   525→  `, periodStart.toISOString(), ...(siteId ? [siteId] : []));
   526→
   527→  return {
   528→    totalRequests: totalCount,
   529→    totalTokens: totalTokens._sum.totalTokens || 0,
   530→    inputTokens: totalTokens._sum.inputTokens || 0,
   531→    outputTokens: totalTokens._sum.outputTokens || 0,
   532→    totalCost: totalCost._sum.costUsd || 0,
   533→    errorCount,
   534→    successRate: totalCount > 0 ? ((totalCount - errorCount) / totalCount) * 100 : 100,
   535→    avgDurationMs: Math.round(avgDuration._avg.durationMs || 0),
   536→    providerBreakdown,
   537→    modelBreakdown,
   538→    dailyUsage,
   539→  };
   540→}

   566→export interface ImageGenerationResponse {
   567→  images: GeneratedImage[];
   568→  model: string;
   569→  costUsd: number;
   570→  durationMs: number;
   571→  providerName: string;
   572→  providerKind: string;
   573→}
   574→
   575→// Approximate cost per image (USD) for known models, keyed by model then size
   576→const IMAGE_MODEL_COSTS: Record<string, Record<string, number>> = {
   577→  'dall-e-2': { '256x256': 0.016, '512x512': 0.016, '1024x1024': 0.02 },
   578→  'dall-e-3': { '1024x1024': 0.040, '1792x1024': 0.080, '1024x1792': 0.080 },
   579→  'gpt-image-1': { '1024x1024': 0.040, '1536x1024': 0.080, '1024x1536': 0.080 },
   580→};
   581→
   582→function getImageCost(modelId: string, size: string): number {
   583→  const modelCosts = IMAGE_MODEL_COSTS[modelId];
   584→  if (modelCosts) {
   585→    return modelCosts[size] ?? modelCosts['1024x1024'] ?? 0.04;
   586→  }
   587→  // Fallback: use default estimate
   588→  return 0.04;
   589→}

   591→async function callOpenAIImageGeneration(
   592→  baseUrl: string,
   593→  apiKey: string,
   594→  model: string,
   595→  prompt: string,
   596→  opts: {
   597→    negativePrompt?: string;
   598→    size?: string;
   599→    quality?: string;
   600→    style?: string;
   601→    n?: number;
   602→    responseFormat?: string;
   603→    apiVersion?: string;
   604→  } = {},
   605→): Promise<{ images: GeneratedImage[] }> {
   606→  const url = opts.apiVersion
   607→    ? `${baseUrl}/images/generations?api-version=${opts.apiVersion}`
   608→    : `${baseUrl}/images/generations`;
   609→
   610→  const body: Record<string, unknown> = {
   611→    model,
   612→    prompt,
   613→    n: opts.n ?? 1,
   614→    size: opts.size ?? '1024x1024',
   615→    response_format: opts.responseFormat ?? 'url',
   616→  };
   617→
   618→  if (opts.quality) body.quality = opts.quality;
   619→  if (opts.style) body.style = opts.style;
   620→
   621→  const res = await fetch(url, {
   622→    method: 'POST',
   623→    headers: {
   624→      'Content-Type': 'application/json',
   625→      'Authorization': `Bearer ${apiKey}`,
   626→    },
   627→    body: JSON.stringify(body),
   628→  });
   629→
   630→  if (!res.ok) {
   631→    const err = await res.text();
   632→    throw new Error(`Image generation API error: ${res.status} — ${err}`);
   633→  }
   634→
   635→  const data = await res.json();
   636→  const images: GeneratedImage[] = (data.data ?? []).map((item: Record<string, unknown>) => ({
   637→    url: (item.url as string) ?? null,
   638→    base64: (item.b64_json as string) ?? null,
   639→    revisedPrompt: (item.revised_prompt as string) ?? null,
   640→  }));
   641→
   642→  return { images };
   643→}

   645→async function callGeminiImageGeneration(
   646→  baseUrl: string,
   647→  apiKey: string,
   648→  model: string,
   649→  prompt: string,
   650→  opts: {
   651→    negativePrompt?: string;
   652→    size?: string;
   653→    n?: number;
   654→    responseFormat?: string;
   655→  } = {},
   656→): Promise<{ images: GeneratedImage[] }> {
   657→  // Parse size to aspect ratio for Gemini Imagen
   658→  const sizeParts = (opts.size ?? '1024x1024').split('x');
   659→  const width = parseInt(sizeParts[0] ?? '1024', 10);
   660→  const height = parseInt(sizeParts[1] ?? '1024', 10);
   661→  let ratio: string;
   662→  if (Math.abs(width - height) < 100) {
   663→    ratio = '1:1'; // square
   664→  } else {
   665→    ratio = width >= height ? '16:9' : '9:16';
   666→  }
   667→
   668→  // Gemini uses Imagen via the generateContent endpoint with image generation config
   669→  // or the dedicated Imagen predict endpoint
   670→  const res = await fetch(
   671→    `${baseUrl}/models/${model}:generateContent?key=${apiKey}`,
   672→    {
   673→      method: 'POST',
   674→      headers: { 'Content-Type': 'application/json' },
   675→      body: JSON.stringify({
   676→        contents: [{ parts: [{ text: prompt }] }],
   677→        generationConfig: {
   678→          responseModalities: ['TEXT', 'IMAGE'],
   679→          // @ts-expect-error -- Gemini-specific field
   680→          imageGenerationConfig: {
   681→            numberOfImages: opts.n ?? 1,
   682→            aspectRatio: ratio,
   683→          },
   684→        },
   685→      }),
   686→    },
   687→  );
   688→
   689→  if (!res.ok) {
   690→    const err = await res.text();
   691→    throw new Error(`Gemini image generation error: ${res.status} — ${err}`);
   692→  }
   693→
   694→  const data = await res.json();
   695→  const images: GeneratedImage[] = [];
   696→
   697→  const parts = data.candidates?.[0]?.content?.parts ?? [];
   698→  for (const part of parts) {
   699→    if (part.inlineData) {
   700→      images.push({
   701→        url: null,
   702→        base64: part.inlineData.data ?? null,
   703→        revisedPrompt: prompt,
   704→      });
   705→    }
   706→  }
   707→
   708→  // Fallback: if no image parts returned, throw
   709→  if (images.length === 0) {
   710→    throw new Error('No images were generated by Gemini');
   711→  }
   712→
   713→  return { images };
   714→}

   716→export async function executeImageGeneration(req: ImageGenerationRequest): Promise<ImageGenerationResponse> {
   717→  const provider = await db.aiProvider.findUnique({
   718→    where: { id: req.providerId },
   719→    include: { models: true },
   720→  });
   721→
   722→  if (!provider) throw new Error('Provider not found');
   723→  if (!provider.isActive) throw new Error('Provider is disabled');
   724→  if (!provider.apiKeyEncrypted) throw new Error('API key not configured');
   725→
   726→  const apiKey = await decrypt(provider.apiKeyEncrypted);
   727→  const config = getProviderConfig(provider.kind);
   728→  const baseUrl = provider.baseUrl || config.defaultBaseUrl;
   729→  const modelId = req.modelId || provider.models.find(m => m.isDefault)?.modelId || provider.models[0]?.modelId;
   730→  if (!modelId) throw new Error('No model available');
   731→
   732→  const startTime = Date.now();
   733→  let images: GeneratedImage[] = [];
   734→  let usedProvider = provider;
   735→
   736→  try {
   737→    if (provider.kind === 'GEMINI') {
   738→      const result = await callGeminiImageGeneration(baseUrl, apiKey, modelId, req.prompt, {
   739→        negativePrompt: req.negativePrompt,
   740→        size: req.size,
   741→        n: req.n,
   742→        responseFormat: req.responseFormat,
   743→      });
   744→      images = result.images;
   745→    } else if (provider.kind === 'AZURE_OPENAI') {
   746→      const azureConfig = JSON.parse(provider.config || '{}');
   747→      const result = await callOpenAIImageGeneration(
   748→        `${baseUrl}/openai/deployments/${azureConfig.deployment || modelId}`,
   749→        apiKey,
   750→        modelId,
   751→        req.prompt,
   752→        { size: req.size, quality: req.quality, style: req.style, n: req.n, responseFormat: req.responseFormat, apiVersion: provider.apiVersion || '2024-02-01' },
   753→      );
   754→      images = result.images;
   755→    } else if (
   756→      provider.kind === 'OPENAI' ||
   757→      provider.kind === 'OPENROUTER' ||
   758→      provider.kind === 'GROQ' ||
   759→      provider.kind === 'DEEPSEEK' ||
   760→      provider.kind === 'OLLAMA'
   761→    ) {
   762→      const result = await callOpenAIImageGeneration(baseUrl, apiKey, modelId, req.prompt, {
   763→        negativePrompt: req.negativePrompt,
   764→        size: req.size,
   765→        quality: req.quality,
   766→        style: req.style,
   767→        n: req.n,
   768→        responseFormat: req.responseFormat,
   769→      });
   770→      images = result.images;
   771→    } else {
   772→      // Attempt OpenAI-compatible for unknown providers
   773→      const result = await callOpenAIImageGeneration(baseUrl, apiKey, modelId, req.prompt, {
   774→        negativePrompt: req.negativePrompt,
   775→        size: req.size,
   776→        quality: req.quality,
   777→        style: req.style,
   778→        n: req.n,
   779→        responseFormat: req.responseFormat,
   780→      });
   781→      images = result.images;
   782→    }
   783→  } catch (err) {
   784→    // Try fallback providers for image generation
   785→    const fallbacks = await db.aiProviderFallback.findMany({
   786→      where: { providerId: provider.id },
   787→      include: { fallback: { include: { models: true } } },
   788→      orderBy: { priority: 'asc' },
   789→    });
   790→
   791→    for (const fb of fallbacks) {
   792→      if (!fb.fallback.isActive || !fb.fallback.apiKeyEncrypted) continue;
   793→      try {
   794→        const fbApiKey = await decrypt(fb.fallback.apiKeyEncrypted);
   795→        const fbBaseUrl = fb.fallback.baseUrl || getProviderConfig(fb.fallback.kind).defaultBaseUrl;
   796→        const fbModel = req.modelId || fb.fallback.models.find(m => m.isDefault)?.modelId || fb.fallback.models[0]?.modelId;
   797→        if (!fbModel) continue;
   798→
   799→        const result = await callOpenAIImageGeneration(fbBaseUrl, fbApiKey, fbModel, req.prompt, {
   800→          size: req.size, quality: req.quality, style: req.style, n: req.n, responseFormat: req.responseFormat,
   801→        });
   802→        images = result.images;
   803→        usedProvider = fb.fallback;
   804→        break;
   805→      } catch {
   806→        continue;
   807→      }
   808→    }
   809→
   810→    if (images.length === 0) throw err;
   811→  }
   812→
   813→  const durationMs = Date.now() - startTime;
   814→  const size = req.size ?? '1024x1024';
   815→  const n = images.length;
   816→  const costUsd = n * getImageCost(modelId, size);
   817→
   818→  // Update provider stats
   819→  await db.aiProvider.update({
   820→    where: { id: usedProvider.id },
   821→    data: {
   822→      lastUsedAt: new Date(),
   823→      latencyMs: durationMs,
   824→      connectionStatus: 'CONNECTED',
   825→      lastError: null,
   826→    },
   827→  });
   828→
   829→  // Log the request
   830→  await db.aiLog.create({
   831→    data: {
   832→      providerId: usedProvider.id,
   833→      providerName: usedProvider.name,
   834→      modelId,
   835→      question: `[IMAGE] ${req.prompt}`,
   836→      response: JSON.stringify({
   837→        imagesGenerated: n,
   838→        size,
   839→        format: req.responseFormat ?? 'url',
   840→        model: modelId,
   841→      }),
   842→      inputTokens: 0,
   843→      outputTokens: 0,
   844→      totalTokens: 0,
   845→      costUsd,
   846→      durationMs,
   847→      status: 'success',
   848→      siteId: req.siteId,
   849→      userId: req.userId,
   850→    },
   851→  });
   852→
   853→  return {
   854→    images,
   855→    model: modelId,
   856→    costUsd,
   857→    durationMs,
   858→    providerName: usedProvider.name,
   859→    providerKind: usedProvider.kind,
   860→  };
   861→}
