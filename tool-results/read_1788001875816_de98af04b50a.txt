     4→// PLATFORM PLANS & PRICING — simplified admin UI.
     5→// ============================================================
     6→// Owner sees compact plan cards (Beta / Pro / Max …) with a quick
     7→// Active toggle, a compact price + features + limits summary,
     8→// and an "Edit Plan" button. The Edit Plan button opens a Dialog
     9→// with three compact sections: Basic Information, Feature Access,
    10→// Usage Limits. An optional collapsed "Client Display" section
    11→// holds the marketing feature list shown to clients on the
    12→// Client Billing page. A "+ Create Plan" button at the top-right
    13→// opens a Create Plan dialog (same sections, blank defaults) that
    14→// POSTs to /api/platform/admin/plans.
    15→//
    16→// All writes go through /api/platform/admin/plans (GET list,
    17→// POST create) + /api/platform/admin/plans/[planId] (PUT update)
    18→// — owner-only — the same shared plan-config cache the Client
    19→// Billing page and MRR read. Server-side entitlement enforcement
    20→// (hasFeature) and usage-limit checks (checkLimit) consume the
    21→// same data, so toggling a feature off here denies it for clients
    22→// on the next request. The Client Billing page is NOT modified
    23→// and continues to read from /api/platform/billing/me.
    24→// ============================================================

    90→/** Resolve the badge id for the shared PlanBadge component. Falls
    91→ *  back to 'beta' for any unknown plan id so future plans still
    92→ *  render with a sensible neutral badge. */
    93→function getPlanBadgeId(planId: string): 'beta' | 'pro' | 'max' {
    94→  if (planId === 'beta' || planId === 'pro' || planId === 'max') return planId;
    95→  return 'beta';
    96→}
    97→
    98→/** Format a single limit value for the compact card summary. -1 →
    99→ *  "Unlimited"; storageBytes → human-readable (1 GB); other limits
   100→ *  use locale-formatted numbers. */
   101→function formatLimitValue(key: LimitKey, value: number): string {
   102→  if (value === UNLIMITED) return 'Unlimited';
   103→  if (key === 'storageBytes') return formatBytes(value);
   104→  return value.toLocaleString();
   105→}

   120→function PlanSummaryCard({
   121→  plan,
   122→  canEdit,
   123→}: {
   124→  plan: PlanConfigData;
   125→  canEdit: boolean;
   126→}) {
   127→  const queryClient = useQueryClient();
   128→  const [editing, setEditing] = useState(false);
   129→
   130→  // Quick active toggle — auto-saves immediately (owner-only).
   131→  // Reuses the same PUT endpoint and the same invalidation chain
   132→  // as the full Edit modal so clients see the new status on the
   133→  // next read.
   134→  const activeMutation = useMutation({
   135→    mutationFn: (active: boolean) =>
   136→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, { active }),
   137→    onSuccess: (data) => {
   138→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   139→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   140→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   141→      toast.success(
   142→        `${data.name} ${data.active ? 'activated' : 'deactivated'} — changes are live for clients.`,
   143→      );
   144→    },
   145→    onError: (err: unknown) => {
   146→      const e = err as { error?: { message?: string }; message?: string };
   147→      toast.error(
   148→        e?.error?.message ?? e?.message ?? 'Unable to change plan status. OWNER access required.',
   149→      );
   150→    },
   151→  });
   152→
   153→  // Feature list shown on the card. Prefer the client-facing marketing
   154→  // copy (plan.features); fall back to the structured entitlement labels
   155→  // so a brand-new plan with no marketing copy still surfaces a readable
   156→  // feature list. Either way the data shown is real plan data — no mock.
   157→  const featureItems =
   158→    plan.features.length > 0
   159→      ? plan.features
   160→      : plan.entitlements.map((k) => ENTITLEMENT_LABELS[k as EntitlementKey] ?? k);
   161→
   162→  return (
   163→    <>
   164→      <div
   165→        className={`group relative flex h-full flex-col overflow-hidden rounded-3xl border border-white/[0.08] bg-[#161616] p-8 transition-colors duration-300 hover:border-white/[0.16] ${
   166→          plan.active ? '' : 'opacity-60'
   167→        }`}
   168→      >
   169→        {/* Header: name + badge + quick active toggle */}
   170→        <div className="flex items-start justify-between gap-3">
   171→          <div className="min-w-0">
   172→            <h3 className="text-2xl font-semibold tracking-tight text-zinc-50">{plan.name}</h3>
   173→            <div className="mt-2.5">
   174→              <PlanBadge planId={getPlanBadgeId(plan.planId)} />
   175→            </div>
   176→          </div>
   177→          <div className="flex shrink-0 items-center gap-2">
   178→            <Label
   179→              htmlFor={`active-${plan.planId}`}
   180→              className="cursor-pointer text-xs text-zinc-500"
   181→            >
   182→              {plan.active ? 'Active' : 'Inactive'}
   183→            </Label>
   184→            <Switch
   185→              id={`active-${plan.planId}`}
   186→              checked={plan.active}
   187→              disabled={!canEdit || activeMutation.isPending}
   188→              onCheckedChange={(v) => activeMutation.mutate(v)}
   189→            />
   190→          </div>
   191→        </div>
   192→
   193→        {/* Price */}
   194→        <div className="mt-8">
   195→          {plan.isFree ? (
   196→            <div className="flex items-baseline gap-2">
   197→              <span className="text-5xl font-semibold tracking-tight text-zinc-50">Free</span>
   198→            </div>
   199→          ) : (
   200→            <div className="flex items-baseline gap-2">
   201→              <span className="text-5xl font-semibold tracking-tight text-zinc-50">
   202→                {formatCurrency(plan.priceMonthly, plan.currency)}
   203→              </span>
   204→              <span className="text-sm text-zinc-500">/ month</span>
   205→            </div>
   206→          )}
   207→          {!plan.isFree && plan.priceYearly > 0 && (
   208→            <p className="mt-1.5 text-sm text-zinc-500">
   209→              {formatCurrency(plan.priceYearly, plan.currency)} / year
   210→            </p>
   211→          )}
   212→        </div>
   213→
   214→        {/* Billing interval pill */}
   215→        <div className="mt-4">
   216→          <span className="inline-flex items-center rounded-full border border-white/10 bg-white/[0.04] px-3 py-1 text-xs font-medium capitalize text-zinc-400">
   217→            {plan.interval} billing
   218→          </span>
   219→        </div>
   220→
   221→        {/* Features */}
   222→        <div className="mt-8">
   223→          <p className="text-xs font-medium uppercase tracking-wider text-zinc-500">Features</p>
   224→          <ul className="mt-3.5 space-y-2.5">
   225→            {featureItems.length === 0 ? (
   226→              <li className="text-sm text-zinc-600">No features configured</li>
   227→            ) : (
   228→              featureItems.map((f, i) => (
   229→                <li key={`${f}-${i}`} className="flex items-start gap-2.5 text-sm text-zinc-300">
   230→                  <Check className="mt-0.5 h-4 w-4 shrink-0 text-zinc-500" />
   231→                  <span className="leading-snug">{f}</span>
   232→                </li>
   233→              ))
   234→            )}
   235→          </ul>
   236→        </div>
   237→
   238→        {/* Usage Limits */}
   239→        <div className="mt-8">
   240→          <p className="text-xs font-medium uppercase tracking-wider text-zinc-500">Usage Limits</p>
   241→          <div className="mt-3.5 grid grid-cols-2 gap-2">
   242→            {LIMIT_KEYS.map((k) => (
   243→              <div
   244→                key={k}
   245→                className="rounded-lg border border-white/[0.06] bg-white/[0.02] px-3 py-2"
   246→              >
   247→                <p className="text-[10px] uppercase tracking-wide text-zinc-500">{LIMIT_LABELS[k]}</p>
   248→                <p className="mt-0.5 text-sm font-medium text-zinc-200">
   249→                  {formatLimitValue(k, plan.limits[k])}
   250→                </p>
   251→              </div>
   252→            ))}
   253→          </div>
   254→        </div>
   255→
   256→        {/* Edit button — pinned to card bottom */}
   257→        <div className="mt-auto pt-8">
   258→          <Button
   259→            variant="outline"
   260→            size="sm"
   261→            className="w-full rounded-full border-white/15 bg-transparent text-zinc-200 hover:bg-white/5 hover:text-white hover:border-white/25"
   262→            disabled={!canEdit}
   263→            onClick={() => setEditing(true)}
   264→          >
   265→            <Pencil className="h-4 w-4 mr-2" />
   266→            Edit Plan
   267→          </Button>
   268→          {!canEdit && (
   269→            <p className="mt-2 text-center text-[10px] text-zinc-600">
   270→              Editing is restricted to the OWNER role.
   271→            </p>
   272→          )}
   273→        </div>
   274→      </div>
   275→
   276→      {editing && (
   277→        <EditPlanDialog plan={plan} open={editing} onOpenChange={setEditing} />
   278→      )}
   279→    </>
   280→  );
   281→}

   285→function EditPlanDialog({
   286→  plan,
   287→  open,
   288→  onOpenChange,
   289→}: {
   290→  plan: PlanConfigData;
   291→  open: boolean;
   292→  onOpenChange: (open: boolean) => void;
   293→}) {
   294→  const queryClient = useQueryClient();
   295→  const [name, setName] = useState(plan.name);
   296→  const [priceMonthly, setPriceMonthly] = useState(String(plan.priceMonthly));
   297→  const [priceYearly, setPriceYearly] = useState(String(plan.priceYearly));
   298→  const [currency, setCurrency] = useState(plan.currency);
   299→  const [interval, setInterval] = useState<'monthly' | 'yearly'>(plan.interval);
   300→  const [active, setActive] = useState(plan.active);
   301→  const [features, setFeatures] = useState(plan.features.join('\n'));
   302→  const [entitlements, setEntitlements] = useState<string[]>(plan.entitlements);
   303→  const [limits, setLimits] = useState<PlanLimits>(plan.limits);
   304→  const [clientDisplayOpen, setClientDisplayOpen] = useState(false);
   305→
   306→  // EditPlanDialog is conditionally rendered by the parent (mounted
   307→  // fresh each time the user opens it), so the useState initializers
   308→  // above already seed local state from the latest server snapshot.
   309→  // No useEffect sync is needed — that would just trigger cascading
   310→  // renders (and the React Compiler correctly flags it).
   311→
   312→  const reset = () => {
   313→    setName(plan.name);
   314→    setPriceMonthly(String(plan.priceMonthly));
   315→    setPriceYearly(String(plan.priceYearly));
   316→    setCurrency(plan.currency);
   317→    setInterval(plan.interval);
   318→    setActive(plan.active);
   319→    setFeatures(plan.features.join('\n'));
   320→    setEntitlements(plan.entitlements);
   321→    setLimits(plan.limits);
   322→  };
   323→
   324→  const saveMutation = useMutation({
   325→    mutationFn: (patch: PlanPatch) =>
   326→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, patch),
   327→    onSuccess: (data) => {
   328→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   329→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   330→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   331→      toast.success(`${data.name} updated — changes are now live for clients.`);
   332→      onOpenChange(false);
   333→    },
   334→    onError: (err: unknown) => {
   335→      const e = err as { error?: { message?: string }; message?: string };
   336→      toast.error(
   337→        e?.error?.message ?? e?.message ?? 'Unable to save plan. You may need OWNER access.',
   338→      );
   339→    },
   340→  });
   341→
   342→  const buildPatch = (): PlanPatch => ({
   343→    name,
   344→    priceMonthly: Number(priceMonthly) || 0,
   345→    priceYearly: Number(priceYearly) || 0,
   346→    currency,
   347→    interval,
   348→    active,
   349→    features: features
   350→      .split('\n')
   351→      .map((f) => f.trim())
   352→      .filter(Boolean),
   353→    entitlements,
   354→    limits: { ...limits, storageBytes: Number(limits.storageBytes) || 0 },
   355→  });
   356→
   357→  return (
   358→    <Dialog open={open} onOpenChange={onOpenChange}>
   359→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   360→        <DialogHeader>
   361→          <div className="flex items-center gap-2">
   362→            <DialogTitle className="text-base">Edit {plan.name}</DialogTitle>
   363→            <PlanBadge planId={getPlanBadgeId(plan.planId)} />
   364→          </div>
   365→          <DialogDescription className="text-xs">
   366→            Single source of truth — changes propagate to the Client Billing page, MRR and
   367→            server-side entitlement enforcement.
   368→          </DialogDescription>
   369→        </DialogHeader>
   370→
   371→        <div className="space-y-5">
   372→          {/* -------------------- Basic Information -------------------- */}
   373→          <section className="space-y-3">
   374→            <h4 className="text-sm font-semibold">Basic Information</h4>
   375→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   376→              <div className="space-y-1">
   377→                <Label className="text-xs">Name</Label>
   378→                <Input
   379→                  value={name}
   380→                  onChange={(e) => setName(e.target.value)}
   381→                  className="h-9"
   382→                />
   383→              </div>
   384→              <div className="space-y-1">
   385→                <Label className="text-xs">Monthly Price</Label>
   386→                <Input
   387→                  type="number"
   388→                  value={priceMonthly}
   389→                  onChange={(e) => setPriceMonthly(e.target.value)}
   390→                  className="h-9"
   391→                />
   392→              </div>
   393→              <div className="space-y-1">
   394→                <Label className="text-xs">Yearly Price</Label>
   395→                <Input
   396→                  type="number"
   397→                  value={priceYearly}
   398→                  onChange={(e) => setPriceYearly(e.target.value)}
   399→                  className="h-9"
   400→                />
   401→              </div>
   402→              <div className="space-y-1">
   403→                <Label className="text-xs">Currency</Label>
   404→                <Input
   405→                  value={currency}
   406→                  onChange={(e) => setCurrency(e.target.value)}
   407→                  className="h-9"
   408→                />
   409→              </div>
   410→            </div>
   411→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   412→              <div className="space-y-1">
   413→                <Label className="text-xs">Billing Interval</Label>
   414→                <Select
   415→                  value={interval}
   416→                  onValueChange={(v) => setInterval(v as 'monthly' | 'yearly')}
   417→                >
   418→                  <SelectTrigger className="h-9">
   419→                    <SelectValue />
   420→                  </SelectTrigger>
   421→                  <SelectContent>
   422→                    <SelectItem value="monthly">monthly</SelectItem>
   423→                    <SelectItem value="yearly">yearly</SelectItem>
   424→                  </SelectContent>
   425→                </Select>
   426→              </div>
   427→              <div className="flex items-center justify-between rounded-md border px-3 h-9 sm:mt-5">
   428→                <Label
   429→                  htmlFor="dialog-active"
   430→                  className="text-xs text-muted-foreground cursor-pointer"
   431→                >
   432→                  Active
   433→                </Label>
   434→                <Switch
   435→                  id="dialog-active"
   436→                  checked={active}
   437→                  onCheckedChange={setActive}
   438→                />
   439→              </div>
   440→            </div>
   441→          </section>
   442→
   443→          <Separator />
   444→
   445→          {/* -------------------- Feature Access -------------------- */}
   446→          <section className="space-y-3">
   447→            <div className="flex items-center justify-between gap-2">
   448→              <div>
   449→                <h4 className="text-sm font-semibold">Feature Access</h4>
   450→                <p className="text-xs text-muted-foreground mt-0.5">
   451→                  Server-side enforced. Disabled features return 403 to clients.
   452→                </p>
   453→              </div>
   454→              <Badge variant="outline" className="text-[10px] font-mono shrink-0">

   521→                <div className="flex items-center gap-2">
   522→                  <span className="text-sm font-semibold">Client Display</span>

   551→        <DialogFooter className="gap-2">

   553→            variant="outline"

   555→            onClick={reset}
