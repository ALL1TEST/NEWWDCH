   279→// -------------------- Category Count Query --------------------
   280→
   281→/**
   282→ * Per-scope category counts. `scope=platform` requests system templates
   283→ * (siteId IS NULL) via the scope=platform query param; `scope=client` uses
   284→ * the default client behavior. The query key is scoped so the two caches
   285→ * never collide.
   286→ */
   287→function useCategoryCounts(scope: 'client' | 'platform' = 'client') {
   288→  return useQuery({
   289→    queryKey: ['email-templates', 'category-counts', scope],
   290→    queryFn: async () => {
   291→      const allCategories = CATEGORIES.filter((c) => c.value !== 'ALL');
   292→      const results = await Promise.allSettled(
   293→        allCategories.map(async (cat) => {
   294→          const res = await getApi<{ data: EmailTemplate[]; meta: { pagination: { total: number } } }>(
   295→            '/api/email-templates',
   296→            {
   297→              category: cat.value,
   298→              page: 1,
   299→              pageSize: 1,
   300→              ...(scope === 'platform' ? { scope: 'platform' } : {}),
   301→            },
   302→            { raw: true },
   303→          );
   304→          return { category: cat.value, count: res?.meta?.pagination?.total ?? 0 };
   305→        }),
   306→      );
   307→      const counts: Record<string, number> = {};
   308→      for (const r of results) {
   309→        if (r.status === 'fulfilled') {
   310→          counts[r.value.category] = r.value.count;
   311→        }
   312→      }
   313→      return counts;
   314→    },
   315→    staleTime: 30_000,
   316→  });
   317→}

   321→export function TemplateList({ onEdit, onPreview, scope = 'client' }: TemplateListProps) {
   322→  const isPlatform = scope === 'platform';
   323→  const moduleName = isPlatform ? 'platform-email-templates' : 'email-templates';
   324→  const queryClient = useQueryClient();
   325→  const navigate = useNavigationStore((s) => s.navigate);
   326→
   327→  // -------------------- Filter State --------------------
   328→  const [category, setCategory] = useState<EmailTemplateCategory | 'ALL'>('ALL');
   329→  const [statusFilter, setStatusFilter] = useState('ALL');
   330→  const [search, setSearch] = useState('');
   331→  const [searchInput, setSearchInput] = useState('');
   332→  const [sortValue, setSortValue] = useState('createdAt:desc');
   333→  const [page, setPage] = useState(1);
   334→  const [pageSize, setPageSize] = useState(DEFAULT_PAGE_SIZE);
   335→
   336→  // -------------------- Dialog State --------------------
   337→  const [deleteTarget, setDeleteTarget] = useState<EmailTemplate | null>(null);
   338→  const [revertTarget, setRevertTarget] = useState<EmailTemplate | null>(null);
   339→  const [sendTestTarget, setSendTestTarget] = useState<string | null>(null);
   340→  const [seedDialogOpen, setSeedDialogOpen] = useState(false);
   341→
   342→  // -------------------- Derived Sort Values --------------------
   343→  const [sortField, sortOrder] = useMemo(() => {
   344→    const [f, o] = sortValue.split(':');
   345→    return [f, o as 'asc' | 'desc'];
   346→  }, [sortValue]);
   347→
   348→  // -------------------- Search Debounce --------------------
   349→  const debounceRef = useRef<ReturnType<typeof setTimeout>>();
   350→
   351→  const updateSearch = useCallback(
   352→    (value: string) => {
   353→      setSearchInput(value);
   354→      if (debounceRef.current) clearTimeout(debounceRef.current);
   355→      debounceRef.current = setTimeout(() => {
   356→        setSearch(value);
   357→        setPage(1);
   358→      }, 300);
   359→    },
   360→    [],
   361→  );
   362→
   363→  // -------------------- Filter Change Handlers --------------------
   364→  const handleCategoryChange = useCallback((value: string) => {
   365→    setCategory(value as EmailTemplateCategory | 'ALL');
   366→    setPage(1);
   367→  }, []);
   368→
   369→  const handleStatusChange = useCallback((value: string) => {
   370→    setStatusFilter(value);
   371→    setPage(1);
   372→  }, []);
   373→
   374→  const handleSortChange = useCallback((value: string) => {
   375→    setSortValue(value);
   376→    setPage(1);
   377→  }, []);
   378→
   379→  // -------------------- Build Query Params --------------------
   380→  const queryParams = useMemo(
   381→    () => ({
   382→      page,
   383→      pageSize,
   384→      sort: sortField,
   385→      order: sortOrder,
   386→      ...(category !== 'ALL' ? { category } : {}),
   387→      ...(statusFilter !== 'ALL' ? { status: statusFilter } : {}),
   388→      ...(search ? { search } : {}),
   389→      // Platform scope requests system templates (siteId IS NULL) and is
   390→      // guarded by requirePlatformAdmin on the server. Client scope omits
   391→      // the param so the default client behavior is preserved.
   392→      ...(isPlatform ? { scope: 'platform' } : {}),
   393→    }),
   394→    [page, pageSize, sortField, sortOrder, category, statusFilter, search, isPlatform],
   395→  );
   396→
   397→  // -------------------- Data Fetching --------------------
   398→  // The query key is scoped so client and platform caches never collide —
   399→  // platform queries are prefixed with the scope to keep them isolated.
   400→  const { data: raw, isLoading } = useQuery({
   401→    queryKey: isPlatform
   402→      ? ['email-templates', 'list', 'platform', queryParams]
   403→      : queryKeys.emailTemplates.list(queryParams),
   404→    queryFn: () =>
   405→      getApi<{ data: EmailTemplate[]; meta: { pagination: { page: number; pageSize: number; total: number; totalPages: number } } }>('/api/email-templates', queryParams, { raw: true }),
   406→    staleTime: 10_000,
   407→  });
   408→
   409→  const templates = (Array.isArray(raw?.data) ? raw.data : []) as EmailTemplate[];
   410→  const pagination = raw?.meta?.pagination;
   411→
   412→  // -------------------- Category Counts --------------------
   413→  const { data: categoryCounts } = useCategoryCounts(scope);
   414→  const totalCount = categoryCounts
   415→    ? Object.values(categoryCounts).reduce((a, b) => a + b, 0)
   416→    : 0;
   417→
   418→  // -------------------- Mutations --------------------
   419→  // Scope-aware invalidation predicate. Platform invalidates the platform
   420→  // cache keys (prefixed with 'platform'); client invalidates the legacy
   421→  // keys. Both scopes also invalidate the detail cache (scope-agnostic,
   422→  // keyed by id) so an edited template's detail refetches.
   423→  const invalidateAll = useCallback(() => {
   424→    if (isPlatform) {
   425→      queryClient.invalidateQueries({ queryKey: ['email-templates', 'list', 'platform'] });
   426→      queryClient.invalidateQueries({ queryKey: ['email-templates', 'category-counts', 'platform'] });
   427→    } else {
   428→      queryClient.invalidateQueries({ queryKey: queryKeys.emailTemplates.all });
   429→      queryClient.invalidateQueries({ queryKey: ['email-templates', 'category-counts', 'client'] });
   430→    }
   431→  }, [queryClient, isPlatform]);
   432→
   433→  const duplicateMutation = useMutation({
   434→    mutationFn: (id: string) => postApi(`/api/email-templates/${id}/duplicate`),
   435→    onSuccess: () => {
   436→      toast.success('Template duplicated successfully');
   437→      invalidateAll();
   438→    },
   439→    onError: (err: Error) => {
   440→      toast.error(err.message || 'Failed to duplicate template');
   441→    },
   442→  });
   443→
   444→  const toggleStatusMutation = useMutation({
   445→    mutationFn: ({ id, status }: { id: string; status: EmailTemplateStatus }) =>
   446→      patchApi(`/api/email-templates/${id}`, { status }),
   447→    onSuccess: () => {
   448→      invalidateAll();
   449→    },
   450→    onError: (err: Error) => {
   451→      toast.error(err.message || 'Failed to update template status');
   452→    },
   453→  });
   454→
   455→  const deleteMutation = useMutation({
   456→    mutationFn: (id: string) => deleteApi(`/api/email-templates/${id}`),
   457→    onSuccess: () => {
   458→      toast.success('Template deleted');
   459→      invalidateAll();
   460→      setDeleteTarget(null);
   461→    },
   462→    onError: (err: Error) => {
   463→      toast.error(err.message || 'Failed to delete template');
   464→    },
   465→  });
   466→
   467→  const revertMutation = useMutation({
   468→    mutationFn: (id: string) => postApi(`/api/email-templates/${id}/revert`),
   469→    onSuccess: () => {
   470→      toast.success('Template reverted to default');
   471→      invalidateAll();
   472→      setRevertTarget(null);
   473→    },
   474→    onError: (err: Error) => {
   475→      toast.error(err.message || 'Failed to revert template');
   476→    },
   477→  });
   478→
   479→  const seedMutation = useMutation({
   480→    mutationFn: () => postApi<{ seeded: number; skipped: number; total: number }>(
   481→      '/api/email-templates/seed',
   482→      // Platform scope is guarded by requirePlatformAdmin on the server.
   483→      isPlatform ? { scope: 'platform' } : {},
   484→    ),
   485→    onSuccess: (res: any) => {
   486→      const data = res?.data ?? res ?? { seeded: 0, skipped: 0, total: 0 };
   487→      const { seeded, skipped } = data;
   488→      invalidateAll();
   489→      if (seeded > 0) {
   490→        toast.success(`${seeded} default template${seeded > 1 ? 's' : ''} created${skipped > 0 ? `, ${skipped} already existed` : ''}`);
   491→      } else if (skipped > 0) {
   492→        toast.info('All default email templates are already available.');
   493→      } else {
   494→        toast.info('No default templates to seed.');
   495→      }
   496→      setSeedDialogOpen(false);
   497→    },
   498→    onError: (err: Error) => {
   499→      toast.error(err.message || 'Failed to seed templates');
   500→    },
   501→  });
   502→
   503→  // -------------------- Navigation Handlers --------------------
   504→  // Platform scope navigates under 'platform-email-templates' so the URL
   505→  // hash is #platform-email-templates/new and the platform module router
   506→  // picks it up. Client scope uses the legacy 'email-templates' module.
   507→  const handleCreate = useCallback(() => {
   508→    navigate(moduleName, 'new');
   509→  }, [navigate, moduleName]);
   510→
   511→  // -------------------- Render --------------------
   512→  // Scope-aware page header. Platform scope uses PlatformPageHeader (keeps
   513→  // the PLATFORM badge) with platform-level copy; client scope uses the
   514→  // legacy PageHeader with the original copy.
   515→  const headerAction = (
   516→    <div className="flex flex-col items-end gap-2">
   517→      <Button size="sm" onClick={handleCreate}>
   518→        <Plus className="h-4 w-4 mr-2" />
   519→        Create Template
   520→      </Button>
   521→      <Button
   522→        size="sm"
   523→        variant="outline"
   524→        onClick={() => setSeedDialogOpen(true)}
   525→        disabled={seedMutation.isPending}
   526→        className="border-amber-300 text-amber-700 hover:bg-amber-50 hover:text-amber-800 dark:border-amber-700 dark:text-amber-400 dark:hover:bg-amber-950/40"
   527→      >
   528→        {seedMutation.isPending ? (
   529→          <Loader2 className="h-3.5 w-3.5 mr-1.5 animate-spin" />
   530→        ) : (
   531→          <Cpu className="h-3.5 w-3.5 mr-1.5" />
   532→        )}
   533→        Seed Defaults
   534→      </Button>
   535→    </div>
   536→  );
   537→
   538→  return (
   539→    <div className="space-y-6">
   540→      {/* Page Header */}
   541→      {isPlatform ? (
   542→        <PlatformPageHeader
   543→          title="Email Templates"
   544→          subtitle="Platform-wide system email templates: welcome, payment, subscription, trial, invoice, and account lifecycle. Visible to all customers and sites."
   545→          actions={headerAction}
   546→        />
   547→      ) : (
   548→        <PageHeader
   549→          title="Email Templates"
   550→          description="Manage email templates for all system notifications, newsletters, and transactional emails."
   551→          action={headerAction}
   552→        />
   553→      )}
   554→
   555→      {/* Category Tabs */}
   556→      <div className="border-b">
   557→        <ScrollArea className="w-full">
   558→          <div className="flex items-center gap-1 pb-px overflow-x-auto">
   559→            {CATEGORIES.map((cat) => {
   560→              const isActive = category === cat.value;
   561→              const count =
   562→                cat.value === 'ALL'
   563→                  ? totalCount
   564→                  : (categoryCounts?.[cat.value] ?? 0);
   565→
   566→              return (
   567→                <button
   568→                  key={cat.value}
   569→                  onClick={() => handleCategoryChange(cat.value)}
   570→                  className={cn(
   571→                    'relative flex items-center gap-1.5 whitespace-nowrap px-3 py-2.5 text-sm font-medium transition-colors border-b-2 -mb-px',
   572→                    isActive
   573→                      ? 'border-primary text-foreground'
   574→                      : 'border-transparent text-muted-foreground hover:text-foreground hover:border-border',
   575→                  )}
   576→                >
   577→                  {cat.label}
   578→                  <span
   579→                    className={cn(
   580→                      'inline-flex h-5 min-w-5 items-center justify-center rounded-full px-1.5 text-[11px] font-medium leading-none',
   581→                      isActive
   582→                        ? 'bg-primary/10 text-primary'
   583→                        : 'bg-muted text-muted-foreground',
   584→                    )}
   585→                  >
   586→                    {count}
   587→                  </span>
   588→                </button>
   589→              );
   590→            })}
   591→          </div>
   592→        </ScrollArea>
   593→      </div>
   594→
   595→      {/* Search & Filter Bar */}
   596→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   597→        <div className="relative flex-1 max-w-sm">
   598→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   599→          <Input
   600→            placeholder="Search templates..."
   601→            value={searchInput}
   602→            onChange={(e) => updateSearch(e.target.value)}
   603→            className="pl-9 h-9"
   604→          />
   605→        </div>
   606→        <div className="flex items-center gap-2">
   607→          <Select value={statusFilter} onValueChange={handleStatusChange}>
   608→            <SelectTrigger size="sm" className="w-[150px]">
   609→              <SelectValue placeholder="All Statuses" />
   610→            </SelectTrigger>
   611→            <SelectContent>
   612→              {STATUS_OPTIONS.map((s) => (
   613→                <SelectItem key={s.value} value={s.value}>
   614→                  {s.label}
   615→                </SelectItem>
   616→              ))}
   617→            </SelectContent>
   618→          </Select>
   619→          <Select value={sortValue} onValueChange={handleSortChange}>
   620→            <SelectTrigger size="sm" className="w-[130px]">
   621→              <SelectValue placeholder="Newest" />
   622→            </SelectTrigger>
   623→            <SelectContent>
   624→              {SORT_OPTIONS.map((s) => (
   625→                <SelectItem key={s.value} value={s.value}>
   626→                  {s.label}
   627→                </SelectItem>
   628→              ))}
   629→            </SelectContent>
   630→          </Select>
   631→        </div>
   632→      </div>
   633→
   634→      {/* Table */}
   635→      <div className="rounded-lg border bg-card">
   636→        {isLoading ? (
   637→          <div className="p-6 space-y-3">
   638→            {Array.from({ length: 5 }).map((_, i) => (
   639→              <div key={i} className="flex items-center gap-4">
   640→                <Skeleton className="h-5 w-48" />
   641→                <Skeleton className="h-5 w-28" />
   642→                <Skeleton className="h-5 w-20" />
   643→                <Skeleton className="h-5 w-20" />
   644→                <Skeleton className="h-5 w-32" />
   645→                <Skeleton className="h-5 w-12" />
   646→                <Skeleton className="h-8 w-8 ml-auto" />
   647→              </div>
   648→            ))}
   649→          </div>
   650→        ) : templates.length === 0 ? (
   651→          <EmptyState
   652→            icon={Mail}
   653→            title="No templates found"
   654→            description={
   655→              search || statusFilter !== 'ALL' || category !== 'ALL'
   656→                ? 'Try adjusting your filters to find what you\'re looking for.'
   657→                : isPlatform
   658→                  ? 'Create your first platform template or seed the defaults to get started.'
   659→                  : 'Create your first email template or seed the defaults to get started.'
   660→            }
   661→            action={
   662→              !search && statusFilter === 'ALL' && category === 'ALL'
   663→                ? {
   664→                    label: 'Seed Default Templates',
   665→                    onClick: () => seedMutation.mutate(),
   666→                  }
   667→                : undefined
   668→            }
   669→          />
   670→        ) : (
   671→          <>
   672→            <Table>
   673→              <TableHeader>
   674→                <TableRow>
   675→                  <TableHead className="pl-4">Template Name</TableHead>
   676→                  <TableHead>Category</TableHead>
   677→                  <TableHead>Status</TableHead>
   678→                  <TableHead>Provider</TableHead>
   679→                  <TableHead>Last Updated</TableHead>
   680→                  <TableHead>Language</TableHead>
   681→                  <TableHead className="pr-4 w-12"></TableHead>
   682→                </TableRow>
   683→              </TableHeader>
   684→              <TableBody>
   685→                {templates.map((template) => (
   686→                  <TableRow
   687→                    key={template.id}
   688→                    className="cursor-pointer group"
   689→                    onClick={() => onEdit(template.id)}
   690→                  >
   691→                    {/* Template Name */}
   692→                    <TableCell className="pl-4">
   693→                      <div className="flex items-center gap-2 min-w-0">
   694→                        <span className="truncate font-medium text-foreground">
   695→                          {template.name}
   696→                        </span>
   697→                        {template.isSystem && (
   698→                          <Badge
   699→                            variant="outline"
   700→                            className="shrink-0 px-1.5 py-0 text-[10px] font-medium text-muted-foreground border-dashed"
   701→                          >
   702→                            System
   703→                          </Badge>
   704→                        )}
   705→                      </div>
   706→                    </TableCell>
   707→
   708→                    {/* Category */}
   709→                    <TableCell>
   710→                      <StatusBadge
   711→                        status={template.category}
   712→                        size="sm"
   713→                      />
   714→                    </TableCell>
   715→
   716→                    {/* Status */}
   717→                    <TableCell>
   718→                      <Badge
   719→                        variant="outline"
   720→                        className={cn(
   721→                          'border-transparent font-medium shrink-0 px-2 py-0.5 text-xs',
   722→                          getStatusColor(template.status),
   723→                        )}
   724→                      >
   725→                        {getStatusLabel(template.status)}
   726→                      </Badge>
   727→                    </TableCell>
   728→
   729→                    {/* Provider */}
   730→                    <TableCell>
   731→                      <Badge
   732→                        variant="outline"
   733→                        className={cn(
   734→                          'border-transparent shrink-0 px-2 py-0.5 text-xs font-medium',
   735→                          PROVIDER_COLORS[template.provider] ??
   736→                            'bg-zinc-100 text-zinc-700 dark:bg-zinc-800 dark:text-zinc-300',
   737→                        )}
   738→                      >
   739→                        {template.provider}
   740→                      </Badge>
   741→                    </TableCell>
   742→
   743→                    {/* Last Updated */}
   744→                    <TableCell className="text-muted-foreground text-sm">
   745→                      {formatRelativeTime(template.updatedAt)}
   746→                    </TableCell>
   747→
   748→                    {/* Language */}
   749→                    <TableCell>
   750→                      <Badge
   751→                        variant="outline"
   752→                        className="shrink-0 px-1.5 py-0 text-[10px] font-bold uppercase"
   753→                      >
   754→                        {template.language || 'EN'}
   755→                      </Badge>
   756→                    </TableCell>
   757→
   758→                    {/* Actions */}
   759→                    <TableCell className="pr-4" onClick={(e) => e.stopPropagation()}>
   760→                      <DropdownMenu>
   761→                        <DropdownMenuTrigger asChild>
   762→                          <Button
   763→                            variant="ghost"
   764→                            size="icon"
   765→                            className="h-8 w-8 opacity-0 group-hover:opacity-100 transition-opacity"
   766→                          >
   767→                            <MoreHorizontal className="h-4 w-4" />
   768→                          </Button>
   769→                        </DropdownMenuTrigger>
   770→                        <DropdownMenuContent align="end" className="w-48">
   771→                          <DropdownMenuItem onClick={() => onEdit(template.id)}>
   772→                            <Pencil className="h-4 w-4 mr-2" />
   773→                            Edit
   774→                          </DropdownMenuItem>
   775→                          <DropdownMenuItem onClick={() => onPreview(template.id)}>
   776→                            <Eye className="h-4 w-4 mr-2" />
   777→                            Preview
   778→                          </DropdownMenuItem>
   779→                          <DropdownMenuItem
   780→                            onClick={() => duplicateMutation.mutate(template.id)}
   781→                            disabled={duplicateMutation.isPending}
   782→                          >
   783→                            <Copy className="h-4 w-4 mr-2" />
   784→                            Duplicate
   785→                          </DropdownMenuItem>
   786→                          <DropdownMenuItem
   787→                            onClick={() => setSendTestTarget(template.id)}
   788→                          >
   789→                            <Send className="h-4 w-4 mr-2" />
   790→                            Send Test
   791→                          </DropdownMenuItem>
   792→                          <DropdownMenuSeparator />
   793→                          <DropdownMenuItem
   794→                            onClick={() =>
   795→                              toggleStatusMutation.mutate({
   796→                                id: template.id,
   797→                                status:
   798→                                  template.status === 'ENABLED'
   799→                                    ? 'DISABLED'
   800→                                    : 'ENABLED',
   801→                              })
   802→                            }
   803→                            disabled={toggleStatusMutation.isPending}
   804→                          >
   805→                            {template.status === 'ENABLED' ? (
   806→                              <>
   807→                                <span className="h-4 w-4 mr-2 inline-flex items-center justify-center rounded-full bg-zinc-200 dark:bg-zinc-700">
   808→                                  <span className="h-2 w-2 rounded-full" />
   809→                                </span>
   810→                                Disable
   811→                              </>
   812→                            ) : (
   813→                              <>
   814→                                <span className="h-4 w-4 mr-2 inline-flex items-center justify-center rounded-full bg-emerald-200 dark:bg-emerald-800">
   815→                                  <span className="h-2 w-2 rounded-full" />
   816→                                </span>
   817→                                Enable
   818→                              </>
   819→                            )}
   820→                          </DropdownMenuItem>
   821→                          {template.isSystem && (
   822→                            <DropdownMenuItem
   823→                              onClick={() => setRevertTarget(template)}
   824→                              className="text-amber-600 dark:text-amber-400"
   825→                            >
   826→                              <RotateCcw className="h-4 w-4 mr-2" />
   827→                              Revert to Default
   828→                            </DropdownMenuItem>
   829→                          )}
   830→                          <DropdownMenuSeparator />
   831→                          <DropdownMenuItem
   832→                            variant="destructive"
   833→                            onClick={() => setDeleteTarget(template)}
   834→                          >
   835→                            <Trash2 className="h-4 w-4 mr-2" />
   836→                            Delete
   837→                          </DropdownMenuItem>
   838→                        </DropdownMenuContent>
   839→                      </DropdownMenu>
   840→                    </TableCell>
   841→                  </TableRow>
   842→                ))}
   843→              </TableBody>
   844→            </Table>
   845→
   846→            {/* Pagination */}
   847→            {pagination && pagination.totalPages > 1 && (
   848→              <div className="flex items-center justify-between border-t px-4 py-3">
   849→                <div className="flex items-center gap-2 text-sm text-muted-foreground">
   850→                  <span>
   851→                    Page {pagination.page} of {pagination.totalPages}
   852→                  </span>
   853→                  <span className="text-muted-foreground/50">·</span>
   854→                  <span>{pagination.total} total</span>
   855→                </div>
   856→                <div className="flex items-center gap-3">
   857→                  <Select
   858→                    value={String(pageSize)}
   859→                    onValueChange={(v) => {
   860→                      setPageSize(Number(v));
   861→                      setPage(1);
   862→                    }}
   863→                  >
   864→                    <SelectTrigger size="sm" className="w-[80px] h-8">
   865→                      <SelectValue />
   866→                    </SelectTrigger>
   867→                    <SelectContent>
   868→                      {PAGE_SIZES.map((s) => (
   869→                        <SelectItem key={s} value={String(s)}>
   870→                          {s}
   871→                        </SelectItem>
   872→                      ))}
   873→                    </SelectContent>
   874→                  </Select>
   875→                  <div className="flex items-center gap-1">
   876→                    <Button
   877→                      variant="outline"
   878→                      size="icon"
   879→                      className="h-8 w-8"
   880→                      disabled={pagination.page <= 1}
   881→                      onClick={() => setPage((p) => p - 1)}
   882→                    >
   883→                      <ChevronLeft className="h-4 w-4" />
   884→                    </Button>
   885→                    <Button
   886→                      variant="outline"
   887→                      size="icon"
   888→                      className="h-8 w-8"
   889→                      disabled={pagination.page >= pagination.totalPages}
   890→                      onClick={() => setPage((p) => p + 1)}
   891→                    >
   892→                      <ChevronRight className="h-4 w-4" />
   893→                    </Button>
   894→                  </div>
   895→                </div>
   896→              </div>
   897→            )}
   898→          </>
   899→        )}
   900→      </div>
   901→
   902→      {/* Send Test Dialog */}
   903→      <SendTestDialog
   904→        open={!!sendTestTarget}
   905→        onOpenChange={(v) => !v && setSendTestTarget(null)}
   906→        templateId={sendTestTarget}
   907→      />
   908→
   909→      {/* Delete Confirm Dialog */}
   910→      <ConfirmDialog
   911→        open={!!deleteTarget}
   912→        onOpenChange={(v) => !v && setDeleteTarget(null)}
   913→        title="Delete Template"
   914→        description={
   915→          deleteTarget
   916→            ? `Are you sure you want to delete "${deleteTarget.name}"? This action cannot be undone.`
   917→            : undefined
   918→        }
   919→        confirmLabel="Delete"
   920→        variant="destructive"
   921→        onConfirm={() => {
   922→          if (deleteTarget) deleteMutation.mutate(deleteTarget.id);
   923→        }}
   924→        isLoading={deleteMutation.isPending}
   925→      />
   926→
   927→      {/* Revert Confirm Dialog */}
   928→      <ConfirmDialog
   929→        open={!!revertTarget}
   930→        onOpenChange={(v) => !v && setRevertTarget(null)}
   931→        title="Revert to Default"
   932→        description={
   933→          revertTarget
   934→            ? `This will reset "${revertTarget.name}" to its original default content. Your current changes will be saved as a version before reverting.`
   935→            : undefined
   936→        }
   937→        confirmLabel="Revert"
   938→        onConfirm={() => {
   939→          if (revertTarget) revertMutation.mutate(revertTarget.id);
   940→        }}
   941→        isLoading={revertMutation.isPending}
   942→      />
   943→
   944→      {/* Seed Defaults Confirm Dialog */}
   945→      <ConfirmDialog
   946→        open={seedDialogOpen}
   947→        onOpenChange={setSeedDialogOpen}
   948→        title="Seed Default Templates"
   949→        description="This will add any missing default system email templates. Existing templates (including your custom ones) will not be modified."
   950→        confirmLabel="Seed Defaults"
   951→        onConfirm={() => seedMutation.mutate()}
   952→        isLoading={seedMutation.isPending}
   953→      />
   954→    </div>
   955→  );
   956→}
