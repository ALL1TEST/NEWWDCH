    95→  excerpt,

   104→  excerpt: string;

   115→    year: 'numeric', month: 'long', day: 'numeric',

   127→          variant="outline"

   134→        </Button>
   135→      </div>

   137→      {/* Article Preview */}
   138→      <div className="mx-auto max-w-3xl px-6 py-12">

   140→        {featuredImageUrl && (
   141→          <div className="mb-8 overflow-hidden rounded-xl">
   142→            <img src={featuredImageUrl} alt={title} className="w-full h-auto object-cover" />

   144→        )}
   145→
   146→        {/* Title */}
   147→        <h1 className="text-3xl font-bold tracking-tight sm:text-4xl">{title || 'Untitled Article'}</h1>
   148→
   149→        {/* Metadata */}

   166→        {tags.length > 0 && (

   168→            {tags.map((tag) => (

   183→          className="prose prose-gray dark:prose-invert max-w-none prose-headings:font-bold prose-h2:text-2xl prose-h3:text-xl prose-p:leading-relaxed prose-a:text-primary"
   184→          dangerouslySetInnerHTML={{ __html: content || '<p class="text-muted-foreground">No content yet.</p>' }}

   441→  }, []);
   442→
   443→  const clearSavedSelection = useCallback(() => {
   444→    setSavedSelectedText('');
   445→    setSelectedText('');
   446→  }, []);
   447→
   448→  // Dialog states
   449→  const [previewOpen, setPreviewOpen] = useState(false);
   450→  const [scheduleOpen, setScheduleOpen] = useState(false);
   451→  const [mediaLibraryOpen, setMediaLibraryOpen] = useState(false);
   452→  const [aiAssistOpen, setAiAssistOpen] = useState(false);
   453→
   454→  // Fetch dropdown data
   455→  const { data: contentTypes } = useQuery({
   456→    queryKey: queryKeys.contentTypes.all,
   457→    queryFn: () => getApi<ContentTypeOption[]>('/api/content-types?pageSize=100'),
   458→    staleTime: 60_000,
   459→  });
   460→  const { data: categories } = useQuery({
   461→    queryKey: queryKeys.categories.all,
   462→    queryFn: () => getApi<CategoryOption[]>('/api/categories?pageSize=200'),
   463→    staleTime: 60_000,
   464→  });
   465→  const { data: allTags, refetch: refetchTags } = useQuery({
   466→    queryKey: queryKeys.tags.all,
   467→    queryFn: () => getApi<TagOption[]>('/api/tags?pageSize=200'),
   468→    staleTime: 60_000,
   469→  });
   470→
   471→  const filteredTags = useMemo(() => {
   472→    if (!allTags) return [];
   473→    if (!tagSearch) return allTags;
   474→    return allTags.filter((t) => t.name.toLowerCase().includes(tagSearch.toLowerCase()));
   475→  }, [allTags, tagSearch]);
   476→
   477→  const {
   478→    register, handleSubmit, control, watch, setValue, getValues, formState: { errors },
   479→  } = useForm<ContentFormValues>({
   480→    resolver: zodResolver(contentFormSchema),
   481→    defaultValues: { title: '', excerpt: '', content: '', status: 'DRAFT', contentTypeId: '', categoryId: '', tagIds: [], seoTitle: '', seoDescription: '' },
   482→  });
   483→
   484→  const selectedTagIds = watch('tagIds');
   485→  const watchedTitle = watch('title');
   486→  const watchedExcerpt = watch('excerpt');
   487→  const watchedSeoTitle = watch('seoTitle');
   488→  const watchedSeoDescription = watch('seoDescription');
   489→
   490→  // Auto-slug
   491→  React.useEffect(() => {
   492→    setSlugValue(slugify(watchedTitle));
   493→  }, [watchedTitle]);
   494→
   495→  // Create mutation
   496→  const createMutation = useMutation({
   497→    mutationFn: (data: ContentFormValues & { scheduledAt?: string }) => {
   498→      const slug = slugify(data.title);
   499→      return postApi<CreatedContent>('/api/content', {
   500→        title: data.title,
   501→        slug,
   502→        excerpt: data.excerpt || undefined,
   503→        content: editorContent || undefined,
   504→        status: data.status,
   505→        contentTypeId: data.contentTypeId,
   506→        categoryId: data.categoryId || undefined,
   507→        tagIds: data.tagIds,
   508→        seoTitle: data.seoTitle || undefined,
   509→        seoDescription: data.seoDescription || undefined,
   510→        featuredImageId: featuredImage?.id || undefined,
   511→        scheduledAt: data.scheduledAt || undefined,
   512→      } as Record<string, unknown>);
   513→    },
   514→    onSuccess: (created) => {
   515→      queryClient.invalidateQueries({ queryKey: queryKeys.content.all });
   516→      toast.success('Content created successfully');
   517→      navigate('content', created.id);
   518→    },
   519→    onError: (err: Error) => toast.error(err.message || 'Failed to create content'),
   520→  });
   521→
   522→  // Create tag mutation
   523→  const createTagMutation = useMutation({
   524→    mutationFn: (name: string) => postApi<TagOption>('/api/tags', { name }),
   525→    onSuccess: (newTag) => {
   526→      refetchTags();
   527→      if (!selectedTagIds.includes(newTag.id)) {
   528→        setValue('tagIds', [...selectedTagIds, newTag.id], { shouldValidate: true });
   529→      }
   530→      setCustomTags((prev) => [...prev, newTag.name]);
   531→      toast.success(`Tag "${newTag.name}" created`);
   532→    },
   533→    onError: (err: Error) => toast.error(err.message || 'Failed to create tag'),
   534→  });
   535→
   536→  // Upload featured image mutation
   537→  const uploadMutation = useMutation({
   538→    mutationFn: async (file: File) => {
   539→      const formData = new FormData();
   540→      formData.append('file', file);
   541→      const res = await fetch('/api/media/upload', {
   542→        method: 'POST',
   543→        body: formData,
   544→      });
   545→      if (!res.ok) {
   546→        const err = await res.json().catch(() => ({}));
   547→        throw new Error(err?.error?.message || 'Upload failed');
   548→      }
   549→      const json = await res.json();
   550→      return json.data as MediaItem;
   551→    },
   552→    onSuccess: (media) => {
