     1→// ============================================================
     2→// STRIPE CLIENT + CONFIGURATION GUARD.
     3→// ============================================================
     4→// This is the SINGLE place that resolves Stripe credentials and
     5→// constructs the Stripe SDK client. The rest of the codebase calls
     6→// `isStripeConfigured()` to gate real billing; when false, checkout
     7→// returns 503 with a clear "not configured" message and NEVER fakes
     8→// a successful payment.
     9→//
    10→// CREDENTIAL RESOLUTION ORDER (per credential type):
    11→//   1. The StripeSettings singleton row in the DB (admin-configured
    12→//      via Platform Admin → Stripe Settings). Secret keys are stored
    13→//      AES-256-GCM encrypted; only the masked form is returned to the
    14→//      frontend. Publishable keys are stored as plaintext.
    15→//   2. process.env.STRIPE_* (fallback — works without any DB row,
    16→//      e.g. on first deploy before the admin has connected Stripe).
    17→//
    18→// So an admin can connect Stripe from the UI without editing .env,
    19→// and existing .env-based deployments keep working.
    20→//
    21→// Stripe SDK is only ever imported server-side (this file is imported
    22→// by API routes and webhook handlers). The publishable key is exposed
    23→// to the client only via /api/billing/checkout (GET) (so the client
    24→// can show the right UI without leaking the secret key).
    25→// ============================================================
    26→
    27→import Stripe from 'stripe';
    28→import { encrypt, decrypt, maskSecret } from '@/lib/encryption';
    29→
    30→/** The Unicode bullet character (U+2022) used by `maskSecret` to mask
    31→ *  the middle of secret keys. Used to detect when the admin has
    32→ *  copy-pasted the MASKED form of a secret back into the input —
    33→ *  storing that would silently break Stripe API calls (Stripe would
    34→ *  reject the masked string as an "Invalid API Key"). */
    35→const MASK_BULLET = '\u2022'; // = '•'
    36→
    37→/** True when the value looks like a masked secret (contains the mask
    38→ *  bullet character). Used to reject masked form values at the API
    39→ *  boundary so an admin can never accidentally persist the masked
    40→ *  form as the real secret. */
    41→export function isMaskedSecretValue(value: string): boolean {
    42→  return typeof value === 'string' && value.includes(MASK_BULLET);
    43→}

    46→// -------------------- DB-backed settings cache --------------------
    47→// In-memory cache of the StripeSettings singleton row. Re-read on
    48→// demand (invalidateStripeSettingsCache below) — call after every
    49→// mutation (save / test-connection) so the next getStripeClient()
    50→// picks up the new credentials without a dev-server restart.
    51→interface ResolvedStripeSettings {
    52→  mode: 'test' | 'live';
    53→  secretKey: string | null; // resolved plaintext (decrypted) for the active mode
    54→  publishableKey: string | null;
    55→  webhookSecret: string | null;
    56→  appUrl: string | null;
    57→}
    58→
    59→let _resolvedCache: ResolvedStripeSettings | null = null;
    60→let _resolvedCacheAt = 0;
    61→const RESOLVED_TTL_MS = 5_000; // 5s — keeps reads fast but picks up edits quickly
    62→
    63→/** Invalidate the in-memory cache of resolved Stripe settings.
    64→ *  Call after every saveStripeSettings() / mutation so the next
    65→ *  getStripeClient() reads the fresh DB row. */
    66→export function invalidateStripeSettingsCache(): void {
    67→  _resolvedCache = null;
    68→  _resolvedCacheAt = 0;
    69→  _client = null; // force the SDK client to be reconstructed on next use
    70→}
    71→
    72→/** Read the StripeSettings singleton row from the DB and decrypt
    73→ *  the secret keys for the ACTIVE mode. Returns null when no row
    74→ *  exists yet. Never throws — every error degrades to null. */
    75→async function loadResolvedSettings(): Promise<ResolvedStripeSettings | null> {
    76→  const { db } = await import('@/lib/db');
    77→  try {
    78→    const row = await db.stripeSettings.findUnique({ where: { id: 'singleton' } });
    79→    if (!row) return null;
    80→    const mode = (row.mode === 'live' ? 'live' : 'test') as 'test' | 'live';
    81→    const secretEncrypted = mode === 'live' ? row.secretKeyLiveEncrypted : row.secretKeyTestEncrypted;
    82→    const webhookEncrypted = mode === 'live' ? row.webhookSecretLiveEncrypted : row.webhookSecretTestEncrypted;
    83→    const publishable = mode === 'live' ? row.publishableKeyLive : row.publishableKeyTest;
    84→
    85→    let secretKey: string | null = null;
    86→    if (secretEncrypted) {
    87→      try {
    88→        secretKey = await decrypt(secretEncrypted);
    89→      } catch {
    90→        secretKey = null;
    91→      }
    92→    }
    93→    let webhookSecret: string | null = null;
    94→    if (webhookEncrypted) {
    95→      try {
    96→        webhookSecret = await decrypt(webhookEncrypted);
    97→      } catch {
    98→        webhookSecret = null;
    99→      }
   100→    }
   101→    return {
   102→      mode,
   103→      secretKey,
   104→      publishableKey: publishable ?? null,
   105→      webhookSecret,
   106→      appUrl: row.appUrl ?? null,
   107→    };
   108→  } catch {
   109→    // Table not yet created, db not reachable, etc. → fall back to env.
   110→    return null;
   111→  }
   112→}
   113→
   114→/** Get the resolved Stripe settings (DB → env fallback), with a
   115→ *  5s in-memory cache so it's cheap to call repeatedly. Public so
   116→ *  the admin settings page can read the masked view + the test
   117→ *  connection route can show what it tested with. */
   118→export async function getResolvedStripeSettings(): Promise<ResolvedStripeSettings & {
   119→  source: 'db' | 'env' | 'none';
   120→}> {
   121→  // 1. DB row (admin-configured).
   122→  const now = Date.now();
   123→  if (_resolvedCache && now - _resolvedCacheAt < RESOLVED_TTL_MS) {
   124→    return { ..._resolvedCache, source: 'db' };
   125→  }
   126→  const fromDb = await loadResolvedSettings();
   127→  if (fromDb && (fromDb.secretKey || fromDb.publishableKey || fromDb.webhookSecret)) {
   128→    _resolvedCache = fromDb;
   129→    _resolvedCacheAt = now;
   130→    return { ...fromDb, source: 'db' };
   131→  }
   132→  // 2. Env fallback.
   133→  const envSettings: ResolvedStripeSettings = {
   134→    mode: process.env.STRIPE_SECRET_KEY?.startsWith('sk_live_') ? 'live' : 'test',
   135→    secretKey: process.env.STRIPE_SECRET_KEY && process.env.STRIPE_SECRET_KEY.trim().length > 0
   136→      ? process.env.STRIPE_SECRET_KEY
   137→      : null,
   138→    publishableKey: process.env.STRIPE_PUBLISHABLE_KEY && process.env.STRIPE_PUBLISHABLE_KEY.trim().length > 0
   139→      ? process.env.STRIPE_PUBLISHABLE_KEY
   140→      : null,
   141→    webhookSecret: process.env.STRIPE_WEBHOOK_SECRET && process.env.STRIPE_WEBHOOK_SECRET.trim().length > 0
   142→      ? process.env.STRIPE_WEBHOOK_SECRET
   143→      : null,
   144→    appUrl: process.env.STRIPE_APP_URL ?? null,
   145→  };
   146→  // If env has no usable secret key either, source is 'none' (the
   147→  // admin UI shows "Stripe is not connected — configure below").
   148→  const hasEnvCreds = Boolean(envSettings.secretKey) || Boolean(envSettings.publishableKey) || Boolean(envSettings.webhookSecret);
   149→  return { ...envSettings, source: hasEnvCreds ? 'env' : 'none' };
   150→}

   152→/** True when Stripe secret key is configured (DB or env). */
   153→export async function isStripeConfiguredAsync(): Promise<boolean> {
   154→  const s = await getResolvedStripeSettings();
   155→  return Boolean(s.secretKey && s.secretKey.trim().length > 0);
   156→}

   158→/** Sync version — checks env vars only. Kept for backwards compat
   159→ *  with callers that haven't been migrated to the async pattern
   160→ *  (e.g. the webhook handler's pre-flight guard). Prefer

   162→export function isStripeConfigured(): boolean {
   163→  return Boolean(process.env.STRIPE_SECRET_KEY && process.env.STRIPE_SECRET_KEY.trim().length > 0);
   164→}

   166→let _client: Stripe | null = null;

   168→/** Get the shared Stripe SDK client. Reads the secret key from DB
   169→ *  first, then falls back to env. Throws if neither is configured —
   170→ *  call isStripeConfiguredAsync() first. */
   171→export async function getStripeClient(): Promise<Stripe> {
   172→  const settings = await getResolvedStripeSettings();
   173→  const key = settings.secretKey;
   174→  if (!key) throw new Error('Stripe is not configured. Set credentials in Platform Admin → Stripe Settings or in .env.');
   175→  // If the cache is fresh and the key matches, reuse the SDK client.
   176→  if (_client && _cachedClientKey === key) return _client;
   177→  _client = new Stripe(key, {
   178→    // Pin to a stable API version. Update together with the SDK.
   179→    apiVersion: '2026-08-26.dahlia' as Stripe.LatestApiVersion,
   180→    typescript: true,
   181→    // Surface errors as detailed objects (don't hide fields).
   182→    maxNetworkRetries: 2,
   183→  });
   184→  _cachedClientKey = key;
   185→  return _client;
   186→}
   187→let _cachedClientKey: string | null = null;

   206→/**
   207→ * Resolve the Stripe Price ID for a (planId, billingInterval) pair.
   208→ * Reads from PlanConfig.stripePriceIdMonthly / stripePriceIdYearly
   209→ * (set via Platform Admin → Edit Plan).
   210→ *
   211→ * Returns null when the plan has no Stripe price wired for the
   212→ * requested interval — the caller MUST refuse to fake checkout.
   213→ *
   214→ * NOTE: this returns the DEFAULT currency's Stripe Price ID. For
   215→ * multi-currency checkout, use resolveStripePriceIdForCurrency()
   216→ * which looks up the per-currency Stripe Price ID in
   217→ * stripePriceIdsByCurrency.
   218→ */
   219→export async function resolveStripePriceId(
   220→  planId: string,
   221→  interval: 'monthly' | 'yearly',
   222→): Promise<string | null> {
   223→  const { db } = await import('@/lib/db');
   224→  const row = await db.planConfig.findUnique({ where: { planId } });
   225→  if (!row) return null;
   226→  if (interval === 'yearly') return row.stripePriceIdYearly ?? null;
   227→  return row.stripePriceIdMonthly ?? null;
   228→}

   230→/**
   231→ * Resolve the Stripe Price ID for a (planId, currency, interval) pair
   232→ * from the authoritative per-currency map stored on PlanConfig. Falls
   233→ * back to the default-currency snapshot when the requested currency
   234→ * has no per-currency Stripe Price wired (legacy compat). Returns
   235→ * null when no Stripe Price is wired for the requested currency +
   236→ * interval — the caller MUST refuse to fake checkout.
   237→ *
   238→ * Currency is normalized to uppercase (USD, EUR, CHF, MAD).
   239→ */
   240→export async function resolveStripePriceIdForCurrency(
   241→  planId: string,
   242→  currency: string,
   243→  interval: 'monthly' | 'yearly',
   244→): Promise<string | null> {
   245→  const { db } = await import('@/lib/db');
   246→  const row = await db.planConfig.findUnique({ where: { planId } });
   247→  if (!row) return null;
   248→  const curUpper = currency.trim().toUpperCase();
   249→  // 1. Try the authoritative per-currency map.
   250→  let byCurrency: Record<string, { monthly: string | null; yearly: string | null }> = {};
   251→  try {
   252→    byCurrency = JSON.parse(row.stripePriceIdsByCurrency || '{}');
   253→  } catch {
   254→    byCurrency = {};
   255→  }
   256→  const entry = byCurrency[curUpper];
   257→  if (entry) {
   258→    const id = interval === 'yearly' ? entry.yearly : entry.monthly;
   259→    if (id && id.trim().length > 0) return id;
   260→  }
   261→  // 2. Fall back to the default-currency snapshot fields. This keeps
   262→  //    backward compatibility with plans that were synced before the
   263→  //    multi-currency fields existed.
   264→  if (interval === 'yearly') return row.stripePriceIdYearly ?? null;
   265→  return row.stripePriceIdMonthly ?? null;
   266→}

   268→/** Verify the Stripe webhook signature. Reads the webhook secret from
   269→ *  DB first, then falls back to STRIPE_WEBHOOK_SECRET. Returns the
   270→ *  verified event or null if verification fails (or Stripe isn't
   271→ *  configured). */
   272→export async function verifyStripeWebhook(
   273→  rawBody: Buffer | string,
   274→  signature: string,
   275→): Promise<Stripe.Event | null> {
   276→  const settings = await getResolvedStripeSettings();
   277→  const secret = settings.webhookSecret;
   278→  if (!secret) return null;
   279→  try {
   280→    const stripe = await getStripeClient();
   281→    return await stripe.webhooks.constructEventAsync(
   282→      rawBody instanceof Buffer ? rawBody : Buffer.from(rawBody),
   283→      signature,
   284→      secret,
   285→    );
   286→  } catch {
   287→    return null;
   288→  }
   289→}

   302→/**
   303→ * Return the Stripe Customer id for `user`, reusing an existing
   304→ * Customer when `existingCustomerId` resolves to a live (non-deleted)
   305→ * Stripe Customer, otherwise creating a new Customer tagged with
   306→ * `metadata.userId`. On retrieve failure of an existing id, falls
   307→ * through to create.
   308→ */
   309→export async function getOrCreateStripeCustomer(
   310→  stripe: Stripe,
   311→  user: { id: string; email: string; name?: string | null },
   312→  existingCustomerId?: string | null,
   313→): Promise<string> {
   314→  if (existingCustomerId) {
   315→    try {
   316→      const customer = await stripe.customers.retrieve(existingCustomerId);
   317→      if (customer && !(customer as { deleted?: boolean }).deleted) {
   318→        return customer.id;
   319→      }
   320→    } catch {
   321→      // fall through to create a new Customer
   322→    }
   323→  }
   324→  const created = await stripe.customers.create({
   325→    email: user.email,
   326→    name: user.name ?? undefined,
   327→    metadata: { userId: user.id },
   328→  });
   329→  return created.id;
   330→}

   332→/**
   333→ * Mirror a local Coupon into a Stripe Coupon + Promotion Code so
   334→ * Stripe Checkout can apply it as a `discounts[]` entry. Reuses
   335→ * existing Stripe objects when present (idempotent). Returns the
   336→ * Stripe Coupon id + Promotion Code id so the caller can cache
   337→ * them on the local Coupon row.
   338→ */
   339→export async function createStripeCouponMirror(
   340→  stripe: Stripe,
   341→  coupon: {
   342→    code: string;
   343→    type: 'percent' | 'fixed';
   344→    value: number;
   345→    currency: string;
   346→    maxRedemptions?: number | null;
   347→    perCustomerLimit?: number | null;
   348→    expiresAt?: string | null;
   349→    startsAt?: string | null;
   350→  },
   351→): Promise<{ stripeCouponId: string; stripePromotionCodeId: string }> {
   352→  // Derive a safe Stripe Coupon id: local_<code_lowered_alphanum>, truncate to 40 chars.
   353→  const derivedId = `local_${coupon.code.toLowerCase().replace(/[^a-z0-9]/g, '')}`.slice(0, 40);
   354→
   355→  // Reuse an existing Stripe Coupon with that id, else create one.
   356→  let stripeCouponId: string;
   357→  try {
   358→    const existing = await stripe.coupons.retrieve(derivedId);
   359→    stripeCouponId = existing.id;
   360→  } catch {
   361→    const created = await stripe.coupons.create({
   362→      id: derivedId,
   363→      percent_off: coupon.type === 'percent' ? coupon.value : undefined,
   364→      amount_off: coupon.type === 'fixed' ? coupon.value * 100 : undefined,
   365→      currency: coupon.type === 'fixed' ? coupon.currency.toLowerCase() : undefined,
   366→      duration: 'once',
   367→      max_redemptions: coupon.maxRedemptions ?? undefined,
   368→      redeem_by: coupon.expiresAt ? Math.floor(new Date(coupon.expiresAt).getTime() / 1000) : undefined,
   369→      metadata: { localCode: coupon.code },
   370→    });
   371→    stripeCouponId = created.id;
   372→  }
   373→
   374→  // Create-or-reuse a Stripe Promotion Code with code: coupon.code.
   375→  let stripePromotionCodeId: string;
   376→  const existingPromos = await stripe.promotionCodes.list({ code: coupon.code, limit: 1 });
   377→  if (existingPromos.data.length > 0) {
   378→    stripePromotionCodeId = existingPromos.data[0].id;
   379→  } else {
   380→    const promo = await stripe.promotionCodes.create({
   381→      coupon: stripeCouponId,
   382→      code: coupon.code,
   383→      max_redemptions: coupon.maxRedemptions ?? undefined,
   384→      expires_at: coupon.expiresAt ? Math.floor(new Date(coupon.expiresAt).getTime() / 1000) : undefined,
   385→      metadata: { localCode: coupon.code },
   386→    });
   387→    stripePromotionCodeId = promo.id;
   388→  }
   389→
   390→  return { stripeCouponId, stripePromotionCodeId };
   391→}
   392→
   393→/**
   394→ * Best-effort teardown of a local Coupon's Stripe mirror: deactivate
   395→ * the Promotion Code (so it can no longer be redeemed) then delete the
   396→ * Stripe Coupon. Never throws — all errors are swallowed. Safe to call
   397→ * with null/undefined ids (no-op).
   398→ */
   399→export async function clearStripeCouponMirror(
   400→  stripe: Stripe,
   401→  stripeCouponId?: string | null,
   402→  stripePromotionCodeId?: string | null,
   403→): Promise<void> {
   404→  if (stripePromotionCodeId) {
   405→    try {
   406→      await stripe.promotionCodes.update(stripePromotionCodeId, { active: false });
   407→    } catch {
   408→      // best-effort — never throw
   409→    }
   410→  }
   411→  if (stripeCouponId) {
   412→    try {
   413→      await stripe.coupons.del(stripeCouponId);
   414→    } catch {
   415→      // best-effort — never throw
   416→    }
   417→  }
   418→}

   420→/**
   421→ * Sync a local PlanConfig to Stripe by ensuring both a monthly and a
   422→ * yearly Stripe Price exist (with `metadata.planId` + `metadata.interval`
   423→ * set so they can be discovered later). Reuses existing Stripe Prices
   424→ * / Products where possible. Returns the resolved { monthly, yearly }
   425→ * Stripe Price ids (existing + any newly created).
   426→ *
   427→ * PRICE-CHANGE DETECTION: when an existing Stripe Price ID is provided
   428→ * AND the corresponding local price (priceMonthly / priceYearly)
   429→ * changed since the last sync, this function retrieves the Stripe
   430→ * Price, compares its `unit_amount` against the locally-configured
   431→ * amount, and — on mismatch — creates a NEW Stripe Price with the
   432→ * updated amount (the OLD Stripe Price is NOT archived — existing
   433→ * subscriptions may still be on it; Stripe keeps them on the original
   434→ * price until they upgrade / cancel / prorate). The local plan row
   435→ * is then updated with the new Stripe Price ID by the caller.
   436→ */
   437→export async function syncPlanToStripe(
   438→  stripe: Stripe,
   439→  plan: {
   440→    planId: string;
   441→    name: string;
   442→    priceMonthly: number;
   443→    priceYearly: number;
   444→    currency: string;
   445→    stripePriceIdMonthly?: string | null;
   446→    stripePriceIdYearly?: string | null;
   447→  },
   448→): Promise<{ stripePriceIdMonthly: string; stripePriceIdYearly: string }> {
   449→  const resolveInterval = async (interval: 'monthly' | 'yearly'): Promise<string> => {
   450→    const existing = interval === 'monthly' ? plan.stripePriceIdMonthly : plan.stripePriceIdYearly;
   451→    const expectedAmount = (interval === 'monthly' ? plan.priceMonthly : plan.priceYearly) * 100;
   452→
   453→    if (existing && existing.trim().length > 0) {
   454→      // PRICE-CHANGE CHECK: retrieve the existing Stripe Price and
   455→      // compare its unit_amount against the locally-configured amount.
   456→      // On mismatch (the admin edited the price), create a NEW Stripe
   457→      // Price with the updated amount. The OLD Stripe Price is kept
   458→      // active — existing Stripe subscriptions on it stay on the
   459→      // original price until they renew / prorate / cancel. New
   460→      // checkouts will use the new Price ID (the local plan row is
   461→      // updated by the caller with the resolved ID).
   462→      try {
   463→        const existingPrice = await stripe.prices.retrieve(existing);
   464→        const existingAmount = existingPrice.unit_amount ?? 0;
   465→        if (existingAmount !== expectedAmount) {
   466→          // Price changed → create a new Stripe Price on the same
   467→          // Product. Reuse the existing Price's `product` so we don't
   468→          // create a duplicate Product.
   469→          const productId =
   470→            typeof existingPrice.product === 'string'
   471→              ? existingPrice.product
   472→              : existingPrice.product && 'id' in existingPrice.product
   473→              ? existingPrice.product.id
   474→              : undefined;
   475→          if (!productId) {
   476→            // Can't resolve product → keep the existing Price ID.
   477→            return existing;
   478→          }
   479→          const stripeInterval: 'month' | 'year' = interval === 'monthly' ? 'month' : 'year';
   480→          const newPrice = await stripe.prices.create({
   481→            unit_amount: expectedAmount,
   482→            currency: plan.currency.toLowerCase(),
   483→            recurring: { interval: stripeInterval },
   484→            product: productId,
   485→            metadata: { planId: plan.planId, interval },
   486→          });
   487→          return newPrice.id;
   488→        }
   489→        // Amount matches → keep the existing Stripe Price ID.
   490→        return existing;
   491→      } catch {
   492→        // Retrieval failed (deleted? permission?) → fall through to
   493→        // the lookup-by-metadata + create branch. The admin can use
   494→        // the explicit "Sync to Stripe" route to surface real errors.
   495→      }
   496→    }
   497→
   498→    // Try to find an existing active Stripe Price with matching metadata.
   499→    const list = await stripe.prices.list({ active: true, limit: 100 });
   500→    const match = list.data.find(
   501→      (p) => p.metadata?.planId === plan.planId && p.metadata?.interval === interval,
   502→    );
   503→    if (match) {
   504→      // Verify the matched Price's amount matches the locally-configured
   505→      // amount — if not, create a fresh Price (same logic as above).
   506→      const matchAmount = match.unit_amount ?? 0;
   507→      if (matchAmount === expectedAmount) return match.id;
   508→      // Amount mismatch → create new on the same Product.
   509→      const productId =
   510→        typeof match.product === 'string'
   511→          ? match.product
   512→          : match.product && 'id' in match.product
   513→          ? match.product.id
   514→          : undefined;
   515→      if (productId) {
   516→        const stripeInterval: 'month' | 'year' = interval === 'monthly' ? 'month' : 'year';
   517→        const newPrice = await stripe.prices.create({
   518→          unit_amount: expectedAmount,
   519→          currency: plan.currency.toLowerCase(),
   520→          recurring: { interval: stripeInterval },
   521→          product: productId,
   522→          metadata: { planId: plan.planId, interval },
   523→        });
   524→        return newPrice.id;
   525→      }
   526→    }
   527→
   528→    // Create-or-reuse a Stripe Product tagged with metadata.planId.
   529→    let productId: string | undefined;
   530→    try {
   531→      const existingProducts = await stripe.products.list({ limit: 100 });
   532→      const existingProduct = existingProducts.data.find(
   533→        (p) => p.metadata?.planId === plan.planId,
   534→      );
   535→      if (existingProduct) productId = existingProduct.id;
   536→    } catch {
   537→      // fall through to create
   538→    }
   539→    if (!productId) {
   540→      const product = await stripe.products.create({
   541→        name: plan.name,
   542→        metadata: { planId: plan.planId },
   543→      });
   544→      productId = product.id;
   545→    }
   546→
   547→    // Stripe's recurring.interval enum is 'day' | 'week' | 'month' | 'year'
   548→    // — map our local 'monthly'/'yearly' terms onto it.
   549→    const stripeInterval: 'month' | 'year' = interval === 'monthly' ? 'month' : 'year';
   550→    const price = await stripe.prices.create({
   551→      unit_amount: expectedAmount,
   552→      currency: plan.currency.toLowerCase(),
   553→      recurring: { interval: stripeInterval },
   554→      product: productId,
   555→      metadata: { planId: plan.planId, interval },
   556→    });
   557→    return price.id;
   558→  };
   559→
   560→  const stripePriceIdMonthly = await resolveInterval('monthly');
   561→  const stripePriceIdYearly = await resolveInterval('yearly');
   562→  return { stripePriceIdMonthly, stripePriceIdYearly };
   563→}

   566→// MULTI-CURRENCY STRIPE SYNC
   567→// ============================================================
   568→// syncPlanToStripeMulti creates/reuses a single Stripe Product (per
   569→// plan) + a Stripe Price PER (currency, interval) — so a single plan
   570→// can be sold in USD, EUR, CHF, MAD at independent price points and
   571→// each (currency, interval) pair has its own real Stripe Price ID.
   572→// Returns the resolved stripePriceIdsByCurrency map so the caller can
   573→// persist it onto PlanConfig.
   574→//
   575→// PRICE-CHANGE DETECTION: per (currency, interval), when an existing
   576→// Stripe Price ID is provided AND the corresponding locally-configured
   577→// amount changed, a NEW Stripe Price is created on the same Product
   578→// with the updated amount. The OLD Price is kept active so existing
   579→// subscriptions on it stay on the original price (Stripe pattern).
   580→//
   581→// ZERO-PRICE SKIP: when a (currency, interval) pair has a 0 price,
   582→// the existing Stripe Price (if any) is left untouched and NO new
   583→// Price is created. The returned ID for that pair is null.
   584→// ============================================================
   585→
   586→export interface SyncMultiInput {
   587→  planId: string;
   588→  name: string;
   589→  defaultCurrency: string;
   590→  /** Map of currency → { monthly, yearly } prices. */
   591→  pricesByCurrency: Record<string, { monthly: number; yearly: number }>;
   592→  /** Existing Stripe Price IDs per currency. */
   593→  stripePriceIdsByCurrency?: Record<string, { monthly: string | null; yearly: string | null }>;
   594→}
   595→
   596→export interface SyncMultiResult {
   597→  stripePriceIdsByCurrency: Record<string, { monthly: string | null; yearly: string | null }>;
   598→  /** Number of NEW Stripe Prices created during this sync. */
   599→  created: number;
   600→}
   601→
   602→export async function syncPlanToStripeMulti(
   603→  stripe: Stripe,
   604→  input: SyncMultiInput,
   605→): Promise<SyncMultiResult> {
   606→  // 1. Ensure the Stripe Product exists (one per planId).
   607→  let productId: string | undefined;
   608→  try {
   609→    const existingProducts = await stripe.products.list({ limit: 100 });
   610→    const existingProduct = existingProducts.data.find(
   611→      (p) => p.metadata?.planId === input.planId,
   612→    );
   613→    if (existingProduct) productId = existingProduct.id;
   614→  } catch {
   615→    // fall through to create
   616→  }
   617→  if (!productId) {
   618→    const product = await stripe.products.create({
   619→      name: input.name,
   620→      metadata: { planId: input.planId },
   621→    });
   622→    productId = product.id;
   623→  }
   624→
   625→  // 2. For each (currency, interval) pair, resolve or create the
   626→  //    appropriate Stripe Price. Currencies with all-zero prices are
   627→  //    skipped (the existing IDs are preserved as-is, no new Prices
   628→  //    created).
   629→  const result: SyncMultiResult['stripePriceIdsByCurrency'] = {};
   630→  let created = 0;
   631→  const currencies = Object.keys(input.pricesByCurrency);
   632→  // Include currencies that had prior Stripe Price IDs even if no
   633→  // current price config — so we preserve the IDs (legacy compat).
   634→  const existingMap = input.stripePriceIdsByCurrency ?? {};
   635→  const allCurrencies = Array.from(new Set([...currencies, ...Object.keys(existingMap)]));
   636→
   637→  for (const rawCur of allCurrencies) {
   638→    const curUpper = rawCur.toUpperCase();
   639→    const price = input.pricesByCurrency[curUpper] ?? { monthly: 0, yearly: 0 };
   640→    const existing = existingMap[curUpper] ?? { monthly: null, yearly: null };
   641→    const out: { monthly: string | null; yearly: string | null } = {
   642→      monthly: existing.monthly,
   643→      yearly: existing.yearly,
   644→    };
   645→    for (const interval of ['monthly', 'yearly'] as const) {
   646→      const amount = interval === 'monthly' ? price.monthly : price.yearly;
   647→      const expectedAmount = amount * 100;
   648→      // ZERO PRICE: skip creating a new Price. Keep the existing ID
   649→      // if any (legacy). Do NOT archive — existing subs may still
   650→      // reference it.
   651→      if (amount <= 0) {
   652→        continue;
   653→      }
   654→      const existingId = interval === 'monthly' ? existing.monthly : existing.yearly;
   655→      let resolvedId: string | null = null;
   656→      // (a) If an existing ID is provided, verify + price-change check.
   657→      if (existingId && existingId.trim().length > 0) {
   658→        try {
   659→          const existingPrice = await stripe.prices.retrieve(existingId);
   660→          const existingAmount = existingPrice.unit_amount ?? 0;
   661→          if (existingAmount === expectedAmount) {
   662→            resolvedId = existingId;
   663→          } else {
   664→            // Price changed → create a NEW Stripe Price on the same
   665→            // Product with the updated amount.
   666→            const stripeInterval: 'month' | 'year' = interval === 'monthly' ? 'month' : 'year';
   667→            const newPrice = await stripe.prices.create({
   668→              unit_amount: expectedAmount,
   669→              currency: curUpper.toLowerCase(),
   670→              recurring: { interval: stripeInterval },
   671→              product: productId,
   672→              metadata: { planId: input.planId, interval, currency: curUpper },
   673→            });
   674→            resolvedId = newPrice.id;
   675→            created++;
   676→          }
   677→        } catch {
   678→          // Retrieval failed (deleted? permission?) — fall through
   679→          // to lookup-by-metadata + create branch.
   680→        }
   681→      }
   682→      // (b) If no resolved ID yet, try to find an existing active
   683→      //     Stripe Price with matching metadata (currency + interval).
   684→      if (!resolvedId) {
   685→        try {
   686→          const list = await stripe.prices.list({ active: true, limit: 100 });
   687→          const match = list.data.find(
   688→            (p) =>
   689→              p.metadata?.planId === input.planId &&
   690→              p.metadata?.interval === interval &&
   691→              p.metadata?.currency === curUpper,
   692→          );
   693→          if (match) {
   694→            const matchAmount = match.unit_amount ?? 0;
   695→            if (matchAmount === expectedAmount) {
   696→              resolvedId = match.id;
   697→            } else {
   698→              // Amount mismatch → create new on the same Product.
   699→              const stripeInterval: 'month' | 'year' = interval === 'monthly' ? 'month' : 'year';
   700→              const newPrice = await stripe.prices.create({
   701→                unit_amount: expectedAmount,
   702→                currency: curUpper.toLowerCase(),
   703→                recurring: { interval: stripeInterval },
   704→                product: productId,
   705→                metadata: { planId: input.planId, interval, currency: curUpper },
   706→              });
   707→              resolvedId = newPrice.id;
   708→              created++;
   709→            }
   710→          }
   711→        } catch {
   712→          // fall through to create branch
   713→        }
   714→      }
   715→      // (c) If still no resolved ID, create a fresh Stripe Price.
   716→      if (!resolvedId) {
   717→        const stripeInterval: 'month' | 'year' = interval === 'monthly' ? 'month' : 'year';
   718→        const newPrice = await stripe.prices.create({
   719→          unit_amount: expectedAmount,
   720→          currency: curUpper.toLowerCase(),
   721→          recurring: { interval: stripeInterval },
   722→          product: productId,
   723→          metadata: { planId: input.planId, interval, currency: curUpper },
   724→        });
   725→        resolvedId = newPrice.id;
   726→        created++;
   727→      }
   728→      if (interval === 'monthly') out.monthly = resolvedId;
   729→      else out.yearly = resolvedId;
   730→    }
   731→    result[curUpper] = out;
   732→  }
   733→  return { stripePriceIdsByCurrency: result, created };
   734→}

   857→/**
   858→ * IDEMPOTENCY LEDGER — claim-for-processing pattern.
   859→ *
   860→ * Stripe retries webhook events up to 3 days. We MUST NOT process the
   861→ * same `evt_...` twice. The flow is:
   862→ *
   863→ *   1. claimWebhookEventForProcessing({ stripeEventId, eventType, ... })
   864→ *      Tries to insert a row with outcome='processing'. Three outcomes:
   865→ *        - No existing row → insert succeeds → returns 'new' (process now).
   866→ *        - Existing row with outcome='processed' → returns 'processed' (skip).
   867→ *        - Existing row with outcome='error' OR stale 'processing'
   868→ *          (older than 10 min — likely a crashed worker) → re-claim it
   869→ *          (update to 'processing') → returns 'retry' (process again).
   870→ *
   871→ *   2. After the handler dispatch:
   872→ *      markWebhookEventOutcome(stripeEventId, 'processed')  // success
   873→ *      markWebhookEventOutcome(stripeEventId, 'error', '...') // failure
   874→ *
   875→ * This pattern (vs the old "insert with outcome='processed' BEFORE
   876→ * dispatch") lets Stripe retries recover from handler failures —
   877→ * previously, a row was marked 'processed' before the handler ran,
   878→ * so a crash left the row 'processed' and the retry was silently
   879→ * dropped, leaving the DB in a partial state.
   880→ *
   881→ * Lazy-imports `@/lib/db` so this file stays safe to import from
   882→ * server-only contexts without forcing Prisma to load eagerly.
   883→ */
   884→const STALE_PROCESSING_MS = 10 * 60_000; // 10 min — a worker that crashed mid-process
   885→
   886→export type WebhookEventClaim = 'new' | 'retry' | 'processed';
   887→
   888→export async function claimWebhookEventForProcessing(params: {
   889→  stripeEventId: string;
   890→  eventType: string;
   891→  apiVersion?: string;
   892→  objectId?: string;
   893→}): Promise<WebhookEventClaim> {
   894→  const { db } = await import('@/lib/db');
   895→  try {
   896→    // Try to insert a fresh 'processing' row. If the unique constraint
   897→    // on stripeEventId rejects the insert, fall through to the existing-row
   898→    // branch.
   899→    await db.webhookEvent.create({
   900→      data: {
   901→        stripeEventId: params.stripeEventId,
   902→        eventType: params.eventType,
   903→        apiVersion: params.apiVersion ?? null,
   904→        objectId: params.objectId ?? null,
   905→        outcome: 'processing',
   906→      },
   907→    });
   908→    return 'new';
   909→  } catch (err: unknown) {
   910→    const e = (err ?? {}) as { code?: string; message?: string };
   911→    if (e.code !== 'P2002' && !e.message?.includes('Unique constraint')) {
   912→      // Any other error → conservative: treat as 'processed' so the
   913→      // webhook skips (request still returns 200 to Stripe).
   914→      return 'processed';
   915→    }
   916→    // Unique constraint violation — a row already exists. Look it up.
   917→    try {
   918→      const existing = await db.webhookEvent.findUnique({
   919→        where: { stripeEventId: params.stripeEventId },
   920→        select: { outcome: true, processedAt: true },
   921→      });
   922→      if (!existing) return 'processed'; // race: row vanished → skip
   923→      if (existing.outcome === 'processed') return 'processed'; // already done → skip
   924→      if (existing.outcome === 'error') {
   925→        // Re-claim → re-process on retry.
   926→        await db.webhookEvent.update({
   927→          where: { stripeEventId: params.stripeEventId },
   928→          data: {
   929→            outcome: 'processing',
   930→            errorMessage: null,
   931→            processedAt: new Date(),
   932→          },
   933→        });
   934→        return 'retry';
   935→      }
   936→      if (existing.outcome === 'processing') {
   937→        // Stale in-progress row (worker crashed mid-process). Re-claim
   938→        // only when stale; otherwise skip to avoid double-processing.
   939→        const age = Date.now() - existing.processedAt.getTime();
   940→        if (age > STALE_PROCESSING_MS) {
   941→          await db.webhookEvent.update({
   942→            where: { stripeEventId: params.stripeEventId },
   943→            data: {
   944→              outcome: 'processing',
   945→              errorMessage: null,
   946→              processedAt: new Date(),
   947→            },
   948→          });
   949→          return 'retry';
   950→        }
   951→        return 'processed'; // still being processed by another worker → skip
   952→      }
   953→      return 'processed';
   954→    } catch {
   955→      // Lookup failed → conservative: skip.
   956→      return 'processed';
   957→    }
   958→  }
   959→}
   960→
   961→/** Update the outcome of a webhook event row after the handler runs.
   962→ *  Best-effort — never throws. Called with 'processed' on success or
   963→ *  'error' on failure (Stripe will retry). */
   964→export async function markWebhookEventOutcome(
   965→  stripeEventId: string,
   966→  outcome: 'processed' | 'error',
   967→  errorMessage?: string | null,
   968→): Promise<void> {
   969→  const { db } = await import('@/lib/db');
   970→  try {
   971→    await db.webhookEvent.update({
   972→      where: { stripeEventId },
   973→      data: {
   974→        outcome,
   975→        errorMessage: errorMessage ? errorMessage.slice(0, 1000) : null,
   976→        processedAt: new Date(),
   977→      },
   978→    });
   979→    if (outcome === 'processed' || outcome === 'error') {
   980→      // Invalidate the resolved-settings cache when a webhook succeeds
   981→      // so subsequent reads pick up any DB changes (e.g., a Payment
   982→      // row created by the webhook). Best-effort.
   983→      invalidateStripeSettingsCache();
   984→    }
   985→  } catch {
   986→    // best-effort — never throw from an outcome update
   987→  }
   988→}
   989→
   990→/**
   991→ * @deprecated Use `claimWebhookEventForProcessing` + `markWebhookEventOutcome`
   992→ *   instead. The new 2-step pattern allows Stripe retries to recover from
   993→ *   handler failures (the old single-step insert marked the row 'processed'
   994→ *   BEFORE dispatch, so retries were silently dropped).
   995→ *
   996→ *   Kept as a backwards-compatible wrapper so callers that haven't migrated
   997→ *   keep working. Calls `claimWebhookEventForProcessing`; returns true only
   998→ *   when the claim is 'new' or 'retry' (i.e. the caller should process the
   999→ *   event). Returns false when the event was already 'processed'.
  1000→ */
  1001→export async function markWebhookEventProcessed(params: {
  1002→  stripeEventId: string;
  1003→  eventType: string;
  1004→  apiVersion?: string;
  1005→  objectId?: string;
  1006→  outcome: 'processed' | 'skipped_duplicate' | 'error';
  1007→  errorMessage?: string;
  1008→}): Promise<boolean> {
  1009→  const claim = await claimWebhookEventForProcessing({
  1010→    stripeEventId: params.stripeEventId,
  1011→    eventType: params.eventType,
  1012→    apiVersion: params.apiVersion,
  1013→    objectId: params.objectId,
  1014→  });
  1015→  if (claim === 'processed') return false;
  1016→  // 'new' or 'retry' — caller will process. If the caller also passed an
  1017→  // outcome, apply it now (for backwards compat with callers that used
  1018→  // the old single-step API to record an outcome at the start).
  1019→  if (params.outcome && params.outcome !== 'processed') {
  1020→    await markWebhookEventOutcome(params.stripeEventId, params.outcome === 'error' ? 'error' : 'processed', params.errorMessage);
  1021→  }
  1022→  return true;
  1023→}

  1091→/** Decrypt a stored encrypted secret (returns null when ciphertext
  1092→ *  is null or decryption fails). */
  1093→async function tryDecrypt(ciphertext: string | null | undefined): Promise<string | null> {
  1094→  if (!ciphertext) return null;
  1095→  try {
  1096→    return await decrypt(ciphertext);
  1097→  } catch {
  1098→    return null;
  1099→  }

  1116→  return {

  1126→      ? row.lastTestStatus

  1141→/** Save (encrypt + upsert) Stripe credentials. Empty-string values
  1142→ *  for secret keys mean "leave unchanged" (the existing encrypted
  1143→ *  value is preserved); null means "clear". Omitted (undefined)
  1144→ *  fields also mean "preserve" — this matters because the frontend
  1145→ *  may omit a field entirely from the JSON body when the admin didn't
  1146→ *  touch it. Publishable keys are written as plaintext (non-secret).
  1147→ *  Returns the masked view. Never returns raw secret values.
  1148→ *
  1149→ *  MASKED-VALUE REJECTION: any non-empty secret/publishable/webhook
  1150→ *  value containing the mask bullet character (•) is rejected with a
  1151→ *  thrown Error. This catches the common mistake of copy-pasting the
  1152→ *  masked display form (e.g. `sk_test_••••abcd`) back into the input
  1153→ *  — which would otherwise be encrypted as the masked string and
  1154→ *  silently break every subsequent Stripe API call with
  1155→ *  "Invalid API Key provided". */
  1156→export async function saveStripeSettings(
  1157→  input: StripeSettingsInput,
  1158→  editorUserId: string,
  1159→): Promise<StripeSettingsView> {
  1160→  const { db } = await import('@/lib/db');
  1161→  const existing = await readStripeSettingsRow();
  1162→
  1163→  // ---- Masked-value rejection (before any encryption / storage). ----
  1164→  // The admin may have copy-pasted the masked form back into the
  1165→  // input — reject it with a clear, actionable error.
  1166→  const maskedChecks: Array<{ value: string | undefined; label: string }> = [
  1167→    { value: input.secretKeyTest, label: 'Secret Key (Test)' },
  1168→    { value: input.secretKeyLive, label: 'Secret Key (Live)' },
  1169→    { value: input.publishableKeyTest, label: 'Publishable Key (Test)' },
  1170→    { value: input.publishableKeyLive, label: 'Publishable Key (Live)' },
  1171→    { value: input.webhookSecretTest, label: 'Webhook Secret (Test)' },
  1172→    { value: input.webhookSecretLive, label: 'Webhook Secret (Live)' },
  1173→  ];
  1174→  for (const c of maskedChecks) {
  1175→    if (c.value && isMaskedSecretValue(c.value)) {
  1176→      throw new Error(
  1177→        `${c.label} looks like a masked value (contains the • character). ` +
  1178→          `Re-enter the full, real Stripe key — do not copy the masked form.`,
  1179→      );
  1180→    }
  1181→  }
  1182→
  1183→  // ---- Secret keys: encrypt when provided; preserve when empty ----
  1184→  // Semantics (NEW, fixes the "Invalid API Key" bug):
  1185→  //   - undefined  → preserve (admin omitted the field entirely)
  1186→  //   - null       → clear (admin explicitly wants to remove it)
  1187→  //   - ''         → preserve (admin left the input empty — "leave unchanged")
  1188→  //   - non-empty  → encrypt + store
  1189→  // Previously, undefined was treated as "clear", which silently wiped
  1190→  // the saved secret every time the admin saved without retyping the
  1191→  // key. That made the next Test Connection fall back to env (often
  1192→  // empty or wrong) → Stripe returned "Invalid API Key".
  1193→  const encryptOrPreserve = async (
  1194→    plaintext: string | undefined | null,
  1195→    existingEncrypted: string | null | undefined,
  1196→  ): Promise<string | null> => {
  1197→    if (plaintext === undefined) return existingEncrypted ?? null; // preserve (omitted)
  1198→    if (plaintext === null) return null; // clear
  1199→    if (plaintext === '') return existingEncrypted ?? null; // preserve (empty input)
  1200→    return await encrypt(plaintext);
  1201→  };
  1202→
  1203→  const secretKeyTestEncrypted = await encryptOrPreserve(
  1204→    input.secretKeyTest,
  1205→    existing?.secretKeyTestEncrypted,
  1206→  );
  1207→  const secretKeyLiveEncrypted = await encryptOrPreserve(
  1208→    input.secretKeyLive,
  1209→    existing?.secretKeyLiveEncrypted,
  1210→  );
  1211→  const webhookSecretTestEncrypted = await encryptOrPreserve(
  1212→    input.webhookSecretTest,
  1213→    existing?.webhookSecretTestEncrypted,
  1214→  );
  1215→  const webhookSecretLiveEncrypted = await encryptOrPreserve(
  1216→    input.webhookSecretLive,
  1217→    existing?.webhookSecretLiveEncrypted,
  1218→  );
  1219→
  1220→  // ---- Publishable keys: plaintext (non-secret). Empty string
  1221→  //      preserves existing; null clears; undefined (omitted) preserves. ----
  1222→  const resolvePub = (
  1223→    v: string | undefined | null,
  1224→    existingVal: string | null,
  1225→  ): string | null => {
  1226→    if (v === undefined) return existingVal ?? null; // preserve (omitted)
  1227→    if (v === null) return null; // clear
  1228→    if (v === '') return existingVal ?? null; // preserve (empty input)
  1229→    return v;
  1230→  };
  1231→  const publishableKeyTest = resolvePub(input.publishableKeyTest, existing?.publishableKeyTest ?? null);
  1232→  const publishableKeyLive = resolvePub(input.publishableKeyLive, existing?.publishableKeyLive ?? null);
  1233→
  1234→  // ---- App URL: empty string = unset (no preserve semantic — caller
  1235→  //      should pass the full URL). ----
  1236→  const appUrl = input.appUrl ?? existing?.appUrl ?? null;
  1237→
  1238→  await db.stripeSettings.upsert({
  1239→    where: { id: 'singleton' },
  1240→    create: {
  1241→      id: 'singleton',
  1242→      mode: input.mode,
  1243→      secretKeyTestEncrypted,
  1244→      secretKeyLiveEncrypted,
  1245→      webhookSecretTestEncrypted,
  1246→      webhookSecretLiveEncrypted,
  1247→      publishableKeyTest,
  1248→      publishableKeyLive,
  1249→      appUrl,
  1250→      updatedBy: editorUserId,
  1251→    },
  1252→    update: {
  1253→      mode: input.mode,
  1254→      secretKeyTestEncrypted,
  1255→      secretKeyLiveEncrypted,
  1256→      webhookSecretTestEncrypted,
  1257→      webhookSecretLiveEncrypted,
  1258→      publishableKeyTest,
  1259→      publishableKeyLive,
  1260→      appUrl,
  1261→      updatedBy: editorUserId,
  1262→    },
  1263→  });
  1264→
  1265→  // Force the next read to pick up the new credentials.
  1266→  invalidateStripeSettingsCache();
  1267→
  1268→  return getStripeSettingsForAdmin();
  1269→}
  1270→
  1271→/** Test a Stripe connection by pinging the read-only /v1/balance
  1272→ *  endpoint. Uses the currently-configured credentials when
  1273→ *  `input` is null/undefined; otherwise constructs a temporary
  1274→ *  client from `input` (so the admin can test BEFORE saving).
  1275→ *  Records the outcome on the singleton row.
  1276→ *
  1277→ *  Returns `{ success: true, accountInfo }` on success, or
  1278→ *  `{ success: false, code, message }` on failure. Never throws. */
  1279→export async function testStripeConnection(
  1280→  input?: StripeSettingsInput,
  1281→): Promise<
  1282→  | { success: true; mode: 'test' | 'live'; accountInfo: { id: string; type: string; country: string; email: string | null; displayName: string | null } }
  1283→  | { success: false; mode: 'test' | 'live'; code: string; message: string }
  1284→> {
  1285→  // Resolve the secret key to test with.
  1286→  let secretKey: string | null = null;
  1287→  let mode: 'test' | 'live' = 'test';
  1288→
  1289→  if (input) {
  1290→    mode = input.mode;
  1291→    const candidate = mode === 'live' ? input.secretKeyLive : input.secretKeyTest;
  1292→    if (candidate && candidate.trim().length > 0) {
  1293→      secretKey = candidate;
  1294→    } else {
  1295→      // Fall back to the stored value for the chosen mode.
  1296→      const row = await readStripeSettingsRow();
  1297→      const enc = mode === 'live' ? row?.secretKeyLiveEncrypted : row?.secretKeyTestEncrypted;
  1298→      secretKey = await tryDecrypt(enc);
  1299→    }
  1300→  } else {
  1301→    const resolved = await getResolvedStripeSettings();
  1302→    secretKey = resolved.secretKey;
  1303→    mode = resolved.mode;
  1304→  }
  1305→
  1306→  if (!secretKey || secretKey.trim().length === 0) {
  1307→    const msg = 'No Stripe secret key is configured for the selected mode. Set the secret key first.';
  1308→    await recordTestOutcome(mode, 'error', msg);
  1309→    return { success: false, mode, code: 'NOT_CONFIGURED', message: msg };
  1310→  }
  1311→
  1312→  // Reject masked values outright — the admin copy-pasted the masked
  1313→  // display form (sk_test_••••abcd) instead of the real key. Stripe
  1314→  // would reject the masked string as an "Invalid API Key" — better
  1315→  // to surface the real cause here with a clear, actionable message.
  1316→  if (isMaskedSecretValue(secretKey)) {
  1317→    const msg =
  1318→      'The provided secret key looks like a masked value (contains the • character). ' +
  1319→      'Re-enter the full, real Stripe secret key — do not copy the masked form.';
  1320→    await recordTestOutcome(mode, 'error', msg);
  1321→    return { success: false, mode, code: 'MASKED_VALUE', message: msg };
  1322→  }
  1323→
  1324→  // Basic key-shape validation — sk_test_ / sk_live_ / rk_test_ /
  1325→  // rk_live_ (restricted keys also work for read-only calls).
  1326→  const keyShapeOk =
  1327→    secretKey.startsWith('sk_test_') ||
  1328→    secretKey.startsWith('sk_live_') ||
  1329→    secretKey.startsWith('rk_test_') ||
  1330→    secretKey.startsWith('rk_live_');
  1331→  if (!keyShapeOk) {
  1332→    const msg = 'The secret key does not look like a Stripe secret key (expected sk_test_… or sk_live_…).';
  1333→    await recordTestOutcome(mode, 'error', msg);
  1334→    return { success: false, mode, code: 'INVALID_KEY_SHAPE', message: msg };
  1335→  }
  1336→
  1337→  // MODE-KEY MISMATCH VALIDATION: verify the key prefix matches the
  1338→  // selected Test/Live mode. Without this, the admin could select
  1339→  // Live mode but provide a sk_test_ key — the Stripe client would
  1340→  // authenticate against the TEST account (since the key is a test
  1341→  // key), silently bypassing the live-account check the admin meant
  1342→  // to perform. Stripe would happily return account info for the
  1343→  // test account, leading the admin to believe the LIVE credentials
  1344→  // were validated when they weren't.
  1345→  const isTestKey = secretKey.startsWith('sk_test_') || secretKey.startsWith('rk_test_');
  1346→  const isLiveKey = secretKey.startsWith('sk_live_') || secretKey.startsWith('rk_live_');
  1347→  if (mode === 'test' && !isTestKey) {
  1348→    const msg =
  1349→      'Mode/key mismatch: Test mode is selected but the secret key is a LIVE key ' +
  1350→      `(starts with ${secretKey.slice(0, 8)}…). Switch to Live mode or paste a Test key (sk_test_…).`;
  1351→    await recordTestOutcome(mode, 'error', msg);
  1352→    return { success: false, mode, code: 'MODE_KEY_MISMATCH', message: msg };
  1353→  }
  1354→  if (mode === 'live' && !isLiveKey) {
  1355→    const msg =
  1356→      'Mode/key mismatch: Live mode is selected but the secret key is a TEST key ' +
  1357→      `(starts with ${secretKey.slice(0, 8)}…). Switch to Test mode or paste a Live key (sk_live_…).`;
  1358→    await recordTestOutcome(mode, 'error', msg);
  1359→    return { success: false, mode, code: 'MODE_KEY_MISMATCH', message: msg };
  1360→  }
  1361→
  1362→  try {
  1363→    // Construct a TEMPORARY client (not the cached singleton) so we
  1364→    // can test credentials that haven't been saved yet.
  1365→    const tempClient = new Stripe(secretKey, {
  1366→      apiVersion: '2026-08-26.dahlia' as Stripe.LatestApiVersion,
  1367→      typescript: true,
  1368→      maxNetworkRetries: 1,
  1369→    });
  1370→    // /v1/balance is read-only and free of side effects — perfect for
  1371→    // testing credentials. It also works with restricted keys that
  1372→    // have the "Balance" read permission.
  1373→    const balance = await tempClient.balance.retrieve();
  1374→    // Resolve the account for richer info (id, country, display name).
  1375→    // This call requires the "Settings" read permission, which secret
  1376→    // keys always have. Fall back gracefully when restricted keys
  1377→    // can't reach it.
  1378→    let accountInfo: { id: string; type: string; country: string; email: string | null; displayName: string | null };
  1379→    try {
  1380→      const acct = await tempClient.accounts.retrieve();
  1381→      accountInfo = {
  1382→        id: acct.id,
  1383→        type: (acct as unknown as { type?: string }).type ?? 'standard',
  1384→        country: (acct as unknown as { country?: string }).country ?? '',
  1385→        email: (acct as unknown as { email?: string | null }).email ?? null,
  1386→        displayName: (acct as unknown as { display_name?: string | null }).display_name ?? null,
  1387→      };
  1388→    } catch {
  1389→      // Restricted key with no Settings read — still a successful
  1390→      // test (the secret key authenticates to Stripe).
  1391→      accountInfo = {
  1392→        id: '(restricted key)',
  1393→        type: 'restricted',
  1394→        country: '',
  1395→        email: null,
  1396→        displayName: null,
  1397→      };
  1398→    }
  1399→    const balanceAvailable = balance.available?.[0]?.amount;
  1400→    await recordTestOutcome(
  1401→      mode,
  1402→      'success',
  1403→      null,
  1404→      `account_id=${accountInfo.id}, available=${balanceAvailable ?? 'n/a'}`,
  1405→    );
  1406→    return { success: true, mode, accountInfo };
  1407→  } catch (err: unknown) {
  1408→    const e = (err ?? {}) as { code?: string; message?: string; type?: string };
  1409→    const code = e.code ?? e.type ?? 'stripe_error';
  1410→    const message = e.message ?? 'Unknown Stripe error';
  1411→    await recordTestOutcome(mode, 'error', `${code}: ${message}`);
  1412→    return { success: false, mode, code, message };
  1413→  }
  1414→}

  1416→/** Persist the last test-connection outcome on the singleton row.
  1417→ *  Best-effort — never throws. */
  1418→async function recordTestOutcome(
  1419→  mode: 'test' | 'live',
  1420→  status: 'success' | 'error',
  1421→  errorMessage: string | null,
  1422→  _detail?: string,
  1423→): Promise<void> {

  1426→    await db.stripeSettings.upsert({
  1427→      where: { id: 'singleton' },
  1428→      create: {
  1429→        id: 'singleton',
  1430→        mode,
  1431→        lastTestStatus: status,
  1432→        lastTestedAt: new Date(),
  1433→        lastTestErrorMessage: errorMessage,
  1434→      },
  1435→      update: {
  1436→        mode,
  1437→        lastTestStatus: status,
  1438→        lastTestedAt: new Date(),
  1439→        lastTestErrorMessage: errorMessage,
  1440→      },
  1441→    });
  1442→    invalidateStripeSettingsCache();
  1443→  } catch {
  1444→    // best-effort
  1445→  }
  1446→}
