   153→   153→export function ProvidersPage() {
   154→   154→  const queryClient = useQueryClient();
   155→   155→  const [page, setPage] = useState(1);
   156→   156→  const [pageSize] = useState(25);
   157→   157→  const [search, setSearch] = useState('');
   158→   158→  const [kindFilter, setKindFilter] = useState<string>('all');
   159→   159→  const [statusFilter, setStatusFilter] = useState<string>('all');
   160→   160→  const [dialogOpen, setDialogOpen] = useState(false);
   161→   161→  const [editingProvider, setEditingProvider] = useState<AiProvider | null>(null);
   162→   162→  const [formData, setFormData] = useState<ProviderFormData>(emptyForm);
   163→   163→  const [showApiKey, setShowApiKey] = useState(false);
   164→   164→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   165→   165→  const [deletingProvider, setDeletingProvider] = useState<AiProvider | null>(null);
   166→   166→
   167→   167→  // Fetch providers
   168→   168→  const { data, isLoading, isError } = useQuery({
   169→   169→    queryKey: queryKeys.aiProviders.list({ page, pageSize, search, kind: kindFilter, connectionStatus: statusFilter }),
   170→   170→    queryFn: () => getApi<PaginatedResponse<AiProvider>>('/api/ai/providers', {
   171→   171→      page,
   172→   172→      pageSize,
   173→   173→      search: search || undefined,
   174→   174→      kind: kindFilter !== 'all' ? kindFilter : undefined,
   175→   175→      connectionStatus: statusFilter !== 'all' ? statusFilter : undefined,
   176→   176→    }),
   177→   177→  });
   178→   178→
   179→   179→  const providers = data?.data ?? [];
   180→   180→  const pagination = data?.pagination;
   181→   181→
   182→   182→  // Create / Update mutation
   183→   183→  const saveMutation = useMutation({
   184→   184→    mutationFn: (body: ProviderFormData) => {
   185→   185→      if (editingProvider) {
   186→   186→        const patchBody: Record<string, unknown> = { ...body };
   187→   187→        if (!patchBody.apiKey) delete patchBody.apiKey;
   188→   188→        return patchApi<AiProvider>(`/api/ai/providers/${editingProvider.id}`, patchBody);
   189→   189→      }
   190→   190→      return postApi<AiProvider>('/api/ai/providers', body);
   191→   191→    },
   192→   192→    onSuccess: () => {
   193→   193→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   194→   194→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   195→   195→      queryClient.invalidateQueries({ queryKey: queryKeys.aiSettings.all });
   196→   196→      queryClient.invalidateQueries({ queryKey: queryKeys.aiPrompts.all });
   197→   197→      toast.success(editingProvider ? 'Provider updated' : 'Provider created');
   198→   198→      handleCloseDialog();
   199→   199→    },
   200→   200→    onError: (err: Error) => {
   201→   201→      toast.error(err.message || 'Failed to save provider');
   202→   202→    },
   203→   203→  });
   204→   204→
   205→   205→  // Delete mutation
   206→   206→  const deleteMutation = useMutation({
   207→   207→    mutationFn: (id: string) => deleteApi(`/api/ai/providers/${id}`),
   208→   208→    onSuccess: () => {
   209→   209→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   210→   210→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   211→   211→      queryClient.invalidateQueries({ queryKey: queryKeys.aiSettings.all });
   212→   212→      queryClient.invalidateQueries({ queryKey: queryKeys.aiPrompts.all });
   213→   213→      toast.success('Provider deleted');
   214→   214→      setDeleteDialogOpen(false);
   215→   215→      setDeletingProvider(null);
   216→   216→    },
   217→   217→    onError: (err: Error) => {
   218→   218→      toast.error(err.message || 'Failed to delete provider');
   219→   219→    },
   220→   220→  });
   221→   221→
   222→   222→  // Test connection mutation
   223→   223→  const testMutation = useMutation({
   224→   224→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/test`),
   225→   225→    onSuccess: (result: unknown) => {
   226→   226→      const res = result as { success?: boolean; latency?: number; message?: string; status?: string };
   227→   227→      if (res.success) {
   228→   228→        toast.success(`Connection successful (${res.latency ?? 0}ms)`);
   229→   229→      } else {
   230→   230→        toast.error(res.message || `Connection ${res.status || 'failed'}`);
   231→   231→      }
   232→   232→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   233→   233→    },
   234→   234→    onError: (err: Error) => {
   235→   235→      toast.error(err.message || 'Connection test failed');
   236→   236→    },
   237→   237→  });
   238→   238→
   239→   239→  // Sync models mutation
   240→   240→  const syncMutation = useMutation({
   241→   241→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/sync-models`),
   242→   242→    onSuccess: (result: unknown) => {
   243→   243→      const res = result as { syncedCount?: number; count?: number };
   244→   244→      toast.success(`Synced ${res.syncedCount ?? res.count ?? 0} models`);
   245→   245→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   246→   246→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   247→   247→    },
   248→   248→    onError: (err: Error) => {
   249→   249→      toast.error(err.message || 'Failed to sync models');
   250→   250→    },
   251→   251→  });
   252→   252→
   253→   253→  // Set default mutation
   254→   254→  const setDefaultMutation = useMutation({
   255→   255→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/set-default`),
   256→   256→    onSuccess: () => {
   257→   257→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   258→   258→      toast.success('Default provider updated');
   259→   259→    },
   260→   260→    onError: (err: Error) => {
   261→   261→      toast.error(err.message || 'Failed to set default');
   262→   262→    },
   263→   263→  });
   264→   264→
   265→   265→  // Toggle active mutation
   266→   266→  const toggleActiveMutation = useMutation({
   267→   267→    mutationFn: ({ id, isActive }: { id: string; isActive: boolean }) =>
   268→   268→      patchApi(`/api/ai/providers/${id}`, { isActive }),
   269→   269→    onSuccess: () => {
   270→   270→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   271→   271→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   272→   272→      queryClient.invalidateQueries({ queryKey: queryKeys.aiSettings.all });
   273→   273→      queryClient.invalidateQueries({ queryKey: queryKeys.aiPrompts.all });
   274→   274→    },
   275→   275→    onError: (err: Error) => {
   276→   276→      toast.error(err.message || 'Failed to update provider');
   277→   277→    },
   278→   278→  });
   279→   279→
   280→   280→  // Handlers
   281→   281→  const handleOpenCreate = () => {
   282→   282→    setEditingProvider(null);
   283→   283→    setFormData(emptyForm);
   284→   284→    setShowApiKey(false);
   285→   285→    setDialogOpen(true);
   286→   286→  };
   287→   287→
   288→   288→  const handleOpenEdit = (provider: AiProvider) => {
   289→   289→    setEditingProvider(provider);
   290→   290→    setFormData({
   291→   291→      name: provider.name,
   292→   292→      kind: provider.kind,
   293→   293→      baseUrl: provider.baseUrl,
   294→   294→      apiKey: '', // masked in edit
   295→   295→      apiVersion: provider.apiVersion ?? '',
   296→   296→      isActive: provider.isActive,
   297→   297→    });
   298→   298→    setShowApiKey(false);
   299→   299→    setDialogOpen(true);
   300→   300→  };
   301→   301→
   302→   302→  const handleCloseDialog = () => {
   303→   303→    setDialogOpen(false);
   304→   304→    setEditingProvider(null);
   305→   305→    setFormData(emptyForm);
   306→   306→    setShowApiKey(false);
   307→   307→  };
   308→   308→
   309→   309→  const handleSave = () => {
   310→   310→    if (!formData.name.trim()) {
   311→   311→      toast.error('Name is required');
   312→   312→      return;
   313→   313→    }
   314→   314→    if (!editingProvider && !formData.apiKey.trim()) {
   315→   315→      toast.error('API key is required');
   316→   316→      return;
   317→   317→    }
   318→   318→    // CUSTOM providers require a Base URL
   319→   319→    if (formData.kind === 'CUSTOM' && !formData.baseUrl.trim()) {
   320→   320→      toast.error('Base URL is required for Custom providers');
   321→   321→      return;
   322→   322→    }
   323→   323→    // Validate Base URL format for CUSTOM providers
   324→   324→    if (formData.kind === 'CUSTOM' && formData.baseUrl.trim()) {
   325→   325→      try {
   326→   326→        const u = new URL(formData.baseUrl);
   327→   327→        if (!['http:', 'https:'].includes(u.protocol)) {
   328→   328→          toast.error('Base URL must use http or https protocol');
   329→   329→          return;
   330→   330→        }
   331→   331→      } catch {
   332→   332→        toast.error('Base URL is not a valid URL');
   333→   333→        return;
   334→   334→      }
   335→   335→    }
   336→   336→    saveMutation.mutate(formData);
   337→   337→  };
   338→   338→

   343→   343→      ...prev,
   344→   344→      kind: providerKind,
   345→   345→      baseUrl:
   346→   346→        !prev.baseUrl || prev.baseUrl === prevDefaultUrl
   347→   347→          ? PROVIDER_CONFIGS[providerKind].defaultUrl
   348→   348→          : prev.baseUrl,
   349→   349→    }));
   350→   350→  };
   351→   351→
   352→   352→  // KPI calculations
   353→   353→  const totalProviders = pagination?.total ?? 0;
   354→   354→  const connectedCount = providers.filter((p) => p.connectionStatus === 'CONNECTED').length;
   355→   355→  const defaultProvider = providers.find((p) => p.isDefault);
   356→   356→
   357→   357→  return (
   358→   358→    <div className="space-y-6">
   359→   359→      {/* KPI Cards */}
   360→   360→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   361→   361→        <Card>
   362→   362→          <CardContent className="p-4 flex items-center gap-4">
   363→   363→            <div className="p-2 rounded-lg bg-zinc-100">
   364→   364→              <Server className="h-5 w-5 text-zinc-600" />
   365→   365→            </div>
   366→   366→            <div>
   367→   367→              <p className="text-sm text-zinc-500">Total Providers</p>
   368→   368→              <div className="text-2xl font-bold">{isLoading ? <Skeleton className="h-8 w-12 inline-block" /> : totalProviders}</div>
   369→   369→            </div>
   370→   370→          </CardContent>
   371→   371→        </Card>
   372→   372→        <Card>
   373→   373→          <CardContent className="p-4 flex items-center gap-4">
   374→   374→            <div className="p-2 rounded-lg bg-green-100">
   375→   375→              <Wifi className="h-5 w-5 text-green-600" />
   376→   376→            </div>
   377→   377→            <div>
   378→   378→              <p className="text-sm text-zinc-500">Connected</p>
   379→   379→              <div className="text-2xl font-bold">{isLoading ? <Skeleton className="h-8 w-12 inline-block" /> : connectedCount}</div>
   380→   380→            </div>
   381→   381→          </CardContent>
   382→   382→        </Card>
   383→   383→        <Card>
   384→   384→          <CardContent className="p-4 flex items-center gap-4">
   385→   385→            <div className="p-2 rounded-lg bg-amber-100">
   386→   386→              <Star className="h-5 w-5 text-amber-600" />
   387→   387→            </div>
   388→   388→            <div>
   389→   389→              <p className="text-sm text-zinc-500">Default Provider</p>
   390→   390→              <div className="text-lg font-bold truncate max-w-[200px]">{isLoading ? <Skeleton className="h-6 w-32 inline-block" /> : defaultProvider?.name ?? 'None'}</div>
   391→   391→            </div>
   392→   392→          </CardContent>
   393→   393→        </Card>
   394→   394→      </div>
   395→   395→
   396→   396→      {/* Filters & Actions */}
   397→   397→      <Card>
   398→   398→        <CardHeader className="pb-4">
   399→   399→          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   400→   400→            <CardTitle className="text-lg">AI Providers</CardTitle>
   401→   401→            <Button onClick={handleOpenCreate} size="sm">
   402→   402→              <Plus className="h-4 w-4 mr-2" />
   403→   403→              Add Provider
   404→   404→            </Button>
   405→   405→          </div>
   406→   406→          <div className="flex flex-col sm:flex-row gap-3 mt-2">
   407→   407→            <div className="relative flex-1">
   408→   408→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-zinc-400" />
   409→   409→              <Input
   410→   410→                placeholder="Search providers..."
   411→   411→                value={search}
   412→   412→                onChange={(e) => { setSearch(e.target.value); setPage(1); }}
   413→   413→                className="pl-9"
   414→   414→              />
   415→   415→            </div>
   416→   416→            <Select value={kindFilter} onValueChange={(v) => { setKindFilter(v); setPage(1); }}>
   417→   417→              <SelectTrigger className="w-full sm:w-[180px]">
   418→   418→                <SelectValue placeholder="Provider Kind" />
   419→   419→              </SelectTrigger>
   420→   420→              <SelectContent>
   421→   421→                <SelectItem value="all">All Kinds</SelectItem>
   422→   422→                {PROVIDER_KINDS.map((k) => (
   423→   423→                  <SelectItem key={k} value={k}>{PROVIDER_CONFIGS[k].label}</SelectItem>
   424→   424→                ))}
   425→   425→              </SelectContent>
   426→   426→            </Select>
   427→   427→            <Select value={statusFilter} onValueChange={(v) => { setStatusFilter(v); setPage(1); }}>
   428→   428→              <SelectTrigger className="w-full sm:w-[180px]">
   429→   429→                <SelectValue placeholder="Status" />
   430→   430→              </SelectTrigger>
   431→   431→              <SelectContent>
   432→   432→                <SelectItem value="all">All Status</SelectItem>
   433→   433→                <SelectItem value="CONNECTED">Connected</SelectItem>
   434→   434→                <SelectItem value="DISCONNECTED">Disconnected</SelectItem>
   435→   435→                <SelectItem value="ERROR">Error</SelectItem>
   436→   436→              </SelectContent>
   437→   437→            </Select>
   438→   438→          </div>
   439→   439→        </CardHeader>
   440→   440→        <CardContent className="p-0">
   441→   441→          <ScrollArea className="max-h-[500px]">
   442→   442→            <Table>
   443→   443→              <TableHeader>
   444→   444→                <TableRow>
   445→   445→                  <TableHead>Name</TableHead>
   446→   446→                  <TableHead>Kind</TableHead>
