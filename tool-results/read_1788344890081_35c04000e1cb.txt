     3→// ============================================================
     4→// PLATFORM STRIPE SETTINGS — connect the platform's Stripe
     5→// account from the admin UI.
     6→// ============================================================
     7→// Owner-only. Lets the Platform Owner:
     8→//   - Switch between Test and Live mode (separate credentials per mode).
     9→//   - Enter the Stripe Secret Key, Publishable Key, and Webhook
    10→//     Signing Secret for each mode. Secret keys are AES-256-GCM
    11→//     encrypted in the DB and NEVER returned to the frontend in
    12→//     plaintext — only the masked form is shown (sk_...xxxx).
    13→//   - Set the public app URL (for Stripe Checkout success/cancel
    14→//     redirects).
    15→//   - "Test Connection" pings Stripe's /v1/balance + /v1/account
    16→//     endpoints with a temporary client (read-only — no side
    17→//     effects) and records the outcome on the singleton row.
    18→//   - See the Webhook Endpoint URL to copy into the Stripe
    19→//     dashboard so Stripe fires events to this platform.
    20→//   - See the last Test Connection outcome + timestamp.
    21→//
    22→// Backend: /api/platform/admin/stripe/settings (GET+PUT),
    23→//          /api/platform/admin/stripe/test-connection (POST).
    24→//
    25→// This page is FINANCIAL SETTINGS ONLY. Customer / payment /
    26→// subscription / refund management lives on the Customers /
    27→// Payments pages — not duplicated here.
    28→// ============================================================

   112→export function PlatformStripeSettingsModule() {
   113→  const queryClient = useQueryClient();
   114→
   115→  // ---------- Load saved settings ----------
   116→  const { data, isLoading, isError, refetch } = useQuery<StripeSettingsView>({
   117→    queryKey: ['platform', 'stripe-settings'],
   118→    queryFn: () => getApi<StripeSettingsView>('/api/platform/admin/stripe/settings'),
   119→    staleTime: 10_000,
   120→    refetchOnWindowFocus: false,
   121→  });
   122→
   123→  if (isLoading) {
   124→    return (
   125→      <div className="space-y-6">
   126→        <PlatformPageHeader
   127→          title="Stripe Settings"
   128→          subtitle="Connect your Stripe account to enable real subscription billing."
   129→        />
   130→        <Card>
   131→          <CardContent className="p-6 flex items-center justify-center h-64">
   132→            <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   133→          </CardContent>
   134→        </Card>
   135→      </div>
   136→    );
   137→  }
   138→
   139→  if (isError || !data) {
   140→    return (
   141→      <div className="space-y-6">
   142→        <PlatformPageHeader
   143→          title="Stripe Settings"
   144→          subtitle="Connect your Stripe account to enable real subscription billing."
   145→        />
   146→        <Card>
   147→          <CardContent className="p-6">
   148→            <ErrorState message="Could not load Stripe settings. Please retry." onRetry={() => refetch()} />
   149→          </CardContent>
   150→        </Card>
   151→      </div>
   152→    );
   153→  }
   154→
   155→  // Render the form once data is available — `key` ensures the form
   156→  // remounts (re-initializing useState) when the saved settings change
   157→  // after a save / test-connection / external mutation. This avoids the
   158→  // setState-in-effect anti-pattern.
   159→  return (
   160→    <StripeSettingsForm
   161→      key={data.lastTestedAt ?? 'fresh'}
   162→      data={data}
   163→      queryClient={queryClient}
   164→    />
   165→  );
   166→}

   168→// -------------------- Form (mounted when data is available) --------------------
   169→
   170→function StripeSettingsForm({
   171→  data,
   172→  queryClient,
   173→}: {
   174→  data: StripeSettingsView;
   175→  queryClient: ReturnType<typeof useQueryClient>;
   176→}) {
   177→  // ---------- Local form state (initialized ONCE from data on mount) ----------
   178→  const [mode, setMode] = useState<'test' | 'live'>(data.mode);
   179→  const [secretKeyTest, setSecretKeyTest] = useState(''); // never pre-fill secret keys
   180→  const [secretKeyLive, setSecretKeyLive] = useState('');
   181→  const [publishableKeyTest, setPublishableKeyTest] = useState(data.publishableKeyTest || '');
   182→  const [publishableKeyLive, setPublishableKeyLive] = useState(data.publishableKeyLive || '');
   183→  const [webhookSecretTest, setWebhookSecretTest] = useState('');
   184→  const [webhookSecretLive, setWebhookSecretLive] = useState('');
   185→  const [appUrl, setAppUrl] = useState(data.appUrl || '');
   186→  const [showSecretTest, setShowSecretTest] = useState(false);
   187→  const [showSecretLive, setShowSecretLive] = useState(false);
   188→  const [showWebhookTest, setShowWebhookTest] = useState(false);
   189→  const [showWebhookLive, setShowWebhookLive] = useState(false);
   190→  const [copied, setCopied] = useState(false);
   191→
   192→  // ---------- Save mutation ----------
   193→  const saveMutation = useMutation({
   194→    mutationFn: async () => {
   195→      // IMPORTANT: send the actual field values (which may be empty
   196→      // string). The backend distinguishes:
   197→      //   - undefined (field omitted from JSON body) → "preserve" (keep existing)
   198→      //   - null → "clear" (set to null)
   199→      //   - '' (empty string) → "preserve" (matches the form's "leave empty
   200→      //     to keep current" placeholder semantics)
   201→      //   - non-empty string → encrypt + store
   202→      // We used to convert '' → undefined via `|| undefined`, which made
   203→      // the backend "clear" the saved secret every time the admin saved
   204→      // without retyping the key → "Invalid API Key" on the next Test
   205→      // Connection. Sending the explicit empty string fixes that.
   206→      // Publishable keys are plaintext and visible — empty string also
   207→      // means "preserve" (the field is pre-populated with the current
   208→      // value so the admin can see + edit it directly).
   209→      return putApi<StripeSettingsView>('/api/platform/admin/stripe/settings', {
   210→        mode,
   211→        secretKeyTest,
   212→        secretKeyLive,
   213→        publishableKeyTest,
   214→        publishableKeyLive,
   215→        webhookSecretTest,
   216→        webhookSecretLive,
   217→        appUrl,
   218→      });
   219→    },
   220→    onSuccess: () => {
   221→      toast.success('Stripe settings saved.');
   222→      queryClient.invalidateQueries({ queryKey: ['platform', 'stripe-settings'] });
   223→      // Clear the secret-key inputs after a successful save — the admin
   224→      // must re-enter the key to view/modify it (defensive: prevents
   225→      // stale plaintext from lingering in the form).
   226→      setSecretKeyTest('');
   227→      setSecretKeyLive('');
   228→      setWebhookSecretTest('');
   229→      setWebhookSecretLive('');
   230→    },
   231→    onError: (err: unknown) => {
   232→      const e = err as { code?: string; message?: string };
   233→      toast.error(e?.message || 'Unable to save Stripe settings.');
   234→    },
   235→  });
   236→
   237→  // ---------- Test connection mutation ----------
   238→  // Tests the UNSAVED credentials when secret-key fields are populated
   239→  // (admin just typed a new key); falls back to the saved/stored
   240→  // credentials when they're empty. Sends the actual field values (not
   241→  // `|| undefined`) so the backend can distinguish "field omitted"
   242→  // (fall back to stored) from "field cleared" (use no key). An empty
   243→  // string is treated the same as omitted for Test Connection (the
   244→  // backend falls back to the stored value for the selected mode).
   245→  const testMutation = useMutation({
   246→    mutationFn: async () => {
   247→      return postApi<TestResult>('/api/platform/admin/stripe/test-connection', {
   248→        mode,
   249→        secretKeyTest,
   250→        secretKeyLive,
   251→      });
   252→    },
   253→    onSuccess: (res) => {
   254→      if (res.success) {
   255→        toast.success(
   256→          `Stripe ${res.mode.toUpperCase()} connection successful. Account: ${res.accountInfo?.id ?? 'unknown'} (${res.accountInfo?.country ?? '?'})`,
   257→        );
   258→      } else {
   259→        toast.error(`Connection failed (${res.code}): ${res.message}`);
   260→      }
   261→      queryClient.invalidateQueries({ queryKey: ['platform', 'stripe-settings'] });
   262→    },
   263→    onError: (err: unknown) => {
   264→      const e = err as { code?: string; message?: string };
   265→      toast.error(e?.message || 'Unable to test Stripe connection.');
   266→    },
   267→  });
   268→
   269→  // ---------- Copy webhook URL ----------
   270→  const copyWebhookUrl = async () => {
   271→    if (!data.webhookUrlHint) return;
   272→    try {
   273→      await navigator.clipboard.writeText(data.webhookUrlHint);
   274→      setCopied(true);
   275→      setTimeout(() => setCopied(false), 2000);
   276→      toast.success('Webhook URL copied to clipboard.');
   277→    } catch {
   278→      toast.error('Could not copy. Select and copy manually.');
   279→    }
   280→  };
   281→
   282→  // -------------------- Render --------------------
   283→
   284→  return (
   285→    <div className="space-y-6">
   286→      <PlatformPageHeader
   287→        title="Stripe Settings"
   288→        subtitle="Connect your Stripe account to enable real subscription billing."
   289→        actions={
   290→          <Button
   291→            variant="outline"
   292→            size="sm"
   293→            onClick={() => testMutation.mutate()}
   294→            disabled={testMutation.isPending}
   295→          >
   296→            {testMutation.isPending ? (
   297→              <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   298→            ) : (
   299→              <Plug className="h-4 w-4 mr-2" />
   300→            )}
   301→            Test Connection
   302→          </Button>
   303→        }
   304→      />
   305→
   306→      {/* -------------------- Status banner -------------------- */}
   307→      <Card>
   308→        <CardContent className="p-4">
   309→          <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   310→            <div className="flex items-center gap-3">
   311→              {data.isConfigured ? (
   312→                <ShieldCheck className="h-5 w-5 text-emerald-600" />
   313→              ) : (
   314→                <ShieldAlert className="h-5 w-5 text-amber-600" />
   315→              )}
   316→              <div>
   317→                <p className="text-sm font-medium">
   318→                  {data.isConfigured
   319→                    ? `Stripe is connected (${data.mode.toUpperCase()} mode)`
   320→                    : 'Stripe is not connected'}
   321→                </p>
   322→                <p className="text-xs text-muted-foreground">
   323→                  {data.isConfigured
   324→                    ? `Credentials source: ${data.activeSource === 'db' ? 'Admin Settings' : '.env'}`
   325→                    : 'Configure credentials below or set them in .env to enable checkout.'}
   326→                </p>
   327→              </div>
   328→            </div>
   329→            {/* Last test result */}
   330→            <div className="flex items-center gap-3">
   331→              {data.lastTestStatus === 'success' && (
   332→                <Badge className="bg-emerald-50 text-emerald-700 border-emerald-200">
   333→                  <CheckCircle2 className="h-3 w-3 mr-1" />
   334→                  Last test: OK
   335→                </Badge>
   336→              )}
   337→              {data.lastTestStatus === 'error' && (
   338→                <Badge className="bg-rose-50 text-rose-700 border-rose-200">
   339→                  <XCircle className="h-3 w-3 mr-1" />
   340→                  Last test: Failed
   341→                </Badge>
   342→              )}
   343→              {data.lastTestedAt && (
   344→                <span className="text-xs text-muted-foreground">
   345→                  {new Date(data.lastTestedAt).toLocaleString()}
   346→                </span>
   347→              )}
   348→            </div>
   349→          </div>
   350→          {data.lastTestStatus === 'error' && data.lastTestErrorMessage && (
   351→            <div className="mt-3 rounded-md border border-rose-200 bg-rose-50 p-3">
   352→              <p className="text-xs font-mono text-rose-700 break-all">{data.lastTestErrorMessage}</p>
   353→            </div>
   354→          )}
   355→        </CardContent>
   356→      </Card>
   357→
   358→      {/* -------------------- Mode + credentials -------------------- */}
   359→      <Card>
   360→        <CardHeader>
   361→          <CardTitle className="text-base flex items-center gap-2">
   362→            <CreditCard className="h-4 w-4" />
   363→            Account Credentials
   364→          </CardTitle>
   365→          <CardDescription>
   366→            Enter your Stripe keys for each mode. Secret keys are AES-256-GCM encrypted at rest and never
   367→            sent back to the frontend in plaintext. Leave a secret field empty to keep the existing value;
   368→            enter a new value to rotate.
   369→          </CardDescription>
   370→        </CardHeader>
   371→        <CardContent className="space-y-5">
   372→          {/* Mode toggle */}
   373→          <div className="flex items-center justify-between rounded-md border bg-muted/30 px-4 py-3">
   374→            <div>
   375→              <p className="text-sm font-medium">Mode</p>
   376→              <p className="text-xs text-muted-foreground mt-0.5">
   377→                Test mode uses your Stripe sandbox keys (sk_test_…). Live mode uses real production keys
   378→                (sk_live_…). Switch modes any time — credentials are stored separately per mode.
   379→              </p>
   380→            </div>
   381→            <div className="flex items-center gap-3">
   382→              <span className={cn('text-xs font-medium', mode === 'test' ? 'text-amber-700' : 'text-muted-foreground')}>
   383→                TEST
   384→              </span>
   385→              <Switch
   386→                checked={mode === 'live'}
   387→                onCheckedChange={(v) => setMode(v ? 'live' : 'test')}
   388→              />
   389→              <span className={cn('text-xs font-medium', mode === 'live' ? 'text-emerald-700' : 'text-muted-foreground')}>
   390→                LIVE
   391→              </span>
   392→            </div>
   393→          </div>
   394→
   395→          <Separator />
   396→
   397→          {/* Test mode credentials */}
   398→          <div className={cn('space-y-4', mode === 'live' && 'opacity-50 pointer-events-none')}>
   399→            <div className="flex items-center justify-between">
   400→              <h4 className="text-sm font-semibold flex items-center gap-2">
   401→                <span className={cn(
   402→                  'inline-flex items-center rounded-full px-2 py-0.5 text-xs font-medium',
   403→                  'bg-amber-50 text-amber-700 border border-amber-200',
   404→                )}>
   405→                  TEST
   406→                </span>
   407→                Test Mode Credentials
   408→              </h4>
   409→              {data.hasSecretKeyTest && (
   410→                <Badge variant="outline" className="text-emerald-700 border-emerald-300 bg-emerald-50">
   411→                  Configured
   412→                </Badge>
   413→              )}
   414→            </div>
   415→            <SecretKeyInput
   416→              label="Secret Key"
   417→              prefix="sk_test_"
   418→              value={secretKeyTest}
   419→              onChange={setSecretKeyTest}
   420→              show={showSecretTest}
   421→              onToggleShow={() => setShowSecretTest((v) => !v)}
   422→              masked={data.secretKeyTestMasked}
   423→              isSet={data.hasSecretKeyTest}
   424→              placeholder="sk_test_…"
   425→            />
   426→            <div className="space-y-1.5">
   427→              <Label className="text-xs">Publishable Key</Label>
   428→              <Input
   429→                value={publishableKeyTest}
   430→                onChange={(e) => setPublishableKeyTest(e.target.value)}
   431→                className="h-9 font-mono text-xs"
   432→                placeholder="pk_test_…"
   433→              />
   434→              <p className="text-[11px] text-muted-foreground">
   435→                Non-secret — safe to expose to the client for Stripe.js.
   436→              </p>
   437→            </div>
   438→            <SecretKeyInput
   439→              label="Webhook Signing Secret"
   440→              prefix="whsec_"
   441→              value={webhookSecretTest}
   442→              onChange={setWebhookSecretTest}
   443→              show={showWebhookTest}
   444→              onToggleShow={() => setShowWebhookTest((v) => !v)}
   445→              masked={data.webhookSecretTestMasked}
   446→              isSet={data.hasWebhookSecretTest}
   447→              placeholder="whsec_…"
   448→            />
   449→          </div>
   450→
   451→          <Separator />
   452→
   453→          {/* Live mode credentials */}
   454→          <div className={cn('space-y-4', mode === 'test' && 'opacity-50 pointer-events-none')}>
   455→            <div className="flex items-center justify-between">
   456→              <h4 className="text-sm font-semibold flex items-center gap-2">
   457→                <span className={cn(
   458→                  'inline-flex items-center rounded-full px-2 py-0.5 text-xs font-medium',
   459→                  'bg-emerald-50 text-emerald-700 border border-emerald-200',
   460→                )}>
   461→                  LIVE
   462→                </span>
   463→                Live Mode Credentials
   464→              </h4>
   465→              {data.hasSecretKeyLive && (
   466→                <Badge variant="outline" className="text-emerald-700 border-emerald-300 bg-emerald-50">
   467→                  Configured
   468→                </Badge>
   469→              )}
   470→            </div>
   471→            <SecretKeyInput
   472→              label="Secret Key"
   473→              prefix="sk_live_"
   474→              value={secretKeyLive}
   475→              onChange={setSecretKeyLive}
   476→              show={showSecretLive}
   477→              onToggleShow={() => setShowSecretLive((v) => !v)}
   478→              masked={data.secretKeyLiveMasked}
   479→              isSet={data.hasSecretKeyLive}
   480→              placeholder="sk_live_…"
   481→            />
   482→            <div className="space-y-1.5">
   483→              <Label className="text-xs">Publishable Key</Label>
   484→              <Input
   485→                value={publishableKeyLive}
   486→                onChange={(e) => setPublishableKeyLive(e.target.value)}
   487→                className="h-9 font-mono text-xs"
   488→                placeholder="pk_live_…"
   489→              />
   490→              <p className="text-[11px] text-muted-foreground">
   491→                Non-secret — safe to expose to the client for Stripe.js.
   492→              </p>
   493→            </div>
   494→            <SecretKeyInput
   495→              label="Webhook Signing Secret"
   496→              prefix="whsec_"
   497→              value={webhookSecretLive}
   498→              onChange={setWebhookSecretLive}
   499→              show={showWebhookLive}
   500→              onToggleShow={() => setShowWebhookLive((v) => !v)}
   501→              masked={data.webhookSecretLiveMasked}
   502→              isSet={data.hasWebhookSecretLive}
   503→              placeholder="whsec_…"
   504→            />
   505→          </div>
   506→
   507→          <Separator />
   508→
   509→          {/* App URL */}
   510→          <div className="space-y-1.5">

   524→          {/* Save button */}
   525→          <div className="flex justify-end gap-2 pt-2">
   526→            <Button
   527→              onClick={() => saveMutation.mutate()}
   528→              disabled={saveMutation.isPending}
   529→            >
   530→              {saveMutation.isPending ? (
   531→                <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   532→              ) : (
   533→                <Save className="h-4 w-4 mr-2" />

   535→              Save Settings
   536→            </Button>
   537→          </div>
   538→        </CardContent>
   539→      </Card>
   540→
   541→      {/* -------------------- Webhook endpoint -------------------- */}
   542→      <Card>

   551→          </CardDescription>

   553→        <CardContent className="space-y-3">
   554→          <div className="flex items-center gap-2">

   558→            <Button variant="outline" size="icon" onClick={copyWebhookUrl} title="Copy">
   559→              {copied ? <Check className="h-4 w-4 text-emerald-600" /> : <Copy className="h-4 w-4" />}
   560→            </Button>
   561→          </div>
