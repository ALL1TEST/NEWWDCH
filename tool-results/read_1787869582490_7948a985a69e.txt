   145→// -------------------- Collapsed-Rail Tooltip Positioning -------------------
   146→/*
   147→ * SINGLE source of truth for the positioning of every tooltip/label that
   148→ * appears when the sidebar is collapsed into the 48px icon rail.
   149→ *
   150→ *   side="right"         → opens to the RIGHT of the 48px collapsed rail,
   151→ *                          fully inside the main viewport (never overlaps
   152→ *                          the rail, never opens to the left/outside).
   153→ *   align="center"       → vertically centers the bubble on the icon's
   154→ *                          32×32 hover target so labels line up across
   155→ *                          every row of the rail (logo, theme, nav items,
   156→ *                          parent items with submenu).
   157→ *   sideOffset=8         → ~8px visible gap from the trigger button's right
   158→ *                          edge (button right edge ~x=40 inside the 48px
   159→ *                          rail, tooltip left edge ~x=48 = the rail's
   160→ *                          right edge — flush with the RAIL, NOT with the
   161→ *                          icon, so the bubble never touches the glyph).
   162→ *   collisionPadding=12 → 12px viewport-edge collision padding so the
   163→ *                          bubble can never be clipped at any viewport
   164→ *                          edge or scroll past the visible area.
   165→ *
   166→ * Portal-based rendering (Radix TooltipPrimitive.Portal → document.body,
   167→ * wired inside src/components/ui/tooltip.tsx) means every tooltip is
   168→ * rendered OUTSIDE the sidebar DOM tree — so the sidebar's
   169→ * overflow-hidden / overflow-x-hidden on the rail container can NEVER
   170→ * clip or hide the bubble. The tooltip floats above every other
   171→ * element at z-50.
   172→ *
   173→ * Used by:
   174→ *   • CollapsedLogoButton         (header "Expand" tooltip)
   175→ *   • SimpleNavItem              (leaf nav items: Dashboard, Articles,
   176→ *                                  Calendar, Media, Users, Comments,
   177→ *                                  Newsletter, SEO, AI, Automation)
   178→ *   • ExpandableNavItem          (parent nav items in EXPANDED state —
   179→ *                                  tooltip is hidden by SidebarMenuButton
   180→ *                                  when state!=="collapsed", but the
   181→ *                                  positioning object is still applied
   182→ *                                  so the prop stays consistent across
   183→ *                                  every nav item renderer)
   184→ *   • CollapsedParentNavItem     (parent nav items in COLLAPSED state —
   185→ *                                  label tooltip shows when the floating
   186→ *                                  popover is CLOSED; when the popover is
   187→ *                                  open, no tooltip shows to avoid visual
   188→ *                                  conflict)
   189→ *
   190→ * The ThemeToggle component (src/components/layout/theme-toggle.tsx) lives
   191→ * in a separate file and inlines the SAME four values — see the comment
   192→ * there for the rationale.
   193→ */

   287→ * Derive the top-level section ID from the current browser hash.
   288→ * Uses EXPLICIT prefix matching — #settings/seo always returns 'Settings', never 'SEO'.

   290→function getSectionFromHash(hash: string): string | null {
   291→  const path = hash.replace(/^#\/?,?/, '').trim();
   292→  if (!path || path === 'dashboard') return null;
   293→
   294→  const firstSegment = path.split('/')[0];
   295→  return ROUTE_PREFIX_TO_SECTION[firstSegment] ?? null;
   296→}

   347→/**
   348→ * Collapsed-rail logo cell. Since the extra collapse-toggle icon was removed
   349→ * from below the logo, THIS cell is the explicit expand control:
   350→ *   • identical 32×32 grid geometry as every other rail icon
   351→ *   • click toggles the sidebar open (SidebarRail edge strip still works)
   352→ * No new icon is introduced and no empty clickable ghost remains.
   353→ *
   354→ * HOVER BEHAVIOR (collapsed state only — this button is rendered ONLY
   355→ * inside the collapsed-rail cluster, so the expanded LogoMark is untouched):
   356→ * The `hovered` prop is LIFTED to the rail level (see AppSidebar's
   357→ * `railHovered` state + the onMouseEnter/onMouseLeave handlers on the
   358→ * <Sidebar> root). This means the swap fires when the mouse is ANYWHERE
   359→ * over the collapsed 48px rail — NOT only when the pointer is directly
   360→ * over the logo cell:
   361→ *   • rail at rest ........  shows the "C" logo mark on its bg-primary
   362→ *                            black box (NORMAL LOGO — background UNCHANGED,
   363→ *                            same as the expanded LogoMark).
   364→ *   • mouse over ANY rail   the "C" is replaced by a PanelLeftOpen icon
   365→ *     item (Dashboard,       AND the button's background becomes
   366→ *     Articles, …, Profile,  transparent so ONLY the Expand icon is
   367→ *     Theme, Settings, …)    visible with no box behind it. The icon
   368→ *                            color switches to muted-foreground (gray)
   369→ *                            so it stays visible against the
   370→ *                            transparent/page bg in both Light and Dark.
   371→ *   • mouse leaves the     restores the "C" logo mark AND its bg-primary
   372→ *     entire rail            box (onMouseLeave on the <Sidebar> root).
   373→ *   • click ............. calls toggleSidebar (existing functionality,
   374→ *                         unchanged).
   375→ *
   376→ * Each rail item keeps its OWN Radix Tooltip label (Dashboard, Articles,
   377→ * Profile, etc.) — the lifted `hovered` prop ONLY drives the logo cell's
   378→ * C↔PanelLeftOpen swap; it does NOT touch any item's tooltip. The logo's
   379→ * own "Expand" Tooltip still fires ONLY when the pointer is directly over
   380→ * the logo button (the Tooltip trigger is the button itself, independent
   381→ * of the `hovered` prop).
   382→ *
   383→ * Background is removed ONLY from the temporary Expand-icon state; the
   384→ * normal logo (at rest) keeps its bg-primary background exactly as-is.
   385→ *
   386→ * Implementation: React state lifted to AppSidebar (railHovered +
   387→ * onMouseEnter/onMouseLeave on <Sidebar>) — NOT CSS :hover / group-hover
   388→ * variants. This is deliberate: Tailwind v4 wraps its hover:/group-hover:
   389→ * utilities inside @media (hover: hover), so on any browser that reports
   390→ * (hover: none) (headless, touch, some preview iframes) those utilities
   391→ * NEVER activate and the swap + bg-transparent would silently fail. React
   392→ * mouse events fire on ANY pointer input regardless of the hover media
   393→ * query, so the behavior is identical in every environment. The
   394→ * conditional className + conditional render swap the bg/color/glyph
   395→ * atomically per state. Works in both Light and Dark.
   396→ */
   397→function CollapsedLogoButton({ hovered }: { hovered: boolean }) {
   398→  const { toggleSidebar } = useSidebar();
   399→  return (
   400→    /* Hover-only "Expand" tooltip: instant (provider delayDuration=0),
   401→       appears on mouse-enter, disappears on mouse-leave. Radix closes it
   402→       on any pointer exit — disableHoverableContent guarantees it can
   403→       never linger or trap the pointer. The Tooltip trigger is the logo
   404→       button itself, so the "Expand" TEXT fires ONLY on direct logo hover
   405→       (NOT on rail-level hover of other items) — the C↔PanelLeftOpen ICON
   406→       swap, by contrast, is driven by the lifted `hovered` prop and fires
   407→       for ANY rail item. */
   408→    <Tooltip disableHoverableContent>
   409→      <TooltipTrigger asChild>
   410→        <button
   411→          type="button"
   412→          onClick={toggleSidebar}
   413→          aria-label="Expand sidebar"
   414→          className={cn(
   415→            'flex h-8 w-8 shrink-0 cursor-pointer items-center justify-center rounded-lg font-bold text-sm outline-none transition-opacity hover:opacity-90 focus-visible:ring-2 focus-visible:ring-ring select-none',
   416→            // NORMAL LOGO STATE (at rest): bg-primary black box +
   417→            // text-primary-foreground white "C" — background UNCHANGED,
   418→            // same as the expanded LogoMark.
   419→            // EXPAND-ICON STATE (on hover): bg-transparent (NO background)
   420→            // + text-muted-foreground gray icon — ONLY this temporary
   421→            // state loses the background. The icon stays visible (gray
   422→            // on transparent/page bg) in both Light and Dark mode.
   423→            hovered
   424→              ? 'bg-transparent text-muted-foreground'
   425→              : 'bg-primary text-primary-foreground',
   426→          )}
   427→        >
   428→          {/* Conditional render: "C" logo at rest, PanelLeftOpen icon on
   429→              hover. h-4 w-4 matches the CollapseToggle's PanelLeftClose
   430→              icon size; the icon is the visual opposite of the collapse
   431→              icon so the collapsed logo reads as a clear "expand"
   432→              affordance when the pointer is over it. */}
   433→          {hovered ? (
   434→            <PanelLeftOpen className="h-4 w-4" />
   435→          ) : (
   436→            <span>C</span>
   437→          )}
   438→          <span className="sr-only">Expand sidebar</span>
   439→        </button>
   440→      </TooltipTrigger>
   441→      {/* Collapsed-rail tooltip — uses the SHARED COLLAPSED_TOOLTIP_PROPS
   442→          constant so the "Expand" label has IDENTICAL positioning (side,
   443→          align, sideOffset, collisionPadding) to every other collapsed-
   444→          rail tooltip (ThemeToggle, SimpleNavItem, ExpandableNavItem,
   445→          CollapsedParentNavItem). Portal-based rendering means the bubble
   446→          floats outside the sidebar DOM at z-50 — never clipped by the
   447→          rail's overflow-hidden. */}
   448→      <TooltipContent {...COLLAPSED_TOOLTIP_PROPS}>Expand</TooltipContent>
   449→    </Tooltip>
   450→  );
   451→}

   521→/**
   522→ * Settings parent item rendered on the COLLAPSED 48px rail.
   523→ * Clicking it opens a portal-based popover to the RIGHT of the icon —
   524→ * the sidebar itself NEVER expands, and the popover anchors to this
   525→ * exact button so the "star/sparkle alignment" grid stays intact.
   526→ */
   527→function CollapsedParentNavItem({
   528→  item,
   529→  currentModule,
   530→  currentSubPage,
   531→}: {
   532→  item: NavItem;
   533→  currentModule: string;
   534→  currentSubPage: string | null;
   535→}) {
   536→  const [floatOpen, setFloatOpen] = useState(false);
   537→  const mod = hrefToModule(item.href);
   538→  const isActive = currentModule === mod;
   539→
   540→  const handleChildNavigate = (
   541→    e: React.MouseEvent<HTMLAnchorElement>,
   542→    child: NavItem,
   543→  ) => {
   544→    e.preventDefault();
   545→    e.stopPropagation();
   546→    setFloatOpen(false); // close the floating submenu after navigation…
   547→    // …but do NOT touch sidebar expansion state (rule #7).
   548→    const hash = child.href.replace(/^#/, '');
   549→    const parts = hash.split('/');
   550→    const childSubPage = parts[1] || null;
   551→    useNavigationStore.getState().navigate(parts[0], null, childSubPage);
   552→  };
   553→
   554→  return (
   555→    <SidebarMenuItem>
   556→      <Popover open={floatOpen} onOpenChange={setFloatOpen}>
   557→        <PopoverTrigger asChild>
   558→          <SidebarMenuButton
   559→            isActive={isActive || floatOpen}
   560→            aria-expanded={floatOpen}
   561→            aria-haspopup="menu"
   562→            // Collapsed-rail tooltip — uses the SHARED COLLAPSED_TOOLTIP_PROPS
   563→            // so the parent item's label has IDENTICAL positioning (side,
   564→            // align, sideOffset, collisionPadding) to every other collapsed-
   565→            // rail tooltip (CollapsedLogoButton, ThemeToggle, SimpleNavItem,
   566→            // ExpandableNavItem). Suppressed when the popover is open so the
   567→            // floating submenu never visually conflicts with a lingering
   568→            // label bubble. Portal-based rendering means the bubble floats
   569→            // outside the sidebar DOM at z-50 — never clipped by the rail's
   570→            // overflow-hidden.
   571→            tooltip={
   572→              floatOpen
   573→                ? undefined
   574→                : { ...COLLAPSED_TOOLTIP_PROPS, children: item.label }
   575→            }
   576→            onClick={(e: React.MouseEvent) => {
   577→              e.preventDefault();
   578→              setFloatOpen((o) => !o);
   579→            }}
   580→          >
   581→            <NavIcon name={item.icon} />
   582→            <span>{item.label}</span>
   583→          </SidebarMenuButton>
   584→        </PopoverTrigger>
   585→        {/* Portal → document.body: immune to sidebar overflow clipping.

   630→function ExpandableNavItem({
   631→  item,
   632→  currentModule,
   633→  currentSubPage,
   634→  isExpanded,
   635→  onToggle,
   636→}: {
   637→  item: NavItem;
   638→  currentModule: string;
   639→  currentSubPage: string | null;
   640→  isExpanded: boolean;
   641→  onToggle: () => void;
   642→}) {
   643→  const mod = hrefToModule(item.href);
   644→  const isActive = currentModule === mod;
   645→  const sectionId = `submenu-${item.label.toLowerCase().replace(/\s+/g, '-')}`;
   646→
   647→  return (
   648→    <SidebarMenuItem>
   649→      <SidebarMenuButton
   650→        // Expandable parent item rendered in EXPANDED state. SidebarMenuButton
   651→        // hides its tooltip whenever state!=="collapsed" (see ui/sidebar.tsx),
   652→        // so this label never shows while expanded — but passing the SAME
   653→        // COLLAPSED_TOOLTIP_PROPS object as every other nav-item renderer
   654→        // keeps the prop consistent across the codebase. If the sidebar
   655→        // toggles between expanded and collapsed while this row stays
   656→        // mounted, the tooltip that appears is identical in positioning to
   657→        // SimpleNavItem / CollapsedParentNavItem / CollapsedLogoButton.
   658→        tooltip={{ ...COLLAPSED_TOOLTIP_PROPS, children: item.label }}
   659→        isActive={isActive}
   660→        onClick={(e: React.MouseEvent) => {
   661→          e.preventDefault();
   662→          onToggle();
   663→        }}
   664→        aria-expanded={isExpanded}
   665→        aria-controls={sectionId}
   666→      >
   667→        <NavIcon name={item.icon} />
   668→        <span>{item.label}</span>
   669→        <ChevronRight
   670→          className={cn(
   671→            'ml-auto h-4 w-4 shrink-0 transition-transform duration-200',
   672→            isExpanded && 'rotate-90',
   673→          )}
   674→        />
   675→      </SidebarMenuButton>
   676→
   677→      {item.badge != null && (
   678→        <SidebarMenuBadge>
   679→          {item.badge}
   680→        </SidebarMenuBadge>
   681→      )}
   682→
   683→      <AccordionSubmenu isOpen={isExpanded} sectionLabel={item.label}>
   684→        {item.children!.map((child) => {
   685→          const hash = child.href.replace(/^#/, '');
   686→          const parts = hash.split('/');
   687→          const childMod = parts[0];
   688→          const childSubPage = parts[1] || null;
   689→          const isChildActive =
   690→            currentModule === childMod &&
   691→            (!childSubPage || currentSubPage === childSubPage);
   692→
   693→          return (
   694→            <SidebarMenuSubItem key={child.label}>
   695→              <SidebarMenuSubButton
   696→                asChild
   697→                isActive={isChildActive}
   698→              >
   699→                <a
   700→                  href={child.href}
   701→                  onClick={(e) => {
   702→                    e.preventDefault();
   703→                    e.stopPropagation();
   704→                    useNavigationStore
   705→                      .getState()
   706→                      .navigate(childMod, null, childSubPage);
   707→                  }}
   708→                >
   709→                  <NavIcon name={child.icon} />
   710→                  <span>{child.label}</span>
   711→                </a>
   712→              </SidebarMenuSubButton>
   713→            </SidebarMenuSubItem>
   714→          );
   715→        })}
   716→      </AccordionSubmenu>
   717→    </SidebarMenuItem>
   718→  );

   770→function NavGroupSection({
   771→  group,
   772→  currentModule,
   773→  currentSubPage,
   774→  openSection,
   775→  onToggleSection,
   776→}: {
   777→  group: NavGroup;
   778→  currentModule: string;
   779→  currentSubPage: string | null;
   780→  openSection: string | null;
   781→  onToggleSection: (label: string) => void;
   782→}) {

   785→  return (
   786→    <SidebarGroup>
   787→      <SidebarGroupContent>
   788→        <SidebarMenu>
   789→          {group.items.map((item) => {
   790→            const hasChildren = item.children && item.children.length > 0;
   791→
   792→            if (hasChildren && isCollapsedRail) {
   793→              /* Collapsed rail → parents with children become floating
   794→                 popovers anchored to their own icon. The rail NEVER
   795→                 auto-expands (#4/#5/#6). */
   796→              return (
   797→                <CollapsedParentNavItem
   798→                  key={item.label}
   799→                  item={item}
   800→                  currentModule={currentModule}
   801→                  currentSubPage={currentSubPage}
   802→                />
   803→              );
   804→            }
   805→
   806→            if (hasChildren) {
   807→              return (
   808→                <ExpandableNavItem
   809→                  key={item.label}
   810→                  item={item}
   811→                  currentModule={currentModule}
   812→                  currentSubPage={currentSubPage}
   813→                  isExpanded={openSection === item.label}
   814→                  onToggle={() => onToggleSection(item.label)}
   815→                />
   816→              );
   817→            }
   818→
   819→            return (
   820→              <SimpleNavItem
   821→                key={item.label}
   822→                item={item}
   823→                currentModule={currentModule}
   824→              />
   825→            );
   826→          })}
   827→        </SidebarMenu>
   828→      </SidebarGroupContent>
   829→    </SidebarGroup>
   830→  );
   831→}
