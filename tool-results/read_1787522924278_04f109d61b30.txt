    91→    91→function getFileIcon(mimeType: string) {
    92→    92→  if (mimeType.startsWith('image/')) return ImageIcon;
    93→    93→  if (mimeType.startsWith('video/')) return Film;
    94→    94→  if (mimeType.startsWith('audio/')) return Music;
    95→    95→  if (mimeType.includes('pdf') || mimeType.includes('document') || mimeType.includes('text')) return FileText;
    96→    96→  return File;
    97→    97→}

   100→   100→/** Flatten a tree of folders into an indented list for select dropdowns */
   101→   101→function flattenFolderTree(
   102→   102→  folders: MediaFolderItem[],
   103→   103→  depth = 0,
   104→   104→): { id: string; name: string; depth: number }[] {
   105→   105→  const result: { id: string; name: string; depth: number }[] = [];
   106→   106→  for (const f of folders) {
   107→   107→    result.push({ id: f.id, name: f.name, depth });
   108→   108→    if (f.children && f.children.length > 0) {
   109→   109→      result.push(...flattenFolderTree(f.children, depth + 1));
   110→   110→    }
   111→   111→  }
   112→   112→  return result;
   113→   113→}
