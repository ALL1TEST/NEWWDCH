   434→// ============================================================
   435→// SMTP Settings Page
   436→// ============================================================
   437→
   438→export function SmtpSettingsPage() {
   439→  const queryClient = useQueryClient();
   440→  const [dialogOpen, setDialogOpen] = useState(false);
   441→  const [editingItem, setEditingItem] = useState<ProviderFormData | null>(null);
   442→  const [deleteTarget, setDeleteTarget] = useState<SmtpSetting | null>(null);
   443→
   444→  // -------------------- Fetch --------------------
   445→
   446→  const {
   447→    data: response,
   448→    isLoading,
   449→    isError,
   450→  } = useQuery<PaginatedResponse<SmtpSetting>>({
   451→    queryKey: queryKeys.smtpSettings.list(),
   452→    queryFn: () => getApi<PaginatedResponse<SmtpSetting>>('/api/smtp-settings', { pageSize: 100 }),
   453→  });
   454→
   455→  const settings = response?.data ?? [];
   456→
   457→  // -------------------- Create/Update Mutation --------------------
   458→
   459→  const upsertMutation = useMutation({
   460→    mutationFn: async (form: ProviderFormData) => {
   461→      if (form.id) {
   462→        const payload: Record<string, unknown> = {
   463→          name: form.name,
   464→          provider: form.provider,
   465→          host: form.host,
   466→          port: form.port,
   467→          username: form.username,
   468→          fromName: form.fromName,
   469→          fromEmail: form.fromEmail,
   470→          replyTo: form.replyTo,
   471→          isDefault: form.isDefault,
   472→          isActive: form.isActive,
   473→        };
   474→        // Only send password if changed (non-empty)
   475→        if (form.password) payload.password = form.password;
   476→        return patchApi<SmtpSetting>(`/api/smtp-settings/${form.id}`, payload);
   477→      }
   478→      return postApi<SmtpSetting>('/api/smtp-settings', {
   479→        name: form.name,
   480→        provider: form.provider,
   481→        host: form.host,
   482→        port: form.port,
   483→        username: form.username,
   484→        password: form.password,
   485→        fromName: form.fromName,
   486→        fromEmail: form.fromEmail,
   487→        replyTo: form.replyTo,
   488→        isDefault: form.isDefault,
   489→        isActive: form.isActive,
   490→      });
   491→    },
   492→    onSuccess: () => {
   493→      queryClient.invalidateQueries({ queryKey: queryKeys.smtpSettings.all });
   494→      setDialogOpen(false);
   495→      setEditingItem(null);
   496→      toast.success(editingItem ? 'Provider updated' : 'Provider added');
   497→    },
   498→    onError: (err: Error) => {
   499→      toast.error(err.message || 'Failed to save provider');
   500→    },
   501→  });
   502→
   503→  // -------------------- Delete Mutation --------------------
   504→
   505→  const deleteMutation = useMutation({
   506→    mutationFn: (id: string) => deleteApi(`/api/smtp-settings/${id}`),
   507→    onSuccess: () => {
   508→      queryClient.invalidateQueries({ queryKey: queryKeys.smtpSettings.all });
   509→      setDeleteTarget(null);
   510→      toast.success('Provider deleted');
   511→    },
   512→    onError: (err: Error) => {
   513→      toast.error(err.message || 'Failed to delete provider');
   514→    },
   515→  });
   516→
   517→  // -------------------- Handlers --------------------
   518→
   519→  const handleOpenCreate = () => {
   520→    setEditingItem(null);
   521→    setDialogOpen(true);
   522→  };
   523→
   524→  const handleOpenEdit = (setting: SmtpSetting) => {
   525→    setEditingItem({
   526→      id: setting.id,
   527→      name: setting.name,
   528→      provider: setting.provider,
   529→      host: setting.host,
   530→      port: setting.port,
   531→      username: setting.username,
   532→      password: '',
   533→      fromName: setting.fromName,
   534→      fromEmail: setting.fromEmail,
   535→      replyTo: setting.replyTo,
   536→      isDefault: setting.isDefault,
   537→      isActive: setting.isActive,
   538→      region: 'us-east-1',
   539→    });
   540→    setDialogOpen(true);
   541→  };
   542→
   543→  const handleSubmit = (form: ProviderFormData) => {
   544→    upsertMutation.mutate(form);
   545→  };
   546→
   547→  const handleDelete = () => {
   548→    if (deleteTarget) {
   549→      deleteMutation.mutate(deleteTarget.id);
   550→    }
   551→  };
   552→
   553→  // -------------------- Render --------------------
   554→
   555→  return (
   556→    <div className="space-y-6">
   557→      {/* ==================== Page Header ==================== */}
   558→      <PageHeader
   559→        title="SMTP Settings"
   560→        description="Configure email delivery providers and SMTP settings for your sites."
   561→        action={
   562→          <Button onClick={handleOpenCreate} size="sm">
   563→            <Plus className="h-4 w-4" />
   564→            Add Provider
   565→          </Button>
   566→        }
   567→      />
   568→
   569→      {/* ==================== Content ==================== */}
   570→      {isLoading ? (
   571→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   572→          {Array.from({ length: 3 }).map((_, i) => (
   573→            <Skeleton key={i} className="h-56 rounded-xl" />
   574→          ))}
   575→        </div>
   576→      ) : isError ? (
   577→        <div className="flex flex-col items-center justify-center py-20 text-center">
   578→          <p className="text-sm text-muted-foreground">Failed to load SMTP settings.</p>
   579→        </div>
   580→      ) : settings.length === 0 ? (
   581→        <EmptyState
   582→          icon={Server}
   583→          title="No Email Providers"
   584→          description="Add your first email delivery provider to start sending transactional and marketing emails."
   585→          action={{
   586→            label: 'Add Provider',
   587→            onClick: handleOpenCreate,
   588→            icon: <Plus className="h-4 w-4" />,
   589→          }}
   590→        />
   591→      ) : (
   592→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   593→          {settings.map((setting) => (
   594→            <ProviderCard
   595→              key={setting.id}
   596→              setting={setting}
   597→              onEdit={() => handleOpenEdit(setting)}
   598→              onDelete={() => setDeleteTarget(setting)}
   599→            />
   600→          ))}
   601→        </div>
   602→      )}
   603→
   604→      {/* ==================== Add/Edit Dialog ==================== */}
   605→      <ProviderDialog
   606→        key={editingItem?.id ?? 'create'}
   607→        open={dialogOpen}
   608→        onOpenChange={(open) => {
   609→          setDialogOpen(open);
   610→          if (!open) setEditingItem(null);
   611→        }}
   612→        data={editingItem}
   613→        onSubmit={handleSubmit}
   614→        isLoading={upsertMutation.isPending}
   615→      />
   616→
   617→      {/* ==================== Delete Confirmation ==================== */}
   618→      <ConfirmDialog
   619→        open={!!deleteTarget}
   620→        onOpenChange={(open) => !open && setDeleteTarget(null)}
   621→        title="Delete Provider"
   622→        description={
   623→          deleteTarget
   624→            ? `Are you sure you want to delete "${deleteTarget.name}"? This action cannot be undone.`
   625→            : ''
   626→        }
   627→        confirmLabel="Delete"
   628→        variant="destructive"
   629→        onConfirm={handleDelete}
   630→        isLoading={deleteMutation.isPending}
   631→      />
   632→    </div>
   633→  );
   634→}

   747→              'font-medium border-transparent text-[10px] leading-4 px-1.5',
