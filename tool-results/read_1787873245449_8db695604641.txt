   140→// -------------------- KPI Card --------------------
   141→function KpiCard({
   142→  label,
   143→  value,
   144→ sublabel,
   145→ icon,
   146→  trend,
   147→  color = 'default',
   148→}: {
   149→  label: string;
   150→  value: string | number;
   151→  sublabel?: string;
   152→  icon: React.ReactNode;
   153→  trend?: 'up' | 'down' | 'neutral';
   154→  color?: 'emerald' | 'amber' | 'violet' | 'rose' | 'default';
   155→}) {
   156→  const colorMap = {
   157→    emerald: 'bg-emerald-50 text-emerald-600 dark:bg-emerald-950/40 dark:text-emerald-400',
   158→    amber: 'bg-amber-50 text-amber-600 dark:bg-amber-950/40 dark:text-amber-400',
   159→    violet: 'bg-violet-50 text-violet-600 dark:bg-violet-950/40 dark:text-violet-400',
   160→    rose: 'bg-rose-50 text-rose-600 dark:bg-rose-950/40 dark:text-rose-400',
   161→    default: 'bg-muted text-muted-foreground',
   162→  };
   163→
   164→  return (
   165→    <Card className="relative overflow-hidden">
   166→      <CardContent className="p-4">
   167→        <div className="flex items-start justify-between">
   168→          <div className="space-y-1">
   169→            <p className="text-xs font-medium text-muted-foreground uppercase tracking-wider">{label}</p>
   170→            <p className="text-2xl font-bold tracking-tight">{value}</p>
   171→            {sublabel && (
   172→              <p className="text-xs text-muted-foreground">{sublabel}</p>
   173→            )}
   174→          </div>
   175→          <div className={cn('h-9 w-9 rounded-lg flex items-center justify-center', colorMap[color])}>
   176→            {icon}
   177→          </div>
   178→        </div>
   179→        {trend && (
   180→          <div className="flex items-center gap-1 mt-2">
   181→            {trend === 'up' && <TrendingUp className="h-3 w-3 text-emerald-500" />}
   182→            {trend === 'down' && <AlertTriangle className="h-3 w-3 text-rose-500" />}
   183→            <span className={cn('text-xs font-medium', trend === 'up' ? 'text-emerald-500' : trend === 'down' ? 'text-rose-500' : 'text-muted-foreground')}>
   184→              {trend === 'up' ? 'Trending up' : trend === 'down' ? 'Needs attention' : 'Stable'}
   185→            </span>
   186→          </div>
   187→        )}
   188→      </CardContent>
   189→    </Card>
   190→  );
   191→}

   258→  const style = typeStyles[action.type];

   260→  return (
   261→    <div className={cn('flex items-center gap-3 py-2.5 px-3 rounded-lg hover:bg-accent/50 transition-colors', style.bg)}>
   262→      <div className="shrink-0">{style.icon}</div>
   263→      <div className="flex-1 min-w-0">
   264→        <div className="flex items-center gap-2">
   265→          <span className="text-xs font-medium text-muted-foreground">{action.site}</span>
   266→          {style.badge}
   267→        </div>
   268→        <p className="text-sm truncate mt-0.5">{action.message}</p>
   269→      </div>
   270→      <div className="flex items-center gap-2 shrink-0">
   271→        <span className="text-xs text-muted-foreground hidden sm:block">{action.time}</span>
   272→        {action.action && (

   278→    </div>
