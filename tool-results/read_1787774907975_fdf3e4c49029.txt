   215→                <span className="truncate">{p.label}</span>

   219→          })}

   270→// -------------------- Validation --------------------
   271→
   272→/** Returns true when the Name field is non-empty AND every required field
   273→ *  for the currently-selected provider has a non-empty value. Fields that
   274→ *  belong to OTHER providers are intentionally NOT checked — the user's
   275→ *  requirement: "Do not validate hidden fields belonging to another
   276→ *  provider." Optional fields (required: false) are never checked. */
   277→function isFormValid(form: StorageForm): boolean {
   278→  if (!form.name.trim()) return false;
   279→  const fields = PROVIDER_FIELDS[form.provider] ?? [];
   280→  return fields.every(
   281→    (f) => !f.required || (form.config[f.key] != null && form.config[f.key].trim() !== ''),
   282→  );
   283→}

   310→export function StoragePage() {
   311→  const queryClient = useQueryClient();
   312→  const [dialogOpen, setDialogOpen] = useState(false);
   313→  const [editingId, setEditingId] = useState<string | null>(null);
   314→  const [form, setForm] = useState<StorageForm>(initialForm);
   315→  const [deleteTarget, setDeleteTarget] = useState<StorageRow | null>(null);
   316→
   317→  const table = useDataTable({ initialSortField: 'createdAt', initialSortOrder: 'desc' });
   318→
   319→  const { data, isLoading } = useQuery({
   320→    queryKey: queryKeys.backupStorage.list({
   321→      page: table.currentPage,
   322→      pageSize: table.pageSize,
   323→      sort: table.sortField,
   324→      order: table.sortOrder,
   325→      search: table.searchValue || undefined,
   326→    }),
   327→    queryFn: () => getApi<ApiResponse<StorageRow[]>>('/api/backups/storage', {
   328→      page: table.currentPage,
   329→      pageSize: table.pageSize,
   330→      sort: table.sortField,
   331→      order: table.sortOrder,
   332→      search: table.searchValue || undefined,
   333→    }, { raw: true }),
   334→    staleTime: 10_000,
   335→  });
   336→
   337→  const storages = data?.data ?? [];
   338→  const pagination = data?.meta?.pagination;
   339→
   340→  // ---- Dual empty-state logic ----
   341→  // TWO distinct empty states:
   342→  //   A) isInitialEmpty — system has ZERO storage destinations AND no active
   343→  //      search. Render the existing full-page "No storage configured" state.
   344→  //   B) isSearchEmpty — destinations exist (or could exist) but the current
   345→  //      search returns zero results. Keep the table/card/headers/footer
   346→  //      visible and render "No storage found" INSIDE the table body via the
   347→  //      DataTable `emptyState` prop (DataTableEmpty spans colSpan=999 so
   348→  //      the headers stay visible above it).
   349→  // Search filtering must only affect table rows, never the page-level
   350→  // empty state — so the full-page EmptyState is gated on `!hasSearch`.
   351→  const hasSearch = !!table.searchValue?.trim();
   352→  const isInitialEmpty = !isLoading && storages.length === 0 && !hasSearch;
   353→  const isSearchEmpty = !isLoading && storages.length === 0 && hasSearch;
   354→
   355→  const createMutation = useMutation({
   356→    mutationFn: (body: { name: string; provider: BackupStorageProvider; config: string }) =>
   357→      postApi('/api/backups/storage', body),
   358→    onSuccess: () => {
   359→      queryClient.invalidateQueries({ queryKey: queryKeys.backupStorage.all });
   360→      toast.success('Storage configuration created');
   361→      closeDialog();
   362→    },
   363→    onError: (err: Error) => toast.error(err.message || 'Failed to create storage configuration'),
   364→  });
   365→
   366→  const updateMutation = useMutation({
   367→    mutationFn: ({ id, body }: { id: string; body: { name: string; provider: BackupStorageProvider; config: string } }) =>
   368→      patchApi(`/api/backups/storage/${id}`, body),
   369→    onSuccess: () => {
   370→      queryClient.invalidateQueries({ queryKey: queryKeys.backupStorage.all });
   371→      toast.success('Storage configuration updated');
   372→      closeDialog();
   373→    },
   374→    onError: (err: Error) => toast.error(err.message || 'Failed to update storage configuration'),
   375→  });
   376→
   377→  const deleteMutation = useMutation({
   378→    mutationFn: (id: string) => deleteApi(`/api/backups/storage/${id}`),
   379→    onSuccess: () => {
   380→      queryClient.invalidateQueries({ queryKey: queryKeys.backupStorage.all });
   381→      setDeleteTarget(null);
   382→      toast.success('Storage configuration deleted');
   383→    },
   384→    onError: (err: Error) => {
   385→      setDeleteTarget(null);
   386→      toast.error(err.message || 'Failed to delete storage configuration');
   387→    },
   388→  });
   389→
   390→  const testConnectionMutation = useMutation({
   391→    mutationFn: (id: string) => postApi(`/api/backups/storage/${id}/test-connection`),
   392→    onSuccess: (_data, id) => {
   393→      queryClient.invalidateQueries({ queryKey: queryKeys.backupStorage.detail(id) });
   394→      toast.success('Connection test passed');
   395→    },
   396→    onError: (err: Error) => toast.error(err.message || 'Connection test failed'),
   397→  });
   398→
   399→  const openCreate = () => {
   400→    setEditingId(null);
   401→    setForm(initialForm);
   402→    setDialogOpen(true);
   403→  };
   404→
   405→  const openEdit = (row: StorageRow) => {
   406→    setEditingId(row.id);
   407→    // Flatten the stored config object into a string map for form inputs.
   408→    // Numbers (e.g. port: 21) are coerced to strings; nullish values become ''.
   409→    const storedConfig = row.config ?? {};
   410→    const formConfig: Record<string, string> = {};
   411→    for (const [k, v] of Object.entries(storedConfig)) {
   412→      formConfig[k] = v == null ? '' : String(v);
   413→    }
   414→    setForm({
   415→      name: row.name,
   416→      provider: row.provider,
   417→      config: formConfig,
   418→    });
   419→    setDialogOpen(true);
   420→  };
   421→
   422→  const closeDialog = () => {
   423→    setDialogOpen(false);
   424→    setEditingId(null);
   425→    setForm(initialForm);
   426→  };
   427→
   428→  // Update a single config field value within the form state. Preserves
   429→  // values for other providers' fields (they stay in `form.config` but are
   430→  // not rendered/validated/submitted when their provider isn't selected).
   431→  const updateConfigField = (key: string, value: string) => {
   432→    setForm((prev) => ({
   433→      ...prev,
   434→      config: { ...prev.config, [key]: value },
   435→    }));
   436→  };
   437→
   438→  const handleSubmit = () => {
   439→    if (!isFormValid(form)) return;
   440→    // Build the config object from ONLY the current provider's fields.
   441→    // This ensures hidden fields (from another provider) are never sent to
   442→    // the API, and the value type matches what the server expects (number
   443→    // fields like `port` are converted from string to Number). Empty
   444→    // optional fields are omitted entirely (the server treats absent keys
   445→    // the same as empty). The API schema expects `config` as a JSON string,
   446→    // so we JSON.stringify before sending.
   447→    const fields = PROVIDER_FIELDS[form.provider] ?? [];
   448→    const configObj: Record<string, unknown> = {};
   449→    for (const f of fields) {
   450→      const raw = form.config[f.key];
   451→      if (raw == null || raw.trim() === '') continue;
   452→      configObj[f.key] = f.type === 'number' ? Number(raw) : raw;
   453→    }
   454→    const body = {
   455→      name: form.name,
   456→      provider: form.provider,
   457→      config: JSON.stringify(configObj),
   458→    };
   459→    if (editingId) {
   460→      updateMutation.mutate({ id: editingId, body });
   461→    } else {
   462→      createMutation.mutate(body);
   463→    }
   464→  };
   465→
   466→  const updateForm = <K extends keyof StorageForm>(key: K, value: StorageForm[K]) => {
   467→    setForm((prev) => ({ ...prev, [key]: value }));
   468→  };
   469→
   470→  const isSaving = createMutation.isPending || updateMutation.isPending;
   471→
   472→  const columns: ColumnDef<StorageRow>[] = [
   473→      ColumnDefHelper.textColumn<StorageRow>({
   474→        id: 'name',
   475→        header: 'Name',
   476→        accessorKey: 'name',
   477→        className: 'font-medium',
   478→      }),
   479→      {
   480→        id: 'provider',
   481→        header: 'Provider',
   482→        accessorKey: 'provider',
   483→        enableSorting: false,
   484→        size: 150,
   485→        cell: ({ getValue }) => <ProviderBadge provider={getValue() as BackupStorageProvider} />,
   486→      },
   487→      {
   488→        id: 'isActive',
   489→        header: 'Status',
   490→        accessorKey: 'isActive',
   491→        enableSorting: false,
   492→        size: 100,
   493→        cell: ({ getValue }) => (
   494→          <Badge
   495→            variant="outline"
   496→            className={cn(
   497→              'border-transparent font-medium',
   498→              getValue()
   499→                ? 'bg-green-100 text-green-700 dark:bg-green-900/30 dark:text-green-400'
   500→                : 'bg-zinc-100 text-zinc-500 dark:bg-zinc-800 dark:text-zinc-400'
   501→            )}
   502→          >
   503→            {getValue() ? 'Active' : 'Inactive'}
   504→          </Badge>
   505→        ),
   506→      },
   507→      {
   508→        id: 'lastTestAt',
   509→        header: 'Last Test',
   510→        accessorKey: 'lastTestAt',
   511→        size: 140,
   512→        cell: ({ getValue }) => {
   513→          const val = getValue() as string | null;
   514→          return (
   515→            <span className="text-xs text-muted-foreground">
   516→              {val ? formatRelativeTime(val) : '—'}
   517→            </span>
   518→          );
   519→        },
   520→      },
   521→      {
   522→        id: 'lastTestResult',
   523→        header: 'Test Result',
   524→        accessorKey: 'lastTestResult',
   525→        enableSorting: false,
   526→        size: 120,
   527→        cell: ({ row }) => {
   528→          const result = row.original.lastTestResult;
   529→          if (!result) return <span className="text-muted-foreground text-xs">—</span>;
   530→          return result === 'SUCCESS' ? (
   531→            <span className="inline-flex items-center gap-1 text-xs font-medium text-green-600 dark:text-green-400">
   532→              <CheckCircle2 className="h-3.5 w-3.5" />
   533→              Passed
   534→            </span>
   535→          ) : (
   536→            <span className="inline-flex items-center gap-1 text-xs font-medium text-red-600 dark:text-red-400">
   537→              <XCircle className="h-3.5 w-3.5" />
   538→              Failed
   539→            </span>
   540→          );
   541→        },
   542→      },
   543→      ColumnDefHelper.actionColumn<StorageRow>({
   544→        id: 'actions',
   545→        size: 50,
   546→        render: (row) => (
   547→          <DropdownMenu>
   548→            <DropdownMenuTrigger asChild>
   549→              <Button variant="ghost" size="icon" className="h-8 w-8" onClick={(e) => e.stopPropagation()}>
   550→                <MoreHorizontal className="h-4 w-4" />
   551→              </Button>
   552→            </DropdownMenuTrigger>
   553→            <DropdownMenuContent align="end">
   554→              <DropdownMenuItem onClick={() => openEdit(row)}>
   555→                <Pencil className="h-4 w-4 mr-2" />
   556→                Edit
   557→              </DropdownMenuItem>
   558→              <DropdownMenuItem
   559→                onClick={() => testConnectionMutation.mutate(row.id)}
   560→                disabled={testConnectionMutation.isPending}
   561→              >
   562→                {testConnectionMutation.isPending ? (
   563→                  <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   564→                ) : (
   565→                  <Plug className="h-4 w-4 mr-2" />
   566→                )}
   567→                Test Connection
   568→              </DropdownMenuItem>
   569→              <DropdownMenuSeparator />
   570→              <DropdownMenuItem variant="destructive" onClick={() => setDeleteTarget(row)}>
   571→                <Trash2 className="h-4 w-4 mr-2" />
   572→                Delete
   573→              </DropdownMenuItem>
   574→            </DropdownMenuContent>
   575→          </DropdownMenu>
   576→        ),
   577→      }),
   578→    ];
   579→
   580→  return (
   581→    <div className="space-y-4">
   582→      <PageHeader
   583→        title="Storage"
   584→        description="Configure backup storage destinations"
   585→        action={
   586→          <Button size="sm" onClick={openCreate}>
   587→            <Plus className="h-4 w-4 mr-2" />
   588→            Add Storage
   589→          </Button>
   590→        }
   591→      />
   592→
   593→      {isInitialEmpty ? (
   594→        <EmptyState
   595→          icon={HardDrive}
   596→          title="No storage configured"
   597→          description="Add a storage destination to save your backups."
   598→          action={{ label: 'Add Storage', onClick: openCreate }}
   599→        />
   600→      ) : (
   601→        <DataTable
   602→          columns={columns}
   603→          data={storages}
   604→          isLoading={isLoading}
   605→          totalItems={pagination?.total ?? 0}
   606→          pageSize={table.pageSize}
   607→          currentPage={table.currentPage}
   608→          onPageChange={(p) => table.setCurrentPage(p)}
   609→          onSortChange={(f, o) => table.setSortField(f, o)}
   610→          sortField={table.sortField}
   611→          sortOrder={table.sortOrder}
   612→          searchPlaceholder="Search storage..."
   613→          searchValue={table.searchValue}
   614→          onSearch={(v) => {
   615→            table.setSearchValue(v);
   616→            table.setCurrentPage(1);
   617→          }}
   618→          getRowId={(row) => row.id}
   619→          // When an active search returns zero results, keep the table/card,
   620→          // headers, search input, and footer visible and render the
   621→          // "No storage found" empty state INSIDE the table body (DataTableEmpty
   622→          // spans colSpan=999 so the headers remain visible above it). The
   623→          // result count (totalItems=0) is passed through so the footer
   624→          // correctly shows 0 matching items.
   625→          emptyMessage="No storage configurations found."
   626→          emptyState={
   627→            isSearchEmpty ? (
   628→              <NoStorageSearchEmpty
   629→                onClear={() => {
   630→                  table.setSearchValue('');
   631→                  table.setCurrentPage(1);
   632→                }}
   633→              />
   634→            ) : undefined
   635→          }
   636→        />
   637→      )}
   638→
   639→      {/* Create/Edit Dialog */}
   640→      <Dialog open={dialogOpen} onOpenChange={(v) => !v && closeDialog()}>
   641→        <DialogContent className="sm:max-w-[520px]">
   642→          <DialogHeader>
   643→            <DialogTitle>{editingId ? 'Edit Storage' : 'Add Storage'}</DialogTitle>
   644→            <DialogDescription>
   645→              {editingId
   646→                ? 'Update storage destination configuration.'
   647→                : 'Configure a new backup storage destination.'}
   648→            </DialogDescription>
   649→          </DialogHeader>
   650→          <div className="space-y-4 py-2 max-h-[60vh] overflow-y-auto pr-1">
   651→            <div className="space-y-2">
   652→              <Label htmlFor="storage-name">Name</Label>
   653→              <Input
   654→                id="storage-name"
   655→                placeholder="e.g., Production S3 Bucket"
   656→                value={form.name}
   657→                onChange={(e) => updateForm('name', e.target.value)}
   658→              />
   659→            </div>
   660→
   661→            <div className="space-y-2">
   662→              <Label htmlFor="storage-provider">Provider</Label>
   663→              <ProviderDropdown
   664→                value={form.provider}
   665→                onChange={(v) => updateForm('provider', v)}
   666→              />
   667→            </div>
   668→
   669→            {/* Dynamic provider-specific configuration fields. Only the
   670→                fields for the currently-selected provider are rendered —
   671→                fields for other providers are hidden (and neither
   672→                validated nor submitted). Switching providers updates this
   673→                list immediately without clearing already-typed values for
   674→                fields shared across providers (e.g. `bucket` is used by
   675→                both AMAZON_S3 and CLOUDFLARE_R2, so switching between them
   676→                preserves the typed bucket name). */}
   677→            {(PROVIDER_FIELDS[form.provider] ?? []).length > 0 && (
   678→              <div className="space-y-3">
   679→                {(PROVIDER_FIELDS[form.provider] ?? []).map((f) => {
   680→                  const fieldValue = form.config[f.key] ?? '';
   681→                  const inputId = `storage-config-${f.key}`;
   682→                  return (
   683→                    <div key={f.key} className="space-y-2">
   684→                      <Label htmlFor={inputId}>
   685→                        {f.label}
   686→                        {f.required && <span className="text-destructive ml-0.5">*</span>}
   687→                      </Label>
   688→                      {f.multiline ? (
   689→                        <Textarea
   690→                          id={inputId}
   691→                          placeholder={f.placeholder}
   692→                          value={fieldValue}
   693→                          onChange={(e) => updateConfigField(f.key, e.target.value)}
   694→                          rows={4}
   695→                          className="font-mono text-xs"
   696→                        />
   697→                      ) : (
   698→                        <Input
   699→                          id={inputId}
   700→                          type={f.type === 'password' ? 'password' : f.type === 'number' ? 'number' : 'text'}
   701→                          placeholder={f.placeholder}
   702→                          value={fieldValue}
   703→                          onChange={(e) => updateConfigField(f.key, e.target.value)}
   704→                          autoComplete={f.type === 'password' ? 'new-password' : undefined}
   705→                        />
   706→                      )}
   707→                      {f.helpText && (
   708→                        <p className="text-xs text-muted-foreground">{f.helpText}</p>
   709→                      )}
   710→                    </div>
   711→                  );
   712→                })}
   713→              </div>
   714→            )}
   715→          </div>
   716→          <DialogFooter>
   717→            <Button variant="outline" onClick={closeDialog}>
   718→              Cancel
   719→            </Button>
   720→            <Button
   721→              onClick={handleSubmit}
   722→              disabled={isSaving || !isFormValid(form)}
   723→            >
   724→              {isSaving && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   725→              {editingId ? 'Update' : 'Create'}
   726→            </Button>
   727→          </DialogFooter>
   728→        </DialogContent>
   729→      </Dialog>
   730→
   731→      {/* Delete Confirmation */}
   732→      <ConfirmDialog
   733→        open={!!deleteTarget}
   734→        onOpenChange={(v) => !v && setDeleteTarget(null)}
   735→        title="Delete Storage Configuration"
   736→        description={
   737→          deleteTarget
   738→            ? `Are you sure you want to delete "${deleteTarget.name}"? Existing backups stored via this configuration will not be affected.`
   739→            : undefined
   740→        }
   741→        confirmLabel="Delete"
   742→        variant="destructive"
   743→        onConfirm={() => {
   744→          if (deleteTarget) deleteMutation.mutate(deleteTarget.id);
   745→        }}
   746→        isLoading={deleteMutation.isPending}
   747→      />
   748→    </div>
   749→  );
   750→}
