   123→// ==================== Helpers ====================
   124→
   125→function formatNumber(n: number | null | undefined): string {
   126→  const num = n ?? 0;
   127→  if (num >= 1000000) return `${(num / 1000000).toFixed(1)}M`;
   128→  if (num >= 1000) return `${(num / 1000).toFixed(1)}K`;
   129→  return num.toLocaleString();
   130→}
   131→
   132→function formatPercent(n: number | null | undefined): string {
   133→  return `${((n ?? 0) * 100).toFixed(2)}%`;
   134→}
   135→
   136→function formatPosition(n: number | null | undefined): string {
   137→  return (n ?? 0).toFixed(1);
   138→}
   139→
   140→function formatDate(dateStr: string | null): string {
   141→  if (!dateStr) return 'Never';
   142→  try {
   143→    return new Date(dateStr).toLocaleString(undefined, {
   144→      month: 'short',
   145→      day: 'numeric',
   146→      year: 'numeric',
   147→      hour: '2-digit',
   148→      minute: '2-digit',
   149→    });
   150→  } catch {
   151→    return 'Unknown';
   152→  }
   153→}
   154→
   155→function formatShortDate(dateStr: string): string {
   156→  try {
   157→    return new Date(dateStr).toLocaleDateString(undefined, {
   158→      month: 'short',
   159→      day: 'numeric',
   160→    });
   161→  } catch {
   162→    return dateStr;
   163→  }
   164→}

   241→// Custom recharts tooltip — shows the exact date + per-day values.
   242→function ChartTooltip({
   243→  active,
   244→  payload,
   245→}: {
   246→  active?: boolean;
   247→  payload?: Array<{ payload: DailyStat }>;
   248→}) {
   249→  if (!active || !payload || payload.length === 0) return null;
   250→  const point = payload[0].payload;
   251→  let dateLabel = point.date;
   252→  try {
   253→    dateLabel = new Date(`${point.date}T00:00:00`).toLocaleDateString(undefined, {
   254→      year: 'numeric',
   255→      month: 'short',
   256→      day: 'numeric',
   257→    });
   258→  } catch {
   259→    /* keep raw date */
   260→  }
   261→  return (
   262→    <div className="rounded-md border bg-popover text-popover-foreground shadow-md px-3 py-2 text-xs min-w-[170px]">
   263→      <div className="font-medium mb-1.5">{dateLabel}</div>
   264→      <div className="flex items-center justify-between gap-3">
   265→        <span className="flex items-center gap-1.5">
   266→          <span className="h-2 w-2 rounded-sm bg-primary" /> Clicks
   267→        </span>
   268→        <span className="font-medium tabular-nums">{point.clicks.toLocaleString()}</span>
   269→      </div>
   270→      <div className="flex items-center justify-between gap-3 mt-0.5">
   271→        <span className="flex items-center gap-1.5">
   272→          <span className="h-2 w-2 rounded-sm bg-primary/30" /> Impressions
   273→        </span>
   274→        <span className="font-medium tabular-nums">{point.impressions.toLocaleString()}</span>
   275→      </div>
   276→      {point.ctr != null && (
   277→        <div className="flex items-center justify-between gap-3 mt-0.5 text-muted-foreground">
   278→          <span>CTR</span>
   279→          <span className="tabular-nums">{(point.ctr * 100).toFixed(2)}%</span>
   280→        </div>
   281→      )}
   282→      {point.position != null && (
   283→        <div className="flex items-center justify-between gap-3 mt-0.5 text-muted-foreground">
   284→          <span>Avg Position</span>
   285→          <span className="tabular-nums">{point.position.toFixed(1)}</span>
   286→        </div>
   287→      )}
   288→    </div>
   289→  );
   290→}

   292→function PerformanceChart({ stats, isLoading, onSync, isSyncing }: PerformanceChartProps) {
   293→  if (isLoading) {
   294→    return (
   295→      <div className="space-y-3">
   296→        <Skeleton className="h-4 w-24" />
   297→        <Skeleton className="h-56 w-full" />
   298→      </div>
   299→    );
   300→  }
   301→
   302→  if (!stats || stats.length === 0) {
   303→    return (
   304→      <div className="flex flex-col items-center justify-center py-16 text-center">
   305→        <BarChart3 className="h-12 w-12 text-muted-foreground/30 mb-3" strokeWidth={1.5} />
   306→        <p className="text-sm font-medium text-muted-foreground">
   307→          No Search Console data available for this period.
   308→        </p>
   309→        <p className="text-xs text-muted-foreground mt-1">
   310→          Try a different range or sync with Search Console to see chart data.
   311→        </p>
   312→        {onSync && (
   313→          <Button
   314→            variant="outline"
   315→            size="sm"
   316→            className="mt-4"
   317→            onClick={onSync}
   318→            disabled={isSyncing}
   319→          >
   320→            {isSyncing ? (
   321→              <Loader2 className="h-3.5 w-3.5 mr-2 animate-spin" />
   322→            ) : (
   323→              <RefreshCw className="h-3.5 w-3.5 mr-2" />
   324→            )}
   325→            Sync Now
   326→          </Button>
   327→        )}
   328→      </div>
   329→    );
   330→  }
   331→
   332→  // Defensive chronological sort (oldest → newest). The API already returns
   333→  // ascending by date, but a mis-sorted payload must never scramble the X-axis.
   334→  const data = [...stats]
   335→    .map((s) => ({
   336→      date: s.date,
   337→      clicks: s.clicks,
   338→      impressions: s.impressions,
   339→      ctr: s.ctr,
   340→      position: s.position,
   341→    }))
   342→    .sort((a, b) => (a.date < b.date ? -1 : a.date > b.date ? 1 : 0));
   343→
   344→  const axisTick = { fontSize: 10, fill: 'hsl(var(--muted-foreground))' };
   345→
   346→  return (
   347→    <div className="space-y-4">
   348→      <div className="flex items-center gap-4 text-xs text-muted-foreground">
   349→        <span className="flex items-center gap-1.5">
   350→          <span className="h-2.5 w-2.5 rounded-sm bg-primary" />
   351→          Clicks
   352→        </span>
   353→        <span className="flex items-center gap-1.5">
   354→          <span className="h-2.5 w-2.5 rounded-sm bg-primary/30" />
   355→          Impressions
   356→        </span>
   357→      </div>
   358→
   359→      {/*
   360→        Dual Y-axis area chart: left axis = Impressions (large numbers),
   361→        right axis = Clicks (small numbers). Without two axes the Clicks series
   362→        would collapse to a flat line at the bottom and be unreadable. Colors
   363→        stay on the existing `primary` hue to preserve the page palette and the
   364→        manual legend above.
   365→      */}
   366→      <div className="h-56 w-full">
   367→        <ResponsiveContainer width="100%" height="100%">
   368→          <AreaChart data={data} margin={{ top: 8, right: 8, left: 0, bottom: 0 }}>
   369→            <defs>
   370→              <linearGradient id="scImpGrad" x1="0" y1="0" x2="0" y2="1">
   371→                <stop offset="0%" stopColor="hsl(var(--primary))" stopOpacity={0.3} />
   372→                <stop offset="100%" stopColor="hsl(var(--primary))" stopOpacity={0.04} />
   373→              </linearGradient>
   374→              <linearGradient id="scClkGrad" x1="0" y1="0" x2="0" y2="1">
   375→                <stop offset="0%" stopColor="hsl(var(--primary))" stopOpacity={0.9} />
   376→                <stop offset="100%" stopColor="hsl(var(--primary))" stopOpacity={0.45} />
   377→              </linearGradient>
   378→            </defs>
   379→            <CartesianGrid
   380→              strokeDasharray="3 3"
   381→              vertical={false}
   382→              stroke="hsl(var(--border))"
   383→              strokeOpacity={0.6}
   384→            />
   385→            <XAxis
   386→              dataKey="date"
   387→              tickFormatter={(value: string) => formatShortDate(value)}
   388→              minTickGap={20}
   389→              tick={axisTick}
   390→              tickLine={false}
   391→              axisLine={{ stroke: 'hsl(var(--border))' }}
   392→            />
   393→            <YAxis
   394→              yAxisId="impressions"
   395→              tickFormatter={(n: number) => formatNumber(n)}
   396→              tick={axisTick}
   397→              tickLine={false}
   398→              axisLine={false}
   399→              width={46}
   400→            />
   401→            <YAxis
   402→              yAxisId="clicks"
   403→              orientation="right"
   404→              tickFormatter={(n: number) => formatNumber(n)}
   405→              tick={axisTick}
   406→              tickLine={false}
   407→              axisLine={false}
   408→              width={40}
   409→            />
   410→            <RechartsTooltip
   411→              content={<ChartTooltip />}
   412→              cursor={{ stroke: 'hsl(var(--border))', strokeDasharray: '3 3' }}
   413→            />
   414→            <Area
   415→              yAxisId="impressions"
   416→              type="monotone"
   417→              dataKey="impressions"
   418→              name="Impressions"
   419→              stroke="hsl(var(--primary))"
   420→              strokeOpacity={0.55}
   421→              strokeWidth={1.5}
   422→              fill="url(#scImpGrad)"
   423→              fillOpacity={1}
   424→              isAnimationActive={false}
   425→            />
   426→            <Area
   427→              yAxisId="clicks"
   428→              type="monotone"
   429→              dataKey="clicks"
   430→              name="Clicks"
   431→              stroke="hsl(var(--primary))"
   432→              strokeWidth={2}
   433→              fill="url(#scClkGrad)"
   434→              fillOpacity={1}
   435→              isAnimationActive={false}
   436→            />
   437→          </AreaChart>
   438→        </ResponsiveContainer>
   439→      </div>
   440→    </div>
   441→  );
   442→}

   651→      setCustomTo(toIsoDate(end));
   652→    }

   694→      totalImpressions,

   721→    onSuccess: () => {
   722→      setConnectUrl('');
   723→      queryClient.invalidateQueries({ queryKey: queryKeys.seoSearchConsole.all });
