     4→     4→// PLATFORM PLANS & PRICING — simplified admin UI.
     5→     5→// ============================================================
     6→     6→// Owner sees compact plan cards (Beta / Pro / Max …) with a quick
     7→     7→// Active toggle, a compact price + features + limits summary,
     8→     8→// and an "Edit Plan" button. The Edit Plan button opens a Dialog
     9→     9→// with three compact sections: Basic Information, Feature Access,
    10→    10→// Usage Limits. An optional collapsed "Client Display" section
    11→    11→// holds the marketing feature list shown to clients on the
    12→    12→// Client Billing page. A "+ Create Plan" button at the top-right
    13→    13→// opens a Create Plan dialog (same sections, blank defaults) that
    14→    14→// POSTs to /api/platform/admin/plans.
    15→    15→//
    16→    16→// All writes go through /api/platform/admin/plans (GET list,
    17→    17→// POST create) + /api/platform/admin/plans/[planId] (PUT update)
    18→    18→// — owner-only — the same shared plan-config cache the Client
    19→    19→// Billing page and MRR read. Server-side entitlement enforcement
    20→    20→// (hasFeature) and usage-limit checks (checkLimit) consume the
    21→    21→// same data, so toggling a feature off here denies it for clients
    22→    22→// on the next request. The Client Billing page is NOT modified
    23→    23→// and continues to read from /api/platform/billing/me.

    88→    88→/** Resolve the badge id for the shared PlanBadge component. Falls
    89→    89→ *  back to 'beta' for any unknown plan id so future plans still
    90→    90→ *  render with a sensible neutral badge. */
    91→    91→function getPlanBadgeId(planId: string): 'beta' | 'pro' | 'max' {

   109→   109→function PlanSummaryCard({
   110→   110→  plan,
   111→   111→  canEdit,
   112→   112→}: {
   113→   113→  plan: PlanConfigData;
   114→   114→  canEdit: boolean;
   115→   115→}) {
   116→   116→  const queryClient = useQueryClient();
   117→   117→  const [editing, setEditing] = useState(false);
   118→   118→
   119→   119→  // Quick active toggle — auto-saves immediately (owner-only).
   120→   120→  // Reuses the same PUT endpoint and the same invalidation chain
   121→   121→  // as the full Edit modal so clients see the new status on the
   122→   122→  // next read.
   123→   123→  const activeMutation = useMutation({
   124→   124→    mutationFn: (active: boolean) =>
   125→   125→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, { active }),
   126→   126→    onSuccess: (data) => {
   127→   127→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   128→   128→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   129→   129→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   130→   130→      toast.success(
   131→   131→        `${data.name} ${data.active ? 'activated' : 'deactivated'} — changes are live for clients.`,
   132→   132→      );
   133→   133→    },
   134→   134→    onError: (err: unknown) => {
   135→   135→      const e = err as { error?: { message?: string }; message?: string };
   136→   136→      toast.error(
   137→   137→        e?.error?.message ?? e?.message ?? 'Unable to change plan status. OWNER access required.',
   138→   138→      );
   139→   139→    },
   140→   140→  });
   141→   141→
   142→   142→  // Feature list shown on the card. Prefer the client-facing marketing
   143→   143→  // copy (plan.features); fall back to the structured entitlement labels
   144→   144→  // so a brand-new plan with no marketing copy still surfaces a readable
   145→   145→  // feature list. Either way the data shown is real plan data — no mock.
   146→   146→  const featureItems =
   147→   147→    plan.features.length > 0
   148→   148→      ? plan.features
   149→   149→      : plan.entitlements.map((k) => ENTITLEMENT_LABELS[k as EntitlementKey] ?? k);
   150→   150→
   151→   151→  return (
   152→   152→    <>
   153→   153→      <div
   154→   154→        className={`group relative flex h-full flex-col overflow-hidden rounded-3xl border bg-card p-8 text-card-foreground transition-colors duration-300 hover:border-foreground/20 ${
   155→   155→          plan.active ? '' : 'opacity-60'
   156→   156→        }`}
   157→   157→      >
   158→   158→        {/* Header: plan name (clean text label) + quick active toggle */}
   159→   159→        <div className="flex items-start justify-between gap-3">
   160→   160→          <div className="min-w-0">
   161→   161→            <h3 className="text-2xl font-semibold tracking-tight text-foreground">{plan.name}</h3>
   162→   162→          </div>
   163→   163→          <div className="flex shrink-0 items-center gap-2">
   164→   164→            <Label
   165→   165→              htmlFor={`active-${plan.planId}`}
   166→   166→              className="cursor-pointer text-xs text-muted-foreground"
   167→   167→            >
   168→   168→              {plan.active ? 'Active' : 'Inactive'}
   169→   169→            </Label>
   170→   170→            <Switch
   171→   171→              id={`active-${plan.planId}`}
   172→   172→              checked={plan.active}
   173→   173→              disabled={!canEdit || activeMutation.isPending}
   174→   174→              onCheckedChange={(v) => activeMutation.mutate(v)}
   175→   175→            />
   176→   176→          </div>
   177→   177→        </div>
   178→   178→
   179→   179→        {/* Price — main monthly price only (yearly price intentionally omitted on the card) */}
   180→   180→        <div className="mt-8">
   181→   181→          {plan.isFree ? (
   182→   182→            <div className="flex items-baseline gap-2">
   183→   183→              <span className="text-5xl font-semibold tracking-tight text-foreground">Free</span>
   184→   184→            </div>
   185→   185→          ) : (
   186→   186→            <div className="flex items-baseline gap-2">
   187→   187→              <span className="text-5xl font-semibold tracking-tight text-foreground">
   188→   188→                {formatCurrency(plan.priceMonthly, plan.currency)}
   189→   189→              </span>
   190→   190→              <span className="text-sm text-muted-foreground">/ month</span>
   191→   191→            </div>
   192→   192→          )}
   193→   193→        </div>
   194→   194→
   195→   195→        {/* Thin horizontal divider between the price section and the features section */}
   196→   196→        <div className="mt-8 h-px bg-border" aria-hidden />
   197→   197→
   198→   198→        {/* Feature items (section label omitted) */}
   199→   199→        <div className="mt-6">
   200→   200→          <ul className="space-y-2.5">
   201→   201→            {featureItems.length === 0 ? (
   202→   202→              <li className="text-sm text-muted-foreground">No features configured</li>
   203→   203→            ) : (
   204→   204→              featureItems.map((f, i) => (
   205→   205→                <li key={`${f}-${i}`} className="flex items-start gap-2.5 text-sm text-card-foreground">
   206→   206→                  <Check className="mt-0.5 h-4 w-4 shrink-0 text-muted-foreground" />
   207→   207→                  <span className="leading-snug">{f}</span>
   208→   208→                </li>
   209→   209→              ))
   210→   210→            )}
   211→   211→          </ul>
   212→   212→        </div>
   213→   213→
   214→   214→        {/* Edit button — pinned to card bottom */}
   215→   215→        <div className="mt-auto pt-8">
   216→   216→          <Button
   217→   217→            variant="outline"
   218→   218→            size="sm"
   219→   219→            className="w-full rounded-full"
   220→   220→            disabled={!canEdit}
   221→   221→            onClick={() => setEditing(true)}
   222→   222→          >
   223→   223→            <Pencil className="h-4 w-4 mr-2" />
   224→   224→            Edit Plan
   225→   225→          </Button>
   226→   226→          {!canEdit && (
   227→   227→            <p className="mt-2 text-center text-[10px] text-muted-foreground">
   228→   228→              Editing is restricted to the OWNER role.
   229→   229→            </p>
   230→   230→          )}
   231→   231→        </div>
   232→   232→      </div>
   233→   233→
   234→   234→      {editing && (
   235→   235→        <EditPlanDialog plan={plan} open={editing} onOpenChange={setEditing} />
   236→   236→      )}
   237→   237→    </>
   238→   238→  );
   239→   239→}

   243→   243→function EditPlanDialog({
   244→   244→  plan,
   245→   245→  open,
   246→   246→  onOpenChange,

   248→   248→  plan: PlanConfigData;

   251→   251→}) {
   252→   252→  const queryClient = useQueryClient();
   253→   253→  const [name, setName] = useState(plan.name);
   254→   254→  const [priceMonthly, setPriceMonthly] = useState(String(plan.priceMonthly));
   255→   255→  const [priceYearly, setPriceYearly] = useState(String(plan.priceYearly));
   256→   256→  const [currency, setCurrency] = useState(plan.currency);
   257→   257→  const [interval, setInterval] = useState<'monthly' | 'yearly'>(plan.interval);

   259→   259→  const [features, setFeatures] = useState(plan.features.join('\n'));
   260→   260→  const [entitlements, setEntitlements] = useState<string[]>(plan.entitlements);
   261→   261→  const [limits, setLimits] = useState<PlanLimits>(plan.limits);
   262→   262→  const [clientDisplayOpen, setClientDisplayOpen] = useState(false);
   263→   263→
   264→   264→  // EditPlanDialog is conditionally rendered by the parent (mounted
   265→   265→  // fresh each time the user opens it), so the useState initializers
   266→   266→  // above already seed local state from the latest server snapshot.
   267→   267→  // No useEffect sync is needed — that would just trigger cascading

   270→   270→  const reset = () => {
   271→   271→    setName(plan.name);
   272→   272→    setPriceMonthly(String(plan.priceMonthly));
   273→   273→    setPriceYearly(String(plan.priceYearly));
   274→   274→    setCurrency(plan.currency);
   275→   275→    setInterval(plan.interval);
   276→   276→    setActive(plan.active);
   277→   277→    setFeatures(plan.features.join('\n'));
   278→   278→    setEntitlements(plan.entitlements);
   279→   279→    setLimits(plan.limits);
   280→   280→  };
   281→   281→
   282→   282→  const saveMutation = useMutation({
   283→   283→    mutationFn: (patch: PlanPatch) =>
   284→   284→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, patch),
   285→   285→    onSuccess: (data) => {
   286→   286→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   287→   287→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   288→   288→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   289→   289→      toast.success(`${data.name} updated — changes are now live for clients.`);
   290→   290→      onOpenChange(false);
   291→   291→    },
   292→   292→    onError: (err: unknown) => {
   293→   293→      const e = err as { error?: { message?: string }; message?: string };
   294→   294→      toast.error(
   295→   295→        e?.error?.message ?? e?.message ?? 'Unable to save plan. You may need OWNER access.',
   296→   296→      );
   297→   297→    },
   298→   298→  });
   299→   299→
   300→   300→  const buildPatch = (): PlanPatch => ({
   301→   301→    name,
   302→   302→    priceMonthly: Number(priceMonthly) || 0,
   303→   303→    priceYearly: Number(priceYearly) || 0,
   304→   304→    currency,
   305→   305→    interval,
   306→   306→    active,
   307→   307→    features: features
   308→   308→      .split('\n')
   309→   309→      .map((f) => f.trim())
   310→   310→      .filter(Boolean),
   311→   311→    entitlements,
   312→   312→    limits: { ...limits, storageBytes: Number(limits.storageBytes) || 0 },
   313→   313→  });
   314→   314→
   315→   315→  return (
   316→   316→    <Dialog open={open} onOpenChange={onOpenChange}>
   317→   317→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   318→   318→        <DialogHeader>
   319→   319→          <div className="flex items-center gap-2">
   320→   320→            <DialogTitle className="text-base">Edit {plan.name}</DialogTitle>
   321→   321→            <PlanBadge planId={getPlanBadgeId(plan.planId)} />
   322→   322→          </div>
   323→   323→          <DialogDescription className="text-xs">
   324→   324→            Single source of truth — changes propagate to the Client Billing page, MRR and
   325→   325→            server-side entitlement enforcement.
   326→   326→          </DialogDescription>
   327→   327→        </DialogHeader>
   328→   328→
   329→   329→        <div className="space-y-5">
   330→   330→          {/* -------------------- Basic Information -------------------- */}
   331→   331→          <section className="space-y-3">
   332→   332→            <h4 className="text-sm font-semibold">Basic Information</h4>
   333→   333→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   334→   334→              <div className="space-y-1">
   335→   335→                <Label className="text-xs">Name</Label>
   336→   336→                <Input
   337→   337→                  value={name}
   338→   338→                  onChange={(e) => setName(e.target.value)}
   339→   339→                  className="h-9"
   340→   340→                />
   341→   341→              </div>
   342→   342→              <div className="space-y-1">
   343→   343→                <Label className="text-xs">Monthly Price</Label>
   344→   344→                <Input
