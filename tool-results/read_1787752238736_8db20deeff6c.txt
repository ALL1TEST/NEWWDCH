   111→  setSearchValue: (value: string) => void;
   112→  toggleSort: (field: string) => void;

   115→  clearSelection: () => void;

   117→
   118→export function useDataTable(options: UseDataTableOptions = {}): UseDataTableReturn {
   119→  const {
   120→    initialPage = 1,
   121→    initialPageSize = DEFAULT_PAGE_SIZE,
   122→    initialSortField = 'createdAt',
   123→    initialSortOrder = 'desc',
   124→  } = options;
   125→
   126→  const [currentPage, setCurrentPage] = useState(initialPage);
   127→  const [pageSize, setPageSize] = useState(initialPageSize);
   128→  const [sortField, setSortFieldRaw] = useState(initialSortField);
   129→  const [sortOrder, setSortOrderRaw] = useState<'asc' | 'desc'>(initialSortOrder);
   130→  const [selectedIds, setSelectedIds] = useState<string[]>([]);
   131→  const [searchValue, setSearchValue] = useState('');
   132→
   133→  const setSortField = useCallback((field: string, order: 'asc' | 'desc') => {
   134→    setSortFieldRaw(field);
   135→    setSortOrderRaw(order);
   136→    setCurrentPage(1);
   137→  }, []);
   138→
   139→  const toggleSort = useCallback(
   140→    (field: string) => {
   141→      if (sortField === field) {
   142→        const next = sortOrder === 'asc' ? 'desc' : 'asc';
   143→        setSortFieldRaw(field);
   144→        setSortOrderRaw(next);
   145→      } else {
   146→        setSortFieldRaw(field);
   147→        setSortOrderRaw('asc');
   148→      }
   149→      setCurrentPage(1);
   150→    },
   151→    [sortField, sortOrder],
   152→  );
   153→
   154→  const toggleRowSelection = useCallback((id: string) => {
   155→    setSelectedIds((prev) =>
   156→      prev.includes(id) ? prev.filter((i) => i !== id) : [...prev, id],
   157→    );
   158→  }, []);
   159→
   160→  const toggleAllSelection = useCallback(
   161→    (allIds: string[], currentDataIds: string[]) => {
   162→      const allSelected = currentDataIds.every((id) => selectedIds.includes(id));
   163→      if (allSelected) {
   164→        setSelectedIds((prev) =>
   165→          prev.filter((id) => !currentDataIds.includes(id)),
   166→        );
   167→      } else {
   168→        const newIds = currentDataIds.filter((id) => !selectedIds.includes(id));
   169→        setSelectedIds((prev) => [...prev, ...newIds]);
   170→      }
   171→    },
   172→    [selectedIds],
   173→  );
   174→
   175→  const clearSelection = useCallback(() => {
   176→    setSelectedIds([]);
   177→  }, []);
   178→
   179→  return {
   180→    currentPage,
   181→    pageSize,
   182→    sortField,
   183→    sortOrder,
   184→    selectedIds,
   185→    searchValue,
   186→    setCurrentPage,
   187→    setPageSize,
   188→    setSortField,
   189→    setSelectedIds,
   190→    setSearchValue,
   191→    toggleSort,
   192→    toggleRowSelection,
   193→    toggleAllSelection,
   194→    clearSelection,
   195→  };
   196→}

   236→export class ColumnDefHelper {
   237→  static textColumn<TData>(options: TextColumnOptions<TData>): ColumnDef<TData> {
   238→    return {
   239→      id: options.id,
   240→      accessorKey: options.accessorKey,
   241→      accessorFn: options.accessorFn,
   242→      header: ({ column }) => {
   243→        const canSort = options.enableSorting !== false;
   244→        if (!canSort) {
   245→          return <span className="font-medium">{options.header}</span>;
   246→        }
   247→        return (
   248→          <button
   249→            className="flex items-center gap-1 hover:text-foreground transition-colors"
   250→            onClick={() => column.toggleSorting(column.getIsSorted() === 'asc')}
   251→          >
   252→            <span className="font-medium">{options.header}</span>
   253→            <SortIcon sorted={column.getIsSorted()} />
   254→          </button>
   255→        );
   256→      },
   257→      size: options.size,
   258→      enableSorting: options.enableSorting !== false,
   259→      enableHiding: options.enableHiding,
   260→      cell: ({ getValue }) => {
   261→        const value = getValue() as string;
   262→        if (!value) return <span className="text-muted-foreground">—</span>;
   263→        if (options.truncate && value.length > options.truncate) {
   264→          return (
   265→            <span className={cn('truncate block max-w-full', options.className)} title={value}>
   266→              {value.slice(0, options.truncate)}...
   267→            </span>
   268→          );
   269→        }
   270→        return <span className={options.className}>{value}</span>;
   271→      },
   272→    };

   498→// -------------------- Pagination --------------------
   499→
   500→function DataTablePagination({
   501→  currentPage,
   502→  pageSize,
   503→  totalItems,
   504→  onPageChange,
   505→  onPageSizeChange,
   506→}: {
   507→  currentPage: number;
   508→  pageSize: number;
   509→  totalItems: number;
   510→  onPageChange: (page: number) => void;
   511→  onPageSizeChange: (size: number) => void;
   512→}) {
   513→  const totalPages = Math.max(1, Math.ceil(totalItems / pageSize));
   514→  const rangeStart = totalItems === 0 ? 0 : (currentPage - 1) * pageSize + 1;
   515→  const rangeEnd = Math.min(currentPage * pageSize, totalItems);
   516→
   517→  return (
   518→    <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between px-2 py-3 border-t">
   519→      <div className="text-xs text-muted-foreground">
   520→        Showing {rangeStart}–{rangeEnd} of {totalItems} items
   521→      </div>
   522→      <div className="flex items-center gap-3">
   523→        <div className="flex items-center gap-1.5">
   524→          <span className="text-xs text-muted-foreground whitespace-nowrap">Rows</span>
   525→          <Select
   526→            value={String(pageSize)}
   527→            onValueChange={(v) => onPageSizeChange(Number(v))}
   528→          >
   529→            <SelectTrigger size="sm" className="w-[70px] h-8">
   530→              <SelectValue />
   531→            </SelectTrigger>
   532→            <SelectContent>
   533→              {PAGE_SIZES.map((size) => (
   534→                <SelectItem key={size} value={String(size)}>
   535→                  {size}
   536→                </SelectItem>
   537→              ))}
   538→            </SelectContent>
   539→          </Select>
   540→        </div>
   541→        <div className="flex items-center gap-1">
   542→          <Button
   543→            variant="outline"
   544→            size="icon"
   545→            className="h-8 w-8"
   546→            onClick={() => onPageChange(1)}
   547→            disabled={currentPage <= 1}
   548→          >
   549→            <ChevronsLeft className="h-4 w-4" />
   550→            <span className="sr-only">First page</span>
   551→          </Button>
   552→          <Button
   553→            variant="outline"
   554→            size="icon"
   555→            className="h-8 w-8"
   556→            onClick={() => onPageChange(currentPage - 1)}
   557→            disabled={currentPage <= 1}
   558→          >
   559→            <ChevronLeft className="h-4 w-4" />
   560→            <span className="sr-only">Previous page</span>
   561→          </Button>
   562→          <span className="text-xs text-muted-foreground min-w-[60px] text-center">
   563→            {currentPage} / {totalPages}
   564→          </span>
   565→          <Button
   566→            variant="outline"
   567→            size="icon"
   568→            className="h-8 w-8"
   569→            onClick={() => onPageChange(currentPage + 1)}
   570→            disabled={currentPage >= totalPages}
   571→          >
   572→            <ChevronRight className="h-4 w-4" />
   573→            <span className="sr-only">Next page</span>
   574→          </Button>
   575→          <Button
   576→            variant="outline"
   577→            size="icon"
   578→            className="h-8 w-8"
   579→            onClick={() => onPageChange(totalPages)}
   580→            disabled={currentPage >= totalPages}
   581→          >
   582→            <ChevronsRight className="h-4 w-4" />
   583→            <span className="sr-only">Last page</span>
   584→          </Button>
   585→        </div>
   586→      </div>
   587→    </div>
   588→  );
   589→}

   593→function TableToolbar<TData, TValue>({
   594→  searchPlaceholder,
   595→  searchValue,
   596→  onSearch,
   597→  selectedIds,
   598→  bulkActions,
   599→  onBulkAction,
   600→  onSelectionChange,
   601→  filterContent,
   602→}: {
   603→  searchPlaceholder?: string;
   604→  searchValue?: string;
   605→  onSearch?: (value: string) => void;
   606→  selectedIds?: string[];
   607→  bulkActions?: BulkAction[];
   608→  onBulkAction?: (action: string, ids: string[]) => void;
   609→  onSelectionChange?: (ids: string[]) => void;
   610→  filterContent?: React.ReactNode;
   611→}) {
   612→  const hasSelection = selectedIds && selectedIds.length > 0;
   613→
   614→  return (
   615→    <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between p-4 pb-2">
   616→      <div className="flex items-center gap-2 flex-1">
   617→        {(onSearch || searchValue !== undefined) && (
   618→          <div className="relative max-w-sm w-full">
   619→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   620→            <Input
   621→              placeholder={searchPlaceholder ?? 'Search...'}
   622→              value={searchValue ?? ''}
   623→              onChange={(e) => onSearch?.(e.target.value)}
   624→              className="pl-9 h-9"
   625→            />
   626→          </div>
   627→        )}
   628→        {filterContent}
   629→      </div>
   630→      {hasSelection && bulkActions && bulkActions.length > 0 && (
   631→        <div className="flex items-center gap-2">
   632→          <span className="text-xs text-muted-foreground">
   633→            {selectedIds.length} selected
   634→          </span>
   635→          <DropdownMenu>
   636→            <DropdownMenuTrigger asChild>
   637→              <Button variant="outline" size="sm">
   638→                <MoreHorizontal className="h-4 w-4 mr-1" />
   639→                Bulk Actions
   640→              </Button>
   641→            </DropdownMenuTrigger>
   642→            <DropdownMenuContent align="end">
   643→              {bulkActions.map((action) => (
   644→                <DropdownMenuItem
   645→                  key={action.label}
   646→                  variant={action.variant}
   647→                  disabled={action.disabled}
   648→                  onClick={() => onBulkAction?.(action.label, selectedIds)}
   649→                >
   650→                  {action.icon}
   651→                  {action.label}
   652→                </DropdownMenuItem>
   653→              ))}
   654→            </DropdownMenuContent>
   655→          </DropdownMenu>
   656→          <Button
   657→            variant="ghost"
   658→            size="sm"
   659→            onClick={() => onSelectionChange?.([])}
   660→          >
   661→            Clear
   662→          </Button>
   663→        </div>
   664→      )}
   665→    </div>
   666→  );
   667→}

   669→// -------------------- Main DataTable Component --------------------
   670→
   671→export function DataTable<TData, TValue>({
   672→  columns,
   673→  data,
   674→  isLoading = false,
   675→  totalItems,
   676→  pageSize,
   677→  currentPage,
   678→  onPageChange,
   679→  onSortChange,
   680→  sortField,
   681→  sortOrder,
   682→  onRowClick,
   683→  selectedIds = [],
   684→  onSelectionChange,
   685→  onBulkAction,
   686→  bulkActions,
   687→  emptyMessage = 'No data found.',
   688→  emptyIcon,
   689→  emptyState,
   690→  searchPlaceholder,
   691→  onSearch,
   692→  searchValue,
   693→  filterContent,
   694→  getRowId,
   695→  onPageSizeChange,
   696→}: DataTableProps<TData, TValue>) {
   697→  const hasRowSelection = !!onSelectionChange;
   698→  const columnCount = columns.length + (hasRowSelection ? 1 : 0);
   699→
   700→  const enhancedColumns = useMemo<ColumnDef<TData, TValue>[]>(() => {
   701→    if (!hasRowSelection) return columns;
   702→
   703→    const selectColumn: ColumnDef<TData, TValue> = {
   704→      id: '__select__',
   705→      size: 40,
   706→      enableSorting: false,
   707→      enableHiding: false,
   708→      header: ({ table }) => {
   709→        const allRowIds = table.getRowModel().rows.map((r) => r.id);
   710→        const allChecked =
   711→          allRowIds.length > 0 && allRowIds.every((id) => selectedIds.includes(id));
   712→        const someChecked = allRowIds.some((id) => selectedIds.includes(id));
   713→
   714→        return (
   715→          <Checkbox
   716→            checked={allChecked ? true : someChecked ? 'indeterminate' : false}
   717→            onCheckedChange={(checked) => {
   718→              if (checked) {
   719→                const allIds = data
   720→                  .map((row, idx) => getRowId?.(row) ?? String(idx))
   721→                  .filter((id) => !selectedIds.includes(id));
   722→                onSelectionChange?.([...selectedIds, ...allIds]);
   723→              } else {
   724→                const currentDataIds = data.map((row, idx) =>
   725→                  getRowId?.(row) ?? String(idx),
   726→                );
   727→                onSelectionChange?.(
   728→                  selectedIds.filter((id) => !currentDataIds.includes(id)),
   729→                );
   730→              }
   731→            }}
   732→            aria-label="Select all rows"
   733→          />
   734→        );
   735→      },
   736→      cell: ({ row }) => {
   737→        const rowId = row.id;
   738→        const isChecked = selectedIds.includes(rowId);
   739→
   740→        return (
   741→          <Checkbox
   742→            checked={isChecked}
   743→            onCheckedChange={(checked) => {
   744→              if (checked) {
   745→                onSelectionChange?.([...selectedIds, rowId]);
   746→              } else {
   747→                onSelectionChange?.(selectedIds.filter((id) => id !== rowId));
   748→              }
   749→            }}
   750→            onClick={(e) => e.stopPropagation()}
   751→            aria-label={`Select row ${rowId}`}
   752→          />
   753→        );
   754→      },
   755→    };
   756→
   757→    return [selectColumn, ...columns];
   758→  }, [columns, hasRowSelection, selectedIds, onSelectionChange, data, getRowId]);
   759→
   760→  const sorting: ColumnSort[] = useMemo(() => {
   761→    if (!sortField) return [];
   762→    return [{ id: sortField, desc: sortOrder === 'desc' }];
   763→  }, [sortField, sortOrder]);
   764→
   765→  const table = useReactTable({
   766→    data,
   767→    columns: enhancedColumns,
   768→    getCoreRowModel: getCoreRowModel(),
   769→    getSortedRowModel: getSortedRowModel(),
   770→    state: {
   771→      sorting,
   772→    },
   773→    onSortingChange: (updater) => {
   774→      const newSorting =
   775→        typeof updater === 'function' ? updater(sorting) : updater;
   776→      if (newSorting.length > 0) {
   777→        onSortChange(newSorting[0].id, newSorting[0].desc ? 'desc' : 'asc');
   778→      }
   779→    },
   780→    manualPagination: true,
   781→    manualSorting: true,
   782→    getRowId: getRowId
   783→      ? (row, index) => getRowId(row) ?? String(index)
   784→      : (_, index) => String(index),
   785→  });
   786→
   787→  const handlePageSizeChange = useCallback(
   788→    (newSize: number) => {
   789→      onPageSizeChange?.(newSize);
   790→      onPageChange(1);
   791→    },
   792→    [onPageChange, onPageSizeChange],
   793→  );
   794→
   795→  const showToolbar =
   796→    (onSearch || searchValue !== undefined) ||
   797→    (filterContent != null) ||
   798→    (selectedIds.length > 0 && bulkActions && bulkActions.length > 0);
   799→
   800→  return (
   801→    <div className="space-y-0">
   802→      {showToolbar && (
   803→        <TableToolbar
   804→          searchPlaceholder={searchPlaceholder}
   805→          searchValue={searchValue}
   806→          onSearch={onSearch}
   807→          selectedIds={selectedIds}
   808→          bulkActions={bulkActions}
   809→          onBulkAction={onBulkAction}
   810→          onSelectionChange={onSelectionChange}
   811→          filterContent={filterContent}
   812→        />
   813→      )}
   814→      <div className="rounded-lg border">
   815→        <Table>
   816→          <TableHeader>
   817→            {table.getHeaderGroups().map((headerGroup) => (
   818→              <TableRow key={headerGroup.id}>
   819→                {headerGroup.headers.map((header) => (
   820→                  <TableHead
   821→                    key={header.id}
   822→                    style={{ width: header.getSize() !== 150 ? header.getSize() : undefined }}
   823→                  >
   824→                    {flexRender(
   825→                      header.column.columnDef.header,
   826→                      header.getContext(),
   827→                    )}
   828→                  </TableHead>
   829→                ))}
   830→              </TableRow>
   831→            ))}
   832→          </TableHeader>
   833→          <TableBody>
   834→            {isLoading ? (
   835→              <SkeletonRows columnCount={columnCount} />
   836→            ) : data.length === 0 ? (
   837→              <DataTableEmpty icon={emptyIcon} message={emptyMessage} state={emptyState} />
   838→            ) : (
   839→              table.getRowModel().rows.map((row) => (
   840→                <TableRow
   841→                  key={row.id}
   842→                  data-state={selectedIds.includes(row.id) ? 'selected' : undefined}
   843→                  className={cn(onRowClick && 'cursor-pointer')}
   844→                  onClick={() => onRowClick?.(row.original)}
   845→                >
   846→                  {row.getVisibleCells().map((cell) => (
   847→                    <TableCell key={cell.id}>
   848→                      {flexRender(
   849→                        cell.column.columnDef.cell,
   850→                        cell.getContext(),
   851→                      )}
   852→                    </TableCell>
   853→                  ))}
   854→                </TableRow>
   855→              ))
   856→            )}
   857→          </TableBody>
   858→        </Table>
   859→      </div>
   860→      <DataTablePagination
   861→        currentPage={currentPage}
   862→        pageSize={pageSize}
   863→        totalItems={totalItems}
   864→        onPageChange={onPageChange}
   865→        onPageSizeChange={handlePageSizeChange}
   866→      />
   867→    </div>
   868→  );
   869→}
