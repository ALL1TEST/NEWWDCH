    85→// ==================== Formatters ====================
    86→
    87→function formatCompactNumber(n: number | null | undefined): string {
    88→  const num = n ?? 0;
    89→  if (num >= 1_000_000) return `${(num / 1_000_000).toFixed(1)}M`;
    90→  if (num >= 1_000) return `${(num / 1_000).toFixed(1)}K`;
    91→  return num.toLocaleString();
    92→}
    93→
    94→function formatCtr(n: number | null | undefined): string {
    95→  return `${((n ?? 0) * 100).toFixed(2)}%`;
    96→}
    97→
    98→function formatPosition(n: number | null | undefined): string {
    99→  return (n ?? 0).toFixed(1);
   100→}

   102→// ==================== Health Score Ring ====================
   103→
   104→function ScoreRing({ score }: { score: number }) {
   105→  const clampedScore = Math.max(0, Math.min(100, score));
   106→  const radius = 54;
   107→  const circumference = 2 * Math.PI * radius;
   108→  const offset = circumference - (clampedScore / 100) * circumference;
   109→  const color = clampedScore >= 80 ? 'text-green-500' : clampedScore >= 50 ? 'text-amber-500' : 'text-red-500';
   110→  const bgColor = clampedScore >= 80 ? 'bg-green-100 dark:bg-green-900/20' : clampedScore >= 50 ? 'bg-amber-100 dark:bg-amber-900/20' : 'bg-red-100 dark:bg-red-900/20';
   111→  const label = clampedScore >= 80 ? 'Healthy' : clampedScore >= 50 ? 'Needs Work' : 'Critical';
   112→  const labelColor = clampedScore >= 80 ? 'text-green-600 dark:text-green-400' : clampedScore >= 50 ? 'text-amber-600 dark:text-amber-400' : 'text-red-600 dark:text-red-400';
   113→
   114→  return (
   115→    <div className="flex flex-col items-center">
   116→      <div className={cn('relative flex items-center justify-center w-36 h-36 rounded-full', bgColor)}>
   117→        <svg className="w-full h-full -rotate-90" viewBox="0 0 120 120">
   118→          <circle cx="60" cy="60" r={radius} fill="none" stroke="currentColor" strokeWidth="6" className="text-border/40" />
   119→          <circle
   120→            cx="60" cy="60" r={radius} fill="none" strokeWidth="6" strokeLinecap="round"
   121→            className={color} strokeDasharray={circumference} strokeDashoffset={offset}
   122→            style={{ transition: 'stroke-dashoffset 1s ease-in-out' }}
   123→          />
   124→        </svg>
   125→        <div className="absolute flex flex-col items-center">
   126→          <span className={cn('text-3xl font-bold tabular-nums', color)}>{clampedScore}</span>
   127→          <span className="text-[10px] text-muted-foreground">/100</span>
   128→        </div>
   129→      </div>
   130→      <p className="text-sm font-semibold mt-3">SEO Health</p>
   131→      <span className={cn('text-xs font-medium mt-0.5', labelColor)}>{label}</span>
   132→    </div>
   133→  );
   134→}

   203→function RecentIssuesSummary({ issues, navigate }: { issues: SeoIssue[]; navigate: NavigateFn }) {
   204→  // Unresolved counts by severity
   205→  const unresolved = issues.filter((i) => !i.isResolved);
   206→  const criticalCount = unresolved.filter((i) => i.severity === 'CRITICAL').length;
   207→  const warningCount = unresolved.filter((i) => i.severity === 'WARNING').length;
   208→  const infoCount = unresolved.filter((i) => i.severity === 'INFO').length;
   209→
   210→  // Only the 3-5 most recent issues
   211→  const recent = issues.slice(0, 5);
   212→
   213→  return (
   214→    <div className="space-y-4">
   215→      {/* Severity summary */}
   216→      <div className="flex flex-wrap items-center gap-2 text-xs">
   217→        <span className="inline-flex items-center gap-1.5 px-2.5 py-1 rounded-md bg-red-100 dark:bg-red-900/30 text-red-700 dark:text-red-400 font-medium">
   218→          <AlertTriangle className="h-3.5 w-3.5" />
   219→          {criticalCount} Critical
   220→        </span>
   221→        <span className="text-muted-foreground/60">·</span>
   222→        <span className="inline-flex items-center gap-1.5 px-2.5 py-1 rounded-md bg-amber-100 dark:bg-amber-900/30 text-amber-700 dark:text-amber-400 font-medium">
   223→          <AlertTriangle className="h-3.5 w-3.5" />
   224→          {warningCount} Warnings
   225→        </span>
   226→        <span className="text-muted-foreground/60">·</span>
   227→        <span className="inline-flex items-center gap-1.5 px-2.5 py-1 rounded-md bg-sky-100 dark:bg-sky-900/30 text-sky-700 dark:text-sky-400 font-medium">
   228→          <Info className="h-3.5 w-3.5" />
   229→          {infoCount} Info
   230→        </span>
   231→      </div>
   232→
   233→      {recent.length > 0 ? (
   234→        <ul className="space-y-2">
   235→          {recent.map((issue) => {
   236→            const sev = SEVERITY_CONFIG[issue.severity] ?? SEVERITY_CONFIG.INFO;
   237→            const SevIcon = sev.icon;
   238→            const fullUrl = issue.pageUrl.startsWith('http') ? issue.pageUrl : `https://cms.example.com${issue.pageUrl}`;
   239→            return (
   240→              <li
   241→                key={issue.id}
   242→                className="flex items-start gap-3 p-3 rounded-lg border bg-card hover:bg-muted/30 transition-colors"
   243→              >
   244→                <Badge
   245→                  variant="outline"
   246→                  className={cn('border-transparent font-medium gap-1 shrink-0 mt-0.5', sev.bg, sev.color)}
   247→                >
   248→                  <SevIcon className="h-3 w-3" />
   249→                  {issue.severity.charAt(0) + issue.severity.slice(1).toLowerCase()}
   250→                </Badge>
   251→                <div className="min-w-0 flex-1 space-y-1">
   252→                  <p className="text-sm font-medium text-foreground leading-snug">
   253→                    {issue.problem}
   254→                  </p>
   255→                  <a
   256→                    href={fullUrl}
   257→                    target="_blank"
   258→                    rel="noopener noreferrer"
   259→                    className="inline-flex items-center gap-1 font-mono text-xs text-primary hover:underline"
   260→                    title={issue.pageUrl}
   261→                  >
   262→                    <span className="truncate max-w-[280px]">{issue.pageUrl}</span>
   263→                    <ExternalLink className="h-2.5 w-2.5 opacity-50 shrink-0" />
   264→                  </a>
   265→                </div>
   266→              </li>
   267→            );
   268→          })}
   269→        </ul>
   270→      ) : (
   271→        <div className="flex flex-col items-center justify-center py-8 text-center">
   272→          <div className="mb-3 text-muted-foreground/30">
   273→            <Shield className="h-10 w-10" strokeWidth={1.5} />
   274→          </div>
   275→          <p className="text-sm font-medium text-foreground">No SEO issues found</p>
   276→          <p className="text-xs text-muted-foreground mt-1">
   277→            Run an audit to detect potential SEO issues.
   278→          </p>
   279→          <Button
   280→            size="sm"
   281→            variant="outline"
   282→            className="mt-3 text-xs"
   283→            onClick={() => navigate('seo', null, 'audit')}
   284→          >
   285→            <ClipboardCheck className="h-3.5 w-3.5 mr-1.5" />
   286→            Run SEO Audit
   287→            <ArrowUpRight className="h-3.5 w-3.5 ml-1" />
   288→          </Button>
   289→        </div>
   290→      )}
   291→
   292→      {recent.length > 0 && (
   293→        <div className="flex justify-end">
   294→          <Button
   295→            variant="ghost"
   296→            size="sm"
   297→            className="text-xs text-muted-foreground"
   298→            onClick={() => navigate('seo', null, 'audit')}
   299→          >
   300→            View All Issues
   301→            <ArrowUpRight className="h-3.5 w-3.5 ml-1" />
   302→          </Button>
   303→        </div>
   304→      )}
   305→    </div>
   306→  );
   307→}
