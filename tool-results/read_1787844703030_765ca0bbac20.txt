   292→   292→ * Collapsed-rail logo cell. Since the extra collapse-toggle icon was removed
   293→   293→ * from below the logo, THIS cell is the explicit expand control:
   294→   294→ *   • identical 32×32 grid geometry as every other rail icon
   295→   295→ *   • click toggles the sidebar open (SidebarRail edge strip still works)
   296→   296→ * No new icon is introduced and no empty clickable ghost remains.
   297→   297→ *
   298→   298→ * HOVER BEHAVIOR (collapsed state only — this button is rendered ONLY
   299→   299→ * inside the collapsed-rail cluster, so the expanded LogoMark is untouched):
   300→   300→ *   • at rest ............ shows the "C" logo mark on its bg-primary black
   301→   301→ *                           box (NORMAL LOGO — background UNCHANGED, same
   302→   302→ *                           as the expanded LogoMark).
   303→   303→ *   • on mouse-enter ..... the "C" is replaced by a PanelLeftOpen icon AND
   304→   304→ *                           the button's background becomes transparent so
   305→   305→ *                           ONLY the Expand icon is visible with no box
   306→   306→ *                           behind it. The icon color switches to
   307→   307→ *                           muted-foreground (gray) so it stays visible
   308→   308→ *                           against the transparent/page bg in both Light
   309→   309→ *                           and Dark mode.
   310→   310→ *   • on mouse-leave ..... restores the "C" logo mark AND its bg-primary box.
   311→   311→ *   • click ............. calls toggleSidebar (existing functionality, unchanged).
   312→   312→ *
   313→   313→ * Background is removed ONLY from the temporary Expand-icon state; the
   314→   314→ * normal logo (at rest) keeps its bg-primary background exactly as-is.
   315→   315→ *
   316→   316→ * Implementation: React state (useState + onMouseEnter/onMouseLeave) — NOT
   317→   317→ * CSS :hover / group-hover variants. This is deliberate: Tailwind v4 wraps
   318→   318→ * its hover:/group-hover: utilities inside @media (hover: hover), so on
   319→   319→ * any browser that reports (hover: none) (headless, touch, some preview
   320→   320→ * iframes) those utilities NEVER activate and the swap + bg-transparent
   321→   321→ * would silently fail. React mouse events fire on ANY pointer input
   322→   322→ * regardless of the hover media query, so the behavior is identical in
   323→   323→ * every environment. The conditional className + conditional render swap
   324→   324→ * the bg/color/glyph atomically per state. Works in both Light and Dark.
   325→   325→ */
   326→   326→function CollapsedLogoButton() {
   327→   327→  const { toggleSidebar } = useSidebar();
   328→   328→  // Local hover state drives the logo↔Expand-icon swap AND the per-state
   329→   329→  // background/color. Using React state (not CSS :hover / group-hover)
   330→   330→  // guarantees the swap works in ALL environments — including browsers
   331→   331→  // that report (hover: none), where Tailwind's @media (hover: hover)-
   332→   332→  // gated hover:/group-hover: variants would never activate.
   333→   333→  const [hovered, setHovered] = useState(false);
   334→   334→  return (
   335→   335→    /* Hover-only "Expand" tooltip: instant (provider delayDuration=0),
   336→   336→       appears on mouse-enter, disappears on mouse-leave. Radix closes it
   337→   337→       on any pointer exit — disableHoverableContent guarantees it can
   338→   338→       never linger or trap the pointer. */
   339→   339→    <Tooltip disableHoverableContent>
   340→   340→      <TooltipTrigger asChild>
   341→   341→        <button
   342→   342→          type="button"
   343→   343→          onClick={toggleSidebar}
   344→   344→          aria-label="Expand sidebar"
   345→   345→          onMouseEnter={() => setHovered(true)}
   346→   346→          onMouseLeave={() => setHovered(false)}
   347→   347→          className={cn(
   348→   348→            'flex h-8 w-8 shrink-0 cursor-pointer items-center justify-center rounded-lg font-bold text-sm outline-none transition-opacity hover:opacity-90 focus-visible:ring-2 focus-visible:ring-ring select-none',
   349→   349→            // NORMAL LOGO STATE (at rest): bg-primary black box +
   350→   350→            // text-primary-foreground white "C" — background UNCHANGED,
   351→   351→            // same as the expanded LogoMark.
   352→   352→            // EXPAND-ICON STATE (on hover): bg-transparent (NO background)
   353→   353→            // + text-muted-foreground gray icon — ONLY this temporary
   354→   354→            // state loses the background. The icon stays visible (gray
   355→   355→            // on transparent/page bg) in both Light and Dark mode.
   356→   356→            hovered
   357→   357→              ? 'bg-transparent text-muted-foreground'
   358→   358→              : 'bg-primary text-primary-foreground',
   359→   359→          )}
   360→   360→        >
   361→   361→          {/* Conditional render: "C" logo at rest, PanelLeftOpen icon on
   362→   362→              hover. h-4 w-4 matches the CollapseToggle's PanelLeftClose
   363→   363→              icon size; the icon is the visual opposite of the collapse
   364→   364→              icon so the collapsed logo reads as a clear "expand"
   365→   365→              affordance when the pointer is over it. */}
   366→   366→          {hovered ? (
   367→   367→            <PanelLeftOpen className="h-4 w-4" />
   368→   368→          ) : (
   369→   369→            <span>C</span>
   370→   370→          )}
   371→   371→          <span className="sr-only">Expand sidebar</span>
   372→   372→        </button>
   373→   373→      </TooltipTrigger>
   374→   374→      <TooltipContent side="right" sideOffset={8}>Expand</TooltipContent>
   375→   375→    </Tooltip>
   376→   376→  );
   377→   377→}

   447→   447→/**
   448→   448→ * Settings parent item rendered on the COLLAPSED 48px rail.
   449→   449→ * Clicking it opens a portal-based popover to the RIGHT of the icon —
   450→   450→ * the sidebar itself NEVER expands, and the popover anchors to this
   451→   451→ * exact button so the "star/sparkle alignment" grid stays intact.
   452→   452→ */
   453→   453→function CollapsedParentNavItem({
   454→   454→  item,
   455→   455→  currentModule,
   456→   456→  currentSubPage,
   457→   457→}: {
   458→   458→  item: NavItem;
   459→   459→  currentModule: string;
   460→   460→  currentSubPage: string | null;
   461→   461→}) {
   462→   462→  const [floatOpen, setFloatOpen] = useState(false);
   463→   463→  const mod = hrefToModule(item.href);
   464→   464→  const isActive = currentModule === mod;
   465→   465→
   466→   466→  const handleChildNavigate = (
   467→   467→    e: React.MouseEvent<HTMLAnchorElement>,
   468→   468→    child: NavItem,
   469→   469→  ) => {
   470→   470→    e.preventDefault();
   471→   471→    e.stopPropagation();
   472→   472→    setFloatOpen(false); // close the floating submenu after navigation…
   473→   473→    // …but do NOT touch sidebar expansion state (rule #7).
   474→   474→    const hash = child.href.replace(/^#/, '');
   475→   475→    const parts = hash.split('/');
   476→   476→    const childSubPage = parts[1] || null;
   477→   477→    useNavigationStore.getState().navigate(parts[0], null, childSubPage);
   478→   478→  };
   479→   479→
   480→   480→  return (
   481→   481→    <SidebarMenuItem>
   482→   482→      <Popover open={floatOpen} onOpenChange={setFloatOpen}>
   483→   483→        <PopoverTrigger asChild>
   484→   484→          <SidebarMenuButton
   485→   485→            isActive={isActive || floatOpen}
   486→   486→            aria-expanded={floatOpen}
   487→   487→            aria-haspopup="menu"
   488→   488→            tooltip={floatOpen ? undefined : item.label}
   489→   489→            onClick={(e: React.MouseEvent) => {
   490→   490→              e.preventDefault();
   491→   491→              setFloatOpen((o) => !o);
   492→   492→            }}
   493→   493→          >
   494→   494→            <NavIcon name={item.icon} />
   495→   495→            <span>{item.label}</span>
   496→   496→          </SidebarMenuButton>
   497→   497→        </PopoverTrigger>
   498→   498→        {/* Portal → document.body: immune to sidebar overflow clipping.
   499→   499→            Radix handles Esc + outside-click closing and re-anchors at any

   666→   666→function NavGroupSection({
   667→   667→  group,
   668→   668→  currentModule,
   669→   669→  currentSubPage,
   670→   670→  openSection,
   671→   671→  onToggleSection,
   672→   672→}: {
   673→   673→  group: NavGroup;
   674→   674→  currentModule: string;
   675→   675→  currentSubPage: string | null;

   689→   689→              /* Collapsed rail → parents with children become floating
   690→   690→                 popovers anchored to their own icon. The rail NEVER
   691→   691→                 auto-expands (#4/#5/#6). */
   692→   692→              return (

   698→   698→                />

   700→   700→            }
   701→   701→
   702→   702→            if (hasChildren) {

   704→   704→                <ExpandableNavItem
   705→   705→                  key={item.label}
   706→   706→                  item={item}
