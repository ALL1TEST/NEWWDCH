     4→// PLATFORM PLANS & PRICING — simplified admin UI.
     5→// ============================================================
     6→// Owner sees compact plan cards (Free / Plus / Pro / Max …) with a
     7→// quick Active toggle, a compact price + features + limits summary,
     8→// and an "Edit Plan" button. The Edit Plan modal has ONE base price
     9→// configuration per plan — NO per-currency matrix:
    10→//
    11→//   Basic Information: Name · Default/Fallback Currency (country/
    12→//   currency selector: 🇺🇸 United States — USD — $) · Auto Currency
    13→//   toggle · Monthly/Yearly Price (only for enabled periods) · Billing

    15→//   then: Stripe Billing (sync + status) · Feature Access · Usage

    18→// AUTO CURRENCY: each plan carries an autoCurrency flag. When ON,
    19→// the customer's currency is resolved SERVER-SIDE from their IP
    20→// (country → currency → plan price for that currency → fallback to
    21→// the plan default when unsupported) — the admin never picks the
    22→// customer's currency. When OFF, every customer is billed in the
    23→// plan's default currency. Per-currency prices (e.g. 90 MAD) come
    24→// from the PLATFORM-level regional config (CountryPricing), not
    25→// from this modal.
    26→//
    27→// All writes go through /api/platform/admin/plans (GET list,
    28→// POST create) + /api/platform/admin/plans/[planId] (PUT update)
    29→// — owner-only — the same shared plan-config cache the Client
    30→// Billing page and MRR read. Server-side entitlement enforcement
    31→// (hasFeature) and usage-limit checks (checkLimit) consume the

   222→/** Count the editor-visible enabled features (the 10 checkboxes —
   223→ *  Platform AI and Client's Own AI API each count as one feature). */
   224→function countEditorFeatures(entitlements: readonly string[]): number {
   225→  return PLAN_EDITOR_FEATURE_KEYS.filter((k) => entitlements.includes(k)).length;
   226→}
   227→
   228→const EDITOR_FEATURE_TOTAL = PLAN_EDITOR_FEATURE_KEYS.length; // 10
   229→
   230→/** Toggle one feature key in the entitlements state — with the ONE
   231→ *  inter-feature dependency rule applied interactively:
   232→ *  - enabling API Access auto-enables Client's Own AI API (required),
   233→ *  - disabling Client's Own AI API auto-disables API Access.
   234→ *  Platform AI is fully independent of both. */
   235→function toggleFeature(cur: string[], key: string): string[] {
   236→  const next = cur.includes(key) ? cur.filter((k) => k !== key) : [...cur, key];
   237→  if (key === 'api_access' && next.includes('api_access') && !next.includes('ai_client')) {
   238→    // Enabling API Access → auto-enable its dependency.
   239→    return [...next, 'ai_client'];
   240→  }
   241→  if (key === 'ai_client' && !next.includes('ai_client') && next.includes('api_access')) {
   242→    // Disabling Client's Own AI API → API Access loses its dependency.
   243→    return next.filter((k) => k !== 'api_access');
   244→  }
   245→  return next;
   246→}

   388→/** Format the LARGE primary price on a plan card — integer amounts.
   389→ *  e.g. (90, 'MAD') → "90 MAD"; (0, 'CHF') → "CHF 0". */
   390→function formatPriceSymbol(amount: number, currency: string): string {

   397→function formatPriceSymbolMonthlyEquiv(amount: number, currency: string): string {
   398→  return formatMoney(amount, currency, 2);

   527→              </span>
   528→            </div>
   529→          ) : effectiveInterval === 'yearly' && !plan.billingMonthly ? (
   530→            // YEARLY-ONLY plan — there is no monthly option, so the
   531→            // yearly total IS the primary price ("$90 / year"). No
   532→            // monthly-equivalent is shown (that would advertise a
   533→            // monthly option the plan does not offer).
   534→            <div className="flex flex-wrap items-baseline gap-2">
   535→              <span className="shrink-0 whitespace-nowrap text-4xl font-semibold leading-none tracking-tight text-foreground">
   536→                {formatPriceSymbol(plan.priceYearly, plan.currency)}
   537→              </span>
   538→              <span className="whitespace-nowrap text-sm text-muted-foreground">/ year</span>
   539→            </div>
   540→          ) : effectiveInterval === 'yearly' ? (
   541→            // Both periods — INLINE: [LARGE $X.XX] [small / month]
   542→            // [small $X / year]. The monthly equivalent (priceYearly /
   543→            // 12) is the dominant large price; the real yearly total
   544→            // stays small/muted beside it.
   545→            //   e.g. Plus →  $7.50 / month   $90 / year
   546→            //        Pro  →  $40.83 / month  $490 / year
   547→            //        Max  →  $82.50 / month  $990 / year
   548→            <div className="flex flex-wrap items-baseline gap-2">
   549→              <span className="shrink-0 whitespace-nowrap text-4xl font-semibold leading-none tracking-tight text-foreground">
   550→                {formatPriceSymbolMonthlyEquiv(plan.priceYearly / 12, plan.currency)}
   551→              </span>
   552→              <span className="whitespace-nowrap text-sm text-muted-foreground">/ month</span>
   553→              <span className="shrink-0 whitespace-nowrap text-sm text-muted-foreground">
   554→                {formatPriceSymbol(plan.priceYearly, plan.currency)} / year
   555→              </span>
   556→            </div>
   557→          ) : (
   558→            <div className="flex flex-wrap items-baseline gap-2">
   559→              <span className="shrink-0 whitespace-nowrap text-4xl font-semibold leading-none tracking-tight text-foreground">
   560→                {formatPriceSymbol(plan.priceMonthly, plan.currency)}
   561→              </span>
   562→              <span className="whitespace-nowrap text-sm text-muted-foreground">/ month</span>
   563→            </div>
   564→          )}
   565→        </div>
   566→
   567→        {/* Thin horizontal divider between the price section and the features section.
   568→             Equidistant from the price above and the feature list below (mt-6 both
   569→             sides) for a consistent, compact vertical rhythm on every card. The

   571→             the price row + plan name above (flush to the content left) while its
   572→             right edge extends into the card padding to match the price row —
   573→             identical in both modes. */}
   574→        <div className="-mr-6 mt-6 h-px bg-border" aria-hidden />
   575→
   576→        {/* Feature items (section label omitted). Same typography on every card:
   577→             text-[15px] leading-normal so features read as a clear medium-size
   578→             block — visibly subordinate to the text-5xl plan name and text-4xl
   579→             price, never competing with them. Check icon is h-4 w-4 with mt-0.5
   580→             so its optical center aligns with the first line of feature text;
   581→             gap-2.5 keeps a consistent icon↔text column gutter; items-start
   582→             pins the icon to the first line even when a feature wraps to 2
   583→             lines, so rows never look vertically misaligned. All cards share
   584→             the card's p-8 left padding, so every feature column starts at the
   585→             same x. */}
   586→        <div className="mt-6">
   587→          <ul className="space-y-2.5">
   588→            {featureItems.length === 0 ? (
   589→              <li className="text-[15px] leading-normal text-muted-foreground">No features configured</li>
   590→            ) : (
   591→              featureItems.map((f, i) => (
   592→                <li key={`${f}-${i}`} className="flex items-start gap-2.5 text-[15px] leading-normal text-card-foreground">
   593→                  <Check className="mt-0.5 h-4 w-4 shrink-0 text-muted-foreground" />
   594→                  <span>{f}</span>
   595→                </li>
   596→              ))
   597→            )}
   598→          </ul>
   599→        </div>
   600→
   601→        {/* Bottom row — Edit Plan (left, flex-1) + quick active toggle
   602→             (right, shrink-0), pinned to the card bottom. The toggle used
   603→             to live in the header's top-right; it was relocated here so the
   604→             name + price row above can use the full card width and the
   605→             price can sit on the same line as the name. The Edit button
   606→             changed from w-full to flex-1 to share the row with the toggle.
   607→             The toggle's logic (auto-save PUT to /api/platform/admin/plans/
   608→             [planId] with { active }) is unchanged. */}
   609→        <div className="mt-auto flex items-center gap-3 pt-8">
   610→          <Button
   611→            variant="outline"
   612→            size="sm"
   613→            className="flex-1 rounded-full"
   614→            disabled={!canEdit}
   615→            onClick={() => setEditing(true)}
   616→          >
   617→            <Pencil className="h-4 w-4 mr-2" />
   618→            Edit Plan
   619→          </Button>
   620→          <div className="flex shrink-0 items-center gap-2">
   621→            <Label
   622→              htmlFor={`active-${plan.planId}`}
   623→              className="cursor-pointer text-xs text-muted-foreground"
   624→            >
   625→              {plan.active ? 'Active' : 'Inactive'}
   626→            </Label>
   627→            <Switch
   628→              id={`active-${plan.planId}`}
   629→              checked={plan.active}
   630→              disabled={!canEdit || activeMutation.isPending}
   631→              onCheckedChange={(v) => activeMutation.mutate(v)}
   632→            />
   633→          </div>
   634→        </div>
   635→        {!canEdit && (
   636→          <p className="mt-2 text-center text-[10px] text-muted-foreground">
   637→            Editing is restricted to the OWNER role.
   638→          </p>
   639→        )}
   640→      </div>
   641→
   642→      {editing && (
   643→        <EditPlanDialog plan={plan} open={editing} onOpenChange={setEditing} />
   644→      )}
   645→    </>
   646→  );
   647→}
