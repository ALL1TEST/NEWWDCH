   127→   127→export function useDataTable(options: UseDataTableOptions = {}): UseDataTableReturn {
   128→   128→  const {
   129→   129→    initialPage = 1,
   130→   130→    initialPageSize = DEFAULT_PAGE_SIZE,
   131→   131→    initialSortField = 'createdAt',
   132→   132→    initialSortOrder = 'desc',
   133→   133→  } = options;
   134→   134→
   135→   135→  const [currentPage, setCurrentPage] = useState(initialPage);
   136→   136→  const [pageSize, setPageSize] = useState(initialPageSize);
   137→   137→  const [sortField, setSortFieldRaw] = useState(initialSortField);
   138→   138→  const [sortOrder, setSortOrderRaw] = useState<'asc' | 'desc'>(initialSortOrder);
   139→   139→  const [selectedIds, setSelectedIds] = useState<string[]>([]);
   140→   140→  const [searchValue, setSearchValue] = useState('');
   141→   141→
   142→   142→  const setSortField = useCallback((field: string, order: 'asc' | 'desc') => {
   143→   143→    setSortFieldRaw(field);
   144→   144→    setSortOrderRaw(order);
   145→   145→    setCurrentPage(1);
   146→   146→  }, []);
   147→   147→
   148→   148→  const toggleSort = useCallback(
   149→   149→    (field: string) => {
   150→   150→      if (sortField === field) {
   151→   151→        const next = sortOrder === 'asc' ? 'desc' : 'asc';
   152→   152→        setSortFieldRaw(field);
   153→   153→        setSortOrderRaw(next);
   154→   154→      } else {
   155→   155→        setSortFieldRaw(field);
   156→   156→        setSortOrderRaw('asc');
   157→   157→      }
   158→   158→      setCurrentPage(1);
   159→   159→    },
   160→   160→    [sortField, sortOrder],
   161→   161→  );
   162→   162→
   163→   163→  const toggleRowSelection = useCallback((id: string) => {
   164→   164→    setSelectedIds((prev) =>
   165→   165→      prev.includes(id) ? prev.filter((i) => i !== id) : [...prev, id],
   166→   166→    );
   167→   167→  }, []);
   168→   168→
   169→   169→  const toggleAllSelection = useCallback(
   170→   170→    (allIds: string[], currentDataIds: string[]) => {
   171→   171→      const allSelected = currentDataIds.every((id) => selectedIds.includes(id));
   172→   172→      if (allSelected) {
   173→   173→        setSelectedIds((prev) =>
   174→   174→          prev.filter((id) => !currentDataIds.includes(id)),
   175→   175→        );
   176→   176→      } else {
   177→   177→        const newIds = currentDataIds.filter((id) => !selectedIds.includes(id));
   178→   178→        setSelectedIds((prev) => [...prev, ...newIds]);
   179→   179→      }
   180→   180→    },
   181→   181→    [selectedIds],
   182→   182→  );
   183→   183→
   184→   184→  const clearSelection = useCallback(() => {
   185→   185→    setSelectedIds([]);
   186→   186→  }, []);
   187→   187→
   188→   188→  return {
   189→   189→    currentPage,
   190→   190→    pageSize,
   191→   191→    sortField,
   192→   192→    sortOrder,
   193→   193→    selectedIds,
   194→   194→    searchValue,
   195→   195→    setCurrentPage,
   196→   196→    setPageSize,
   197→   197→    setSortField,
   198→   198→    setSelectedIds,
   199→   199→    setSearchValue,
   200→   200→    toggleSort,
   201→   201→    toggleRowSelection,
   202→   202→    toggleAllSelection,
   203→   203→    clearSelection,
   204→   204→  };
   205→   205→}

   509→   509→function DataTablePagination({
   510→   510→  currentPage,
   511→   511→  pageSize,
   512→   512→  totalItems,
   513→   513→  onPageChange,
   514→   514→  onPageSizeChange,
   515→   515→}: {
   516→   516→  currentPage: number;
   517→   517→  pageSize: number;
   518→   518→  totalItems: number;
   519→   519→  onPageChange: (page: number) => void;
   520→   520→  onPageSizeChange: (size: number) => void;
   521→   521→}) {
   522→   522→  const totalPages = Math.max(1, Math.ceil(totalItems / pageSize));
   523→   523→  const rangeStart = totalItems === 0 ? 0 : (currentPage - 1) * pageSize + 1;
   524→   524→  const rangeEnd = Math.min(currentPage * pageSize, totalItems);
   525→   525→
   526→   526→  return (
   527→   527→    <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between px-4 py-3 border-t">
   528→   528→      <div className="text-xs text-muted-foreground">
   529→   529→        Showing {rangeStart}–{rangeEnd} of {totalItems} items
   530→   530→      </div>
   531→   531→      <div className="flex items-center gap-3">
   532→   532→        <div className="flex items-center gap-1.5">
   533→   533→          <span className="text-xs text-muted-foreground whitespace-nowrap">Rows</span>
   534→   534→          <Select
   535→   535→            value={String(pageSize)}
   536→   536→            onValueChange={(v) => onPageSizeChange(Number(v))}
   537→   537→          >
   538→   538→            <SelectTrigger size="sm" className="w-[70px] h-8">
   539→   539→              <SelectValue />
   540→   540→            </SelectTrigger>
   541→   541→            <SelectContent>
   542→   542→              {PAGE_SIZES.map((size) => (
   543→   543→                <SelectItem key={size} value={String(size)}>
   544→   544→                  {size}
   545→   545→                </SelectItem>
   546→   546→              ))}
   547→   547→            </SelectContent>
   548→   548→          </Select>
   549→   549→        </div>
   550→   550→        <div className="flex items-center gap-1">
   551→   551→          <Button
   552→   552→            variant="outline"
   553→   553→            size="icon"
   554→   554→            className="h-8 w-8"
   555→   555→            onClick={() => onPageChange(1)}
   556→   556→            disabled={currentPage <= 1}
   557→   557→          >
   558→   558→            <ChevronsLeft className="h-4 w-4" />
   559→   559→            <span className="sr-only">First page</span>
   560→   560→          </Button>
   561→   561→          <Button
   562→   562→            variant="outline"
   563→   563→            size="icon"
   564→   564→            className="h-8 w-8"
   565→   565→            onClick={() => onPageChange(currentPage - 1)}
   566→   566→            disabled={currentPage <= 1}
   567→   567→          >
   568→   568→            <ChevronLeft className="h-4 w-4" />
   569→   569→            <span className="sr-only">Previous page</span>
   570→   570→          </Button>
   571→   571→          <span className="text-xs text-muted-foreground min-w-[60px] text-center">
   572→   572→            {currentPage} / {totalPages}
   573→   573→          </span>
   574→   574→          <Button
   575→   575→            variant="outline"
   576→   576→            size="icon"
   577→   577→            className="h-8 w-8"
   578→   578→            onClick={() => onPageChange(currentPage + 1)}
   579→   579→            disabled={currentPage >= totalPages}
   580→   580→          >
   581→   581→            <ChevronRight className="h-4 w-4" />
   582→   582→            <span className="sr-only">Next page</span>
   583→   583→          </Button>
   584→   584→          <Button
   585→   585→            variant="outline"
   586→   586→            size="icon"
   587→   587→            className="h-8 w-8"
   588→   588→            onClick={() => onPageChange(totalPages)}
   589→   589→            disabled={currentPage >= totalPages}
   590→   590→          >
   591→   591→            <ChevronsRight className="h-4 w-4" />
   592→   592→            <span className="sr-only">Last page</span>
   593→   593→          </Button>
   594→   594→        </div>
   595→   595→      </div>
   596→   596→    </div>
   597→   597→  );
   598→   598→}

   602→   602→function TableToolbar<TData, TValue>({
   603→   603→  searchPlaceholder,
   604→   604→  searchValue,
   605→   605→  onSearch,
   606→   606→  selectedIds,
   607→   607→  bulkActions,
   608→   608→  onBulkAction,
   609→   609→  onSelectionChange,
   610→   610→  filterContent,
   611→   611→}: {
   612→   612→  searchPlaceholder?: string;
   613→   613→  searchValue?: string;
   614→   614→  onSearch?: (value: string) => void;
   615→   615→  selectedIds?: string[];
   616→   616→  bulkActions?: BulkAction[];
   617→   617→  onBulkAction?: (action: string, ids: string[]) => void;
   618→   618→  onSelectionChange?: (ids: string[]) => void;
   619→   619→  filterContent?: React.ReactNode;
   620→   620→}) {
   621→   621→  const hasSelection = selectedIds && selectedIds.length > 0;
   622→   622→
   623→   623→  return (
   624→   624→    <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between p-4 pb-2">
   625→   625→      <div className="flex items-center gap-2 flex-1">
   626→   626→        {(onSearch || searchValue !== undefined) && (
   627→   627→          <div className="relative max-w-sm w-full">
   628→   628→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   629→   629→            <Input
   630→   630→              placeholder={searchPlaceholder ?? 'Search...'}
   631→   631→              value={searchValue ?? ''}
   632→   632→              onChange={(e) => onSearch?.(e.target.value)}
   633→   633→              className="pl-9 h-9"
   634→   634→            />
   635→   635→          </div>
   636→   636→        )}
   637→   637→        {filterContent}
   638→   638→      </div>
   639→   639→      {hasSelection && bulkActions && bulkActions.length > 0 && (
   640→   640→        <div className="flex items-center gap-2">
   641→   641→          <span className="text-xs text-muted-foreground">
   642→   642→            {selectedIds.length} selected
   643→   643→          </span>
   644→   644→          <DropdownMenu>
   645→   645→            <DropdownMenuTrigger asChild>
   646→   646→              <Button variant="outline" size="sm">
   647→   647→                <MoreHorizontal className="h-4 w-4 mr-1" />
   648→   648→                Bulk Actions
   649→   649→              </Button>
   650→   650→            </DropdownMenuTrigger>
   651→   651→            <DropdownMenuContent align="end">
   652→   652→              {bulkActions.map((action) => (
   653→   653→                <DropdownMenuItem
   654→   654→                  key={action.label}
   655→   655→                  variant={action.variant}
   656→   656→                  disabled={action.disabled}
   657→   657→                  onClick={() => onBulkAction?.(action.label, selectedIds)}
   658→   658→                >
   659→   659→                  {action.icon}
   660→   660→                  {action.label}
   661→   661→                </DropdownMenuItem>
   662→   662→              ))}
   663→   663→            </DropdownMenuContent>
   664→   664→          </DropdownMenu>
   665→   665→          <Button
   666→   666→            variant="ghost"
   667→   667→            size="sm"
   668→   668→            onClick={() => onSelectionChange?.([])}
   669→   669→          >
   670→   670→            Clear
   671→   671→          </Button>
   672→   672→        </div>
   673→   673→      )}
   674→   674→    </div>
   675→   675→  );
   676→   676→}

   678→   678→// -------------------- Main DataTable Component --------------------
   679→   679→
   680→   680→export function DataTable<TData, TValue>({
   681→   681→  columns,
   682→   682→  data,
   683→   683→  isLoading = false,
   684→   684→  totalItems,
   685→   685→  pageSize,
   686→   686→  currentPage,
   687→   687→  onPageChange,
   688→   688→  onSortChange,
   689→   689→  sortField,
   690→   690→  sortOrder,
   691→   691→  onRowClick,
   692→   692→  selectedIds = [],
   693→   693→  onSelectionChange,
   694→   694→  onBulkAction,
   695→   695→  bulkActions,
   696→   696→  emptyMessage = 'No data found.',
   697→   697→  emptyIcon,
   698→   698→  emptyState,
   699→   699→  searchPlaceholder,
   700→   700→  onSearch,
   701→   701→  searchValue,
   702→   702→  filterContent,
   703→   703→  getRowId,
   704→   704→  onPageSizeChange,
   705→   705→  tableFixed = false,
   706→   706→  tableMinWidth,
   707→   707→}: DataTableProps<TData, TValue>) {
   708→   708→  const hasRowSelection = !!onSelectionChange;
   709→   709→  const columnCount = columns.length + (hasRowSelection ? 1 : 0);
   710→   710→
   711→   711→  const enhancedColumns = useMemo<ColumnDef<TData, TValue>[]>(() => {
   712→   712→    if (!hasRowSelection) return columns;
   713→   713→
   714→   714→    const selectColumn: ColumnDef<TData, TValue> = {
   715→   715→      id: '__select__',
   716→   716→      size: 40,
   717→   717→      enableSorting: false,
   718→   718→      enableHiding: false,
   719→   719→      header: ({ table }) => {
   720→   720→        const allRowIds = table.getRowModel().rows.map((r) => r.id);
   721→   721→        const allChecked =
   722→   722→          allRowIds.length > 0 && allRowIds.every((id) => selectedIds.includes(id));
   723→   723→        const someChecked = allRowIds.some((id) => selectedIds.includes(id));
   724→   724→
   725→   725→        return (
   726→   726→          <Checkbox
   727→   727→            checked={allChecked ? true : someChecked ? 'indeterminate' : false}
   728→   728→            onCheckedChange={(checked) => {
   729→   729→              if (checked) {
   730→   730→                const allIds = data
   731→   731→                  .map((row, idx) => getRowId?.(row) ?? String(idx))
   732→   732→                  .filter((id) => !selectedIds.includes(id));
   733→   733→                onSelectionChange?.([...selectedIds, ...allIds]);
   734→   734→              } else {
   735→   735→                const currentDataIds = data.map((row, idx) =>
   736→   736→                  getRowId?.(row) ?? String(idx),
   737→   737→                );
   738→   738→                onSelectionChange?.(
   739→   739→                  selectedIds.filter((id) => !currentDataIds.includes(id)),
   740→   740→                );
   741→   741→              }
   742→   742→            }}
   743→   743→            aria-label="Select all rows"
   744→   744→          />
   745→   745→        );
   746→   746→      },
   747→   747→      cell: ({ row }) => {
   748→   748→        const rowId = row.id;
   749→   749→        const isChecked = selectedIds.includes(rowId);
   750→   750→
   751→   751→        return (
   752→   752→          <Checkbox
   753→   753→            checked={isChecked}
   754→   754→            onCheckedChange={(checked) => {
   755→   755→              if (checked) {
   756→   756→                onSelectionChange?.([...selectedIds, rowId]);
   757→   757→              } else {
   758→   758→                onSelectionChange?.(selectedIds.filter((id) => id !== rowId));
   759→   759→              }
   760→   760→            }}
   761→   761→            onClick={(e) => e.stopPropagation()}
   762→   762→            aria-label={`Select row ${rowId}`}
   763→   763→          />
   764→   764→        );
   765→   765→      },
   766→   766→    };
   767→   767→
   768→   768→    return [selectColumn, ...columns];
   769→   769→  }, [columns, hasRowSelection, selectedIds, onSelectionChange, data, getRowId]);
   770→   770→
   771→   771→  const sorting: ColumnSort[] = useMemo(() => {
   772→   772→    if (!sortField) return [];
   773→   773→    return [{ id: sortField, desc: sortOrder === 'desc' }];
   774→   774→  }, [sortField, sortOrder]);
   775→   775→
   776→   776→  const table = useReactTable({
   777→   777→    data,
   778→   778→    columns: enhancedColumns,
   779→   779→    getCoreRowModel: getCoreRowModel(),
   780→   780→    getSortedRowModel: getSortedRowModel(),
   781→   781→    state: {
   782→   782→      sorting,
   783→   783→    },
   784→   784→    onSortingChange: (updater) => {
   785→   785→      const newSorting =
   786→   786→        typeof updater === 'function' ? updater(sorting) : updater;
   787→   787→      if (newSorting.length > 0) {
   788→   788→        onSortChange(newSorting[0].id, newSorting[0].desc ? 'desc' : 'asc');
   789→   789→      }
   790→   790→    },
   791→   791→    manualPagination: true,
   792→   792→    manualSorting: true,
   793→   793→    getRowId: getRowId
   794→   794→      ? (row, index) => getRowId(row) ?? String(index)
   795→   795→      : (_, index) => String(index),
   796→   796→  });
   797→   797→
   798→   798→  const handlePageSizeChange = useCallback(
   799→   799→    (newSize: number) => {
   800→   800→      onPageSizeChange?.(newSize);
   801→   801→      onPageChange(1);
   802→   802→    },
   803→   803→    [onPageChange, onPageSizeChange],
   804→   804→  );
   805→   805→
   806→   806→  const showToolbar =
   807→   807→    (onSearch || searchValue !== undefined) ||
   808→   808→    (filterContent != null) ||
   809→   809→    (selectedIds.length > 0 && bulkActions && bulkActions.length > 0);
   810→   810→
   811→   811→  const tableStyle = tableMinWidth ? { minWidth: `${tableMinWidth}px` } : undefined;
   812→   812→  const tableClassName = cn(tableFixed && 'table-fixed');
   813→   813→
   814→   814→  return (
   815→   815→    <div className="rounded-lg border overflow-hidden">
   816→   816→      {showToolbar && (
   817→   817→        <TableToolbar
   818→   818→          searchPlaceholder={searchPlaceholder}
   819→   819→          searchValue={searchValue}
   820→   820→          onSearch={onSearch}
   821→   821→          selectedIds={selectedIds}
   822→   822→          bulkActions={bulkActions}
   823→   823→          onBulkAction={onBulkAction}
   824→   824→          onSelectionChange={onSelectionChange}
   825→   825→          filterContent={filterContent}
   826→   826→        />
   827→   827→      )}
   828→   828→      <Table className={tableClassName} style={tableStyle}>
   829→   829→        <TableHeader>
   830→   830→          {table.getHeaderGroups().map((headerGroup) => (
   831→   831→            <TableRow key={headerGroup.id}>
   832→   832→              {headerGroup.headers.map((header) => (
   833→   833→                <TableHead
   834→   834→                  key={header.id}
   835→   835→                  style={{ width: header.getSize() !== 150 ? header.getSize() : undefined }}
   836→   836→                >
   837→   837→                  {flexRender(
   838→   838→                    header.column.columnDef.header,
   839→   839→                    header.getContext(),
   840→   840→                  )}
   841→   841→                </TableHead>
   842→   842→              ))}
   843→   843→            </TableRow>
   844→   844→          ))}
   845→   845→        </TableHeader>
   846→   846→        <TableBody>
   847→   847→          {isLoading ? (
   848→   848→            <SkeletonRows columnCount={columnCount} />
   849→   849→          ) : data.length === 0 ? (
   850→   850→            <DataTableEmpty icon={emptyIcon} message={emptyMessage} state={emptyState} />
   851→   851→          ) : (
   852→   852→            table.getRowModel().rows.map((row) => (
   853→   853→              <TableRow
   854→   854→                key={row.id}
   855→   855→                data-state={selectedIds.includes(row.id) ? 'selected' : undefined}
   856→   856→                className={cn(onRowClick && 'cursor-pointer')}
   857→   857→                onClick={() => onRowClick?.(row.original)}
   858→   858→              >
   859→   859→                {row.getVisibleCells().map((cell) => (
   860→   860→                  <TableCell key={cell.id}>
   861→   861→                    {flexRender(
   862→   862→                      cell.column.columnDef.cell,
   863→   863→                      cell.getContext(),
   864→   864→                    )}
   865→   865→                  </TableCell>
   866→   866→                ))}
   867→   867→              </TableRow>
   868→   868→            ))
   869→   869→          )}
   870→   870→        </TableBody>
   871→   871→      </Table>
   872→   872→      <DataTablePagination
   873→   873→        currentPage={currentPage}
   874→   874→        pageSize={pageSize}
   875→   875→        totalItems={totalItems}
   876→   876→        onPageChange={onPageChange}
   877→   877→        onPageSizeChange={handlePageSizeChange}
   878→   878→      />
   879→   879→    </div>
   880→   880→  );
   881→   881→}
