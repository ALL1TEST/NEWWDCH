   107→   107→// ==================== Validation ====================
   108→   108→
   109→   109→function validateRedirectForm(form: RedirectFormData, isEdit: boolean): string | null {
   110→   110→  if (!form.fromPath.trim()) return 'From URL is required.';
   111→   111→  if (!form.fromPath.startsWith('/')) return 'From URL must start with "/".';
   112→   112→  if (!form.toPath.trim()) return 'To URL is required.';
   113→   113→  if (!form.toPath.startsWith('/')) return 'To URL must start with "/".';
   114→   114→  if (!isEdit && form.fromPath.trim() === form.toPath.trim()) {
   115→   115→    return 'From URL and To URL cannot be the same (self-redirect).';
   116→   116→  }
   117→   117→  if (form.fromPath.trim().toLowerCase() === form.toPath.trim().toLowerCase()) {
   118→   118→    return 'From URL and To URL cannot be the same (self-redirect).';
   119→   119→  }
   120→   120→  return null;
   121→   121→}

   276→   276→    mutationFn: (csv: string) => postApi<CsvValidationResult>('/api/redirects/bulk?action=import', { csvContent: csv }),
   277→   277→    onSuccess: (data) => { setValidation(data as unknown as CsvValidationResult); setStep('preview'); },
   278→   278→    onError: () => toast.error('Failed to validate CSV'),
   279→   279→  });
   280→   280→
   281→   281→  const importMutation = useMutation({
   282→   282→    mutationFn: () => postApi<CsvImportResult>('/api/redirects/bulk?action=import&confirm=true', { csvContent }),
   283→   283→    onSuccess: (data) => { setImportResult(data as unknown as CsvImportResult); setStep('done'); queryClient.invalidateQueries({ queryKey: queryKeys.redirects.all }); },

   287→   287→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   288→   288→    const f = e.target.files?.[0];
   289→   289→    if (!f) return;
   290→   290→    setFile(f);
   291→   291→    const reader = new FileReader();
   292→   292→    reader.onload = (ev) => { const text = ev.target?.result as string; setCsvContent(text); setValidation(null); setImportResult(null); setStep('upload'); };
   293→   293→    reader.readAsText(f);
   294→   294→  };

   297→   297→    e.preventDefault();

   304→   304→  };
   305→   305→
   306→   306→  const handleValidate = () => { if (csvContent.trim()) validateMutation.mutate(csvContent); };
   307→   307→  const handleImport = () => { importMutation.mutate(); };
   308→   308→  const handleClose = () => { onOpenChange(false); setTimeout(() => { setCsvContent(''); setFile(null); setValidation(null); setImportResult(null); setStep('upload'); }, 200); };
   309→   309→

   311→   311→    <Dialog open={open} onOpenChange={handleClose}>
   312→   312→      <DialogContent className="sm:max-w-[600px] max-h-[80vh] overflow-y-auto">

   326→   326→/old-page,/new-page,301" value={csvContent} onChange={(e) => setCsvContent(e.target.value)} />
   327→   327→            </div>
   328→   328→            <DialogFooter>
   329→   329→              <Button variant="outline" onClick={handleClose}>Cancel</Button>
   330→   330→              <Button onClick={handleValidate} disabled={!csvContent.trim() || validateMutation.isPending}>{validateMutation.isPending && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}Validate & Preview</Button>
   331→   331→            </DialogFooter>
   332→   332→          </div>

   336→   336→          <div className="space-y-4 py-2">
   337→   337→            <div className="grid grid-cols-3 gap-3">
