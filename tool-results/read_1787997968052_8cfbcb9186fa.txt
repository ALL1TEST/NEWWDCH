   185→// -------------------- Provider Dropdown (custom, portaled, downward-first) -------------------
   186→
   187→/** Custom categorized provider dropdown.
   188→ *
   189→ *  Why this exists as a fully custom component (instead of the shadcn
   190→ *  Popover/Select): the Add Storage modal has a fixed header, a scrollable
   191→ *  form body and a fixed footer. A Radix-Popover-based dropdown rendered
   192→ *  inside that body inherits two problems:
   193→ *
   194→ *    1. Positioning — Radix auto-flips the panel to whichever side has more
   195→ *       room, which often meant the panel opened UPWARD over the modal
   196→ *       header even when there was plenty of space below the trigger.
   197→ *    2. Wheel scroll — the panel's inner scrollable region was being
   198→ *       starved of wheel events by the surrounding modal scroll container,
   199→ *       so hovering the list and scrolling did nothing.
   200→ *
   201→ *  This implementation fixes both by:
   202→ *
   203→ *    - Rendering the panel through React's `createPortal(...)` directly
   204→ *      onto `document.body`, so the modal's `overflow`/scroll container
   205→ *      is no longer an ancestor of the panel — it cannot clip the panel
   206→ *      and cannot intercept its wheel events.
   207→ *    - Computing the panel's `position: fixed` rect from the trigger's
   208→ *      `getBoundingClientRect()`, ALWAYS preferring to open DOWNWARD. The
   209→ *      panel flips upward only when downward genuinely cannot fit a 200px
   210→ *      minimum and upward has more room.
   211→ *    - Capping the panel height to `min(360px, available viewport space)`
   212→ *      with `overflow-y: auto` + `overscroll-behavior: contain`, so:
   213→ *        * mouse wheel over the panel scrolls the provider options (the
   214→ *          panel is the wheel target's nearest scrollable ancestor, and
   215→ *          containment prevents scroll-chaining back to the modal body);
   216→ *        * mouse wheel over the modal body (outside the panel) still
   217→ *          scrolls the modal body — we never attach a modal-level wheel
   218→ *          handler and never call `preventDefault()` on wheel.
   219→ *    - Tracking scroll/resize (capture phase, any element) so the panel
   220→ *      re-positions itself to follow the trigger when the modal body
   221→ *      scrolls.
   222→ *    - Full keyboard navigation (Arrow/Home/End/Enter/Escape) with the
   223→ *      active option kept in view.
   224→ *    - Outside-click (mousedown) closes; trigger click toggles; selecting
   225→ *      an option closes and returns focus to the trigger.
   226→ */
   227→function ProviderDropdown({
   228→  value,
   229→  onChange,
   230→  disabled,
   231→}: {
   232→  value: BackupStorageProvider;
   233→  onChange: (v: BackupStorageProvider) => void;
   234→  disabled?: boolean;
   235→}) {
   236→  const [open, setOpen] = useState(false);
   237→  const [activeIndex, setActiveIndex] = useState(-1);
   238→  const [panelStyle, setPanelStyle] = useState<React.CSSProperties>({});
   239→  const triggerRef = useRef<HTMLButtonElement>(null);
   240→  const panelRef = useRef<HTMLDivElement>(null);
   241→
   242→  // Refs to the latest values so the open-effect (which attaches window
   243→  // listeners) does NOT re-run on every value/onChange/activeIndex change.
   244→  // The listeners read from these refs instead.
   245→  const valueRef = useRef(value);
   246→  valueRef.current = value;
   247→  const onChangeRef = useRef(onChange);
   248→  onChangeRef.current = onChange;
   249→  const activeIndexRef = useRef(activeIndex);
   250→  activeIndexRef.current = activeIndex;
   251→
   252→  // Pre-compute the categorized structure (for rendering) and a flat
   253→  // index map (for keyboard navigation) once — the registry is static.
   254→  const { categorized, flatOptions, flatIndexOf } = useMemo(() => {
   255→    const cats = getProvidersByCategory();
   256→    const flat: { id: BackupStorageProvider; name: string }[] = [];
   257→    const idxMap = new Map<BackupStorageProvider, number>();
   258→    cats.forEach(({ providers }) => {
   259→      providers.forEach((p) => {
   260→        idxMap.set(p.id, flat.length);
   261→        flat.push({ id: p.id, name: p.name });
   262→      });
   263→    });
   264→    return { categorized: cats, flatOptions: flat, flatIndexOf: idxMap };
   265→  }, []);
   266→
   267→  const selected = getProviderDefinition(value);
   268→
   269→  /** Recompute the panel's fixed position from the trigger's current rect.
   270→   *  Prefers opening DOWNWARD. Flips upward only when downward cannot fit
   271→   *  a 200px minimum AND upward has more room. Closes the dropdown if the
   272→   *  trigger has scrolled entirely out of the viewport. */
   273→  const updatePosition = useCallback(() => {
   274→    const trigger = triggerRef.current;
   275→    if (!trigger) return;
   276→    const rect = trigger.getBoundingClientRect();
   277→    if (rect.bottom < 0 || rect.top > window.innerHeight) {
   278→      setOpen(false);
   279→      return;
   280→    }
   281→    const MIN_HEIGHT = 200;
   282→    const SAFETY = 16; // px margin from the viewport edge
   283→    const spaceBelow = window.innerHeight - rect.bottom - SAFETY;
   284→    const spaceAbove = rect.top - SAFETY;
   285→    const placement: 'below' | 'above' =
   286→      spaceBelow >= MIN_HEIGHT || spaceBelow >= spaceAbove ? 'below' : 'above';
   287→    const available = placement === 'below' ? spaceBelow : spaceAbove;
   288→    // Cap to 360px (per spec); never smaller than 120px so the panel is
   289→    // always usable even on very short viewports.
   290→    const maxHeight = Math.max(120, Math.min(360, available));
   291→    const top =
   292→      placement === 'below' ? rect.bottom + 4 : rect.top - 4 - maxHeight;
   293→    setPanelStyle({
   294→      position: 'fixed',
   295→      top: `${top}px`,
   296→      left: `${rect.left}px`,
   297→      width: `${rect.width}px`,
   298→      maxHeight: `${maxHeight}px`,
   299→      zIndex: 60,
   300→    });
   301→  }, []);
   302→
   303→  // Open/close lifecycle. Runs only when `open` changes (refs hold the
   304→  // latest value/onChange/activeIndex so the listeners stay stable).
   305→  useLayoutEffect(() => {
   306→    if (!open) return;
   307→
   308→    // Initial active option = currently selected provider (so the
   309→    // highlight is on the user's current value when the panel opens).
   310→    const idx = flatIndexOf.get(valueRef.current) ?? 0;
   311→    setActiveIndex(idx);
   312→    updatePosition();
   313→
   314→    // Move focus to the panel so keyboard events originate from inside
   315→    // the dropdown (Arrow/Home/End/Enter are then scoped to it). Deferred
   316→    // one frame so the portaled element is mounted before we focus it.
   317→    const raf = requestAnimationFrame(() => panelRef.current?.focus());
   318→
   319→    // ---- Listeners -----------------------------------------------------
   320→
   321→    // Reposition on ANY scroll event (capture phase catches scrolls on
   322→    // any descendant — including the modal body's internal overflow-y-auto
   323→    // — so the panel always tracks the trigger). Harmless when the panel's
   324→    // own inner scroll fires (recomputes the same position).
   325→    const onScroll = () => updatePosition();
   326→    const onResize = () => updatePosition();
   327→    window.addEventListener('scroll', onScroll, true);
   328→    window.addEventListener('resize', onResize);
   329→
   330→    // Outside mousedown closes the dropdown (but not the modal — the
   331→    // modal's own onPointerDownOutside still calls preventDefault to
   332→    // keep the modal alive). Trigger mousedown is allowed through so the
   333→    // trigger's onClick can toggle.
   334→    const onPointerDown = (e: MouseEvent) => {
   335→      const t = e.target as Node | null;
   336→      if (!t) return;
   337→      if (panelRef.current?.contains(t)) return;
   338→      if (triggerRef.current?.contains(t)) return;
   339→      setOpen(false);
   340→    };
   341→    document.addEventListener('mousedown', onPointerDown);
   342→
   343→    // True only when the currently-focused element is inside the dropdown
   344→    // (trigger or panel). Used to scope Arrow/Home/End/Enter keys so they
   345→    // don't hijack typing in the Name field while the panel happens to be
   346→    // open. Escape works from anywhere.
   347→    const isInDropdown = () => {
   348→      const active = document.activeElement;
   349→      if (!active) return false;
   350→      if (panelRef.current?.contains(active)) return true;
   351→      if (triggerRef.current?.contains(active)) return true;
   352→      return false;
   353→    };
   354→
   355→    const onKeyDown = (e: KeyboardEvent) => {
   356→      if (e.key === 'Escape') {
   357→        e.preventDefault();
   358→        setOpen(false);
   359→        if (isInDropdown()) triggerRef.current?.focus();
   360→        return;
   361→      }
   362→      if (!isInDropdown()) return; // don't hijack other inputs
   363→      switch (e.key) {
   364→        case 'ArrowDown':
   365→          e.preventDefault();
   366→          setActiveIndex((i) =>
   367→            i < 0 ? 0 : Math.min(i + 1, flatOptions.length - 1),
   368→          );
   369→          break;
   370→        case 'ArrowUp':
   371→          e.preventDefault();
   372→          setActiveIndex((i) => (i <= 0 ? 0 : i - 1));
   373→          break;
   374→        case 'Home':
   375→          e.preventDefault();
   376→          setActiveIndex(0);
   377→          break;
   378→        case 'End':
   379→          e.preventDefault();
   380→          setActiveIndex(flatOptions.length - 1);
   381→          break;
   382→        case 'Enter':
   383→          e.preventDefault();
   384→          {
   385→            const i = activeIndexRef.current;
   386→            if (i >= 0 && i < flatOptions.length) {
   387→              onChangeRef.current(flatOptions[i].id);
   388→              setOpen(false);
   389→              triggerRef.current?.focus();
   390→            }
   391→          }
   392→          break;
   393→        default:
   394→          break;
   395→      }
   396→    };
   397→    document.addEventListener('keydown', onKeyDown);
   398→
   399→    // If focus leaves the dropdown entirely (e.g. user clicks into the
   400→    // Name field or Tabs away), close — same behavior as Radix Select.
   401→    const onFocusIn = (e: FocusEvent) => {
   402→      const t = e.target as Node | null;
   403→      if (!t) return;
   404→      if (panelRef.current?.contains(t)) return;
   405→      if (triggerRef.current?.contains(t)) return;
   406→      setOpen(false);
   407→    };
   408→    document.addEventListener('focusin', onFocusIn);
   409→
   410→    return () => {
   411→      cancelAnimationFrame(raf);
   412→      window.removeEventListener('scroll', onScroll, true);
   413→      window.removeEventListener('resize', onResize);
   414→      document.removeEventListener('mousedown', onPointerDown);
   415→      document.removeEventListener('keydown', onKeyDown);
   416→      document.removeEventListener('focusin', onFocusIn);
   417→    };
   418→  }, [open, flatIndexOf, flatOptions, updatePosition]);
   419→
   420→  // Keep the active (highlighted) option scrolled into view inside the
   421→  // panel whenever it changes (keyboard nav, hover, or initial open).
   422→  useLayoutEffect(() => {
   423→    if (!open) return;
   424→    const el = panelRef.current?.querySelector<HTMLElement>(
   425→      `[data-index="${activeIndex}"]`,
   426→    );
   427→    el?.scrollIntoView({ block: 'nearest' });
   428→  }, [activeIndex, open]);
   429→
   430→  const handleTriggerClick = () => {
   431→    if (disabled) return;
   432→    setOpen((o) => !o);
   433→  };
   434→
   435→  /** Wheel handler attached to the panel itself. We ONLY call
   436→   *  `stopPropagation()` — never `preventDefault()`. This means:
   437→   *    - native scroll on the panel (its `overflow-y:auto`) still works
   438→   *      normally — the wheel target's nearest scrollable ancestor is the
   439→   *      panel, so the browser scrolls the panel;
   440→   *    - the wheel event does not bubble up to any ancestor handler that
   441→   *      might try to scroll the modal body or the page;
   442→   *    - `overscroll-behavior:contain` (set inline on the panel) further
   443→   *      prevents scroll-chaining to the modal body when the panel
   444→   *      reaches its top/bottom boundary.
   445→   *  Mouse wheel over the modal body (outside the panel) is never
   446→   *  intercepted here and continues to scroll the modal body normally. */
   447→  const handleWheel = (e: React.WheelEvent) => {
   448→    e.stopPropagation();
   449→  };
   450→
   451→  return (
   452→    <>
   453→      <button
   454→        ref={triggerRef}
   455→        type="button"
   456→        aria-haspopup="listbox"
   457→        aria-expanded={open}
   458→        disabled={disabled}
   459→        onClick={handleTriggerClick}
   460→        className="flex h-9 w-full items-center justify-between rounded-md border border-input bg-transparent px-3 py-2 text-sm shadow-sm transition-[color,box-shadow] hover:bg-accent/50 focus:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:border-ring disabled:cursor-not-allowed disabled:opacity-50"
   461→      >
   462→        <span className="truncate text-left text-foreground">
   463→          {selected?.name ?? 'Select provider'}
   464→        </span>
   465→        <ChevronDown
   466→          className={cn(
   467→            'h-4 w-4 opacity-50 transition-transform shrink-0',
   468→            open && 'rotate-180',
   469→          )}
   470→        />
   471→      </button>
   472→
   473→      {/* The panel is rendered via createPortal directly onto document.body
   474→          so the modal's overflow/scroll-container is no longer an ancestor.
   475→          It cannot clip the panel and cannot intercept the panel's wheel
   476→          events. The panel only mounts once `panelStyle.top` has been
   477→          computed by the layout effect above (so it never flashes at the
   478→          wrong position). */}
   479→      {open &&
   480→        panelStyle.top !== undefined &&
   481→        createPortal(
   482→          <div
   483→            ref={panelRef}
   484→            role="listbox"
   485→            tabIndex={-1}
   486→            onWheel={handleWheel}
   487→            style={{
   488→              ...panelStyle,
   489→              overflowY: 'auto',
   490→              overscrollBehavior: 'contain',
   491→              // Body is set to `pointer-events: none` while the Radix Dialog
   492→              // is open (Radix's scroll-lock). `pointer-events` is inherited
   493→              // by default, so a panel portaled to `document.body` would
   494→              // inherit `none` and become click/wheel-through. Explicitly
   495→              // re-enable pointer events on the panel so it actually
   496→              // captures mouse wheel and click events.
   497→              pointerEvents: 'auto',
   498→            }}
   499→            className="storage-modal-scroll animate-in fade-in-0 zoom-in-95 slide-in-from-top-2 bg-popover text-popover-foreground rounded-md border shadow-md p-1 outline-none"
   500→          >
   501→            {categorized.map(({ category, label, providers }) => (
   502→              <div key={category} className="py-1">
   503→                <div className="px-2 py-1 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground/70">
   504→                  {label}
   505→                </div>
   506→                {providers.map((p) => {
   507→                  const flatIdx = flatIndexOf.get(p.id) ?? 0;
   508→                  const isSelected = p.id === value;
   509→                  const isActive = flatIdx === activeIndex;
   510→                  return (
   511→                    <button
   512→                      key={p.id}
   513→                      type="button"
   514→                      role="option"
   515→                      aria-selected={isSelected}
   516→                      data-index={flatIdx}
   517→                      onMouseEnter={() => setActiveIndex(flatIdx)}
   518→                      onClick={() => {
   519→                        onChange(p.id);
   520→                        setOpen(false);
   521→                        triggerRef.current?.focus();
   522→                      }}
   523→                      className={cn(
   524→                        'flex w-full items-center justify-between rounded-sm px-2 py-1.5 text-sm outline-none transition-colors',
   525→                        isActive ? 'bg-accent' : 'hover:bg-accent/70',
   526→                        isSelected && 'font-medium',
   527→                      )}
   528→                    >
   529→                      <span className="truncate">{p.name}</span>
   530→                      {isSelected && (
   531→                        <Check className="h-4 w-4 text-primary shrink-0" />
   532→                      )}
   533→                    </button>
   534→                  );
   535→                })}
   536→              </div>
   537→            ))}
   538→          </div>,
   539→          document.body,
   540→        )}
   541→    </>
   542→  );
   543→}

   560→// -------------------- Validation helpers --------------------
   561→
   562→/** True when the Name field is non-empty AND every required field for the
   563→ *  currently-selected provider has a non-empty value. For providers that
   564→ *  require a validated connection, the connection state must also be
   565→ *  'connected'. */
   566→function isFormValid(form: StorageForm): boolean {
   567→  if (!form.name.trim()) return false;
   568→  const def = getProviderDefinition(form.provider);
   569→  if (!def) return false;
   570→
   571→  const allFieldsFilled = def.fields.every((f) => {
   572→    if (!f.required) return true;
   573→    const v = form.config[f.key];
   574→    if (v == null) return false;
   575→    return f.type === 'switch' ? true : v.trim() !== '';
   576→  });
   577→  if (!allFieldsFilled) return false;
   578→
   579→  if (def.requiresConnection && form.connection !== 'connected') return false;
   580→  return true;
   581→}
   582→
   583→/** Returns the inline error message for a single field, or '' when the
   584→ *  field is valid. Required fields that are empty show their message only
   585→ *  after the user has blurred the field OR attempted to submit — we never
   586→ *  surface a generic aggregate "N issues" summary. The Name field follows
   587→ *  the same rule. */
   588→function fieldError(
   589→  form: StorageForm,
   590→  field: { key: string; label: string; required: boolean; type: string },
   591→  submitAttempted: boolean,
   592→): string {
   593→  if (!field.required) return '';
   594→  if (field.type === 'switch') return '';
   595→  const v = form.config[field.key];
   596→  const isEmpty = v == null || v.trim() === '';
   597→  if (!isEmpty) return '';
   598→  const wasTouched = form.touched[field.key] === true;
   599→  if (!wasTouched && !submitAttempted) return '';
   600→  return `${field.label} is required`;
   601→}
   602→
   603→/** Signature for the connection/credential-group fields of the current
   604→ *  provider. Used to detect whether the user has edited any such field
   605→ *  since the last Test/Connect — if so, the 'connected' state is stale. */
   606→function connectionSignature(form: StorageForm): string {
   607→  const def = getProviderDefinition(form.provider);
   608→  if (!def) return '';
   609→  const groups: FieldGroup[] =
   610→    def.connectionType === 'oauth' ? ['connection'] : ['credentials'];
   611→  return def.fields
   612→    .filter((f) => groups.includes(f.group))
   613→    .map((f) => `${f.key}=${form.config[f.key] ?? ''}`)
   614→    .join('|');
   615→}
   616→
   617→/** True when all required connection/credential fields are filled, so the
   618→ *  Test/Connect button can be enabled. */
   619→function canValidateConnection(form: StorageForm): boolean {
   620→  const def = getProviderDefinition(form.provider);
   621→  if (!def || def.connectionType === 'none') return false;
   622→  const groups: FieldGroup[] =
   623→    def.connectionType === 'oauth' ? ['connection'] : ['credentials'];
   624→  return def.fields
   625→    .filter((f) => groups.includes(f.group))
   626→    .every((f) => {
   627→      if (!f.required) return true;
   628→      const v = form.config[f.key];
   629→      return v != null && v.trim() !== '';
   630→    });
   631→}
   632→
   633→/** Detects password-type connection/credential fields still holding the
   634→ *  API mask placeholder ('••••••••'). When editing an existing storage,
   635→ *  secrets come back masked; re-testing with a mask sends a bogus secret.
   636→ *  We block the test and ask the user to re-enter the secret. */
   637→function maskedSecretFields(form: StorageForm): string[] {
   638→  const def = getProviderDefinition(form.provider);
   639→  if (!def) return [];
   640→  const groups: FieldGroup[] =
   641→    def.connectionType === 'oauth' ? ['connection'] : ['credentials'];
   642→  return def.fields
   643→    .filter((f) => f.type === 'password' && groups.includes(f.group))
   644→    .map((f) => f.label)
   645→    .filter((label) => label && label.length > 0)
   646→    .filter((_, i) => {
   647→      const f = def.fields
   648→        .filter((x) => x.type === 'password' && groups.includes(x.group))[i];
   649→      const v = form.config[f.key];
   650→      return typeof v === 'string' && v.includes('•');
   651→    });
   652→}
   653→
   654→/** Build the config object from the current provider's fields only. Empty
   655→ *  optional fields are omitted. Number fields → Number. Switch → boolean. */
   656→function buildConfigObject(form: StorageForm): Record<string, unknown> {
   657→  const def = getProviderDefinition(form.provider);
   658→  if (!def) return {};
   659→  const configObj: Record<string, unknown> = {};
   660→  for (const field of def.fields) {
   661→    const raw = form.config[field.key];
   662→    if (raw == null || raw.trim() === '') continue;
   663→    if (field.type === 'number') {
   664→      const n = Number(raw);
   665→      if (!Number.isNaN(n)) configObj[field.key] = n;
   666→    } else if (field.type === 'switch') {
   667→      configObj[field.key] = raw === 'true';
   668→    } else {
   669→      configObj[field.key] = raw;
   670→    }
   671→  }
   672→  return configObj;
   673→}

  1512→// -------------------- Config Field Renderer --------------------
  1513→// Renders a single provider field generically. Password fields mask
  1514→// their value (type=password + a lock affordance). Switch fields use
  1515→// the Switch component. Multiline password fields (e.g. GCS private

  1518→
  1519→function ConfigField({
  1520→  field,
  1521→  value,
  1522→  onChange,
  1523→  onBlur,
  1524→  error,
  1525→  disabled,
  1526→}: {

  1528→  value: string;

  1530→  onBlur?: () => void;
  1531→  error?: string;
  1532→  disabled?: boolean;
  1533→}) {
  1534→  const inputId = `storage-config-${field.key}`;
  1535→  const showError = !!error;
  1536→  return (
  1537→    <div className="space-y-1.5">
  1538→      <Label htmlFor={inputId} className="text-xs">
  1539→        {field.label}
  1540→        {field.required && <span className="text-destructive ml-0.5">*</span>}
  1541→      </Label>

  1543→        <div className="flex items-center gap-2 h-9">
  1544→          <Switch
  1545→            id={inputId}
  1546→            checked={value === 'true'}
  1547→            onCheckedChange={(checked) => onChange(checked ? 'true' : 'false')}
  1548→            disabled={disabled}
  1549→          />
  1550→          <span className="text-xs text-muted-foreground">
  1551→            {value === 'true' ? 'Enabled' : 'Disabled'}
  1552→          </span>

  1554→      ) : field.type === 'password' && field.multiline ? (
  1555→        <Textarea
  1556→          id={inputId}
  1557→          placeholder={field.placeholder}
  1558→          value={value}
  1559→          onChange={(e) => onChange(e.target.value)}
  1560→          onBlur={onBlur}
  1561→          rows={4}
  1562→          disabled={disabled}

  1566→      ) : (
  1567→        <div className="relative">
  1568→          <Input
  1569→            id={inputId}
  1570→            type={field.type === 'password' ? 'password' : field.type === 'number' ? 'number' : 'text'}
  1571→            placeholder={field.placeholder}
  1572→            value={value}
  1573→            onChange={(e) => onChange(e.target.value)}

  1579→          />
  1580→          {field.type === 'password' && value && (
  1581→            <Lock className="absolute right-3 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground/60 pointer-events-none" />
  1582→          )}
  1583→        </div>
  1584→      )}
  1585→      {showError ? (
  1586→        <p className="text-xs text-destructive">{error}</p>
  1587→      ) : field.helpText ? (
  1588→        <p className="text-xs text-muted-foreground leading-relaxed">{field.helpText}</p>
  1589→      ) : null}
  1590→    </div>
  1591→  );
  1592→}
