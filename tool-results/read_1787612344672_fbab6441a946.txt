   204→   204→    mutationFn: (id: string) => deleteApi(`/api/ai/providers/${id}`),

   226→   226→        toast.error(res.message || `Connection ${res.status || 'failed'}`);
   227→   227→      }
   228→   228→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   229→   229→    },
   230→   230→    onError: (err: Error) => {
   231→   231→      toast.error(err.message || 'Connection test failed');
   232→   232→    },
   233→   233→  });

   235→   235→  // Sync models mutation

   240→   240→      toast.success(`Synced ${res.syncedCount ?? res.count ?? 0} models`);
   241→   241→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   242→   242→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   243→   243→    },
   244→   244→    onError: (err: Error) => {
   245→   245→      toast.error(err.message || 'Failed to sync models');
   246→   246→    },
   247→   247→  });

   258→   258→    },
   259→   259→  });

   266→   266→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });

   268→   268→    },

   284→   284→    setFormData({
   285→   285→      name: provider.name,
   286→   286→      kind: provider.kind,

   348→   348→            </div>
   349→   349→          </CardContent>
   350→   350→        </Card>
