    99→function getFileIcon(mimeType: string) {
   100→  if (mimeType.startsWith('image/')) return ImageIcon;
   101→  if (mimeType.startsWith('video/')) return Film;
   102→  if (mimeType.startsWith('audio/')) return Music;
   103→  if (mimeType.includes('pdf') || mimeType.includes('document') || mimeType.includes('text')) return FileText;
   104→  return File;
   105→}
   106→function isImageType(mimeType: string) { return mimeType.startsWith('image/'); }

   108→/** Flatten a tree of folders into an indented list for select dropdowns */
   109→function flattenFolderTree(
   110→  folders: MediaFolderItem[],
   111→  depth = 0,
   112→): { id: string; name: string; depth: number }[] {
   113→  const result: { id: string; name: string; depth: number }[] = [];
   114→  for (const f of folders) {
   115→    result.push({ id: f.id, name: f.name, depth });
   116→    if (f.children && f.children.length > 0) {
   117→      result.push(...flattenFolderTree(f.children, depth + 1));
   118→    }
   119→  }
   120→  return result;
   121→}

   513→  // Each file is uploaded in its own request, so one failed file does NOT

   515→  // the console AND in the toast so the user can see exactly what failed.
   516→  const uploadMutation = useMutation({
   517→    mutationFn: async (files: File[]) => {
   518→      const url = buildUploadUrl();
   519→      const results: MediaItemRow[] = [];
   520→      const failures: { name: string; reason: string }[] = [];
   521→
   522→      for (const file of files) {
   523→        const formData = new FormData();
   524→        formData.append('file', file);
   525→        if (currentFolderId) formData.append('folderId', currentFolderId);
   526→        if (user?.id) formData.append('uploadedById', user.id);
   527→
   528→        try {
   529→          const res = await fetch(url, { method: 'POST', body: formData });
   530→          // Always try to read the JSON envelope — even on error responses
   531→          // the backend returns `{ error: { code, message }, meta }`.
   532→          const json = (await res.json().catch(() => null)) as
   533→            | { data?: MediaItemRow | MediaItemRow[]; error?: { code?: string; message?: string } }
   534→            | null;
   535→
   536→          if (!res.ok || !json?.data) {
   537→            const reason =
   538→              json?.error?.message ||
   539→              (res.status === 413 ? 'File is too large' :
   540→               res.status === 415 ? 'File type not supported' :
   541→               res.status === 401 ? 'Authentication required' :
   542→               res.status === 403 ? 'Not allowed' :
   543→               res.status === 404 ? 'Upload endpoint not found' :
   544→               res.status >= 500 ? 'Server error' :
   545→               `HTTP ${res.status}`) ||
   546→              res.statusText ||
   547→              'Upload failed';
   548→            failures.push({ name: file.name, reason });
   549→            console.error(
   550→              `[MEDIA:UPLOAD] ${file.name} failed —`,
   551→              `status=${res.status}`,
   552→              `code=${json?.error?.code ?? 'n/a'}`,
   553→              `reason=${reason}`,
   554→            );
   555→            continue;
   556→          }
   557→
   558→          const data = json.data;
   559→          if (Array.isArray(data)) results.push(...data);
   560→          else results.push(data);
   561→        } catch (err) {
   562→          // Network error / aborted request — surface the actual reason
   563→          const reason = err instanceof Error ? err.message : 'Network request failed';
   564→          failures.push({ name: file.name, reason });
   565→          console.error(`[MEDIA:UPLOAD] ${file.name} network error —`, err);
   566→        }
   567→      }
   568→
   569→      // Always refresh the media list — even partial success means new
   570→      // files exist on the server and should appear.
   571→      const summary: UploadSummary = {
   572→        succeeded: results.length,
   573→        failed: failures,
   574→      };
   575→      // If EVERY file failed, throw so React Query triggers onError — but
   576→      // include the structured summary so the handler can show specifics.
   577→      if (results.length === 0 && files.length > 0) {
   578→        const err = new Error(
   579→          failures.length === 1
   580→            ? `Upload failed: ${failures[0].reason}`
   581→            : `All ${failures.length} files failed`,
   582→        ) as Error & { uploadSummary?: UploadSummary };
   583→        err.uploadSummary = summary;
   584→        throw err;
   585→      }
   586→      return summary;
   587→    },
   588→    onSuccess: (summary) => {
   589→      invalidateMediaAndFolders();
   590→      setUploadDialogOpen(false);
   591→      setUploadFiles([]);
   592→
   593→      if (summary.failed.length === 0) {
   594→        const n = summary.succeeded;
   595→        toast.success(n === 1 ? '1 file uploaded successfully' : `${n} files uploaded successfully`);
   596→      } else {
   597→        const failedNames = summary.failed.map((f) => f.name).join(', ');
   598→        toast.warning(
   599→          `${summary.succeeded} file${summary.succeeded === 1 ? '' : 's'} uploaded, ${summary.failed.length} failed: ${failedNames}`,
   600→          { duration: 6000 },
   601→        );
   602→      }
   603→    },
   604→    onError: (err: Error & { uploadSummary?: UploadSummary }) => {
   605→      const summary = err.uploadSummary;
   606→      if (summary && summary.failed.length > 0) {
   607→        const first = summary.failed[0];
   608→        toast.error(
   609→          summary.failed.length === 1
   610→            ? `Upload failed: ${first.reason}`
   611→            : `All ${summary.failed.length} files failed. First reason: ${first.reason}`,
   612→          { duration: 8000 },
   613→        );
   614→      } else {
   615→        toast.error(err.message || 'Upload failed');
   616→      }
   617→    },
   618→  });
