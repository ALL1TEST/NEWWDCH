   229→/**
   230→ * Derive the top-level section ID from the current browser hash.
   231→ * Uses EXPLICIT prefix matching — #settings/seo always returns 'Settings', never 'SEO'.
   232→ */
   233→function getSectionFromHash(hash: string): string | null {
   234→  const path = hash.replace(/^#\/?,?/, '').trim();
   235→  if (!path || path === 'dashboard') return null;
   236→
   237→  const firstSegment = path.split('/')[0];
   238→  return ROUTE_PREFIX_TO_SECTION[firstSegment] ?? null;
   239→}

   376→/*

   378→ *   • Inline submenu visibility  → controlled ONLY by openSection/onToggle

   389→/**
   390→ * Settings parent item rendered on the COLLAPSED 48px rail.
   391→ * Clicking it opens a portal-based popover to the RIGHT of the icon —
   392→ * the sidebar itself NEVER expands, and the popover anchors to this
   393→ * exact button so the "star/sparkle alignment" grid stays intact.
   394→ */
   395→function CollapsedParentNavItem({
   396→  item,
   397→  currentModule,
   398→  currentSubPage,
   399→}: {
   400→  item: NavItem;
   401→  currentModule: string;
   402→  currentSubPage: string | null;
   403→}) {
   404→  const [floatOpen, setFloatOpen] = useState(false);
   405→  const mod = hrefToModule(item.href);
   406→  const isActive = currentModule === mod;
   407→
   408→  const handleChildNavigate = (
   409→    e: React.MouseEvent<HTMLAnchorElement>,
   410→    child: NavItem,
   411→  ) => {
   412→    e.preventDefault();
   413→    e.stopPropagation();
   414→    setFloatOpen(false); // close the floating submenu after navigation…
   415→    // …but do NOT touch sidebar expansion state (rule #7).
   416→    const hash = child.href.replace(/^#/, '');
   417→    const parts = hash.split('/');
   418→    const childSubPage = parts[1] || null;
   419→    useNavigationStore.getState().navigate(parts[0], null, childSubPage);
   420→  };
   421→
   422→  return (
   423→    <SidebarMenuItem>
   424→      <Popover open={floatOpen} onOpenChange={setFloatOpen}>
   425→        <PopoverTrigger asChild>
   426→          <SidebarMenuButton
   427→            isActive={isActive || floatOpen}
   428→            aria-expanded={floatOpen}
   429→            aria-haspopup="menu"
   430→            tooltip={floatOpen ? undefined : item.label}
   431→            onClick={(e: React.MouseEvent) => {
   432→              e.preventDefault();
   433→              setFloatOpen((o) => !o);
   434→            }}
   435→          >
   436→            <NavIcon name={item.icon} />
   437→            <span>{item.label}</span>
   438→          </SidebarMenuButton>
   439→        </PopoverTrigger>
   440→        {/* Portal → document.body: immune to sidebar overflow clipping.
   441→            Radix handles Esc + outside-click closing and re-anchors at any
   442→            viewport size/zoom via avoidCollisions. */}
   443→        <PopoverContent
   444→          side="right"
   445→          align="start"
   446→          sideOffset={6}
   447→          collisionPadding={12}
   448→          className="w-56 rounded-md border bg-popover p-1.5 shadow-md"
   449→          onEscapeKeyDown={() => setFloatOpen(false)}
   450→        >
   451→          <ul
   452→            role="menu"
   453→            aria-label={`${item.label} submenu`}
   454→            className="flex min-w-0 list-none flex-col gap-0.5 p-0 m-0"
   455→          >
   456→            {item.children!.map((child) => {
   457→              const hash = child.href.replace(/^#/, '');
   458→              const parts = hash.split('/');
   459→              const childSubPage = parts[1] || null;
   460→              const isChildActive =
   461→                currentModule === parts[0] &&
   462→                (!childSubPage || currentSubPage === childSubPage);
   463→              return (
   464→                <SidebarMenuSubItem key={child.label} className="list-none p-0">
   465→                  <SidebarMenuSubButton asChild isActive={isChildActive}>
   466→                    <a
   467→                      href={child.href}
   468→                      role="menuitem"
   469→                      onClick={(e) => handleChildNavigate(e, child)}
   470→                    >
   471→                      <NavIcon name={child.icon} />
   472→                      <span>{child.label}</span>
   473→                    </a>
   474→                  </SidebarMenuSubButton>
   475→                </SidebarMenuSubItem>
   476→              );
   477→            })}
   478→          </ul>
   479→        </PopoverContent>
   480→      </Popover>
   481→    </SidebarMenuItem>
   482→  );
   483→}

   485→function ExpandableNavItem({
   486→  item,
   487→  currentModule,
   488→  currentSubPage,
   489→  isExpanded,
   490→  onToggle,
   491→}: {
   492→  item: NavItem;
   493→  currentModule: string;
   494→  currentSubPage: string | null;
   495→  isExpanded: boolean;
   496→  onToggle: () => void;
   497→}) {
   498→  const mod = hrefToModule(item.href);
   499→  const isActive = currentModule === mod;
   500→  const sectionId = `submenu-${item.label.toLowerCase().replace(/\s+/g, '-')}`;
   501→
   502→  return (
   503→    <SidebarMenuItem>
   504→      <SidebarMenuButton
   505→        tooltip={item.label}
   506→        isActive={isActive}
   507→        onClick={(e: React.MouseEvent) => {
   508→          e.preventDefault();
   509→          onToggle();
   510→        }}
   511→        aria-expanded={isExpanded}
   512→        aria-controls={sectionId}
   513→      >
   514→        <NavIcon name={item.icon} />
   515→        <span>{item.label}</span>
   516→        <ChevronRight
   517→          className={cn(
   518→            'ml-auto h-4 w-4 shrink-0 transition-transform duration-200',
   519→            isExpanded && 'rotate-90',
   520→          )}
   521→        />
   522→      </SidebarMenuButton>
   523→
   524→      {item.badge != null && (
   525→        <SidebarMenuBadge>
   526→          {item.badge}
   527→        </SidebarMenuBadge>
   528→      )}
   529→
   530→      <AccordionSubmenu isOpen={isExpanded} sectionLabel={item.label}>
   531→        {item.children!.map((child) => {
   532→          const hash = child.href.replace(/^#/, '');
   533→          const parts = hash.split('/');
   534→          const childMod = parts[0];
   535→          const childSubPage = parts[1] || null;
   536→          const isChildActive =
   537→            currentModule === childMod &&
   538→            (!childSubPage || currentSubPage === childSubPage);
   539→
   540→          return (
   541→            <SidebarMenuSubItem key={child.label}>
   542→              <SidebarMenuSubButton
   543→                asChild
   544→                isActive={isChildActive}
   545→              >
   546→                <a
   547→                  href={child.href}
   548→                  onClick={(e) => {
   549→                    e.preventDefault();
   550→                    e.stopPropagation();
   551→                    useNavigationStore
   552→                      .getState()
   553→                      .navigate(childMod, null, childSubPage);
   554→                  }}
   555→                >
   556→                  <NavIcon name={child.icon} />
   557→                  <span>{child.label}</span>
   558→                </a>
   559→              </SidebarMenuSubButton>
   560→            </SidebarMenuSubItem>
   561→          );
   562→        })}
   563→      </AccordionSubmenu>
   564→    </SidebarMenuItem>
   565→  );
   566→}

   608→function NavGroupSection({
   609→  group,
   610→  currentModule,
   611→  currentSubPage,
   612→  openSection,
   613→  onToggleSection,
   614→}: {
   615→  group: NavGroup;
   616→  currentModule: string;
   617→  currentSubPage: string | null;
   618→  openSection: string | null;
   619→  onToggleSection: (label: string) => void;
   620→}) {
   621→  const isCollapsedRail = useCollapsedRail();
   622→
   623→  return (
   624→    <SidebarGroup>
   625→      <SidebarGroupContent>
   626→        <SidebarMenu>
   627→          {group.items.map((item) => {
   628→            const hasChildren = item.children && item.children.length > 0;
   629→
   630→            if (hasChildren && isCollapsedRail) {
   631→              /* Collapsed rail → parents with children become floating
   632→                 popovers anchored to their own icon. The rail NEVER
   633→                 auto-expands (#4/#5/#6). */
   634→              return (
   635→                <CollapsedParentNavItem
   636→                  key={item.label}
   637→                  item={item}
   638→                  currentModule={currentModule}
   639→                  currentSubPage={currentSubPage}
   640→                />
   641→              );
   642→            }
   643→
   644→            if (hasChildren) {
   645→              return (
   646→                <ExpandableNavItem
   647→                  key={item.label}
   648→                  item={item}
   649→                  currentModule={currentModule}
   650→                  currentSubPage={currentSubPage}
   651→                  isExpanded={openSection === item.label}
   652→                  onToggle={() => onToggleSection(item.label)}
   653→                />
   654→              );
   655→            }
   656→
   657→            return (
   658→              <SimpleNavItem
   659→                key={item.label}
   660→                item={item}
   661→                currentModule={currentModule}
   662→              />
   663→            );
   664→          })}
   665→        </SidebarMenu>

   673→export function AppSidebar() {
   674→  const user = useAuthStore((s) => s.user);
   675→  // Active plan — the footer role badge shows the plan NAME (e.g. "Beta")
   676→  // instead of the static user role, styled with the plan's amber accent.
   677→  const { currentPlan } = useSubscriptionStore();
   678→  const currentModule = useNavigationStore((s) => s.currentModule);
   679→  const currentSubPage = useNavigationStore((s) => s.currentSubPage);
   680→  const openCommandPalette = useCommandPaletteStore((s) => s.open);
   681→  // Sidebar context hook — MUST run before any early return.
   682→  const sidebarCtx = useSidebar();
   683→
   684→  const userRole = user?.role;
   685→  const pagePermissions = user?.pagePermissions ?? null;
   686→  const visibleItems = useMemo(() => {
   687→    if (!userRole) return [];
   688→    return getVisibleNavItems(userRole, NAV_ITEMS, pagePermissions);
   689→  }, [userRole, pagePermissions]);
   690→
   691→  /*
   692→   * SINGLE SOURCE OF TRUTH for the expanded top-level section.
   693→   *
   694→   * manualOverride is a tri-state:
   695→   *   null  → not overridden; use route-derived value
   696→   *   ""   → explicitly closed by user
   697→   *   "SEO" → explicitly opened by user
   698→   *
   699→   * When the module changes (navigation event), we reset manualOverride
   700→   * to null so the route-derived value takes over.
   701→   */
   702→  const [manualOverride, setManualOverride] = useState<string | null>(null);
   703→  // Track previous module to detect navigation changes synchronously during render.
   704→  // This is the React-approved pattern for "derived state from props".
   705→  const [prevModule, setPrevModule] = useState(currentModule);
   706→  if (currentModule !== prevModule) {
   707→    setPrevModule(currentModule);
   708→    setManualOverride(null);
   709→  }
   710→
   711→  // Derive the section from the current route hash (EXPLICIT prefix matching)
   712→  const routeDerivedSection = useMemo(() => {
   713→    return getSectionFromHash(window.location.hash);
   714→  }, [currentModule, currentSubPage]);
   715→
   716→  // Compute the actual open section
   717→  const openSection = useMemo(() => {
   718→    if (manualOverride !== null) {
   719→      // User manually toggled: empty string = closed, non-empty = that section
   720→      return manualOverride || null;
   721→    }
   722→    return routeDerivedSection;
   723→  }, [manualOverride, routeDerivedSection]);
   724→
   725→  // Toggle a section open/closed
   726→  const handleToggleSection = useCallback(
   727→    (label: string) => {
   728→      setManualOverride((prev) => {
   729→        // Determine what's currently open (considering route-derived if no override)
   730→        const currentlyOpen =
   731→          prev !== null ? (prev || null) : routeDerivedSection;
   732→        // If clicking the same section that's open, close it
   733→        if (currentlyOpen === label) {
   734→          return ''; // explicitly closed
   735→        }
   736→        // Otherwise, open the clicked section (closing any other)
   737→        return label;
   738→      });
   739→    },
   740→    [routeDerivedSection],
   741→  );
   742→
   743→  if (!user) return null;
   744→
   745→  const groups = buildNavGroups(visibleItems);
   746→
   747→  return (

   749→      {/* ---- Header: one 32px icon cell per row, all centered at x=24px ---- */}
   750→      <SidebarHeader className="px-2 py-3 shrink-0">
   751→        {/* Expanded: [logo][title][…spacer…][Search][Collapse toggle].
   752→            Per instruction the Search icon sits directly NEXT TO the
   753→            Collapse toggle at the far right — NOT next to the "CMS Admin"
   754→            name (the old position was removed, never duplicated). */}

   756→          <LogoMark />
   757→          <span className="truncate font-semibold text-sm tracking-tight whitespace-nowrap text-text-primary">
   758→            CMS Admin
   759→          </span>
   760→          <div className="ml-auto flex shrink-0 items-center gap-1">
   761→            <Tooltip>
   762→              <TooltipTrigger asChild>
   763→                <Button
   764→                  variant="ghost"
   765→                  size="icon"
   766→                  className="h-8 w-8 shrink-0 rounded-md"
   767→                  onClick={openCommandPalette}
   768→                  aria-label="Search"
   769→                >
   770→                  <Search className="h-4 w-4" />
   771→                  <span className="sr-only">Search</span>
   772→                </Button>
   773→              </TooltipTrigger>
   774→              <TooltipContent side="bottom">Search</TooltipContent>
   775→            </Tooltip>
   776→            <CollapseToggle side="bottom" />
   777→          </div>
   778→        </div>
   779→        {/* Collapsed rail: ONLY the "C" logo. The previously stacked

   782→            Expand affordance: the logo cell itself toggles the sidebar

   785→        <div className="hidden flex-col items-center group-data-[collapsible=icon]:flex">
