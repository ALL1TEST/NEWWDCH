     7→// (src/modules/backups/backups-list-page.tsx): same PageHeader
     8→// + Table layout, same Create Backup Dialog, same Restore /
     9→// Delete ConfirmDialogs, same Verify + Download actions, same
    10→// badge components (Scope / Type / Encryption / Verification /

    12→// No-Search-Results patterns.

    20→// GET + POST routes were extended with a `scope=platform`
    21→// query/body param guarded by requirePlatformAdmin (PLATFORM_ADMIN
    22→// OR OWNER). The per-id endpoints (GET/PATCH/DELETE/verify/restore/
    23→// download) are scope-agnostic and work as-is.

   148→// -------------------- Badge Components --------------------
   149→
   150→function ScopeBadge({ scope }: { scope: BackupScope }) {
   151→  return (
   152→    <Badge
   153→      variant="outline"
   154→      className={cn('border-transparent font-medium', SCOPE_BADGE_CLASSES[scope])}
   155→    >
   156→      {labelize(scope)}
   157→    </Badge>
   158→  );
   159→}
   160→
   161→function TypeBadge({ type }: { type: BackupType }) {
   162→  const classes: Record<BackupType, string> = {
   163→    AUTOMATED: 'bg-teal-100 text-teal-700 dark:bg-teal-900/30 dark:text-teal-400',
   164→    MANUAL: 'bg-orange-100 text-orange-700 dark:bg-orange-900/30 dark:text-orange-400',
   165→  };
   166→  return (
   167→    <Badge
   168→      variant="outline"
   169→      className={cn('border-transparent font-medium', classes[type])}
   170→    >
   171→      {labelize(type)}
   172→    </Badge>
   173→  );
   174→}
   175→
   176→function EncryptionBadge({ status }: { status: string }) {
   177→  if (status !== 'ENCRYPTED') {
   178→    return <span className="text-muted-foreground text-xs">—</span>;
   179→  }
   180→  return (
   181→    <Badge
   182→      variant="outline"
   183→      className="border-transparent bg-green-100 text-green-700 dark:bg-green-900/30 dark:text-green-400 font-medium"
   184→    >
   185→      <Lock className="h-3 w-3 mr-1" />
   186→      Encrypted
   187→    </Badge>
   188→  );
   189→}
   190→
   191→function VerificationBadge({ status }: { status: string | null }) {
   192→  if (!status || status === 'PENDING' || status === 'SKIPPED') {
   193→    return <span className="text-muted-foreground text-xs">—</span>;
   194→  }
   195→  return <StatusBadge status={status} size="sm" />;
   196→}

   226→// -------------------- Main Component --------------------
   227→
   228→export function PlatformBackupsModule() {
   229→  const queryClient = useQueryClient();
   230→  const currentUserId = useAuthStore((s) => s.user?.id ?? null);
   231→
   232→  // -------------------- UI State --------------------
   233→  const [dialogOpen, setDialogOpen] = useState(false);
   234→  const [form, setForm] = useState<CreateBackupForm>(initialForm);
   235→  const [deleteTarget, setDeleteTarget] = useState<BackupRow | null>(null);
   236→  const [restoreTarget, setRestoreTarget] = useState<BackupRow | null>(null);
   237→
   238→  // -------------------- Search State (debounced, client-side) --------------------
   239→  const [searchInput, setSearchInput] = useState('');
   240→  const [search, setSearch] = useState('');
   241→  const debounceRef = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
   242→
   243→  const updateSearch = useCallback((value: string) => {
   244→    setSearchInput(value);
   245→    if (debounceRef.current) clearTimeout(debounceRef.current);
   246→    debounceRef.current = setTimeout(() => {
   247→      setSearch(value);
   248→    }, 300);
   249→  }, []);
   250→
   251→  const handleClearSearch = useCallback(() => {
   252→    setSearchInput('');
   253→    setSearch('');
   254→  }, []);
   255→
   256→  const updateForm = useCallback(
   257→    (key: keyof CreateBackupForm, value: string | boolean) => {
   258→      setForm((prev) => ({ ...prev, [key]: value }));
   259→    },
   260→    [],
   261→  );
   262→
   263→  // -------------------- Data fetching --------------------
   264→  // GET /api/backups?scope=platform — platform admin sees ALL backups
   265→  // across all sites (no site filter). The queryKey includes `search`
   266→  // per spec; the API call itself does NOT send search server-side
   267→  // because we filter client-side by name (mirrors spec wording).
   268→  const { data: raw, isLoading, isError, refetch } = useQuery({
   269→    queryKey: ['platform-backups', search],
   270→    queryFn: () =>
   271→      getApi<ListResponse>(
   272→        '/api/backups',
   273→        { scope: 'platform', pageSize: 20 },
   274→        { raw: true },
   275→      ),
   276→    staleTime: 10_000,
   277→  });
   278→
   279→  const backups = useMemo(() => {
   280→    const list = (Array.isArray(raw?.data) ? raw.data : []) as BackupRow[];
   281→    if (!search.trim()) return list;
   282→    const q = search.trim().toLowerCase();
   283→    return list.filter((b) =>
   284→      (b.name ?? '').toLowerCase().includes(q) ||
   285→      (b.note ?? '').toLowerCase().includes(q),
   286→    );
   287→  }, [raw, search]);
   288→
   289→  // Fetch configured storage destinations for the dropdown.
   290→  // Platform admins have no active site context, so api-client does not
   291→  // inject siteId — the storage endpoint returns ALL destinations across
   292→  // all sites. The `?scope=platform` is passed for explicitness (it's a
   293→  // no-op on the storage route today).
   294→  const { data: storageDestinationsData } = useQuery({
   295→    queryKey: ['platform-backups-storage-destinations'],
   296→    queryFn: () =>
   297→      getApi<ListResponse>('/api/backups/storage', {
   298→        scope: 'platform',
   299→        pageSize: 100,
   300→      }, { raw: true }),
   301→    staleTime: 30_000,
   302→  });
   303→  const storageDestinations = useMemo(() => {
   304→    const list = (Array.isArray(storageDestinationsData?.data)
   305→      ? storageDestinationsData.data
   306→      : []) as StorageDestination[];
   307→    return list.filter((s) => s.isActive);
   308→  }, [storageDestinationsData]);
   309→
   310→  const hasSearch = !!search.trim();
   311→  const isInitialEmpty = !isLoading && backups.length === 0 && !hasSearch;
   312→  const isSearchEmpty = !isLoading && backups.length === 0 && hasSearch;
   313→
   314→  // -------------------- Mutations --------------------
   315→  const createMutation = useMutation({
   316→    mutationFn: () =>
   317→      postApi('/api/backups', {
   318→        scope: 'platform',
   319→        backupScope: form.backupScope,
   320→        name: form.name.trim(),
   321→        description: form.description.trim(),
   322→        storageId: form.storageId || undefined,
   323→        encryptionEnabled: form.encryptionEnabled,
   324→      } as Record<string, unknown>),
   325→    onSuccess: () => {
   326→      queryClient.invalidateQueries({ queryKey: ['platform-backups'] });
   327→      toast.success('Backup creation started');
   328→      setDialogOpen(false);
   329→      setForm(initialForm);
   330→    },
   331→    onError: (err: Error) => toast.error(err.message || 'Failed to create backup'),
   332→  });
   333→
   334→  const verifyMutation = useMutation({
   335→    mutationFn: (id: string) =>
   336→      postApi(`/api/backups/${id}/verify`, {
   337→        ...(currentUserId ? { createdById: currentUserId } : {}),
   338→      }),
   339→    onSuccess: () => {
   340→      queryClient.invalidateQueries({ queryKey: ['platform-backups'] });
   341→      toast.success('Verification started');
   342→    },
   343→    onError: (err: Error) => toast.error(err.message || 'Failed to verify backup'),
   344→  });
   345→
   346→  const restoreMutation = useMutation({
   347→    mutationFn: (id: string) =>
   348→      postApi(`/api/backups/${id}/restore`, {
   349→        ...(currentUserId ? { createdById: currentUserId } : {}),
   350→      }),
   351→    onSuccess: () => {
   352→      queryClient.invalidateQueries({ queryKey: ['platform-backups'] });
   353→      setRestoreTarget(null);
   354→      toast.success('Restore initiated successfully');
   355→    },
   356→    onError: (err: Error) => {
   357→      setRestoreTarget(null);
   358→      toast.error(err.message || 'Failed to restore backup');
   359→    },
   360→  });
   361→
   362→  const deleteMutation = useMutation({
   363→    mutationFn: (id: string) => deleteApi(`/api/backups/${id}`),
   364→    onSuccess: () => {
   365→      queryClient.invalidateQueries({ queryKey: ['platform-backups'] });
   366→      setDeleteTarget(null);
   367→      toast.success('Backup deleted');
   368→    },
   369→    onError: (err: Error) => {
   370→      setDeleteTarget(null);
   371→      toast.error(err.message || 'Failed to delete backup');
   372→    },
   373→  });
   374→
   375→  // -------------------- Render --------------------
   376→  return (
   377→    <div className="space-y-6">
   378→      <PlatformPageHeader
   379→        title="Backups"
   380→        subtitle="Platform-wide backup management. Restore, verify, and download backups across all customers and sites."
   381→        actions={
   382→          <Button size="sm" onClick={() => setDialogOpen(true)}>
   383→            <Plus className="h-4 w-4 mr-2" />
   384→            Create Backup
   385→          </Button>
   386→        }
   387→      />
   388→
   389→      {/* Search bar */}
   390→      <div className="relative max-w-sm">
   391→        <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   392→        <Input
   393→          placeholder="Search backups by name..."
   394→          value={searchInput}
   395→          onChange={(e) => updateSearch(e.target.value)}
   396→          className="pl-9 h-9"
   397→        />
   398→      </div>
   399→
   400→      {/* Backup History Table — same visual structure as the Client
   401→          Backups page: bordered Card + plain Table with overflow-x-auto
   402→          and the same 11 columns (Name / Scope / Type / Size / Storage /
   403→          Encryption / Verification / Status / Duration / Created /
   404→          Actions). */}
   405→      <div className="rounded-lg border bg-card overflow-x-auto">
   406→        {isLoading ? (
   407→          // Loading: 5 skeleton table rows matching the 11-column layout.
   408→          <div className="p-4 space-y-3">
   409→            {Array.from({ length: 5 }).map((_, i) => (
   410→              <div key={i} className="flex items-center gap-4">
   411→                <Skeleton className="h-5 w-40" />
   412→                <Skeleton className="h-5 w-20" />
   413→                <Skeleton className="h-5 w-16" />
   414→                <Skeleton className="h-5 w-16" />
   415→                <Skeleton className="h-5 w-24" />
   416→                <Skeleton className="h-5 w-20" />
   417→                <Skeleton className="h-5 w-20" />
   418→                <Skeleton className="h-5 w-16" />
   419→                <Skeleton className="h-5 w-16" />
   420→                <Skeleton className="h-5 w-32" />
   421→                <Skeleton className="h-8 w-8 ml-auto" />
   422→              </div>
   423→            ))}
   424→          </div>
   425→        ) : isError ? (
   426→          <div className="py-6">
   427→            <ErrorState
   428→              message="Unable to load platform backups. Make sure you are signed in as a platform admin."
   429→              onRetry={() => refetch()}
   430→            />
   431→          </div>
   432→        ) : isInitialEmpty ? (
   433→          <EmptyState
   434→            icon={DatabaseBackup}
   435→            title="No backups recorded"
   436→            description="When backups complete they will appear here. Create your first platform-wide backup to get started."
   437→            action={{
   438→              label: 'Create Backup',
   439→              onClick: () => setDialogOpen(true),
   440→              icon: <Plus className="h-4 w-4" />,
   441→            }}
   442→          />
   443→        ) : (
   444→          <Table>
   445→            <TableHeader>
   446→              <TableRow>
   447→                <TableHead className="pl-4">Name</TableHead>
   448→                <TableHead>Scope</TableHead>
   449→                <TableHead>Type</TableHead>
   450→                <TableHead>Size</TableHead>
   451→                <TableHead>Storage</TableHead>
   452→                <TableHead>Encryption</TableHead>
   453→                <TableHead>Verification</TableHead>
   454→                <TableHead>Status</TableHead>
   455→                <TableHead>Duration</TableHead>
   456→                <TableHead>Created</TableHead>
   457→                <TableHead className="pr-4 text-right">Actions</TableHead>
   458→              </TableRow>
   459→            </TableHeader>
   460→            <TableBody>
   461→              {isSearchEmpty ? (
   462→                <TableRow>
   463→                  <TableCell colSpan={11} className="p-0">
   464→                    <NoSearchResultsEmpty onClear={handleClearSearch} />
   465→                  </TableCell>
   466→                </TableRow>
   467→              ) : (
   468→                backups.map((b) => (
   469→                  <TableRow key={b.id} className="group">
   470→                    <TableCell className="pl-4">
   471→                      <span className="font-medium text-sm" title={b.name}>
   472→                        {b.name || '—'}
   473→                      </span>
   474→                    </TableCell>
   475→                    <TableCell>
   476→                      <ScopeBadge scope={b.scope} />
   477→                    </TableCell>
   478→                    <TableCell>
   479→                      <TypeBadge type={b.type} />
   480→                    </TableCell>
   481→                    <TableCell>
   482→                      <span className="tabular-nums text-sm text-muted-foreground">
   483→                        {formatFileSize(b.size)}
   484→                      </span>
   485→                    </TableCell>
   486→                    <TableCell>
   487→                      <span
   488→                        className="text-xs text-muted-foreground"
   489→                        title={labelize(b.storageProvider)}
   490→                      >
   491→                        {labelize(b.storageProvider)}
   492→                      </span>
   493→                    </TableCell>
   494→                    <TableCell>
   495→                      <EncryptionBadge status={b.encryptionStatus} />
   496→                    </TableCell>
   497→                    <TableCell>
   498→                      <VerificationBadge status={b.verificationStatus} />
   499→                    </TableCell>
   500→                    <TableCell>
   501→                      <StatusBadge status={b.status} size="sm" />
   502→                    </TableCell>
   503→                    <TableCell>
   504→                      <span className="tabular-nums text-xs text-muted-foreground">
   505→                        {formatDurationMs(b.durationMs)}
   506→                      </span>
   507→                    </TableCell>
   508→                    <TableCell>
   509→                      <span
   510→                        className="text-xs text-muted-foreground"
   511→                        title={formatRelativeTime(b.createdAt)}
   512→                      >
   513→                        {formatRelativeTime(b.createdAt)}
   514→                      </span>
   515→                    </TableCell>
   516→                    <TableCell className="pr-4 text-right">
   517→                      <DropdownMenu>
   518→                        <DropdownMenuTrigger asChild>
   519→                          <Button
   520→                            variant="ghost"
   521→                            size="icon"
   522→                            className="h-8 w-8"
   523→                            onClick={(e) => e.stopPropagation()}
   524→                          >
   525→                            <MoreHorizontal className="h-4 w-4" />
   526→                          </Button>
   527→                        </DropdownMenuTrigger>
   528→                        <DropdownMenuContent align="end">
   529→                          <DropdownMenuItem
   530→                            onClick={() =>
   531→                              window.open(`/api/backups/${b.id}/download`, '_blank')
   532→                            }
   533→                          >
   534→                            <Download className="h-4 w-4 mr-2" />
   535→                            Download
   536→                          </DropdownMenuItem>
   537→                          <DropdownMenuItem
   538→                            onClick={() => verifyMutation.mutate(b.id)}
   539→                            disabled={
   540→                              b.status !== 'COMPLETED' || verifyMutation.isPending
   541→                            }
   542→                          >
   543→                            <ShieldCheck className="h-4 w-4 mr-2" />
   544→                            Verify
   545→                          </DropdownMenuItem>
   546→                          <DropdownMenuItem
   547→                            onClick={() => setRestoreTarget(b)}
   548→                            disabled={b.status !== 'COMPLETED'}
   549→                          >
   550→                            <RotateCcw className="h-4 w-4 mr-2" />
   551→                            Restore
   552→                          </DropdownMenuItem>
   553→                          <DropdownMenuSeparator />
   554→                          <DropdownMenuItem
   555→                            variant="destructive"
   556→                            onClick={() => setDeleteTarget(b)}
   557→                          >
   558→                            <Trash2 className="h-4 w-4 mr-2" />
   559→                            Delete
   560→                          </DropdownMenuItem>
   561→                        </DropdownMenuContent>
   562→                      </DropdownMenu>
   563→                    </TableCell>
   564→                  </TableRow>
   565→                ))
   566→              )}
   567→            </TableBody>
   568→          </Table>
   569→        )}
   570→      </div>
   571→
   572→      {/* Create Backup Dialog */}
   573→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   574→        <DialogContent className="sm:max-w-[480px]">
   575→          <DialogHeader>
   576→            <DialogTitle>Create Backup</DialogTitle>
   577→            <DialogDescription>
   578→              Start a new platform-wide backup. The backup will be visible across all customers and sites.
   579→            </DialogDescription>
   580→          </DialogHeader>
   581→          <div className="space-y-4 py-2">
   582→            <div className="space-y-2">
   583→              <Label htmlFor="backup-name">Name</Label>
   584→              <Input
   585→                id="backup-name"
   586→                placeholder="e.g., Daily platform backup"
   587→                value={form.name}
   588→                onChange={(e) => updateForm('name', e.target.value)}
   589→              />
   590→            </div>
   591→            <div className="space-y-2">
   592→              <Label htmlFor="backup-desc">Description</Label>
   593→              <Textarea
   594→                id="backup-desc"
   595→                placeholder="Optional description..."
   596→                value={form.description}
   597→                onChange={(e) => updateForm('description', e.target.value)}
   598→                rows={2}
   599→              />
   600→            </div>
