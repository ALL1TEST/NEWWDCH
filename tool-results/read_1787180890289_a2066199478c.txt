   312→function AIGenerateDialog({
   313→  open,
   314→  onOpenChange,
   315→  idea,
   316→}: {
   317→  open: boolean;
   318→  onOpenChange: (open: boolean) => void;
   319→  idea: ArticleIdea | null;
   320→}) {
   321→  const [brief, setBrief] = useState('');
   322→  const [keywords, setKeywords] = useState('');
   323→  const [writingStyle, setWritingStyle] = useState('Professional');
   324→  const [targetLength, setTargetLength] = useState('Medium (800-1200 words)');
   325→  const [numberOfDrafts, setNumberOfDrafts] = useState('1');
   326→  const [includeCta, setIncludeCta] = useState(false);
   327→
   328→  const navigate = useNavigationStore((s) => s.navigate);
   329→
   330→  const generateMutation = useMutation({
   331→    mutationFn: () =>
   332→      postApi('/api/content/ai-generate', {
   333→        title: idea?.title || '',
   334→        brief: brief || idea?.description || '',
   335→        keywords: keywords || idea?.keywords?.join(', ') || '',
   336→        writingStyle,
   337→        targetLength,
   338→        numberOfDrafts: parseInt(numberOfDrafts),
   339→        includeCta,
   340→      }),
   341→    onSuccess: (result) => {
   342→      const draft = result.data?.drafts?.[0];
   343→      if (draft) {
   344→        toast.success('Article draft generated!');
   345→        onOpenChange(false);
   346→        navigate('content', null, 'create');
   347→      }
   348→    },
   349→    onError: (err: Error) => toast.error(err.message || 'Failed to generate'),
   350→  });
   351→
   352→  React.useEffect(() => {
   353→    if (idea) {
   354→      setBrief(idea.description);
   355→      setKeywords(idea.keywords?.join(', ') || '');
   356→    }
   357→  }, [idea]);
   358→
   359→  return (
   360→    <Dialog open={open} onOpenChange={onOpenChange}>
   361→      <DialogContent className="sm:max-w-lg">
   362→        <DialogHeader>
   363→          <DialogTitle className="flex items-center gap-2">

   375→              {idea?.title}
   376→            </div>
   377→          </div>
   378→
   379→          {/* Brief */}
   380→          <div className="space-y-1.5">
   381→            <Label className="text-xs text-muted-foreground">Brief / Description</Label>
   382→            <Textarea
   383→              value={brief}
