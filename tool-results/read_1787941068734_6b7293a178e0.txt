   184→// -------------------- Provider Dropdown (custom, portaled, downward-first) -------------------
   185→
   186→/** Custom categorized provider dropdown.
   187→ *
   188→ *  Why this exists as a fully custom component (instead of the shadcn
   189→ *  Popover/Select): the Add Storage modal has a fixed header, a scrollable
   190→ *  form body and a fixed footer. A Radix-Popover-based dropdown rendered
   191→ *  inside that body inherits two problems:
   192→ *
   193→ *    1. Positioning — Radix auto-flips the panel to whichever side has more
   194→ *       room, which often meant the panel opened UPWARD over the modal
   195→ *       header even when there was plenty of space below the trigger.
   196→ *    2. Wheel scroll — the panel's inner scrollable region was being
   197→ *       starved of wheel events by the surrounding modal scroll container,
   198→ *       so hovering the list and scrolling did nothing.
   199→ *
   200→ *  This implementation fixes both by:
   201→ *
   202→ *    - Rendering the panel through React's `createPortal(...)` directly
   203→ *      onto `document.body`, so the modal's `overflow`/scroll container
   204→ *      is no longer an ancestor of the panel — it cannot clip the panel
   205→ *      and cannot intercept its wheel events.
   206→ *    - Computing the panel's `position: fixed` rect from the trigger's
   207→ *      `getBoundingClientRect()`, ALWAYS preferring to open DOWNWARD. The
   208→ *      panel flips upward only when downward genuinely cannot fit a 200px
   209→ *      minimum and upward has more room.
   210→ *    - Capping the panel height to `min(360px, available viewport space)`
   211→ *      with `overflow-y: auto` + `overscroll-behavior: contain`, so:
   212→ *        * mouse wheel over the panel scrolls the provider options (the
   213→ *          panel is the wheel target's nearest scrollable ancestor, and
   214→ *          containment prevents scroll-chaining back to the modal body);
   215→ *        * mouse wheel over the modal body (outside the panel) still
   216→ *          scrolls the modal body — we never attach a modal-level wheel
   217→ *          handler and never call `preventDefault()` on wheel.
   218→ *    - Tracking scroll/resize (capture phase, any element) so the panel
   219→ *      re-positions itself to follow the trigger when the modal body
   220→ *      scrolls.
   221→ *    - Full keyboard navigation (Arrow/Home/End/Enter/Escape) with the
   222→ *      active option kept in view.
   223→ *    - Outside-click (mousedown) closes; trigger click toggles; selecting
   224→ *      an option closes and returns focus to the trigger.
   225→ */
   226→function ProviderDropdown({
   227→  value,
   228→  onChange,
   229→  disabled,
   230→}: {
   231→  value: BackupStorageProvider;
   232→  onChange: (v: BackupStorageProvider) => void;
   233→  disabled?: boolean;
   234→}) {
   235→  const [open, setOpen] = useState(false);
   236→  const [activeIndex, setActiveIndex] = useState(-1);
   237→  const [panelStyle, setPanelStyle] = useState<React.CSSProperties>({});
   238→  const triggerRef = useRef<HTMLButtonElement>(null);
   239→  const panelRef = useRef<HTMLDivElement>(null);
   240→
   241→  // Refs to the latest values so the open-effect (which attaches window
   242→  // listeners) does NOT re-run on every value/onChange/activeIndex change.
   243→  // The listeners read from these refs instead.
   244→  const valueRef = useRef(value);
   245→  valueRef.current = value;
   246→  const onChangeRef = useRef(onChange);
   247→  onChangeRef.current = onChange;
   248→  const activeIndexRef = useRef(activeIndex);
   249→  activeIndexRef.current = activeIndex;
   250→
   251→  // Pre-compute the categorized structure (for rendering) and a flat
   252→  // index map (for keyboard navigation) once — the registry is static.
   253→  const { categorized, flatOptions, flatIndexOf } = useMemo(() => {
   254→    const cats = getProvidersByCategory();
   255→    const flat: { id: BackupStorageProvider; name: string }[] = [];
   256→    const idxMap = new Map<BackupStorageProvider, number>();
   257→    cats.forEach(({ providers }) => {
   258→      providers.forEach((p) => {
   259→        idxMap.set(p.id, flat.length);
   260→        flat.push({ id: p.id, name: p.name });
   261→      });
   262→    });
   263→    return { categorized: cats, flatOptions: flat, flatIndexOf: idxMap };
   264→  }, []);
   265→
   266→  const selected = getProviderDefinition(value);
   267→
   268→  /** Recompute the panel's fixed position from the trigger's current rect.
   269→   *  Prefers opening DOWNWARD. Flips upward only when downward cannot fit
   270→   *  a 200px minimum AND upward has more room. Closes the dropdown if the
   271→   *  trigger has scrolled entirely out of the viewport. */
   272→  const updatePosition = useCallback(() => {
   273→    const trigger = triggerRef.current;
   274→    if (!trigger) return;
   275→    const rect = trigger.getBoundingClientRect();
   276→    if (rect.bottom < 0 || rect.top > window.innerHeight) {
   277→      setOpen(false);
   278→      return;
   279→    }
   280→    const MIN_HEIGHT = 200;
   281→    const SAFETY = 16; // px margin from the viewport edge
   282→    const spaceBelow = window.innerHeight - rect.bottom - SAFETY;
   283→    const spaceAbove = rect.top - SAFETY;
   284→    const placement: 'below' | 'above' =
   285→      spaceBelow >= MIN_HEIGHT || spaceBelow >= spaceAbove ? 'below' : 'above';
   286→    const available = placement === 'below' ? spaceBelow : spaceAbove;
   287→    // Cap to 360px (per spec); never smaller than 120px so the panel is
   288→    // always usable even on very short viewports.
   289→    const maxHeight = Math.max(120, Math.min(360, available));
   290→    const top =
   291→      placement === 'below' ? rect.bottom + 4 : rect.top - 4 - maxHeight;
   292→    setPanelStyle({
   293→      position: 'fixed',
   294→      top: `${top}px`,
   295→      left: `${rect.left}px`,
   296→      width: `${rect.width}px`,
   297→      maxHeight: `${maxHeight}px`,
   298→      zIndex: 60,
   299→    });
   300→  }, []);
   301→
   302→  // Open/close lifecycle. Runs only when `open` changes (refs hold the
   303→  // latest value/onChange/activeIndex so the listeners stay stable).
   304→  useLayoutEffect(() => {
   305→    if (!open) return;
   306→
   307→    // Initial active option = currently selected provider (so the
   308→    // highlight is on the user's current value when the panel opens).
   309→    const idx = flatIndexOf.get(valueRef.current) ?? 0;
   310→    setActiveIndex(idx);
   311→    updatePosition();
   312→
   313→    // Move focus to the panel so keyboard events originate from inside
   314→    // the dropdown (Arrow/Home/End/Enter are then scoped to it). Deferred
   315→    // one frame so the portaled element is mounted before we focus it.
   316→    const raf = requestAnimationFrame(() => panelRef.current?.focus());
   317→
   318→    // ---- Listeners -----------------------------------------------------
   319→
   320→    // Reposition on ANY scroll event (capture phase catches scrolls on
   321→    // any descendant — including the modal body's internal overflow-y-auto
   322→    // — so the panel always tracks the trigger). Harmless when the panel's
   323→    // own inner scroll fires (recomputes the same position).
   324→    const onScroll = () => updatePosition();
   325→    const onResize = () => updatePosition();
   326→    window.addEventListener('scroll', onScroll, true);
   327→    window.addEventListener('resize', onResize);
   328→
   329→    // Outside mousedown closes the dropdown (but not the modal — the
   330→    // modal's own onPointerDownOutside still calls preventDefault to
   331→    // keep the modal alive). Trigger mousedown is allowed through so the
   332→    // trigger's onClick can toggle.
   333→    const onPointerDown = (e: MouseEvent) => {
   334→      const t = e.target as Node | null;
   335→      if (!t) return;
   336→      if (panelRef.current?.contains(t)) return;
   337→      if (triggerRef.current?.contains(t)) return;
   338→      setOpen(false);
   339→    };
   340→    document.addEventListener('mousedown', onPointerDown);
   341→
   342→    // True only when the currently-focused element is inside the dropdown
   343→    // (trigger or panel). Used to scope Arrow/Home/End/Enter keys so they
   344→    // don't hijack typing in the Name field while the panel happens to be
   345→    // open. Escape works from anywhere.
   346→    const isInDropdown = () => {
   347→      const active = document.activeElement;
   348→      if (!active) return false;
   349→      if (panelRef.current?.contains(active)) return true;
   350→      if (triggerRef.current?.contains(active)) return true;
   351→      return false;
   352→    };
   353→
   354→    const onKeyDown = (e: KeyboardEvent) => {
   355→      if (e.key === 'Escape') {
   356→        e.preventDefault();
   357→        setOpen(false);
   358→        if (isInDropdown()) triggerRef.current?.focus();
   359→        return;
   360→      }
   361→      if (!isInDropdown()) return; // don't hijack other inputs
   362→      switch (e.key) {
   363→        case 'ArrowDown':
   364→          e.preventDefault();
   365→          setActiveIndex((i) =>
   366→            i < 0 ? 0 : Math.min(i + 1, flatOptions.length - 1),
   367→          );
   368→          break;
   369→        case 'ArrowUp':
   370→          e.preventDefault();
   371→          setActiveIndex((i) => (i <= 0 ? 0 : i - 1));
   372→          break;
   373→        case 'Home':
   374→          e.preventDefault();
   375→          setActiveIndex(0);
   376→          break;
   377→        case 'End':
   378→          e.preventDefault();
   379→          setActiveIndex(flatOptions.length - 1);
   380→          break;
   381→        case 'Enter':
   382→          e.preventDefault();
   383→          {
   384→            const i = activeIndexRef.current;
   385→            if (i >= 0 && i < flatOptions.length) {
   386→              onChangeRef.current(flatOptions[i].id);
   387→              setOpen(false);
   388→              triggerRef.current?.focus();
   389→            }
   390→          }
   391→          break;
   392→        default:
   393→          break;
   394→      }
   395→    };
   396→    document.addEventListener('keydown', onKeyDown);
   397→
   398→    // If focus leaves the dropdown entirely (e.g. user clicks into the
   399→    // Name field or Tabs away), close — same behavior as Radix Select.
   400→    const onFocusIn = (e: FocusEvent) => {
   401→      const t = e.target as Node | null;
   402→      if (!t) return;
   403→      if (panelRef.current?.contains(t)) return;
   404→      if (triggerRef.current?.contains(t)) return;
   405→      setOpen(false);
   406→    };
   407→    document.addEventListener('focusin', onFocusIn);
   408→
   409→    return () => {
   410→      cancelAnimationFrame(raf);
   411→      window.removeEventListener('scroll', onScroll, true);
   412→      window.removeEventListener('resize', onResize);
   413→      document.removeEventListener('mousedown', onPointerDown);
   414→      document.removeEventListener('keydown', onKeyDown);
   415→      document.removeEventListener('focusin', onFocusIn);
   416→    };
   417→  }, [open, flatIndexOf, flatOptions, updatePosition]);
   418→
   419→  // Keep the active (highlighted) option scrolled into view inside the
   420→  // panel whenever it changes (keyboard nav, hover, or initial open).
   421→  useLayoutEffect(() => {
   422→    if (!open) return;
   423→    const el = panelRef.current?.querySelector<HTMLElement>(
   424→      `[data-index="${activeIndex}"]`,
   425→    );
   426→    el?.scrollIntoView({ block: 'nearest' });
   427→  }, [activeIndex, open]);
   428→
   429→  const handleTriggerClick = () => {
   430→    if (disabled) return;
   431→    setOpen((o) => !o);
   432→  };
   433→
   434→  /** Wheel handler attached to the panel itself. We ONLY call
   435→   *  `stopPropagation()` — never `preventDefault()`. This means:
   436→   *    - native scroll on the panel (its `overflow-y:auto`) still works
   437→   *      normally — the wheel target's nearest scrollable ancestor is the
   438→   *      panel, so the browser scrolls the panel;
   439→   *    - the wheel event does not bubble up to any ancestor handler that
   440→   *      might try to scroll the modal body or the page;
   441→   *    - `overscroll-behavior:contain` (set inline on the panel) further
   442→   *      prevents scroll-chaining to the modal body when the panel
   443→   *      reaches its top/bottom boundary.
   444→   *  Mouse wheel over the modal body (outside the panel) is never
   445→   *  intercepted here and continues to scroll the modal body normally. */
   446→  const handleWheel = (e: React.WheelEvent) => {
   447→    e.stopPropagation();
   448→  };
   449→
   450→  return (
   451→    <>
   452→      <button
   453→        ref={triggerRef}
   454→        type="button"
   455→        aria-haspopup="listbox"
   456→        aria-expanded={open}
   457→        disabled={disabled}
   458→        onClick={handleTriggerClick}
   459→        className="flex h-9 w-full items-center justify-between rounded-md border border-input bg-transparent px-3 py-2 text-sm shadow-sm transition-[color,box-shadow] hover:bg-accent/50 focus:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:border-ring disabled:cursor-not-allowed disabled:opacity-50"
   460→      >
   461→        <span className="truncate text-left text-foreground">
   462→          {selected?.name ?? 'Select provider'}
   463→        </span>
   464→        <ChevronDown
   465→          className={cn(
   466→            'h-4 w-4 opacity-50 transition-transform shrink-0',
   467→            open && 'rotate-180',
   468→          )}
   469→        />
   470→      </button>
   471→
   472→      {/* The panel is rendered via createPortal directly onto document.body
   473→          so the modal's overflow/scroll-container is no longer an ancestor.
   474→          It cannot clip the panel and cannot intercept the panel's wheel
   475→          events. The panel only mounts once `panelStyle.top` has been
   476→          computed by the layout effect above (so it never flashes at the
   477→          wrong position). */}
   478→      {open &&
   479→        panelStyle.top !== undefined &&
   480→        createPortal(
   481→          <div
   482→            ref={panelRef}
   483→            role="listbox"
   484→            tabIndex={-1}
   485→            onWheel={handleWheel}
   486→            style={{
   487→              ...panelStyle,
   488→              overflowY: 'auto',
   489→              overscrollBehavior: 'contain',
   490→              // Body is set to `pointer-events: none` while the Radix Dialog
   491→              // is open (Radix's scroll-lock). `pointer-events` is inherited
   492→              // by default, so a panel portaled to `document.body` would
   493→              // inherit `none` and become click/wheel-through. Explicitly
   494→              // re-enable pointer events on the panel so it actually
   495→              // captures mouse wheel and click events.
   496→              pointerEvents: 'auto',
   497→            }}
   498→            className="storage-modal-scroll animate-in fade-in-0 zoom-in-95 slide-in-from-top-2 bg-popover text-popover-foreground rounded-md border shadow-md p-1 outline-none"
   499→          >
   500→            {categorized.map(({ category, label, providers }) => (
   501→              <div key={category} className="py-1">
   502→                <div className="px-2 py-1 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground/70">
   503→                  {label}
   504→                </div>
   505→                {providers.map((p) => {
   506→                  const flatIdx = flatIndexOf.get(p.id) ?? 0;
   507→                  const isSelected = p.id === value;
   508→                  const isActive = flatIdx === activeIndex;
   509→                  return (
   510→                    <button
   511→                      key={p.id}
   512→                      type="button"
   513→                      role="option"
   514→                      aria-selected={isSelected}
   515→                      data-index={flatIdx}
   516→                      onMouseEnter={() => setActiveIndex(flatIdx)}
   517→                      onClick={() => {
   518→                        onChange(p.id);
   519→                        setOpen(false);
   520→                        triggerRef.current?.focus();
   521→                      }}
   522→                      className={cn(
   523→                        'flex w-full items-center justify-between rounded-sm px-2 py-1.5 text-sm outline-none transition-colors',
   524→                        isActive ? 'bg-accent' : 'hover:bg-accent/70',
   525→                        isSelected && 'font-medium',
   526→                      )}
   527→                    >
   528→                      <span className="truncate">{p.name}</span>
   529→                      {isSelected && (
   530→                        <Check className="h-4 w-4 text-primary shrink-0" />
   531→                      )}
   532→                    </button>
   533→                  );
   534→                })}
   535→              </div>
   536→            ))}
   537→          </div>,
   538→          document.body,
   539→        )}
   540→    </>
   541→  );
   542→}

   559→// -------------------- Validation helpers --------------------
   560→
   561→/** True when the Name field is non-empty AND every required field for the
   562→ *  currently-selected provider has a non-empty value. For providers that
   563→ *  require a validated connection, the connection state must also be
   564→ *  'connected'. */
   565→function isFormValid(form: StorageForm): boolean {
   566→  if (!form.name.trim()) return false;
   567→  const def = getProviderDefinition(form.provider);
   568→  if (!def) return false;
   569→
   570→  const allFieldsFilled = def.fields.every((f) => {
   571→    if (!f.required) return true;
   572→    const v = form.config[f.key];
   573→    if (v == null) return false;
   574→    return f.type === 'switch' ? true : v.trim() !== '';
   575→  });
   576→  if (!allFieldsFilled) return false;
   577→
   578→  if (def.requiresConnection && form.connection !== 'connected') return false;
   579→  return true;
   580→}
   581→
   582→/** Returns the inline error message for a single field, or '' when the
   583→ *  field is valid. Required fields that are empty show their message only
   584→ *  after the user has blurred the field OR attempted to submit — we never
   585→ *  surface a generic aggregate "N issues" summary. The Name field follows
   586→ *  the same rule. */
   587→function fieldError(
   588→  form: StorageForm,
   589→  field: { key: string; label: string; required: boolean; type: string },
   590→  submitAttempted: boolean,
   591→): string {
   592→  if (!field.required) return '';
   593→  if (field.type === 'switch') return '';
   594→  const v = form.config[field.key];
   595→  const isEmpty = v == null || v.trim() === '';
   596→  if (!isEmpty) return '';
   597→  const wasTouched = form.touched[field.key] === true;
   598→  if (!wasTouched && !submitAttempted) return '';
   599→  return `${field.label} is required`;
   600→}
   601→
   602→/** Signature for the connection/credential-group fields of the current
   603→ *  provider. Used to detect whether the user has edited any such field
   604→ *  since the last Test/Connect — if so, the 'connected' state is stale. */
   605→function connectionSignature(form: StorageForm): string {
   606→  const def = getProviderDefinition(form.provider);
   607→  if (!def) return '';
   608→  const groups: FieldGroup[] =
   609→    def.connectionType === 'oauth' ? ['connection'] : ['credentials'];
   610→  return def.fields
   611→    .filter((f) => groups.includes(f.group))
   612→    .map((f) => `${f.key}=${form.config[f.key] ?? ''}`)
   613→    .join('|');
   614→}
   615→
   616→/** True when all required connection/credential fields are filled, so the
   617→ *  Test/Connect button can be enabled. */
   618→function canValidateConnection(form: StorageForm): boolean {
   619→  const def = getProviderDefinition(form.provider);
   620→  if (!def || def.connectionType === 'none') return false;
   621→  const groups: FieldGroup[] =
   622→    def.connectionType === 'oauth' ? ['connection'] : ['credentials'];
   623→  return def.fields
   624→    .filter((f) => groups.includes(f.group))
   625→    .every((f) => {
   626→      if (!f.required) return true;
   627→      const v = form.config[f.key];
   628→      return v != null && v.trim() !== '';
   629→    });
   630→}
   631→
   632→/** Detects password-type connection/credential fields still holding the
   633→ *  API mask placeholder ('••••••••'). When editing an existing storage,
   634→ *  secrets come back masked; re-testing with a mask sends a bogus secret.
   635→ *  We block the test and ask the user to re-enter the secret. */
   636→function maskedSecretFields(form: StorageForm): string[] {
   637→  const def = getProviderDefinition(form.provider);
   638→  if (!def) return [];
   639→  const groups: FieldGroup[] =
   640→    def.connectionType === 'oauth' ? ['connection'] : ['credentials'];
   641→  return def.fields
   642→    .filter((f) => f.type === 'password' && groups.includes(f.group))
   643→    .map((f) => f.label)
   644→    .filter((label) => label && label.length > 0)
   645→    .filter((_, i) => {
   646→      const f = def.fields
   647→        .filter((x) => x.type === 'password' && groups.includes(x.group))[i];
   648→      const v = form.config[f.key];
   649→      return typeof v === 'string' && v.includes('•');
   650→    });
   651→}
   652→
   653→/** Build the config object from the current provider's fields only. Empty
   654→ *  optional fields are omitted. Number fields → Number. Switch → boolean. */
   655→function buildConfigObject(form: StorageForm): Record<string, unknown> {
   656→  const def = getProviderDefinition(form.provider);
   657→  if (!def) return {};
   658→  const configObj: Record<string, unknown> = {};
   659→  for (const field of def.fields) {
   660→    const raw = form.config[field.key];
   661→    if (raw == null || raw.trim() === '') continue;
   662→    if (field.type === 'number') {
   663→      const n = Number(raw);
   664→      if (!Number.isNaN(n)) configObj[field.key] = n;
   665→    } else if (field.type === 'switch') {
   666→      configObj[field.key] = raw === 'true';
   667→    } else {
   668→      configObj[field.key] = raw;
   669→    }
   670→  }
   671→  return configObj;
   672→}

  1472→// -------------------- Config Field Renderer --------------------
  1473→// Renders a single provider field generically. Password fields mask
  1474→// their value (type=password + a lock affordance). Switch fields use
  1475→// the Switch component. Multiline password fields (e.g. GCS private
  1476→// key) render as a masked Textarea. The component is fully driven by
  1477→// the ProviderField definition — it has no provider-specific logic.
  1478→
  1479→function ConfigField({
  1480→  field,
  1481→  value,
  1482→  onChange,
  1483→  onBlur,
  1484→  error,
  1485→  disabled,
  1486→}: {
  1487→  field: ProviderField;
  1488→  value: string;
  1489→  onChange: (v: string) => void;
  1490→  onBlur?: () => void;
  1491→  error?: string;
  1492→  disabled?: boolean;
  1493→}) {
  1494→  const inputId = `storage-config-${field.key}`;
  1495→  const showError = !!error;
  1496→  return (
  1497→    <div className="space-y-1.5">
  1498→      <Label htmlFor={inputId} className="text-xs">
  1499→        {field.label}
  1500→        {field.required && <span className="text-destructive ml-0.5">*</span>}
  1501→      </Label>
  1502→      {field.type === 'switch' ? (
  1503→        <div className="flex items-center gap-2 h-9">
  1504→          <Switch
  1505→            id={inputId}
  1506→            checked={value === 'true'}
  1507→            onCheckedChange={(checked) => onChange(checked ? 'true' : 'false')}
  1508→            disabled={disabled}
  1509→          />
  1510→          <span className="text-xs text-muted-foreground">
  1511→            {value === 'true' ? 'Enabled' : 'Disabled'}
  1512→          </span>
  1513→        </div>
  1514→      ) : field.type === 'password' && field.multiline ? (
  1515→        <Textarea
  1516→          id={inputId}
  1517→          placeholder={field.placeholder}
  1518→          value={value}
  1519→          onChange={(e) => onChange(e.target.value)}
  1520→          onBlur={onBlur}
  1521→          rows={4}
  1522→          disabled={disabled}
  1523→          aria-invalid={showError || undefined}
  1524→          className={cn('font-mono text-xs', showError && 'border-destructive focus-visible:ring-destructive')}
  1525→        />
  1526→      ) : (
  1527→        <div className="relative">
  1528→          <Input
  1529→            id={inputId}
  1530→            type={field.type === 'password' ? 'password' : field.type === 'number' ? 'number' : 'text'}
  1531→            placeholder={field.placeholder}
  1532→            value={value}
  1533→            onChange={(e) => onChange(e.target.value)}
  1534→            onBlur={onBlur}
  1535→            disabled={disabled}
  1536→            aria-invalid={showError || undefined}
  1537→            autoComplete={field.type === 'password' ? 'new-password' : undefined}
  1538→            className={cn(field.type === 'password' ? 'pr-9' : '', showError && 'border-destructive focus-visible:ring-destructive')}
  1539→          />
  1540→          {field.type === 'password' && value && (
  1541→            <Lock className="absolute right-3 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground/60 pointer-events-none" />
  1542→          )}
  1543→        </div>
  1544→      )}
  1545→      {showError ? (
  1546→        <p className="text-xs text-destructive">{error}</p>
  1547→      ) : field.helpText ? (
  1548→        <p className="text-xs text-muted-foreground leading-relaxed">{field.helpText}</p>
  1549→      ) : null}
  1550→    </div>
  1551→  );
  1552→}
