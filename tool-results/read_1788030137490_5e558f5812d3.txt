     1→// ============================================================
     2→// PLATFORM DATA SERVICE — Centralized SaaS platform dataset
     3→// ============================================================
     4→// This is the SINGLE source of truth for the entire SaaS platform.
     5→// Both the Client Billing experience AND the Platform Admin
     6→// Dashboard read from and mutate this same in-memory dataset, so
     7→// a plan change, cancellation, or payment made on the client side
     8→// is immediately reflected on the admin side, and vice-versa.
     9→//
    10→// Consistency rules (enforced by derivation, never hardcoded):
    11→//  - totalCustomers  = customers.length (non-deleted)
    12→//  - activeSubscriptions = subscriptions where status === 'active'
    13→//  - mrr = sum of normalized monthly price over active PAID subs
    14→//  - totalSites = sites.length
    15→//  - planDistribution = subscriptions grouped by plan (all statuses)
    16→//  - statusCounts = subscriptions grouped by status
    17→//  - recentCustomers = customers sorted by createdAt desc
    18→//  - recentPayments = payments sorted by date desc
    19→//  - usage = aggregated from sites/content/media/AI records
    20→//  - alerts = derived from failed payments + past-due + storage
    21→//
    22→// Mock data is deterministic (no Math.random) so numbers are stable
    23→// across renders and never contradict each other.
    24→// ============================================================

   150→// -------------------- Plans (delegated to DB-backed plan-config) --------------------
   151→// The editable plan definitions live in the PlanConfig table and are cached
   152→// by ./plan-config — the SINGLE source of truth. The owner edits there via
   153→// the Platform Admin; this module (admin overview + client billing) and the
   154→// client billing API both read the same cached values, so an owner price
   155→// change propagates to the client billing page and to MRR on the next read.
   156→// The PLANS array below is a live view re-spliced whenever the config changes.
   157→
   158→import { getPlanConfigsSync, subscribe as subscribePlanConfig, type PlanConfigData } from './plan-config';
   159→
   160→function toPlan(d: PlanConfigData): Plan {
   161→  return {
   162→    id: d.planId as PlanId,
   163→    name: d.name,
   164→    price: d.priceMonthly,
   165→    priceMonthly: d.priceMonthly,
   166→    priceYearly: d.priceYearly,
   167→    currency: d.currency,
   168→    interval: d.interval,
   169→    isFree: d.isFree,
   170→    active: d.active,
   171→    features: d.features,
   172→    entitlements: d.entitlements,
   173→    limits: d.limits,
   174→  };
   175→}
   176→
   177→/** Live view of the DB-backed plan configs. Mutated in place on every
   178→ *  config refresh so existing references stay valid. */
   179→export const PLANS: Plan[] = [];
   180→
   181→function refreshPLANS(): void {
   182→  const data = getPlanConfigsSync();
   183→  PLANS.length = 0;
   184→  for (const d of data) PLANS.push(toPlan(d));
   185→}
   186→refreshPLANS();
   187→subscribePlanConfig(refreshPLANS);
   188→
   189→export function getPlan(id: PlanId): Plan {
   190→  return PLANS.find((p) => p.id === id) ?? PLANS[0];
   191→}
   192→
   193→/** Normalized monthly price. Yearly subscriptions contribute priceYearly/12
   194→ *  so MRR is a true monthly run-rate. */
   195→export function monthlyPrice(plan: Plan, interval: BillingInterval): number {
   196→  if (interval === 'yearly') return Math.round(plan.priceYearly / 12);
   197→  return plan.priceMonthly;
   198→}

   205→function isoDaysAgo(days: number): string {
   206→  const d = new Date();
   207→  d.setDate(d.getDate() - days);
   208→  d.setHours(9, 15, 0, 0);
   209→  return d.toISOString();
   210→}
   211→function isoDaysAhead(days: number): string {
   212→  const d = new Date();
   213→  d.setDate(d.getDate() + days);
   214→  d.setHours(9, 15, 0, 0);
   215→  return d.toISOString();
   216→}

   390→function buildRevenueSeries(currentMrr: number): RevenuePoint[] {
   391→  const trend = [0.41, 0.47, 0.54, 0.62, 0.71, 0.80, 0.89, 1.0];
   392→  const out: RevenuePoint[] = [];
   393→  const now = new Date();
   394→  for (let i = 7; i >= 0; i--) {
   395→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   396→    const factor = trend[7 - i];
   397→    out.push({ month: MONTHS[d.getMonth()], revenue: Math.round(currentMrr * factor) });
   398→  }
   399→  return out;
   400→}

   417→function createStore(): PlatformStore {
   418→  const customers: Customer[] = CUSTOMER_SEED.map((c, i) => ({
   419→    ...c,
   420→    id: `cus_${String(i + 1).padStart(3, '0')}`,

   431→    media: s.media,
   432→    storageBytes: s.storageBytes,
   433→    createdAt: isoDaysAgo(s.daysAgo),
   434→  }));
   435→
   436→  const payments: Payment[] = PAYMENT_SEED.map((p, i) => {
   437→    const c = customers[p.customerIdx];
   438→    return {
   439→      id: `pay_${String(i + 1).padStart(3, '0')}`,
   440→      customerId: c.id,
   441→      planId: p.planId,
   442→      amount: p.amount,
   443→      currency: 'CHF',

   445→      method: p.method,
   446→      date: isoDaysAgo(p.daysAgo),
   447→      invoiceNumber: `INV-2026-${String(1000 + i)}`,

   452→    ...a,
   453→    id: `aud_${String(i + 1).padStart(3, '0')}`,
   454→  }));
   455→
   456→  _auditCounter = audit.length;
   457→  _paymentCounter = payments.length;
   458→
   459→  return { customers, sites, payments, audit };

   627→export async function listPayments(opts?: { status?: PaymentStatus | 'all'; search?: string }): Promise<(Payment & { customerName: string; customerEmail: string })[]> {
   628→  // REAL DB-backed (replaces the in-memory PAYMENT_SEED mock). The admin
   629→  // Payments table now reads the live Payment table, joined to the User
   630→  // (customer) + Subscription. Stripe webhooks (invoice.paid /
   631→  // checkout.session.completed / payment_intent.*) and the dev seed
   632→  // script write these rows, so this is the actual relational state.
   633→  //
   634→  // Server-side filtering:
   635→  //   - status: exact match on Payment.status
   636→  //   - search: case-insensitive substring over payment id, invoice
   637→  //     number, Stripe invoice/payment-intent/charge IDs, and the
   638→  //     customer's name + email (via the User join).
   639→  //
   640→  // Returns the SAME shape the UI has always consumed
   641→  // (Payment & { customerName, customerEmail }) so the Payments page is
   642→  // unchanged — only the data source switched from mock to real DB. The
   643→  // summary values (total payments, paid count, paid total) are derived
   644→  // client-side from this array by the Payments page (already dynamic).
   645→  const { db } = await import('@/lib/db');
   646→
   647→  const where: { status?: string; OR?: unknown[] } = {};
   648→  if (opts?.status && opts.status !== 'all') {
   649→    where.status = opts.status;
   650→  }
   651→  if (opts?.search) {
   652→    const q = opts.search;
   653→    where.OR = [
   654→      { id: { contains: q } },
   655→      { invoiceNumber: { contains: q } },
   656→      { stripeInvoiceId: { contains: q } },
   657→      { stripePaymentIntentId: { contains: q } },
   658→      { stripeChargeId: { contains: q } },
   659→      { user: { OR: [{ name: { contains: q } }, { email: { contains: q } }] } },
   660→    ];
   661→  }
   662→
   663→  const rows = await db.payment.findMany({
   664→    where,
   665→    include: { user: { select: { id: true, name: true, email: true } }, subscription: true },
   666→    orderBy: [{ paidAt: 'desc' }, { createdAt: 'desc' }],
   667→    take: 500,
   668→  });
   669→
   670→  const mapped = rows.map((r) => mapPaymentRow(r));
   671→  // Precise effective-date sort (paidAt ?? createdAt desc) so pending /
   672→  // failed rows (paidAt null) interleave by their createdAt.

   677→/** Normalize a Prisma Payment row (with user included) to the public
   678→ *  Payment interface the admin UI consumes. The shape is identical to
   679→ *  the legacy mock Payment so the UI is unchanged; only the data source
   680→ *  switched to the real DB. */
   681→function mapPaymentRow(r: {
   682→  id: string;
   683→  userId: string;
   684→  planId: string;
   685→  amount: number;
   686→  currency: string;
   687→  status: string;
   688→  method: string | null;
   689→  invoiceNumber: string | null;
   690→  stripeInvoiceId: string | null;
   691→  stripePaymentIntentId: string | null;
   692→  stripeChargeId: string | null;
   693→  paymentMethodType: string | null;
   694→  paymentMethodDetails: string | null;
   695→  paidAt: Date | null;
   696→  createdAt: Date;
   697→  user: { id: string; name: string | null; email: string } | null;
   698→}): Payment & { customerName: string; customerEmail: string } {
   699→  const planId = (['free', 'plus', 'pro', 'max'].includes(r.planId) ? r.planId : 'free') as PlanId;
   700→  const status = (['paid', 'pending', 'failed', 'refunded'].includes(r.status)
   701→    ? r.status
   702→    : 'pending') as PaymentStatus;
   703→  return {
   704→    id: r.id,
   705→    customerId: r.userId,
   706→    planId,
   707→    amount: r.amount,
   708→    currency: r.currency,
   709→    status,
   710→    method: r.method ?? deriveMethodLabel(r.paymentMethodType, r.paymentMethodDetails) ?? '—',
   711→    date: (r.paidAt ?? r.createdAt).toISOString(),
   712→    invoiceNumber: r.invoiceNumber ?? r.stripeInvoiceId ?? '—',
   713→    // Stripe transaction details — surfaced in the admin Payments table
   714→    // so every row is verifiably tied to a real Stripe PaymentIntent /
   715→    // Charge / Invoice. Null/undefined when Stripe isn't configured.
   716→    stripeInvoiceId: r.stripeInvoiceId,
   717→    stripePaymentIntentId: r.stripePaymentIntentId,
   718→    stripeChargeId: r.stripeChargeId,
   719→    customerName: r.user?.name ?? r.user?.email ?? '—',
   720→    customerEmail: r.user?.email ?? '—',
   721→  };
   722→}

   724→/** Build a human-readable payment-method label from the stored metadata,
   725→ *  e.g. "Visa ••4242". Falls back to the capitalized type. */
   726→function deriveMethodLabel(type: string | null, detailsJson: string | null): string | null {
   727→  if (!type) return null;
   728→  if (type === 'card' && detailsJson) {
   729→    try {
   730→      const d = JSON.parse(detailsJson) as { brand?: string; last4?: string };
   731→      if (d.brand && d.last4) return `${capitalize(d.brand)} ••${d.last4}`;
   732→      if (d.brand) return capitalize(d.brand);
   733→    } catch {
   734→      // ignore malformed JSON
   735→    }
   736→  }
   737→  return capitalize(type);
   738→}

   740→function capitalize(s: string): string {
   741→  return s.charAt(0).toUpperCase() + s.slice(1);

  1104→ *
  1105→ * For users WITH a DB Subscription row:
  1106→ *   - status, billingInterval, currentPeriodEnd, trialEnd come from the DB.
  1107→ *   - freeTrialExpired is computed from trialEnd + plan.isFree + status.
  1108→ *   - paymentHistory comes from the DB Payment table (real Stripe invoices
  1109→ *     when Stripe is configured; empty for free plans).

  1112→ *   - Falls back to getClientBilling() (in-memory customer or Free plan).

  1114→export async function getClientBillingAsync(user: BillingUser): Promise<ClientBillingState> {
  1115→  // Bypass users: return the synthetic Internal plan (no DB lookup).
  1116→  if (isBillingBypass(user)) return getClientBilling(user);
  1117→
  1118→  // No user id → can't look up DB Subscription. Fall back to legacy.
  1119→  if (!user.id) return getClientBilling(user);
  1120→
  1121→  // Try the DB Subscription first.
  1122→  const subState = await getClientSubscriptionState({
  1123→    id: user.id,
  1124→    email: user.email,
  1125→    role: user.role ?? '',
  1126→    billingMode: user.billingMode,
  1127→  });
  1128→
  1129→  // No DB subscription row → fall back to the in-memory legacy state.
  1130→  if (!subState.subscription) {
  1131→    return getClientBilling(user);
  1132→  }
  1133→

  1135→  // (NOT from the in-memory customer — the DB is the source of truth).
  1136→  const plan = toPlan(subState.plan);
  1137→
  1138→  // Load real payment history from the DB Payment table.
  1139→  const dbPayments = await getUserPayments(user.id);

  1142→    customer: null, // no in-memory customer when DB subscription is the source
  1143→    plan,

  1162→    currentPeriodEnd: subState.currentPeriodEnd,
  1163→    freeTrialExpiresAt: subState.freeTrialExpiresAt,

  1183→/**
  1184→ * Change plan — for FREE plans this writes a real DB Subscription row
  1185→ * (with the configured free-trial duration if any). For PAID plans
  1186→ * this function REFUSES and the caller must redirect the user to
  1187→ * /api/billing/checkout (Stripe). This is the only way to change plan
  1188→ * without going through Stripe — and it ONLY works for free plans.
  1189→ *
  1190→ * For backwards compatibility with the legacy in-memory demo customers,
  1191→ * if the user matches a CUSTOMER_SEED entry we ALSO mutate that entry
  1192→ * (so the Platform Admin Customers page stays consistent). The DB
  1193→ * Subscription row is the source of truth for the Client Billing page.
  1194→ */
  1195→export async function clientChangePlan(
  1196→  user: BillingUser,
  1197→  newPlanId: PlanId,
  1198→): Promise<ClientBillingState | null> {
  1199→  if (isBillingBypass(user)) return getClientBilling(user); // bypass: no plan change
  1200→  if (!user.id) return null; // no user id → can't write DB Subscription
  1201→
  1202→  // Look up the target plan from the live cache.
  1203→  const target = getPlan(newPlanId);
  1204→  if (!target.active) return null;
  1205→
  1206→  // For FREE plans: write a DB Subscription row via subscribeToFreePlan.
  1207→  if (target.isFree) {
  1208→    const result = await subscribeToFreePlan(user.id, newPlanId);
  1209→    if (!result.ok) return null;
  1210→  } else {
  1211→    // PAID plan: REFUSE. The user must use /api/billing/checkout (Stripe).
  1212→    // Returning null here signals "must use checkout". The Client Billing
  1213→    // UI detects this and shows the Upgrade button → /api/billing/checkout.
  1214→    return null;
  1215→  }
  1216→
  1217→  // Also mutate the in-memory demo customer (if one exists for this email)
  1218→  // so the Platform Admin Customers page stays consistent with the new state.
  1219→  const s = store();
  1220→  const c = s.customers.find((cu) => cu.email.toLowerCase() === user.email.toLowerCase());
  1221→  if (c) changeCustomerPlan(c.id, newPlanId, user.email);
  1222→
  1223→  return getClientBillingAsync(user);
  1224→}
  1225→
  1226→/**
  1227→ * Cancel subscription — writes to the DB Subscription row (the source
  1228→ * of truth) and also mutates the legacy in-memory demo customer if one
  1229→ * exists. Returns the updated ClientBillingState.
  1230→ */
  1231→export async function clientCancelSubscription(user: BillingUser): Promise<ClientBillingState | null> {
  1232→  if (isBillingBypass(user)) return getClientBilling(user); // bypass: nothing to cancel
  1233→  if (!user.id) return null;
  1234→
  1235→  // Cancel the DB Subscription (the authoritative record).
  1236→  await cancelDbSubscription(user.id);
  1237→
  1238→  // Also mutate the in-memory demo customer if one exists.
  1239→  const s = store();
  1240→  const c = s.customers.find((cu) => cu.email.toLowerCase() === user.email.toLowerCase());
  1241→  if (c) cancelSubscription(c.id, user.email);
  1242→
  1243→  return getClientBillingAsync(user);
  1244→}

  1246→// ============================================================
  1247→// PLATFORM EVENTS — derived, unified platform-level feed
  1248→// ============================================================
  1249→// Aggregates platform-level events from the SAME centralized
  1250→// dataset that powers Platform Overview / Customers / Payments /
  1251→// Subscriptions / Audit / Alerts. PURE read-only function — no new
  1252→// state, no persistence. The /api/platform/admin/notifications
  1253→// endpoint paginates and filters this list; mark-as-read / delete
  1254→// are documented no-ops on the API side because the feed is derived
  1255→// fresh on every request (read-state would live in a separate
  1256→// table in a real implementation).
  1257→//
  1258→// Event types follow the same NotificationType shape the Client
  1259→// Notifications page uses: INFO | SUCCESS | WARNING | ERROR |
  1260→// ACTION_REQUIRED. This lets the Platform Notifications UI reuse
  1261→// the client page's NotificationCard / NOTIFICATION_TYPE_CONFIG
  1262→// verbatim.
  1263→// ============================================================
  1264→
  1265→export type PlatformEventType = 'INFO' | 'SUCCESS' | 'WARNING' | 'ERROR' | 'ACTION_REQUIRED';
  1266→
  1267→export interface PlatformEvent {
  1268→  id: string;
  1269→  type: PlatformEventType;
  1270→  title: string;
  1271→  message: string;
  1272→  isRead: boolean;
  1273→  createdAt: string;
  1274→}
  1275→
  1276→function isoMinutesAgo(mins: number): string {
  1277→  return new Date(Date.now() - mins * 60_000).toISOString();
  1278→}
  1279→
  1280→/** Map alert id → a deterministic "recent" timestamp (no Math.random,
  1281→ *  stable across renders for a given wall-clock minute). */
  1282→function alertTimestamp(alertId: string): string {
  1283→  switch (alertId) {
  1284→    case 'alert-failed-payments': return isoMinutesAgo(30);
  1285→    case 'alert-past-due': return isoMinutesAgo(120);
  1286→    case 'alert-storage': return isoMinutesAgo(360);
  1287→    case 'alert-new-customers': return isoMinutesAgo(60 * 24);
  1288→    default: return isoMinutesAgo(60);
  1289→  }
  1290→}

  1292→/** Derive the unified platform events feed. Reads from listCustomers,
  1293→ *  listPayments, listSubscriptions, getAuditLog, getAlerts — all of
  1294→ *  which already exist and power the rest of the Platform Admin. */
  1295→export function getPlatformEvents(): PlatformEvent[] {
  1296→  const events: PlatformEvent[] = [];
  1297→
  1298→  // 1. Customer registrations — INFO "New customer registered"
  1299→  for (const c of listCustomers()) {
  1300→    events.push({
  1301→      id: `evt-cust-${c.id}`,
  1302→      type: 'INFO',
  1303→      title: 'New customer registered',
  1304→      message: `${c.name} (${c.email}) signed up from ${c.country} on the ${c.planId.toUpperCase()} plan.`,
  1305→      isRead: false,
  1306→      createdAt: c.createdAt,
  1307→    });
  1308→  }
  1309→
  1310→  // 2. Payments — SUCCESS / ERROR / WARNING / INFO by status
  1311→  for (const p of listPayments()) {
  1312→    let type: PlatformEventType = 'INFO';
  1313→    let title = 'Payment update';
  1314→    if (p.status === 'paid') {
  1315→      type = 'SUCCESS';
  1316→      title = 'Successful payment';
  1317→    } else if (p.status === 'failed') {
  1318→      type = 'ERROR';
  1319→      title = 'Failed payment';
  1320→    } else if (p.status === 'refunded') {
  1321→      type = 'WARNING';
  1322→      title = 'Payment refunded';
  1323→    } else if (p.status === 'pending') {
  1324→      type = 'INFO';
  1325→      title = 'Payment pending';
  1326→    }
  1327→    events.push({
  1328→      id: `evt-pay-${p.id}`,
  1329→      type,
  1330→      title,
  1331→      message: `${p.customerName} — ${p.currency} ${p.amount} via ${p.method}. Invoice ${p.invoiceNumber}. Status: ${p.status}.`,
  1332→      isRead: false,
  1333→      createdAt: p.date,
  1334→    });
  1335→  }
  1336→
  1337→  // 3. Subscriptions — INFO created, WARNING cancelled, ACTION_REQUIRED trial ending
  1338→  for (const s of listSubscriptions()) {
  1339→    const plan = getPlan(s.planId);
  1340→    events.push({
  1341→      id: `evt-sub-${s.id}`,
  1342→      type: 'INFO',
  1343→      title: 'Subscription created',
  1344→      message: `${s.name} subscribed to the ${s.planName} plan (${s.billingInterval} billing, ${plan.currency} ${s.monthlyPrice}/mo).`,
  1345→      isRead: false,
  1346→      createdAt: s.subscriptionStart,
  1347→    });
  1348→
  1349→    if (s.subscriptionStatus === 'cancelled') {
  1350→      // Cancellation happened at some point after creation — there is no
  1351→      // dedicated timestamp in the dataset, so use subscriptionStart as a
  1352→      // stable proxy (still sorts in the past, like a historical event).
  1353→      events.push({
  1354→        id: `evt-sub-cancel-${s.id}`,
  1355→        type: 'WARNING',
  1356→        title: 'Subscription cancelled',
  1357→        message: `${s.name} cancelled the ${s.planName} plan subscription. Access has been revoked at the end of the current period.`,
  1358→        isRead: false,
  1359→        createdAt: s.subscriptionStart,
  1360→      });
  1361→    }
  1362→
  1363→    if (s.subscriptionStatus === 'trial') {
  1364→      const trialEnd = s.trialEnd ?? s.nextBillingAt ?? s.subscriptionStart;
  1365→      events.push({
  1366→        id: `evt-sub-trial-${s.id}`,
  1367→        type: 'ACTION_REQUIRED',
  1368→        title: 'Trial ending soon',
  1369→        message: `${s.name}'s ${s.planName} plan trial ends on ${trialEnd}. Convert to a paid subscription before then to retain access.`,
  1370→        isRead: false,
  1371→        createdAt: trialEnd,
  1372→      });
  1373→    }
  1374→  }
  1375→
  1376→  // 4. Audit log entries — INFO with action as title, detail as message
  1377→  for (const a of getAuditLog(50)) {
  1378→    events.push({
  1379→      id: `evt-aud-${a.id}`,
  1380→      type: 'INFO',
  1381→      title: a.action,
  1382→      message: a.detail && a.detail.length > 0 ? `${a.detail} — target: ${a.target}` : a.target,
  1383→      isRead: false,
  1384→      createdAt: a.timestamp,
  1385→    });
  1386→  }
  1387→
  1388→  // 5. Alerts — ERROR / WARNING / INFO by severity
  1389→  for (const alert of getAlerts()) {
  1390→    let type: PlatformEventType = 'INFO';
  1391→    if (alert.severity === 'critical') type = 'ERROR';
  1392→    else if (alert.severity === 'warning') type = 'WARNING';
  1393→    events.push({
  1394→      id: `evt-alert-${alert.id}`,
  1395→      type,
  1396→      title: alert.title,
  1397→      message: alert.message,
  1398→      isRead: false,
  1399→      createdAt: alertTimestamp(alert.id),
  1400→    });
  1401→  }
  1402→
  1403→  // Newest first — consistent with the Client Notifications feed.
  1404→  return events.sort(
  1405→    (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime(),
  1406→  );
  1407→}
