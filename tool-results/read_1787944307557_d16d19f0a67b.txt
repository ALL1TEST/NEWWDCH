   533→export async function POST(_request: NextRequest) {
   534→  const id = reqId();
   535→
   536→  try {
   537→    // Find a known user to use as createdById
   538→    const firstUser = await db.user.findFirst({ select: { id: true } });
   539→    if (!firstUser) {
   540→      return NextResponse.json(
   541→        { error: { code: 'NO_USER', message: 'No users found in the database. Create a user first.' }, meta: { requestId: id } },
   542→        { status: 400 },
   543→      );
   544→    }
   545→
   546→    const createdById = firstUser.id;
   547→    let seeded = 0;
   548→    let skipped = 0;
   549→
   550→    for (const tpl of DEFAULT_TEMPLATES) {
   551→      const existing = await db.emailTemplate.findUnique({ where: { slug: tpl.slug } });
   552→      if (existing) {
   553→        skipped++;
   554→        continue;
   555→      }
   556→
   557→      await db.emailTemplate.create({
   558→        data: {
   559→          name: tpl.name,
   560→          slug: tpl.slug,
   561→          subject: tpl.subject,
   562→          previewText: tpl.previewText,
   563→          htmlBody: tpl.htmlBody,
   564→          defaultBody: tpl.htmlBody,
   565→          category: tpl.category as 'CUSTOMER_EMAILS' | 'AUTHENTICATION' | 'NEWSLETTER' | 'MARKETING' | 'TRANSACTIONAL' | 'NOTIFICATIONS' | 'BILLING' | 'SYSTEM',
   566→          status: 'ENABLED',
   567→          provider: 'SMTP',
   568→          isSystem: true,
   569→          createdById,
   570→        },
   571→      });
   572→      seeded++;
   573→    }
   574→
   575→    return NextResponse.json({
   576→      data: { seeded, skipped, total: DEFAULT_TEMPLATES.length },
   577→      meta: { requestId: id },
   578→    });
   579→  } catch (error) {
   580→    console.error(`[EMAIL_TEMPLATES:SEED] ${id} —`, error);
   581→    return NextResponse.json(
   582→      { error: { code: 'INTERNAL_ERROR', message: 'Failed to seed email templates' }, meta: { requestId: id } },
   583→      { status: 500 },
   584→    );
   585→  }
   586→}
