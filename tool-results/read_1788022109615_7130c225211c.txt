    99→    99→/** Display symbol for a currency code. Plans & Pricing shows a
   100→   100→ *  symbol (not the ISO text code) per the visual spec — e.g. "$7.50"
   101→   101→ *  not "USD 7.50" / "CHF 7.50". The stored currency code in the DB is
   102→   102→ *  unaffected; this is a display-only mapping. Unknown codes default
   103→   103→ *  to '$' so the price always renders as a compact symbol + number. */
   104→   104→function currencySymbol(code: string): string {
   105→   105→  const map: Record<string, string> = {
   106→   106→    USD: '$',
   107→   107→    EUR: '€',
   108→   108→    GBP: '£',
   109→   109→    JPY: '¥',
   110→   110→    CNY: '¥',
   111→   111→  };
   112→   112→  return map[(code ?? '').toUpperCase()] ?? '$';
   113→   113→}
   114→   114→
   115→   115→/** Format an integer amount with the display symbol, no decimals.
   116→   116→ *  e.g. (90, 'USD') → "$90"; (0, 'CHF') → "$0". */
   117→   117→function formatPriceSymbol(amount: number, currency: string): string {
   118→   118→  return `${currencySymbol(currency)}${amount.toLocaleString(undefined, {
   119→   119→    minimumFractionDigits: 0,
   120→   120→    maximumFractionDigits: 0,
   121→   121→  })}`;
   122→   122→}
   123→   123→
   124→   124→/** Format with 2 decimals — used for the monthly-equivalent of a
   125→   125→ *  yearly price (yearly 90 → "$7.50"). Always two decimals so the
   126→   126→ *  large primary price reads "$7.50" / "$40.83" / "$82.50". */
   127→   127→function formatPriceSymbolMonthlyEquiv(amount: number, currency: string): string {
   128→   128→  return `${currencySymbol(currency)}${amount.toLocaleString(undefined, {
   129→   129→    minimumFractionDigits: 2,
   130→   130→    maximumFractionDigits: 2,
   131→   131→  })}`;
   132→   132→}
