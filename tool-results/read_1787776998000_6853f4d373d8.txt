   446→   446→// -------------------- Validation --------------------
   447→   447→
   448→   448→/** Returns true when the Name field is non-empty AND every required field for
   449→   449→ *  the currently-selected provider has a non-empty value. For providers that
   450→   450→ *  require a validated connection (R2/FTP/OAuth), the connection state must
   451→   451→ *  also be 'connected'. Fields belonging to OTHER providers are never
   452→   452→ *  checked — they were cleared on provider switch. Optional fields
   453→   453→ *  (required: false) are never checked. */
   454→   454→function isFormValid(form: StorageForm): boolean {
   455→   455→  if (!form.name.trim()) return false;
   456→   456→  const config = PROVIDER_CONFIG[form.provider];
   457→   457→  if (!config) return false;
   458→   458→
   459→   459→  // All required fields for this provider must be filled. For OAuth
   460→   460→  // providers, destination fields (e.g. folder) are only reachable after a
   461→   461→  // successful connection, so a 'connected' state implies the connection
   462→   462→  // fields are filled — but we still verify destination required fields.
   463→   463→  const allFieldsFilled = config.fields.every((f) => {
   464→   464→    if (!f.required) return true;
   465→   465→    const v = form.config[f.key];
   466→   466→    if (v == null) return false;
   467→   467→    return f.type === 'switch' ? true : v.trim() !== '';
   468→   468→  });
   469→   469→  if (!allFieldsFilled) return false;
   470→   470→
   471→   471→  if (config.requiresConnection && form.connection !== 'connected') {
   472→   472→    return false;
   473→   473→  }
   474→   474→  return true;
   475→   475→}
   476→   476→
   477→   477→/** Compute a signature for the connection-group fields of the current
   478→   478→ *  provider. Used to detect whether the user has edited any connection
   479→   479→ *  field since the last Test/Connect — if so, the previous 'connected'
   480→   480→ *  state is invalidated and they must re-validate. */
   481→   481→function connectionSignature(form: StorageForm): string {
   482→   482→  const config = PROVIDER_CONFIG[form.provider];
   483→   483→  if (!config) return '';
   484→   484→  return config.fields
   485→   485→    .filter((f) => f.group === 'connection')
   486→   486→    .map((f) => `${f.key}=${form.config[f.key] ?? ''}`)
   487→   487→    .join('|');
   488→   488→}
   489→   489→
   490→   490→/** True when all required connection-group fields are filled, so the
   491→   491→ *  Test/Connect button can be enabled. */
   492→   492→function canValidateConnection(form: StorageForm): boolean {
   493→   493→  const config = PROVIDER_CONFIG[form.provider];
   494→   494→  if (!config || config.connectionType === 'none') return false;
   495→   495→  return config.fields
   496→   496→    .filter((f) => f.group === 'connection')
   497→   497→    .every((f) => {
   498→   498→      if (!f.required) return true;
   499→   499→      const v = form.config[f.key];
   500→   500→      return v != null && v.trim() !== '';
   501→   501→    });
   502→   502→}

   827→   827→      } else {

   890→   890→      setForm((prev) => ({ ...prev, connection: 'failed', connectionMessage: message }));

   895→   895→  // ---- OAuth Connect (Google Drive/Dropbox/OneDrive create flow) ----
   896→   896→  // OAuth providers can't complete a real authorization roundtrip in this
   897→   897→  // sandbox (no public callback URL), so the "Connect" button validates
   898→   898→  // that all required OAuth credential fields are present and marks the
   899→   899→  // connection as configured. This is the explicit "integration-ready
   900→   900→  // structure" allowed by the task: the stored config (client ID, client
   901→   901→  // secret, refresh token) is exactly what the backup service needs to run
   902→   902→  // the real OAuth refresh against the provider in a production
   903→   903→  // environment. The label says "Credentials configured" rather than
   904→   904→  // pretending a live OAuth session was established.
