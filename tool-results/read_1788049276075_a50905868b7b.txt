     3→// ============================================================
     4→// PLATFORM STRIPE SETTINGS — connect the platform's Stripe
     5→// account from the admin UI.
     6→// ============================================================
     7→// Owner-only. Lets the Platform Owner:
     8→//   - Switch between Test and Live mode (separate credentials per mode).
     9→//   - Enter the Stripe Secret Key, Publishable Key, and Webhook
    10→//     Signing Secret for each mode. Secret keys are AES-256-GCM
    11→//     encrypted in the DB and NEVER returned to the frontend in
    12→//     plaintext — only the masked form is shown (sk_...xxxx).
    13→//   - Set the public app URL (for Stripe Checkout success/cancel
    14→//     redirects).
    15→//   - "Test Connection" pings Stripe's /v1/balance + /v1/account
    16→//     endpoints with a temporary client (read-only — no side
    17→//     effects) and records the outcome on the singleton row.
    18→//   - See the Webhook Endpoint URL to copy into the Stripe
    19→//     dashboard so Stripe fires events to this platform.
    20→//   - See the last Test Connection outcome + timestamp.
    21→//
    22→// Backend: /api/platform/admin/stripe/settings (GET+PUT),
    23→//          /api/platform/admin/stripe/test-connection (POST).
    24→//
    25→// This page is FINANCIAL SETTINGS ONLY. Customer / payment /
    26→// subscription / refund management lives on the Customers /
    27→// Payments pages — not duplicated here.
    28→// ============================================================

    71→// -------------------- Types --------------------
    72→
    73→interface StripeSettingsView {
    74→  mode: 'test' | 'live';
    75→  secretKeyTestMasked: string;
    76→  secretKeyLiveMasked: string;
    77→  publishableKeyTest: string;
    78→  publishableKeyLive: string;
    79→  webhookSecretTestMasked: string;
    80→  webhookSecretLiveMasked: string;
    81→  appUrl: string;
    82→  lastTestStatus: 'success' | 'error' | null;
    83→  lastTestedAt: string | null;
    84→  lastTestErrorMessage: string | null;
    85→  hasSecretKeyTest: boolean;
    86→  hasSecretKeyLive: boolean;
    87→  hasWebhookSecretTest: boolean;
    88→  hasWebhookSecretLive: boolean;
    89→  hasPublishableKeyTest: boolean;
    90→  hasPublishableKeyLive: boolean;
    91→  activeSource: 'db' | 'env' | 'none';
    92→  isConfigured: boolean;
    93→  webhookUrlHint: string;
    94→}
    95→
    96→interface TestResult {
    97→  success: boolean;
    98→  mode: 'test' | 'live';
    99→  accountInfo?: {
   100→    id: string;
   101→    type: string;
   102→    country: string;
   103→    email: string | null;
   104→    displayName: string | null;
   105→  };
   106→  code?: string;
   107→  message?: string;
   108→}

   110→// -------------------- Page (loader shell) --------------------
   111→
   112→export function PlatformStripeSettingsModule() {
   113→  const queryClient = useQueryClient();
   114→
   115→  // ---------- Load saved settings ----------
   116→  const { data, isLoading, isError, refetch } = useQuery<StripeSettingsView>({
   117→    queryKey: ['platform', 'stripe-settings'],
   118→    queryFn: () => getApi<StripeSettingsView>('/api/platform/admin/stripe/settings'),
   119→    staleTime: 10_000,
   120→    refetchOnWindowFocus: false,
   121→  });
   122→
   123→  if (isLoading) {
   124→    return (
   125→      <div className="space-y-6">
   126→        <PlatformPageHeader
   127→          title="Stripe Settings"
   128→          subtitle="Connect your Stripe account to enable real subscription billing."
   129→        />
   130→        <Card>
   131→          <CardContent className="p-6 flex items-center justify-center h-64">
   132→            <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   133→          </CardContent>
   134→        </Card>
   135→      </div>
   136→    );
   137→  }
   138→
   139→  if (isError || !data) {
   140→    return (
   141→      <div className="space-y-6">
   142→        <PlatformPageHeader
   143→          title="Stripe Settings"
   144→          subtitle="Connect your Stripe account to enable real subscription billing."
   145→        />
   146→        <Card>
   147→          <CardContent className="p-6">
   148→            <ErrorState message="Could not load Stripe settings. Please retry." onRetry={() => refetch()} />
   149→          </CardContent>
   150→        </Card>
   151→      </div>
   152→    );
   153→  }
   154→
   155→  // Render the form once data is available — `key` ensures the form
   156→  // remounts (re-initializing useState) when the saved settings change
   157→  // after a save / test-connection / external mutation. This avoids the
   158→  // setState-in-effect anti-pattern.
   159→  return (
   160→    <StripeSettingsForm
   161→      key={data.lastTestedAt ?? 'fresh'}
   162→      data={data}
   163→      queryClient={queryClient}
   164→    />
   165→  );
   166→}

   168→// -------------------- Form (mounted when data is available) --------------------
   169→
   170→function StripeSettingsForm({
   171→  data,
   172→  queryClient,
   173→}: {
   174→  data: StripeSettingsView;
   175→  queryClient: ReturnType<typeof useQueryClient>;
   176→}) {
   177→  // ---------- Local form state (initialized ONCE from data on mount) ----------
   178→  const [mode, setMode] = useState<'test' | 'live'>(data.mode);
   179→  const [secretKeyTest, setSecretKeyTest] = useState(''); // never pre-fill secret keys
   180→  const [secretKeyLive, setSecretKeyLive] = useState('');
   181→  const [publishableKeyTest, setPublishableKeyTest] = useState(data.publishableKeyTest || '');
   182→  const [publishableKeyLive, setPublishableKeyLive] = useState(data.publishableKeyLive || '');
   183→  const [webhookSecretTest, setWebhookSecretTest] = useState('');
   184→  const [webhookSecretLive, setWebhookSecretLive] = useState('');
   185→  const [appUrl, setAppUrl] = useState(data.appUrl || '');
   186→  const [showSecretTest, setShowSecretTest] = useState(false);
   187→  const [showSecretLive, setShowSecretLive] = useState(false);
   188→  const [showWebhookTest, setShowWebhookTest] = useState(false);
   189→  const [showWebhookLive, setShowWebhookLive] = useState(false);
   190→  const [copied, setCopied] = useState(false);
   191→
   192→  // ---------- Save mutation ----------
   193→  const saveMutation = useMutation({
   194→    mutationFn: async () => {
   195→      // Empty string = "leave unchanged" for secret keys + webhook secrets.
   196→      // Publishable keys are written as plaintext (non-secret) — empty
   197→      // means "clear" since the admin can see the current value.
   198→      return putApi<StripeSettingsView>('/api/platform/admin/stripe/settings', {
   199→        mode,
   200→        secretKeyTest: secretKeyTest || undefined,
   201→        secretKeyLive: secretKeyLive || undefined,
   202→        publishableKeyTest: publishableKeyTest || undefined,
   203→        publishableKeyLive: publishableKeyLive || undefined,
   204→        webhookSecretTest: webhookSecretTest || undefined,
   205→        webhookSecretLive: webhookSecretLive || undefined,
   206→        appUrl: appUrl || undefined,
   207→      });
   208→    },
   209→    onSuccess: () => {
   210→      toast.success('Stripe settings saved.');
   211→      queryClient.invalidateQueries({ queryKey: ['platform', 'stripe-settings'] });
   212→      // Clear the secret-key inputs after a successful save — the admin
   213→      // must re-enter the key to view/modify it (defensive: prevents
   214→      // stale plaintext from lingering in the form).
   215→      setSecretKeyTest('');
   216→      setSecretKeyLive('');
   217→      setWebhookSecretTest('');
   218→      setWebhookSecretLive('');
   219→    },
   220→    onError: (err: unknown) => {
   221→      const e = err as { code?: string; message?: string };
   222→      toast.error(e?.message || 'Unable to save Stripe settings.');
   223→    },
   224→  });
   225→
   226→  // ---------- Test connection mutation ----------
   227→  // Tests the UNSAVED credentials when secret-key fields are populated;
   228→  // falls back to the saved/stored credentials when they're empty.
   229→  const testMutation = useMutation({
   230→    mutationFn: async () => {
   231→      return postApi<TestResult>('/api/platform/admin/stripe/test-connection', {
   232→        mode,
   233→        secretKeyTest: secretKeyTest || undefined,
   234→        secretKeyLive: secretKeyLive || undefined,
   235→      });
   236→    },
   237→    onSuccess: (res) => {
   238→      if (res.success) {
   239→        toast.success(
   240→          `Stripe ${res.mode.toUpperCase()} connection successful. Account: ${res.accountInfo?.id ?? 'unknown'} (${res.accountInfo?.country ?? '?'})`,
   241→        );
   242→      } else {
   243→        toast.error(`Connection failed (${res.code}): ${res.message}`);
   244→      }
   245→      queryClient.invalidateQueries({ queryKey: ['platform', 'stripe-settings'] });
   246→    },
   247→    onError: (err: unknown) => {
   248→      const e = err as { code?: string; message?: string };
   249→      toast.error(e?.message || 'Unable to test Stripe connection.');
   250→    },
   251→  });
   252→
   253→  // ---------- Copy webhook URL ----------
   254→  const copyWebhookUrl = async () => {
   255→    if (!data.webhookUrlHint) return;
   256→    try {
   257→      await navigator.clipboard.writeText(data.webhookUrlHint);
   258→      setCopied(true);
   259→      setTimeout(() => setCopied(false), 2000);
   260→      toast.success('Webhook URL copied to clipboard.');
   261→    } catch {
   262→      toast.error('Could not copy. Select and copy manually.');
   263→    }
   264→  };
   265→
   266→  // -------------------- Render --------------------
   267→
   268→  return (
   269→    <div className="space-y-6">
   270→      <PlatformPageHeader
   271→        title="Stripe Settings"
   272→        subtitle="Connect your Stripe account to enable real subscription billing."
   273→        actions={
   274→          <Button
   275→            variant="outline"
   276→            size="sm"
   277→            onClick={() => testMutation.mutate()}
   278→            disabled={testMutation.isPending}
   279→          >
   280→            {testMutation.isPending ? (
   281→              <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   282→            ) : (
   283→              <Plug className="h-4 w-4 mr-2" />
   284→            )}
   285→            Test Connection
   286→          </Button>
   287→        }
   288→      />
   289→
   290→      {/* -------------------- Status banner -------------------- */}
   291→      <Card>
   292→        <CardContent className="p-4">
   293→          <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   294→            <div className="flex items-center gap-3">
   295→              {data.isConfigured ? (
   296→                <ShieldCheck className="h-5 w-5 text-emerald-600" />
   297→              ) : (
   298→                <ShieldAlert className="h-5 w-5 text-amber-600" />
   299→              )}
   300→              <div>
   301→                <p className="text-sm font-medium">
   302→                  {data.isConfigured
   303→                    ? `Stripe is connected (${data.mode.toUpperCase()} mode)`
   304→                    : 'Stripe is not connected'}
   305→                </p>
   306→                <p className="text-xs text-muted-foreground">
   307→                  {data.isConfigured
   308→                    ? `Credentials source: ${data.activeSource === 'db' ? 'Admin Settings' : '.env'}`
   309→                    : 'Configure credentials below or set them in .env to enable checkout.'}
   310→                </p>
   311→              </div>
   312→            </div>
   313→            {/* Last test result */}
   314→            <div className="flex items-center gap-3">
   315→              {data.lastTestStatus === 'success' && (
   316→                <Badge className="bg-emerald-50 text-emerald-700 border-emerald-200">
   317→                  <CheckCircle2 className="h-3 w-3 mr-1" />
   318→                  Last test: OK
   319→                </Badge>
   320→              )}
   321→              {data.lastTestStatus === 'error' && (
   322→                <Badge className="bg-rose-50 text-rose-700 border-rose-200">
   323→                  <XCircle className="h-3 w-3 mr-1" />
   324→                  Last test: Failed
   325→                </Badge>
   326→              )}
   327→              {data.lastTestedAt && (
   328→                <span className="text-xs text-muted-foreground">
   329→                  {new Date(data.lastTestedAt).toLocaleString()}
   330→                </span>
   331→              )}
   332→            </div>
   333→          </div>
   334→          {data.lastTestStatus === 'error' && data.lastTestErrorMessage && (
   335→            <div className="mt-3 rounded-md border border-rose-200 bg-rose-50 p-3">
   336→              <p className="text-xs font-mono text-rose-700 break-all">{data.lastTestErrorMessage}</p>
   337→            </div>
   338→          )}
   339→        </CardContent>
   340→      </Card>
   341→
   342→      {/* -------------------- Mode + credentials -------------------- */}
   343→      <Card>
   344→        <CardHeader>
   345→          <CardTitle className="text-base flex items-center gap-2">
   346→            <CreditCard className="h-4 w-4" />
   347→            Account Credentials
   348→          </CardTitle>
   349→          <CardDescription>
   350→            Enter your Stripe keys for each mode. Secret keys are AES-256-GCM encrypted at rest and never
   351→            sent back to the frontend in plaintext. Leave a secret field empty to keep the existing value;
   352→            enter a new value to rotate.
   353→          </CardDescription>
   354→        </CardHeader>
   355→        <CardContent className="space-y-5">
   356→          {/* Mode toggle */}
   357→          <div className="flex items-center justify-between rounded-md border bg-muted/30 px-4 py-3">
   358→            <div>
   359→              <p className="text-sm font-medium">Mode</p>
   360→              <p className="text-xs text-muted-foreground mt-0.5">
   361→                Test mode uses your Stripe sandbox keys (sk_test_…). Live mode uses real production keys
   362→                (sk_live_…). Switch modes any time — credentials are stored separately per mode.
   363→              </p>
   364→            </div>
   365→            <div className="flex items-center gap-3">
   366→              <span className={cn('text-xs font-medium', mode === 'test' ? 'text-amber-700' : 'text-muted-foreground')}>
   367→                TEST
   368→              </span>
   369→              <Switch
   370→                checked={mode === 'live'}
   371→                onCheckedChange={(v) => setMode(v ? 'live' : 'test')}
   372→              />
   373→              <span className={cn('text-xs font-medium', mode === 'live' ? 'text-emerald-700' : 'text-muted-foreground')}>
   374→                LIVE
   375→              </span>
   376→            </div>
   377→          </div>
   378→
   379→          <Separator />
   380→
   381→          {/* Test mode credentials */}
   382→          <div className={cn('space-y-4', mode === 'live' && 'opacity-50 pointer-events-none')}>
   383→            <div className="flex items-center justify-between">
   384→              <h4 className="text-sm font-semibold flex items-center gap-2">
   385→                <span className={cn(
   386→                  'inline-flex items-center rounded-full px-2 py-0.5 text-xs font-medium',
   387→                  'bg-amber-50 text-amber-700 border border-amber-200',
   388→                )}>
   389→                  TEST
   390→                </span>
   391→                Test Mode Credentials
   392→              </h4>
   393→              {data.hasSecretKeyTest && (
   394→                <Badge variant="outline" className="text-emerald-700 border-emerald-300 bg-emerald-50">
   395→                  Configured
   396→                </Badge>
   397→              )}
   398→            </div>
   399→            <SecretKeyInput
   400→              label="Secret Key"
   401→              prefix="sk_test_"
   402→              value={secretKeyTest}
   403→              onChange={setSecretKeyTest}
   404→              show={showSecretTest}
   405→              onToggleShow={() => setShowSecretTest((v) => !v)}
   406→              masked={data.secretKeyTestMasked}
   407→              isSet={data.hasSecretKeyTest}
   408→              placeholder="sk_test_…"
   409→            />
   410→            <div className="space-y-1.5">
   411→              <Label className="text-xs">Publishable Key</Label>
   412→              <Input
   413→                value={publishableKeyTest}
   414→                onChange={(e) => setPublishableKeyTest(e.target.value)}
   415→                className="h-9 font-mono text-xs"
   416→                placeholder="pk_test_…"
   417→              />
   418→              <p className="text-[11px] text-muted-foreground">
   419→                Non-secret — safe to expose to the client for Stripe.js.
   420→              </p>
   421→            </div>
   422→            <SecretKeyInput
   423→              label="Webhook Signing Secret"
   424→              prefix="whsec_"
   425→              value={webhookSecretTest}
   426→              onChange={setWebhookSecretTest}
   427→              show={showWebhookTest}
   428→              onToggleShow={() => setShowWebhookTest((v) => !v)}
   429→              masked={data.webhookSecretTestMasked}
   430→              isSet={data.hasWebhookSecretTest}
   431→              placeholder="whsec_…"
   432→            />
   433→          </div>
   434→
   435→          <Separator />
   436→
   437→          {/* Live mode credentials */}
   438→          <div className={cn('space-y-4', mode === 'test' && 'opacity-50 pointer-events-none')}>
   439→            <div className="flex items-center justify-between">
   440→              <h4 className="text-sm font-semibold flex items-center gap-2">
   441→                <span className={cn(
   442→                  'inline-flex items-center rounded-full px-2 py-0.5 text-xs font-medium',
   443→                  'bg-emerald-50 text-emerald-700 border border-emerald-200',
   444→                )}>
   445→                  LIVE
   446→                </span>
   447→                Live Mode Credentials
   448→              </h4>
   449→              {data.hasSecretKeyLive && (
   450→                <Badge variant="outline" className="text-emerald-700 border-emerald-300 bg-emerald-50">
   451→                  Configured
   452→                </Badge>
   453→              )}
   454→            </div>
   455→            <SecretKeyInput
   456→              label="Secret Key"
   457→              prefix="sk_live_"
   458→              value={secretKeyLive}
   459→              onChange={setSecretKeyLive}
   460→              show={showSecretLive}
   461→              onToggleShow={() => setShowSecretLive((v) => !v)}
   462→              masked={data.secretKeyLiveMasked}
   463→              isSet={data.hasSecretKeyLive}
   464→              placeholder="sk_live_…"
   465→            />
   466→            <div className="space-y-1.5">
   467→              <Label className="text-xs">Publishable Key</Label>
   468→              <Input
   469→                value={publishableKeyLive}
   470→                onChange={(e) => setPublishableKeyLive(e.target.value)}
   471→                className="h-9 font-mono text-xs"
   472→                placeholder="pk_live_…"
   473→              />
   474→              <p className="text-[11px] text-muted-foreground">
   475→                Non-secret — safe to expose to the client for Stripe.js.
   476→              </p>
   477→            </div>
   478→            <SecretKeyInput
   479→              label="Webhook Signing Secret"
   480→              prefix="whsec_"
   481→              value={webhookSecretLive}
   482→              onChange={setWebhookSecretLive}
   483→              show={showWebhookLive}
   484→              onToggleShow={() => setShowWebhookLive((v) => !v)}
   485→              masked={data.webhookSecretLiveMasked}
   486→              isSet={data.hasWebhookSecretLive}
   487→              placeholder="whsec_…"
   488→            />
   489→          </div>
   490→
   491→          <Separator />
   492→
   493→          {/* App URL */}
   494→          <div className="space-y-1.5">
   495→            <Label className="text-xs">Public App URL</Label>
   496→            <Input
   497→              value={appUrl}
   498→              onChange={(e) => setAppUrl(e.target.value)}
   499→              className="h-9"
   500→              placeholder="https://your-platform.com"
   501→            />
   502→            <p className="text-[11px] text-muted-foreground">
   503→              Base URL for Stripe Checkout success / cancel redirects. Falls back to STRIPE_APP_URL env
   504→              var or http://localhost:3000.
   505→            </p>
   506→          </div>
   507→
   508→          {/* Save button */}
   509→          <div className="flex justify-end gap-2 pt-2">
   510→            <Button
   511→              onClick={() => saveMutation.mutate()}
   512→              disabled={saveMutation.isPending}
   513→            >
   514→              {saveMutation.isPending ? (
   515→                <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   516→              ) : (
   517→                <Save className="h-4 w-4 mr-2" />
   518→              )}
   519→              Save Settings
   520→            </Button>
   521→          </div>
   522→        </CardContent>
   523→      </Card>
   524→
   525→      {/* -------------------- Webhook endpoint -------------------- */}
   526→      <Card>
   527→        <CardHeader>
   528→          <CardTitle className="text-base flex items-center gap-2">
   529→            <Webhook className="h-4 w-4" />
   530→            Webhook Endpoint
   531→          </CardTitle>
   532→          <CardDescription>
   533→            Add this URL to your Stripe Dashboard → Developers → Webhooks. Stripe will fire events
   534→            here when subscriptions change, invoices are paid, payments fail, etc.
   535→          </CardDescription>
   536→        </CardHeader>
   537→        <CardContent className="space-y-3">
   538→          <div className="flex items-center gap-2">
   539→            <code className="flex-1 rounded-md border bg-muted px-3 py-2 text-xs font-mono break-all">
   540→              {data.webhookUrlHint}
   541→            </code>
   542→            <Button variant="outline" size="icon" onClick={copyWebhookUrl} title="Copy">
   543→              {copied ? <Check className="h-4 w-4 text-emerald-600" /> : <Copy className="h-4 w-4" />}
   544→            </Button>
   545→          </div>
   546→          <div className="rounded-md border border-amber-200 bg-amber-50 p-3">
   547→            <p className="text-xs text-amber-900 leading-relaxed">
   548→              <strong>Important:</strong> Use the Webhook Signing Secret (<code className="font-mono">whsec_…</code>)
   549→              that Stripe generates for THIS endpoint — paste it into the {mode === 'live' ? 'Live' : 'Test'} Mode Webhook Signing Secret field above.
   550→              Never use the same signing secret across environments.
   551→            </p>
   552→          </div>
   553→          <div className="flex items-center gap-2 text-xs text-muted-foreground">
   554→            <Sparkles className="h-3 w-3" />
   555→            <span>
   556→              Events handled: <code className="font-mono">checkout.session.completed</code>,{' '}
