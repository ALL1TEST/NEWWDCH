     1→// ============================================================
     2→// PLATFORM DATA SERVICE — Centralized SaaS platform dataset
     3→// ============================================================
     4→// This is the SINGLE source of truth for the entire SaaS platform.
     5→// Both the Client Billing experience AND the Platform Admin
     6→// Dashboard read from and mutate this same in-memory dataset, so
     7→// a plan change, cancellation, or payment made on the client side
     8→// is immediately reflected on the admin side, and vice-versa.
     9→//
    10→// Consistency rules (enforced by derivation, never hardcoded):
    11→//  - totalCustomers  = customers.length (non-deleted)
    12→//  - activeSubscriptions = subscriptions where status === 'active'
    13→//  - mrr = sum of normalized monthly price over active PAID subs
    14→//  - totalSites = sites.length
    15→//  - planDistribution = subscriptions grouped by plan (all statuses)
    16→//  - statusCounts = subscriptions grouped by status
    17→//  - recentCustomers = customers sorted by createdAt desc
    18→//  - recentPayments = payments sorted by date desc
    19→//  - usage = aggregated from sites/content/media/AI records
    20→//  - alerts = derived from failed payments + past-due + storage
    21→//
    22→// Mock data is deterministic (no Math.random) so numbers are stable
    23→// across renders and never contradict each other.
    24→// ============================================================

   141→// -------------------- Plans (delegated to DB-backed plan-config) --------------------
   142→// The editable plan definitions live in the PlanConfig table and are cached
   143→// by ./plan-config — the SINGLE source of truth. The owner edits there via
   144→// the Platform Admin; this module (admin overview + client billing) and the
   145→// client billing API both read the same cached values, so an owner price
   146→// change propagates to the client billing page and to MRR on the next read.
   147→// The PLANS array below is a live view re-spliced whenever the config changes.
   148→
   149→import { getPlanConfigsSync, subscribe as subscribePlanConfig, type PlanConfigData } from './plan-config';
   150→
   151→function toPlan(d: PlanConfigData): Plan {
   152→  return {
   153→    id: d.planId as PlanId,
   154→    name: d.name,
   155→    price: d.priceMonthly,
   156→    priceMonthly: d.priceMonthly,
   157→    priceYearly: d.priceYearly,
   158→    currency: d.currency,
   159→    interval: d.interval,
   160→    isFree: d.isFree,
   161→    active: d.active,
   162→    features: d.features,
   163→    entitlements: d.entitlements,
   164→    limits: d.limits,
   165→  };
   166→}
   167→
   168→/** Live view of the DB-backed plan configs. Mutated in place on every
   169→ *  config refresh so existing references stay valid. */
   170→export const PLANS: Plan[] = [];
   171→
   172→function refreshPLANS(): void {
   173→  const data = getPlanConfigsSync();
   174→  PLANS.length = 0;
   175→  for (const d of data) PLANS.push(toPlan(d));
   176→}
   177→refreshPLANS();
   178→subscribePlanConfig(refreshPLANS);
   179→
   180→export function getPlan(id: PlanId): Plan {
   181→  return PLANS.find((p) => p.id === id) ?? PLANS[0];
   182→}
   183→
   184→/** Normalized monthly price. Yearly subscriptions contribute priceYearly/12
   185→ *  so MRR is a true monthly run-rate. */
   186→export function monthlyPrice(plan: Plan, interval: BillingInterval): number {
   187→  if (interval === 'yearly') return Math.round(plan.priceYearly / 12);
   188→  return plan.priceMonthly;
   189→}

   196→function isoDaysAgo(days: number): string {
   197→  const d = new Date();
   198→  d.setDate(d.getDate() - days);
   199→  d.setHours(9, 15, 0, 0);
   200→  return d.toISOString();
   201→}
   202→function isoDaysAhead(days: number): string {
   203→  const d = new Date();
   204→  d.setDate(d.getDate() + days);
   205→  d.setHours(9, 15, 0, 0);
   206→  return d.toISOString();
   207→}

   367→function buildRevenueSeries(currentMrr: number): RevenuePoint[] {
   368→  const trend = [0.41, 0.47, 0.54, 0.62, 0.71, 0.80, 0.89, 1.0];
   369→  const out: RevenuePoint[] = [];
   370→  const now = new Date();
   371→  for (let i = 7; i >= 0; i--) {
   372→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   373→    const factor = trend[7 - i];
   374→    out.push({ month: MONTHS[d.getMonth()], revenue: Math.round(currentMrr * factor) });
   375→  }
   376→  return out;
   377→}

   383→interface PlatformStore {
   384→  customers: Customer[];
   385→  sites: PlatformSite[];
   386→  payments: Payment[];
   387→  audit: AuditEntry[];

   389→
   390→let _store: PlatformStore | null = null;
   391→let _auditCounter = 0;
   392→let _paymentCounter = 0;
   393→
   394→function createStore(): PlatformStore {
   395→  const customers: Customer[] = CUSTOMER_SEED.map((c, i) => ({
   396→    ...c,
   397→    id: `cus_${String(i + 1).padStart(3, '0')}`,
   398→  }));
   399→
   400→  const sites: PlatformSite[] = SITE_SEED.map((s, i) => ({
   401→    id: `site_p${String(i + 1).padStart(3, '0')}`,
   402→    customerId: customers[s.customerIdIdx].id,
   403→    name: s.name,
   404→    slug: s.slug,
   405→    domain: s.domain,
   406→    status: s.status,
   407→    articles: s.articles,
   408→    media: s.media,
   409→    storageBytes: s.storageBytes,
   410→    createdAt: isoDaysAgo(s.daysAgo),
   411→  }));
   412→
   413→  const payments: Payment[] = PAYMENT_SEED.map((p, i) => {
   414→    const c = customers[p.customerIdx];
   415→    return {
   416→      id: `pay_${String(i + 1).padStart(3, '0')}`,
   417→      customerId: c.id,
   418→      planId: p.planId,
   419→      amount: p.amount,
   420→      currency: 'CHF',
   421→      status: p.status,
   422→      method: p.method,
   423→      date: isoDaysAgo(p.daysAgo),
   424→      invoiceNumber: `INV-2026-${String(1000 + i)}`,
   425→    };
   426→  });
   427→
   428→  const audit: AuditEntry[] = INITIAL_AUDIT.map((a, i) => ({
   429→    ...a,
   430→    id: `aud_${String(i + 1).padStart(3, '0')}`,
   431→  }));
   432→
   433→  _auditCounter = audit.length;
   434→  _paymentCounter = payments.length;
   435→
   436→  return { customers, sites, payments, audit };
   437→}

   560→    list = list.filter((c) => c.subscriptionStatus === opts.status);
   561→  }
   562→  return list
   563→    .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   564→    .map((c) => ({ ...c, siteCount: s.sites.filter((si) => si.customerId === c.id).length }));

   657→export function getAlerts(): PlatformAlert[] {
   658→  const s = store();
   659→  const alerts: PlatformAlert[] = [];
   660→
   661→  const failed = s.payments.filter((p) => p.status === 'failed');
   662→  if (failed.length > 0) {
   663→    alerts.push({
   664→      id: 'alert-failed-payments',
   665→      severity: 'critical',
   666→      title: 'Failed payments',
   667→      message: `${failed.length} payment${failed.length === 1 ? '' : 's'} failed and need attention`,
   668→      action: { label: 'View Payments', module: 'platform-payments' },
   669→      time: 'recent',
   670→    });
   671→  }
   672→
   673→  const pastDue = s.customers.filter((c) => c.subscriptionStatus === 'past_due');
   674→  if (pastDue.length > 0) {
   675→    alerts.push({
   676→      id: 'alert-past-due',
   677→      severity: 'warning',
   678→      title: 'Past-due subscriptions',
   679→      message: `${pastDue.length} subscription${pastDue.length === 1 ? '' : 's'} past due`,
   680→      action: { label: 'View Subscriptions', module: 'platform-subscriptions' },
   681→      time: 'recent',
   682→    });
   683→  }
   684→
   685→  const nearLimit = s.customers.filter((c) => {
   686→    const used = s.sites
   687→      .filter((si) => si.customerId === c.id)
   688→      .reduce((a, si) => a + si.storageBytes, 0);
   689→    return c.storageLimitBytes > 0 && used / c.storageLimitBytes > 0.8;
   690→  });
   691→  if (nearLimit.length > 0) {
   692→    alerts.push({
   693→      id: 'alert-storage',
   694→      severity: 'warning',
   695→      title: 'Storage limit',
   696→      message: `${nearLimit.length} customer${nearLimit.length === 1 ? '' : 's'} approaching storage limit`,
   697→      action: { label: 'View Customers', module: 'platform-customers' },
   698→      time: 'recent',
   699→    });
   700→  }
   701→
   702→  const weekAgo = Date.now() - 7 * 24 * 3600_000;
   703→  const newCustomers = s.customers.filter((c) => new Date(c.createdAt).getTime() > weekAgo);
   704→  if (newCustomers.length > 0) {
   705→    alerts.push({
   706→      id: 'alert-new-customers',
   707→      severity: 'info',
   708→      title: 'New customers',
   709→      message: `${newCustomers.length} new customer${newCustomers.length === 1 ? '' : 's'} registered this week`,
   710→      action: { label: 'View Customers', module: 'platform-customers' },
   711→      time: 'recent',
   712→    });
   713→  }
   714→
   715→  return alerts;
   716→}

   724→export function changeCustomerPlan(customerId: string, newPlanId: PlanId, actor: string): Customer | null {
   725→  const s = store();
   726→  const c = s.customers.find((cu) => cu.id === customerId);
   727→  if (!c) return null;
   728→  const oldPlan = getPlan(c.planId).name;
   729→  c.planId = newPlanId;
   730→  c.subscriptionStatus = 'active';
   731→  c.trialEnd = null;
   732→  c.subscriptionStart = new Date().toISOString();
   733→  const plan = getPlan(newPlanId);
   734→  c.nextBillingAt = isoDaysAhead(30);
   735→
   736→  if (!plan.isFree) {
   737→    const amount = c.billingInterval === 'yearly' ? plan.price * 12 : plan.price;
   738→    s.payments.unshift({
   739→      id: nextPaymentId(),
   740→      customerId: c.id,
   741→      planId: newPlanId,
   742→      amount,
   743→      currency: 'CHF',
   744→      status: 'paid',
   745→      method: c.email.includes('example') ? 'Visa ••4242' : 'Invoice',
   746→      date: new Date().toISOString(),
   747→      invoiceNumber: `INV-2026-${String(1000 + _paymentCounter)}`,
   748→    });
   749→  }
   750→
   751→  appendAudit({
   752→    actor,
   753→    action: 'changed plan',
   754→    target: `Customer: ${c.name}`,
   755→    detail: `${oldPlan} → ${plan.name}`,
   756→    severity: 'info',
   757→  });
   758→  return c;
   759→}

   825→/** Current resource usage for a customer (sync). Derived from the same
   826→ *  centralized dataset as the admin overview — never an independent number. */
   827→export function getCustomerUsageSync(email: string): CustomerUsage {
   828→  const s = store();
   829→  const customer = s.customers.find((c) => c.email.toLowerCase() === email.toLowerCase());
   830→  if (!customer) {
   831→    return { sites: 0, storageBytes: 0, aiWords: 0, aiArticles: 0, automationRuns: 0 };
   832→  }
   833→  const sites = s.sites.filter((si) => si.customerId === customer.id);
   834→  const totalArticles = sites.reduce((a, si) => a + si.articles, 0);
   835→  const storageBytes = sites.reduce((a, si) => a + si.storageBytes, 0);
   836→  const aiArticles = Math.round(totalArticles * 0.28);
   837→  return {
   838→    sites: sites.length,
   839→    storageBytes,
   840→    aiWords: aiArticles * 1850,
   841→    aiArticles,
   842→    automationRuns: 0, // not tracked per-customer in the demo dataset
   843→  };
   844→}

   848→export interface ClientBillingState {
   849→  customer: Customer | null;
   850→  plan: Plan;
   851→  allPlans: Plan[];
   852→  status: SubscriptionStatus;
   853→  trialEnd: string | null;
   854→  nextBillingAt: string | null;
   855→  paymentHistory: Payment[];
   856→  /** Billing mode of the authenticated user. INTERNAL/EXEMPT users have a
   857→   *  billing bypass (full access, not a paying customer). */
   858→  billingMode: 'EXTERNAL' | 'INTERNAL' | 'EXEMPT';
   859→  isInternal: boolean;

   862→/** Synthetic "Internal" plan for owner / billing-bypass users. Carries every
   863→ *  entitlement and a zero price; the user is not a paying customer. */
   864→const INTERNAL_PLAN: Plan = {
   865→  id: 'beta',
   866→  name: 'Internal',
   867→  price: 0,
   868→  priceMonthly: 0,
   869→  priceYearly: 0,
   870→  currency: 'CHF',
   871→  interval: 'monthly',
   872→  isFree: true,
   873→  active: true,
   874→  features: ['Full platform access', 'All features enabled', 'Billing bypass', 'Not counted in MRR'],
   875→  entitlements: ['automation', 'ai_content', 'advanced_analytics', 'custom_domains', 'api_access', 'white_label', 'audit_log', 'advanced_seo', 'newsletter'],
   876→  limits: { maxSites: -1, storageBytes: -1, aiWords: -1, aiArticles: -1, automationRuns: -1 },
   877→};

   879→interface BillingUser {
   880→  email: string;
   881→  role?: string;
   882→  billingMode?: string;
   883→}
   884→
   885→function isBillingBypass(user: BillingUser): boolean {
   886→  return user.role === 'OWNER' || user.billingMode === 'INTERNAL' || user.billingMode === 'EXEMPT';
   887→}
   888→
   889→export function getClientBilling(user: BillingUser): ClientBillingState {
   890→  const s = store();
   891→  const customer = s.customers.find((c) => c.email.toLowerCase() === user.email.toLowerCase()) ?? null;
   892→  const bypass = isBillingBypass(user);
   893→  const billingMode = (user.billingMode as 'EXTERNAL' | 'INTERNAL' | 'EXEMPT') ?? 'EXTERNAL';
   894→
   895→  if (bypass) {
   896→    return {
   897→      customer: null,
   898→      plan: INTERNAL_PLAN,
   899→      allPlans: PLANS.filter((p) => p.active),
   900→      status: 'active',
   901→      trialEnd: null,
   902→      nextBillingAt: null,
   903→      paymentHistory: [],
   904→      billingMode,
   905→      isInternal: true,
   906→    };
   907→  }
   908→
   909→  const plan = customer ? getPlan(customer.planId) : getPlan('beta');
   910→  const paymentHistory = customer
   911→    ? s.payments
   912→        .filter((p) => p.customerId === customer.id)
   913→        .sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())
   914→    : [];
   915→  return {
   916→    customer,
   917→    plan,
   918→    allPlans: PLANS.filter((p) => p.active),
   919→    status: customer?.subscriptionStatus ?? 'trial',
   920→    trialEnd: customer?.trialEnd ?? null,
   921→    nextBillingAt: customer?.nextBillingAt ?? null,
   922→    paymentHistory,
   923→    billingMode,
   924→    isInternal: false,
   925→  };
   926→}

   929→export function getClientBillingByEmail(email: string): ClientBillingState {

   933→export function clientChangePlan(user: BillingUser, newPlanId: PlanId): ClientBillingState | null {
   934→  if (isBillingBypass(user)) return getClientBilling(user); // bypass: no plan change
   935→  const s = store();
   936→  const c = s.customers.find((cu) => cu.email.toLowerCase() === user.email.toLowerCase());
   937→  if (!c) return null;
   938→  // Reject change to an inactive plan.
   939→  const target = getPlan(newPlanId);
   940→  if (!target.active) return null;
   941→  changeCustomerPlan(c.id, newPlanId, user.email);
   942→  return getClientBilling(user);
   943→}

   945→export function clientCancelSubscription(user: BillingUser): ClientBillingState | null {
   946→  if (isBillingBypass(user)) return getClientBilling(user); // bypass: nothing to cancel
   947→  const s = store();
   948→  const c = s.customers.find((cu) => cu.email.toLowerCase() === user.email.toLowerCase());
   949→  if (!c) return null;
   950→  cancelSubscription(c.id, user.email);
   951→  return getClientBilling(user);
   952→}
