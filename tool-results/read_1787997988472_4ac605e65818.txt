    27→// Realistic checksum (sha256 hex, 64 chars). Static values for stability.
    28→function fakeChecksum(seed: number): string {
    29→  const hex = '0123456789abcdef';
    30→  let s = '';
    31→  for (let i = 0; i < 64; i++) {
    32→    s += hex[(seed * (i + 7) + i * 13) % 16];
    33→  }
    34→  return s;
    35→}

   532→type LogStatus = 'SUCCESS' | 'FAILED' | 'IN_PROGRESS';
   533→

   535→  action: LogAction;
   536→  status: LogStatus;
   537→  backupIdx: number | null; // index into BACKUPS (resolved to backupId)
   538→  databaseSize: number | null;
   539→  fileCount: number | null;
   540→  archiveSize: number | null;

   542→  storageProvider: Provider | null;

   569→        : null,
   570→      errorMessage: b.status === 'FAILED' ? b.note : null,
   571→      warnings: b.verificationStatus === 'WARNING' ? 'Checksum mismatch on 2 of 1842 files. Manual review recommended.' : null,

   723→}

   820→  }

   827→      data: {
   828→        backupId: l.backupIdx !== null ? backupIds[l.backupIdx] : null,

   832→        fileCount: l.fileCount,

   847→  // ---- Summary ----
