   105→function formatNumber(n: number | null | undefined): string {
   106→  const num = n ?? 0;
   107→  if (num >= 1000000) return `${(num / 1000000).toFixed(1)}M`;
   108→  if (num >= 1000) return `${(num / 1000).toFixed(1)}K`;
   109→  return num.toLocaleString();
   110→}
   111→
   112→function formatPercent(n: number | null | undefined): string {
   113→  return `${((n ?? 0) * 100).toFixed(2)}%`;
   114→}
   115→
   116→function formatPosition(n: number | null | undefined): string {
   117→  return (n ?? 0).toFixed(1);
   118→}
   119→
   120→function formatDate(dateStr: string | null): string {
   121→  if (!dateStr) return 'Never';
   122→  try {
   123→    return new Date(dateStr).toLocaleString(undefined, {
   124→      month: 'short',
   125→      day: 'numeric',
   126→      year: 'numeric',
   127→      hour: '2-digit',
   128→      minute: '2-digit',
   129→    });
   130→  } catch {
   131→    return 'Unknown';
   132→  }
   133→}
   134→
   135→function formatShortDate(dateStr: string): string {
   136→  try {
   137→    return new Date(dateStr).toLocaleDateString(undefined, {
   138→      month: 'short',
   139→      day: 'numeric',
   140→    });
   141→  } catch {
   142→    return dateStr;
   143→  }
   144→}

   207→function PerformanceChart({ stats, isLoading }: { stats: DailyStat[]; isLoading: boolean }) {

   211→        <Skeleton className="h-4 w-24" />

   223→          Sync with Search Console to see chart data
   224→        </p>
   225→      </div>
   226→    );
   227→  }
   228→
   229→  const maxClicks = Math.max(...stats.map((s) => s.clicks), 1);
   230→  const maxImpressions = Math.max(...stats.map((s) => s.impressions), 1);
   231→
   232→  return (
   233→    <div className="space-y-4">
   234→      <div className="flex items-center justify-between">
   235→        <div className="flex items-center gap-4 text-xs text-muted-foreground">

   237→            <span className="h-2.5 w-2.5 rounded-sm bg-primary" />
   238→            Clicks
   239→          </span>
   240→          <span className="flex items-center gap-1.5">
   241→            <span className="h-2.5 w-2.5 rounded-sm bg-primary/30" />
   242→            Impressions
   243→          </span>
   244→        </div>
   245→        <Badge variant="outline" className="text-[10px] font-normal bg-background/80">
   246→          Last 14 days
   247→        </Badge>
   248→      </div>
   249→
   250→      <div className="flex items-end gap-[3px] sm:gap-1.5 h-48">
   251→        {stats.map((stat) => {
   252→          const clickHeight = Math.max((stat.clicks / maxClicks) * 100, stat.clicks > 0 ? 4 : 0);
   253→          const impHeight = Math.max((stat.impressions / maxImpressions) * 100, stat.impressions > 0 ? 4 : 0);
   254→
   255→          return (
   256→            <div
   257→              key={stat.date}
   258→              className="flex-1 flex flex-col items-center gap-0.5 group relative min-w-0"
   259→            >
   260→              {/* Tooltip */}
   261→              <div className="absolute -top-16 left-1/2 -translate-x-1/2 hidden group-hover:flex flex-col items-center gap-0.5 bg-popover border rounded-md shadow-md px-2 py-1.5 z-10 pointer-events-none">
   262→                <span className="text-[10px] text-muted-foreground whitespace-nowrap">
   263→                  {formatShortDate(stat.date)}
   264→                </span>
   265→                <span className="text-[10px] font-medium">
   266→                  {stat.clicks} clicks · {stat.impressions} imp
   267→                </span>
   268→              </div>
   269→
   270→              {/* Bars container */}
   271→              <div className="flex items-end gap-[1px] w-full">
   272→                {/* Impressions bar (wider, lighter) */}
   273→                <div
   274→                  className="flex-1 rounded-t-sm bg-primary/20 transition-all duration-300"
   275→                  style={{ height: `${impHeight * 0.85}%` }}
   276→                />
   277→                {/* Clicks bar (narrower, solid) */}
   278→                <div
   279→                  className="w-[40%] rounded-t-sm bg-primary transition-all duration-300"
   280→                  style={{ height: `${clickHeight}%` }}
   281→                />
   282→              </div>
   283→            </div>
   284→          );
   285→        })}
   286→      </div>
   287→
   288→      {/* Date labels */}
   289→      <div className="flex justify-between text-[10px] text-muted-foreground/60">
   290→        <span>{formatShortDate(stats[0]?.date ?? '')}</span>
   291→        <span>{formatShortDate(stats[stats.length - 1]?.date ?? '')}</span>
   292→      </div>
   293→    </div>
   294→  );
   295→}

   463→function SeoSearchConsolePageInner() {
   464→  const queryClient = useQueryClient();
   465→  const [connectUrl, setConnectUrl] = useState('');
   466→
   467→  // Main query
   468→  const { data, isLoading, error } = useQuery({
   469→    queryKey: queryKeys.seoSearchConsole.all,
   470→    queryFn: () => getApi<SearchConsoleData>('/api/seo/search-console'),
   471→    staleTime: 30_000,
   472→  });
   473→
   474→  // Stats query (14 days)
   475→  const { data: statsData, isLoading: statsLoading } = useQuery({
   476→    queryKey: queryKeys.seoSearchConsoleStats.list(14),
   477→    queryFn: () => getApi<DailyStat[]>('/api/seo/search-console/stats', { days: 14 }),
   478→    staleTime: 60_000,
   479→    enabled: data?.connection?.status === 'CONNECTED',
   480→  });
   481→
   482→  // Queries query
   483→  const { data: queriesData, isLoading: queriesLoading } = useQuery({
   484→    queryKey: queryKeys.seoSearchConsoleQueries.list({ page: 1, pageSize: 10 }),
   485→    queryFn: () => getApi<PaginatedList<QueryItem>>('/api/seo/search-console/queries', { page: 1, pageSize: 10 }),
   486→    staleTime: 60_000,
   487→    enabled: data?.connection?.status === 'CONNECTED',
   488→  });
   489→
   490→  // Pages query
   491→  const { data: pagesData, isLoading: pagesLoading } = useQuery({
   492→    queryKey: queryKeys.seoSearchConsolePages.list({ page: 1, pageSize: 10 }),
   493→    queryFn: () => getApi<PaginatedList<PageItem>>('/api/seo/search-console/pages', { page: 1, pageSize: 10 }),
   494→    staleTime: 60_000,
   495→    enabled: data?.connection?.status === 'CONNECTED',
   496→  });
   497→
   498→  const isConnected = data?.connection?.status === 'CONNECTED';
   499→  const connection = data?.connection;
   500→  const summary = data?.summary;
   501→
   502→  // Connect mutation
   503→  const connectMutation = useMutation({
   504→    mutationFn: () => postApi('/api/seo/search-console', { siteUrl: connectUrl }),
   505→    onSuccess: () => {
   506→      setConnectUrl('');
   507→      queryClient.invalidateQueries({ queryKey: queryKeys.seoSearchConsole.all });
   508→      toast.success('Search Console connected successfully');
   509→    },
   510→    onError: () => {
   511→      toast.error('Failed to connect Search Console');
   512→    },
   513→  });
   514→
   515→  // Disconnect mutation
   516→  const disconnectMutation = useMutation({
   517→    mutationFn: () => deleteApi('/api/seo/search-console'),
   518→    onSuccess: () => {
   519→      queryClient.invalidateQueries({ queryKey: queryKeys.seoSearchConsole.all });
   520→      toast.success('Search Console disconnected');
   521→    },
   522→    onError: () => {
   523→      toast.error('Failed to disconnect Search Console');
   524→    },
   525→  });
   526→
   527→  // Sync mutation
   528→  const syncMutation = useMutation({
   529→    mutationFn: () => patchApi('/api/seo/search-console?action=sync'),
   530→    onSuccess: () => {
   531→      queryClient.invalidateQueries({ queryKey: queryKeys.seoSearchConsole.all });
   532→      queryClient.invalidateQueries({ queryKey: queryKeys.seoSearchConsoleStats.all });
   533→      queryClient.invalidateQueries({ queryKey: queryKeys.seoSearchConsoleQueries.all });
   534→      queryClient.invalidateQueries({ queryKey: queryKeys.seoSearchConsolePages.all });
   535→      toast.success('Search Console data synced successfully');
   536→    },
   537→    onError: () => {
   538→      toast.error('Failed to sync Search Console data');
   539→    },
   540→  });
   541→
   542→  return (
   543→    <div className="space-y-6">
   544→      <PageHeader
   545→        title="Search Console"
   546→        description="Monitor your site's performance in Google Search"
   547→      />
   548→
   549→      {/* Error state */}
   550→      {error && (
   551→        <Card className="border-red-200 dark:border-red-900/50 bg-red-50 dark:bg-red-950/20">
   552→          <CardContent className="p-4 flex items-center gap-3">
   553→            <AlertTriangle className="h-5 w-5 text-red-500 shrink-0" />
   554→            <p className="text-sm text-red-700 dark:text-red-400">
   555→              Failed to load Search Console data. Please try again later.
   556→            </p>
   557→          </CardContent>
   558→        </Card>
   559→      )}
   560→
   561→      {isLoading ? (
   562→        <SearchConsoleSkeleton />
   563→      ) : data ? (
   564→        <>
   565→          {/* Connection Status Card */}
   566→          <Card className="p-6">
   567→            <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   568→              <div className="flex items-center gap-3">
   569→                <div className={cn(
   570→                  'flex h-10 w-10 shrink-0 items-center justify-center rounded-lg',
   571→                  isConnected
   572→                    ? 'bg-green-100 dark:bg-green-900/30'
   573→                    : 'bg-zinc-100 dark:bg-zinc-800',
   574→                )}>
   575→                  <Plug className={cn(
   576→                    'h-5 w-5',
   577→                    isConnected
   578→                      ? 'text-green-600 dark:text-green-400'
   579→                      : 'text-muted-foreground',
   580→                  )} />
   581→                </div>
   582→                <div>
   583→                  <div className="flex items-center gap-2">
   584→                    <h3 className="font-semibold text-sm">Connection Status</h3>
   585→                    <ConnectionBadge status={connection?.status ?? 'DISCONNECTED'} />
   586→                  </div>
   587→                  {connection?.siteUrl && isConnected && (
