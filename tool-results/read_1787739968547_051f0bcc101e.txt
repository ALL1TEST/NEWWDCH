   112→// ==================== Helpers ====================

   114→function formatNumber(n: number | null | undefined): string {
   115→  const num = n ?? 0;
   116→  if (num >= 1000000) return `${(num / 1000000).toFixed(1)}M`;
   117→  if (num >= 1000) return `${(num / 1000).toFixed(1)}K`;
   118→  return num.toLocaleString();
   119→}
   120→
   121→function formatPercent(n: number | null | undefined): string {
   122→  return `${((n ?? 0) * 100).toFixed(2)}%`;
   123→}
   124→
   125→function formatPosition(n: number | null | undefined): string {
   126→  return (n ?? 0).toFixed(1);
   127→}
   128→
   129→function formatDate(dateStr: string | null): string {
   130→  if (!dateStr) return 'Never';
   131→  try {
   132→    return new Date(dateStr).toLocaleString(undefined, {
   133→      month: 'short',
   134→      day: 'numeric',
   135→      year: 'numeric',
   136→      hour: '2-digit',
   137→      minute: '2-digit',
   138→    });
   139→  } catch {
   140→    return 'Unknown';
   141→  }
   142→}
   143→
   144→function formatShortDate(dateStr: string): string {
   145→  try {
   146→    return new Date(dateStr).toLocaleDateString(undefined, {
   147→      month: 'short',
   148→      day: 'numeric',
   149→    });
   150→  } catch {
   151→    return dateStr;
   152→  }
   153→}

   247→            variant="outline"

   268→  return (

   270→      <div className="flex items-center justify-between">

   322→                />
   323→                {/* Clicks bar (narrower, solid) */}

   325→                  className="w-[40%] rounded-t-sm bg-primary transition-all duration-300"
   326→                  style={{ height: `${clickHeight}%` }}
   327→                />

   336→        <span>{formatShortDate(stats[0]?.date ?? '')}</span>
   337→        <span>{formatShortDate(stats[stats.length - 1]?.date ?? '')}</span>
   338→      </div>

   514→
   515→function SeoSearchConsolePageInner() {
   516→  const queryClient = useQueryClient();
   517→  const [connectUrl, setConnectUrl] = useState('');
   518→  // Date range for the Performance Chart (days back from today).
   519→  const [chartDays, setChartDays] = useState(14);
   520→
   521→  // Main query
   522→  const { data, isLoading, error } = useQuery({
   523→    queryKey: queryKeys.seoSearchConsole.all,
   524→    queryFn: () => getApi<SearchConsoleData>('/api/seo/search-console'),
   525→    staleTime: 30_000,
   526→  });
   527→
   528→  // Stats query — uses the user-selected `chartDays` range.
   529→  const isConnected = data?.connection?.status === 'CONNECTED';
   530→  const { data: statsData, isLoading: statsLoading } = useQuery({
   531→    queryKey: queryKeys.seoSearchConsoleStats.list(chartDays),
   532→    queryFn: () => getApi<DailyStat[]>('/api/seo/search-console/stats', { days: chartDays }),
   533→    staleTime: 60_000,
   534→    enabled: isConnected,
   535→  });
   536→
   537→  // Queries query
   538→  const { data: queriesData, isLoading: queriesLoading } = useQuery({
   539→    queryKey: queryKeys.seoSearchConsoleQueries.list({ page: 1, pageSize: 10 }),
   540→    queryFn: () => getApi<PaginatedList<QueryItem>>('/api/seo/search-console/queries', { page: 1, pageSize: 10 }),
   541→    staleTime: 60_000,
   542→    enabled: data?.connection?.status === 'CONNECTED',
   543→  });
   544→
   545→  // Pages query
   546→  const { data: pagesData, isLoading: pagesLoading } = useQuery({
   547→    queryKey: queryKeys.seoSearchConsolePages.list({ page: 1, pageSize: 10 }),
   548→    queryFn: () => getApi<PaginatedList<PageItem>>('/api/seo/search-console/pages', { page: 1, pageSize: 10 }),
   549→    staleTime: 60_000,
   550→    enabled: data?.connection?.status === 'CONNECTED',
   551→  });
   552→
   553→  const connection = data?.connection;
   554→  const summary = data?.summary;
   555→
   556→  // Connect mutation
   557→  const connectMutation = useMutation({
   558→    mutationFn: () => postApi('/api/seo/search-console', { siteUrl: connectUrl }),
   559→    onSuccess: () => {
   560→      setConnectUrl('');
   561→      queryClient.invalidateQueries({ queryKey: queryKeys.seoSearchConsole.all });
   562→      toast.success('Search Console connected successfully');
   563→    },
   564→    onError: () => {
   565→      toast.error('Failed to connect Search Console');
   566→    },
   567→  });
   568→
   569→  // Disconnect mutation
   570→  const disconnectMutation = useMutation({
   571→    mutationFn: () => deleteApi('/api/seo/search-console'),
   572→    onSuccess: () => {
   573→      queryClient.invalidateQueries({ queryKey: queryKeys.seoSearchConsole.all });
   574→      toast.success('Search Console disconnected');
   575→    },
   576→    onError: () => {
   577→      toast.error('Failed to disconnect Search Console');
   578→    },
   579→  });
   580→
   581→  // Sync mutation
   582→  const syncMutation = useMutation({
   583→    mutationFn: () => patchApi('/api/seo/search-console?action=sync'),
   584→    onSuccess: () => {
   585→      queryClient.invalidateQueries({ queryKey: queryKeys.seoSearchConsole.all });
   586→      queryClient.invalidateQueries({ queryKey: queryKeys.seoSearchConsoleStats.all });
   587→      queryClient.invalidateQueries({ queryKey: queryKeys.seoSearchConsoleQueries.all });
   588→      queryClient.invalidateQueries({ queryKey: queryKeys.seoSearchConsolePages.all });
   589→      toast.success('Search Console data synced successfully');
   590→    },
   591→    onError: () => {
   592→      toast.error('Failed to sync Search Console data');
