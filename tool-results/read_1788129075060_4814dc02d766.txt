   134→// Map the API `relatedEntityType` to the UI category. 'security' is
   135→// a synthetic UI bucket — the API stores 'webhook'/'stripe' entities,
   136→// which the UI shows under Security.
   137→function mapEntityCategory(entityType: string | null | undefined): NotificationCategory {
   138→  switch (entityType) {
   139→    case 'customer':
   140→      return 'customers';
   141→    case 'payment':
   142→      return 'payments';
   143→    case 'subscription':
   144→      return 'subscriptions';
   145→    case 'webhook':
   146→    case 'stripe':
   147→      return 'security';
   148→    case 'coupon':
   149→    case 'plan':
   150→    case 'system':
   151→    default:
   152→      return 'system';
   153→  }
   154→}

   156→const PAGE_SIZE = 25;

   158→// ==================== Date grouping ====================
   159→
   160→type DateGroupKey = 'today' | 'yesterday' | 'thisWeek' | 'older';
   161→
   162→const DATE_GROUP_LABEL: Record<DateGroupKey, string> = {
   163→  today: 'Today',
   164→  yesterday: 'Yesterday',
   165→  thisWeek: 'Earlier this week',
   166→  older: 'Older',
   167→};
   168→
   169→const DATE_GROUP_ORDER: DateGroupKey[] = ['today', 'yesterday', 'thisWeek', 'older'];
   170→
   171→function dateGroupKey(iso: string): DateGroupKey {
   172→  const now = new Date();
   173→  const d = new Date(iso);
   174→  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
   175→  const ts = d.getTime();
   176→  if (ts >= startOfToday) return 'today';
   177→  if (ts >= startOfToday - 86_400_000) return 'yesterday';
   178→  if (ts >= startOfToday - 7 * 86_400_000) return 'thisWeek';
   179→  return 'older';
   180→}

   205→        security: 'webhook', // primary entity type for the Security bucket
   206→      };

   208→    }

   213→  // table (populated by the Stripe webhook + the periodic platform-scan

   216→    data,
   217→    isLoading,
   218→    fetchNextPage,
   219→    hasNextPage,
   220→    isFetchingNextPage,
   221→    refetch,

   223→    queryKey: ['platform-notifications', queryParams],
   224→    queryFn: ({ pageParam = 1 }) =>
   225→      getApi<ApiResponse<NotificationItem[]>>(
   226→        '/api/platform/admin/notifications',

   233→      if (!pagination) return undefined;
   234→      const { page, totalPages } = pagination;
   235→      return page < totalPages ? page + 1 : undefined;

   237→    staleTime: 0,
   238→    refetchOnMount: true,
   239→  });

   242→  // server-side so the bell + this page subtitle stay in sync).

   244→    queryKey: ['platform-admin', 'notifications', 'unread-count'],

   246→    staleTime: 5_000,

   257→  // (client-side).

   269→    if (q) {
   270→      list = list.filter(
   271→        (n) =>
   272→          n.title.toLowerCase().includes(q) ||
   273→          n.message.toLowerCase().includes(q),

   275→    }
   276→    return list;
   277→  }, [allNotifications, activeFilter, search]);
   278→

   284→      thisWeek: [],

   291→      key: k,
   292→      label: DATE_GROUP_LABEL[k],

   315→            ...p,
   316→            data: (p?.data ?? []).map((n: NotificationItem) =>

   319→          })),

   337→          ...old,
   338→          pages: old.pages.map((p: any) => ({
   339→            ...p,
   340→            data: (p?.data ?? []).map((n: NotificationItem) =>

   343→          })),
   344→        };
   345→      });

   347→    onSuccess: () => {
   348→      queryClient.invalidateQueries({ queryKey: ['platform-notifications'] });

   363→            ...p,

   379→      deleteApi(`/api/platform/admin/notifications/${id}`),

   385→          ...old,

   387→            ...p,
   388→            data: (p?.data ?? []).filter((n: NotificationItem) => n.id !== id),

   445→    [isFetchingNextPage, hasNextPage, fetchNextPage],

   447→
   448→  const subtitle =
   449→    unreadCount > 0

   537→      {/* ==================== Notification list ==================== */}

   621→
   622→interface NotificationRowProps {
   623→  notification: NotificationItem;
   624→  onMarkRead: (id: string) => void;
   625→  onMarkUnread: (id: string) => void;
   626→  onDelete: (id: string) => void;
   627→}
   628→
   629→function NotificationRow({
   630→  notification,
   631→  onMarkRead,
   632→  onMarkUnread,
   633→  onDelete,
   634→}: NotificationRowProps) {
   635→  const config = NOTIFICATION_TYPE_CONFIG[notification.type] ?? NOTIFICATION_TYPE_CONFIG.INFO;
   636→  const isUnread = !notification.isRead;
   637→
   638→  const handleClick = useCallback(() => {
   639→    if (isUnread) onMarkRead(notification.id);
   640→  }, [isUnread, notification.id, onMarkRead]);
   641→
   642→  return (
   643→    <div
   644→      role="button"
   645→      tabIndex={0}
   646→      onClick={handleClick}
   647→      onKeyDown={(e) => {
   648→        if (e.key === 'Enter' || e.key === ' ') {
   649→          e.preventDefault();
   650→          handleClick();
   651→        }
   652→      }}
   653→      className={cn(
   654→        'group relative flex items-start gap-3 px-4 py-3 transition-colors cursor-pointer',
   655→        'hover:bg-accent/40',
   656→        isUnread ? 'bg-primary/[0.03]' : 'bg-card',
   657→      )}
   658→    >
   659→      {/* Unread left accent bar */}
   660→      {isUnread && (
   661→        <span className="absolute left-0 top-0 bottom-0 w-0.5 bg-primary" aria-hidden />
   662→      )}
   663→
   664→      {/* Type icon avatar */}
   665→      <div
   666→        className={cn(
   667→          'flex h-8 w-8 shrink-0 items-center justify-center rounded-full mt-0.5',
   668→          config.color,
   669→        )}
   670→      >
   671→        {config.icon}
   672→      </div>
   673→
   674→      {/* Content */}
   675→      <div className="flex-1 min-w-0 pr-2">
   676→        <div className="flex items-center gap-2">
   677→          <h4
   678→            className={cn(
   679→              'text-sm truncate',
   680→              isUnread ? 'font-semibold' : 'font-medium',
   681→            )}
   682→          >
   683→            {notification.title}
   684→          </h4>
   685→        </div>
   686→        <p className="text-xs text-muted-foreground mt-0.5 line-clamp-2">
   687→          {notification.message}
   688→        </p>
   689→      </div>
   690→
   691→      {/* Timestamp + unread dot */}
   692→      <div className="shrink-0 flex flex-col items-end gap-1.5 pt-0.5">
   693→        <span className="text-[11px] text-muted-foreground whitespace-nowrap">
   694→          {formatRelativeTime(notification.createdAt)}
   695→        </span>
   696→        {isUnread && (
   697→          <span
   698→            className="h-1.5 w-1.5 rounded-full bg-primary"
   699→            aria-label="Unread"
   700→          />
   701→        )}
   702→      </div>
   703→
   704→      {/* Hover actions */}
   705→      <div className="absolute right-2 bottom-2 flex gap-0.5 opacity-0 group-hover:opacity-100 focus-within:opacity-100 transition-opacity">
   706→        <button
   707→          type="button"
   708→          title={isUnread ? 'Mark as read' : 'Mark as unread'}
   709→          onClick={(e) => {
   710→            e.stopPropagation();
   711→            if (isUnread) onMarkRead(notification.id);
   712→            else onMarkUnread(notification.id);
   713→          }}
   714→          className="h-7 w-7 inline-flex items-center justify-center rounded-md text-muted-foreground hover:bg-accent hover:text-foreground"
   715→        >
   716→          {isUnread ? <Check className="h-3.5 w-3.5" /> : <Undo2 className="h-3.5 w-3.5" />}
   717→        </button>
   718→        <button
   719→          type="button"
   720→          title="Delete"
   721→          onClick={(e) => {
   722→            e.stopPropagation();
   723→            onDelete(notification.id);
   724→          }}
   725→          className="h-7 w-7 inline-flex items-center justify-center rounded-md text-muted-foreground hover:bg-rose-50 hover:text-rose-600 dark:hover:bg-rose-950/40 dark:hover:text-rose-400"
   726→        >
   727→          <Trash2 className="h-3.5 w-3.5" />
   728→        </button>
   729→      </div>
   730→    </div>
   731→  );
   732→}
