   319→function AIGenerateDialog({
   320→  open,
   321→  onOpenChange,
   322→  idea,
   323→}: {
   324→  open: boolean;
   325→  onOpenChange: (open: boolean) => void;
   326→  idea: ArticleIdea | null;
   327→}) {
   328→  const [brief, setBrief] = useState('');
   329→  const [keywords, setKeywords] = useState('');
   330→  const [writingStyle, setWritingStyle] = useState('Professional');
   331→  const [targetLength, setTargetLength] = useState('Medium (800-1200 words)');
   332→  const [numberOfDrafts, setNumberOfDrafts] = useState('1');
   333→  const [includeCta, setIncludeCta] = useState(false);
   334→
   335→  const navigate = useNavigationStore((s) => s.navigate);
   336→
   337→  const generateMutation = useMutation({
   338→    mutationFn: () =>
   339→      postApi('/api/content/ai-generate', {
   340→        title: idea?.title || '',
   341→        brief: brief || idea?.description || '',
   342→        keywords: keywords || idea?.keywords?.join(', ') || '',
   343→        writingStyle,
   344→        targetLength,
   345→        numberOfDrafts: parseInt(numberOfDrafts),
   346→        includeCta,
   347→      }),
   348→    onSuccess: (result) => {
   349→      const draft = result.data?.drafts?.[0];
   350→      if (draft) {
   351→        toast.success('Article draft generated!');
   352→        onOpenChange(false);
   353→        navigate('content', null, 'create');
   354→      }
   355→    },
   356→    onError: (err: Error) => toast.error(err.message || 'Failed to generate'),
   357→  });
   358→
   359→  React.useEffect(() => {
   360→    if (idea) {
   361→      setBrief(idea.description);
   362→      setKeywords(idea.keywords?.join(', ') || '');
   363→    }
   364→  }, [idea]);
   365→
   366→  return (
   367→    <Dialog open={open} onOpenChange={onOpenChange}>
   368→      <DialogContent className="sm:max-w-lg">
   369→        <DialogHeader>
   370→          <DialogTitle className="flex items-center gap-2">
   371→            <Sparkles className="h-5 w-5 text-amber-500" />
   372→            Generate Article with AI
   373→          </DialogTitle>
   374→          <DialogDescription>Configure and generate an article draft from this idea.</DialogDescription>
   375→        </DialogHeader>
   376→
   377→        <div className="space-y-4">
   378→          {/* Title */}
   379→          <div className="space-y-1.5">
   380→            <Label className="text-xs text-muted-foreground">Article Title</Label>
   381→            <div className="rounded-md border bg-muted/50 px-3 py-2 text-sm font-medium">
   382→              {idea?.title}
   383→            </div>
   384→          </div>
   385→
   386→          {/* Brief */}
   387→          <div className="space-y-1.5">
   388→            <Label className="text-xs text-muted-foreground">Brief / Description</Label>
   389→            <Textarea
   390→              value={brief}
   391→              onChange={(e) => setBrief(e.target.value)}
   392→              placeholder="Describe what the article should cover..."
   393→              rows={3}
   394→              className="text-sm resize-none"
   395→            />
   396→          </div>
   397→
   398→          {/* Keywords */}
   399→          <div className="space-y-1.5">
   400→            <Label className="text-xs text-muted-foreground">Keywords</Label>
   401→            <Input
   402→              value={keywords}
   403→              onChange={(e) => setKeywords(e.target.value)}
   404→              placeholder="comma, separated, keywords"

   406→            />
   407→          </div>

   409→          {/* Output Settings Grid */}
   410→          <div className="grid grid-cols-2 gap-3">
   411→            <div className="space-y-1.5">

   414→                <SelectTrigger className="h-9 text-sm"><SelectValue /></SelectTrigger>
   415→                <SelectContent>
   416→                  <SelectItem value="1">1 Draft</SelectItem>

   493→    queryKey: queryKeys.categories.all,
   494→    queryFn: () => getApi<{ data: { id: string; name: string; slug?: string }[] } | { id: string; name: string; slug?: string }[]>('/api/categories?pageSize=200'),
   495→    enabled: open,
   496→    staleTime: 30_000,
   497→  });
   498→  const { data: tagsData, isLoading: tagsLoading } = useQuery({
   499→    queryKey: queryKeys.tags.all,
   500→    queryFn: () => getApi<{ data: { id: string; name: string; slug?: string; color?: string }[] } | { id: string; name: string; slug?: string; color?: string }[]>('/api/tags?pageSize=200'),
   501→    enabled: open,
   502→    staleTime: 30_000,
   503→  });
   504→
   505→  // Normalize responses — API may return either an array or { data: [...] }
   506→  const categories = useMemo(() => {
   507→    const d = categoriesData as unknown;
   508→    if (Array.isArray(d)) return d as { id: string; name: string; slug?: string }[];
   509→    if (d && typeof d === 'object' && 'data' in (d as Record<string, unknown>)) {
   510→      return (d as { data: { id: string; name: string; slug?: string }[] }).data;
   511→    }
   512→    return [];
   513→  }, [categoriesData]);
   514→  const tags = useMemo(() => {
   515→    const d = tagsData as unknown;
   516→    if (Array.isArray(d)) return d as { id: string; name: string; slug?: string; color?: string }[];
   517→    if (d && typeof d === 'object' && 'data' in (d as Record<string, unknown>)) {
   518→      return (d as { data: { id: string; name: string; slug?: string; color?: string }[] }).data;
   519→    }
   520→    return [];
   521→  }, [tagsData]);
   522→
   523→  const createCategoryMutation = useMutation({
   524→    mutationFn: (name: string) => postApi('/api/categories', { name }),
   525→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: queryKeys.categories.all }); toast.success('Category created'); },
   526→    onError: (err: Error) => toast.error(err.message || 'Failed to create category'),
   527→  });
   528→  const createTagMutation = useMutation({
   529→    mutationFn: (name: string) => postApi('/api/tags', { name }),
   530→    onSuccess: () => { queryClient.invalidateQueries({ queryKey: queryKeys.tags.all }); toast.success('Tag created'); },
   531→    onError: (err: Error) => toast.error(err.message || 'Failed to create tag'),
   532→  });
   533→  const deleteCategoryMutation = useMutation({
