   158→// -------------------- Collapsed-Rail Tooltip Positioning -------------------
   159→/*
   160→ * SINGLE source of truth for the positioning of every tooltip/label that
   161→ * appears when the sidebar is collapsed into the 48px icon rail.
   162→ *
   163→ *   side="right"         → opens to the RIGHT of the 48px collapsed rail,
   164→ *                          fully inside the main viewport (never overlaps
   165→ *                          the rail, never opens to the left/outside).
   166→ *   align="center"       → vertically centers the bubble on the icon's
   167→ *                          32×32 hover target so labels line up across
   168→ *                          every row of the rail (logo, theme, nav items,
   169→ *                          parent items with submenu).
   170→ *   sideOffset=8         → ~8px visible gap from the trigger button's right
   171→ *                          edge (button right edge ~x=40 inside the 48px
   172→ *                          rail, tooltip left edge ~x=48 = the rail's
   173→ *                          right edge — flush with the RAIL, NOT with the
   174→ *                          icon, so the bubble never touches the glyph).
   175→ *   collisionPadding=12 → 12px viewport-edge collision padding so the
   176→ *                          bubble can never be clipped at any viewport
   177→ *                          edge or scroll past the visible area.
   178→ *
   179→ * Portal-based rendering (Radix TooltipPrimitive.Portal → document.body,
   180→ * wired inside src/components/ui/tooltip.tsx) means every tooltip is
   181→ * rendered OUTSIDE the sidebar DOM tree — so the sidebar's
   182→ * overflow-hidden / overflow-x-hidden on the rail container can NEVER
   183→ * clip or hide the bubble. The tooltip floats above every other
   184→ * element at z-50.
   185→ *
   186→ * Used by:
   187→ *   • CollapsedLogoButton         (header "Expand" tooltip)
   188→ *   • SimpleNavItem              (leaf nav items: Dashboard, Articles,
   189→ *                                  Calendar, Media, Users, Comments,
   190→ *                                  Newsletter, SEO, AI, Automation)
   191→ *   • ExpandableNavItem          (parent nav items in EXPANDED state —
   192→ *                                  tooltip is hidden by SidebarMenuButton
   193→ *                                  when state!=="collapsed", but the
   194→ *                                  positioning object is still applied
   195→ *                                  so the prop stays consistent across
   196→ *                                  every nav item renderer)
   197→ *   • CollapsedParentNavItem     (parent nav items in COLLAPSED state —
   198→ *                                  label tooltip shows when the floating
   199→ *                                  popover is CLOSED; when the popover is
   200→ *                                  open, no tooltip shows to avoid visual
   201→ *                                  conflict)
   202→ *
   203→ * The ThemeToggle component (src/components/layout/theme-toggle.tsx) lives
   204→ * in a separate file and inlines the SAME four values — see the comment
   205→ * there for the rationale.
   206→ */

   452→/**
   453→ * Collapsed-rail logo cell. Since the extra collapse-toggle icon was removed
   454→ * from below the logo, THIS cell is the explicit expand control:
   455→ *   • identical 32×32 grid geometry as every other rail icon
   456→ *   • click toggles the sidebar open (SidebarRail edge strip still works)
   457→ * No new icon is introduced and no empty clickable ghost remains.
   458→ *
   459→ * HOVER BEHAVIOR (collapsed state only — this button is rendered ONLY
   460→ * inside the collapsed-rail cluster, so the expanded LogoMark is untouched):
   461→ * The `hovered` prop is LIFTED to the rail level (see AppSidebar's
   462→ * `railHovered` state + the onMouseEnter/onMouseLeave handlers on the
   463→ * <Sidebar> root). This means the swap fires when the mouse is ANYWHERE
   464→ * over the collapsed 48px rail — NOT only when the pointer is directly
   465→ * over the logo cell:
   466→ *   • rail at rest ........  shows the "C" logo mark on its bg-primary
   467→ *                            black box (NORMAL LOGO — background UNCHANGED,
   468→ *                            same as the expanded LogoMark).
   469→ *   • mouse over ANY rail   the "C" is replaced by a PanelLeftOpen icon
   470→ *     item (Dashboard,       AND the button's background becomes
   471→ *     Articles, …, Profile,  transparent so ONLY the Expand icon is
   472→ *     Theme, Settings, …)    visible with no box behind it. The icon
   473→ *                            color switches to muted-foreground (gray)
   474→ *                            so it stays visible against the
   475→ *                            transparent/page bg in both Light and Dark.
   476→ *   • mouse leaves the     restores the "C" logo mark AND its bg-primary
   477→ *     entire rail            box (onMouseLeave on the <Sidebar> root).
   478→ *   • click ............. calls toggleSidebar (existing functionality,
   479→ *                         unchanged).
   480→ *
   481→ * Each rail item keeps its OWN Radix Tooltip label (Dashboard, Articles,
   482→ * Profile, etc.) — the lifted `hovered` prop ONLY drives the logo cell's
   483→ * C↔PanelLeftOpen swap; it does NOT touch any item's tooltip. The logo's
   484→ * own "Expand" Tooltip still fires ONLY when the pointer is directly over
   485→ * the logo button (the Tooltip trigger is the button itself, independent
   486→ * of the `hovered` prop).
   487→ *
   488→ * Background is removed ONLY from the temporary Expand-icon state; the
   489→ * normal logo (at rest) keeps its bg-primary background exactly as-is.
   490→ *
   491→ * Implementation: React state lifted to AppSidebar (railHovered +
   492→ * onMouseEnter/onMouseLeave on <Sidebar>) — NOT CSS :hover / group-hover
   493→ * variants. This is deliberate: Tailwind v4 wraps its hover:/group-hover:
   494→ * utilities inside @media (hover: hover), so on any browser that reports
   495→ * (hover: none) (headless, touch, some preview iframes) those utilities
   496→ * NEVER activate and the swap + bg-transparent would silently fail. React
   497→ * mouse events fire on ANY pointer input regardless of the hover media
   498→ * query, so the behavior is identical in every environment. The
   499→ * conditional className + conditional render swap the bg/color/glyph
   500→ * atomically per state. Works in both Light and Dark.
   501→ */
   502→function CollapsedLogoButton({ hovered }: { hovered: boolean }) {
   503→  const { toggleSidebar } = useSidebar();
   504→  return (
   505→    /* Hover-only "Expand" tooltip: instant (provider delayDuration=0),
   506→       appears on mouse-enter, disappears on mouse-leave. Radix closes it
   507→       on any pointer exit — disableHoverableContent guarantees it can
   508→       never linger or trap the pointer. The Tooltip trigger is the logo
   509→       button itself, so the "Expand" TEXT fires ONLY on direct logo hover
   510→       (NOT on rail-level hover of other items) — the C↔PanelLeftOpen ICON
   511→       swap, by contrast, is driven by the lifted `hovered` prop and fires
   512→       for ANY rail item. */
   513→    <Tooltip disableHoverableContent>
   514→      <TooltipTrigger asChild>
   515→        <button
   516→          type="button"
   517→          onClick={toggleSidebar}
   518→          aria-label="Expand sidebar"
   519→          className={cn(
   520→            'flex h-8 w-8 shrink-0 cursor-pointer items-center justify-center rounded-lg font-bold text-sm outline-none transition-opacity hover:opacity-90 focus-visible:ring-2 focus-visible:ring-ring select-none',
   521→            // NORMAL LOGO STATE (at rest): bg-primary black box +
   522→            // text-primary-foreground white "C" — background UNCHANGED,
   523→            // same as the expanded LogoMark.
   524→            // EXPAND-ICON STATE (on hover): bg-transparent (NO background)
   525→            // + text-muted-foreground gray icon — ONLY this temporary
   526→            // state loses the background. The icon stays visible (gray
   527→            // on transparent/page bg) in both Light and Dark mode.
   528→            hovered
   529→              ? 'bg-transparent text-muted-foreground'
   530→              : 'bg-primary text-primary-foreground',
   531→          )}
   532→        >
   533→          {/* Conditional render: "C" logo at rest, PanelLeftOpen icon on
   534→              hover. h-4 w-4 matches the CollapseToggle's PanelLeftClose
   535→              icon size; the icon is the visual opposite of the collapse
   536→              icon so the collapsed logo reads as a clear "expand"
   537→              affordance when the pointer is over it. */}
   538→          {hovered ? (
   539→            <PanelLeftOpen className="h-4 w-4" />
   540→          ) : (
   541→            <span>C</span>
   542→          )}
   543→          <span className="sr-only">Expand sidebar</span>
   544→        </button>
   545→      </TooltipTrigger>
   546→      {/* Collapsed-rail tooltip — uses the SHARED COLLAPSED_TOOLTIP_PROPS
   547→          constant so the "Expand" label has IDENTICAL positioning (side,
   548→          align, sideOffset, collisionPadding) to every other collapsed-
   549→          rail tooltip (ThemeToggle, SimpleNavItem, ExpandableNavItem,
   550→          CollapsedParentNavItem). Portal-based rendering means the bubble
   551→          floats outside the sidebar DOM at z-50 — never clipped by the
   552→          rail's overflow-hidden. */}
   553→      <TooltipContent {...COLLAPSED_TOOLTIP_PROPS}>Expand</TooltipContent>
   554→    </Tooltip>
   555→  );
   556→}

   626→/**
   627→ * Settings parent item rendered on the COLLAPSED 48px rail.
   628→ * Clicking it opens a portal-based popover to the RIGHT of the icon —
   629→ * the sidebar itself NEVER expands, and the popover anchors to this
   630→ * exact button so the "star/sparkle alignment" grid stays intact.
   631→ */
   632→function CollapsedParentNavItem({
   633→  item,
   634→  currentModule,
   635→  currentSubPage,
   636→}: {
   637→  item: NavItem;
   638→  currentModule: string;
   639→  currentSubPage: string | null;
   640→}) {
   641→  const [floatOpen, setFloatOpen] = useState(false);
   642→  const mod = hrefToModule(item.href);
   643→  const isActive = currentModule === mod;
   644→
   645→  const handleChildNavigate = (
   646→    e: React.MouseEvent<HTMLAnchorElement>,
   647→    child: NavItem,
   648→  ) => {
   649→    e.preventDefault();
   650→    e.stopPropagation();
   651→    setFloatOpen(false); // close the floating submenu after navigation…
   652→    // …but do NOT touch sidebar expansion state (rule #7).
   653→    const hash = child.href.replace(/^#/, '');
   654→    const parts = hash.split('/');
   655→    const childSubPage = parts[1] || null;
   656→    useNavigationStore.getState().navigate(parts[0], null, childSubPage);
   657→  };
   658→
   659→  return (
   660→    <SidebarMenuItem>
   661→      <Popover open={floatOpen} onOpenChange={setFloatOpen}>
   662→        <PopoverTrigger asChild>
   663→          <SidebarMenuButton
   664→            isActive={isActive || floatOpen}
   665→            aria-expanded={floatOpen}
   666→            aria-haspopup="menu"
   667→            // Collapsed-rail tooltip — uses the SHARED COLLAPSED_TOOLTIP_PROPS
   668→            // so the parent item's label has IDENTICAL positioning (side,
   669→            // align, sideOffset, collisionPadding) to every other collapsed-
   670→            // rail tooltip (CollapsedLogoButton, ThemeToggle, SimpleNavItem,
   671→            // ExpandableNavItem). Suppressed when the popover is open so the
   672→            // floating submenu never visually conflicts with a lingering
   673→            // label bubble. Portal-based rendering means the bubble floats
   674→            // outside the sidebar DOM at z-50 — never clipped by the rail's
   675→            // overflow-hidden.
   676→            tooltip={
   677→              floatOpen
   678→                ? undefined
   679→                : { ...COLLAPSED_TOOLTIP_PROPS, children: item.label }
   680→            }
   681→            onClick={(e: React.MouseEvent) => {
   682→              e.preventDefault();
   683→              setFloatOpen((o) => !o);
   684→            }}
   685→          >
   686→            <NavIcon name={item.icon} />
   687→            <span>{item.label}</span>
   688→          </SidebarMenuButton>
   689→        </PopoverTrigger>
   690→        {/* Portal → document.body: immune to sidebar overflow clipping.
   691→            Radix handles Esc + outside-click closing and re-anchors at any
   692→            viewport size/zoom via avoidCollisions. */}
   693→        <PopoverContent
   694→          side="right"
   695→          align="start"
   696→          sideOffset={6}
   697→          collisionPadding={12}
   698→          className="w-56 rounded-md border bg-popover p-1.5 shadow-md"
   699→          onEscapeKeyDown={() => setFloatOpen(false)}
   700→        >
   701→          <ul
   702→            role="menu"
   703→            aria-label={`${item.label} submenu`}
   704→            className="flex min-w-0 list-none flex-col gap-0.5 p-0 m-0"
   705→          >
   706→            {item.children!.map((child) => {
   707→              const hash = child.href.replace(/^#/, '');
   708→              const parts = hash.split('/');
   709→              const childSubPage = parts[1] || null;
   710→              const isChildActive =
   711→                currentModule === parts[0] &&
   712→                (!childSubPage || currentSubPage === childSubPage);
   713→              return (
   714→                <SidebarMenuSubItem key={child.label} className="list-none p-0">
   715→                  <SidebarMenuSubButton asChild isActive={isChildActive}>
   716→                    <a
   717→                      href={child.href}
   718→                      role="menuitem"
   719→                      onClick={(e) => handleChildNavigate(e, child)}
   720→                    >
   721→                      <NavIcon name={child.icon} />
   722→                      <span>{child.label}</span>
   723→                    </a>
   724→                  </SidebarMenuSubButton>
   725→                </SidebarMenuSubItem>
   726→              );
   727→            })}
   728→          </ul>
   729→        </PopoverContent>
   730→      </Popover>
   731→    </SidebarMenuItem>
   732→  );
   733→}

   735→function ExpandableNavItem({
   736→  item,
   737→  currentModule,
   738→  currentSubPage,
   739→  isExpanded,
   740→  onToggle,
   741→}: {
   742→  item: NavItem;
   743→  currentModule: string;
   744→  currentSubPage: string | null;
   745→  isExpanded: boolean;
   746→  onToggle: () => void;
   747→}) {
   748→  const mod = hrefToModule(item.href);
   749→  const isActive = currentModule === mod;
   750→  const sectionId = `submenu-${item.label.toLowerCase().replace(/\s+/g, '-')}`;
   751→
   752→  return (
   753→    <SidebarMenuItem>
   754→      <SidebarMenuButton
   755→        // Expandable parent item rendered in EXPANDED state. SidebarMenuButton
   756→        // hides its tooltip whenever state!=="collapsed" (see ui/sidebar.tsx),
   757→        // so this label never shows while expanded — but passing the SAME
   758→        // COLLAPSED_TOOLTIP_PROPS object as every other nav-item renderer
   759→        // keeps the prop consistent across the codebase. If the sidebar
   760→        // toggles between expanded and collapsed while this row stays
   761→        // mounted, the tooltip that appears is identical in positioning to
   762→        // SimpleNavItem / CollapsedParentNavItem / CollapsedLogoButton.
   763→        tooltip={{ ...COLLAPSED_TOOLTIP_PROPS, children: item.label }}
   764→        isActive={isActive}
   765→        onClick={(e: React.MouseEvent) => {
   766→          e.preventDefault();
   767→          onToggle();
   768→        }}
   769→        aria-expanded={isExpanded}
   770→        aria-controls={sectionId}
   771→      >
   772→        <NavIcon name={item.icon} />
   773→        <span>{item.label}</span>
   774→        <ChevronRight
   775→          className={cn(
   776→            'ml-auto h-4 w-4 shrink-0 transition-transform duration-200',
   777→            isExpanded && 'rotate-90',
   778→          )}
   779→        />
   780→      </SidebarMenuButton>
   781→
   782→      {item.badge != null && (
   783→        <SidebarMenuBadge>
   784→          {item.badge}
   785→        </SidebarMenuBadge>
   786→      )}
   787→
   788→      <AccordionSubmenu isOpen={isExpanded} sectionLabel={item.label}>
   789→        {item.children!.map((child) => {
   790→          const hash = child.href.replace(/^#/, '');
   791→          const parts = hash.split('/');
   792→          const childMod = parts[0];
   793→          const childSubPage = parts[1] || null;
   794→          const isChildActive =
   795→            currentModule === childMod &&
   796→            (!childSubPage || currentSubPage === childSubPage);
   797→
   798→          return (
   799→            <SidebarMenuSubItem key={child.label}>
   800→              <SidebarMenuSubButton
   801→                asChild
   802→                isActive={isChildActive}
   803→              >
   804→                <a
   805→                  href={child.href}
   806→                  onClick={(e) => {
   807→                    e.preventDefault();
   808→                    e.stopPropagation();
   809→                    useNavigationStore
   810→                      .getState()
   811→                      .navigate(childMod, null, childSubPage);
   812→                  }}
   813→                >
   814→                  <NavIcon name={child.icon} />
   815→                  <span>{child.label}</span>
   816→                </a>
   817→              </SidebarMenuSubButton>
   818→            </SidebarMenuSubItem>
   819→          );
   820→        })}
   821→      </AccordionSubmenu>
   822→    </SidebarMenuItem>
   823→  );
   824→}

   875→function NavGroupSection({
   876→  group,
   877→  currentModule,
   878→  currentSubPage,
   879→  openSection,
   880→  onToggleSection,
   881→}: {
   882→  group: NavGroup;
   883→  currentModule: string;
   884→  currentSubPage: string | null;
   885→  openSection: string | null;
   886→  onToggleSection: (label: string) => void;
   887→}) {
   888→  const isCollapsedRail = useCollapsedRail();
   889→
   890→  return (
   891→    <SidebarGroup>
   892→      <SidebarGroupContent>
   893→        <SidebarMenu>
   894→          {group.items.map((item) => {
   895→            const hasChildren = item.children && item.children.length > 0;
   896→
   897→            if (hasChildren && isCollapsedRail) {
   898→              /* Collapsed rail → parents with children become floating
   899→                 popovers anchored to their own icon. The rail NEVER
   900→                 auto-expands (#4/#5/#6). */
   901→              return (
   902→                <CollapsedParentNavItem
   903→                  key={item.label}
   904→                  item={item}
   905→                  currentModule={currentModule}
   906→                  currentSubPage={currentSubPage}
   907→                />
   908→              );
   909→            }
   910→
   911→            if (hasChildren) {
   912→              return (
   913→                <ExpandableNavItem
   914→                  key={item.label}
   915→                  item={item}
   916→                  currentModule={currentModule}
   917→                  currentSubPage={currentSubPage}
   918→                  isExpanded={openSection === item.label}
   919→                  onToggle={() => onToggleSection(item.label)}
   920→                />
   921→              );
   922→            }
   923→
   924→            return (
   925→              <SimpleNavItem
   926→                key={item.label}
   927→                item={item}
   928→                currentModule={currentModule}
   929→              />
   930→            );
   931→          })}
   932→        </SidebarMenu>
   933→      </SidebarGroupContent>
   934→    </SidebarGroup>
   935→  );
   936→}

   947→  const openCommandPalette = useCommandPaletteStore((s) => s.open);
   948→  // Sidebar context hook — MUST run before any early return.
   949→  const sidebarCtx = useSidebar();
   950→
   951→  const userRole = user?.role;
   952→  const pagePermissions = user?.pagePermissions ?? null;
   953→  const isPlatformAdmin = userRole === 'PLATFORM_ADMIN' || userRole === 'OWNER';
   954→  const visibleItems = useMemo(() => {
   955→    if (!userRole) return [];
   956→    // Platform admins see the dedicated platform nav, not the client CMS nav.
   957→    const sourceItems = isPlatformAdmin ? PLATFORM_NAV_ITEMS : NAV_ITEMS;
   958→    return getVisibleNavItems(userRole, sourceItems, pagePermissions);
   959→  }, [userRole, pagePermissions, isPlatformAdmin]);
   960→
   961→  /*
   962→   * SINGLE SOURCE OF TRUTH for the expanded top-level section.
   963→   *
   964→   * manualOverride is a tri-state:
   965→   *   null  → not overridden; use route-derived value
   966→   *   ""   → explicitly closed by user
   967→   *   "SEO" → explicitly opened by user
   968→   *
   969→   * When the module changes (navigation event), we reset manualOverride
   970→   * to null so the route-derived value takes over.
   971→   */
   972→  const [manualOverride, setManualOverride] = useState<string | null>(null);
   973→  // Track previous module to detect navigation changes synchronously during render.
   974→  // This is the React-approved pattern for "derived state from props".
   975→  const [prevModule, setPrevModule] = useState(currentModule);
   976→  if (currentModule !== prevModule) {
   977→    setPrevModule(currentModule);
   978→    setManualOverride(null);
   979→  }
   980→
   981→  // Derive the section from the current route hash (EXPLICIT prefix matching)
   982→  const routeDerivedSection = useMemo(() => {
   983→    return getSectionFromHash(window.location.hash);
   984→  }, [currentModule, currentSubPage]);
   985→
   986→  // Compute the actual open section
   987→  const openSection = useMemo(() => {
   988→    if (manualOverride !== null) {
   989→      // User manually toggled: empty string = closed, non-empty = that section
   990→      return manualOverride || null;
   991→    }
   992→    return routeDerivedSection;
   993→  }, [manualOverride, routeDerivedSection]);
   994→
   995→  // Toggle a section open/closed
   996→  const handleToggleSection = useCallback(
   997→    (label: string) => {
   998→      setManualOverride((prev) => {
   999→        // Determine what's currently open (considering route-derived if no override)
  1000→        const currentlyOpen =
  1001→          prev !== null ? (prev || null) : routeDerivedSection;
  1002→        // If clicking the same section that's open, close it
  1003→        if (currentlyOpen === label) {
  1004→          return ''; // explicitly closed
  1005→        }
  1006→        // Otherwise, open the clicked section (closing any other)
  1007→        return label;
  1008→      });
  1009→    },
  1010→    [routeDerivedSection],
  1011→  );
  1012→
  1013→  // Rail-level hover state for the COLLAPSED sidebar. When the mouse is
  1014→  // anywhere over the 48px collapsed rail (any icon — logo, nav items,
  1015→  // Settings, Theme, Notifications, Profile), this flips to true and the

  1020→  // (independent of this state); the logo's own "Expand" Tooltip still
  1021→  // fires only on direct logo hover. React mouse events (not CSS :hover)
  1022→  // so it works in headless / (hover:none) environments too. The
  1023→  // handlers attach to <Sidebar> which forwards ...props to the visible
  1024→  // fixed sidebar-container div, so mouseenter fires the moment the
  1025→  // pointer enters the rail and mouseleave fires only when it leaves the

  1027→  const [railHovered, setRailHovered] = useState(false);

  1029→  if (!user) return null;
  1030→
