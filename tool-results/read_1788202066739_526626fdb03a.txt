    52→const GB_FACTOR = 1024 * 1024 * 1024;
    53→
    54→/** Server-side currency resolution context (from /api/platform/billing/me
    55→ *  → customerCurrencyResolution). The customer never selects a currency —
    56→ *  the server detects it from the request IP. */
    57→interface CustomerCurrencyResolution {
    58→  currency: string;
    59→  countryCode: string;
    60→  countryName: string;
    61→  source: 'ip' | 'default' | 'local';
    62→}
    63→
    64→/** Server-resolved FINAL price for one plan (from /api/platform/billing/me
    65→ *  → planPricing[planId]). Computed by the same resolveCustomerPricing
    66→ *  the checkout route uses — the price/currency displayed here is exactly
    67→ *  what Stripe charges. */
    68→interface ServerPlanPricing {
    69→  planId: string;
    70→  currency: string;
    71→  monthly: number;
    72→  yearly: number;
    73→  source: 'ip' | 'default' | 'local' | 'plan';
    74→  supported: boolean;
    75→  detectedCurrency: string;
    76→  countryCode: string;
    77→  countryName: string;
    78→  regional: boolean;
    79→}

    81→type BillingStateWithCurrency = ClientBillingState & {

    87→  planPricing?: Record<string, ServerPlanPricing>;

    90→/** Resolve the price + currency to display for a plan. Prefers the
    91→ *  SERVER-RESOLVED pricing (planPricing — the same values checkout
    92→ *  charges); falls back to a local pricesByCurrency lookup for older
    93→ *  API responses. */
    94→function resolvePlanPricing(
    95→  plan: {
    96→    id: string;
    97→    pricesByCurrency?: Record<string, { monthly: number; yearly: number }>;
    98→    priceMonthly: number;
    99→    priceYearly: number;
   100→    currency: string;
   101→    interval: 'monthly' | 'yearly';
   102→  },
   103→  serverPricing: ServerPlanPricing | undefined,
   104→  customerCurrency: string,
   105→): { monthly: number; yearly: number; currency: string } {
   106→  if (serverPricing) {
   107→    return {
   108→      monthly: serverPricing.monthly,
   109→      yearly: serverPricing.yearly,
   110→      currency: serverPricing.currency,
   111→    };
   112→  }
   113→  // Legacy fallback — local lookup (display only).
   114→  const entry = plan.pricesByCurrency?.[customerCurrency];
   115→  if (entry) return { monthly: entry.monthly, yearly: entry.yearly, currency: customerCurrency };
   116→  return { monthly: plan.priceMonthly, yearly: plan.priceYearly, currency: plan.currency };
   117→}
   118→
   119→/** One customer-facing plan feature: a label line (rendered with a ✓)
   120→ *  plus optional indented sub-lines. Used for the Platform AI block —
   121→ *  its description + the plan's AI usage limits sit NEXT TO the ✓
   122→ *  Platform AI feature so the customer immediately understands what
   123→ *  is included (never "✓ Platform AI" alone when Platform AI is
   124→ *  enabled). */
   125→interface PlanFeatureItem {
   126→  label: string;
   127→  subLines?: string[];
   128→}
   129→
   130→/** Compact AI-limit number for the customer-facing plan cards:
   131→ *  200000 → "200K", 25000 → "25K", 1500 → "1.5K", 100 → "100",
   132→ *  -1 → "Unlimited". */
   133→function formatAiCount(n: number): string {
   134→  if (n === UNLIMITED) return 'Unlimited';
   135→  if (n >= 1000) {
   136→    const k = n / 1000;
   137→    return `${k % 1 === 0 ? String(k) : k.toFixed(1)}K`;
   138→  }
   139→  return String(n);
   140→}
   141→
   142→/** Derive the customer-facing feature list from the plan's entitlements +
   143→ *  a few standard items (max sites, storage, support tier). Uses the
   144→ *  legacy plan.features array when it has entries (legacy compat for
   145→ *  plans that still have manually-maintained marketing copy, e.g. the
   146→ *  synthetic Internal plan). The structured entitlements are the
   147→ *  single source of truth — the admin no longer maintains a separate
   148→ *  marketing list per plan.
   149→ *
   150→ *  PLATFORM AI: the ✓ Platform AI feature item carries its own
   151→ *  description ("AI provided by the platform") plus the plan's three
   152→ *  AI usage limits as nested sub-lines — shown ONLY while Platform AI
   153→ *  is enabled (Client's Own AI API-only plans never show platform AI
   154→ *  limits; own-API usage is not counted against them). */
   155→function derivePlanFeatures(plan: {
   156→  features: string[];
   157→  entitlements: string[];
   158→  isFree: boolean;
   159→  limits: {
   160→    maxSites: number;
   161→    storageBytes: number;
   162→    aiArticlesPerMonth?: number;
   163→    aiWordsPerMonth?: number;
   164→    aiImagesPerMonth?: number;
   165→  };
   166→}): PlanFeatureItem[] {
   167→  if (plan.features.length > 0) {
   168→    // Legacy marketing copy — plain items, no sub-lines.
   169→    return plan.features.map((f) => ({ label: f }));
   170→  }
   171→  const items: PlanFeatureItem[] = [];
   172→  items.push(
   173→    plan.limits.maxSites === -1
   174→      ? { label: 'Unlimited sites' }
   175→      : { label: `Up to ${plan.limits.maxSites} sites` },
   176→  );
   177→  const gb = plan.limits.storageBytes / GB_FACTOR;
   178→  if (gb >= 1) {
   179→    items.push({ label: `${Math.floor(gb)} GB storage` });
   180→  } else if (plan.limits.storageBytes > 0) {
   181→    items.push({ label: `${plan.limits.storageBytes} bytes storage` });
   182→  } else {
   183→    items.push({ label: 'No storage' });
   184→  }
   185→  items.push({ label: plan.isFree ? 'Community support' : 'Priority support' });
   186→  // Enabled features — ONLY the enabled ones are displayed (each
   187→  // entitlement key renders its label). Platform AI nests its limits
   188→  // directly next to the feature; Client's Own AI API states that
   189→  // platform AI limits do not apply to it. The legacy 'ai_content'
   190→  // alias is skipped (it duplicates the AI feature keys).
   191→  for (const e of plan.entitlements) {
   192→    if (e === 'ai_content') continue;
   193→    const label = ENTITLEMENT_LABELS[e as EntitlementKey];
   194→    if (!label) continue;
   195→    if (e === 'ai_platform') {
   196→      const articles = plan.limits.aiArticlesPerMonth ?? 0;
   197→      const words = plan.limits.aiWordsPerMonth ?? 0;
   198→      const images = plan.limits.aiImagesPerMonth ?? 0;
   199→      items.push({
   200→        label,
   201→        subLines: [
   202→          'AI provided by the platform',
   203→          `${formatAiCount(articles)} AI articles / month`,
   204→          `${formatAiCount(words)} AI words / month`,
   205→          `${formatAiCount(images)} AI images / month`,
   206→        ],
   207→      });
   208→      continue;
   209→    }
   210→    if (e === 'ai_client') {
   211→      items.push({
   212→        label,
   213→        subLines: ["Your own AI provider — platform AI usage limits don't apply"],
   214→      });
   215→      continue;
   216→    }
   217→    items.push({ label });
   218→  }
   219→  return items;
   220→}
   221→
   222→/** Render the derived plan feature items — one ✓ row per item with
   223→ *  optional indented sub-lines (the Platform AI limits next to the
   224→ *  ✓ Platform AI feature). */
   225→function PlanFeatureList({ items }: { items: PlanFeatureItem[] }) {
   226→  return (
   227→    <ul className="space-y-2">
   228→      {items.map((item, idx) => (
   229→        <li key={`${item.label}-${idx}`} className="flex items-start gap-2 text-sm">
   230→          <Check className="h-4 w-4 text-muted-foreground shrink-0 mt-0.5" />
   231→          <div className="min-w-0">
   232→            <span className="block">{item.label}</span>
   233→            {item.subLines?.map((s) => (
   234→              <span key={s} className="block text-xs text-muted-foreground leading-snug mt-0.5">
   235→                {s}
   236→              </span>
   237→            ))}
   238→          </div>
   239→        </li>
   240→      ))}
   241→    </ul>
   242→  );
   243→}

   308→            code?: string;
   309→            status?: number;

   333→          'This plan does not have a Stripe Price ID configured. An admin must wire it via Platform Admin → Edit Plan → Stripe Price ID.',

   530→  //   - Paid plan → checkout API (Stripe Checkout Session) — the body
   531→  //     sends ONLY planId + interval; the currency/price is resolved
   532→  //     SERVER-SIDE from the request IP (the frontend cannot pick it).
   533→  //     The interval is the plan's ENABLED billing period — the same
   534→  //     one its card displays (single-period plans pin to their only
   535→  //     period; both-period plans use the customer's toggle choice),
   536→  //     so the charged amount always matches.
   537→  const handleSelectPlan = (
   538→    plan: {
   539→      id: PlanId;
   540→      name: string;
   541→      price: number;
   542→      isFree: boolean;
   543→      billingMonthly?: boolean;
   544→      billingYearly?: boolean;
   545→      interval: 'monthly' | 'yearly';
   546→    },
   547→    interval: 'monthly' | 'yearly',
   548→  ) => {
   549→    if (plan.isFree) {
   550→      changePlanMutation.mutate({
   551→        planId: plan.id,
   552→        planName: plan.name,
   553→        isUpgrade: isHigherPlan(plan),
   554→      });
   555→    } else {
   556→      checkoutMutation.mutate({
   557→        planId: plan.id,
   558→        planName: plan.name,
   559→        interval,
   560→      });
   561→    }
   562→  };
   563→
   564→  return (
   565→    <div className="max-w-4xl mx-auto space-y-6">
   566→      {/* Page Header */}
   567→      <div>
   568→        <h1 className="text-xl font-bold tracking-tight text-foreground">{t('billing.title')}</h1>
   569→        <p className="text-sm text-muted-foreground mt-1">{t('billing.description')}</p>
   570→        {/* Auto Currency indicator — shows the customer's resolved
   571→            currency + the source (IP-detected / platform default /
   572→            local development). The server resolves this from the
   573→            request IP; the client cannot change currency manually. */}
   574→        {currencySource && (
   575→          <div className="mt-3 flex flex-wrap items-center gap-2">
   576→            <Badge variant="outline" className="text-[10px] font-mono">
   577→              Currency: {customerCurrency}
   578→            </Badge>
   579→            <span className="text-[11px] text-muted-foreground">
   580→              {currencySource === 'ip'
   581→                ? `Auto-detected from your location${customerCountryName ? ` (${customerCountryName})` : ''}`
   582→                : currencySource === 'default'
   583→                  ? 'Platform default'
   584→                  : 'Local development'}
   585→            </span>
   586→          </div>
   587→        )}
   588→      </div>
   589→
   590→      {/* Free-trial-expired banner — surfaces server-side enforcement */}
   591→      {freeTrialExpired && freeTrialExpiresAt && (
   592→        <Card className="border-amber-500/40 bg-amber-50 dark:bg-amber-950/40">
   593→          <CardContent className="flex items-start gap-3 py-4">
   594→            <AlertTriangle className="mt-0.5 h-5 w-5 shrink-0 text-amber-600 dark:text-amber-400" />
   595→            <div className="text-sm">
   596→              <p className="font-semibold text-amber-700 dark:text-amber-300">
   597→                Free trial expired
   598→              </p>
   599→              <p className="text-muted-foreground">
   600→                Your free access expired on {formatDate(freeTrialExpiresAt)}. Upgrade to a paid plan to continue using gated features.
   601→              </p>
   602→            </div>
   603→          </CardContent>
   604→        </Card>
   605→      )}
   606→
   607→      {/* Current Subscription */}
   608→      <Card>
   609→        <CardHeader>
   610→          <CardTitle className="text-base">{t('billing.currentPlan')}</CardTitle>
   611→        </CardHeader>
   612→        <CardContent className="space-y-4">
   613→          <div className="flex items-start justify-between">
   614→            <div className="space-y-1">
   615→              <div className="flex items-center gap-2">
   616→                <CreditCard className="h-5 w-5 text-muted-foreground" />
   617→                <span className="font-semibold text-lg">{currentPlan.name}</span>
   618→                <Badge
   619→                  variant="outline"
   620→                  className={`text-[10px] font-semibold ${getPlanBadgeClasses(currentStorePlan.badgeVariant)}`}
   621→                >
   622→                  {currentPlan.name}
   623→                </Badge>
   624→              </div>
   625→              <p className="text-sm text-muted-foreground ml-7">
   626→                {currentPrice === 0
   627→                  ? t('billing.free')
   628→                  : `${formatMoney(currentPrice, currentDisplayCurrency)}/${normalizeInterval(currentInterval)}`}
   629→              </p>
   630→              {currentPrice > 0 &&
   631→                planPricing?.[currentPlan.id] &&
   632→                planPricing[currentPlan.id].supported &&
   633→                customerCountryName && (
   634→                  <p className="text-[11px] text-muted-foreground ml-7">
   635→                    Priced for {customerCountryName} ({currentDisplayCurrency})
   636→                  </p>
   637→                )}
   638→            </div>
   639→            <Badge
   640→              variant={status === 'active' ? 'default' : 'outline'}
   641→              className="capitalize"
   642→            >
   643→              {status}
   644→            </Badge>
   645→          </div>
   646→
   647→          {/* Current period end / next billing date (DB Subscription source of truth) */}
   648→          {currentPeriodEnd && !isCancelled && (
   649→            <div className="flex items-center gap-2 text-sm text-muted-foreground bg-muted/50 rounded-md p-3">
   650→              <Calendar className="h-4 w-4 shrink-0" />
   651→              <span>
   652→                Next billing: <span className="font-medium text-foreground">{formatDate(currentPeriodEnd)}</span>
   653→              </span>
   654→            </div>
   655→          )}
   656→
   657→          {trialEnd && (
   658→            <div className="flex items-center gap-2 text-sm text-muted-foreground bg-muted/50 rounded-md p-3">
   659→              <Clock className="h-4 w-4 shrink-0" />
   660→              <span>
   661→                {t('billing.trialActive')} — ends {formatDate(trialEnd)}
   662→              </span>
   663→            </div>
   664→          )}
   665→
   666→          {/* Included in your plan — the plan's usage limits (sites,
   667→              storage) plus the ENABLED feature checkmarks (disabled
   668→              features are never displayed). Platform AI shows its AI
   669→              limits NESTED next to the ✓ Platform AI feature (only
   670→              while Platform AI is enabled). Same derivation as the
   671→              plan cards. */}
   672→          <div className="space-y-2">
   673→            <p className="text-xs font-medium text-muted-foreground uppercase tracking-wide">
   674→              Included in your plan
   675→            </p>
   676→            <PlanFeatureList items={derivePlanFeatures(currentPlan)} />
   677→          </div>
   678→
   679→          <Separator />
   680→
   681→          {isCancelled ? (
   682→            <div className="flex items-center justify-end">
   683→              <p className="text-xs text-muted-foreground italic">
   684→                Your subscription is cancelled
   685→              </p>
   686→            </div>
   687→          ) : (
   688→            <div className="flex flex-col sm:flex-row gap-3">
   689→              <div className="flex-1" />
   690→              <div className="flex gap-2">
   691→                <Button variant="outline" size="sm" disabled>
   692→                  {t('billing.managePayment')}
   693→                </Button>
   694→                <Button
   695→                  variant="outline"
   696→                  size="sm"
   697→                  className="text-destructive hover:text-destructive"
   698→                  disabled={cancelMutation.isPending}
   699→                  onClick={handleCancel}
   700→                >
   701→                  {cancelMutation.isPending ? (
   702→                    <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   703→                  ) : null}
   704→                  {t('billing.cancelSubscription')}
   705→                </Button>
   706→              </div>
   707→            </div>
   708→          )}
   709→        </CardContent>
   710→      </Card>
   711→
   712→      {/* Other Plans */}
   713→      <div>
   714→        <h2 className="text-base font-semibold mb-4">{t('billing.otherPlans')}</h2>
   715→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   716→          {otherPlans.map((plan) => {
   717→            const storePlan = getStorePlan(plan.id);
   718→            // Server-resolved FINAL price — the same values checkout charges.
   719→            const planPricingResolved = resolvePlanPricing(plan, planPricing?.[plan.id], customerCurrency);
   720→            const planDisplayCurrency = planPricingResolved.currency;
   721→            const planSupported = planPricing?.[plan.id]?.supported ?? true;
   722→            // ---- Enabled billing periods (admin-configured per plan) ----
   723→            // A disabled period NEVER appears on the card and is never
   724→            // sent to checkout (the backend rejects it too — 400).
   725→            const planHasMonthly = plan.billingMonthly ?? true;
   726→            const planHasYearly = plan.billingYearly ?? true;
   727→            const bothPeriods = planHasMonthly && planHasYearly;
   728→            // The card's displayed period: the customer's toggle choice
   729→            // when both are enabled; the ONLY enabled period otherwise
   730→            // (defaults to the plan's default cadence).
   731→            const cardInterval: 'monthly' | 'yearly' = bothPeriods
   732→              ? (periodByPlan[plan.id] ?? plan.interval)
   733→              : planHasMonthly
   734→                ? 'monthly'
   735→                : 'yearly';
   736→            const planPrice =
   737→              cardInterval === 'yearly' ? planPricingResolved.yearly : planPricingResolved.monthly;
   738→            const isBusy =
   739→              (changePlanMutation.isPending && changePlanMutation.variables?.planId === plan.id) ||
   740→              (checkoutMutation.isPending && checkoutMutation.variables?.planId === plan.id);
   741→            return (
   742→              <Card
   743→                key={plan.id}
   744→                className={`relative ${getPlanCardBorderClasses(storePlan.badgeVariant)}`}
   745→              >
   746→                <CardHeader className="pb-3">
   747→                  <div className="flex items-center justify-between">
   748→                    <CardTitle className="text-base">{plan.name}</CardTitle>
   749→                    <Badge
   750→                      variant="outline"
   751→                      className={`text-[10px] font-semibold ${getPlanBadgeClasses(storePlan.badgeVariant)}`}
   752→                    >
   753→                      {plan.name}
   754→                    </Badge>
   755→                  </div>
   756→                </CardHeader>
   757→                <CardContent className="space-y-4">
   758→                  <div>
   759→                    {/* Monthly / Yearly switch — ONLY for plans with BOTH
   760→                        periods enabled. Single-period plans have no
   761→                        switch: their only period is always shown. */}
   762→                    {bothPeriods && !plan.isFree && (
   763→                      <div className="inline-flex items-center rounded-full border bg-muted/40 p-0.5 mb-2">
   764→                        <button
   765→                          type="button"
   766→                          onClick={() =>
   767→                            setPeriodByPlan((cur) => ({ ...cur, [plan.id]: 'monthly' }))
   768→                          }
   769→                          className={`h-7 rounded-full px-4 text-xs font-medium transition-colors ${
   770→                            cardInterval === 'monthly'
   771→                              ? 'bg-foreground text-background'
   772→                              : 'text-muted-foreground hover:text-foreground'
   773→                          }`}
   774→                        >
   775→                          Monthly
   776→                        </button>
   777→                        <button
   778→                          type="button"
   779→                          onClick={() =>
   780→                            setPeriodByPlan((cur) => ({ ...cur, [plan.id]: 'yearly' }))
   781→                          }
   782→                          className={`h-7 rounded-full px-4 text-xs font-medium transition-colors ${
   783→                            cardInterval === 'yearly'
   784→                              ? 'bg-foreground text-background'
   785→                              : 'text-muted-foreground hover:text-foreground'
   786→                          }`}
   787→                        >
   788→                          Yearly
   789→                        </button>
   790→                      </div>
   791→                    )}
   792→                    <span className="text-2xl font-bold">
   793→                      {planPrice === 0 ? t('billing.free') : formatMoney(planPrice, planDisplayCurrency)}
   794→                    </span>
   795→                    {planPrice > 0 && (
   796→                      <span className="text-sm text-muted-foreground ml-1">
   797→                        /{normalizeInterval(cardInterval)}
   798→                      </span>
   799→                    )}
   800→                    {planPrice > 0 && customerCountryName && planSupported && (
   801→                      <p className="text-[11px] text-muted-foreground mt-1">
   802→                        Priced for {customerCountryName} ({planDisplayCurrency})
   803→                      </p>
   804→                    )}
   805→                    {planPrice > 0 && !planSupported && planPricing?.[plan.id] && (
   806→                      <p className="text-[11px] text-muted-foreground mt-1">
   807→                        Priced in {planDisplayCurrency} (plan default) — {planPricing[plan.id].detectedCurrency} not
   808→                        configured for this plan
   809→                      </p>
   810→                    )}
   811→                  </div>
   812→                  <Separator />
   813→                  <PlanFeatureList items={derivePlanFeatures(plan)} />
   814→                  <Button
   815→                    className="w-full"
   816→                    variant={isHigherPlan(plan) ? 'default' : 'outline'}
   817→                    disabled={isBusy}
   818→                    onClick={() => handleSelectPlan(plan, cardInterval)}
   819→                  >
   820→                    {isBusy ? (
   821→                      <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   822→                    ) : null}
   823→                    {getActionLabel(plan)}
   824→                  </Button>
   825→                </CardContent>
   826→              </Card>
   827→            );
   828→          })}
   829→        </div>
   830→      </div>
   831→
   832→      {/* Payment History */}
   833→      <Card>
   834→        <CardHeader>
   835→          <CardTitle className="text-base">{t('billing.paymentHistory')}</CardTitle>
   836→        </CardHeader>
   837→        <CardContent>
   838→          {billingState.paymentHistory.length === 0 ? (
   839→            <div className="flex flex-col items-center justify-center py-12 text-center">
   840→              <div className="h-12 w-12 rounded-full bg-muted flex items-center justify-center mb-4">
   841→                <Receipt className="h-6 w-6 text-muted-foreground" />
   842→              </div>
   843→              <h3 className="font-medium text-sm">{t('billing.noPayments')}</h3>
   844→            </div>
   845→          ) : (
   846→            <div className="overflow-x-auto">
   847→              <table className="w-full text-sm">
   848→                <thead>
   849→                  <tr className="border-b text-left">
   850→                    <th className="pb-2 font-medium text-xs text-muted-foreground">{t('billing.invoice')}</th>
   851→                    <th className="pb-2 font-medium text-xs text-muted-foreground">Plan</th>
   852→                    <th className="pb-2 font-medium text-xs text-muted-foreground text-right">{t('billing.amount')}</th>
   853→                    <th className="pb-2 font-medium text-xs text-muted-foreground">{t('billing.status')}</th>
   854→                    <th className="pb-2 font-medium text-xs text-muted-foreground">Method</th>
   855→                    <th className="pb-2 font-medium text-xs text-muted-foreground text-right">{t('billing.date')}</th>
   856→                  </tr>
   857→                </thead>
   858→                <tbody className="divide-y">
   859→                  {billingState.paymentHistory.map((p: Payment) => {
   860→                    const planForPayment = billingState.allPlans.find((pl) => pl.id === p.planId);
   861→                    return (
   862→                      <tr key={p.id} className="hover:bg-accent/30 transition-colors">
   863→                        <td className="py-2.5 pr-4">
   864→                          <span className="font-mono text-xs">{p.invoiceNumber}</span>
   865→                        </td>
   866→                        <td className="py-2.5 pr-4">
   867→                          <Badge
   868→                            variant="outline"
   869→                            className={`text-[10px] font-semibold ${getPlanBadgeClasses(getStorePlan(p.planId).badgeVariant)}`}
   870→                          >
   871→                            {planForPayment?.name ?? p.planId}
   872→                          </Badge>
   873→                        </td>
   874→                        <td className="py-2.5 pr-4 text-right font-medium">
   875→                          {formatCurrency(p.amount, p.currency)}
   876→                        </td>
   877→                        <td className="py-2.5 pr-4">
   878→                          <PaymentStatusBadge status={p.status} />
   879→                        </td>
   880→                        <td className="py-2.5 pr-4 text-xs text-muted-foreground">{p.method}</td>
   881→                        <td className="py-2.5 text-right text-xs text-muted-foreground">
   882→                          {formatDate(p.date)}
   883→                        </td>
   884→                      </tr>
   885→                    );
   886→                  })}
   887→                </tbody>
   888→              </table>
   889→            </div>
   890→          )}
   891→        </CardContent>
   892→      </Card>
   893→    </div>
   894→  );
