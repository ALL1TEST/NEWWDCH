   102→/** One customer-facing plan feature: a label line (rendered with a ✓)
   103→ *  plus optional indented sub-lines. Used for the Platform AI block —
   104→ *  its description + the plan's AI usage limits sit NEXT TO the ✓
   105→ *  Platform AI feature so the customer immediately understands what
   106→ *  is included (never "✓ Platform AI" alone when Platform AI is
   107→ *  enabled). */
   108→interface PlanFeatureItem {
   109→  label: string;
   110→  subLines?: string[];
   111→}
   112→
   113→/** Compact AI-limit number for the customer-facing plan cards:
   114→ *  200000 → "200K", 25000 → "25K", 1500 → "1.5K", 100 → "100",
   115→ *  -1 → "Unlimited". */
   116→function formatAiCount(n: number): string {
   117→  if (n === UNLIMITED) return 'Unlimited';
   118→  if (n >= 1000) {
   119→    const k = n / 1000;
   120→    return `${k % 1 === 0 ? String(k) : k.toFixed(1)}K`;
   121→  }
   122→  return String(n);
   123→}
   124→
   125→/** Derive the customer-facing feature list from the plan's entitlements +
   126→ *  a few standard items (max sites, storage, support tier). Uses the
   127→ *  legacy plan.features array when it has entries (legacy compat for
   128→ *  plans that still have manually-maintained marketing copy, e.g. the
   129→ *  synthetic Internal plan). The structured entitlements are the
   130→ *  single source of truth — the admin no longer maintains a separate
   131→ *  marketing list per plan.
   132→ *
   133→ *  PLATFORM AI: the ✓ Platform AI feature item carries its own
   134→ *  description ("AI provided by the platform") plus the plan's AI
   135→ *  usage limits (article + image generations per month — words/tokens
   136→ *  are never metered) as nested sub-lines — shown ONLY while Platform
   137→ *  AI is enabled (Client's Own AI API-only plans never show platform AI
   138→ *  limits; own-API usage is not counted against them). */
   139→function derivePlanFeatures(plan: {
   140→  features: string[];
   141→  entitlements: string[];
   142→  isFree: boolean;
   143→  limits: {
   144→    maxSites: number;
   145→    storageBytes: number;
   146→    aiArticlesPerMonth?: number;
   147→    aiImagesPerMonth?: number;
   148→  };
   149→}): PlanFeatureItem[] {
   150→  if (plan.features.length > 0) {
   151→    // Legacy marketing copy — plain items, no sub-lines.
   152→    return plan.features.map((f) => ({ label: f }));
   153→  }
   154→  const items: PlanFeatureItem[] = [];
   155→  items.push(
   156→    plan.limits.maxSites === -1
   157→      ? { label: 'Unlimited sites' }
   158→      : { label: `Up to ${plan.limits.maxSites} sites` },
   159→  );
   160→  const gb = plan.limits.storageBytes / GB_FACTOR;
   161→  if (gb >= 1) {
   162→    items.push({ label: `${Math.floor(gb)} GB storage` });
   163→  } else if (plan.limits.storageBytes > 0) {
   164→    items.push({ label: `${plan.limits.storageBytes} bytes storage` });
   165→  } else {
   166→    items.push({ label: 'No storage' });
   167→  }
   168→  items.push({ label: plan.isFree ? 'Community support' : 'Priority support' });
   169→  // Enabled features — ONLY the enabled ones are displayed (each
   170→  // entitlement key renders its label). Platform AI nests its limits
   171→  // directly next to the feature; Client's Own AI API states that
   172→  // platform AI limits do not apply to it. The legacy 'ai_content'
   173→  // alias is skipped (it duplicates the AI feature keys).
   174→  for (const e of plan.entitlements) {
   175→    if (e === 'ai_content') continue;
   176→    const label = ENTITLEMENT_LABELS[e as EntitlementKey];
   177→    if (!label) continue;
   178→    if (e === 'ai_platform') {
   179→      const articles = plan.limits.aiArticlesPerMonth ?? 0;
   180→      const images = plan.limits.aiImagesPerMonth ?? 0;
   181→      // Display-only: the monthly usage-limit logic is unchanged; the
   182→      // customer-facing lines just drop the "/ month" suffix (they are
   183→      // usage limits, not prices) and render as separate lines.
   184→      items.push({
   185→        label,
   186→        subLines: [
   187→          'AI provided by the platform',
   188→          `${formatAiCount(articles)} AI articles`,
   189→          `${formatAiCount(images)} AI images`,
   190→        ],
   191→      });
   192→      continue;
   193→    }
   194→    if (e === 'ai_client') {
   195→      items.push({
   196→        label,
   197→        subLines: ["Your own AI provider — platform AI usage limits don't apply"],
   198→      });
   199→      continue;
   200→    }
   201→    items.push({ label });
   202→  }
   203→  return items;
   204→}
   205→
   206→/** Render the derived plan feature items — one ✓ row per item with
   207→ *  optional indented sub-lines (the Platform AI limits next to the
   208→ *  ✓ Platform AI feature). */
   209→function PlanFeatureList({ items }: { items: PlanFeatureItem[] }) {
   210→  return (
   211→    <ul className="space-y-2">
   212→      {items.map((item, idx) => (
   213→        <li key={`${item.label}-${idx}`} className="flex items-start gap-2 text-sm">
   214→          <Check className="h-4 w-4 text-muted-foreground shrink-0 mt-0.5" />
   215→          <div className="min-w-0">
   216→            <span className="block">{item.label}</span>
   217→            {item.subLines?.map((s) => (
   218→              <span key={s} className="block text-xs text-muted-foreground leading-snug mt-0.5">
   219→                {s}
   220→              </span>
   221→            ))}
   222→          </div>
   223→        </li>
   224→      ))}
   225→    </ul>
   226→  );
   227→}

   531→  // interval (the period they are actually billed on).
   532→  const showPeriodSelector = otherPlans.some((p) => p.billingYearly ?? true);
   533→  const effectivePeriod: 'monthly' | 'yearly' =
   534→    globalPeriod ?? (currentInterval === 'yearly' ? 'yearly' : 'monthly');
   535→
   536→  // Billing period of the CURRENT subscription (used for the factual
   537→  // "Billed monthly / Billed yearly" line — the price/currency itself
   538→  // is NOT duplicated in this section; the plan cards and the Stripe
   539→  // Customer Portal carry the amounts).
   540→
   541→  const isHigherPlan = (plan: { price: number }) => plan.price > currentPlan.price;
   542→  const getActionLabel = (plan: { price: number; isFree: boolean }) => {
   543→    if (plan.isFree) return t('billing.changePlan');
   544→    if (isHigherPlan(plan)) return t('billing.upgrade');
   545→    if (plan.price < currentPlan.price) return t('billing.downgrade');
   546→    return t('billing.changePlan');
   547→  };
   548→
   549→  const handleCancel = () => {
   550→    if (window.confirm('Are you sure you want to cancel your subscription?')) {
   551→      cancelMutation.mutate();
   552→    }
   553→  };
   554→
   555→  // Handle a plan change / upgrade:
   556→  //   - Free plan → direct change-plan API (no Stripe needed)
   557→  //   - Paid plan → checkout API (Stripe Checkout Session) — the body
   558→  //     sends ONLY planId + interval; the currency/price is resolved
   559→  //     SERVER-SIDE from the request IP (the frontend cannot pick it).
   560→  //     The interval is the card's DISPLAYED billing period — the
   561→  //     global Monthly/Yearly selection when the plan supports it,
   562→  //     otherwise the plan's only enabled period (the backend rejects
   563→  //     a disabled interval) — so the charged amount always matches.
   564→  const handleSelectPlan = (
   565→    plan: {
   566→      id: PlanId;
   567→      name: string;
   568→      price: number;
   569→      isFree: boolean;
   570→      billingMonthly?: boolean;
   571→      billingYearly?: boolean;
   572→      interval: 'monthly' | 'yearly';
   573→    },
   574→    interval: 'monthly' | 'yearly',
   575→  ) => {
   576→    if (plan.isFree) {
   577→      changePlanMutation.mutate({
   578→        planId: plan.id,
   579→        planName: plan.name,
   580→        isUpgrade: isHigherPlan(plan),
   581→      });
   582→    } else {
   583→      checkoutMutation.mutate({
   584→        planId: plan.id,
   585→        planName: plan.name,
   586→        interval,
   587→      });
   588→    }
   589→  };
   590→
   591→  return (
   592→    <div className="max-w-6xl mx-auto space-y-8">
   593→      {/* Page Header */}
   594→      <div>
   595→        <h1 className="text-xl font-bold tracking-tight text-foreground">{t('billing.title')}</h1>
   596→      </div>
   597→
   598→      {/* Free-trial-expired banner — surfaces server-side enforcement */}
   599→      {freeTrialExpired && freeTrialExpiresAt && (
   600→        <Card className="border-amber-500/40 bg-amber-50 dark:bg-amber-950/40">
   601→          <CardContent className="flex items-start gap-3 py-4">
   602→            <AlertTriangle className="mt-0.5 h-5 w-5 shrink-0 text-amber-600 dark:text-amber-400" />
   603→            <div className="text-sm">
   604→              <p className="font-semibold text-amber-700 dark:text-amber-300">
   605→                Free trial expired
   606→              </p>
   607→              <p className="text-muted-foreground">
   608→                Your free access expired on {formatDate(freeTrialExpiresAt)}. Upgrade to a paid plan to continue using gated features.
   609→              </p>
   610→            </div>
   611→          </CardContent>
   612→        </Card>
   613→      )}
   614→
   615→      {/* Current Subscription — clean SaaS billing layout:
   616→          plan name + subscription status (header), billing period,
   617→          included features (with Platform AI + its limits when
   618→          enabled), and the billing actions. Price/currency is NOT
   619→          duplicated here — the plan cards and the Stripe Customer
   620→          Portal carry the amounts. */}
   621→      <Card>
   622→        <CardHeader className="flex flex-row flex-wrap items-center justify-between gap-2 space-y-0">
   623→          <CardTitle className="text-base">{t('billing.currentPlan')}</CardTitle>
   624→          {/* Subscription status — the SAME green "Active" badge
   625→              treatment the dashboard uses for statuses (SubStatusBadge). */}
   626→          <SubStatusBadge status={status} />
   627→        </CardHeader>
   628→        <CardContent className="space-y-4">
   629→          <div className="space-y-1 min-w-0">
   630→            <span className="text-2xl font-semibold tracking-tight">{currentPlan.name}</span>
   631→            {/* Factual billing period of the current subscription —
   632→                no price/currency presentation. */}
   633→            <p className="text-sm text-muted-foreground">
   634→              {currentInterval === 'yearly' ? 'Billed yearly' : 'Billed monthly'}
   635→            </p>
   636→          </div>
   637→
   638→          {trialEnd && (
   639→            <div className="flex items-center gap-2 text-sm text-muted-foreground bg-muted/50 rounded-md p-3">
   640→              <Clock className="h-4 w-4 shrink-0" />
   641→              <span>
   642→                {t('billing.trialActive')} — ends {formatDate(trialEnd)}
   643→              </span>
   644→            </div>
   645→          )}
   646→
   647→          {/* Included in your plan — the plan's usage limits (sites,
   648→              storage) plus the ENABLED feature checkmarks (disabled
   649→              features are never displayed). Platform AI shows its AI
   650→              limits NESTED next to the ✓ Platform AI feature (only
   651→              while Platform AI is enabled). Same derivation as the
   652→              plan cards. */}
   653→          <div className="space-y-2">
   654→            <p className="text-xs font-medium text-muted-foreground uppercase tracking-wide">
   655→              Included in your plan
   656→            </p>
   657→            <PlanFeatureList items={derivePlanFeatures(currentPlan)} />
   658→          </div>
   659→
   660→          <Separator />
   661→
   662→          {isCancelled ? (
   663→            <div className="flex items-center justify-end">
   664→              <p className="text-xs text-muted-foreground italic">
   665→                Your subscription is cancelled
   666→              </p>
   667→            </div>
   668→          ) : (
   669→            <div className="flex flex-col sm:flex-row gap-3">
   670→              <div className="flex-1" />
   671→              {/* flex-wrap: the two action buttons never overflow the
   672→                  card on narrow (mobile) viewports — they wrap as whole
   673→                  units instead (display-only fix; handlers unchanged). */}
   674→              <div className="flex flex-wrap gap-2">
   675→                {/* Manage Payment Method — opens the REAL Stripe
   676→                    Customer Portal (payment method, card details,
   677→                    billing information) for this customer. */}
   678→                <Button
   679→                  size="sm"
   680→                  disabled={portalMutation.isPending}
   681→                  onClick={() => portalMutation.mutate()}
   682→                >
   683→                  {portalMutation.isPending ? (
   684→                    <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   685→                  ) : null}
   686→                  {t('billing.managePayment')}
   687→                </Button>
   688→                <Button
   689→                  variant="outline"
   690→                  size="sm"
   691→                  className="text-destructive hover:text-destructive"
   692→                  disabled={cancelMutation.isPending}
   693→                  onClick={handleCancel}
   694→                >
   695→                  {cancelMutation.isPending ? (
   696→                    <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   697→                  ) : null}
   698→                  {t('billing.cancelSubscription')}
   699→                </Button>
   700→              </div>
   701→            </div>
   702→          )}
   703→        </CardContent>
   704→      </Card>
   705→
   706→      {/* Other Plans — ONE global Monthly / Yearly selector above ALL
   707→          cards (never repeated inside a card). Card layout: large plan
   708→          name → large price → divider → feature list with check icons →
   709→          action button pinned to the bottom. */}
   710→      <div>
   711→        <div className="flex flex-wrap items-center justify-between gap-3 mb-5">
   712→          <h2 className="text-base font-semibold">{t('billing.otherPlans')}</h2>
   713→          {/* Global billing-period selector — shown only when at least
   714→              one available plan supports yearly billing; it then drives
   715→              the displayed billing period of every plan card below. */}
   716→          {showPeriodSelector && (
   717→            <div
   718→              role="group"
   719→              aria-label="Billing period"
   720→              className="inline-flex items-center rounded-full border bg-muted/40 p-1"
   721→            >
   722→              <button
   723→                type="button"
   724→                onClick={() => setGlobalPeriod('monthly')}
   725→                aria-pressed={effectivePeriod === 'monthly'}
   726→                className={`h-8 rounded-full px-5 text-xs font-medium transition-colors ${
   727→                  effectivePeriod === 'monthly'
   728→                    ? 'bg-foreground text-background shadow-sm'
   729→                    : 'text-muted-foreground hover:text-foreground'
   730→                }`}
   731→              >
   732→                Monthly
   733→              </button>
   734→              <button
   735→                type="button"
   736→                onClick={() => setGlobalPeriod('yearly')}
   737→                aria-pressed={effectivePeriod === 'yearly'}
   738→                className={`h-8 rounded-full px-5 text-xs font-medium transition-colors ${
   739→                  effectivePeriod === 'yearly'
   740→                    ? 'bg-foreground text-background shadow-sm'
   741→                    : 'text-muted-foreground hover:text-foreground'
   742→                }`}
   743→              >
   744→                Yearly
   745→              </button>
   746→            </div>
   747→          )}
   748→        </div>
   749→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-5">
   750→          {otherPlans.map((plan) => {
   751→            const storePlan = getStorePlan(plan.id);
   752→            // Display price/currency = the plan's CONFIGURED base fields —
   753→            // the SAME source the Platform Admin Plans & Pricing page reads.
   754→            const planDisplayCurrency = plan.currency;
   755→            // ---- Enabled billing periods (admin-configured per plan) ----
   756→            const planHasMonthly = plan.billingMonthly ?? true;
   757→            const planHasYearly = plan.billingYearly ?? true;
   758→            // The card's displayed period: the GLOBAL selection when this
   759→            // plan supports it; otherwise the plan's ONLY enabled period
   760→            // (pinned — existing billing logic: a disabled period is
   761→            // never displayed and never sent to checkout, so no
   762→            // invalid/fake price can appear; the backend rejects a
   763→            // disabled interval too).
   764→            const followsSelection =
   765→              effectivePeriod === 'monthly' ? planHasMonthly : planHasYearly;
   766→            const cardInterval: 'monthly' | 'yearly' = followsSelection
   767→              ? effectivePeriod
   768→              : planHasMonthly
   769→                ? 'monthly'
   770→                : planHasYearly
   771→                  ? 'yearly'
   772→                  : plan.interval;
   773→            const planPrice =
   774→              cardInterval === 'yearly' ? plan.priceYearly : plan.priceMonthly;
   775→            const isBusy =
   776→              (changePlanMutation.isPending && changePlanMutation.variables?.planId === plan.id) ||
   777→              (checkoutMutation.isPending && checkoutMutation.variables?.planId === plan.id);
   778→            return (
   779→              <Card
   780→                key={plan.id}
   781→                className={`relative flex flex-col ${getPlanCardBorderClasses(storePlan.badgeVariant)}`}
   782→              >
   783→                <CardContent className="flex flex-col flex-1">
   784→                  {/* Large plan name — keeps the plan's color identity via
   785→                      the card border + its outline badge (text + border
   786→                      color only, NO filled background — one identical
   787→                      treatment for every plan). */}
   788→                  <div className="flex flex-wrap items-start justify-between gap-2">
   789→                    <h3 className="text-2xl font-semibold tracking-tight text-foreground">
   790→                      {plan.name}
   791→                    </h3>
   792→                    <Badge
   793→                      variant="outline"
   794→                      className={`text-[10px] font-semibold shrink-0 bg-transparent ${getPlanBadgeOutlineClasses(storePlan.badgeVariant)}`}
   795→                    >
   796→                      {plan.name}
   797→                    </Badge>
   798→                  </div>
   799→                  {/* Large price — explicit period suffix so the amount is
   800→                      always unambiguous (also for pinned-period cards); a
   801→                      zero price shows the plain formatted amount ("$0")
   802→                      with no suffix, like Platform Admin Plans & Pricing. */}
   803→                  <div className="mt-4">
   804→                    <div className="flex items-baseline gap-1.5">
   805→                      <span className="text-4xl font-bold tracking-tight text-foreground">
   806→                        {formatMoney(planPrice, planDisplayCurrency)}
   807→                      </span>
   808→                      {planPrice > 0 && (
   809→                        <span className="text-sm text-muted-foreground whitespace-nowrap">
   810→                          / {cardInterval === 'yearly' ? 'year' : 'month'}
   811→                        </span>
   812→                      )}
   813→                    </div>
   814→                    {/* A plan that cannot follow the global selection keeps
   815→                        its only enabled period — stated explicitly (never
   816→                        an invalid price for the selected period). */}
   817→                    {!plan.isFree && !followsSelection && planHasMonthly !== planHasYearly && (
   818→                      <p className="text-[11px] text-muted-foreground mt-1.5">
   819→                        {planHasMonthly ? 'Monthly billing only' : 'Yearly billing only'}
   820→                      </p>
   821→                    )}
   822→                  </div>
   823→                  <Separator className="my-5" />
   824→                  {/* Feature list with check icons */}
   825→                  <div className="flex-1">
   826→                    <PlanFeatureList items={derivePlanFeatures(plan)} />
   827→                  </div>
   828→                  {/* Action button pinned to the card bottom (label +
   829→                      behavior unchanged: Upgrade / Downgrade / Change
   830→                      Plan per the existing subscription logic). */}
   831→                  <Button
   832→                    className="mt-6 w-full"
   833→                    variant={isHigherPlan(plan) ? 'default' : 'outline'}
   834→                    disabled={isBusy}
   835→                    onClick={() => handleSelectPlan(plan, cardInterval)}
   836→                  >
   837→                    {isBusy ? (
   838→                      <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   839→                    ) : null}
   840→                    {getActionLabel(plan)}
   841→                  </Button>
   842→                </CardContent>
   843→              </Card>
   844→            );
   845→          })}
   846→        </div>
   847→      </div>
   848→
   849→      {/* Payment History — simple 3-column table (Plan | Status |
   850→          Date) over the REAL payment/subscription history data.
   851→          Invoice / Amount / Method columns were removed; amounts and
   852→          invoices are managed in the Stripe Customer Portal. */}
   853→      <Card>
   854→        <CardHeader>
   855→          <CardTitle className="text-base">{t('billing.paymentHistory')}</CardTitle>
   856→        </CardHeader>
   857→        <CardContent>
   858→          {billingState.paymentHistory.length === 0 ? (
   859→            <div className="flex flex-col items-center justify-center py-12 text-center">
   860→              <div className="h-12 w-12 rounded-full bg-muted flex items-center justify-center mb-4">
   861→                <Receipt className="h-6 w-6 text-muted-foreground" />
   862→              </div>
   863→              <h3 className="font-medium text-sm">{t('billing.noPayments')}</h3>
   864→            </div>
   865→          ) : (
   866→            <div className="overflow-x-auto">
   867→              <table className="w-full text-sm">
   868→                <thead>
   869→                  <tr className="border-b text-left">
   870→                    <th className="pb-2 font-medium text-xs text-muted-foreground">Plan</th>
   871→                    <th className="pb-2 font-medium text-xs text-muted-foreground">{t('billing.status')}</th>
   872→                    <th className="pb-2 font-medium text-xs text-muted-foreground text-right">{t('billing.date')}</th>
   873→                  </tr>
   874→                </thead>
   875→                <tbody className="divide-y">
   876→                  {billingState.paymentHistory.map((p: Payment) => {
   877→                    const planForPayment = billingState.allPlans.find((pl) => pl.id === p.planId);
   878→                    return (
   879→                      <tr key={p.id} className="hover:bg-accent/30 transition-colors">
   880→                        {/* Plan — the actual plan name of the payment,
   881→                            in the page's uniform outline badge
   882→                            treatment. */}
   883→                        <td className="py-2.5 pr-4">
   884→                          <Badge
   885→                            variant="outline"
   886→                            className={`text-[10px] font-semibold bg-transparent ${getPlanBadgeOutlineClasses(getStorePlan(p.planId).badgeVariant)}`}
   887→                          >
   888→                            {planForPayment?.name ?? p.planId}
   889→                          </Badge>
   890→                        </td>
   891→                        {/* Status — existing green/success payment
   892→                            status badge (Paid). */}
   893→                        <td className="py-2.5 pr-4">
   894→                          <PaymentStatusBadge status={p.status} />
   895→                        </td>
   896→                        {/* Date — the actual payment date. */}
   897→                        <td className="py-2.5 text-right text-xs text-muted-foreground">
   898→                          {formatDate(p.date)}
   899→                        </td>
   900→                      </tr>
   901→                    );
   902→                  })}
   903→                </tbody>
   904→              </table>
   905→            </div>
   906→          )}
   907→        </CardContent>
   908→      </Card>
   909→    </div>
   910→  );
   911→}
