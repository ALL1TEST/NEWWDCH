   158→// -------------------- Collapsed-Rail Tooltip Positioning -------------------
   159→/*
   160→ * SINGLE source of truth for the positioning of every tooltip/label that
   161→ * appears when the sidebar is collapsed into the 48px icon rail.
   162→ *
   163→ *   side="right"         → opens to the RIGHT of the 48px collapsed rail,
   164→ *                          fully inside the main viewport (never overlaps
   165→ *                          the rail, never opens to the left/outside).
   166→ *   align="center"       → vertically centers the bubble on the icon's
   167→ *                          32×32 hover target so labels line up across
   168→ *                          every row of the rail (logo, theme, nav items,
   169→ *                          parent items with submenu).
   170→ *   sideOffset=8         → ~8px visible gap from the trigger button's right
   171→ *                          edge (button right edge ~x=40 inside the 48px
   172→ *                          rail, tooltip left edge ~x=48 = the rail's
   173→ *                          right edge — flush with the RAIL, NOT with the
   174→ *                          icon, so the bubble never touches the glyph).
   175→ *   collisionPadding=12 → 12px viewport-edge collision padding so the
   176→ *                          bubble can never be clipped at any viewport
   177→ *                          edge or scroll past the visible area.
   178→ *
   179→ * Portal-based rendering (Radix TooltipPrimitive.Portal → document.body,
   180→ * wired inside src/components/ui/tooltip.tsx) means every tooltip is
   181→ * rendered OUTSIDE the sidebar DOM tree — so the sidebar's
   182→ * overflow-hidden / overflow-x-hidden on the rail container can NEVER
   183→ * clip or hide the bubble. The tooltip floats above every other
   184→ * element at z-50.
   185→ *
   186→ * Used by:
   187→ *   • CollapsedLogoButton         (header "Expand" tooltip)
   188→ *   • SimpleNavItem              (leaf nav items: Dashboard, Articles,
   189→ *                                  Calendar, Media, Users, Comments,
   190→ *                                  Newsletter, SEO, AI, Automation)
   191→ *   • ExpandableNavItem          (parent nav items in EXPANDED state —
   192→ *                                  tooltip is hidden by SidebarMenuButton
   193→ *                                  when state!=="collapsed", but the
   194→ *                                  positioning object is still applied
   195→ *                                  so the prop stays consistent across
   196→ *                                  every nav item renderer)
   197→ *   • CollapsedParentNavItem     (parent nav items in COLLAPSED state —
   198→ *                                  label tooltip shows when the floating
   199→ *                                  popover is CLOSED; when the popover is
   200→ *                                  open, no tooltip shows to avoid visual
   201→ *                                  conflict)
   202→ *
   203→ * The ThemeToggle component (src/components/layout/theme-toggle.tsx) lives
   204→ * in a separate file and inlines the SAME four values — see the comment
   205→ * there for the rationale.
   206→ */

   417→/**
   418→ * Collapsed-rail logo cell. Since the extra collapse-toggle icon was removed
   419→ * from below the logo, THIS cell is the explicit expand control:
   420→ *   • identical 32×32 grid geometry as every other rail icon
   421→ *   • click toggles the sidebar open (SidebarRail edge strip still works)
   422→ * No new icon is introduced and no empty clickable ghost remains.
   423→ *
   424→ * HOVER BEHAVIOR (collapsed state only — this button is rendered ONLY
   425→ * inside the collapsed-rail cluster, so the expanded LogoMark is untouched):
   426→ * The `hovered` prop is LIFTED to the rail level (see AppSidebar's
   427→ * `railHovered` state + the onMouseEnter/onMouseLeave handlers on the
   428→ * <Sidebar> root). This means the swap fires when the mouse is ANYWHERE
   429→ * over the collapsed 48px rail — NOT only when the pointer is directly
   430→ * over the logo cell:
   431→ *   • rail at rest ........  shows the "C" logo mark on its bg-primary
   432→ *                            black box (NORMAL LOGO — background UNCHANGED,
   433→ *                            same as the expanded LogoMark).
   434→ *   • mouse over ANY rail   the "C" is replaced by a PanelLeftOpen icon
   435→ *     item (Dashboard,       AND the button's background becomes
   436→ *     Articles, …, Profile,  transparent so ONLY the Expand icon is
   437→ *     Theme, Settings, …)    visible with no box behind it. The icon
   438→ *                            color switches to muted-foreground (gray)
   439→ *                            so it stays visible against the
   440→ *                            transparent/page bg in both Light and Dark.
   441→ *   • mouse leaves the     restores the "C" logo mark AND its bg-primary
   442→ *     entire rail            box (onMouseLeave on the <Sidebar> root).
   443→ *   • click ............. calls toggleSidebar (existing functionality,
   444→ *                         unchanged).
   445→ *
   446→ * Each rail item keeps its OWN Radix Tooltip label (Dashboard, Articles,
   447→ * Profile, etc.) — the lifted `hovered` prop ONLY drives the logo cell's
   448→ * C↔PanelLeftOpen swap; it does NOT touch any item's tooltip. The logo's
   449→ * own "Expand" Tooltip still fires ONLY when the pointer is directly over
   450→ * the logo button (the Tooltip trigger is the button itself, independent
   451→ * of the `hovered` prop).
   452→ *
   453→ * Background is removed ONLY from the temporary Expand-icon state; the
   454→ * normal logo (at rest) keeps its bg-primary background exactly as-is.
   455→ *
   456→ * Implementation: React state lifted to AppSidebar (railHovered +
   457→ * onMouseEnter/onMouseLeave on <Sidebar>) — NOT CSS :hover / group-hover
   458→ * variants. This is deliberate: Tailwind v4 wraps its hover:/group-hover:
   459→ * utilities inside @media (hover: hover), so on any browser that reports
   460→ * (hover: none) (headless, touch, some preview iframes) those utilities
   461→ * NEVER activate and the swap + bg-transparent would silently fail. React
   462→ * mouse events fire on ANY pointer input regardless of the hover media
   463→ * query, so the behavior is identical in every environment. The
   464→ * conditional className + conditional render swap the bg/color/glyph
   465→ * atomically per state. Works in both Light and Dark.
   466→ */
   467→function CollapsedLogoButton({ hovered }: { hovered: boolean }) {
   468→  const { toggleSidebar } = useSidebar();
   469→  return (
   470→    /* Hover-only "Expand" tooltip: instant (provider delayDuration=0),
   471→       appears on mouse-enter, disappears on mouse-leave. Radix closes it
   472→       on any pointer exit — disableHoverableContent guarantees it can
   473→       never linger or trap the pointer. The Tooltip trigger is the logo
   474→       button itself, so the "Expand" TEXT fires ONLY on direct logo hover
   475→       (NOT on rail-level hover of other items) — the C↔PanelLeftOpen ICON
   476→       swap, by contrast, is driven by the lifted `hovered` prop and fires
   477→       for ANY rail item. */
   478→    <Tooltip disableHoverableContent>
   479→      <TooltipTrigger asChild>
   480→        <button
   481→          type="button"
   482→          onClick={toggleSidebar}
   483→          aria-label="Expand sidebar"
   484→          className={cn(
   485→            'flex h-8 w-8 shrink-0 cursor-pointer items-center justify-center rounded-lg font-bold text-sm outline-none transition-opacity hover:opacity-90 focus-visible:ring-2 focus-visible:ring-ring select-none',
   486→            // NORMAL LOGO STATE (at rest): bg-primary black box +
   487→            // text-primary-foreground white "C" — background UNCHANGED,
   488→            // same as the expanded LogoMark.
   489→            // EXPAND-ICON STATE (on hover): bg-transparent (NO background)
   490→            // + text-muted-foreground gray icon — ONLY this temporary
   491→            // state loses the background. The icon stays visible (gray
   492→            // on transparent/page bg) in both Light and Dark mode.
   493→            hovered
   494→              ? 'bg-transparent text-muted-foreground'
   495→              : 'bg-primary text-primary-foreground',
   496→          )}
   497→        >
   498→          {/* Conditional render: "C" logo at rest, PanelLeftOpen icon on
   499→              hover. h-4 w-4 matches the CollapseToggle's PanelLeftClose
   500→              icon size; the icon is the visual opposite of the collapse
   501→              icon so the collapsed logo reads as a clear "expand"
   502→              affordance when the pointer is over it. */}
   503→          {hovered ? (
   504→            <PanelLeftOpen className="h-4 w-4" />
   505→          ) : (
   506→            <span>C</span>
   507→          )}
   508→          <span className="sr-only">Expand sidebar</span>
   509→        </button>
   510→      </TooltipTrigger>
   511→      {/* Collapsed-rail tooltip — uses the SHARED COLLAPSED_TOOLTIP_PROPS
   512→          constant so the "Expand" label has IDENTICAL positioning (side,
   513→          align, sideOffset, collisionPadding) to every other collapsed-
   514→          rail tooltip (ThemeToggle, SimpleNavItem, ExpandableNavItem,
   515→          CollapsedParentNavItem). Portal-based rendering means the bubble
   516→          floats outside the sidebar DOM at z-50 — never clipped by the
   517→          rail's overflow-hidden. */}
   518→      <TooltipContent {...COLLAPSED_TOOLTIP_PROPS}>Expand</TooltipContent>
   519→    </Tooltip>
   520→  );
   521→}

   605→}) {
   606→  const [floatOpen, setFloatOpen] = useState(false);
   607→  const mod = hrefToModule(item.href);
   608→  const isActive = currentModule === mod;
   609→
   610→  const handleChildNavigate = (
   611→    e: React.MouseEvent<HTMLAnchorElement>,
   612→    child: NavItem,
   613→  ) => {
   614→    e.preventDefault();
   615→    e.stopPropagation();
   616→    setFloatOpen(false); // close the floating submenu after navigation…
   617→    // …but do NOT touch sidebar expansion state (rule #7).
   618→    const hash = child.href.replace(/^#/, '');
   619→    const parts = hash.split('/');
   620→    const childSubPage = parts[1] || null;
   621→    useNavigationStore.getState().navigate(parts[0], null, childSubPage);
   622→  };
   623→
   624→  return (
   625→    <SidebarMenuItem>
   626→      <Popover open={floatOpen} onOpenChange={setFloatOpen}>
   627→        <PopoverTrigger asChild>
   628→          <SidebarMenuButton
   629→            isActive={isActive || floatOpen}
   630→            aria-expanded={floatOpen}
   631→            aria-haspopup="menu"
   632→            // Collapsed-rail tooltip — uses the SHARED COLLAPSED_TOOLTIP_PROPS
   633→            // so the parent item's label has IDENTICAL positioning (side,
   634→            // align, sideOffset, collisionPadding) to every other collapsed-
   635→            // rail tooltip (CollapsedLogoButton, ThemeToggle, SimpleNavItem,
   636→            // ExpandableNavItem). Suppressed when the popover is open so the
   637→            // floating submenu never visually conflicts with a lingering
   638→            // label bubble. Portal-based rendering means the bubble floats
   639→            // outside the sidebar DOM at z-50 — never clipped by the rail's
   640→            // overflow-hidden.
   641→            tooltip={
   642→              floatOpen
   643→                ? undefined
   644→                : { ...COLLAPSED_TOOLTIP_PROPS, children: item.label }
   645→            }
   646→            onClick={(e: React.MouseEvent) => {
   647→              e.preventDefault();
   648→              setFloatOpen((o) => !o);
   649→            }}
   650→          >
   651→            <NavIcon name={item.icon} />
   652→            <span>{item.label}</span>
   653→          </SidebarMenuButton>
   654→        </PopoverTrigger>
   655→        {/* Portal → document.body: immune to sidebar overflow clipping.
   656→            Radix handles Esc + outside-click closing and re-anchors at any
   657→            viewport size/zoom via avoidCollisions. */}
   658→        <PopoverContent
   659→          side="right"
   660→          align="start"
   661→          sideOffset={6}
   662→          collisionPadding={12}
   663→          className="w-56 rounded-md border bg-popover p-1.5 shadow-md"
   664→          onEscapeKeyDown={() => setFloatOpen(false)}
   665→        >
   666→          <ul
   667→            role="menu"
   668→            aria-label={`${item.label} submenu`}
   669→            className="flex min-w-0 list-none flex-col gap-0.5 p-0 m-0"
   670→          >
   671→            {item.children!.map((child) => {
   672→              const hash = child.href.replace(/^#/, '');
   673→              const parts = hash.split('/');
   674→              const childSubPage = parts[1] || null;
   675→              const isChildActive =
   676→                currentModule === parts[0] &&
   677→                (!childSubPage || currentSubPage === childSubPage);
   678→              return (
   679→                <SidebarMenuSubItem key={child.label} className="list-none p-0">
   680→                  <SidebarMenuSubButton asChild isActive={isChildActive}>
   681→                    <a
   682→                      href={child.href}
   683→                      role="menuitem"
   684→                      onClick={(e) => handleChildNavigate(e, child)}
   685→                    >
   686→                      <NavIcon name={child.icon} />
   687→                      <span>{child.label}</span>
   688→                    </a>
   689→                  </SidebarMenuSubButton>
   690→                </SidebarMenuSubItem>
   691→              );
   692→            })}
   693→          </ul>
   694→        </PopoverContent>
   695→      </Popover>
   696→    </SidebarMenuItem>
   697→  );
   698→}

   700→function ExpandableNavItem({
   701→  item,
   702→  currentModule,
   703→  currentSubPage,
   704→  isExpanded,
   705→  onToggle,
   706→}: {
   707→  item: NavItem;
   708→  currentModule: string;
   709→  currentSubPage: string | null;
   710→  isExpanded: boolean;
   711→  onToggle: () => void;
   712→}) {
   713→  const mod = hrefToModule(item.href);
   714→  const isActive = currentModule === mod;
   715→  const sectionId = `submenu-${item.label.toLowerCase().replace(/\s+/g, '-')}`;
   716→
   717→  return (
   718→    <SidebarMenuItem>
   719→      <SidebarMenuButton
   720→        // Expandable parent item rendered in EXPANDED state. SidebarMenuButton
   721→        // hides its tooltip whenever state!=="collapsed" (see ui/sidebar.tsx),
   722→        // so this label never shows while expanded — but passing the SAME
   723→        // COLLAPSED_TOOLTIP_PROPS object as every other nav-item renderer
   724→        // keeps the prop consistent across the codebase. If the sidebar
   725→        // toggles between expanded and collapsed while this row stays
   726→        // mounted, the tooltip that appears is identical in positioning to
   727→        // SimpleNavItem / CollapsedParentNavItem / CollapsedLogoButton.
   728→        tooltip={{ ...COLLAPSED_TOOLTIP_PROPS, children: item.label }}
   729→        isActive={isActive}
   730→        onClick={(e: React.MouseEvent) => {
   731→          e.preventDefault();
   732→          onToggle();
   733→        }}
   734→        aria-expanded={isExpanded}
   735→        aria-controls={sectionId}
   736→      >
   737→        <NavIcon name={item.icon} />
   738→        <span>{item.label}</span>
   739→        <ChevronRight
   740→          className={cn(
   741→            'ml-auto h-4 w-4 shrink-0 transition-transform duration-200',
   742→            isExpanded && 'rotate-90',
   743→          )}
   744→        />
   745→      </SidebarMenuButton>
   746→
   747→      {item.badge != null && (
   748→        <SidebarMenuBadge>
   749→          {item.badge}
   750→        </SidebarMenuBadge>
   751→      )}
   752→
   753→      <AccordionSubmenu isOpen={isExpanded} sectionLabel={item.label}>
   754→        {item.children!.map((child) => {
   755→          const hash = child.href.replace(/^#/, '');
   756→          const parts = hash.split('/');
   757→          const childMod = parts[0];
   758→          const childSubPage = parts[1] || null;
   759→          const isChildActive =
   760→            currentModule === childMod &&
   761→            (!childSubPage || currentSubPage === childSubPage);
   762→
   763→          return (
   764→            <SidebarMenuSubItem key={child.label}>
   765→              <SidebarMenuSubButton
   766→                asChild
   767→                isActive={isChildActive}
   768→              >
   769→                <a
   770→                  href={child.href}
   771→                  onClick={(e) => {
   772→                    e.preventDefault();
   773→                    e.stopPropagation();
   774→                    useNavigationStore
   775→                      .getState()
   776→                      .navigate(childMod, null, childSubPage);
   777→                  }}
   778→                >
   779→                  <NavIcon name={child.icon} />
   780→                  <span>{child.label}</span>
   781→                </a>
   782→              </SidebarMenuSubButton>
   783→            </SidebarMenuSubItem>
   784→          );
   785→        })}
   786→      </AccordionSubmenu>
   787→    </SidebarMenuItem>
   788→  );
   789→}

   840→function NavGroupSection({
   841→  group,
   842→  currentModule,
   843→  currentSubPage,
   844→  openSection,
   845→  onToggleSection,
   846→}: {
   847→  group: NavGroup;
   848→  currentModule: string;
   849→  currentSubPage: string | null;

   861→
   862→            if (hasChildren && isCollapsedRail) {
   863→              /* Collapsed rail → parents with children become floating
   864→                 popovers anchored to their own icon. The rail NEVER
   865→                 auto-expands (#4/#5/#6). */
   866→              return (
   867→                <CollapsedParentNavItem
   868→                  key={item.label}
   869→                  item={item}
   870→                  currentModule={currentModule}
   871→                  currentSubPage={currentSubPage}
   872→                />
   873→              );
   874→            }
   875→
   876→            if (hasChildren) {
   877→              return (
   878→                <ExpandableNavItem
   879→                  key={item.label}
   880→                  item={item}
   881→                  currentModule={currentModule}
   882→                  currentSubPage={currentSubPage}

   887→            }
   888→
   889→            return (
   890→              <SimpleNavItem
   891→                key={item.label}
   892→                item={item}
   893→                currentModule={currentModule}
   894→              />
   895→            );
   896→          })}
   897→        </SidebarMenu>
   898→      </SidebarGroupContent>
   899→    </SidebarGroup>
   900→  );
   901→}
