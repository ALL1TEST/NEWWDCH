   150→// -------------------- Collapsed-Rail Tooltip Positioning -------------------
   151→/*
   152→ * SINGLE source of truth for the positioning of every tooltip/label that
   153→ * appears when the sidebar is collapsed into the 48px icon rail.
   154→ *
   155→ *   side="right"         → opens to the RIGHT of the 48px collapsed rail,
   156→ *                          fully inside the main viewport (never overlaps
   157→ *                          the rail, never opens to the left/outside).
   158→ *   align="center"       → vertically centers the bubble on the icon's
   159→ *                          32×32 hover target so labels line up across
   160→ *                          every row of the rail (logo, theme, nav items,
   161→ *                          parent items with submenu).
   162→ *   sideOffset=8         → ~8px visible gap from the trigger button's right
   163→ *                          edge (button right edge ~x=40 inside the 48px
   164→ *                          rail, tooltip left edge ~x=48 = the rail's
   165→ *                          right edge — flush with the RAIL, NOT with the
   166→ *                          icon, so the bubble never touches the glyph).
   167→ *   collisionPadding=12 → 12px viewport-edge collision padding so the
   168→ *                          bubble can never be clipped at any viewport
   169→ *                          edge or scroll past the visible area.
   170→ *
   171→ * Portal-based rendering (Radix TooltipPrimitive.Portal → document.body,
   172→ * wired inside src/components/ui/tooltip.tsx) means every tooltip is
   173→ * rendered OUTSIDE the sidebar DOM tree — so the sidebar's
   174→ * overflow-hidden / overflow-x-hidden on the rail container can NEVER
   175→ * clip or hide the bubble. The tooltip floats above every other
   176→ * element at z-50.
   177→ *
   178→ * Used by:
   179→ *   • CollapsedLogoButton         (header "Expand" tooltip)
   180→ *   • SimpleNavItem              (leaf nav items: Dashboard, Articles,
   181→ *                                  Calendar, Media, Users, Comments,
   182→ *                                  Newsletter, SEO, AI, Automation)
   183→ *   • ExpandableNavItem          (parent nav items in EXPANDED state —
   184→ *                                  tooltip is hidden by SidebarMenuButton
   185→ *                                  when state!=="collapsed", but the
   186→ *                                  positioning object is still applied
   187→ *                                  so the prop stays consistent across
   188→ *                                  every nav item renderer)
   189→ *   • CollapsedParentNavItem     (parent nav items in COLLAPSED state —
   190→ *                                  label tooltip shows when the floating
   191→ *                                  popover is CLOSED; when the popover is
   192→ *                                  open, no tooltip shows to avoid visual
   193→ *                                  conflict)
   194→ *
   195→ * The ThemeToggle component (src/components/layout/theme-toggle.tsx) lives
   196→ * in a separate file and inlines the SAME four values — see the comment
   197→ * there for the rationale.

   404→/**
   405→ * Collapsed-rail logo cell. Since the extra collapse-toggle icon was removed
   406→ * from below the logo, THIS cell is the explicit expand control:
   407→ *   • identical 32×32 grid geometry as every other rail icon
   408→ *   • click toggles the sidebar open (SidebarRail edge strip still works)
   409→ * No new icon is introduced and no empty clickable ghost remains.
   410→ *
   411→ * HOVER BEHAVIOR (collapsed state only — this button is rendered ONLY
   412→ * inside the collapsed-rail cluster, so the expanded LogoMark is untouched):
   413→ * The `hovered` prop is LIFTED to the rail level (see AppSidebar's
   414→ * `railHovered` state + the onMouseEnter/onMouseLeave handlers on the
   415→ * <Sidebar> root). This means the swap fires when the mouse is ANYWHERE
   416→ * over the collapsed 48px rail — NOT only when the pointer is directly
   417→ * over the logo cell:
   418→ *   • rail at rest ........  shows the "C" logo mark on its bg-primary
   419→ *                            black box (NORMAL LOGO — background UNCHANGED,
   420→ *                            same as the expanded LogoMark).
   421→ *   • mouse over ANY rail   the "C" is replaced by a PanelLeftOpen icon
   422→ *     item (Dashboard,       AND the button's background becomes
   423→ *     Articles, …, Profile,  transparent so ONLY the Expand icon is
   424→ *     Theme, Settings, …)    visible with no box behind it. The icon
   425→ *                            color switches to muted-foreground (gray)
   426→ *                            so it stays visible against the
   427→ *                            transparent/page bg in both Light and Dark.
   428→ *   • mouse leaves the     restores the "C" logo mark AND its bg-primary
   429→ *     entire rail            box (onMouseLeave on the <Sidebar> root).
   430→ *   • click ............. calls toggleSidebar (existing functionality,
   431→ *                         unchanged).
   432→ *
   433→ * Each rail item keeps its OWN Radix Tooltip label (Dashboard, Articles,
   434→ * Profile, etc.) — the lifted `hovered` prop ONLY drives the logo cell's
   435→ * C↔PanelLeftOpen swap; it does NOT touch any item's tooltip. The logo's
   436→ * own "Expand" Tooltip still fires ONLY when the pointer is directly over
   437→ * the logo button (the Tooltip trigger is the button itself, independent
   438→ * of the `hovered` prop).
   439→ *
   440→ * Background is removed ONLY from the temporary Expand-icon state; the
   441→ * normal logo (at rest) keeps its bg-primary background exactly as-is.
   442→ *
   443→ * Implementation: React state lifted to AppSidebar (railHovered +
   444→ * onMouseEnter/onMouseLeave on <Sidebar>) — NOT CSS :hover / group-hover
   445→ * variants. This is deliberate: Tailwind v4 wraps its hover:/group-hover:
   446→ * utilities inside @media (hover: hover), so on any browser that reports
   447→ * (hover: none) (headless, touch, some preview iframes) those utilities
   448→ * NEVER activate and the swap + bg-transparent would silently fail. React
   449→ * mouse events fire on ANY pointer input regardless of the hover media
   450→ * query, so the behavior is identical in every environment. The
   451→ * conditional className + conditional render swap the bg/color/glyph
   452→ * atomically per state. Works in both Light and Dark.
   453→ */
   454→function CollapsedLogoButton({ hovered }: { hovered: boolean }) {
   455→  const { toggleSidebar } = useSidebar();
   456→  return (
   457→    /* Hover-only "Expand" tooltip: instant (provider delayDuration=0),
   458→       appears on mouse-enter, disappears on mouse-leave. Radix closes it
   459→       on any pointer exit — disableHoverableContent guarantees it can
   460→       never linger or trap the pointer. The Tooltip trigger is the logo
   461→       button itself, so the "Expand" TEXT fires ONLY on direct logo hover
   462→       (NOT on rail-level hover of other items) — the C↔PanelLeftOpen ICON
   463→       swap, by contrast, is driven by the lifted `hovered` prop and fires
   464→       for ANY rail item. */
   465→    <Tooltip disableHoverableContent>
   466→      <TooltipTrigger asChild>
   467→        <button
   468→          type="button"
   469→          onClick={toggleSidebar}
   470→          aria-label="Expand sidebar"
   471→          className={cn(
   472→            'flex h-8 w-8 shrink-0 cursor-pointer items-center justify-center rounded-lg font-bold text-sm outline-none transition-opacity hover:opacity-90 focus-visible:ring-2 focus-visible:ring-ring select-none',
   473→            // NORMAL LOGO STATE (at rest): bg-primary black box +
   474→            // text-primary-foreground white "C" — background UNCHANGED,
   475→            // same as the expanded LogoMark.
   476→            // EXPAND-ICON STATE (on hover): bg-transparent (NO background)
   477→            // + text-muted-foreground gray icon — ONLY this temporary
   478→            // state loses the background. The icon stays visible (gray
   479→            // on transparent/page bg) in both Light and Dark mode.
   480→            hovered
   481→              ? 'bg-transparent text-muted-foreground'
   482→              : 'bg-primary text-primary-foreground',
   483→          )}
   484→        >
   485→          {/* Conditional render: "C" logo at rest, PanelLeftOpen icon on
   486→              hover. h-4 w-4 matches the CollapseToggle's PanelLeftClose
   487→              icon size; the icon is the visual opposite of the collapse
   488→              icon so the collapsed logo reads as a clear "expand"
   489→              affordance when the pointer is over it. */}
   490→          {hovered ? (
   491→            <PanelLeftOpen className="h-4 w-4" />
   492→          ) : (
   493→            <span>C</span>
   494→          )}
   495→          <span className="sr-only">Expand sidebar</span>
   496→        </button>
   497→      </TooltipTrigger>
   498→      {/* Collapsed-rail tooltip — uses the SHARED COLLAPSED_TOOLTIP_PROPS
   499→          constant so the "Expand" label has IDENTICAL positioning (side,
   500→          align, sideOffset, collisionPadding) to every other collapsed-
   501→          rail tooltip (ThemeToggle, SimpleNavItem, ExpandableNavItem,
   502→          CollapsedParentNavItem). Portal-based rendering means the bubble
   503→          floats outside the sidebar DOM at z-50 — never clipped by the
   504→          rail's overflow-hidden. */}
   505→      <TooltipContent {...COLLAPSED_TOOLTIP_PROPS}>Expand</TooltipContent>
   506→    </Tooltip>
   507→  );
   508→}

   601→    e.preventDefault();

   603→    setFloatOpen(false); // close the floating submenu after navigation…
   604→    // …but do NOT touch sidebar expansion state (rule #7).
   605→    const hash = child.href.replace(/^#/, '');
   606→    const parts = hash.split('/');
   607→    const childSubPage = parts[1] || null;
   608→    useNavigationStore.getState().navigate(parts[0], null, childSubPage);
   609→  };
   610→
   611→  return (
   612→    <SidebarMenuItem>
   613→      <Popover open={floatOpen} onOpenChange={setFloatOpen}>
   614→        <PopoverTrigger asChild>
   615→          <SidebarMenuButton
   616→            isActive={isActive || floatOpen}
   617→            aria-expanded={floatOpen}
   618→            aria-haspopup="menu"
   619→            // Collapsed-rail tooltip — uses the SHARED COLLAPSED_TOOLTIP_PROPS
   620→            // so the parent item's label has IDENTICAL positioning (side,
   621→            // align, sideOffset, collisionPadding) to every other collapsed-
   622→            // rail tooltip (CollapsedLogoButton, ThemeToggle, SimpleNavItem,
   623→            // ExpandableNavItem). Suppressed when the popover is open so the
   624→            // floating submenu never visually conflicts with a lingering
   625→            // label bubble. Portal-based rendering means the bubble floats
   626→            // outside the sidebar DOM at z-50 — never clipped by the rail's
   627→            // overflow-hidden.
   628→            tooltip={
   629→              floatOpen
   630→                ? undefined
   631→                : { ...COLLAPSED_TOOLTIP_PROPS, children: item.label }
   632→            }
   633→            onClick={(e: React.MouseEvent) => {
   634→              e.preventDefault();
   635→              setFloatOpen((o) => !o);
   636→            }}
   637→          >
   638→            <NavIcon name={item.icon} />
   639→            <span>{item.label}</span>
   640→          </SidebarMenuButton>
   641→        </PopoverTrigger>
   642→        {/* Portal → document.body: immune to sidebar overflow clipping.
   643→            Radix handles Esc + outside-click closing and re-anchors at any
   644→            viewport size/zoom via avoidCollisions. */}
   645→        <PopoverContent
   646→          side="right"
   647→          align="start"
   648→          sideOffset={6}
   649→          collisionPadding={12}
   650→          className="w-56 rounded-md border bg-popover p-1.5 shadow-md"
   651→          onEscapeKeyDown={() => setFloatOpen(false)}
   652→        >
   653→          <ul
   654→            role="menu"
   655→            aria-label={`${item.label} submenu`}
   656→            className="flex min-w-0 list-none flex-col gap-0.5 p-0 m-0"
   657→          >
   658→            {item.children!.map((child) => {
   659→              const hash = child.href.replace(/^#/, '');
   660→              const parts = hash.split('/');
   661→              const childSubPage = parts[1] || null;
   662→              const isChildActive =
   663→                currentModule === parts[0] &&
   664→                (!childSubPage || currentSubPage === childSubPage);
   665→              return (
   666→                <SidebarMenuSubItem key={child.label} className="list-none p-0">

   679→            })}
   680→          </ul>

   687→function ExpandableNavItem({
   688→  item,
   689→  currentModule,
   690→  currentSubPage,
   691→  isExpanded,
   692→  onToggle,
   693→}: {
   694→  item: NavItem;
   695→  currentModule: string;
   696→  currentSubPage: string | null;
   697→  isExpanded: boolean;
   698→  onToggle: () => void;
   699→}) {
   700→  const mod = hrefToModule(item.href);
   701→  const isActive = currentModule === mod;

   705→    <SidebarMenuItem>

   708→        // hides its tooltip whenever state!=="collapsed" (see ui/sidebar.tsx),
   709→        // so this label never shows while expanded — but passing the SAME
   710→        // COLLAPSED_TOOLTIP_PROPS object as every other nav-item renderer
   711→        // keeps the prop consistent across the codebase. If the sidebar
   712→        // toggles between expanded and collapsed while this row stays
   713→        // mounted, the tooltip that appears is identical in positioning to
   714→        // SimpleNavItem / CollapsedParentNavItem / CollapsedLogoButton.

   718→          e.preventDefault();

   720→        }}

   722→        aria-controls={sectionId}

   724→        <NavIcon name={item.icon} />

   726→        <ChevronRight
   727→          className={cn(
   728→            'ml-auto h-4 w-4 shrink-0 transition-transform duration-200',
   729→            isExpanded && 'rotate-90',
   730→          )}
   731→        />

   734→      {item.badge != null && (
   735→        <SidebarMenuBadge>
   736→          {item.badge}
   737→        </SidebarMenuBadge>
   738→      )}
   739→
   740→      <AccordionSubmenu isOpen={isExpanded} sectionLabel={item.label}>
   741→        {item.children!.map((child) => {
   742→          const hash = child.href.replace(/^#/, '');
   743→          const parts = hash.split('/');
   744→          const childMod = parts[0];
   745→          const childSubPage = parts[1] || null;
   746→          const isChildActive =
   747→            currentModule === childMod &&
   748→            (!childSubPage || currentSubPage === childSubPage);
   749→
   750→          return (
   751→            <SidebarMenuSubItem key={child.label}>
   752→              <SidebarMenuSubButton
   753→                asChild
   754→                isActive={isChildActive}
   755→              >
   756→                <a
   757→                  href={child.href}
   758→                  onClick={(e) => {
   759→                    e.preventDefault();
   760→                    e.stopPropagation();
   761→                    useNavigationStore
   762→                      .getState()
   763→                      .navigate(childMod, null, childSubPage);
   764→                  }}
   765→                >
   766→                  <NavIcon name={child.icon} />
   767→                  <span>{child.label}</span>
   768→                </a>
   769→              </SidebarMenuSubButton>
   770→            </SidebarMenuSubItem>
   771→          );
   772→        })}
   773→      </AccordionSubmenu>
   774→    </SidebarMenuItem>
   775→  );

   827→function NavGroupSection({
   828→  group,
   829→  currentModule,

   831→  openSection,
   832→  onToggleSection,

   834→  group: NavGroup;
