   465→// ============================================================
   466→// CHANGE EMAIL DIALOG
   467→// ------------------------------------------------------------
   468→// Modal with:
   469→//   • Current email (read-only display)
   470→//   • New email input (validated client + server-side)
   471→//   • Current password input (server-verified)
   472→// Server uniqueness check at /api/auth/change-email.
   473→// On success: toast + update auth store + close.
   474→// ============================================================
   475→
   476→function ChangeEmailDialog({
   477→  open,
   478→  onOpenChange,
   479→  currentEmail,
   480→  onEmailChanged,
   481→}: {
   482→  open: boolean;
   483→  onOpenChange: (v: boolean) => void;
   484→  currentEmail: string;
   485→  onEmailChanged: (newEmail: string) => void;
   486→}) {
   487→  const [newEmail, setNewEmail] = useState('');
   488→  const [currentPassword, setCurrentPassword] = useState('');
   489→  const [showPw, setShowPw] = useState(false);
   490→  const [isSaving, setIsSaving] = useState(false);
   491→  const [localError, setLocalError] = useState<string | null>(null);
   492→
   493→  useEffect(() => {
   494→    if (!open) {
   495→      setNewEmail('');
   496→      setCurrentPassword('');
   497→      setShowPw(false);
   498→      setLocalError(null);
   499→    }
   500→  }, [open]);
   501→
   502→  const emailValid = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(newEmail.trim());
   503→  const isDifferent = newEmail.trim().toLowerCase() !== currentEmail.toLowerCase();
   504→
   505→  const handleSubmit = async () => {
   506→    setLocalError(null);
   507→    if (!newEmail.trim()) {
   508→      setLocalError('Please enter a new email address.');
   509→      return;
   510→    }
   511→    if (!emailValid) {
   512→      setLocalError('Please enter a valid email address.');
   513→      return;
   514→    }
   515→    if (!isDifferent) {
   516→      setLocalError('New email must be different from your current email.');
   517→      return;
   518→    }
   519→    if (!currentPassword) {
   520→      setLocalError('Please enter your current password to confirm.');
   521→      return;
   522→    }
   523→
   524→    setIsSaving(true);
   525→    try {
   526→      const res = await postApi<{ email: string }>('/api/auth/change-email', {
   527→        currentPassword,
   528→        newEmail: newEmail.trim(),
   529→      });
   530→      onEmailChanged(res.email);
   531→      toast.success('Email address updated successfully.');
   532→      onOpenChange(false);
   533→    } catch (err) {
   534→      const msg = err instanceof Error ? err.message : 'Failed to change email.';
   535→      setLocalError(msg);
   536→    } finally {
   537→      setIsSaving(false);
   538→    }
   539→  };
   540→
   541→  return (
   542→    <Dialog open={open} onOpenChange={(v) => { if (!isSaving) onOpenChange(v); }}>
   543→      <DialogContent className="sm:max-w-md">
   544→        <DialogHeader>
   545→          <DialogTitle className="flex items-center gap-2">
   546→            <Mail className="h-5 w-5" />
   547→            Change Email Address
   548→          </DialogTitle>
   549→          <DialogDescription>
   550→            Enter your new email and current password to confirm the change.
   551→          </DialogDescription>
   552→        </DialogHeader>
   553→        <div className="py-2 space-y-4">
   554→          <div className="grid gap-2">
   555→            <Label className="text-xs text-muted-foreground">Current Email</Label>
   556→            <div className="rounded-md border bg-muted/50 px-3 py-2 text-sm text-muted-foreground">
   557→              {currentEmail}
   558→            </div>
   559→          </div>
   560→          <div className="grid gap-2">
   561→            <Label htmlFor="new-email" className="text-xs">New Email Address</Label>
   562→            <div className="relative">
   563→              <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
