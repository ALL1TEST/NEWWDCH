    19→import {

    30→  Plug,

    34→

    39→  CardTitle,
    40→  CardDescription,
    41→} from '@/components/ui/card';
    42→import { Button } from '@/components/ui/button';
    43→import { Input } from '@/components/ui/input';
    44→import { Label } from '@/components/ui/label';
    45→import { Switch } from '@/components/ui/switch';
    46→import { Separator } from '@/components/ui/separator';
    47→import {
    48→  Select,

    51→  SelectTrigger,
    52→  SelectValue,
    53→} from '@/components/ui/select';

    56→
    57→// -------------------- Types --------------------
    58→
    59→type Encryption = 'none' | 'SSL' | 'STARTTLS';
    60→
    61→interface SmtpSettingsResponse {
    62→  id: string | null;
    63→  provider: string;
    64→  host: string;
    65→  port: number;
    66→  encryption: Encryption;
    67→  username: string;
    68→  password: string;
    69→  fromName: string;
    70→  fromEmail: string;
    71→  replyTo: string;
    72→  timeout: number;
    73→  isActive: boolean;
    74→  isDefault: boolean;
    75→}
    76→
    77→interface TestResponse {
    78→  success: boolean;
    79→  message: string;
    80→  messageId?: string;
    81→}
    82→
    83→interface ApiErrorPayload {
    84→  error: { code: string; message: string };
    85→  meta: { requestId: string };
    86→}
    87→
    88→// -------------------- Constants --------------------
    89→
    90→const PASSWORD_PLACEHOLDER = '••••••••';
    91→
    92→const ENCRYPTION_OPTIONS: {
    93→  value: Encryption;
    94→  label: string;
    95→  hint: string;
    96→  port: number;
    97→}[] = [
    98→  { value: 'STARTTLS', label: 'STARTTLS', hint: 'Recommended · port 587', port: 587 },
    99→  { value: 'SSL', label: 'SSL/TLS', hint: 'Implicit TLS · port 465', port: 465 },
   100→  { value: 'none', label: 'None', hint: 'No encryption · port 25', port: 25 },
   101→];
   102→
   103→// -------------------- Page --------------------
   104→
   105→export function SmtpSettingsPage() {
   106→  const queryClient = useQueryClient();
   107→
   108→  // ---------- Load saved settings ----------
   109→  const { data, isLoading } = useQuery<SmtpSettingsResponse | null>({
   110→    queryKey: ['settings', 'smtp'],
   111→    queryFn: () => getApi<SmtpSettingsResponse | null>('/api/settings/smtp'),
   112→    staleTime: 10_000,
   113→  });
   114→
   115→  // ---------- Draft state (local overrides on top of saved values) ----------
   116→  // We never sync useEffect→state. Instead we derive "current display value"
   117→  // from `saved + draft override`. Dirty = draft is non-empty.
   118→  const [draft, setDraft] = useState<Partial<SmtpSettingsResponse>>({});
   119→  const [showPassword, setShowPassword] = useState(false);
   120→
   121→  // Password UI state — separate so we can support the masked-placeholder pattern.
   122→  const [passwordInput, setPasswordInput] = useState<string>(PASSWORD_PLACEHOLDER);
   123→
   124→  // ---------- Derived values ----------
   125→  const saved = data ?? null;
   126→  const hasSavedPassword = (saved?.password ?? '').length > 0;
   127→
   128→  const current: SmtpSettingsResponse = {
   129→    id: saved?.id ?? null,
   130→    provider: saved?.provider ?? 'SMTP',
   131→    host: draft.host ?? saved?.host ?? '',
   132→    port: draft.port ?? saved?.port ?? 587,
   133→    encryption: draft.encryption ?? saved?.encryption ?? 'STARTTLS',
   134→    username: draft.username ?? saved?.username ?? '',
   135→    password: hasSavedPassword ? PASSWORD_PLACEHOLDER : '',
   136→    fromName: draft.fromName ?? saved?.fromName ?? '',
   137→    fromEmail: draft.fromEmail ?? saved?.fromEmail ?? '',
   138→    replyTo: draft.replyTo ?? saved?.replyTo ?? '',
   139→    timeout: draft.timeout ?? saved?.timeout ?? 10,
   140→    isActive: draft.isActive ?? saved?.isActive ?? true,
   141→    isDefault: saved?.isDefault ?? true,
   142→  };
   143→
   144→  const isDirty =
   145→    Object.keys(draft).length > 0 || passwordInput !== PASSWORD_PLACEHOLDER;
   146→
   147→  // ---------- Setters ----------
   148→  const update = <K extends keyof SmtpSettingsResponse>(
   149→    key: K,
   150→    value: SmtpSettingsResponse[K],
   151→  ) => {
   152→    setDraft((prev) => ({ ...prev, [key]: value }));
   153→  };
   154→
   155→  // When encryption changes, also auto-suggest the matching default port.
   156→  const handleEncryptionChange = (value: Encryption) => {
   157→    const option = ENCRYPTION_OPTIONS.find((o) => o.value === value);
   158→    setDraft((prev) => ({
   159→      ...prev,
   160→      encryption: value,
   161→      port: option ? option.port : prev.port,
   162→    }));
   163→  };
   164→
   165→  // ---------- Save mutation ----------
   166→  const saveMutation = useMutation({
   167→    mutationFn: () =>
   168→      putApi<SmtpSettingsResponse>('/api/settings/smtp', {
   169→        provider: current.provider,
   170→        host: current.host,
   171→        port: current.port,
   172→        encryption: current.encryption,
   173→        username: current.username,
   174→        // Send the masked placeholder when password is unchanged,
   175→        // OR send the new password when user typed something else.
   176→        password: passwordInput,
   177→        fromName: current.fromName,
   178→        fromEmail: current.fromEmail,
   179→        replyTo: current.replyTo,
   180→        timeout: current.timeout,
   181→        isActive: current.isActive,
   182→      }),
   183→    onSuccess: () => {
   184→      queryClient.invalidateQueries({ queryKey: ['settings', 'smtp'] });
   185→      setDraft({});
   186→      setPasswordInput(PASSWORD_PLACEHOLDER);
   187→      toast.success('SMTP settings saved successfully');
   188→    },
   189→    onError: (err: unknown) => {
   190→      const message =
   191→        (err as ApiErrorPayload)?.error?.message ??
   192→        (err instanceof Error ? err.message : 'Failed to save SMTP settings');
   193→      toast.error(message);
   194→    },
   195→  });
   196→
   197→  // ---------- Test connection mutation ----------
   198→  const [testResult, setTestResult] = useState<
   199→    { ok: boolean; message: string } | null
   200→  >(null);
   201→
   202→  const testMutation = useMutation({
   203→    mutationFn: () => {
   204→      // If dirty, send current draft; otherwise let backend use saved settings.
   205→      const payload = isDirty
   206→        ? {
   207→            settings: {
   208→              provider: current.provider,
   209→              host: current.host,
   210→              port: current.port,
   211→              encryption: current.encryption,
   212→              username: current.username,
   213→              password: passwordInput,
   214→              fromName: current.fromName,
   215→              fromEmail: current.fromEmail,
   216→              replyTo: current.replyTo,
   217→              timeout: current.timeout,
   218→              isActive: current.isActive,
   219→            },
   220→          }
   221→        : {};
   222→      return postApi<TestResponse>('/api/settings/smtp/test', payload);
   223→    },
   224→    onSuccess: (data) => {
   225→      setTestResult({ ok: true, message: data.message });
   226→      toast.success(data.message);
   227→    },
   228→    onError: (err: unknown) => {
   229→      const message =
   230→        (err as ApiErrorPayload)?.error?.message ??
   231→        (err instanceof Error ? err.message : 'SMTP connection test failed');
   232→      setTestResult({ ok: false, message });
   233→      toast.error(message);
   234→    },
   235→  });
   236→
   237→  // ---------- Send test email mutation ----------
   238→  const [testEmail, setTestEmail] = useState('');
   239→  const [emailResult, setEmailResult] = useState<
   240→    { ok: boolean; message: string } | null
   241→  >(null);
   242→
   243→  const sendEmailMutation = useMutation({
   244→    mutationFn: () => {
   245→      const payload = {
   246→        email: testEmail,
   247→        ...(isDirty
   248→          ? {
   249→              settings: {
   250→                provider: current.provider,
   251→                host: current.host,
   252→                port: current.port,
   253→                encryption: current.encryption,
   254→                username: current.username,
   255→                password: passwordInput,
   256→                fromName: current.fromName,
   257→                fromEmail: current.fromEmail,
   258→                replyTo: current.replyTo,
   259→                timeout: current.timeout,
   260→                isActive: current.isActive,
   261→              },
   262→            }
   263→          : {}),
   264→      };
   265→      return postApi<TestResponse>('/api/settings/smtp/test-email', payload);
   266→    },
   267→    onSuccess: (data) => {
   268→      setEmailResult({ ok: true, message: data.message });
   269→      toast.success(data.message);
   270→    },
   271→    onError: (err: unknown) => {
   272→      const message =
   273→        (err as ApiErrorPayload)?.error?.message ??
   274→        (err instanceof Error ? err.message : 'Failed to send test email');
   275→      setEmailResult({ ok: false, message });
   276→      toast.error(message);
   277→    },
   278→  });
   279→
   280→  // ---------- Reset (discard local draft) ----------
   281→  const handleDiscard = () => {
   282→    setDraft({});
   283→    setPasswordInput(PASSWORD_PLACEHOLDER);
   284→    setTestResult(null);
   285→    setEmailResult(null);
   286→    toast.info('Changes discarded');
   287→  };
   288→
   289→  // ---------- Loading skeleton ----------
   290→  if (isLoading) {
   291→    return (
   292→      <div className="space-y-6 p-6">
   293→        <div className="h-8 w-48 animate-pulse bg-muted rounded" />
   294→        {Array.from({ length: 3 }).map((_, i) => (
   295→          <Card key={i}>
   296→            <CardContent className="p-6 space-y-4">
   297→              {Array.from({ length: 4 }).map((_, j) => (
   298→                <div key={j} className="h-10 w-full animate-pulse bg-muted rounded" />
   299→              ))}
   300→            </CardContent>
   301→          </Card>
   302→        ))}
   303→      </div>
   304→    );
   305→  }
   306→
   307→  const encryptionOption = ENCRYPTION_OPTIONS.find(
   308→    (o) => o.value === current.encryption,
   309→  );
   310→
   311→  return (
   312→    <div className="space-y-6 p-6">
   313→      {/* Page Header */}
   314→      <div>
   315→        <h1 className="text-2xl font-bold tracking-tight sm:text-3xl">
   316→          SMTP Settings
   317→        </h1>
   318→        <p className="mt-1 text-sm text-muted-foreground">
   319→          Configure email delivery for transactional messages, notifications, and newsletters.
   320→        </p>
   321→      </div>
   322→
   323→      {/* 1. Email Sending */}
   324→      <Card>
   325→        <CardHeader className="pb-4">
   326→          <div className="flex items-center gap-2">
   327→            <SettingsIcon className="h-4 w-4 text-muted-foreground" />
   328→            <CardTitle className="text-sm font-semibold">Email Sending</CardTitle>
   329→          </div>
   330→          <CardDescription>
   331→            Toggle email delivery on or off for the entire site.
   332→          </CardDescription>
   333→        </CardHeader>
   334→        <CardContent>
   335→          <div className="flex items-center justify-between rounded-lg border p-4">
   336→            <div className="space-y-0.5">
   337→              <Label htmlFor="isActive" className="text-sm font-medium">
   338→                Enable Email Sending
   339→              </Label>
   340→              <p className="text-xs text-muted-foreground">
   341→                When disabled, the CMS will not attempt to send any outgoing emails.
   342→              </p>
   343→            </div>
   344→            <Switch
   345→              id="isActive"
   346→              checked={current.isActive}
   347→              onCheckedChange={(v) => update('isActive', v)}
   348→            />
   349→          </div>
   350→        </CardContent>
   351→      </Card>
   352→
   353→      {/* 2. SMTP Connection */}
   354→      <Card>
   355→        <CardHeader className="pb-4">
   356→          <div className="flex items-center gap-2">
   357→            <Server className="h-4 w-4 text-muted-foreground" />
   358→            <CardTitle className="text-sm font-semibold">SMTP Connection</CardTitle>
   359→          </div>
   360→          <CardDescription>
   361→            Mail server endpoint details and encryption mode.
   362→          </CardDescription>
   363→        </CardHeader>
   364→        <CardContent className="space-y-4">
   365→          <div className="grid gap-4 sm:grid-cols-2">
   366→            <div className="space-y-1.5 sm:col-span-2">
   367→              <Label htmlFor="host" className="text-sm font-medium">
   368→                SMTP Host
   369→              </Label>
   370→              <Input
   371→                id="host"
   372→                value={current.host}
   373→                onChange={(e) => update('host', e.target.value)}
   374→                placeholder="smtp.example.com"
   375→                autoComplete="off"
   376→                spellCheck={false}
   377→              />
   378→              <p className="text-xs text-muted-foreground">
   379→                The hostname or IP address of your SMTP server.
   380→              </p>
   381→            </div>
   382→
   383→            <div className="space-y-1.5">
   384→              <Label htmlFor="port" className="text-sm font-medium">
   385→                Port
   386→              </Label>
   387→              <Input
   388→                id="port"
   389→                type="number"
   390→                value={current.port}
   391→                onChange={(e) => update('port', Number(e.target.value))}
   392→                placeholder="587"
   393→                min={1}
   394→                max={65535}
   395→              />
   396→              <p className="text-xs text-muted-foreground">
   397→                Suggested: <span className="font-medium">{encryptionOption?.port ?? 587}</span> for{' '}
   398→                {encryptionOption?.label ?? 'STARTTLS'}.
   399→              </p>
   400→            </div>
   401→
   402→            <div className="space-y-1.5">
   403→              <Label htmlFor="encryption" className="text-sm font-medium">
   404→                Encryption
   405→              </Label>
   406→              <Select
   407→                value={current.encryption}
   408→                onValueChange={(v) => handleEncryptionChange(v as Encryption)}
   409→              >
   410→                <SelectTrigger id="encryption">
   411→                  <SelectValue />
   412→                </SelectTrigger>
   413→                <SelectContent>
   414→                  {ENCRYPTION_OPTIONS.map((opt) => (
   415→                    <SelectItem key={opt.value} value={opt.value}>
   416→                      <div className="flex flex-col">
   417→                        <span className="font-medium">{opt.label}</span>
   418→                        <span className="text-xs text-muted-foreground">
   419→                          {opt.hint}
   420→                        </span>
   421→                      </div>
   422→                    </SelectItem>
   423→                  ))}
   424→                </SelectContent>
   425→              </Select>
   426→              <p className="text-xs text-muted-foreground">
   427→                STARTTLS is recommended for modern mail servers.
   428→              </p>
   429→            </div>
   430→
   431→            <div className="space-y-1.5 sm:col-span-2">
   432→              <Label htmlFor="timeout" className="text-sm font-medium">
   433→                Connection Timeout (seconds)
   434→              </Label>
   435→              <Input
   436→                id="timeout"
   437→                type="number"
   438→                value={current.timeout}
   439→                onChange={(e) => update('timeout', Number(e.target.value))}
   440→                min={1}
   441→                max={120}
   442→                className="sm:w-40"
   443→              />
   444→              <p className="text-xs text-muted-foreground">
   445→                How long to wait before aborting a connection attempt. Default is 10 seconds.
   446→              </p>
   447→            </div>
   448→          </div>
   449→        </CardContent>
   450→      </Card>
   451→
   452→      {/* 3. Authentication */}
   453→      <Card>
   454→        <CardHeader className="pb-4">
   455→          <div className="flex items-center gap-2">
   456→            <ShieldCheck className="h-4 w-4 text-muted-foreground" />
   457→            <CardTitle className="text-sm font-semibold">Authentication</CardTitle>
   458→          </div>
   459→          <CardDescription>
   460→            Credentials used to authenticate with the SMTP server.
   461→          </CardDescription>
   462→        </CardHeader>
   463→        <CardContent className="space-y-4">
   464→          <div className="grid gap-4 sm:grid-cols-2">
   465→            <div className="space-y-1.5">
   466→              <Label htmlFor="username" className="text-sm font-medium">
   467→                Username
   468→              </Label>
   469→              <Input
   470→                id="username"
   471→                value={current.username}
   472→                onChange={(e) => update('username', e.target.value)}
   473→                placeholder="user@example.com"
   474→                autoComplete="off"
   475→                spellCheck={false}
   476→              />
   477→              <p className="text-xs text-muted-foreground">
   478→                Often the same as your From Email address.
   479→              </p>
   480→            </div>
   481→
   482→            <div className="space-y-1.5">
   483→              <Label htmlFor="password" className="text-sm font-medium">
   484→                Password
   485→              </Label>
   486→              {passwordInput === PASSWORD_PLACEHOLDER ? (
   487→                // Saved password — show masked placeholder + "Change" button
   488→                <div className="flex items-center gap-2">
   489→                  <div
   490→                    id="password"
   491→                    className="flex h-9 w-full items-center rounded-md border border-input bg-transparent px-3 text-sm text-muted-foreground"
   492→                  >
   493→                    {PASSWORD_PLACEHOLDER}
   494→                  </div>
   495→                  <Button
   496→                    type="button"
   497→                    variant="outline"
   498→                    size="sm"
   499→                    onClick={() => setPasswordInput('')}
   500→                  >
   501→                    Change
   502→                  </Button>
   503→                </div>
   504→              ) : (
   505→                // Editable password input with show/hide toggle
   506→                <div className="relative">
   507→                  <Input
   508→                    id="password"
   509→                    type={showPassword ? 'text' : 'password'}
   510→                    value={passwordInput}
   511→                    onChange={(e) => {
   512→                      const val = e.target.value;
   513→                      // If the user typed over the masked placeholder, stray bullets
   514→                      // may be embedded in the new value. Strip them out.
   515→                      if (val.includes('•')) {
   516→                        setPasswordInput(val.replace(/•/g, ''));
   517→                      } else {
   518→                        setPasswordInput(val);
   519→                      }
   520→                    }}
   521→                    placeholder="Enter new password"
   522→                    autoComplete="new-password"
   523→                    spellCheck={false}
   524→                    className="pr-10"
   525→                  />
   526→                  <Button
   527→                    type="button"
   528→                    variant="ghost"
   529→                    size="icon"
   530→                    className="absolute right-0 top-0 h-9 w-9 hover:bg-transparent"
   531→                    onClick={() => setShowPassword((s) => !s)}
   532→                    aria-label={showPassword ? 'Hide password' : 'Show password'}
   533→                  >
   534→                    {showPassword ? (
   535→                      <EyeOff className="h-4 w-4 text-muted-foreground" />
   536→                    ) : (
   537→                      <Eye className="h-4 w-4 text-muted-foreground" />
   538→                    )}
   539→                  </Button>
   540→                </div>
   541→              )}
   542→              <div className="flex items-center gap-2 text-xs">
   543→                {hasSavedPassword ? (
   544→                  <span className="inline-flex items-center gap-1 text-emerald-600 dark:text-emerald-500">
   545→                    <CheckCircle2 className="h-3 w-3" />• saved
   546→                  </span>
   547→                ) : (
   548→                  <span className="text-muted-foreground">No password saved yet</span>
   549→                )}
   550→                <span className="text-muted-foreground">
   551→                  · Encrypted at rest with AES-256-GCM
   552→                </span>
   553→              </div>
   554→            </div>
   555→          </div>
   556→        </CardContent>
   557→      </Card>
   558→
   559→      {/* 4. Sender Identity */}
   560→      <Card>
   561→        <CardHeader className="pb-4">
   562→          <div className="flex items-center gap-2">
   563→            <Mail className="h-4 w-4 text-muted-foreground" />
   564→            <CardTitle className="text-sm font-semibold">Sender Identity</CardTitle>
   565→          </div>
   566→          <CardDescription>
   567→            The From and Reply-To addresses used for outgoing emails.
   568→          </CardDescription>
   569→        </CardHeader>
   570→        <CardContent className="space-y-4">
   571→          <div className="grid gap-4 sm:grid-cols-2">
   572→            <div className="space-y-1.5">
   573→              <Label htmlFor="fromName" className="text-sm font-medium">
   574→                From Name
   575→              </Label>
   576→              <Input
   577→                id="fromName"
   578→                value={current.fromName}
   579→                onChange={(e) => update('fromName', e.target.value)}
   580→                placeholder="Acme Inc."
   581→              />
   582→              <p className="text-xs text-muted-foreground">
   583→                Display name shown in the recipient&apos;s inbox.
   584→              </p>
   585→            </div>
   586→
   587→            <div className="space-y-1.5">
   588→              <Label htmlFor="fromEmail" className="text-sm font-medium">
   589→                From Email
   590→              </Label>
   591→              <Input
   592→                id="fromEmail"
   593→                type="email"
   594→                value={current.fromEmail}
   595→                onChange={(e) => update('fromEmail', e.target.value)}
   596→                placeholder="noreply@example.com"
   597→                autoComplete="off"
   598→                spellCheck={false}
   599→              />
   600→              <p className="text-xs text-muted-foreground">
   601→                Address that sends the emails. Must be allowed by your SMTP provider.
   602→              </p>
   603→            </div>
   604→
   605→            <div className="space-y-1.5 sm:col-span-2">
   606→              <Label htmlFor="replyTo" className="text-sm font-medium">
   607→                Reply-To (optional)
   608→              </Label>
   609→              <Input
   610→                id="replyTo"
   611→                type="email"
   612→                value={current.replyTo}
   613→                onChange={(e) => update('replyTo', e.target.value)}
   614→                placeholder="support@example.com"
   615→                autoComplete="off"
