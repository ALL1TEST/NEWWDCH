     3→// ============================================================
     4→// PLATFORM PLANS & PRICING — simplified admin UI.
     5→// ============================================================
     6→// Owner sees compact plan cards (Free / Plus / Pro / Max …) with a
     7→// quick Active toggle, a compact price + features + limits summary,
     8→// and an "Edit Plan" button. The Edit Plan modal has ONE base price
     9→// configuration per plan — NO per-currency matrix:
    10→//
    11→//   Basic Information: Name · Default/Fallback Currency (country/
    12→//   currency selector: 🇺🇸 United States — USD — $) · Auto Currency
    13→//   toggle · Monthly Price · Yearly Price · Billing Interval · Active
    14→//   then: Stripe Billing (sync + status) · Feature Access · Usage
    15→//   Limits.
    16→//
    17→// AUTO CURRENCY: each plan carries an autoCurrency flag. When ON,
    18→// the customer's currency is resolved SERVER-SIDE from their IP
    19→// (country → currency → plan price for that currency → fallback to
    20→// the plan default when unsupported) — the admin never picks the
    21→// customer's currency. When OFF, every customer is billed in the
    22→// plan's default currency. Per-currency prices (e.g. 90 MAD) come
    23→// from the PLATFORM-level regional config (CountryPricing), not
    24→// from this modal.
    25→//
    26→// All writes go through /api/platform/admin/plans (GET list,
    27→// POST create) + /api/platform/admin/plans/[planId] (PUT update)
    28→// — owner-only — the same shared plan-config cache the Client
    29→// Billing page and MRR read. Server-side entitlement enforcement
    30→// (hasFeature) and usage-limit checks (checkLimit) consume the
    31→// same data, so toggling a feature off here denies it for clients
    32→// on the next request.
    33→// ============================================================

   105→/** Resolve the badge id for the shared PlanBadge component. Falls
   106→ *  back to 'free' for any unknown plan id so future plans still
   107→ *  render with a sensible neutral badge. */
   108→function getPlanBadgeId(planId: string): 'free' | 'plus' | 'pro' | 'max' {
   109→  if (planId === 'free' || planId === 'plus' || planId === 'pro' || planId === 'max') return planId;
   110→  // Legacy 'beta' plan id (pre-migration) → render as 'plus' badge.
   111→  if (planId === 'beta') return 'plus';
   112→  return 'free';
   113→}

   115→// -------------------- Price formatting --------------------
   116→// formatMoney comes from the shared currency catalog (client-safe):
   117→//   (9, 'USD') → "$9" · (90, 'MAD') → "90 MAD" · (9, 'CHF') → "CHF 9".
   118→
   119→/** Format the LARGE primary price on a plan card — integer amounts.
   120→ *  e.g. (90, 'MAD') → "90 MAD"; (0, 'CHF') → "CHF 0". */
   121→function formatPriceSymbol(amount: number, currency: string): string {
   122→  return formatMoney(amount, currency, 0);
   123→}
   124→
   125→/** Format with 2 decimals — used for the monthly-equivalent of a
   126→ *  yearly price (yearly 90 → "CHF 7.50"). Always two decimals so the
   127→ *  large primary price reads "$7.50" / "CHF 40.83" / "€82.50". */
   128→function formatPriceSymbolMonthlyEquiv(amount: number, currency: string): string {
   129→  return formatMoney(amount, currency, 2);
   130→}

   146→  plan,
   147→  canEdit,
   148→  billingInterval,
   149→}: {
   150→  plan: PlanConfigData;
   151→  canEdit: boolean;
   152→  billingInterval: 'monthly' | 'yearly';
   153→}) {
   154→  const queryClient = useQueryClient();
   155→  const [editing, setEditing] = useState(false);
   156→
   157→  // Quick active toggle — auto-saves immediately (owner-only).
   158→  // Reuses the same PUT endpoint and the same invalidation chain
   159→  // as the full Edit modal so clients see the new status on the
   160→  // next read.
   161→  const activeMutation = useMutation({
   162→    mutationFn: (active: boolean) =>
   163→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, { active }),
   164→    onSuccess: (data) => {
   165→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   166→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });

   171→    },
   172→    onError: (err: unknown) => {
   173→      const e = err as { error?: { message?: string }; message?: string };
   174→      toast.error(
   175→        e?.error?.message ?? e?.message ?? 'Unable to change plan status. OWNER access required.',

   177→    },
   178→  });
   179→
   180→  // Feature list shown on the card. Prefer the client-facing marketing
   181→  // copy (plan.features); fall back to the structured entitlement labels
   182→  // so a brand-new plan with no marketing copy still surfaces a readable
   183→  // feature list. Either way the data shown is real plan data — no mock.
   184→  const featureItems =
   185→    plan.features.length > 0
   186→      ? plan.features
   187→      : plan.entitlements.map((k) => ENTITLEMENT_LABELS[k as EntitlementKey] ?? k);
   188→
   189→  return (
   190→    <>
   191→      <div
   192→        className={`group relative flex h-full flex-col overflow-hidden rounded-3xl border bg-card p-8 text-card-foreground transition-colors duration-300 hover:border-foreground/20 ${
   193→          plan.active ? '' : 'opacity-60'
   194→        }`}
   195→      >
   196→        {/* Header: plan name (LARGEST typographic element on the card).
   197→             The price sits BELOW the name on its own row (see price
   198→             block below), not on the same line — so the name has the

   202→          <h3 className="text-5xl font-bold tracking-tight text-foreground">{plan.name}</h3>
   203→        </div>
   204→
   205→        {/* Price — sits BELOW the plan name on its own row. Reflects
   206→             the global billing-interval selector (Monthly / Yearly).
   207→             The price's LEFT edge aligns with the plan name's first

   209→             Plus/Pro, "M" of Max (no left breakout, so the price column
   210→             is flush with the name column above it). A RIGHT-side

   213→             line at wider card widths; flex-wrap lets it wrap gracefully
   214→             at narrower widths. Only the price VALUES change between
   215→             Monthly and Yearly.
   216→
   217→             Typography (IDENTICAL in both modes, same on all 4 cards):
   218→               - Main price: text-4xl font-semibold (36px) — slightly
   219→                 smaller than the plan name (text-5xl = 48px), still
   220→                 clearly larger than the labels (so the monthly-equiv
   221→                 reads as the MAIN price)
   222→               - /month, /year: text-sm text-muted-foreground (14px) —
   223→                 small muted (the yearly total is smaller/muted beside
   224→                 the main monthly-equiv price)
   225→               - Free:    $0
   226→               - Monthly: $X / month
   227→               - Yearly:  $X.XX / month   $X / year
   228→             For Yearly, the monthly equivalent (priceYearly / 12,
   229→             computed dynamically, never hardcoded) is the LARGE primary
   230→             price; the actual yearly total is shown smaller and muted
   231→             beside it on the same line. "/ month" belongs to the large
   232→             monthly price; "/ year" belongs to the small yearly total.
   233→             Width budget (measured): card content (p-8) ≈ 245-269px
   234→             (varies with viewport) + right -mr-6 breakout (24px) ≈
   235→             269-293px. Widest case = Max Yearly: $82.50(134) + gap-2(8)
   236→             + /month(55) + gap-2(8) + $990/year(81) = 286px → fits at
   237→             wider card widths, wraps to 2 lines at narrower widths
   238→             (flex-wrap). shrink-0 + whitespace-nowrap + leading-none on
   239→             each price span keep each span's internals from breaking;
   240→             uniform gap-2 (8px) between every price element (no ml-2)
   241→             so wrapped lines stay left-aligned. */}
   242→        <div className="-mr-6 mt-6">
   243→          {plan.isFree ? (
   244→            <div className="flex flex-wrap items-baseline gap-2">
   245→              <span className="shrink-0 whitespace-nowrap text-4xl font-semibold leading-none tracking-tight text-foreground">
   246→                {formatPriceSymbol(0, plan.currency)}
   247→              </span>
   248→            </div>
   249→          ) : billingInterval === 'yearly' ? (
   250→            // Yearly — INLINE: [LARGE $X.XX] [small / month]
   251→            // [small $X / year]. The monthly equivalent (priceYearly /
   252→            // 12) is the dominant large price; the real yearly total
   253→            // stays small/muted beside it.
   254→            //   e.g. Plus →  $7.50 / month   $90 / year
   255→            //        Pro  →  $40.83 / month  $490 / year
   256→            //        Max  →  $82.50 / month  $990 / year
   257→            <div className="flex flex-wrap items-baseline gap-2">
   258→              <span className="shrink-0 whitespace-nowrap text-4xl font-semibold leading-none tracking-tight text-foreground">
   259→                {formatPriceSymbolMonthlyEquiv(plan.priceYearly / 12, plan.currency)}
   260→              </span>
   261→              <span className="whitespace-nowrap text-sm text-muted-foreground">/ month</span>
   262→              <span className="shrink-0 whitespace-nowrap text-sm text-muted-foreground">
   263→                {formatPriceSymbol(plan.priceYearly, plan.currency)} / year
   264→              </span>
   265→            </div>
   266→          ) : (
   267→            <div className="flex flex-wrap items-baseline gap-2">
   268→              <span className="shrink-0 whitespace-nowrap text-4xl font-semibold leading-none tracking-tight text-foreground">
   269→                {formatPriceSymbol(plan.priceMonthly, plan.currency)}
   270→              </span>
   271→              <span className="whitespace-nowrap text-sm text-muted-foreground">/ month</span>
   272→            </div>
   273→          )}
   274→        </div>
   275→
   276→        {/* Thin horizontal divider between the price section and the features section.
   277→             Equidistant from the price above and the feature list below (mt-6 both
   278→             sides) for a consistent, compact vertical rhythm on every card. The
   279→             divider uses -mr-6 (right breakout only) so its left edge aligns with
   280→             the price row + plan name above (flush to the content left) while its
   281→             right edge extends into the card padding to match the price row —
   282→             identical in both modes. */}
   283→        <div className="-mr-6 mt-6 h-px bg-border" aria-hidden />
   284→
   285→        {/* Feature items (section label omitted). Same typography on every card:
   286→             text-[15px] leading-normal so features read as a clear medium-size
   287→             block — visibly subordinate to the text-5xl plan name and text-4xl
   288→             price, never competing with them. Check icon is h-4 w-4 with mt-0.5
   289→             so its optical center aligns with the first line of feature text;
   290→             gap-2.5 keeps a consistent icon↔text column gutter; items-start
   291→             pins the icon to the first line even when a feature wraps to 2
   292→             lines, so rows never look vertically misaligned. All cards share
   293→             the card's p-8 left padding, so every feature column starts at the
   294→             same x. */}
   295→        <div className="mt-6">
   296→          <ul className="space-y-2.5">
   297→            {featureItems.length === 0 ? (
   298→              <li className="text-[15px] leading-normal text-muted-foreground">No features configured</li>
   299→            ) : (
   300→              featureItems.map((f, i) => (
   301→                <li key={`${f}-${i}`} className="flex items-start gap-2.5 text-[15px] leading-normal text-card-foreground">
   302→                  <Check className="mt-0.5 h-4 w-4 shrink-0 text-muted-foreground" />
   303→                  <span>{f}</span>
   304→                </li>
   305→              ))
   306→            )}
   307→          </ul>
   308→        </div>
   309→
   310→        {/* Bottom row — Edit Plan (left, flex-1) + quick active toggle
   311→             (right, shrink-0), pinned to the card bottom. The toggle used
   312→             to live in the header's top-right; it was relocated here so the
   313→             name + price row above can use the full card width and the
   314→             price can sit on the same line as the name. The Edit button
   315→             changed from w-full to flex-1 to share the row with the toggle.
   316→             The toggle's logic (auto-save PUT to /api/platform/admin/plans/
   317→             [planId] with { active }) is unchanged. */}
   318→        <div className="mt-auto flex items-center gap-3 pt-8">
   319→          <Button
   320→            variant="outline"
   321→            size="sm"
   322→            className="flex-1 rounded-full"
   323→            disabled={!canEdit}
   324→            onClick={() => setEditing(true)}
   325→          >
   326→            <Pencil className="h-4 w-4 mr-2" />
   327→            Edit Plan
   328→          </Button>
   329→          <div className="flex shrink-0 items-center gap-2">
   330→            <Label
   331→              htmlFor={`active-${plan.planId}`}
   332→              className="cursor-pointer text-xs text-muted-foreground"
   333→            >
   334→              {plan.active ? 'Active' : 'Inactive'}
   335→            </Label>
   336→            <Switch
   337→              id={`active-${plan.planId}`}
   338→              checked={plan.active}
   339→              disabled={!canEdit || activeMutation.isPending}
   340→              onCheckedChange={(v) => activeMutation.mutate(v)}

   342→          </div>
   343→        </div>
   344→        {!canEdit && (
   345→          <p className="mt-2 text-center text-[10px] text-muted-foreground">
   346→            Editing is restricted to the OWNER role.
   347→          </p>
   348→        )}
   349→      </div>
   350→
   351→      {editing && (
   352→        <EditPlanDialog plan={plan} open={editing} onOpenChange={setEditing} />
   353→      )}
   354→    </>
   355→  );
   356→}

   358→// -------------------- Edit Plan Dialog --------------------
   359→
   360→function EditPlanDialog({
   361→  plan,
   362→  open,
   363→  onOpenChange,
   364→}: {
   365→  plan: PlanConfigData;
   366→  open: boolean;
   367→  onOpenChange: (open: boolean) => void;
   368→}) {
   369→  const queryClient = useQueryClient();
   370→  const [name, setName] = useState(plan.name);
   371→  // ONE base price configuration — the plan's default/fallback
   372→  // currency (country/currency selector) + monthly/yearly prices
   373→  // denominated in it. NO per-currency matrix: other currency prices
   374→  // (e.g. 90 MAD) are platform-level regional config, resolved
   375→  // server-side per customer.
   376→  const [currency, setCurrency] = useState((plan.currency ?? 'CHF').toUpperCase());
   377→  const [autoCurrency, setAutoCurrency] = useState(plan.autoCurrency ?? true);
   378→  const [priceMonthly, setPriceMonthly] = useState(String(plan.priceMonthly ?? 0));
   379→  const [priceYearly, setPriceYearly] = useState(String(plan.priceYearly ?? 0));
   380→  const [interval, setInterval] = useState<'monthly' | 'yearly'>(plan.interval);
   381→  const [active, setActive] = useState(plan.active);
   382→  const [entitlements, setEntitlements] = useState<string[]>(plan.entitlements);
   383→  const [limits, setLimits] = useState<PlanLimits>(plan.limits);
   384→  // Free-trial duration (only used when the base price is 0).
   385→  const [freePlanDurationDays, setFreePlanDurationDays] = useState<string>(
   386→    plan.freePlanDurationDays == null ? '' : String(plan.freePlanDurationDays),
   387→  );
   388→  const [stripeOpen, setStripeOpen] = useState(false);
   389→  // Stripe Price IDs per currency — maintained by the Sync to Stripe
   390→  // button (never hand-edited; shown as compact wired-currency badges).
   391→  const [wiredMap, setWiredMap] = useState<StripePriceIdsByCurrency>(
   392→    plan.stripePriceIdsByCurrency ?? {},
   393→  );
   394→
   395→  // The EditPlanDialog is conditionally rendered by the parent (mounted
   396→  // fresh each time the user opens it), so the useState initializers
   397→  // above already seed local state from the latest server snapshot.
   398→  // No useEffect sync is needed — that would just trigger cascading
   399→  // renders (and the React Compiler correctly flags it).
   400→
   401→  const monthlyNum = Number(priceMonthly) || 0;
   402→  const yearlyNum = Number(priceYearly) || 0;
   403→
   404→  const reset = () => {
   405→    setName(plan.name);
   406→    setCurrency((plan.currency ?? 'CHF').toUpperCase());
   407→    setAutoCurrency(plan.autoCurrency ?? true);
   408→    setPriceMonthly(String(plan.priceMonthly ?? 0));
   409→    setPriceYearly(String(plan.priceYearly ?? 0));
   410→    setInterval(plan.interval);
   411→    setActive(plan.active);
   412→    setEntitlements(plan.entitlements);
   413→    setLimits(plan.limits);
   414→    setFreePlanDurationDays(plan.freePlanDurationDays == null ? '' : String(plan.freePlanDurationDays));
   415→    setWiredMap(plan.stripePriceIdsByCurrency ?? {});
   416→  };
   417→
   418→  const saveMutation = useMutation({
   419→    mutationFn: (patch: PlanPatch) =>
   420→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, patch),
   421→    onSuccess: (data) => {
   422→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   423→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   424→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   425→      toast.success(`${data.name} updated — changes are now live for clients.`);
   426→      onOpenChange(false);
   427→    },
   428→    onError: (err: unknown) => {
   429→      const e = err as { error?: { message?: string }; message?: string };
   430→      toast.error(
   431→        e?.error?.message ?? e?.message ?? 'Unable to save plan. You may need OWNER access.',
   432→      );
   433→    },
   434→  });
   435→
   436→  // Sync this plan to Stripe (creates/reuses the Stripe Product + one
   437→  // Stripe Price per (currency, interval) pair and writes the resolved
   438→  // stripePriceIdsByCurrency map back onto the plan row). Surfaces
   439→  // Stripe errors inline so the admin can fix them.
   440→  const syncToStripeMutation = useMutation({
   441→    mutationFn: () =>
   442→      postApi<{
   443→        planId: string;
   444→        stripePriceIdsByCurrency: StripePriceIdsByCurrency;
   445→        created: number;
   446→        defaultCurrencySnapshot: { monthly: string | null; yearly: string | null };
   447→      }>(`/api/platform/admin/plans/${plan.planId}/sync-stripe`),
   448→    onSuccess: (data) => {
   449→      // Reflect the synced per-currency Stripe Price IDs in the local
   450→      // state so the admin can see them without re-opening the dialog.
   451→      setWiredMap(data.stripePriceIdsByCurrency);
   452→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   453→      toast.success(
   454→        data.created > 0
   455→          ? `Synced to Stripe — ${data.created} new Price(s) created.`
   456→          : 'Stripe Prices are already in sync.',
   457→      );
   458→    },
   459→    onError: (err: unknown) => {
   460→      const e = err as { code?: string; message?: string; error?: { code?: string; message?: string } };
   461→      const code = e?.code ?? e?.error?.code;
   462→      const msg = e?.message ?? e?.error?.message;
   463→      if (code === 'PAYMENT_PROVIDER_NOT_CONFIGURED') {
   464→        toast.error('Stripe is not connected. Configure credentials in Platform Admin → Stripe Settings first.');
   465→      } else if (code === 'VALIDATION_ERROR') {
   466→        toast.error(msg || 'Free plans do not need Stripe — set a price above 0 first.');
   467→      } else {
   468→        toast.error(msg || 'Unable to sync plan to Stripe.');
   469→      }
   470→    },
   471→  });
   472→
   473→  // A plan is free when its base price is 0 for both intervals.
   474→  const isFreeDerived = monthlyNum === 0 && yearlyNum === 0;
   475→
   476→  const buildPatch = (): PlanPatch => {
   477→    return {
   478→      name,
   479→      // The ONE base price configuration — the backend derives the
   480→      // default currency's pricesByCurrency entry from these and
   481→      // preserves the platform's other-currency regional prices.
   482→      currency,
   483→      autoCurrency,
   484→      priceMonthly: monthlyNum,
   485→      priceYearly: yearlyNum,
   486→      interval,
   487→      isFree: isFreeDerived,
   488→      freePlanDurationDays: isFreeDerived ? (freePlanDurationDays.trim() === '' ? null : Number(freePlanDurationDays) || null) : null,
   489→      // No pricesByCurrency / stripePriceIdsByCurrency in the patch:
   490→      // per-currency prices are platform-level config and Stripe Price
   491→      // IDs are managed by the sync (both preserved on save).
   492→      active,
   493→      // features intentionally omitted — the backend derives the
