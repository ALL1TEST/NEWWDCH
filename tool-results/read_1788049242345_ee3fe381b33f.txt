     1→// ============================================================
     2→// STRIPE CLIENT + CONFIGURATION GUARD.
     3→// ============================================================
     4→// This is the SINGLE place that resolves Stripe credentials and
     5→// constructs the Stripe SDK client. The rest of the codebase calls
     6→// `isStripeConfigured()` to gate real billing; when false, checkout
     7→// returns 503 with a clear "not configured" message and NEVER fakes
     8→// a successful payment.
     9→//
    10→// CREDENTIAL RESOLUTION ORDER (per credential type):
    11→//   1. The StripeSettings singleton row in the DB (admin-configured
    12→//      via Platform Admin → Stripe Settings). Secret keys are stored
    13→//      AES-256-GCM encrypted; only the masked form is returned to the
    14→//      frontend. Publishable keys are stored as plaintext.
    15→//   2. process.env.STRIPE_* (fallback — works without any DB row,
    16→//      e.g. on first deploy before the admin has connected Stripe).
    17→//
    18→// So an admin can connect Stripe from the UI without editing .env,
    19→// and existing .env-based deployments keep working.
    20→//
    21→// Stripe SDK is only ever imported server-side (this file is imported
    22→// by API routes and webhook handlers). The publishable key is exposed
    23→// to the client only via /api/billing/checkout (GET) (so the client
    24→// can show the right UI without leaking the secret key).
    25→// ============================================================
    26→
    27→import Stripe from 'stripe';
    28→import { encrypt, decrypt, maskSecret } from '@/lib/encryption';
    29→
    30→// -------------------- DB-backed settings cache --------------------
    31→// In-memory cache of the StripeSettings singleton row. Re-read on
    32→// demand (invalidateStripeSettingsCache below) — call after every
    33→// mutation (save / test-connection) so the next getStripeClient()
    34→// picks up the new credentials without a dev-server restart.
    35→interface ResolvedStripeSettings {
    36→  mode: 'test' | 'live';
    37→  secretKey: string | null; // resolved plaintext (decrypted) for the active mode
    38→  publishableKey: string | null;
    39→  webhookSecret: string | null;
    40→  appUrl: string | null;
    41→}
    42→
    43→let _resolvedCache: ResolvedStripeSettings | null = null;
    44→let _resolvedCacheAt = 0;
    45→const RESOLVED_TTL_MS = 5_000; // 5s — keeps reads fast but picks up edits quickly
    46→
    47→/** Invalidate the in-memory cache of resolved Stripe settings.
    48→ *  Call after every saveStripeSettings() / mutation so the next
    49→ *  getStripeClient() reads the fresh DB row. */
    50→export function invalidateStripeSettingsCache(): void {
    51→  _resolvedCache = null;
    52→  _resolvedCacheAt = 0;
    53→  _client = null; // force the SDK client to be reconstructed on next use
    54→}
    55→
    56→/** Read the StripeSettings singleton row from the DB and decrypt
    57→ *  the secret keys for the ACTIVE mode. Returns null when no row
    58→ *  exists yet. Never throws — every error degrades to null. */
    59→async function loadResolvedSettings(): Promise<ResolvedStripeSettings | null> {
    60→  const { db } = await import('@/lib/db');
    61→  try {
    62→    const row = await db.stripeSettings.findUnique({ where: { id: 'singleton' } });
    63→    if (!row) return null;
    64→    const mode = (row.mode === 'live' ? 'live' : 'test') as 'test' | 'live';
    65→    const secretEncrypted = mode === 'live' ? row.secretKeyLiveEncrypted : row.secretKeyTestEncrypted;
    66→    const webhookEncrypted = mode === 'live' ? row.webhookSecretLiveEncrypted : row.webhookSecretTestEncrypted;
    67→    const publishable = mode === 'live' ? row.publishableKeyLive : row.publishableKeyTest;
    68→
    69→    let secretKey: string | null = null;
    70→    if (secretEncrypted) {
    71→      try {
    72→        secretKey = await decrypt(secretEncrypted);
    73→      } catch {
    74→        secretKey = null;
    75→      }
    76→    }
    77→    let webhookSecret: string | null = null;
    78→    if (webhookEncrypted) {
    79→      try {
    80→        webhookSecret = await decrypt(webhookEncrypted);
    81→      } catch {
    82→        webhookSecret = null;
    83→      }
    84→    }
    85→    return {
    86→      mode,
    87→      secretKey,
    88→      publishableKey: publishable ?? null,
    89→      webhookSecret,
    90→      appUrl: row.appUrl ?? null,
    91→    };
    92→  } catch {
    93→    // Table not yet created, db not reachable, etc. → fall back to env.
    94→    return null;
    95→  }
    96→}
    97→
    98→/** Get the resolved Stripe settings (DB → env fallback), with a
    99→ *  5s in-memory cache so it's cheap to call repeatedly. Public so
   100→ *  the admin settings page can read the masked view + the test
   101→ *  connection route can show what it tested with. */
   102→export async function getResolvedStripeSettings(): Promise<ResolvedStripeSettings & {
   103→  source: 'db' | 'env' | 'none';
   104→}> {
   105→  // 1. DB row (admin-configured).
   106→  const now = Date.now();
   107→  if (_resolvedCache && now - _resolvedCacheAt < RESOLVED_TTL_MS) {
   108→    return { ..._resolvedCache, source: 'db' };
   109→  }
   110→  const fromDb = await loadResolvedSettings();
   111→  if (fromDb && (fromDb.secretKey || fromDb.publishableKey || fromDb.webhookSecret)) {
   112→    _resolvedCache = fromDb;
   113→    _resolvedCacheAt = now;
   114→    return { ...fromDb, source: 'db' };
   115→  }
   116→  // 2. Env fallback.
   117→  const envSettings: ResolvedStripeSettings = {
   118→    mode: process.env.STRIPE_SECRET_KEY?.startsWith('sk_live_') ? 'live' : 'test',
   119→    secretKey: process.env.STRIPE_SECRET_KEY && process.env.STRIPE_SECRET_KEY.trim().length > 0
   120→      ? process.env.STRIPE_SECRET_KEY
   121→      : null,
   122→    publishableKey: process.env.STRIPE_PUBLISHABLE_KEY && process.env.STRIPE_PUBLISHABLE_KEY.trim().length > 0
   123→      ? process.env.STRIPE_PUBLISHABLE_KEY
   124→      : null,
   125→    webhookSecret: process.env.STRIPE_WEBHOOK_SECRET && process.env.STRIPE_WEBHOOK_SECRET.trim().length > 0
   126→      ? process.env.STRIPE_WEBHOOK_SECRET
   127→      : null,
   128→    appUrl: process.env.STRIPE_APP_URL ?? null,
   129→  };
   130→  // If env has no usable secret key either, source is 'none' (the
   131→  // admin UI shows "Stripe is not connected — configure below").
   132→  const hasEnvCreds = Boolean(envSettings.secretKey) || Boolean(envSettings.publishableKey) || Boolean(envSettings.webhookSecret);
   133→  return { ...envSettings, source: hasEnvCreds ? 'env' : 'none' };
   134→}

   136→/** True when Stripe secret key is configured (DB or env). */
   137→export async function isStripeConfiguredAsync(): Promise<boolean> {
   138→  const s = await getResolvedStripeSettings();
   139→  return Boolean(s.secretKey && s.secretKey.trim().length > 0);

   152→/** Get the shared Stripe SDK client. Reads the secret key from DB
   153→ *  first, then falls back to env. Throws if neither is configured —
   154→ *  call isStripeConfiguredAsync() first. */
   155→export async function getStripeClient(): Promise<Stripe> {
   156→  const settings = await getResolvedStripeSettings();
   157→  const key = settings.secretKey;
   158→  if (!key) throw new Error('Stripe is not configured. Set credentials in Platform Admin → Stripe Settings or in .env.');
   159→  // If the cache is fresh and the key matches, reuse the SDK client.
   160→  if (_client && _cachedClientKey === key) return _client;
   161→  _client = new Stripe(key, {
   162→    // Pin to a stable API version. Update together with the SDK.
   163→    apiVersion: '2026-08-26.dahlia' as Stripe.LatestApiVersion,
   164→    typescript: true,
   165→    // Surface errors as detailed objects (don't hide fields).
   166→    maxNetworkRetries: 2,
   167→  });
   168→  _cachedClientKey = key;
   169→  return _client;
   170→}

   190→/**

   192→ * Reads from PlanConfig.stripePriceIdMonthly / stripePriceIdYearly
   193→ * (set via Platform Admin → Edit Plan).

   195→ * Returns null when the plan has no Stripe price wired for the

   198→export async function resolveStripePriceId(
   199→  planId: string,
   200→  interval: 'monthly' | 'yearly',
   201→): Promise<string | null> {
   202→  const { db } = await import('@/lib/db');
   203→  const row = await db.planConfig.findUnique({ where: { planId } });
   204→  if (!row) return null;
   205→  if (interval === 'yearly') return row.stripePriceIdYearly ?? null;
   206→  return row.stripePriceIdMonthly ?? null;
   207→}

   209→/** Verify the Stripe webhook signature. Reads the webhook secret from
   210→ *  DB first, then falls back to STRIPE_WEBHOOK_SECRET. Returns the
   211→ *  verified event or null if verification fails (or Stripe isn't
   212→ *  configured). */
   213→export async function verifyStripeWebhook(
   214→  rawBody: Buffer | string,
   215→  signature: string,
   216→): Promise<Stripe.Event | null> {
   217→  const settings = await getResolvedStripeSettings();
   218→  const secret = settings.webhookSecret;
   219→  if (!secret) return null;
   220→  try {
   221→    const stripe = await getStripeClient();
   222→    return await stripe.webhooks.constructEventAsync(
   223→      rawBody instanceof Buffer ? rawBody : Buffer.from(rawBody),
   224→      signature,
   225→      secret,
   226→    );
   227→  } catch {
   228→    return null;
   229→  }
   230→}

   243→/**
   244→ * Return the Stripe Customer id for `user`, reusing an existing
   245→ * Customer when `existingCustomerId` resolves to a live (non-deleted)
   246→ * Stripe Customer, otherwise creating a new Customer tagged with
   247→ * `metadata.userId`. On retrieve failure of an existing id, falls
   248→ * through to create.
   249→ */
   250→export async function getOrCreateStripeCustomer(
   251→  stripe: Stripe,
   252→  user: { id: string; email: string; name?: string | null },
   253→  existingCustomerId?: string | null,
   254→): Promise<string> {
   255→  if (existingCustomerId) {
   256→    try {
   257→      const customer = await stripe.customers.retrieve(existingCustomerId);
   258→      if (customer && !(customer as { deleted?: boolean }).deleted) {
   259→        return customer.id;
   260→      }
   261→    } catch {
   262→      // fall through to create a new Customer
   263→    }
   264→  }
   265→  const created = await stripe.customers.create({
   266→    email: user.email,
   267→    name: user.name ?? undefined,
   268→    metadata: { userId: user.id },
   269→  });
   270→  return created.id;
   271→}

   273→/**
   274→ * Mirror a local Coupon into a Stripe Coupon + Promotion Code so
   275→ * Stripe Checkout can apply it as a `discounts[]` entry. Reuses
   276→ * existing Stripe objects when present (idempotent). Returns the
   277→ * Stripe Coupon id + Promotion Code id so the caller can cache
   278→ * them on the local Coupon row.
   279→ */
   280→export async function createStripeCouponMirror(
   281→  stripe: Stripe,
   282→  coupon: {
   283→    code: string;
   284→    type: 'percent' | 'fixed';
   285→    value: number;
   286→    currency: string;
   287→    maxRedemptions?: number | null;
   288→    perCustomerLimit?: number | null;
   289→    expiresAt?: string | null;
   290→    startsAt?: string | null;
   291→  },
   292→): Promise<{ stripeCouponId: string; stripePromotionCodeId: string }> {
   293→  // Derive a safe Stripe Coupon id: local_<code_lowered_alphanum>, truncate to 40 chars.
   294→  const derivedId = `local_${coupon.code.toLowerCase().replace(/[^a-z0-9]/g, '')}`.slice(0, 40);
   295→
   296→  // Reuse an existing Stripe Coupon with that id, else create one.
   297→  let stripeCouponId: string;
   298→  try {
   299→    const existing = await stripe.coupons.retrieve(derivedId);
   300→    stripeCouponId = existing.id;
   301→  } catch {
   302→    const created = await stripe.coupons.create({
   303→      id: derivedId,
   304→      percent_off: coupon.type === 'percent' ? coupon.value : undefined,
   305→      amount_off: coupon.type === 'fixed' ? coupon.value * 100 : undefined,
   306→      currency: coupon.type === 'fixed' ? coupon.currency.toLowerCase() : undefined,
   307→      duration: 'once',
   308→      max_redemptions: coupon.maxRedemptions ?? undefined,
   309→      redeem_by: coupon.expiresAt ? Math.floor(new Date(coupon.expiresAt).getTime() / 1000) : undefined,
   310→      metadata: { localCode: coupon.code },
   311→    });
   312→    stripeCouponId = created.id;
   313→  }
   314→
   315→  // Create-or-reuse a Stripe Promotion Code with code: coupon.code.
   316→  let stripePromotionCodeId: string;
   317→  const existingPromos = await stripe.promotionCodes.list({ code: coupon.code, limit: 1 });
   318→  if (existingPromos.data.length > 0) {
   319→    stripePromotionCodeId = existingPromos.data[0].id;
   320→  } else {
   321→    const promo = await stripe.promotionCodes.create({
   322→      coupon: stripeCouponId,
   323→      code: coupon.code,
   324→      max_redemptions: coupon.maxRedemptions ?? undefined,
   325→      expires_at: coupon.expiresAt ? Math.floor(new Date(coupon.expiresAt).getTime() / 1000) : undefined,
   326→      metadata: { localCode: coupon.code },
   327→    });
   328→    stripePromotionCodeId = promo.id;
   329→  }
   330→
   331→  return { stripeCouponId, stripePromotionCodeId };
   332→}
   333→
   334→/**
   335→ * Best-effort teardown of a local Coupon's Stripe mirror: deactivate
   336→ * the Promotion Code (so it can no longer be redeemed) then delete the
   337→ * Stripe Coupon. Never throws — all errors are swallowed. Safe to call
   338→ * with null/undefined ids (no-op).
   339→ */
   340→export async function clearStripeCouponMirror(
   341→  stripe: Stripe,
   342→  stripeCouponId?: string | null,
   343→  stripePromotionCodeId?: string | null,
   344→): Promise<void> {
   345→  if (stripePromotionCodeId) {
   346→    try {
   347→      await stripe.promotionCodes.update(stripePromotionCodeId, { active: false });
   348→    } catch {
   349→      // best-effort — never throw
   350→    }
   351→  }
   352→  if (stripeCouponId) {
   353→    try {
   354→      await stripe.coupons.del(stripeCouponId);
   355→    } catch {
   356→      // best-effort — never throw
   357→    }
   358→  }
   359→}

   361→/**
   362→ * Sync a local PlanConfig to Stripe by ensuring both a monthly and a
   363→ * yearly Stripe Price exist (with `metadata.planId` + `metadata.interval`
   364→ * set so they can be discovered later). Reuses existing Stripe Prices
   365→ * / Products where possible. Returns the resolved { monthly, yearly }
   366→ * Stripe Price ids (existing + any newly created).
   367→ */
   368→export async function syncPlanToStripe(
   369→  stripe: Stripe,
   370→  plan: {
   371→    planId: string;
   372→    name: string;
   373→    priceMonthly: number;
   374→    priceYearly: number;
   375→    currency: string;
   376→    stripePriceIdMonthly?: string | null;
   377→    stripePriceIdYearly?: string | null;
   378→  },
   379→): Promise<{ stripePriceIdMonthly: string; stripePriceIdYearly: string }> {
   380→  const resolveInterval = async (interval: 'monthly' | 'yearly'): Promise<string> => {
   381→    const existing = interval === 'monthly' ? plan.stripePriceIdMonthly : plan.stripePriceIdYearly;
   382→    if (existing && existing.trim().length > 0) return existing;
   383→
   384→    // Try to find an existing active Stripe Price with matching metadata.
   385→    const list = await stripe.prices.list({ active: true, limit: 100 });
   386→    const match = list.data.find(
   387→      (p) => p.metadata?.planId === plan.planId && p.metadata?.interval === interval,
   388→    );
   389→    if (match) return match.id;
   390→
   391→    // Create-or-reuse a Stripe Product tagged with metadata.planId.
   392→    let productId: string | undefined;
   393→    try {
   394→      const existingProducts = await stripe.products.list({ limit: 100 });
   395→      const existingProduct = existingProducts.data.find(
   396→        (p) => p.metadata?.planId === plan.planId,
   397→      );
   398→      if (existingProduct) productId = existingProduct.id;
   399→    } catch {
   400→      // fall through to create
   401→    }
   402→    if (!productId) {
   403→      const product = await stripe.products.create({
   404→        name: plan.name,
   405→        metadata: { planId: plan.planId },
   406→      });
   407→      productId = product.id;
   408→    }
   409→
   410→    // Stripe's recurring.interval enum is 'day' | 'week' | 'month' | 'year'
   411→    // — map our local 'monthly'/'yearly' terms onto it.
   412→    const stripeInterval: 'month' | 'year' = interval === 'monthly' ? 'month' : 'year';
   413→    const price = await stripe.prices.create({
   414→      unit_amount: (interval === 'monthly' ? plan.priceMonthly : plan.priceYearly) * 100,
   415→      currency: plan.currency.toLowerCase(),
   416→      recurring: { interval: stripeInterval },
   417→      product: productId,
   418→      metadata: { planId: plan.planId, interval },
   419→    });
   420→    return price.id;
   421→  };
   422→
   423→  const stripePriceIdMonthly = await resolveInterval('monthly');
   424→  const stripePriceIdYearly = await resolveInterval('yearly');
   425→  return { stripePriceIdMonthly, stripePriceIdYearly };
   426→}

   428→/**
   429→ * Cancel a Stripe Subscription either at period end (subscriber keeps
   430→ * access until the period ends) or immediately. On failure, throws a
   431→ * typed Error with the Stripe error code/message included.
   432→ */
   433→export async function cancelStripeSubscription(
   434→  stripe: Stripe,
   435→  stripeSubscriptionId: string,
   436→  atPeriodEnd: boolean,
   437→): Promise<void> {
   438→  try {
   439→    if (atPeriodEnd) {
   440→      await stripe.subscriptions.update(stripeSubscriptionId, {
   441→        cancel_at_period_end: true,
   442→      });
   443→    } else {
   444→      await stripe.subscriptions.del(stripeSubscriptionId);
   445→    }
   446→  } catch (err: unknown) {
   447→    const e = (err ?? {}) as { code?: string; message?: string };
   448→    const code = e.code ?? 'stripe_error';
   449→    const message = e.message ?? 'Unknown Stripe error';
   450→    throw new Error(`cancelStripeSubscription failed (${code}): ${message}`);
   451→  }
   452→}

   479→/**
   480→ * Swap the underlying Stripe Price on an existing Subscription (single
   481→ * item assumed). Uses `proration_behavior: 'create_prorations'` so the
   482→ * customer is credited / charged the difference. On failure, throws a
   483→ * typed Error.
   484→ */
   485→export async function updateSubscriptionPrice(
   486→  stripe: Stripe,
   487→  stripeSubscriptionId: string,
   488→  newPriceId: string,
   489→): Promise<void> {
   490→  try {
   491→    const sub = await stripe.subscriptions.retrieve(stripeSubscriptionId);
   492→    const existingItem = sub.items.data[0];
   493→    if (existingItem) {
   494→      await stripe.subscriptions.update(stripeSubscriptionId, {
   495→        items: [{ id: existingItem.id, price: newPriceId }],
   496→        proration_behavior: 'create_prorations',
   497→      });
   498→    }
   499→  } catch (err: unknown) {
   500→    const e = (err ?? {}) as { code?: string; message?: string };
   501→    const code = e.code ?? 'stripe_error';
   502→    const message = e.message ?? 'Unknown Stripe error';
   503→    throw new Error(`updateSubscriptionPrice failed (${code}): ${message}`);
   504→  }
   505→}

   507→/**
   508→ * Pure helper (no Stripe call) that extracts coupon snapshot info from
   509→ * a Stripe Invoice. Returns `{ couponCode: null, stripeCouponId: null,
   510→ * discountAmount: null }` when the invoice has no discount, otherwise
   511→ * returns the localCode (when present on the Stripe Coupon metadata)
   512→ * or the Stripe Coupon id as `couponCode`, the Stripe Coupon id, and
   513→ * the discount amount in major units.
   514→ *
   515→ * Cast through `unknown` because the dahlia API version's type defs
   516→ * lag the actual API surface (some fields are missing / mistyped).
   517→ */
   518→export function extractInvoiceCoupon(invoice: Stripe.Invoice): {
   519→  couponCode: string | null;
   520→  stripeCouponId: string | null;
   521→  discountAmount: number | null;
   522→} {
   523→  const inv = invoice as unknown as {
   524→    discount?: {
   525→      coupon?: {
   526→        id: string;
   527→        metadata?: { localCode?: string } | null;
   528→        amount_off?: number | null;
   529→        percent_off?: number | null;
   530→      } | null;
   531→    } | null;
   532→    total_discount_amounts?: Array<{
   533→      amount: number;
   534→      coupon?: { id: string; metadata?: { localCode?: string } | null } | null;
   535→    }> | null;
   536→  };
   537→
   538→  if (inv.discount?.coupon) {
   539→    const stripeCouponId = inv.discount.coupon.id;
   540→    const couponCode = inv.discount.coupon.metadata?.localCode ?? inv.discount.coupon.id;
   541→    const rawAmount = inv.total_discount_amounts?.[0]?.amount ?? 0;
   542→    const discountAmount = rawAmount / 100;
   543→    return { couponCode, stripeCouponId, discountAmount };
   544→  }
   545→
   546→  return { couponCode: null, stripeCouponId: null, discountAmount: null };
   547→}

   549→/**
   550→ * Record a Stripe webhook event id in the WebhookEvent idempotency
   551→ * ledger. Returns `true` when the event was newly recorded (i.e. the
   552→ * caller should process it) or `false` when the event was already
   553→ * recorded (a duplicate — the caller MUST skip processing and return
   554→ * 200 to Stripe). Conservative: any error → returns `false` so the
   555→ * webhook skips; the request still returns 200 to Stripe either way
   556→ * (the outer route handler swallows the ledger error).
   557→ *
   558→ * Lazy-imports `@/lib/db` so this file stays safe to import from
   559→ * server-only contexts without forcing Prisma to load eagerly.
   560→ */
   561→export async function markWebhookEventProcessed(params: {
   562→  stripeEventId: string;
   563→  eventType: string;
   564→  apiVersion?: string;
   565→  objectId?: string;
   566→  outcome: 'processed' | 'skipped_duplicate' | 'error';
   567→  errorMessage?: string;
   568→}): Promise<boolean> {
   569→  const { db } = await import('@/lib/db');
   570→  try {
   571→    await db.webhookEvent.create({
   572→      data: {
   573→        stripeEventId: params.stripeEventId,
   574→        eventType: params.eventType,
   575→        apiVersion: params.apiVersion ?? null,
   576→        objectId: params.objectId ?? null,
   577→        outcome: params.outcome,
   578→        errorMessage: params.errorMessage ?? null,
   579→      },
   580→    });
   581→    return true;
   582→  } catch (err: unknown) {
   583→    const e = (err ?? {}) as { code?: string; message?: string };
   584→    if (e.code === 'P2002' || e.message?.includes('Unique constraint')) {
   585→      // duplicate → already processed
   586→      return false;
   587→    }
   588→    // conservative: any other error → treat as duplicate so the
   589→    // webhook skips processing (request still returns 200 to Stripe).
   590→    return false;
   591→  }
   592→}

   594→// ============================================================
   595→// ADMIN STRIPE SETTINGS — credential storage + Test Connection.
   596→// ============================================================
   597→// saveStripeSettings: encrypts secret keys (sk_*, whsec_*) and
   598→//   upserts the singleton row. Publishable keys (pk_*) are stored
   599→//   as plaintext. Returns the masked view (safe for the admin UI).
   600→//
   601→// getStripeSettingsForAdmin: returns the masked view + last test
   602→//   result + webhook URL hint. Never returns raw secret values —
   603→//   the admin UI shows masked strings (sk_...xxxx) and only ever
   604→//   sends a fresh plaintext key when rotating.
   605→//
   606→// testStripeConnection: constructs a temporary Stripe client from
   607→//   the provided (or stored) credentials and pings Stripe's
   608→//   /v1/balance endpoint (read-only, no side effects). On success
   609→//   returns account info; on failure returns the Stripe error code
   610→//   + message. Records the outcome on the singleton row so the UI
   611→//   can show the last test result.
   612→// ============================================================

   614→export interface StripeSettingsInput {
   615→  mode: 'test' | 'live';
   616→  // Plaintext credentials from the admin UI. Empty string = "leave
   617→  // unchanged" (the existing encrypted value is kept); null/undefined
   618→  // = "clear" (the field is set to null).
   619→  secretKeyTest?: string;
   620→  secretKeyLive?: string;
   621→  publishableKeyTest?: string;
   622→  publishableKeyLive?: string;
   623→  webhookSecretTest?: string;
   624→  webhookSecretLive?: string;
   625→  appUrl?: string;
   626→}

   628→export interface StripeSettingsView {
   629→  mode: 'test' | 'live';
   630→  secretKeyTestMasked: string; // masked or '' when not configured
   631→  secretKeyLiveMasked: string;
   632→  publishableKeyTest: string; // plaintext (non-secret)
   633→  publishableKeyLive: string;
   634→  webhookSecretTestMasked: string;
   635→  webhookSecretLiveMasked: string;
   636→  appUrl: string;
   637→  lastTestStatus: 'success' | 'error' | null;
   638→  lastTestedAt: string | null;
   639→  lastTestErrorMessage: string | null;
   640→  // Whether each credential is configured (so the UI can show ✓/✗).
   641→  hasSecretKeyTest: boolean;
   642→  hasSecretKeyLive: boolean;
   643→  hasWebhookSecretTest: boolean;
   644→  hasWebhookSecretLive: boolean;
   645→  hasPublishableKeyTest: boolean;
   646→  hasPublishableKeyLive: boolean;
   647→  // The source the active Stripe client resolves to ('db' | 'env' | 'none').
   648→  activeSource: 'db' | 'env' | 'none';
   649→  isConfigured: boolean;
   650→}

   654→/** Read the StripeSettings singleton (or null when no row exists). */
   655→async function readStripeSettingsRow() {
   656→  const { db } = await import('@/lib/db');
   657→  return db.stripeSettings.findUnique({ where: { id: 'singleton' } });

   660→/** Decrypt a stored encrypted secret (returns null when ciphertext
   661→ *  is null or decryption fails). */
   662→async function tryDecrypt(ciphertext: string | null | undefined): Promise<string | null> {
   663→  if (!ciphertext) return null;
   664→  try {
   665→    return await decrypt(ciphertext);
   666→  } catch {
   667→    return null;
   668→  }
   669→}

   671→/** Build the safe masked view for the admin UI. Never returns raw
   672→ *  secret values. */
   673→export async function getStripeSettingsForAdmin(): Promise<StripeSettingsView> {
   674→  const row = await readStripeSettingsRow();
   675→
   676→  const secretKeyTest = await tryDecrypt(row?.secretKeyTestEncrypted);
   677→  const secretKeyLive = await tryDecrypt(row?.secretKeyLiveEncrypted);
   678→  const webhookSecretTest = await tryDecrypt(row?.webhookSecretTestEncrypted);
   679→  const webhookSecretLive = await tryDecrypt(row?.webhookSecretLiveEncrypted);
   680→  const publishableKeyTest = row?.publishableKeyTest ?? '';
   681→  const publishableKeyLive = row?.publishableKeyLive ?? '';
   682→
   683→  const resolved = await getResolvedStripeSettings();
   684→
   685→  return {
   686→    mode: (row?.mode === 'live' ? 'live' : 'test'),
   687→    secretKeyTestMasked: secretKeyTest ? maskSecret(secretKeyTest) : EMPTY_MASKED,
   688→    secretKeyLiveMasked: secretKeyLive ? maskSecret(secretKeyLive) : EMPTY_MASKED,
   689→    publishableKeyTest,
   690→    publishableKeyLive,
   691→    webhookSecretTestMasked: webhookSecretTest ? maskSecret(webhookSecretTest) : EMPTY_MASKED,
   692→    webhookSecretLiveMasked: webhookSecretLive ? maskSecret(webhookSecretLive) : EMPTY_MASKED,
   693→    appUrl: row?.appUrl ?? '',
   694→    lastTestStatus: (row?.lastTestStatus === 'success' || row?.lastTestStatus === 'error')
   695→      ? row.lastTestStatus
   696→      : null,
   697→    lastTestedAt: row?.lastTestedAt?.toISOString() ?? null,
   698→    lastTestErrorMessage: row?.lastTestErrorMessage ?? null,
   699→    hasSecretKeyTest: Boolean(secretKeyTest),
   700→    hasSecretKeyLive: Boolean(secretKeyLive),
   701→    hasWebhookSecretTest: Boolean(webhookSecretTest),
   702→    hasWebhookSecretLive: Boolean(webhookSecretLive),
   703→    hasPublishableKeyTest: Boolean(publishableKeyTest),
   704→    hasPublishableKeyLive: Boolean(publishableKeyLive),
   705→    activeSource: resolved.source,
   706→    isConfigured: Boolean(resolved.secretKey && resolved.secretKey.trim().length > 0),
   707→  };
   708→}

   710→/** Save (encrypt + upsert) Stripe credentials. Empty-string values
   711→ *  for secret keys mean "leave unchanged" (the existing encrypted
   712→ *  value is preserved); null/undefined means "clear". Publishable
   713→ *  keys are written as plaintext (non-secret). Returns the masked
   714→ *  view. Never returns raw secret values. */
   715→export async function saveStripeSettings(
   716→  input: StripeSettingsInput,
   717→  editorUserId: string,
   718→): Promise<StripeSettingsView> {
   719→  const { db } = await import('@/lib/db');
   720→  const existing = await readStripeSettingsRow();
   721→
   722→  // ---- Secret keys: encrypt when provided; preserve when empty string ----
   723→  const encryptOrPreserve = async (
   724→    plaintext: string | undefined,
   725→    existingEncrypted: string | null | undefined,
   726→  ): Promise<string | null> => {
   727→    if (plaintext === undefined || plaintext === null) return null; // clear
   728→    if (plaintext === '') return existingEncrypted ?? null; // preserve
   729→    return await encrypt(plaintext);
   730→  };
   731→
   732→  const secretKeyTestEncrypted = await encryptOrPreserve(
   733→    input.secretKeyTest,
   734→    existing?.secretKeyTestEncrypted,
   735→  );
   736→  const secretKeyLiveEncrypted = await encryptOrPreserve(
   737→    input.secretKeyLive,
   738→    existing?.secretKeyLiveEncrypted,
   739→  );
   740→  const webhookSecretTestEncrypted = await encryptOrPreserve(
   741→    input.webhookSecretTest,
   742→    existing?.webhookSecretTestEncrypted,
   743→  );
   744→  const webhookSecretLiveEncrypted = await encryptOrPreserve(
   745→    input.webhookSecretLive,
   746→    existing?.webhookSecretLiveEncrypted,
   747→  );
   748→
   749→  // ---- Publishable keys: plaintext (non-secret). Empty string
   750→  //      preserves existing; null clears. ----
   751→  const resolvePub = (v: string | undefined, existingVal: string | null): string | null => {
   752→    if (v === undefined || v === null) return null;
   753→    if (v === '') return existingVal ?? null;
   754→    return v;
   755→  };
   756→  const publishableKeyTest = resolvePub(input.publishableKeyTest, existing?.publishableKeyTest ?? null);
   757→  const publishableKeyLive = resolvePub(input.publishableKeyLive, existing?.publishableKeyLive ?? null);
   758→
   759→  // ---- App URL: empty string = unset (no preserve semantic — caller
   760→  //      should pass the full URL). ----
   761→  const appUrl = input.appUrl ?? existing?.appUrl ?? null;
   762→
   763→  await db.stripeSettings.upsert({
   764→    where: { id: 'singleton' },
   765→    create: {
   766→      id: 'singleton',
   767→      mode: input.mode,
   768→      secretKeyTestEncrypted,
   769→      secretKeyLiveEncrypted,
   770→      webhookSecretTestEncrypted,
   771→      webhookSecretLiveEncrypted,
   772→      publishableKeyTest,
   773→      publishableKeyLive,
   774→      appUrl,
   775→      updatedBy: editorUserId,
   776→    },
   777→    update: {
   778→      mode: input.mode,
   779→      secretKeyTestEncrypted,
   780→      secretKeyLiveEncrypted,
   781→      webhookSecretTestEncrypted,
   782→      webhookSecretLiveEncrypted,
   783→      publishableKeyTest,
   784→      publishableKeyLive,
   785→      appUrl,
   786→      updatedBy: editorUserId,
   787→    },
   788→  });
   789→
   790→  // Force the next read to pick up the new credentials.
   791→  invalidateStripeSettingsCache();
   792→
   793→  return getStripeSettingsForAdmin();
   794→}

   796→/** Test a Stripe connection by pinging the read-only /v1/balance
   797→ *  endpoint. Uses the currently-configured credentials when
   798→ *  `input` is null/undefined; otherwise constructs a temporary
   799→ *  client from `input` (so the admin can test BEFORE saving).
   800→ *  Records the outcome on the singleton row.
   801→ *
   802→ *  Returns `{ success: true, accountInfo }` on success, or
   803→ *  `{ success: false, code, message }` on failure. Never throws. */
   804→export async function testStripeConnection(
   805→  input?: StripeSettingsInput,
   806→): Promise<
   807→  | { success: true; mode: 'test' | 'live'; accountInfo: { id: string; type: string; country: string; email: string | null; displayName: string | null } }
   808→  | { success: false; mode: 'test' | 'live'; code: string; message: string }
   809→> {
   810→  // Resolve the secret key to test with.
   811→  let secretKey: string | null = null;
   812→  let mode: 'test' | 'live' = 'test';
   813→
   814→  if (input) {
   815→    mode = input.mode;
   816→    const candidate = mode === 'live' ? input.secretKeyLive : input.secretKeyTest;
   817→    if (candidate && candidate.trim().length > 0) {
   818→      secretKey = candidate;
   819→    } else {
   820→      // Fall back to the stored value for the chosen mode.
   821→      const row = await readStripeSettingsRow();
   822→      const enc = mode === 'live' ? row?.secretKeyLiveEncrypted : row?.secretKeyTestEncrypted;
   823→      secretKey = await tryDecrypt(enc);
   824→    }
   825→  } else {
   826→    const resolved = await getResolvedStripeSettings();
   827→    secretKey = resolved.secretKey;
   828→    mode = resolved.mode;
   829→  }
   830→
   831→  if (!secretKey || secretKey.trim().length === 0) {
   832→    const msg = 'No Stripe secret key is configured for the selected mode. Set the secret key first.';
   833→    await recordTestOutcome(mode, 'error', msg);
   834→    return { success: false, mode, code: 'NOT_CONFIGURED', message: msg };
   835→  }
   836→
   837→  // Basic key-shape validation — sk_test_ / sk_live_ / rk_test_ /
   838→  // rk_live_ (restricted keys also work for read-only calls).
   839→  const keyShapeOk =
   840→    secretKey.startsWith('sk_test_') ||
   841→    secretKey.startsWith('sk_live_') ||
   842→    secretKey.startsWith('rk_test_') ||
   843→    secretKey.startsWith('rk_live_');
   844→  if (!keyShapeOk) {
   845→    const msg = 'The secret key does not look like a Stripe secret key (expected sk_test_… or sk_live_…).';
   846→    await recordTestOutcome(mode, 'error', msg);
   847→    return { success: false, mode, code: 'INVALID_KEY_SHAPE', message: msg };
   848→  }
   849→
   850→  try {
   851→    // Construct a TEMPORARY client (not the cached singleton) so we
   852→    // can test credentials that haven't been saved yet.
   853→    const tempClient = new Stripe(secretKey, {
   854→      apiVersion: '2026-08-26.dahlia' as Stripe.LatestApiVersion,
   855→      typescript: true,
   856→      maxNetworkRetries: 1,
   857→    });
   858→    // /v1/balance is read-only and free of side effects — perfect for
   859→    // testing credentials. It also works with restricted keys that
   860→    // have the "Balance" read permission.
   861→    const balance = await tempClient.balance.retrieve();
   862→    // Resolve the account for richer info (id, country, display name).
   863→    // This call requires the "Settings" read permission, which secret
   864→    // keys always have. Fall back gracefully when restricted keys
   865→    // can't reach it.
   866→    let accountInfo: { id: string; type: string; country: string; email: string | null; displayName: string | null };
   867→    try {
   868→      const acct = await tempClient.accounts.retrieve();
   869→      accountInfo = {
   870→        id: acct.id,
   871→        type: (acct as unknown as { type?: string }).type ?? 'standard',
   872→        country: (acct as unknown as { country?: string }).country ?? '',
   873→        email: (acct as unknown as { email?: string | null }).email ?? null,
   874→        displayName: (acct as unknown as { display_name?: string | null }).display_name ?? null,
   875→      };
   876→    } catch {
   877→      // Restricted key with no Settings read — still a successful
   878→      // test (the secret key authenticates to Stripe).
   879→      accountInfo = {
   880→        id: '(restricted key)',
   881→        type: 'restricted',
   882→        country: '',
   883→        email: null,
   884→        displayName: null,
   885→      };
   886→    }
   887→    const balanceAvailable = balance.available?.[0]?.amount;
   888→    await recordTestOutcome(
   889→      mode,
   890→      'success',
   891→      null,
   892→      `account_id=${accountInfo.id}, available=${balanceAvailable ?? 'n/a'}`,
   893→    );
   894→    return { success: true, mode, accountInfo };
   895→  } catch (err: unknown) {
   896→    const e = (err ?? {}) as { code?: string; message?: string; type?: string };
   897→    const code = e.code ?? e.type ?? 'stripe_error';
   898→    const message = e.message ?? 'Unknown Stripe error';
   899→    await recordTestOutcome(mode, 'error', `${code}: ${message}`);
   900→    return { success: false, mode, code, message };
   901→  }
   902→}

   904→/** Persist the last test-connection outcome on the singleton row.
   905→ *  Best-effort — never throws. */
   906→async function recordTestOutcome(
   907→  mode: 'test' | 'live',
   908→  status: 'success' | 'error',
   909→  errorMessage: string | null,
   910→  _detail?: string,
   911→): Promise<void> {
   912→  try {
   913→    const { db } = await import('@/lib/db');
   914→    await db.stripeSettings.upsert({
   915→      where: { id: 'singleton' },
   916→      create: {
   917→        id: 'singleton',
   918→        mode,
   919→        lastTestStatus: status,
   920→        lastTestedAt: new Date(),
   921→        lastTestErrorMessage: errorMessage,
   922→      },
   923→      update: {
   924→        mode,
   925→        lastTestStatus: status,
   926→        lastTestedAt: new Date(),
   927→        lastTestErrorMessage: errorMessage,
   928→      },
   929→    });
   930→    invalidateStripeSettingsCache();
   931→  } catch {
   932→    // best-effort
   933→  }
   934→}
