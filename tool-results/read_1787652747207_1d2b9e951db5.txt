    50→// -------------------- Model Resolution Helper --------------------
    51→// The frontend sends DB cuids as `modelId` (e.g. "m-openai-gpt5"), but the
    52→// upstream provider APIs need the actual model string (e.g. "gpt-5").
    53→// This helper resolves a DB cuid to the AiModel row, validates ownership +
    54→// active status + type, and returns the model row. If no modelId is provided,
    55→// it falls back to AI Settings defaults, then the provider's default model.
    56→
    57→interface ResolvedModel {
    58→  modelId: string;        // upstream model string (e.g. "gpt-5")
    59→  modelDbId: string;      // DB cuid (e.g. "m-openai-gpt5")
    60→  inputCostPer1k: number | null;
    61→  outputCostPer1k: number | null;
    62→  type: string;           // 'TEXT' | 'IMAGE'
    63→}
    64→
    65→async function resolveModel(
    66→  providerId: string,
    67→  modelId: string | undefined,
    68→  expectedType: 'TEXT' | 'IMAGE',
    69→  providerModels: Array<{ id: string; modelId: string; isActive: boolean; isDefault: boolean; type: string; inputCostPer1k: number | null; outputCostPer1k: number | null }>,
    70→): Promise<ResolvedModel> {
    71→  // If a modelId is provided, it's a DB cuid — look it up
    72→  if (modelId) {
    73→    const model = providerModels.find((m) => m.id === modelId);
    74→    if (!model) {
    75→      throw new Error('The selected model was not found for this provider. Please select a valid model.');
    76→    }
    77→    if (model.providerId !== undefined && model.providerId !== providerId) {
    78→      throw new Error('The selected model does not belong to the selected provider.');
    79→    }
    80→    if (!model.isActive) {
    81→      throw new Error('The selected model is inactive. Please activate it or select another model.');
    82→    }
    83→    if (model.type?.toUpperCase() !== expectedType) {
    84→      throw new Error(`The selected model is not a ${expectedType.toLowerCase()} model. Please select a ${expectedType.toLowerCase()} model.`);
    85→    }
    86→    return {
    87→      modelId: model.modelId,
    88→      modelDbId: model.id,
    89→      inputCostPer1k: model.inputCostPer1k,
    90→      outputCostPer1k: model.outputCostPer1k,
    91→      type: model.type,
    92→    };
    93→  }
    94→
    95→  // No modelId provided — fall back to AI Settings defaults
    96→  const settings = await db.aiSettings.findUnique({ where: { scope: 'global' } });
    97→  const settingsModelId = expectedType === 'TEXT' ? settings?.defaultModelId : settings?.imageModelId;
    98→  if (settingsModelId) {
    99→    const model = providerModels.find((m) => m.id === settingsModelId && m.isActive && m.type?.toUpperCase() === expectedType);
   100→    if (model) {
   101→      return {
   102→        modelId: model.modelId,
   103→        modelDbId: model.id,
   104→        inputCostPer1k: model.inputCostPer1k,
   105→        outputCostPer1k: model.outputCostPer1k,
   106→        type: model.type,
   107→      };
   108→    }
   109→  }
   110→
   111→  // Fall back to the provider's default model of the correct type
   112→  const defaultModel = providerModels.find((m) => m.isActive && m.isDefault && m.type?.toUpperCase() === expectedType)
   113→    ?? providerModels.find((m) => m.isActive && m.type?.toUpperCase() === expectedType);
   114→  if (!defaultModel) {
   115→    throw new Error(`No active ${expectedType.toLowerCase()} model is configured for this provider. Please add or activate a model.`);
   116→  }
   117→  return {
   118→    modelId: defaultModel.modelId,
   119→    modelDbId: defaultModel.id,
   120→    inputCostPer1k: defaultModel.inputCostPer1k,
   121→    outputCostPer1k: defaultModel.outputCostPer1k,
   122→    type: defaultModel.type,
   123→  };
   124→}

   128→export async function executeChat(req: ChatRequest): Promise<ChatResponse> {
   129→  const provider = await db.aiProvider.findUnique({
   130→    where: { id: req.providerId },
   131→    include: { models: true },
   132→  });
   133→
   134→  if (!provider) throw new Error('Provider not found');
   135→  if (!provider.isActive) throw new Error('Provider is disabled. Please activate it first.');
   136→  if (!provider.apiKeyEncrypted) throw new Error('API key not configured for this provider.');
   137→
   138→  // Resolve + validate the model (handles DB cuid → model string, type=TEXT, active, belongs-to-provider)
   139→  const resolved = await resolveModel(req.providerId, req.modelId, 'TEXT', provider.models);
   140→  const modelId = resolved.modelId;
   141→
   142→  // Apply AI Settings defaults for temperature/maxTokens if not provided
   143→  const settings = await db.aiSettings.findUnique({ where: { scope: 'global' } });
   144→  const temperature = req.temperature ?? settings?.defaultTemperature ?? 0.7;
   145→  const maxTokens = req.maxTokens ?? settings?.defaultMaxTokens ?? 2048;
   146→
   147→  const apiKey = await decrypt(provider.apiKeyEncrypted);
   148→  const config = getProviderConfig(provider.kind);
   149→  const baseUrl = provider.baseUrl || config.defaultBaseUrl;
   150→  // CUSTOM providers have no defaultBaseUrl — they must have one set explicitly.
   151→  if (!baseUrl) {
   152→    throw new Error('No Base URL configured for this custom provider. Please edit the provider and set a Base URL.');
   153→  }
   154→
   155→  const startTime = Date.now();
   156→  let inputTokens = 0;
   157→  let outputTokens = 0;
   158→  let content = '';
   159→  let usedProvider = provider;
   160→  let usedModelId = modelId;
   161→
   162→  try {
   163→    if (provider.kind === 'ANTHROPIC') {
   164→      const result = await callAnthropic(baseUrl, apiKey, modelId, req.messages, {
   165→        temperature, maxTokens, jsonMode: req.jsonMode,
   166→      });
   167→      inputTokens = result.inputTokens;
   168→      outputTokens = result.outputTokens;
   169→      content = result.content;
   170→    } else if (provider.kind === 'GEMINI') {
   171→      const result = await callGemini(baseUrl, apiKey, modelId, req.messages, {
   172→        temperature, maxTokens, jsonMode: req.jsonMode,
   173→      });
   174→      inputTokens = result.inputTokens;
   175→      outputTokens = result.outputTokens;
   176→      content = result.content;
   177→    } else {
   178→      // OpenAI-compatible (OpenAI, Groq, DeepSeek, Custom)
   179→      const result = await callOpenAI(baseUrl, apiKey, modelId, req.messages, {
   180→        temperature, maxTokens,
   181→        topP: req.topP,
   182→        frequencyPenalty: req.frequencyPenalty,
   183→        presencePenalty: req.presencePenalty,
   184→        jsonMode: req.jsonMode,
   185→      });
   186→      inputTokens = result.inputTokens;
   187→      outputTokens = result.outputTokens;
   188→      content = result.content;
   189→    }
   190→  } catch (err) {
   191→    // Try fallback providers
   192→    const fallbacks = await db.aiProviderFallback.findMany({
   193→      where: { providerId: provider.id },
   194→      include: { fallback: { include: { models: true } } },
   195→      orderBy: { priority: 'asc' },
   196→    });
   197→
   198→    let lastError = err instanceof Error ? err : new Error('Unknown error');
   199→
   200→    for (const fb of fallbacks) {
   201→      if (!fb.fallback.isActive || !fb.fallback.apiKeyEncrypted) continue;
   202→      try {
   203→        const fbResolved = await resolveModel(fb.fallback.id, undefined, 'TEXT', fb.fallback.models);
   204→        const fbApiKey = await decrypt(fb.fallback.apiKeyEncrypted);
   205→        const fbConfig = getProviderConfig(fb.fallback.kind);
   206→        const fbBaseUrl = fb.fallback.baseUrl || fbConfig.defaultBaseUrl;
   207→        // CUSTOM providers have no defaultBaseUrl — skip if the admin hasn't set one.
   208→        if (!fbBaseUrl) continue;
   209→
   210→        let fbResult: { content: string; inputTokens: number; outputTokens: number };
   211→        if (fb.fallback.kind === 'ANTHROPIC') {
   212→          fbResult = await callAnthropic(fbBaseUrl, fbApiKey, fbResolved.modelId, req.messages, { temperature, maxTokens });
   213→        } else if (fb.fallback.kind === 'GEMINI') {
   214→          fbResult = await callGemini(fbBaseUrl, fbApiKey, fbResolved.modelId, req.messages, { temperature, maxTokens });
   215→        } else {
   216→          // OpenAI-compatible (OpenAI, Groq, DeepSeek, Custom)
   217→          fbResult = await callOpenAI(fbBaseUrl, fbApiKey, fbResolved.modelId, req.messages, { temperature, maxTokens });
   218→        }
   219→        inputTokens = fbResult.inputTokens;
   220→        outputTokens = fbResult.outputTokens;
   221→        content = fbResult.content;
   222→        usedProvider = fb.fallback;
   223→        usedModelId = fbResolved.modelId;
   224→        lastError = null as unknown as Error;
   225→        break;
   226→      } catch (fbErr) {
   227→        lastError = fbErr instanceof Error ? fbErr : new Error('Fallback failed');
   228→        continue;
   229→      }
   230→    }
   231→
   232→    if (!content) {
   233→      // Log the failed request
   234→      const durationMs = Date.now() - startTime;
   235→      await db.aiLog.create({
   236→        data: {
   237→          providerId: provider.id,
   238→          providerName: provider.name,
   239→          modelId,
   240→          question: req.messages.map((m) => m.content).join('\n'),
   241→          response: null,
   242→          inputTokens: 0,
   243→          outputTokens: 0,
   244→          totalTokens: 0,
   245→          costUsd: 0,
   246→          durationMs,
   247→          status: 'error',
   248→          errorMessage: lastError?.message ?? 'Unknown error',
   249→          siteId: req.siteId,
   250→          userId: req.userId,
   251→        },
   252→      }).catch(() => { /* logging failure shouldn't mask the original error */ });
   253→      throw lastError ?? new Error('Chat request failed');
   254→    }
   255→  }
   256→
   257→  const durationMs = Date.now() - startTime;
   258→  const totalTokens = inputTokens + outputTokens;
   259→  const costUsd = (inputTokens / 1000) * (resolved.inputCostPer1k || 0)
   260→    + (outputTokens / 1000) * (resolved.outputCostPer1k || 0);
   261→
   262→  // Update provider
   263→  await db.aiProvider.update({
   264→    where: { id: usedProvider.id },
   265→    data: {
   266→      lastUsedAt: new Date(),
   267→      latencyMs: durationMs,
   268→      connectionStatus: 'CONNECTED',
   269→      lastError: null,
   270→    },
   271→  });
   272→
   273→  // Log the request
   274→  await db.aiLog.create({
   275→    data: {
   276→      providerId: usedProvider.id,
   277→      providerName: usedProvider.name,
   278→      modelId: usedModelId,
   279→      question: req.messages.map((m) => m.content).join('\n'),
   280→      response: content,
   281→      inputTokens,
   282→      outputTokens,
   283→      totalTokens,
   284→      costUsd,
   285→      durationMs,
   286→      status: 'success',
   287→      siteId: req.siteId,
   288→      userId: req.userId,
   289→    },
   290→  });
   291→
   292→  return {
   293→    content,
   294→    model: usedModelId,
   295→    inputTokens,
   296→    outputTokens,
   297→    totalTokens,
   298→    costUsd,
   299→    durationMs,
   300→    providerName: usedProvider.name,
   301→  };
   302→}

   304→// -------------------- Provider-specific call implementations --------------------

   306→async function callOpenAI(
   307→  baseUrl: string,
   308→  apiKey: string,
   309→  model: string,
   310→  messages: ChatMessage[],
   311→  opts: { temperature?: number; maxTokens?: number; topP?: number; frequencyPenalty?: number; presencePenalty?: number; jsonMode?: boolean; apiVersion?: string } = {},
   312→) {
   313→  const url = opts.apiVersion
   314→    ? `${baseUrl}?api-version=${opts.apiVersion}`
   315→    : `${baseUrl}/chat/completions`;
   316→
   317→  const body: Record<string, unknown> = {
   318→    model,
   319→    messages,
   320→    temperature: opts.temperature ?? 0.7,
   321→    max_tokens: opts.maxTokens ?? 2048,
   322→  };
   323→  if (opts.topP !== undefined) body.top_p = opts.topP;
   324→  if (opts.frequencyPenalty !== undefined) body.frequency_penalty = opts.frequencyPenalty;
   325→  if (opts.presencePenalty !== undefined) body.presence_penalty = opts.presencePenalty;
   326→  if (opts.jsonMode) {
   327→    body.response_format = { type: 'json_object' };
   328→  }
   329→
   330→  const res = await fetch(url, {
   331→    method: 'POST',
   332→    headers: {
   333→      'Content-Type': 'application/json',
   334→      'Authorization': `Bearer ${apiKey}`,
   335→    },
   336→    body: JSON.stringify(body),
   337→  });
   338→
   339→  if (!res.ok) {
   340→    const err = await res.text();
   341→    throw new Error(`OpenAI API error: ${res.status} — ${err}`);
   342→  }
   343→
   344→  const data = await res.json();
   345→  return {
   346→    content: data.choices?.[0]?.message?.content || '',
   347→    inputTokens: data.usage?.prompt_tokens || 0,
   348→    outputTokens: data.usage?.completion_tokens || 0,
   349→  };
   350→}

   352→async function callAnthropic(
   353→  baseUrl: string,
   354→  apiKey: string,
   355→  model: string,
   356→  messages: ChatMessage[],
   357→  opts: { temperature?: number; maxTokens?: number; jsonMode?: boolean } = {},
   358→) {
   359→  const systemMsg = messages.find(m => m.role === 'system');
   360→  const chatMsgs = messages.filter(m => m.role !== 'system');
   361→
   362→  const body: Record<string, unknown> = {
   363→    model,
   364→    messages: chatMsgs.map(m => ({ role: m.role, content: m.content })),
   365→    max_tokens: opts.maxTokens ?? 2048,
   366→    temperature: opts.temperature ?? 0.7,
   367→  };
   368→  if (systemMsg) body.system = systemMsg.content;
   369→
   370→  const res = await fetch(`${baseUrl}/messages`, {
   371→    method: 'POST',
   372→    headers: {
   373→      'Content-Type': 'application/json',
   374→      'x-api-key': apiKey,
   375→      'anthropic-version': '2023-06-01',
   376→    },
   377→    body: JSON.stringify(body),
   378→  });
   379→
   380→  if (!res.ok) {
   381→    const err = await res.text();
   382→    throw new Error(`Anthropic API error: ${res.status} — ${err}`);
   383→  }
   384→
   385→  const data = await res.json();
   386→  return {
   387→    content: data.content?.[0]?.text || '',
   388→    inputTokens: data.usage?.input_tokens || 0,
   389→    outputTokens: data.usage?.output_tokens || 0,
   390→  };
   391→}

   393→async function callGemini(
   394→  baseUrl: string,
   395→  apiKey: string,
   396→  model: string,
   397→  messages: ChatMessage[],
   398→  opts: { temperature?: number; maxTokens?: number; jsonMode?: boolean } = {},
   399→) {
   400→  const systemMsg = messages.find(m => m.role === 'system');
   401→  const chatMsgs = messages.filter(m => m.role !== 'system');
   402→
   403→  const body: Record<string, unknown> = {
   404→    contents: chatMsgs.map(m => ({
   405→      role: m.role === 'assistant' ? 'model' : 'user',
   406→      parts: [{ text: m.content }],
   407→    })),
   408→    generationConfig: {
   409→      temperature: opts.temperature ?? 0.7,
   410→      maxOutputTokens: opts.maxTokens ?? 2048,
   411→    },
   412→  };
   413→
   414→  const res = await fetch(
   415→    `${baseUrl}/models/${model}:generateContent?key=${apiKey}`,
   416→    {
   417→      method: 'POST',
   418→      headers: { 'Content-Type': 'application/json' },
   419→      body: JSON.stringify(body),
   420→    },
   421→  );
   422→
   423→  if (!res.ok) {
   424→    const err = await res.text();
   425→    throw new Error(`Gemini API error: ${res.status} — ${err}`);
   426→  }
   427→
   428→  const data = await res.json();
   429→  return {
   430→    content: data.candidates?.[0]?.content?.parts?.[0]?.text || '',
   431→    inputTokens: data.usageMetadata?.promptTokenCount || 0,
   432→    outputTokens: data.usageMetadata?.candidatesTokenCount || 0,
   433→  };
   434→}

   438→export async function healthCheck(providerId: string): Promise<HealthCheckResult> {
   439→  const provider = await db.aiProvider.findUnique({
   440→    where: { id: providerId },
   441→    include: { models: { where: { isActive: true } } },
   442→  });
   443→  if (!provider) throw new Error('Provider not found');
   444→  if (!provider.apiKeyEncrypted) return { status: 'DISCONNECTED', latencyMs: 0, error: 'No API key configured' };
   445→
   446→  const apiKey = await decrypt(provider.apiKeyEncrypted);
   447→  const config = getProviderConfig(provider.kind);
   448→  const baseUrl = provider.baseUrl || config.defaultBaseUrl;
   449→  // CUSTOM providers have no defaultBaseUrl — they must have one set explicitly.
   450→  if (!baseUrl) {
   451→    return { status: 'DISCONNECTED', latencyMs: 0, error: 'No Base URL configured for this custom provider. Please edit the provider and set a Base URL.' };
   452→  }
   453→
   454→  const start = Date.now();
   455→  try {
   456→    let models: ProviderModel[] = [];
   457→
   458→    if (provider.kind === 'ANTHROPIC') {
   459→      // Anthropic has no /models endpoint — send a minimal chat request to verify the API key.
   460→      // Use the provider's first active TEXT model, or fall back to a known-good default.
   461→      const testModel = provider.models.find((m) => m.type?.toUpperCase() === 'TEXT')?.modelId
   462→        ?? config.defaultModels.find((m) => !isImageModelId(m.modelId))?.modelId
   463→        ?? 'claude-3-5-haiku-20241022';
   464→      const res = await fetch(`${baseUrl}/messages`, {
   465→        method: 'POST',
   466→        headers: { 'Content-Type': 'application/json', 'x-api-key': apiKey, 'anthropic-version': '2023-06-01' },
   467→        body: JSON.stringify({ model: testModel, max_tokens: 1, messages: [{ role: 'user', content: 'hi' }] }),
   468→      });
   469→      if (!res.ok) {
   470→        const errBody = await res.text().catch(() => '');
   471→        throw new Error(`HTTP ${res.status}${errBody ? `: ${errBody.slice(0, 200)}` : ''}`);
   472→      }
   473→      models = config.defaultModels;
   474→    } else if (provider.kind === 'GEMINI') {
   475→      const res = await fetch(`${baseUrl}/models?key=${apiKey}`);
   476→      if (!res.ok) {
   477→        const errBody = await res.text().catch(() => '');
   478→        throw new Error(`HTTP ${res.status}${errBody ? `: ${errBody.slice(0, 200)}` : ''}`);
   479→      }
   480→      models = config.defaultModels;
   481→    } else if (config.modelsEndpoint) {
   482→      // OpenAI-compatible (OpenAI, Groq, DeepSeek, Custom)
   483→      const res = await fetch(`${baseUrl}${config.modelsEndpoint}`, {
   484→        headers: { 'Authorization': `Bearer ${apiKey}` },
   485→      });
   486→      if (!res.ok) {
   487→        const errBody = await res.text().catch(() => '');
   488→        throw new Error(`HTTP ${res.status}${errBody ? `: ${errBody.slice(0, 200)}` : ''}`);
   489→      }
   490→      const data = await res.json();
   491→      if (data.data && Array.isArray(data.data)) {
   492→        models = data.data.map((m: { id: string }) => ({
   493→          modelId: m.id, name: m.id, contextLength: 0,
   494→          inputCostPer1k: 0, outputCostPer1k: 0,
   495→          supportsImages: false, supportsVision: false,
   496→          supportsFunctionCalling: false, supportsJsonMode: false,
   497→          supportsStreaming: true, supportsTools: false,
   498→        }));
   499→      } else {
   500→        models = config.defaultModels;
   501→      }
   502→    }
   503→
   504→    const latencyMs = Date.now() - start;
   505→
   506→    await db.aiProvider.update({
   507→      where: { id: providerId },
   508→      data: {
   509→        connectionStatus: 'CONNECTED',
   510→        latencyMs,
   511→        lastHealthCheckAt: new Date(),
   512→        lastError: null,
   513→      },
   514→    });
   515→
   516→    return { status: 'CONNECTED', latencyMs, availableModels: models };
   517→  } catch (err) {
   518→    const latencyMs = Date.now() - start;
   519→    const errorMsg = err instanceof Error ? err.message : 'Unknown error';
   520→
   521→    await db.aiProvider.update({
   522→      where: { id: providerId },
   523→      data: {
   524→        connectionStatus: 'ERROR',
   525→        latencyMs,
   526→        lastError: errorMsg,
   527→        lastHealthCheckAt: new Date(),
   528→      },
   529→    });
   530→
   531→    return { status: 'ERROR', latencyMs, error: errorMsg };
   532→  }
   533→}

   535→// -------------------- Sync Models --------------------
   536→
   537→export async function syncModels(providerId: string): Promise<number> {
   538→  const provider = await db.aiProvider.findUnique({ where: { id: providerId } });
   539→  if (!provider) throw new Error('Provider not found');
   540→  if (!provider.apiKeyEncrypted) throw new Error('No API key configured for this provider.');
   541→
   542→  const apiKey = await decrypt(provider.apiKeyEncrypted);
   543→  const config = getProviderConfig(provider.kind);
   544→  const baseUrl = provider.baseUrl || config.defaultBaseUrl;
   545→  // CUSTOM providers have no defaultBaseUrl — they must have one set explicitly.
   546→  if (!baseUrl) {
   547→    throw new Error('No Base URL configured for this custom provider. Please edit the provider and set a Base URL.');
   548→  }
   549→
   550→  let fetchedModels: ProviderModel[] = config.defaultModels;
   551→
   552→  // Try to fetch from API if endpoint exists (Anthropic has no /models endpoint)
   553→  if (config.modelsEndpoint && provider.kind !== 'ANTHROPIC') {
   554→    try {
   555→      const res = await fetch(`${baseUrl}${config.modelsEndpoint}`, {
   556→        headers: { 'Authorization': `Bearer ${apiKey}` },
   557→      });
   558→      if (res.ok) {
   559→        const data = await res.json();
   560→        if (data.data && Array.isArray(data.data)) {
   561→          const existingIds = new Set(config.defaultModels.map((m) => m.modelId));
   562→          const newModels = data.data
   563→            .filter((m: { id: string }) => !existingIds.has(m.id))
   564→            .map((m: { id: string }) => ({
   565→              modelId: m.id, name: m.id, contextLength: 0,
   566→              inputCostPer1k: 0, outputCostPer1k: 0,
   567→              supportsImages: false, supportsVision: false,
   568→              supportsFunctionCalling: false, supportsJsonMode: false,
   569→              supportsStreaming: true, supportsTools: false,
   570→            }));
   571→          fetchedModels = [...config.defaultModels, ...newModels];
   572→        }
   573→      }
   574→    } catch { /* use defaults */ }
   575→  }
   576→
   577→  // Determine the correct type for each model
   578→  const isImage = (mid: string) => isImageModelId(mid);
   579→
   580→  // Upsert models + set type correctly
   581→  let count = 0;
   582→  for (const model of fetchedModels) {
   583→    const modelType = isImage(model.modelId) ? 'IMAGE' : 'TEXT';
   584→    await db.aiModel.upsert({
   585→      where: { providerId_modelId: { providerId, modelId: model.modelId } },
   586→      update: {
   587→        name: model.name,
   588→        contextLength: model.contextLength,
   589→        inputCostPer1k: model.inputCostPer1k,
   590→        outputCostPer1k: model.outputCostPer1k,
   591→        supportsImages: model.supportsImages,
   592→        supportsVision: model.supportsVision,
   593→        supportsFunctionCalling: model.supportsFunctionCalling,
   594→        supportsJsonMode: model.supportsJsonMode,
   595→        supportsStreaming: model.supportsStreaming,
   596→        supportsTools: model.supportsTools,
   597→        type: modelType,
   598→        lastSyncedAt: new Date(),
   599→      },
   600→      create: {
   601→        providerId,
   602→        modelId: model.modelId,
   603→        name: model.name,
   604→        contextLength: model.contextLength,
   605→        inputCostPer1k: model.inputCostPer1k,
   606→        outputCostPer1k: model.outputCostPer1k,
   607→        supportsImages: model.supportsImages,
   608→        supportsVision: model.supportsVision,
   609→        supportsFunctionCalling: model.supportsFunctionCalling,
   610→        supportsJsonMode: model.supportsJsonMode,
   611→        supportsStreaming: model.supportsStreaming,
   612→        supportsTools: model.supportsTools,
   613→        type: modelType,
   614→        isActive: true,
   615→        lastSyncedAt: new Date(),
   616→      },
   617→    });
   618→    count++;
   619→  }
   620→
   621→  // Ensure there's a default TEXT and (if applicable) IMAGE model for this provider
   622→  const textDefault = await db.aiModel.findFirst({
   623→    where: { providerId, type: 'TEXT', isActive: true },
   624→    orderBy: { createdAt: 'asc' },
   625→  });
   626→  if (textDefault && !fetchedModels.some((m) => isImage(m.modelId))) {
   627→    // No image models in this sync — only ensure text default
   628→    const anyDefault = await db.aiModel.findFirst({ where: { providerId, isDefault: true } });
   629→    if (!anyDefault) {
   630→      await db.aiModel.update({ where: { id: textDefault.id }, data: { isDefault: true } });
   631→    }
   632→  }
   633→
   634→  await db.aiProvider.update({
   635→    where: { id: providerId },
   636→    data: { lastSyncAt: new Date() },
   637→  });
   638→
   639→  return count;
   640→}

   654→export async function getUsageAnalytics(siteId?: string, period: 'day' | 'week' | 'month' = 'month') {
   655→  const now = new Date();
   656→  const periodStart = new Date();
   657→  if (period === 'day') periodStart.setHours(now.getHours() - 24);
   658→  else if (period === 'week') periodStart.setDate(now.getDate() - 7);
   659→  else periodStart.setMonth(now.getMonth() - 1);
   660→
   661→  const where: Prisma.AiLogWhereInput = {
   662→    createdAt: { gte: periodStart },
   663→    ...(siteId ? { siteId } : {}),
   664→  };
   665→
   666→  const [totalCount, totalTokens, totalCost, errorCount, avgDuration] = await Promise.all([
   667→    db.aiLog.count({ where }),
   668→    db.aiLog.aggregate({ where, _sum: { totalTokens: true, inputTokens: true, outputTokens: true } }),
   669→    db.aiLog.aggregate({ where, _sum: { costUsd: true } }),
   670→    db.aiLog.count({ where: { ...where, status: 'error' } }),
   671→    db.aiLog.aggregate({ where: { ...where, status: 'success' }, _avg: { durationMs: true } }),
   672→  ]);
   673→
   674→  // Provider breakdown
   675→  const providerBreakdown = await db.aiLog.groupBy({
   676→    by: ['providerName'],
   677→    where,
   678→    _sum: { totalTokens: true, costUsd: true, inputTokens: true, outputTokens: true },
   679→    _count: true,
   680→    orderBy: { _sum: { costUsd: 'desc' } },
   681→  });
   682→
   683→  // Model breakdown
   684→  const modelBreakdown = await db.aiLog.groupBy({
   685→    by: ['modelId'],
   686→    where,
   687→    _sum: { totalTokens: true, costUsd: true },
   688→    _count: true,
   689→    orderBy: { _count: { id: 'desc' } },
   690→    take: 10,
   691→  });
   692→
   693→  // Daily usage chart
   694→  const dailyUsage = await db.$queryRawUnsafe<Array<{ date: string; count: number; tokens: number; cost: number }>>(`
   695→    SELECT DATE(createdAt) as date, COUNT(*) as count, COALESCE(SUM(totalTokens), 0) as tokens, COALESCE(SUM(costUsd), 0) as cost
   696→    FROM AiLog
   697→    WHERE createdAt >= ?
   698→    ${siteId ? 'AND siteId = ?' : ''}
   699→    GROUP BY DATE(createdAt)
   700→    ORDER BY date ASC
   701→  `, periodStart.toISOString(), ...(siteId ? [siteId] : []));
   702→
   703→  return {
   704→    totalRequests: totalCount,
   705→    totalTokens: totalTokens._sum.totalTokens || 0,
   706→    inputTokens: totalTokens._sum.inputTokens || 0,
   707→    outputTokens: totalTokens._sum.outputTokens || 0,
   708→    totalCost: totalCost._sum.costUsd || 0,
   709→    errorCount,
   710→    successRate: totalCount > 0 ? ((totalCount - errorCount) / totalCount) * 100 : 100,
   711→    avgDurationMs: Math.round(avgDuration._avg.durationMs || 0),
   712→    providerBreakdown,
   713→    modelBreakdown,
   714→    dailyUsage,
   715→  };
   716→}

   751→// Approximate cost per image (USD) for known models, keyed by model then size
   752→const IMAGE_MODEL_COSTS: Record<string, Record<string, number>> = {
   753→  'dall-e-2': { '256x256': 0.016, '512x512': 0.016, '1024x1024': 0.02 },
   754→  'dall-e-3': { '1024x1024': 0.040, '1792x1024': 0.080, '1024x1792': 0.080 },
   755→  'gpt-image-1': { '1024x1024': 0.040, '1536x1024': 0.080, '1024x1536': 0.080 },
   756→};
   757→
   758→function getImageCost(modelId: string, size: string): number {
   759→  const modelCosts = IMAGE_MODEL_COSTS[modelId];
   760→  if (modelCosts) {
   761→    return modelCosts[size] ?? modelCosts['1024x1024'] ?? 0.04;
   762→  }
   763→  // Fallback: use default estimate
   764→  return 0.04;
   765→}

   767→async function callOpenAIImageGeneration(
   768→  baseUrl: string,
   769→  apiKey: string,
   770→  model: string,
   771→  prompt: string,
   772→  opts: {
   773→    negativePrompt?: string;
   774→    size?: string;
   775→    quality?: string;
   776→    style?: string;
   777→    n?: number;
   778→    responseFormat?: string;
   779→    apiVersion?: string;
   780→  } = {},
   781→): Promise<{ images: GeneratedImage[] }> {
   782→  const url = opts.apiVersion
   783→    ? `${baseUrl}/images/generations?api-version=${opts.apiVersion}`
   784→    : `${baseUrl}/images/generations`;
   785→
   786→  const body: Record<string, unknown> = {
   787→    model,
   788→    prompt,
   789→    n: opts.n ?? 1,
   790→    size: opts.size ?? '1024x1024',
   791→    response_format: opts.responseFormat ?? 'url',
   792→  };
   793→
   794→  if (opts.quality) body.quality = opts.quality;
   795→  if (opts.style) body.style = opts.style;
   796→
   797→  const res = await fetch(url, {
   798→    method: 'POST',
   799→    headers: {
   800→      'Content-Type': 'application/json',
   801→      'Authorization': `Bearer ${apiKey}`,
   802→    },
   803→    body: JSON.stringify(body),
   804→  });
   805→
   806→  if (!res.ok) {
   807→    const err = await res.text();
   808→    throw new Error(`Image generation API error: ${res.status} — ${err}`);
   809→  }
   810→
   811→  const data = await res.json();
   812→  const images: GeneratedImage[] = (data.data ?? []).map((item: Record<string, unknown>) => ({
   813→    url: (item.url as string) ?? null,
   814→    base64: (item.b64_json as string) ?? null,
   815→    revisedPrompt: (item.revised_prompt as string) ?? null,
   816→  }));
   817→
   818→  return { images };
   819→}

   821→async function callGeminiImageGeneration(
   822→  baseUrl: string,
   823→  apiKey: string,
   824→  model: string,
   825→  prompt: string,
   826→  opts: {
   827→    negativePrompt?: string;
   828→    size?: string;
   829→    n?: number;
   830→    responseFormat?: string;
   831→  } = {},
   832→): Promise<{ images: GeneratedImage[] }> {
   833→  // Parse size to aspect ratio for Gemini Imagen
   834→  const sizeParts = (opts.size ?? '1024x1024').split('x');
   835→  const width = parseInt(sizeParts[0] ?? '1024', 10);
   836→  const height = parseInt(sizeParts[1] ?? '1024', 10);
   837→  let ratio: string;
   838→  if (Math.abs(width - height) < 100) {
   839→    ratio = '1:1'; // square
   840→  } else {
   841→    ratio = width >= height ? '16:9' : '9:16';
   842→  }
   843→
   844→  // Gemini uses Imagen via the generateContent endpoint with image generation config
   845→  // or the dedicated Imagen predict endpoint
   846→  const res = await fetch(
   847→    `${baseUrl}/models/${model}:generateContent?key=${apiKey}`,
   848→    {
   849→      method: 'POST',
   850→      headers: { 'Content-Type': 'application/json' },
   851→      body: JSON.stringify({
   852→        contents: [{ parts: [{ text: prompt }] }],
   853→        generationConfig: {
   854→          responseModalities: ['TEXT', 'IMAGE'],
   855→          // @ts-expect-error -- Gemini-specific field
   856→          imageGenerationConfig: {
   857→            numberOfImages: opts.n ?? 1,
   858→            aspectRatio: ratio,
   859→          },
   860→        },
   861→      }),
   862→    },
   863→  );
   864→
   865→  if (!res.ok) {
   866→    const err = await res.text();
   867→    throw new Error(`Gemini image generation error: ${res.status} — ${err}`);
   868→  }
   869→
   870→  const data = await res.json();
   871→  const images: GeneratedImage[] = [];
   872→
   873→  const parts = data.candidates?.[0]?.content?.parts ?? [];
   874→  for (const part of parts) {
   875→    if (part.inlineData) {
   876→      images.push({
   877→        url: null,
   878→        base64: part.inlineData.data ?? null,
   879→        revisedPrompt: prompt,
   880→      });
   881→    }
   882→  }
   883→
   884→  // Fallback: if no image parts returned, throw
   885→  if (images.length === 0) {
   886→    throw new Error('No images were generated by Gemini');
   887→  }
   888→
   889→  return { images };
   890→}

   892→export async function executeImageGeneration(req: ImageGenerationRequest): Promise<ImageGenerationResponse> {
   893→  const provider = await db.aiProvider.findUnique({
   894→    where: { id: req.providerId },
   895→    include: { models: true },
   896→  });
   897→
   898→  if (!provider) throw new Error('Provider not found');
   899→  if (!provider.isActive) throw new Error('Provider is disabled. Please activate it first.');
   900→  if (!provider.apiKeyEncrypted) throw new Error('API key not configured for this provider.');
   901→
   902→  // Resolve + validate the model (type must be IMAGE)
   903→  const resolved = await resolveModel(req.providerId, req.modelId, 'IMAGE', provider.models);
   904→  const modelId = resolved.modelId;
   905→
   906→  const apiKey = await decrypt(provider.apiKeyEncrypted);
   907→  const config = getProviderConfig(provider.kind);
   908→  const baseUrl = provider.baseUrl || config.defaultBaseUrl;
   909→  // CUSTOM providers have no defaultBaseUrl — they must have one set explicitly.
   910→  if (!baseUrl) {
   911→    throw new Error('No Base URL configured for this custom provider. Please edit the provider and set a Base URL.');
   912→  }
   913→
   914→  const startTime = Date.now();
   915→  let images: GeneratedImage[] = [];
   916→  let usedProvider = provider;
   917→  let usedModelId = modelId;
   918→
   919→  try {
   920→    if (provider.kind === 'GEMINI') {
   921→      const result = await callGeminiImageGeneration(baseUrl, apiKey, modelId, req.prompt, {
   922→        negativePrompt: req.negativePrompt,
   923→        size: req.size,
   924→        n: req.n,
   925→        responseFormat: req.responseFormat,
   926→      });
   927→      images = result.images;
   928→    } else if (provider.kind === 'OPENAI' || provider.kind === 'CUSTOM') {
   929→      // OpenAI and Custom (OpenAI-compatible) providers support image generation
   930→      const result = await callOpenAIImageGeneration(baseUrl, apiKey, modelId, req.prompt, {
   931→        negativePrompt: req.negativePrompt,
   932→        size: req.size,
   933→        quality: req.quality,
   934→        style: req.style,
   935→        n: req.n,
   936→        responseFormat: req.responseFormat,
   937→      });
   938→      images = result.images;
   939→    } else {
   940→      // GROQ and DEEPSEEK do not support image generation
   941→      throw new Error(`${config.name} does not support image generation. Please use OpenAI, Gemini, or a Custom OpenAI-compatible provider.`);
   942→    }
   943→  } catch (err) {
   944→    // Try fallback providers for image generation
   945→    const fallbacks = await db.aiProviderFallback.findMany({
   946→      where: { providerId: provider.id },
   947→      include: { fallback: { include: { models: true } } },
   948→      orderBy: { priority: 'asc' },
   949→    });
   950→
   951→    let lastError = err instanceof Error ? err : new Error('Unknown error');
   952→
   953→    for (const fb of fallbacks) {
   954→      if (!fb.fallback.isActive || !fb.fallback.apiKeyEncrypted) continue;
   955→      // Only try fallbacks that support image generation (OpenAI, Gemini, or Custom OpenAI-compatible)
   956→      if (fb.fallback.kind !== 'OPENAI' && fb.fallback.kind !== 'GEMINI' && fb.fallback.kind !== 'CUSTOM') continue;
   957→      try {
   958→        const fbResolved = await resolveModel(fb.fallback.id, undefined, 'IMAGE', fb.fallback.models);
   959→        const fbApiKey = await decrypt(fb.fallback.apiKeyEncrypted);
   960→        const fbConfig = getProviderConfig(fb.fallback.kind);
   961→        const fbBaseUrl = fb.fallback.baseUrl || fbConfig.defaultBaseUrl;
   962→        // CUSTOM providers have no defaultBaseUrl — skip if not set.
   963→        if (!fbBaseUrl) continue;
   964→
   965→        let fbResult: { images: GeneratedImage[] };
   966→        if (fb.fallback.kind === 'GEMINI') {
   967→          fbResult = await callGeminiImageGeneration(fbBaseUrl, fbApiKey, fbResolved.modelId, req.prompt, {
   968→            negativePrompt: req.negativePrompt, size: req.size, n: req.n, responseFormat: req.responseFormat,
   969→          });
   970→        } else {
   971→          // OpenAI or Custom (OpenAI-compatible)
   972→          fbResult = await callOpenAIImageGeneration(fbBaseUrl, fbApiKey, fbResolved.modelId, req.prompt, {
   973→            size: req.size, quality: req.quality, style: req.style, n: req.n, responseFormat: req.responseFormat,
   974→          });
   975→        }
   976→        images = fbResult.images;
   977→        usedProvider = fb.fallback;
   978→        usedModelId = fbResolved.modelId;
   979→        lastError = null as unknown as Error;
   980→        break;
   981→      } catch (fbErr) {
   982→        lastError = fbErr instanceof Error ? fbErr : new Error('Fallback failed');
   983→        continue;
   984→      }
   985→    }
   986→
   987→    if (images.length === 0) {
   988→      // Log the failed request
   989→      const durationMs = Date.now() - startTime;
   990→      await db.aiLog.create({
   991→        data: {
   992→          providerId: provider.id,
   993→          providerName: provider.name,
   994→          modelId,
   995→          question: `[IMAGE] ${req.prompt}`,
   996→          response: null,
   997→          inputTokens: 0,
   998→          outputTokens: 0,
   999→          totalTokens: 0,
  1000→          costUsd: 0,
  1001→          durationMs,
  1002→          status: 'error',
  1003→          errorMessage: lastError?.message ?? 'Unknown error',
  1004→          siteId: req.siteId,
  1005→          userId: req.userId,
  1006→        },
  1007→      }).catch(() => { /* logging failure shouldn't mask the original error */ });
  1008→      throw lastError ?? new Error('Image generation failed');
  1009→    }
  1010→  }
  1011→
  1012→  const durationMs = Date.now() - startTime;
  1013→  const size = req.size ?? '1024x1024';
  1014→  const n = images.length;
  1015→  const costUsd = n * getImageCost(usedModelId, size);
  1016→
  1017→  // Update provider stats
  1018→  await db.aiProvider.update({
  1019→    where: { id: usedProvider.id },
  1020→    data: {
  1021→      lastUsedAt: new Date(),
  1022→      latencyMs: durationMs,
  1023→      connectionStatus: 'CONNECTED',
  1024→      lastError: null,
  1025→    },
  1026→  });
  1027→
  1028→  // Log the request
  1029→  await db.aiLog.create({
  1030→    data: {
  1031→      providerId: usedProvider.id,
  1032→      providerName: usedProvider.name,
  1033→      modelId: usedModelId,
  1034→      question: `[IMAGE] ${req.prompt}`,
  1035→      response: JSON.stringify({
  1036→        imagesGenerated: n,
  1037→        size,
  1038→        format: req.responseFormat ?? 'url',
  1039→        model: usedModelId,
  1040→      }),
  1041→      inputTokens: 0,
  1042→      outputTokens: 0,
  1043→      totalTokens: 0,
  1044→      costUsd,
  1045→      durationMs,
  1046→      status: 'success',
  1047→      siteId: req.siteId,
  1048→      userId: req.userId,
  1049→    },
  1050→  });
  1051→
  1052→  return {
  1053→    images,
  1054→    model: usedModelId,
  1055→    costUsd,
  1056→    durationMs,
  1057→    providerName: usedProvider.name,
  1058→    providerKind: usedProvider.kind,
  1059→  };
  1060→}
