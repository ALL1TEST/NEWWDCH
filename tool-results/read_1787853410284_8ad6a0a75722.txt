   167→// -------------------- Comment Text (Read More / Read Less) --------------------
   168→// Measures the actual rendered height of the paragraph against its
   169→// truncated (2-line) height. If scrollHeight > clientHeight, the text
   170→// IS being truncated and "Read More" is shown. This is based on the
   171→// actual rendered content, not an arbitrary character threshold.
   172→
   173→function CommentText({
   174→  content,
   175→  commentId,
   176→  isExpanded,
   177→  onToggle,
   178→}: {
   179→  content: string;
   180→  commentId: string;
   181→  isExpanded: boolean;
   182→  onToggle: (id: string) => void;
   183→}) {
   184→  const paraRef = useRef<HTMLParagraphElement>(null);
   185→  const [isTruncated, setIsTruncated] = useState(false);
   186→
   187→  useEffect(() => {
   188→    const el = paraRef.current;
   189→    if (!el) return;
   190→
   191→    const check = () => {
   192→      if (isExpanded) {
   193→        // When expanded, always show "Read Less"
   194→        setIsTruncated(true);
   195→        return;
   196→      }
   197→      // Compare scrollHeight (full content) vs clientHeight (visible area).
   198→      // If scrollHeight > clientHeight, the text IS being clipped by
   199→      // line-clamp-2 — "Read More" should appear.
   200→      const isClipped = el.scrollHeight > el.clientHeight + 2;
   201→      setIsTruncated(isClipped);
   202→    };
   203→
   204→    // Use requestAnimationFrame to ensure the DOM has settled
   205→    const raf = requestAnimationFrame(check);
   206→
   207→    // Re-check on resize (container width may change)
   208→    const observer = new ResizeObserver(() => requestAnimationFrame(check));
   209→    observer.observe(el);
   210→
   211→    return () => {
   212→      cancelAnimationFrame(raf);
   213→      observer.disconnect();
   214→    };
   215→  }, [content, isExpanded]);
   216→
   217→  return (
   218→    <>
   219→      <p
   220→        ref={paraRef}
   221→        className={cn(
   222→          'text-sm text-foreground/90 mt-1 leading-relaxed',
   223→          !isExpanded && 'line-clamp-2',
   224→        )}
   225→      >
   226→        {content}
   227→      </p>
   228→      {isTruncated && (
   229→        <button
   230→          type="button"
   231→          className="text-xs text-primary hover:underline mt-0.5"
   232→          onClick={() => onToggle(commentId)}
   233→        >
   234→          {isExpanded ? 'Read Less' : 'Read More'}
   235→        </button>
   236→      )}
   237→    </>
   238→  );
   239→}

   272→const API_KEY_MASK = '••••••••';
   273→
   274→interface ProviderDraft {
   275→  commentsEnabled?: boolean;
   276→  spamDetection?: boolean;
   277→  spamProvider?: string;
   278→  // Custom provider fields
   279→  customProviderName?: string;
   280→  customApiEndpoint?: string;

   287→
   288→function CommentSettingsCard() {
   289→  const queryClient = useQueryClient();
   290→
   291→  const { data: settingsData, isLoading } = useQuery({
   292→    queryKey: ['settings', 'discussion', 'comments-page'],
   293→    queryFn: () => getApi<Record<string, string> | null>('/api/settings?category=DISCUSSION'),
   294→    staleTime: 10_000,
   295→  });
   296→
   297→  // Hydrate from saved settings. API keys come back as '[ENCRYPTED]' when
   298→  // a value is stored — we show the mask placeholder instead of the real
   299→  // secret. Empty string means no key has been saved yet.
   300→  const savedCommentsEnabled = settingsData?.enable_comments !== 'false';
   301→  const savedSpamDetection = settingsData?.comment_auto_spam_detection === 'true';
   302→  const savedSpamProvider = settingsData?.comment_spam_provider || 'none';
   303→
   304→  const savedCustomProviderName = settingsData?.comment_spam_provider_name || '';
   305→  const savedCustomApiEndpoint = settingsData?.comment_spam_api_endpoint || '';
   306→  const hasSavedCustomApiKey = settingsData?.comment_spam_api_key && settingsData.comment_spam_api_key !== '';
   307→  const savedCustomEnabled = settingsData?.comment_spam_enabled !== 'false';
   308→
   309→  const savedAkismetBlogUrl = settingsData?.akismet_blog_url || '';
   310→  const hasSavedAkismetApiKey = settingsData?.akismet_api_key && settingsData.akismet_api_key !== '';
   311→
   312→  const [draft, setDraft] = useState<ProviderDraft>({});
   313→
   314→  const commentsEnabled = draft.commentsEnabled ?? savedCommentsEnabled;
   315→  const spamDetection = draft.spamDetection ?? savedSpamDetection;
   316→  const spamProvider = draft.spamProvider ?? savedSpamProvider;
   317→
   318→  // Custom-provider draft values (fall back to saved).
   319→  const customProviderName = draft.customProviderName ?? savedCustomProviderName;
   320→  const customApiEndpoint = draft.customApiEndpoint ?? savedCustomApiEndpoint;
   321→  // API key: show mask when a key is saved AND the user hasn't typed a new one.
   322→  const customApiKey = draft.customApiKey ?? (hasSavedCustomApiKey ? API_KEY_MASK : '');
   323→  const customEnabled = draft.customEnabled ?? savedCustomEnabled;
   324→
   325→  // Akismet draft values.
   326→  const akismetApiKey = draft.akismetApiKey ?? (hasSavedAkismetApiKey ? API_KEY_MASK : '');
   327→  const akismetBlogUrl = draft.akismetBlogUrl ?? savedAkismetBlogUrl;
   328→
   329→  const isDirty = Object.keys(draft).length > 0;
   330→
   331→  const saveMutation = useMutation({
   332→    mutationFn: () => {
   333→      const settings: Array<{ key: string; value: string; type?: string; category: string }> = [
   334→        { key: 'enable_comments', value: String(commentsEnabled), type: 'BOOLEAN', category: 'DISCUSSION' },
   335→        { key: 'comment_auto_spam_detection', value: String(spamDetection), type: 'BOOLEAN', category: 'DISCUSSION' },
   336→        { key: 'comment_spam_provider', value: spamProvider, type: 'STRING', category: 'DISCUSSION' },
   337→      ];
   338→
   339→      // Custom provider config — only include the API key when the user
   340→      // typed something OTHER than the mask placeholder (so submitting the
   341→      // mask preserves the existing stored secret).
   342→      if (spamProvider === 'custom') {
   343→        settings.push(
   344→          { key: 'comment_spam_provider_name', value: customProviderName, type: 'STRING', category: 'DISCUSSION' },
   345→          { key: 'comment_spam_api_endpoint', value: customApiEndpoint, type: 'URL', category: 'DISCUSSION' },
   346→          { key: 'comment_spam_enabled', value: String(customEnabled), type: 'BOOLEAN', category: 'DISCUSSION' },
   347→        );
   348→        if (customApiKey && customApiKey !== API_KEY_MASK) {
   349→          settings.push({ key: 'comment_spam_api_key', value: customApiKey, type: 'ENCRYPTED', category: 'DISCUSSION' });
   350→        }
   351→      }
   352→
   353→      // Akismet config — same mask-preservation logic.
   354→      if (spamProvider === 'akismet') {
   355→        settings.push(
   356→          { key: 'akismet_blog_url', value: akismetBlogUrl, type: 'URL', category: 'DISCUSSION' },
   357→        );
   358→        if (akismetApiKey && akismetApiKey !== API_KEY_MASK) {
   359→          settings.push({ key: 'akismet_api_key', value: akismetApiKey, type: 'ENCRYPTED', category: 'DISCUSSION' });
   360→        }
   361→      }
   362→
   363→      return postApi('/api/settings', { settings });
   364→    },
   365→    onSuccess: () => {
   366→      setDraft({});
   367→      queryClient.invalidateQueries({ queryKey: ['settings'] });
   368→      toast.success('Comment settings saved');
   369→    },
   370→    onError: (err: Error) => toast.error(err.message || 'Failed to save settings'),
   371→  });
   372→
   373→  if (isLoading) {
   374→    return (
   375→      <Card className="p-4">
   376→        <div className="flex items-center gap-3">
   377→          <Skeleton className="h-6 w-32" />
   378→          <Skeleton className="h-6 w-40" />
   379→          <Skeleton className="h-6 w-48" />
   380→        </div>
   381→      </Card>
   382→    );
   383→  }
   384→
   385→  return (
   386→    <Card className="p-4">
   387→      {/* Top row — global comment settings */}
   388→      <div className="flex flex-wrap items-center gap-x-6 gap-y-3">
   389→        {/* Enable Comments */}
   390→        <div className="flex items-center gap-2">
   391→          <Switch
   392→            id="enable-comments"
   393→            checked={commentsEnabled}
   394→            onCheckedChange={(v) => setDraft((prev) => ({ ...prev, commentsEnabled: v }))}
   395→          />
   396→          <Label htmlFor="enable-comments" className="text-sm font-medium cursor-pointer">
   397→            Enable Comments
   398→          </Label>
   399→        </div>
   400→
   401→        <Separator orientation="vertical" className="h-6 hidden sm:block" />
   402→
   403→        {/* Auto Spam Detection */}
   404→        <div className="flex items-center gap-2">
   405→          <Shield className="h-3.5 w-3.5 text-muted-foreground" />
   406→          <Switch
   407→            id="spam-detection"
   408→            checked={spamDetection}
   409→            onCheckedChange={(v) => setDraft((prev) => ({ ...prev, spamDetection: v }))}
   410→          />
   411→          <Label htmlFor="spam-detection" className="text-sm font-medium cursor-pointer">
   412→            Auto Spam Detection
   413→          </Label>
   414→        </div>
   415→
   416→        {/* Spam Provider (conditional on spam detection being on) */}
   417→        {spamDetection && (
   418→          <div className="flex items-center gap-2">
   419→            <Label className="text-sm text-muted-foreground">Provider:</Label>
   420→            <Select
   421→              value={spamProvider}
   422→              onValueChange={(v) => setDraft((prev) => ({ ...prev, spamProvider: v }))}

   424→              <SelectTrigger size="sm" className="w-[120px] h-8">
   425→                <SelectValue />

   428→                {SPAM_PROVIDERS.map((p) => (
   429→                  <SelectItem key={p.value} value={p.value}>

   434→            </Select>
   435→          </div>

   437→
   438→        {/* Save button — only visible when there are unsaved changes */}
   439→        <Button
   440→          size="sm"
   441→          className={cn('ml-auto', !isDirty && 'invisible')}
   442→          onClick={() => saveMutation.mutate()}
   443→          disabled={saveMutation.isPending}

   445→          {saveMutation.isPending ? (

   451→
   452→      {/* ---------- Provider configuration sections ---------- */}

   454→      {/* Custom provider config — only when Provider = Custom */}

   457→          <div className="flex items-center justify-between mb-3">

   460→              {/* Helper text as a small info tooltip to keep the UI clean */}
   461→              <Tooltip>

   463→                  <button type="button" className="text-muted-foreground/60 hover:text-muted-foreground transition-colors" onClick={(e) => e.preventDefault()}>
