    90→/** Resolve the badge id for the shared PlanBadge component. Falls
    91→ *  back to 'free' for any unknown plan id so future plans still
    92→ *  render with a sensible neutral badge. */
    93→function getPlanBadgeId(planId: string): 'free' | 'plus' | 'pro' | 'max' {
    94→  if (planId === 'free' || planId === 'plus' || planId === 'pro' || planId === 'max') return planId;

    96→  if (planId === 'beta') return 'plus';
    97→  return 'free';
    98→}

   100→// -------------------- Price formatting helpers --------------------
   101→
   102→/** Display symbol for a currency code. Plans & Pricing shows a
   103→ *  symbol (not the ISO text code) per the visual spec — e.g. "$7.50"
   104→ *  not "USD 7.50" / "CHF 7.50". The stored currency code in the DB is
   105→ *  unaffected; this is a display-only mapping. Unknown codes default
   106→ *  to '$' so the price always renders as a compact symbol + number. */
   107→function currencySymbol(code: string): string {
   108→  const map: Record<string, string> = {
   109→    USD: '$',
   110→    EUR: '€',
   111→    GBP: '£',
   112→    JPY: '¥',
   113→    CNY: '¥',
   114→  };
   115→  return map[(code ?? '').toUpperCase()] ?? '$';
   116→}
   117→
   118→/** Format an integer amount with the display symbol, no decimals.
   119→ *  e.g. (90, 'USD') → "$90"; (0, 'CHF') → "$0". */
   120→function formatPriceSymbol(amount: number, currency: string): string {
   121→  return `${currencySymbol(currency)}${amount.toLocaleString(undefined, {
   122→    minimumFractionDigits: 0,
   123→    maximumFractionDigits: 0,
   124→  })}`;
   125→}
   126→
   127→/** Format with 2 decimals — used for the monthly-equivalent of a
   128→ *  yearly price (yearly 90 → "$7.50"). Always two decimals so the
   129→ *  large primary price reads "$7.50" / "$40.83" / "$82.50". */
   130→function formatPriceSymbolMonthlyEquiv(amount: number, currency: string): string {
   131→  return `${currencySymbol(currency)}${amount.toLocaleString(undefined, {
   132→    minimumFractionDigits: 2,
   133→    maximumFractionDigits: 2,
   134→  })}`;
   135→}

   301→          <ul className="space-y-2.5">

   331→            <Pencil className="h-4 w-4 mr-2" />

   363→// -------------------- Edit Plan Dialog --------------------
   364→
   365→function EditPlanDialog({
   366→  plan,
   367→  open,
   368→  onOpenChange,
   369→}: {
   370→  plan: PlanConfigData;
   371→  open: boolean;
   372→  onOpenChange: (open: boolean) => void;
   373→}) {
   374→  const queryClient = useQueryClient();
   375→  const [name, setName] = useState(plan.name);
   376→  const [priceMonthly, setPriceMonthly] = useState(String(plan.priceMonthly));
   377→  const [priceYearly, setPriceYearly] = useState(String(plan.priceYearly));
   378→  // Currency is NOT admin-editable per-plan — it is derived from the
   379→  // platform's default country (CountryPricing). We display it as
   380→  // read-only so the admin knows what currency the prices are in, but
   381→  // never send it in the patch (the backend keeps the plan's stored
   382→  // currency). This keeps currency handling consistent between the
   383→  // plan, checkout, client billing page, and Stripe.
   384→  const currency = plan.currency;
   385→  const [interval, setInterval] = useState<'monthly' | 'yearly'>(plan.interval);
   386→  const [active, setActive] = useState(plan.active);
   387→  const [features, setFeatures] = useState(plan.features.join('\n'));
   388→  const [entitlements, setEntitlements] = useState<string[]>(plan.entitlements);
   389→  const [limits, setLimits] = useState<PlanLimits>(plan.limits);
   390→  // New (Task 64): free-trial duration + Stripe Price IDs.
   391→  const [freePlanDurationDays, setFreePlanDurationDays] = useState<string>(
   392→    plan.freePlanDurationDays == null ? '' : String(plan.freePlanDurationDays),
   393→  );
   394→  const [stripePriceIdMonthly, setStripePriceIdMonthly] = useState(plan.stripePriceIdMonthly ?? '');
   395→  const [stripePriceIdYearly, setStripePriceIdYearly] = useState(plan.stripePriceIdYearly ?? '');
   396→  const [clientDisplayOpen, setClientDisplayOpen] = useState(false);
   397→  const [stripeOpen, setStripeOpen] = useState(false);
   398→
   399→  // The EditPlanDialog is conditionally rendered by the parent (mounted
   400→  // fresh each time the user opens it), so the useState initializers
   401→  // above already seed local state from the latest server snapshot.
   402→  // No useEffect sync is needed — that would just trigger cascading
   403→  // renders (and the React Compiler correctly flags it).
   404→
   405→  const reset = () => {
   406→    setName(plan.name);
   407→    setPriceMonthly(String(plan.priceMonthly));
   408→    setPriceYearly(String(plan.priceYearly));
   409→    setInterval(plan.interval);
   410→    setActive(plan.active);
   411→    setFeatures(plan.features.join('\n'));
   412→    setEntitlements(plan.entitlements);
   413→    setLimits(plan.limits);
   414→    setFreePlanDurationDays(plan.freePlanDurationDays == null ? '' : String(plan.freePlanDurationDays));
   415→    setStripePriceIdMonthly(plan.stripePriceIdMonthly ?? '');
   416→    setStripePriceIdYearly(plan.stripePriceIdYearly ?? '');
   417→  };
   418→
   419→  const saveMutation = useMutation({
   420→    mutationFn: (patch: PlanPatch) =>
   421→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, patch),
   422→    onSuccess: (data) => {
   423→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   424→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   425→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   426→      toast.success(`${data.name} updated — changes are now live for clients.`);
   427→      onOpenChange(false);
   428→    },
   429→    onError: (err: unknown) => {
   430→      const e = err as { error?: { message?: string }; message?: string };
   431→      toast.error(
   432→        e?.error?.message ?? e?.message ?? 'Unable to save plan. You may need OWNER access.',
   433→      );
   434→    },
   435→  });
   436→
   437→  // Sync this plan to Stripe (creates/reuses the Stripe Product + monthly +
   438→  // yearly Prices and writes the resolved Stripe Price IDs back onto the
   439→  // plan row). Surfaces Stripe errors inline so the admin can fix them.
   440→  const syncToStripeMutation = useMutation({
   441→    mutationFn: () =>
   442→      postApi<{
   443→        planId: string;
   444→        stripePriceIdMonthly: string;
   445→        stripePriceIdYearly: string;
   446→        created: boolean;
   447→      }>(`/api/platform/admin/plans/${plan.planId}/sync-stripe`),
   448→    onSuccess: (data) => {
   449→      // Reflect the synced Stripe Price IDs in the local form state so the
   450→      // admin can see them without re-opening the dialog.
   451→      setStripePriceIdMonthly(data.stripePriceIdMonthly);
   452→      setStripePriceIdYearly(data.stripePriceIdYearly);
   453→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   454→      toast.success(
   455→        data.created
   456→          ? `Synced to Stripe — created monthly (${data.stripePriceIdMonthly.slice(0, 14)}…) + yearly (${data.stripePriceIdYearly.slice(0, 14)}…) Prices.`
   457→          : 'Stripe Prices are already in sync.',
   458→      );
   459→    },
   460→    onError: (err: unknown) => {
   461→      const e = err as { code?: string; message?: string; error?: { code?: string; message?: string } };
   462→      const code = e?.code ?? e?.error?.code;
   463→      const msg = e?.message ?? e?.error?.message;
   464→      if (code === 'PAYMENT_PROVIDER_NOT_CONFIGURED') {
   465→        toast.error('Stripe is not connected. Configure credentials in Platform Admin → Stripe Settings first.');
   466→      } else if (code === 'VALIDATION_ERROR') {
   467→        toast.error(msg || 'Free plans do not need Stripe — set a price above 0 first.');
   468→      } else {
   469→        toast.error(msg || 'Unable to sync plan to Stripe.');
   470→      }
   471→    },
   472→  });
   473→
   474→  const buildPatch = (): PlanPatch => {
   475→    const monthly = Number(priceMonthly) || 0;
   476→    const yearly = Number(priceYearly) || 0;
   477→    const isFreeDerived = monthly === 0 && yearly === 0;
   478→    return {
   479→      name,
   480→      priceMonthly: monthly,
   481→      priceYearly: yearly,
   482→      // currency is intentionally NOT sent — the backend keeps the plan's
   483→      // stored currency (derived from the platform default country). The
   484→      // admin never manually chooses currency per plan.
   485→      interval,
   486→      isFree: isFreeDerived,
   487→      freePlanDurationDays: isFreeDerived ? (freePlanDurationDays.trim() === '' ? null : Number(freePlanDurationDays) || null) : null,
   488→      stripePriceIdMonthly: stripePriceIdMonthly.trim() === '' ? null : stripePriceIdMonthly.trim(),
   489→      stripePriceIdYearly: stripePriceIdYearly.trim() === '' ? null : stripePriceIdYearly.trim(),
