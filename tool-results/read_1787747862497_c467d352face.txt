    58→// ==================== Validation ====================
    59→
    60→function validateRobots(content: string): ValidationWarning[] {
    61→  const warnings: ValidationWarning[] = [];
    62→  const lines = content.split('\n');
    63→
    64→  if (!content.trim()) {
    65→    warnings.push({ type: 'error', message: 'Robots.txt content is empty', line: 1 });
    66→    return warnings;
    67→  }
    68→
    69→  const hasUserAgent = lines.some((line) =>
    70→    line.toLowerCase().trimStart().startsWith('user-agent:'),
    71→  );
    72→  if (!hasUserAgent) {
    73→    warnings.push({ type: 'error', message: 'No "User-agent:" directive found — crawlers may ignore your rules' });
    74→  }
    75→
    76→  // Check Sitemap URL validity
    77→  lines.forEach((line, idx) => {
    78→    if (line.toLowerCase().trimStart().startsWith('sitemap:')) {
    79→      const url = line.slice(line.indexOf(':') + 1).trim();
    80→      if (url && !url.startsWith('http://') && !url.startsWith('https://')) {
    81→        warnings.push({ type: 'warning', message: `Sitemap URL should start with http:// or https://: "${url}"`, line: idx + 1 });
    82→      }
    83→    }
    84→  });
    85→
    86→  // Check for dangerous Disallow: / in User-agent: * groups
    87→  let inWildcardGroup = false;
    88→  let lastLineWasUserAgent = false;
    89→  lines.forEach((line, idx) => {
    90→    const trimmed = line.trim();
    91→    if (!trimmed || trimmed.startsWith('#')) return;
    92→    const colonIdx = trimmed.indexOf(':');
    93→    if (colonIdx === -1) return;
    94→    const directive = trimmed.slice(0, colonIdx).trim().toLowerCase();
    95→    const value = trimmed.slice(colonIdx + 1).trim();
    96→
    97→    if (directive === 'user-agent') {
    98→      if (lastLineWasUserAgent) {
    99→        // continuation of multi-agent group
   100→      } else {
   101→        inWildcardGroup = value === '*';
   102→      }
   103→      lastLineWasUserAgent = true;
   104→    } else {
   105→      lastLineWasUserAgent = false;
   106→      if (directive === 'disallow' && value === '/' && inWildcardGroup) {
   107→        warnings.push({
   108→          type: 'error',
   109→          message: 'This rule blocks ALL crawlers from accessing the entire website. Search engines will not crawl or index any pages.',
   110→          line: idx + 1,
   111→        });
   112→      }
   113→    }
   114→  });
   115→
   116→  // Parse into groups for conflict detection
   117→  interface RobotGroup { agents: string[]; allowPaths: string[]; disallowPaths: string[]; }
   118→  const groups: RobotGroup[] = [];
   119→  let currentGroup: RobotGroup | null = null;
   120→  lastLineWasUserAgent = false;
   121→
   122→  for (const line of lines) {
   123→    const trimmed = line.trim();
   124→    if (!trimmed || trimmed.startsWith('#')) continue;
   125→    const colonIdx = trimmed.indexOf(':');
   126→    if (colonIdx === -1) continue;
   127→    const directive = trimmed.slice(0, colonIdx).trim().toLowerCase();
   128→    const value = trimmed.slice(colonIdx + 1).trim();
   129→
   130→    if (directive === 'user-agent') {
   131→      if (lastLineWasUserAgent && currentGroup) {
   132→        currentGroup.agents.push(value.toLowerCase());
   133→      } else {
   134→        currentGroup = { agents: [value.toLowerCase()], allowPaths: [], disallowPaths: [] };
   135→        groups.push(currentGroup);
   136→      }
   137→      lastLineWasUserAgent = true;
   138→    } else {
   139→      lastLineWasUserAgent = false;
   140→      if (!currentGroup) continue;
   141→      if (directive === 'allow' && value) currentGroup.allowPaths.push(value.toLowerCase());
   142→      else if (directive === 'disallow' && value) currentGroup.disallowPaths.push(value.toLowerCase());
   143→    }
   144→  }
   145→
   146→  // Conflicting Allow/Disallow within same group
   147→  for (const group of groups) {
   148→    const conflicts = group.allowPaths.filter((p) => group.disallowPaths.includes(p));
   149→    if (conflicts.length > 0) {
   150→      warnings.push({ type: 'warning', message: `Conflicting Allow/Disallow for paths: ${conflicts.join(', ')} (User-agent: ${group.agents.join(', ')})` });
   151→    }
   152→  }
   153→
   154→  // Duplicate user-agent groups
   155→  const agentCounts = new Map<string, number>();
   156→  for (const group of groups) for (const a of group.agents) agentCounts.set(a, (agentCounts.get(a) || 0) + 1);
   157→  for (const [agent, count] of agentCounts) {
   158→    if (count > 1) warnings.push({ type: 'warning', message: `Duplicate User-agent "${agent}" — rules may conflict` });
   159→  }
   160→
   161→  // Invalid directives
   162→  const validDirectives = new Set(['user-agent', 'disallow', 'allow', 'sitemap', 'crawl-delay', 'request-rate', 'host', 'clean-param']);
   163→  lines.forEach((line, idx) => {
   164→    const trimmed = line.trim();
   165→    if (!trimmed || trimmed.startsWith('#')) return;
   166→    const colonIdx = trimmed.indexOf(':');
   167→    if (colonIdx === -1) {
   168→      warnings.push({ type: 'warning', message: `Invalid syntax (missing colon): "${trimmed.slice(0, 50)}"`, line: idx + 1 });
   169→    } else {
   170→      const directive = trimmed.slice(0, colonIdx).trim().toLowerCase();
   171→      if (!validDirectives.has(directive)) {
   172→        warnings.push({ type: 'warning', message: `Unrecognized directive: "${directive}"`, line: idx + 1 });
   173→      }
   174→    }
   175→  });
   176→
   177→  return warnings;
   178→}

   180→// ==================== Default Content ====================
   181→
   182→function getDefaultContent(domain: string): string {
   183→  return `User-agent: *\nAllow: /\n\nSitemap: https://${domain}/sitemap.xml`;
   184→}

   186→// ==================== Syntax Highlighted Line ====================
   187→
   188→// Read-only syntax-highlighted line for the Preview modal. Deliberately
   189→// has NO line numbers — line numbers are editor chrome and would make the
   190→// served-response preview feel like a duplicate of the Editor. The Editor
   191→// (CodeEditor) is the only place that shows line numbers.
   192→function HighlightedLine({ line }: { line: string }) {
   193→  const trimmed = line.trim();
   194→
   195→  // Comment
   196→  if (trimmed.startsWith('#')) {
   197→    return <div><span className="text-muted-foreground/60 italic">{line}</span></div>;
   198→  }
   199→
   200→  // Empty line
   201→  if (!trimmed) {
   202→    return <div>&nbsp;</div>;
   203→  }
   204→
   205→  // Directive line
   206→  const colonIdx = line.indexOf(':');
   207→  if (colonIdx === -1) {
   208→    return <div><span className="text-red-500">{line}</span></div>;
   209→  }
   210→
   211→  const directive = line.slice(0, colonIdx + 1);
   212→  const rest = line.slice(colonIdx + 1);
   213→  const directiveLower = directive.toLowerCase().trim();
   214→
   215→  let directiveColor = 'text-foreground';
   216→  if (directiveLower === 'user-agent:') directiveColor = 'text-violet-600 dark:text-violet-400 font-medium';
   217→  else if (directiveLower === 'disallow:') directiveColor = 'text-red-600 dark:text-red-400 font-medium';
   218→  else if (directiveLower === 'allow:') directiveColor = 'text-green-600 dark:text-green-400 font-medium';
   219→  else if (directiveLower === 'sitemap:') directiveColor = 'text-sky-600 dark:text-sky-400 font-medium';
   220→  else directiveColor = 'text-amber-600 dark:text-amber-400';
   221→
   222→  return (
   223→    <div>
   224→      <span className={directiveColor}>{directive}</span>
   225→      <span className="text-foreground/80">{rest}</span>
   226→    </div>
   227→  );
   228→}

   230→// ==================== Status Indicator ====================
   231→
   232→function StatusIndicator({ warnings }: { warnings: ValidationWarning[] }) {
   233→  if (warnings.length === 0) {
   234→    return (
   235→      <span className="inline-flex items-center gap-1.5 text-xs font-medium text-green-600 dark:text-green-400">
   236→        <span className="h-2 w-2 rounded-full bg-green-500" />
   237→        Valid
   238→      </span>
   239→    );
   240→  }
   241→  const hasErrors = warnings.some((w) => w.type === 'error');
   242→  if (hasErrors) {
   243→    return (
   244→      <span className="inline-flex items-center gap-1.5 text-xs font-medium text-red-600 dark:text-red-400">
   245→        <span className="h-2 w-2 rounded-full bg-red-500" />
   246→        Invalid
   247→      </span>
   248→    );
   249→  }
   250→  return (
   251→    <span className="inline-flex items-center gap-1.5 text-xs font-medium text-amber-600 dark:text-amber-400">
   252→      <span className="h-2 w-2 rounded-full bg-amber-500" />
   253→      Has warnings
   254→    </span>
   255→  );
   256→}

   258→// ==================== Code Editor ====================
   259→
   260→function CodeEditor({
   261→  content,
   262→  onChange,
   263→  warningLines,
   264→}: {
   265→  content: string;
   266→  onChange: (val: string) => void;
   267→  warningLines: Set<number>;
   268→}) {
   269→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   270→  const lineNumbersRef = useRef<HTMLDivElement>(null);
   271→  const lines = content.split('\n');
   272→
   273→  // Sync scroll between line numbers and textarea
   274→  const handleScroll = useCallback(() => {
   275→    if (textareaRef.current && lineNumbersRef.current) {
   276→      lineNumbersRef.current.scrollTop = textareaRef.current.scrollTop;
   277→    }
   278→  }, []);
   279→
   280→  return (
   281→    <div className="relative rounded-lg border border-border bg-zinc-50 dark:bg-zinc-950/50 overflow-hidden">
   282→      <div className="flex">
   283→        {/* Line numbers */}
   284→        <div
   285→          ref={lineNumbersRef}
   286→          className="flex-shrink-0 overflow-hidden bg-zinc-100 dark:bg-zinc-900/50 border-r border-border py-3 select-none"
   287→          style={{ width: '3.5rem' }}
   288→        >
   289→          {lines.map((_, i) => (
   290→            <div
   291→              key={i}
   292→              className={cn(
   293→                'text-right pr-3 text-xs leading-6 font-mono',
   294→                warningLines.has(i + 1)
   295→                  ? 'text-red-500 dark:text-red-400 bg-red-50 dark:bg-red-950/30'
   296→                  : 'text-muted-foreground/40',
   297→              )}
   298→            >
   299→              {i + 1}
   300→            </div>
   301→          ))}
   302→        </div>
   303→        {/* Textarea */}
   304→        <textarea
   305→          ref={textareaRef}
   306→          className="flex-1 min-h-[400px] bg-transparent px-4 py-3 font-mono text-sm leading-6 resize-y border-0 focus-visible:outline-none focus-visible:ring-0 text-foreground/90"
   307→          value={content}
   308→          onChange={(e) => onChange(e.target.value)}
   309→          onScroll={handleScroll}
   310→          spellCheck={false}
   311→          placeholder="User-agent: *&#10;Allow: /"
   312→        />
   313→      </div>
   314→    </div>
   315→  );
   316→}

   318→// ==================== Main Page ====================
   319→
   320→export function SeoRobotsPage() {
   321→  const queryClient = useQueryClient();
   322→  const [content, setContent] = useState('');
   323→  const [isDirty, setIsDirty] = useState(false);
   324→  const [previewOpen, setPreviewOpen] = useState(false);
   325→  const [restoreConfirmOpen, setRestoreConfirmOpen] = useState(false);
   326→  const [blockAllConfirmOpen, setBlockAllConfirmOpen] = useState(false);
   327→  const activeSite = useSiteStore((s) => s.getActiveSite());
   328→  const domain = activeSite?.domain ?? 'cms.example.com';
   329→
   330→  const { data: robots, isLoading, error } = useQuery({
   331→    queryKey: queryKeys.seoRobots.all,
   332→    queryFn: () => getApi<RobotsData>('/api/seo/robots'),
   333→    staleTime: 30_000,
   334→  });
   335→
   336→  const serverContent = robots?.content ?? '';
   337→  // Sync content from server when loaded — use a key-based approach to avoid
   338→  // calling setState inside an effect (React Compiler lint rule).
   339→  // We store the last synced server content and compare in render.
   340→  const [lastSynced, setLastSynced] = useState('');
   341→  if (serverContent && serverContent !== lastSynced && !isDirty) {
   342→    setLastSynced(serverContent);
   343→    setContent(serverContent);
   344→  }
   345→
   346→  const handleContentChange = useCallback((val: string) => {
   347→    setContent(val);
   348→    setIsDirty(true);
   349→  }, []);
   350→
   351→  const saveMutation = useMutation({
   352→    mutationFn: (newContent: string) => putApi('/api/seo/robots', { content: newContent }),
   353→    onSuccess: () => {
   354→      setIsDirty(false);
   355→      queryClient.invalidateQueries({ queryKey: queryKeys.seoRobots.all });
   356→      toast.success('Robots.txt saved successfully');
   357→    },
   358→    onError: () => {
   359→      toast.error('Failed to save robots.txt');
   360→    },
   361→  });
   362→
   363→  const handleRestore = useCallback(() => {
   364→    const defaultContent = getDefaultContent(domain);
   365→    setContent(defaultContent);
   366→    setIsDirty(true);
   367→    setRestoreConfirmOpen(false);
   368→    toast.info('Restored to default robots.txt template');
   369→  }, [domain]);
   370→
   371→  const warnings = useMemo(() => validateRobots(content), [content]);
   372→  const warningLines = useMemo(() => {
   373→    const s = new Set<number>();
   374→    warnings.forEach((w) => { if (w.line) s.add(w.line); });
   375→    return s;
   376→  }, [warnings]);
   377→
   378→  const hasBlockAllError = warnings.some((w) => w.message.includes('blocks ALL crawlers'));
   379→  const hasErrors = warnings.some((w) => w.type === 'error');
   380→  const hasWarningsOnly = warnings.length > 0 && !hasErrors;
   381→
   382→  const handleSaveClick = useCallback(() => {
   383→    if (hasBlockAllError) {
   384→      setBlockAllConfirmOpen(true);
   385→    } else {
   386→      saveMutation.mutate(content);
   387→    }
   388→  }, [hasBlockAllError, saveMutation, content]);
   389→
   390→  const lineCount = content.split('\n').length;
   391→
   392→  return (
   393→    <div className="space-y-4">
   394→      {/* Error state */}
   395→      {error && (
   396→        <Card className="border-red-200 dark:border-red-900/50 bg-red-50 dark:bg-red-950/20">
   397→          <CardContent className="p-4 flex items-center gap-3">
   398→            <AlertTriangle className="h-5 w-5 text-red-500 shrink-0" />
   399→            <p className="text-sm text-red-700 dark:text-red-400">
   400→              Failed to load robots.txt. You can still edit and save.
   401→            </p>
   402→          </CardContent>
   403→        </Card>
   404→      )}
   405→
   406→      {/* Validation Warnings */}
   407→      {warnings.length > 0 && (
   408→        <Card className={cn(
   409→          'border',
   410→          warnings.some((w) => w.type === 'error')
   411→            ? 'border-red-200 dark:border-red-900/50 bg-red-50 dark:bg-red-950/20'
   412→            : 'border-amber-200 dark:border-amber-900/50 bg-amber-50 dark:bg-amber-950/20'
   413→        )}>
   414→          <CardContent className="p-4 space-y-2">
   415→            <div className={cn(
   416→              'flex items-center gap-2 text-sm font-medium',
   417→              warnings.some((w) => w.type === 'error')
   418→                ? 'text-red-700 dark:text-red-400'
   419→                : 'text-amber-700 dark:text-amber-400'
   420→            )}>
   421→              {warnings.some((w) => w.type === 'error')
   422→                ? <XCircle className="h-4 w-4 shrink-0" />
   423→                : <AlertTriangle className="h-4 w-4 shrink-0" />}
   424→              {warnings.some((w) => w.type === 'error') ? 'Validation Errors' : 'Validation Warnings'}
   425→            </div>
   426→            {warnings.map((w, i) => (
   427→              <div
   428→                key={i}
   429→                className={cn(
   430→                  'flex items-start gap-2 text-xs pl-6',
   431→                  w.type === 'error'
   432→                    ? 'text-red-600 dark:text-red-400'
   433→                    : 'text-amber-600 dark:text-amber-400',
   434→                )}
   435→              >
   436→                <span>
   437→                  {w.line && <span className="text-muted-foreground/60">Line {w.line}: </span>}
   438→                  {w.message}
   439→                </span>
   440→              </div>
   441→            ))}
   442→          </CardContent>
   443→        </Card>
   444→      )}
   445→
   446→      {/* Editor Card */}
   447→      <Card className="p-6">
   448→        {isLoading ? (
   449→          <div className="space-y-4">
   450→            <Skeleton className="h-5 w-40" />
   451→            <Skeleton className="h-96 w-full" />
   452→            <Skeleton className="h-9 w-64" />
   453→          </div>
   454→        ) : (
   455→          <>
   456→            {/* Toolbar */}
   457→            <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3 mb-4">
   458→              <div className="flex items-center gap-3">
   459→                <div className="flex items-center gap-2">
   460→                  <FileCode className="h-5 w-5 text-muted-foreground" />
   461→                  <h3 className="font-semibold text-sm">Editor</h3>
   462→                </div>
   463→                <StatusIndicator warnings={warnings} />
   464→                {isDirty ? (
   465→                  <Badge variant="outline" className="text-xs font-normal text-amber-600 dark:text-amber-400 border-amber-300 dark:border-amber-700">
   466→                    Unsaved changes
   467→                  </Badge>
   468→                ) : content.trim() ? (
   469→                  <Badge variant="outline" className="text-xs font-normal text-green-600 dark:text-green-400 border-green-300 dark:border-green-700">
   470→                    Saved
   471→                  </Badge>
   472→                ) : null}
   473→              </div>
   474→              <div className="flex flex-wrap gap-2">
   475→                <Button
   476→                  onClick={handleSaveClick}
   477→                  disabled={saveMutation.isPending || !isDirty}
   478→                  size="sm"
   479→                >
   480→                  {saveMutation.isPending ? (
   481→                    <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   482→                  ) : (
   483→                    <Save className="h-4 w-4 mr-2" />
   484→                  )}
   485→                  Save
   486→                </Button>
   487→                <Button variant="outline" size="sm" onClick={() => setRestoreConfirmOpen(true)}>
   488→                  <RotateCcw className="h-4 w-4 mr-2" />
   489→                  Restore Default
   490→                </Button>
   491→                <Button variant="outline" size="sm" onClick={() => setPreviewOpen(true)}>
   492→                  <Eye className="h-4 w-4 mr-2" />
   493→                  Preview Result
   494→                </Button>
   495→              </div>
   496→            </div>
   497→
   498→            {/* Code Editor with line numbers */}
   499→            <CodeEditor
   500→              content={content}
   501→              onChange={handleContentChange}
   502→              warningLines={warningLines}
   503→            />
   504→
   505→            {/* Footer bar */}
   506→            <div className="flex items-center justify-between mt-3 text-xs text-muted-foreground">
   507→              <span>{lineCount} lines · {content.length.toLocaleString()} characters</span>
   508→              <div className="flex items-center gap-1.5">
   509→                {isDirty ? (
   510→                  <>
   511→                    <span className="h-1.5 w-1.5 rounded-full bg-amber-500" />
   512→                    Modified
   513→                  </>
   514→                ) : (
   515→                  <>
   516→                    <CheckCircle2 className="h-3 w-3 text-green-500" />
   517→                    Saved
   518→                  </>
   519→                )}
   520→              </div>
   521→            </div>
   522→          </>
   523→        )}
   524→      </Card>
   525→
   526→      {/* Preview Dialog — renders the actual /robots.txt HTTP response */}
   527→      <Dialog open={previewOpen} onOpenChange={setPreviewOpen}>
   528→        <DialogContent className="max-w-3xl max-h-[85vh]">
   529→          <DialogHeader>
   530→            <DialogTitle className="flex items-center gap-2">
   531→              <FileCode className="h-4 w-4" />
   532→              Robots.txt Preview
   533→            </DialogTitle>
   534→          </DialogHeader>
   535→          <div className="space-y-3">
   536→            <div className="text-xs text-muted-foreground">
   537→              This is the exact response served at{' '}
   538→              <code className="font-mono text-primary">/robots.txt</code>. Read-only.
   539→            </div>
   540→
   541→            {/* HTTP response frame — mimics the served response, not the editor */}
   542→            <div className="overflow-hidden rounded-lg border border-border">
   543→              {/* Response status / headers bar */}
   544→              <div className="flex flex-wrap items-center gap-x-4 gap-y-1 border-b border-border bg-muted/40 px-3 py-2 text-[11px] font-mono">
   545→                <span className="flex items-center gap-1.5">
   546→                  <span className="text-muted-foreground">GET</span>
   547→                  <span className="text-foreground font-medium">/robots.txt</span>
   548→                </span>
   549→                <span
   550→                  className={cn(
   551→                    'inline-flex items-center gap-1 rounded px-1.5 py-0.5 font-semibold',
   552→                    hasErrors
   553→                      ? 'bg-red-100 text-red-700 dark:bg-red-950/40 dark:text-red-400'
   554→                      : 'bg-green-100 text-green-700 dark:bg-green-950/40 dark:text-green-400',
   555→                  )}
   556→                >
   557→                  {hasErrors ? '200 OK · Invalid' : '200 OK'}
   558→                </span>
   559→                <span className="text-muted-foreground">text/plain</span>
   560→                <span className="text-muted-foreground">utf-8</span>
   561→                <span className="text-muted-foreground">{content.length} bytes</span>
   562→              </div>
   563→
   564→              {/* Read-only response body — syntax-highlighted, NO line
   565→                  numbers (line numbers are editor chrome; their absence
   566→                  keeps this clearly a served-response preview, not an
   567→                  editor duplicate). The body is non-editable: it's a
   568→                  <pre>-style block with select-text cursor, no focus ring. */}
   569→              <div className="overflow-auto bg-zinc-50 dark:bg-zinc-950/50 p-4 max-h-[50vh] select-text">
   570→                <div className="font-mono text-sm leading-6 whitespace-pre-wrap cursor-default">
   571→                  {content.trim() ? (
   572→                    content.split('\n').map((line, i) => (
   573→                      <HighlightedLine key={i} line={line} />
   574→                    ))
   575→                  ) : (
   576→                    <span className="text-muted-foreground/50 italic"># robots.txt is empty</span>
   577→                  )}
   578→                </div>
   579→              </div>
   580→            </div>
   581→
   582→            {/* Validation state footer — reflects the current validation state */}
   583→            <div className="space-y-1.5">
   584→              {warnings.length === 0 && content.trim() && (
   585→                <div className="flex items-center gap-2 text-xs text-green-600 dark:text-green-400">
   586→                  <CheckCircle2 className="h-4 w-4 shrink-0" />
   587→                  No validation issues found — robots.txt is valid
   588→                </div>
   589→              )}
   590→              {!content.trim() && (
   591→                <div className="flex items-center gap-2 text-xs text-amber-600 dark:text-amber-400">
   592→                  <AlertTriangle className="h-4 w-4 shrink-0" />
   593→                  robots.txt is empty — crawlers will have no rules to follow
   594→                </div>
   595→              )}
   596→              {hasErrors && (
   597→                <div className="space-y-1">
   598→                  {warnings
   599→                    .filter((w) => w.type === 'error')
   600→                    .map((w, i) => (
   601→                      <div
   602→                        key={`e${i}`}
   603→                        className="flex items-start gap-2 text-xs text-red-600 dark:text-red-400"
   604→                      >
   605→                        <XCircle className="h-3.5 w-3.5 shrink-0 mt-0.5" />
   606→                        <span>
   607→                          {w.line && (
   608→                            <span className="text-muted-foreground/70">Line {w.line}: </span>
   609→                          )}
   610→                          {w.message}
   611→                        </span>
   612→                      </div>
   613→                    ))}
   614→                </div>
   615→              )}
   616→              {hasWarningsOnly && (
   617→                <div className="space-y-1">
   618→                  {warnings
   619→                    .filter((w) => w.type === 'warning')
   620→                    .map((w, i) => (
   621→                      <div
   622→                        key={`w${i}`}
   623→                        className="flex items-start gap-2 text-xs text-amber-600 dark:text-amber-400"
   624→                      >
   625→                        <AlertTriangle className="h-3.5 w-3.5 shrink-0 mt-0.5" />
   626→                        <span>
   627→                          {w.line && (
   628→                            <span className="text-muted-foreground/70">Line {w.line}: </span>
   629→                          )}
   630→                          {w.message}
   631→                        </span>
   632→                      </div>
   633→                    ))}
   634→                </div>
   635→              )}
   636→            </div>
   637→          </div>
   638→        </DialogContent>
   639→      </Dialog>
   640→
   641→      {/* Restore Default Confirmation */}
   642→      <AlertDialog open={restoreConfirmOpen} onOpenChange={setRestoreConfirmOpen}>
   643→        <AlertDialogContent>
   644→          <AlertDialogHeader>
   645→            <AlertDialogTitle>Restore Default Robots.txt?</AlertDialogTitle>
   646→            <AlertDialogDescription>
   647→              This will replace the current content with the recommended default robots.txt configuration. Your current changes will be lost.
   648→            </AlertDialogDescription>
   649→          </AlertDialogHeader>
   650→          <AlertDialogFooter>
   651→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   652→            <AlertDialogAction onClick={handleRestore}>
   653→              Restore Default
   654→            </AlertDialogAction>
   655→          </AlertDialogFooter>
   656→        </AlertDialogContent>
   657→      </AlertDialog>
   658→
   659→      {/* Block All Confirmation — SEO Safety */}
   660→      <AlertDialog open={blockAllConfirmOpen} onOpenChange={setBlockAllConfirmOpen}>
   661→        <AlertDialogContent>
   662→          <AlertDialogHeader>
   663→            <AlertDialogTitle className="flex items-center gap-2">
   664→              <Shield className="h-5 w-5 text-red-500" />
   665→              Warning: This Blocks All Crawlers
   666→            </AlertDialogTitle>
   667→            <AlertDialogDescription className="space-y-2">
   668→              <span className="block">
   669→                Your robots.txt contains <code className="font-mono text-red-600 dark:text-red-400 font-medium">Disallow: /</code> for <code className="font-mono font-medium">User-agent: *</code>.
   670→              </span>
   671→              <span className="block">
   672→                This will prevent search engines from crawling and indexing your entire website. This can severely harm your SEO.
   673→              </span>
   674→              <span className="block font-medium">
   675→                Are you sure you want to save this configuration?
   676→              </span>
   677→            </AlertDialogDescription>
   678→          </AlertDialogHeader>
   679→          <AlertDialogFooter>
   680→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   681→            <AlertDialogAction
   682→              className="bg-red-600 hover:bg-red-700"
   683→              onClick={() => {
   684→                setBlockAllConfirmOpen(false);
   685→                saveMutation.mutate(content);
   686→              }}
   687→            >
   688→              Yes, Save Anyway
   689→            </AlertDialogAction>
   690→          </AlertDialogFooter>
   691→        </AlertDialogContent>
   692→      </AlertDialog>
   693→    </div>
   694→  );
   695→}
