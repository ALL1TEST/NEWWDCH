   114→   114→export function useDataTable(options: UseDataTableOptions = {}): UseDataTableReturn {
   115→   115→  const {
   116→   116→    initialPage = 1,
   117→   117→    initialPageSize = DEFAULT_PAGE_SIZE,
   118→   118→    initialSortField = 'createdAt',
   119→   119→    initialSortOrder = 'desc',
   120→   120→  } = options;
   121→   121→
   122→   122→  const [currentPage, setCurrentPage] = useState(initialPage);
   123→   123→  const [pageSize, setPageSize] = useState(initialPageSize);
   124→   124→  const [sortField, setSortFieldRaw] = useState(initialSortField);
   125→   125→  const [sortOrder, setSortOrderRaw] = useState<'asc' | 'desc'>(initialSortOrder);
   126→   126→  const [selectedIds, setSelectedIds] = useState<string[]>([]);
   127→   127→  const [searchValue, setSearchValue] = useState('');
   128→   128→
   129→   129→  const setSortField = useCallback((field: string, order: 'asc' | 'desc') => {
   130→   130→    setSortFieldRaw(field);
   131→   131→    setSortOrderRaw(order);
   132→   132→    setCurrentPage(1);
   133→   133→  }, []);
   134→   134→
   135→   135→  const toggleSort = useCallback(
   136→   136→    (field: string) => {
   137→   137→      if (sortField === field) {
   138→   138→        const next = sortOrder === 'asc' ? 'desc' : 'asc';
   139→   139→        setSortFieldRaw(field);
   140→   140→        setSortOrderRaw(next);
   141→   141→      } else {
   142→   142→        setSortFieldRaw(field);
   143→   143→        setSortOrderRaw('asc');
   144→   144→      }
   145→   145→      setCurrentPage(1);
   146→   146→    },
   147→   147→    [sortField, sortOrder],
   148→   148→  );
   149→   149→
   150→   150→  const toggleRowSelection = useCallback((id: string) => {
   151→   151→    setSelectedIds((prev) =>
   152→   152→      prev.includes(id) ? prev.filter((i) => i !== id) : [...prev, id],
   153→   153→    );
   154→   154→  }, []);
   155→   155→
   156→   156→  const toggleAllSelection = useCallback(
   157→   157→    (allIds: string[], currentDataIds: string[]) => {
   158→   158→      const allSelected = currentDataIds.every((id) => selectedIds.includes(id));
   159→   159→      if (allSelected) {
   160→   160→        setSelectedIds((prev) =>
   161→   161→          prev.filter((id) => !currentDataIds.includes(id)),
   162→   162→        );
   163→   163→      } else {
   164→   164→        const newIds = currentDataIds.filter((id) => !selectedIds.includes(id));
   165→   165→        setSelectedIds((prev) => [...prev, ...newIds]);
   166→   166→      }
   167→   167→    },
   168→   168→    [selectedIds],
   169→   169→  );
   170→   170→
   171→   171→  const clearSelection = useCallback(() => {
   172→   172→    setSelectedIds([]);
   173→   173→  }, []);
   174→   174→
   175→   175→  return {
   176→   176→    currentPage,
   177→   177→    pageSize,
   178→   178→    sortField,
   179→   179→    sortOrder,
   180→   180→    selectedIds,
   181→   181→    searchValue,
   182→   182→    setCurrentPage,
   183→   183→    setPageSize,
   184→   184→    setSortField,
   185→   185→    setSelectedIds,
   186→   186→    setSearchValue,
   187→   187→    toggleSort,
   188→   188→    toggleRowSelection,
   189→   189→    toggleAllSelection,
   190→   190→    clearSelection,
   191→   191→  };
   192→   192→}

   483→   483→// -------------------- Pagination --------------------
   484→   484→
   485→   485→function DataTablePagination({
   486→   486→  currentPage,
   487→   487→  pageSize,
   488→   488→  totalItems,
   489→   489→  onPageChange,
   490→   490→  onPageSizeChange,
   491→   491→}: {
   492→   492→  currentPage: number;
   493→   493→  pageSize: number;
   494→   494→  totalItems: number;
   495→   495→  onPageChange: (page: number) => void;
   496→   496→  onPageSizeChange: (size: number) => void;
   497→   497→}) {
   498→   498→  const totalPages = Math.max(1, Math.ceil(totalItems / pageSize));
   499→   499→  const rangeStart = totalItems === 0 ? 0 : (currentPage - 1) * pageSize + 1;
   500→   500→  const rangeEnd = Math.min(currentPage * pageSize, totalItems);
   501→   501→
   502→   502→  return (
   503→   503→    <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between px-2 py-3 border-t">
   504→   504→      <div className="text-xs text-muted-foreground">
   505→   505→        Showing {rangeStart}–{rangeEnd} of {totalItems} items
   506→   506→      </div>

   537→   537→          <Button

   539→   539→            size="icon"
   540→   540→            className="h-8 w-8"
   541→   541→            onClick={() => onPageChange(currentPage - 1)}
   542→   542→            disabled={currentPage <= 1}
   543→   543→          >
   544→   544→            <ChevronLeft className="h-4 w-4" />
   545→   545→            <span className="sr-only">Previous page</span>
   546→   546→          </Button>
   547→   547→          <span className="text-xs text-muted-foreground min-w-[60px] text-center">
   548→   548→            {currentPage} / {totalPages}
   549→   549→          </span>
   550→   550→          <Button
   551→   551→            variant="outline"
   552→   552→            size="icon"
   553→   553→            className="h-8 w-8"
   554→   554→            onClick={() => onPageChange(currentPage + 1)}
   555→   555→            disabled={currentPage >= totalPages}
   556→   556→          >
   557→   557→            <ChevronRight className="h-4 w-4" />
   558→   558→            <span className="sr-only">Next page</span>

   560→   560→          <Button
   561→   561→            variant="outline"
   562→   562→            size="icon"
   563→   563→            className="h-8 w-8"
   564→   564→            onClick={() => onPageChange(totalPages)}
   565→   565→            disabled={currentPage >= totalPages}
   566→   566→          >
   567→   567→            <ChevronsRight className="h-4 w-4" />
   568→   568→            <span className="sr-only">Last page</span>

   570→   570→        </div>
   571→   571→      </div>
   572→   572→    </div>
   573→   573→  );

   578→   578→function TableToolbar<TData, TValue>({
   579→   579→  searchPlaceholder,
   580→   580→  searchValue,
   581→   581→  onSearch,
   582→   582→  selectedIds,
   583→   583→  bulkActions,
   584→   584→  onBulkAction,
   585→   585→  onSelectionChange,
   586→   586→  filterContent,
   587→   587→}: {
   588→   588→  searchPlaceholder?: string;
   589→   589→  searchValue?: string;
   590→   590→  onSearch?: (value: string) => void;
   591→   591→  selectedIds?: string[];
   592→   592→  bulkActions?: BulkAction[];
   593→   593→  onBulkAction?: (action: string, ids: string[]) => void;
   594→   594→  onSelectionChange?: (ids: string[]) => void;
   595→   595→  filterContent?: React.ReactNode;
   596→   596→}) {
   597→   597→  const hasSelection = selectedIds && selectedIds.length > 0;
   598→   598→
   599→   599→  return (
   600→   600→    <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between p-4 pb-2">
   601→   601→      <div className="flex items-center gap-2 flex-1">
   602→   602→        {(onSearch || searchValue !== undefined) && (
   603→   603→          <div className="relative max-w-sm w-full">
   604→   604→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   605→   605→            <Input
   606→   606→              placeholder={searchPlaceholder ?? 'Search...'}
   607→   607→              value={searchValue ?? ''}
   608→   608→              onChange={(e) => onSearch?.(e.target.value)}
   609→   609→              className="pl-9 h-9"
   610→   610→            />
   611→   611→          </div>
   612→   612→        )}
   613→   613→        {filterContent}
   614→   614→      </div>
   615→   615→      {hasSelection && bulkActions && bulkActions.length > 0 && (
   616→   616→        <div className="flex items-center gap-2">
   617→   617→          <span className="text-xs text-muted-foreground">
   618→   618→            {selectedIds.length} selected
   619→   619→          </span>
   620→   620→          <DropdownMenu>
   621→   621→            <DropdownMenuTrigger asChild>
   622→   622→              <Button variant="outline" size="sm">
   623→   623→                <MoreHorizontal className="h-4 w-4 mr-1" />
   624→   624→                Bulk Actions
   625→   625→              </Button>
   626→   626→            </DropdownMenuTrigger>
   627→   627→            <DropdownMenuContent align="end">
   628→   628→              {bulkActions.map((action) => (
   629→   629→                <DropdownMenuItem
   630→   630→                  key={action.label}
   631→   631→                  variant={action.variant}
   632→   632→                  disabled={action.disabled}
   633→   633→                  onClick={() => onBulkAction?.(action.label, selectedIds)}
   634→   634→                >
   635→   635→                  {action.icon}
   636→   636→                  {action.label}
   637→   637→                </DropdownMenuItem>
   638→   638→              ))}
   639→   639→            </DropdownMenuContent>
   640→   640→          </DropdownMenu>
   641→   641→          <Button
   642→   642→            variant="ghost"
   643→   643→            size="sm"
   644→   644→            onClick={() => onSelectionChange?.([])}
   645→   645→          >
   646→   646→            Clear
   647→   647→          </Button>
   648→   648→        </div>
   649→   649→      )}
   650→   650→    </div>
   651→   651→  );
   652→   652→}

   654→   654→// -------------------- Main DataTable Component --------------------
   655→   655→
   656→   656→export function DataTable<TData, TValue>({
   657→   657→  columns,
   658→   658→  data,
   659→   659→  isLoading = false,
   660→   660→  totalItems,
   661→   661→  pageSize,
   662→   662→  currentPage,
   663→   663→  onPageChange,
   664→   664→  onSortChange,
   665→   665→  sortField,
   666→   666→  sortOrder,
   667→   667→  onRowClick,
   668→   668→  selectedIds = [],
   669→   669→  onSelectionChange,
   670→   670→  onBulkAction,
   671→   671→  bulkActions,
   672→   672→  emptyMessage = 'No data found.',
   673→   673→  emptyIcon,
   674→   674→  searchPlaceholder,
   675→   675→  onSearch,
   676→   676→  searchValue,
   677→   677→  filterContent,
   678→   678→  getRowId,
   679→   679→}: DataTableProps<TData, TValue>) {
   680→   680→  const hasRowSelection = !!onSelectionChange;
   681→   681→  const columnCount = columns.length + (hasRowSelection ? 1 : 0);
   682→   682→
   683→   683→  const enhancedColumns = useMemo<ColumnDef<TData, TValue>[]>(() => {
   684→   684→    if (!hasRowSelection) return columns;
   685→   685→
   686→   686→    const selectColumn: ColumnDef<TData, TValue> = {
   687→   687→      id: '__select__',
   688→   688→      size: 40,
   689→   689→      enableSorting: false,
   690→   690→      enableHiding: false,
   691→   691→      header: ({ table }) => {
   692→   692→        const allRowIds = table.getRowModel().rows.map((r) => r.id);
   693→   693→        const allChecked =
   694→   694→          allRowIds.length > 0 && allRowIds.every((id) => selectedIds.includes(id));
   695→   695→        const someChecked = allRowIds.some((id) => selectedIds.includes(id));
   696→   696→
   697→   697→        return (
   698→   698→          <Checkbox
   699→   699→            checked={allChecked ? true : someChecked ? 'indeterminate' : false}
   700→   700→            onCheckedChange={(checked) => {
   701→   701→              if (checked) {
   702→   702→                const allIds = data
   703→   703→                  .map((row, idx) => getRowId?.(row) ?? String(idx))
   704→   704→                  .filter((id) => !selectedIds.includes(id));
   705→   705→                onSelectionChange?.([...selectedIds, ...allIds]);
   706→   706→              } else {
   707→   707→                const currentDataIds = data.map((row, idx) =>
   708→   708→                  getRowId?.(row) ?? String(idx),
   709→   709→                );
   710→   710→                onSelectionChange?.(
   711→   711→                  selectedIds.filter((id) => !currentDataIds.includes(id)),
   712→   712→                );
   713→   713→              }
   714→   714→            }}
   715→   715→            aria-label="Select all rows"
   716→   716→          />
   717→   717→        );
   718→   718→      },
   719→   719→      cell: ({ row }) => {
   720→   720→        const rowId = row.id;
   721→   721→        const isChecked = selectedIds.includes(rowId);
   722→   722→
   723→   723→        return (
   724→   724→          <Checkbox
   725→   725→            checked={isChecked}
   726→   726→            onCheckedChange={(checked) => {
   727→   727→              if (checked) {
   728→   728→                onSelectionChange?.([...selectedIds, rowId]);
   729→   729→              } else {
   730→   730→                onSelectionChange?.(selectedIds.filter((id) => id !== rowId));
   731→   731→              }
   732→   732→            }}
   733→   733→            onClick={(e) => e.stopPropagation()}
   734→   734→            aria-label={`Select row ${rowId}`}
   735→   735→          />
   736→   736→        );
   737→   737→      },
   738→   738→    };
   739→   739→
   740→   740→    return [selectColumn, ...columns];
   741→   741→  }, [columns, hasRowSelection, selectedIds, onSelectionChange, data, getRowId]);
   742→   742→
   743→   743→  const sorting: ColumnSort[] = useMemo(() => {
   744→   744→    if (!sortField) return [];
   745→   745→    return [{ id: sortField, desc: sortOrder === 'desc' }];
   746→   746→  }, [sortField, sortOrder]);
   747→   747→
   748→   748→  const table = useReactTable({
   749→   749→    data,
   750→   750→    columns: enhancedColumns,
   751→   751→    getCoreRowModel: getCoreRowModel(),
   752→   752→    getSortedRowModel: getSortedRowModel(),
   753→   753→    state: {
   754→   754→      sorting,
   755→   755→    },
   756→   756→    onSortingChange: (updater) => {
   757→   757→      const newSorting =
   758→   758→        typeof updater === 'function' ? updater(sorting) : updater;
   759→   759→      if (newSorting.length > 0) {
   760→   760→        onSortChange(newSorting[0].id, newSorting[0].desc ? 'desc' : 'asc');
   761→   761→      }
   762→   762→    },
   763→   763→    manualPagination: true,
   764→   764→    manualSorting: true,
   765→   765→    getRowId: getRowId
   766→   766→      ? (row, index) => getRowId(row) ?? String(index)
   767→   767→      : (_, index) => String(index),
   768→   768→  });
   769→   769→
   770→   770→  const handlePageSizeChange = useCallback(
   771→   771→    (newSize: number) => {
   772→   772→      onPageChange(1);
   773→   773→    },
   774→   774→    [onPageChange],
   775→   775→  );
   776→   776→
   777→   777→  const showToolbar =
   778→   778→    (onSearch || searchValue !== undefined) ||
   779→   779→    (filterContent != null) ||
   780→   780→    (selectedIds.length > 0 && bulkActions && bulkActions.length > 0);
   781→   781→
   782→   782→  return (
   783→   783→    <div className="space-y-0">
   784→   784→      {showToolbar && (
   785→   785→        <TableToolbar
   786→   786→          searchPlaceholder={searchPlaceholder}
   787→   787→          searchValue={searchValue}
   788→   788→          onSearch={onSearch}
   789→   789→          selectedIds={selectedIds}
   790→   790→          bulkActions={bulkActions}
   791→   791→          onBulkAction={onBulkAction}
   792→   792→          onSelectionChange={onSelectionChange}
   793→   793→          filterContent={filterContent}
   794→   794→        />
   795→   795→      )}
   796→   796→      <div className="rounded-lg border">
   797→   797→        <Table>
   798→   798→          <TableHeader>
   799→   799→            {table.getHeaderGroups().map((headerGroup) => (
   800→   800→              <TableRow key={headerGroup.id}>
   801→   801→                {headerGroup.headers.map((header) => (
   802→   802→                  <TableHead
   803→   803→                    key={header.id}
   804→   804→                    style={{ width: header.getSize() !== 150 ? header.getSize() : undefined }}
   805→   805→                  >
   806→   806→                    {flexRender(
   807→   807→                      header.column.columnDef.header,
   808→   808→                      header.getContext(),
   809→   809→                    )}
   810→   810→                  </TableHead>
   811→   811→                ))}
   812→   812→              </TableRow>
   813→   813→            ))}
   814→   814→          </TableHeader>
   815→   815→          <TableBody>
   816→   816→            {isLoading ? (
   817→   817→              <SkeletonRows columnCount={columnCount} />
   818→   818→            ) : data.length === 0 ? (
   819→   819→              <DataTableEmpty icon={emptyIcon} message={emptyMessage} />
   820→   820→            ) : (
   821→   821→              table.getRowModel().rows.map((row) => (
   822→   822→                <TableRow
   823→   823→                  key={row.id}
   824→   824→                  data-state={selectedIds.includes(row.id) ? 'selected' : undefined}
   825→   825→                  className={cn(onRowClick && 'cursor-pointer')}
   826→   826→                  onClick={() => onRowClick?.(row.original)}
   827→   827→                >
   828→   828→                  {row.getVisibleCells().map((cell) => (
   829→   829→                    <TableCell key={cell.id}>
   830→   830→                      {flexRender(
   831→   831→                        cell.column.columnDef.cell,
   832→   832→                        cell.getContext(),
   833→   833→                      )}
   834→   834→                    </TableCell>
   835→   835→                  ))}
   836→   836→                </TableRow>
   837→   837→              ))
   838→   838→            )}
   839→   839→          </TableBody>
   840→   840→        </Table>
   841→   841→      </div>
   842→   842→      <DataTablePagination
   843→   843→        currentPage={currentPage}
   844→   844→        pageSize={pageSize}
   845→   845→        totalItems={totalItems}
   846→   846→        onPageChange={onPageChange}
   847→   847→        onPageSizeChange={handlePageSizeChange}
   848→   848→      />
   849→   849→    </div>
   850→   850→  );
   851→   851→}
