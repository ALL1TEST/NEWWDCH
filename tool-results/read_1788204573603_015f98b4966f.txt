   142→// -------------------- Error Diagnostic Parser --------------------
   143→// Parses the `lastError` string (stored by healthCheck as "HTTP {status}: {body}")
   144→// and returns structured, user-friendly diagnostic info. NEVER exposes API keys.
   145→
   146→interface ErrorDiagnostic {
   147→  httpStatus: number | null;
   148→  errorType: string | null;       // e.g. "authentication_error", "request_forbidden"
   149→  errorMessage: string | null;    // the provider's own message
   150→  category: string;               // our human-readable category
   151→  suggestion: string | null;      // suggested fix
   152→  rawSnippet: string | null;      // truncated sanitized raw error (no keys)
   153→}
   154→
   155→// Patterns that look like API keys or secrets — stripped from any displayed text
   156→const SECRET_PATTERNS = [
   157→  /sk-[a-zA-Z0-9-_]{8,}/gi,
   158→  /sk-ant-[a-zA-Z0-9-_]{8,}/gi,
   159→  /Bearer\s+[a-zA-Z0-9-_]{8,}/gi,
   160→  /api[_-]?key[=:]\s*["']?[a-zA-Z0-9-_]{8,}["']?/gi,
   161→  /x-api-key:\s*[a-zA-Z0-9-_]{8,}/gi,
   162→  /\b[A-Za-z0-9+/]{40,}={0,2}\b/g, // base64-looking blobs
   163→];
   164→
   165→function sanitizeSecrets(text: string): string {
   166→  let cleaned = text;
   167→  for (const pattern of SECRET_PATTERNS) {
   168→    cleaned = cleaned.replace(pattern, '[REDACTED]');
   169→  }
   170→  return cleaned;
   171→}
   172→
   173→function parseErrorDiagnostic(lastError: string | null): ErrorDiagnostic | null {
   174→  if (!lastError || lastError.trim() === '') return null;
   175→
   176→  const sanitized = sanitizeSecrets(lastError);
   177→
   178→  // Extract HTTP status: pattern is "HTTP {status}:" or "HTTP {status} "
   179→  const statusMatch = sanitized.match(/HTTP\s+(\d{3})/i);
   180→  const httpStatus = statusMatch ? parseInt(statusMatch[1], 10) : null;
   181→
   182→  // Try to parse the JSON body after "HTTP {status}:"
   183→  let errorType: string | null = null;
   184→  let errorMessage: string | null = null;
   185→  let rawSnippet: string | null = null;
   186→
   187→  const bodyMatch = sanitized.match(/HTTP\s+\d{3}\s*:?\s*([\s\S]*)/i);
   188→  if (bodyMatch && bodyMatch[1]) {
   189→    const bodyStr = bodyMatch[1].trim();
   190→    try {
   191→      const parsed = JSON.parse(bodyStr);
   192→      if (parsed?.error?.message) errorMessage = sanitizeSecrets(String(parsed.error.message));
   193→      if (parsed?.error?.type) errorType = sanitizeSecrets(String(parsed.error.type));
   194→      if (parsed?.error?.code && !errorType) errorType = sanitizeSecrets(String(parsed.error.code));
   195→    } catch {
   196→      // Not JSON — use the raw body (sanitized) as the message
   197→      if (bodyStr) errorMessage = sanitizeSecrets(bodyStr.slice(0, 200));
   198→    }
   199→  } else {
   200→    // No HTTP status prefix — use the raw error string
   201→    errorMessage = sanitizeSecrets(sanitized.slice(0, 200));
   202→  }
   203→
   204→  rawSnippet = sanitized.slice(0, 300);
   205→
   206→  // Map HTTP status → category + suggestion
   207→  let category = 'Connection Error';
   208→  let suggestion: string | null = null;
   209→
   210→  if (httpStatus !== null) {
   211→    if (httpStatus === 401) {
   212→      category = 'Authentication Failed (401)';
   213→      suggestion = 'The API key appears to be invalid or expired. Verify the key in the provider settings and update it if necessary.';
   214→    } else if (httpStatus === 403) {
   215→      category = 'Access Denied (403)';
   216→      suggestion = 'Access was denied by the provider. This may be due to region restrictions, account permissions, or an expired plan. Check your provider account status.';
   217→    } else if (httpStatus === 404) {
   218→      category = 'Endpoint Not Found (404)';
   219→      suggestion = 'The Base URL or models endpoint was not found. Verify the Base URL is correct and includes the API version path (e.g. /v1).';
   220→    } else if (httpStatus === 429) {
   221→      category = 'Rate Limit Exceeded (429)';
   222→      suggestion = 'Too many requests were sent. Wait a moment and try again. Consider upgrading your provider plan for higher rate limits.';
   223→    } else if (httpStatus >= 500) {
   224→      category = `Provider Server Error (${httpStatus})`;
   225→      suggestion = 'The provider\'s server encountered an error. Try again later. If the problem persists, check the provider\'s status page.';
   226→    } else if (httpStatus >= 400) {
   227→      category = `Request Error (${httpStatus})`;
   228→      suggestion = 'The request was rejected by the provider. Review the error details and adjust your configuration.';
   229→    }
   230→  } else {
   231→    // No HTTP status — likely a network/timeout error
   232→    const lower = sanitized.toLowerCase();
   233→    if (lower.includes('timeout') || lower.includes('timed out')) {
   234→      category = 'Timeout';
   235→      suggestion = 'The provider did not respond within the timeout. The provider may be slow or unreachable. Try again or check the provider\'s status.';
   236→    } else if (lower.includes('fetch failed') || lower.includes('enotfound') || lower.includes('econnrefused') || lower.includes('network')) {
   237→      category = 'Network Error';
   238→      suggestion = 'Could not reach the provider. Verify the Base URL is correct, check your network connection, and ensure the provider endpoint is accessible.';
   239→    } else if (lower.includes('ssl') || lower.includes('certificate')) {
   240→      category = 'SSL/Certificate Error';
   241→      suggestion = 'There was an SSL/TLS certificate issue. Verify the Base URL uses https and the provider\'s certificate is valid.';
   242→    }
   243→  }
   244→
   245→  return { httpStatus, errorType, errorMessage, category, suggestion, rawSnippet };
   246→}

   248→// Format the lastHealthCheckAt timestamp for display
   249→function formatLastAttempt(iso: string | null): string | null {
   250→  if (!iso) return null;
   251→  try {
   252→    return new Date(iso).toLocaleString();
   253→  } catch {
   254→    return null;
   255→  }

   275→export function ProvidersPage() {
   276→  const queryClient = useQueryClient();
   277→  const [page, setPage] = useState(1);
   278→  const [pageSize] = useState(25);
   279→  const [search, setSearch] = useState('');
   280→  const [kindFilter, setKindFilter] = useState<string>('all');
   281→  const [statusFilter, setStatusFilter] = useState<string>('all');
   282→  const [dialogOpen, setDialogOpen] = useState(false);
   283→  const [editingProvider, setEditingProvider] = useState<AiProvider | null>(null);
   284→  const [formData, setFormData] = useState<ProviderFormData>(emptyForm);
   285→  const [showApiKey, setShowApiKey] = useState(false);
   286→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   287→  const [deletingProvider, setDeletingProvider] = useState<AiProvider | null>(null);
   288→
   289→  // Fetch providers
   290→  const { data, isLoading, isError } = useQuery({
   291→    queryKey: queryKeys.aiProviders.list({ page, pageSize, search, kind: kindFilter, connectionStatus: statusFilter }),
   292→    queryFn: () => getApi<PaginatedResponse<AiProvider>>('/api/ai/providers', {
   293→      page,
   294→      pageSize,
   295→      search: search || undefined,
   296→      kind: kindFilter !== 'all' ? kindFilter : undefined,
   297→      connectionStatus: statusFilter !== 'all' ? statusFilter : undefined,
   298→    }),
   299→  });
   300→
   301→  const providers = data?.data ?? [];
   302→  const pagination = data?.pagination;
   303→
   304→  // Create / Update mutation
   305→  const saveMutation = useMutation({
   306→    mutationFn: (body: ProviderFormData) => {
   307→      if (editingProvider) {
   308→        const patchBody: Record<string, unknown> = { ...body };
   309→        if (!patchBody.apiKey) delete patchBody.apiKey;
   310→        return patchApi<AiProvider>(`/api/ai/providers/${editingProvider.id}`, patchBody);
   311→      }
   312→      return postApi<AiProvider>('/api/ai/providers', body);
   313→    },
   314→    onSuccess: () => {
   315→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   316→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   317→      queryClient.invalidateQueries({ queryKey: queryKeys.aiSettings.all });
   318→      queryClient.invalidateQueries({ queryKey: queryKeys.aiPrompts.all });
   319→      toast.success(editingProvider ? 'Provider updated' : 'Provider created');
   320→      handleCloseDialog();
   321→    },
   322→    onError: (err: Error) => {
   323→      toast.error(err.message || 'Failed to save provider');
   324→    },
   325→  });
   326→
   327→  // Delete mutation
   328→  const deleteMutation = useMutation({
   329→    mutationFn: (id: string) => deleteApi(`/api/ai/providers/${id}`),
   330→    onSuccess: () => {
   331→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   332→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   333→      queryClient.invalidateQueries({ queryKey: queryKeys.aiSettings.all });
   334→      queryClient.invalidateQueries({ queryKey: queryKeys.aiPrompts.all });
   335→      queryClient.invalidateQueries({ queryKey: queryKeys.aiLogs.all });
   336→      toast.success('Provider deleted');
   337→      setDeleteDialogOpen(false);
   338→      setDeletingProvider(null);
   339→    },
   340→    onError: (err: Error) => {
   341→      toast.error(err.message || 'Failed to delete provider');
   342→    },
   343→  });
   344→
   345→  // Test connection mutation
   346→  const testMutation = useMutation({
   347→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/test`),
   348→    onSuccess: (result: unknown) => {
   349→      const res = result as { success?: boolean; latency?: number; message?: string; status?: string };
   350→      if (res.success) {
   351→        toast.success(`Connection successful (${res.latency ?? 0}ms)`);
   352→      } else {
   353→        toast.error(res.message || `Connection ${res.status || 'failed'}`);
   354→      }
   355→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   356→    },
   357→    onError: (err: Error) => {
   358→      toast.error(err.message || 'Connection test failed');
   359→    },
   360→  });
   361→
   362→  // Sync models mutation
   363→  const syncMutation = useMutation({
   364→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/sync-models`),
   365→    onSuccess: (result: unknown) => {
   366→      const res = result as { syncedCount?: number; count?: number };
   367→      toast.success(`Synced ${res.syncedCount ?? res.count ?? 0} models`);
   368→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   369→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   370→    },
   371→    onError: (err: Error) => {
   372→      toast.error(err.message || 'Failed to sync models');
   373→    },
   374→  });
   375→
   376→  // Set default mutation
   377→  const setDefaultMutation = useMutation({
   378→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/set-default`),
   379→    onSuccess: () => {
   380→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   381→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   382→      queryClient.invalidateQueries({ queryKey: queryKeys.aiSettings.all });
   383→      queryClient.invalidateQueries({ queryKey: queryKeys.aiPrompts.all });
   384→      queryClient.invalidateQueries({ queryKey: queryKeys.aiLogs.all });
   385→      toast.success('Default provider updated');
   386→    },
   387→    onError: (err: Error) => {
   388→      toast.error(err.message || 'Failed to set default');
   389→    },
   390→  });
   391→
   392→  // Toggle active mutation
   393→  const toggleActiveMutation = useMutation({
   394→    mutationFn: ({ id, isActive }: { id: string; isActive: boolean }) =>
   395→      patchApi(`/api/ai/providers/${id}`, { isActive }),
   396→    onSuccess: () => {
   397→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   398→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   399→      queryClient.invalidateQueries({ queryKey: queryKeys.aiSettings.all });
   400→      queryClient.invalidateQueries({ queryKey: queryKeys.aiPrompts.all });
   401→    },
   402→    onError: (err: Error) => {
   403→      toast.error(err.message || 'Failed to update provider');
   404→    },
   405→  });
   406→
   407→  // Handlers
   408→  const handleOpenCreate = () => {
   409→    setEditingProvider(null);
   410→    setFormData(emptyForm);
   411→    setShowApiKey(false);
   412→    setDialogOpen(true);
   413→  };
   414→
   415→  const handleOpenEdit = (provider: AiProvider) => {
   416→    setEditingProvider(provider);
   417→    setFormData({
   418→      name: provider.name,
   419→      kind: provider.kind,
   420→      baseUrl: provider.baseUrl,
   421→      apiKey: '', // masked in edit
   422→      apiVersion: provider.apiVersion ?? '',
   423→      isActive: provider.isActive,
   424→    });
   425→    setShowApiKey(false);
   426→    setDialogOpen(true);
   427→  };
   428→
   429→  const handleCloseDialog = () => {
   430→    setDialogOpen(false);
   431→    setEditingProvider(null);
   432→    setFormData(emptyForm);
   433→    setShowApiKey(false);
   434→  };
   435→
   436→  const handleSave = () => {
   437→    if (!formData.name.trim()) {
   438→      toast.error('Name is required');
   439→      return;
   440→    }
   441→    if (!editingProvider && !formData.apiKey.trim()) {
   442→      toast.error('API key is required');
   443→      return;
   444→    }
   445→    // CUSTOM providers require a Base URL
   446→    if (formData.kind === 'CUSTOM' && !formData.baseUrl.trim()) {
   447→      toast.error('Base URL is required for Custom providers');
   448→      return;
   449→    }
   450→    // Validate Base URL format for CUSTOM providers
   451→    if (formData.kind === 'CUSTOM' && formData.baseUrl.trim()) {
   452→      try {
   453→        const u = new URL(formData.baseUrl);
   454→        if (!['http:', 'https:'].includes(u.protocol)) {
   455→          toast.error('Base URL must use http or https protocol');
   456→          return;
   457→        }
   458→      } catch {
   459→        toast.error('Base URL is not a valid URL');
   460→        return;
   461→      }
   462→    }
   463→    saveMutation.mutate(formData);
   464→  };
   465→
   466→  const handleKindChange = (kind: string) => {
   467→    const providerKind = kind as AiProviderKind;
   468→    const prevDefaultUrl = PROVIDER_CONFIGS[formData.kind]?.defaultUrl ?? '';
   469→    setFormData((prev) => ({
   470→      ...prev,
   471→      kind: providerKind,
   472→      baseUrl:
   473→        !prev.baseUrl || prev.baseUrl === prevDefaultUrl
   474→          ? PROVIDER_CONFIGS[providerKind].defaultUrl
   475→          : prev.baseUrl,
   476→    }));
   477→  };
   478→
   479→  // KPI calculations
   480→  const totalProviders = pagination?.total ?? 0;
   481→  const connectedCount = providers.filter((p) => p.connectionStatus === 'CONNECTED').length;
   482→  const defaultProvider = providers.find((p) => p.isDefault);
   483→
   484→  return (
   485→    <div className="space-y-6">
   486→      {/* KPI Cards */}
   487→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   488→        <Card>
   489→          <CardContent className="p-4 flex items-center gap-4">
   490→            <div className="p-2 rounded-lg bg-zinc-100">
   491→              <Server className="h-5 w-5 text-zinc-600 dark:text-zinc-400" />
   492→            </div>
   493→            <div>
   494→              <p className="text-sm text-zinc-500">Total Providers</p>
   495→              <div className="text-2xl font-bold">{isLoading ? <Skeleton className="h-8 w-12 inline-block" /> : totalProviders}</div>
   496→            </div>
   497→          </CardContent>
   498→        </Card>
   499→        <Card>
   500→          <CardContent className="p-4 flex items-center gap-4">
   501→            <div className="p-2 rounded-lg bg-green-100">
   502→              <Wifi className="h-5 w-5 text-green-600" />
   503→            </div>
   504→            <div>
