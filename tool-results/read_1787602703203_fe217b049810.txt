   199→interface InviteFormData {

   207→function InviteUserDialog({
   208→  open,
   209→  onOpenChange,
   210→  onSubmit,
   211→  isLoading,
   212→  editMode = false,
   213→  initialData,
   214→}: {
   215→  open: boolean;
   216→  onOpenChange: (open: boolean) => void;
   217→  onSubmit: (data: InviteFormData) => void;
   218→  isLoading: boolean;
   219→  editMode?: boolean;
   220→  initialData?: InviteFormData | null;
   221→}) {
   222→  const [form, setForm] = useState<InviteFormData>({ email: '', name: '', role: 'AUTHOR', assignedSites: [], sitePermissions: [] });
   223→  const [errors, setErrors] = useState<Record<string, string>>({});
   224→  const [siteSearch, setSiteSearch] = useState('');
   225→  const [showSiteDropdown, setShowSiteDropdown] = useState(false);
   226→  const [permissionsExpanded, setPermissionsExpanded] = useState(false);
   227→  const dropdownRef = React.useRef<HTMLDivElement>(null);
   228→
   229→  const { data: sitesData } = useQuery({
   230→    queryKey: ['sites-list-invite'],
   231→    queryFn: () => getApi<SiteOption[]>('/api/sites'),
   232→    enabled: open,
   233→  });
   234→  const sites = sitesData ?? [];
   235→
   236→  React.useEffect(() => {
   237→    const handler = (e: MouseEvent) => {
   238→      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node)) {
   239→        setShowSiteDropdown(false);
   240→      }
   241→    };
   242→    if (showSiteDropdown) document.addEventListener('mousedown', handler);
   243→    return () => document.removeEventListener('mousedown', handler);
   244→  }, [showSiteDropdown]);
   245→
   246→  React.useEffect(() => {
   247→    if (open) {
   248→      if (editMode && initialData) {
   249→        setForm(initialData);
   250→      } else {
   251→        setForm({ email: '', name: '', role: 'AUTHOR', assignedSites: [], sitePermissions: [] });
   252→      }
   253→      setErrors({});
   254→      setSiteSearch('');
   255→      setPermissionsExpanded(false);
   256→    }
   257→  }, [open, editMode, initialData]);
   258→
   259→  const filteredSites = React.useMemo(() => {
   260→    if (!siteSearch.trim()) return sites;
   261→    const q = siteSearch.toLowerCase();
   262→    return sites.filter((s) => s.name.toLowerCase().includes(q) || (s.domain && s.domain.toLowerCase().includes(q)));
   263→  }, [sites, siteSearch]);
   264→
   265→  const selectedSiteObjects = React.useMemo(() => {
   266→    return form.assignedSites.map((id) => sites.find((s) => s.id === id)).filter(Boolean) as SiteOption[];
   267→  }, [form.assignedSites, sites]);
   268→
   269→  const handleRoleChange = (role: string) => {
   270→    const presetPerms = ROLE_PRESETS[role] ?? [];
   271→    setForm((p) => ({ ...p, role, sitePermissions: presetPerms }));
   272→  };
   273→
   274→  const togglePermission = (key: string) => {
   275→    setForm((p) => ({
   276→      ...p,
   277→      sitePermissions: p.sitePermissions.includes(key)
   278→        ? p.sitePermissions.filter((k) => k !== key)
   279→        : [...p.sitePermissions, key],
   280→    }));
   281→  };
   282→
   283→  const addSite = (id: string) => {
   284→    if (!form.assignedSites.includes(id)) {
   285→      setForm((p) => ({ ...p, assignedSites: [...p.assignedSites, id] }));
   286→    }
   287→    setShowSiteDropdown(false);
   288→    setSiteSearch('');
   289→  };
   290→
   291→  const removeSite = (id: string) => {
   292→    setForm((p) => ({ ...p, assignedSites: p.assignedSites.filter((s) => s !== id) }));
   293→  };
   294→
   295→  const addAllSites = () => {
   296→    setForm((p) => ({ ...p, assignedSites: sites.map((s) => s.id) }));
   297→    setShowSiteDropdown(false);
   298→  };
   299→
   300→  const handleSubmit = () => {
   301→    const errs: Record<string, string> = {};
   302→    if (!form.email.trim()) errs.email = 'Email is required';
   303→    else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) errs.email = 'Invalid email address';
   304→    if (Object.keys(errs).length > 0) { setErrors(errs); return; }
   305→    onSubmit({ ...form, email: form.email.trim(), name: form.name.trim() });
   306→  };
   307→
   308→  return (
   309→    <Dialog open={open} onOpenChange={onOpenChange}>
   310→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   311→        <DialogHeader>
   312→          <DialogTitle>{editMode ? 'Edit User' : 'Invite User'}</DialogTitle>
   313→          <DialogDescription>
   314→            {editMode
   315→              ? 'Update user details, role, and permissions.'
   316→              : 'Send an invitation email to add a new team member to your organization.'}
   317→          </DialogDescription>
   318→        </DialogHeader>
   319→        <div className="grid gap-6 py-2">
   320→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   321→            <div className="grid gap-2">
   322→              <Label htmlFor="invite-email">Email <span className="text-destructive">*</span></Label>
   323→              <Input id="invite-email" type="email" value={form.email}
   324→                onChange={(e) => { setForm((p) => ({ ...p, email: e.target.value })); setErrors((p) => { const n = { ...p }; delete n.email; return n; }); }}
   325→                placeholder="user@example.com" autoFocus />
   326→              {errors.email && <p className="text-xs text-destructive">{errors.email}</p>}
   327→            </div>
   328→            <div className="grid gap-2">
   329→              <Label htmlFor="invite-name">Name</Label>
   330→              <Input id="invite-name" value={form.name}
   331→                onChange={(e) => setForm((p) => ({ ...p, name: e.target.value }))}
   332→                placeholder="Full name (optional)" />
   333→            </div>
   334→          </div>
   335→          <div className="grid gap-2">
   336→            <Label htmlFor="invite-role">Role Preset</Label>
   337→            <Select value={form.role} onValueChange={handleRoleChange}>
   338→              <SelectTrigger id="invite-role"><SelectValue placeholder="Select a role" /></SelectTrigger>
   339→              <SelectContent>
   340→                {Object.keys(ROLE_PRESETS).map((role) => (
   341→                  <SelectItem key={role} value={role}>
   342→                    {role === 'SUPER_ADMIN' ? 'Super Admin' : role.charAt(0) + role.slice(1).toLowerCase().replace(/_/g, ' ')}
   343→                  </SelectItem>
   344→                ))}
   345→              </SelectContent>
   346→            </Select>
   347→            <p className="text-xs text-muted-foreground">Selecting a role automatically sets recommended permissions. You can customize below.</p>
   348→          </div>
   349→          <div className="grid gap-2">
   350→            <Label>Assigned Sites</Label>
   351→            <div className="relative" ref={dropdownRef}>
   352→              <div className="flex flex-wrap gap-1.5 min-h-[42px] p-2 border rounded-md bg-background cursor-text"
   353→                onClick={() => setShowSiteDropdown(true)}>
   354→                {selectedSiteObjects.length === 0 && !showSiteDropdown && (
   355→                  <span className="text-sm text-muted-foreground">Search and select sites...</span>
   356→                )}
   357→                {selectedSiteObjects.map((site) => (
   358→                  <Badge key={site.id} variant="secondary" className="gap-1.5 pr-1 py-1 text-xs font-normal">
   359→                    {site.logo ? <img src={site.logo} alt="" className="h-3.5 w-3.5 rounded-sm object-cover" /> : <Globe className="h-3.5 w-3.5 text-muted-foreground" />}
   360→                    <span>{site.name}</span>
   361→                    {site.domain && <span className="text-muted-foreground">({site.domain})</span>}
   362→                    <button type="button" className="ml-0.5 hover:bg-muted rounded-full p-0.5"
   363→                      onClick={(e) => { e.stopPropagation(); removeSite(site.id); }}>
   364→                      <svg xmlns="http://www.w3.org/2000/svg" width="10" height="10" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round"><path d="M18 6 6 18"/><path d="m6 6 12 12"/></svg>
   365→                    </button>
   366→                  </Badge>
   367→                ))}
   368→                {showSiteDropdown && (
   369→                  <input type="text" className="flex-1 min-w-[120px] bg-transparent border-none outline-none text-sm"
   370→                    placeholder="Search sites..." value={siteSearch} onChange={(e) => setSiteSearch(e.target.value)}
   371→                    autoFocus onKeyDown={(e) => { if (e.key === 'Escape') setShowSiteDropdown(false); }} />
   372→                )}
   373→              </div>
   374→              {showSiteDropdown && (
   375→                <div className="absolute z-50 mt-1 w-full bg-popover border rounded-md shadow-lg max-h-48 overflow-y-auto">
   376→                  <button type="button" className="w-full px-3 py-2 text-sm text-left hover:bg-accent flex items-center gap-2 border-b"
   377→                    onClick={addAllSites}>
   378→                    <Globe className="h-4 w-4 text-muted-foreground" />
   379→                    <span>Select All Sites</span>
   380→                    <span className="text-xs text-muted-foreground ml-auto">({sites.length})</span>
   381→                  </button>
   382→                  {filteredSites.length === 0 ? (
   383→                    <div className="px-3 py-2 text-sm text-muted-foreground">No sites found</div>
   384→                  ) : (
   385→                    filteredSites.map((site) => {
   386→                      const isSelected = form.assignedSites.includes(site.id);
   387→                      return (
   388→                        <button key={site.id} type="button"
   389→                          className={cn('w-full px-3 py-2 text-sm text-left hover:bg-accent flex items-center gap-3', isSelected && 'bg-accent')}
   390→                          onClick={() => isSelected ? removeSite(site.id) : addSite(site.id)}>
   391→                          {site.logo ? <img src={site.logo} alt="" className="h-5 w-5 rounded object-cover" />
   392→                            : <div className="h-5 w-5 rounded bg-muted flex items-center justify-center"><Globe className="h-3 w-3 text-muted-foreground" /></div>}
   393→                          <div className="flex-1 min-w-0">
   394→                            <div className="truncate font-medium">{site.name}</div>
   395→                            {site.domain && <div className="text-xs text-muted-foreground truncate">{site.domain}</div>}
   396→                          </div>
   397→                          {isSelected && <div className="h-4 w-4 rounded-full bg-primary flex items-center justify-center"><svg xmlns="http://www.w3.org/2000/svg" width="10" height="10" viewBox="0 0 24 24" fill="none" stroke="white" strokeWidth="3" strokeLinecap="round" strokeLinejoin="round"><polyline points="20 6 9 17 4 12"/></svg></div>}
   398→                        </button>
   399→                      );
   400→                    })
   401→                  )}
   402→                </div>
   403→              )}
   404→            </div>
   405→            <p className="text-xs text-muted-foreground">Assign the user to one or multiple websites. Leave empty for all sites.</p>
   406→          </div>
   407→          <div className="grid gap-2">
   408→            <div className="flex items-center justify-between">
   409→              <Label>Site Permissions</Label>
   410→              <button type="button" className="text-xs text-muted-foreground hover:text-foreground transition-colors"
   411→                onClick={() => setPermissionsExpanded(!permissionsExpanded)}>
   412→                {permissionsExpanded ? 'Collapse' : `Expand (${form.sitePermissions.length} selected)`}
   413→              </button>
   414→            </div>
   415→            {permissionsExpanded ? (
   416→              <div className="border rounded-lg divide-y max-h-64 overflow-y-auto">
   417→                {SITE_PERMISSIONS.map((perm) => (
   418→                  <label key={perm.key} className="flex items-start gap-3 px-3 py-2.5 hover:bg-accent/50 cursor-pointer transition-colors">
   419→                    <input type="checkbox" className="mt-0.5 h-4 w-4 rounded border-input"
   420→                      checked={form.sitePermissions.includes(perm.key)} onChange={() => togglePermission(perm.key)} />
   421→                    <div className="flex-1 min-w-0">
   422→                      <div className="text-sm font-medium">{perm.label}</div>
   423→                      <div className="text-xs text-muted-foreground mt-0.5">{perm.description}</div>
   424→                    </div>
   425→                  </label>
   426→                ))}
   427→              </div>
   428→            ) : (
   429→              <div className="border rounded-lg p-3">
   430→                <div className="flex flex-wrap gap-1.5">
   431→                  {form.sitePermissions.length === 0 ? (
   432→                    <span className="text-sm text-muted-foreground">No permissions selected</span>
   433→                  ) : (
   434→                    form.sitePermissions.map((key) => {
   435→                      const perm = SITE_PERMISSIONS.find((p) => p.key === key);
   436→                      return perm ? <Badge key={key} variant="secondary" className="text-xs font-normal">{perm.label}</Badge> : null;
   437→                    })
   438→                  )}
   439→                </div>
   440→              </div>
   441→            )}
   442→          </div>
   443→        </div>
   444→        <DialogFooter>
   445→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={isLoading}>Cancel</Button>
   446→          <Button onClick={handleSubmit} disabled={isLoading || !form.email.trim()}>
   447→            {isLoading && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   448→            {editMode ? 'Save Changes' : 'Send Invitation'}
   449→          </Button>
   450→        </DialogFooter>
   451→      </DialogContent>
   452→    </Dialog>
   453→  );
   454→}

   485→      ...(roleFilter !== 'all' ? { role: roleFilter } : {}),
   486→      ...(statusFilter !== 'all' ? { status: statusFilter } : {}),
   487→    }),
   488→    [
   489→      table.currentPage,
   490→      table.pageSize,
   491→      table.sortField,
   492→      table.sortOrder,
   493→      table.searchValue,
   494→      roleFilter,
   495→      statusFilter,
   496→    ],
   497→  );
   498→
   499→  // Fetch users — use raw:true to get the full ApiResponse envelope
   500→  // (getApi without raw unwraps the `data` field, losing pagination info)
   501→  const { data: rawData, isLoading } = useQuery({
   502→    queryKey: queryKeys.users.list(queryParams),
   503→    queryFn: () =>
   504→      getApi<{ data: UserRow[]; meta: { pagination: { page: number; pageSize: number; total: number; totalPages: number } } }>(
   505→        '/api/users',
   506→        queryParams,
   507→        { raw: true },
   508→      ),
   509→    staleTime: 10_000,
   510→  });
   511→
   512→  const users = rawData?.data ?? [];
   513→  const pagination = rawData?.meta?.pagination;
   514→  const totalItems = pagination?.total ?? 0;
   515→
   516→  // Delete mutation
   517→  const deleteMutation = useMutation({
   518→    mutationFn: (id: string) => deleteApi(`/api/users/${id}`),
   519→    onSuccess: () => {
   520→      queryClient.invalidateQueries({ queryKey: queryKeys.users.all });
   521→      setDeleteTarget(null);
   522→    },
   523→  });
   524→
   525→  // Suspend/Activate toggle mutation
   526→  const toggleStatusMutation = useMutation({
   527→    mutationFn: ({
   528→      id,
   529→      status,
   530→    }: {
   531→      id: string;
   532→      status: UserStatus;
   533→    }) =>
   534→      patchApi(`/api/users/${id}`, { status }),
   535→    onSuccess: () => {
   536→      queryClient.invalidateQueries({ queryKey: queryKeys.users.all });
   537→      setSuspendTarget(null);
   538→    },
   539→  });
   540→
   541→  // Invite mutation
   542→  const inviteMutation = useMutation({
   543→    mutationFn: (data: InviteFormData) => postApi('/api/users/invite', data),
   544→    onSuccess: () => {
   545→      queryClient.invalidateQueries({ queryKey: queryKeys.users.all });
   546→      setInviteDialogOpen(false);
   547→    },
   548→  });
   549→
   550→  // Edit dialog state — reuses the Invite User form in edit mode
   551→  const [editDialogOpen, setEditDialogOpen] = useState(false);
   552→  const [editingUser, setEditingUser] = useState<UserRow | null>(null);
   553→
   554→  // Open the Invite User-style dialog in edit mode
   555→  const openEditDialog = useCallback((user: UserRow) => {
   556→    setEditingUser(user);
   557→    setEditDialogOpen(true);
   558→  }, []);
   559→
   560→  // Edit mutation (updates existing user)
   561→  const editMutation = useMutation({
   562→    mutationFn: ({ id, data }: { id: string; data: InviteFormData }) =>
   563→      patchApi(`/api/users/${id}`, {
   564→        name: data.name,
   565→        email: data.email,
   566→        role: data.role,
   567→      }),
   568→    onSuccess: () => {
   569→      queryClient.invalidateQueries({ queryKey: queryKeys.users.all });
   570→      setEditDialogOpen(false);
   571→      setEditingUser(null);
   572→      toast.success('User updated successfully');
   573→    },
   574→    onError: (err: Error) => {
   575→      toast.error(err.message || 'Failed to update user');
   576→    },
   577→  });
   578→
   579→  const handleSuspendToggle = useCallback(
   580→    (user: UserRow) => {
   581→      const newStatus: UserStatus =
   582→        user.status === 'SUSPENDED' ? 'ACTIVE' : 'SUSPENDED';
   583→      toggleStatusMutation.mutate({ id: user.id, status: newStatus });
   584→      setSuspendTarget(null);
   585→    },
   586→    [toggleStatusMutation],
   587→  );
   588→
   589→  // Column definitions
   590→  const columns = useMemo<ColumnDef<UserRow>[]>(
   591→    () => [
   592→      {
   593→        id: 'name',
   594→        header: ({ column }) => (
   595→          <button
   596→            className="flex items-center gap-1 hover:text-foreground transition-colors"
   597→            onClick={() =>
   598→              column.toggleSorting(column.getIsSorted() === 'asc')
   599→            }
   600→          >
   601→            <span className="font-medium">User</span>
   602→          </button>
   603→        ),
   604→        accessorFn: (row) => row.name ?? row.email,
   605→        size: 240,
   606→        enableSorting: true,
   607→        cell: ({ row }) => {
   608→          const user = row.original;
   609→          return (
   610→            <button
   611→              className="flex items-center gap-2.5 hover:opacity-80 transition-opacity text-left w-full"
   612→              onClick={(e) => {
   613→                e.stopPropagation();
   614→                openEditDialog(row);
   615→              }}
   616→            >
   617→              <AvatarWithFallback
   618→                src={user.avatar ?? undefined}
   619→                name={user.name ?? user.email}
   620→                size="sm"
   621→              />
   622→              <div className="min-w-0 flex-1">
   623→                <div className="truncate font-medium text-sm">
   624→                  {user.name || 'Unnamed'}
   625→                </div>
   626→              </div>
   627→            </button>
   628→          );
   629→        },
   630→      } as ColumnDef<UserRow>,
   631→      ColumnDefHelper.textColumn<UserRow>({
   632→        id: 'email',
   633→        header: 'Email',
   634→        accessorKey: 'email',
   635→        truncate: 40,
   636→        enableSorting: false,
   637→      }),
   638→      {
   639→        id: 'role',
   640→        header: 'Role',
   641→        accessorKey: 'role',
   642→        size: 130,
   643→        enableSorting: true,
   644→        cell: ({ getValue }) => {
   645→          const role = getValue() as UserRole;
   646→          return <RoleBadge role={role} />;
   647→        },
   648→      } as ColumnDef<UserRow>,
   649→      {
   650→        id: 'status',
   651→        header: 'Status',
   652→        accessorKey: 'status',
   653→        size: 120,
   654→        enableSorting: true,
   655→        cell: ({ getValue }) => {
   656→          const status = getValue() as string;
   657→          return <StatusBadge status={status} size="sm" />;
   658→        },
   659→      } as ColumnDef<UserRow>,
   660→      ColumnDefHelper.dateColumn<UserRow>({
   661→        id: 'lastLoginAt',
   662→        header: 'Last Login',
   663→        accessorKey: 'lastLoginAt',
   664→        format: (d) => formatRelativeTime(d),
   665→      }),
   666→      ColumnDefHelper.dateColumn<UserRow>({
   667→        id: 'createdAt',
   668→        header: 'Created',
   669→        accessorKey: 'createdAt',
   670→        format: (d) => formatDate(d),
   671→      }),
   672→      ColumnDefHelper.actionColumn<UserRow>({
   673→        id: 'actions',
   674→        size: 60,
   675→        render: (row) => (
   676→          <DropdownMenu>
   677→            <DropdownMenuTrigger asChild>
   678→              <Button
   679→                variant="ghost"
   680→                size="icon"
   681→                className="h-8 w-8"
   682→                onClick={(e) => e.stopPropagation()}
   683→              >
   684→                <MoreHorizontal className="h-4 w-4" />
   685→                <span className="sr-only">Actions</span>
   686→              </Button>
   687→            </DropdownMenuTrigger>
   688→            <DropdownMenuContent align="end">
   689→              <DropdownMenuItem onClick={() => openEditDialog(row)}>
   690→                <Pencil className="h-4 w-4 mr-2" />
   691→                Edit
   692→              </DropdownMenuItem>
   693→              <DropdownMenuSeparator />
   694→              <DropdownMenuItem
   695→                onClick={() => setSuspendTarget(row)}
   696→              >
   697→                {row.status === 'SUSPENDED' ? (
   698→                  <>
   699→                    <UserCheck className="h-4 w-4 mr-2" />
   700→                    Activate
   701→                  </>
   702→                ) : (
   703→                  <>
   704→                    <UserX className="h-4 w-4 mr-2" />
   705→                    Suspend
   706→                  </>
   707→                )}
   708→              </DropdownMenuItem>
   709→              <DropdownMenuSeparator />
   710→              <DropdownMenuItem
   711→                variant="destructive"
   712→                onClick={() => setDeleteTarget(row)}
   713→              >
   714→                <Trash2 className="h-4 w-4 mr-2" />
   715→                Delete
   716→              </DropdownMenuItem>
   717→            </DropdownMenuContent>
   718→          </DropdownMenu>
   719→        ),
   720→      }),
   721→    ],
   722→    [openEditDialog],
   723→  );
   724→
   725→  // Filter controls
   726→  const filterContent = (
   727→    <div className="flex items-center gap-2">
   728→      <Select
   729→        value={roleFilter}
   730→        onValueChange={(v) => {
   731→          setRoleFilter(v);
   732→          table.setCurrentPage(1);
   733→        }}
   734→      >
   735→        <SelectTrigger size="sm" className="w-[130px] h-9">
   736→          <SelectValue placeholder="All Roles" />
   737→        </SelectTrigger>
   738→        <SelectContent>
   739→          <SelectItem value="all">All Roles</SelectItem>
   740→          {ROLE_OPTIONS.map((opt) => (
   741→            <SelectItem key={opt.value} value={opt.value}>
   742→              {opt.label}
   743→            </SelectItem>
   744→          ))}
   745→        </SelectContent>
   746→      </Select>
   747→      <Select
   748→        value={statusFilter}
   749→        onValueChange={(v) => {
   750→          setStatusFilter(v);
   751→          table.setCurrentPage(1);
   752→        }}
   753→      >
   754→        <SelectTrigger size="sm" className="w-[130px] h-9">
   755→          <SelectValue placeholder="All Statuses" />
   756→        </SelectTrigger>
   757→        <SelectContent>
   758→          <SelectItem value="all">All Statuses</SelectItem>
   759→          {STATUS_OPTIONS.map((opt) => (
   760→            <SelectItem key={opt.value} value={opt.value}>
   761→              {opt.label}
   762→            </SelectItem>
   763→          ))}
   764→        </SelectContent>
   765→      </Select>
   766→    </div>
   767→  );
   768→
   769→  return (
   770→    <div className="space-y-4">
   771→      <PageHeader
   772→        title="Users"
   773→        description="Manage user accounts and permissions"
   774→        action={
   775→          <Button
   776→            size="sm"
   777→            onClick={() => setInviteDialogOpen(true)}
   778→          >
   779→            <UserPlus className="h-4 w-4 mr-2" />
   780→            Invite User
   781→          </Button>
   782→        }
   783→      />
   784→
   785→      <DataTable
   786→        columns={columns}
   787→        data={users}
   788→        isLoading={isLoading}
   789→        totalItems={totalItems}
   790→        pageSize={table.pageSize}
   791→        currentPage={table.currentPage}
   792→        onPageChange={(p) => table.setCurrentPage(p)}
   793→        onSortChange={(field, order) => table.setSortField(field, order)}
   794→        sortField={table.sortField}
   795→        sortOrder={table.sortOrder}
   796→        onRowClick={(row) => openEditDialog(row)}
   797→        searchPlaceholder="Search users..."
   798→        searchValue={table.searchValue}
   799→        onSearch={(v) => {
   800→          table.setSearchValue(v);
   801→          table.setCurrentPage(1);
   802→        }}
   803→        filterContent={filterContent}
   804→        getRowId={(row) => row.id}
   805→        emptyMessage="No users found."
   806→      />
   807→
   808→      {/* Invite Dialog */}
   809→      <InviteUserDialog
   810→        open={inviteDialogOpen}
   811→        onOpenChange={setInviteDialogOpen}
   812→        onSubmit={(d) => inviteMutation.mutate(d)}
   813→        isLoading={inviteMutation.isPending}
   814→      />
   815→
   816→      {/* Edit User Dialog — reuses InviteUserDialog in edit mode */}
   817→      <InviteUserDialog
   818→        open={editDialogOpen}
   819→        onOpenChange={(open) => {
   820→          setEditDialogOpen(open);
   821→          if (!open) setEditingUser(null);
   822→        }}
   823→        onSubmit={(data) => {
   824→          if (editingUser) {
   825→            editMutation.mutate({ id: editingUser.id, data });
   826→          }
   827→        }}
   828→        isLoading={editMutation.isPending}
   829→        editMode
   830→        initialData={editingUser ? {
   831→          email: editingUser.email,
   832→          name: editingUser.name ?? '',
   833→          role: editingUser.role,
   834→          assignedSites: [],
   835→          sitePermissions: [],
   836→        } : null}
   837→      />
   838→
   839→      {/* Delete Confirmation */}
   840→      <ConfirmDialog
   841→        open={!!deleteTarget}
   842→        onOpenChange={(open) => !open && setDeleteTarget(null)}
   843→        title="Delete User"
   844→        description={
   845→          deleteTarget
   846→            ? `Are you sure you want to delete "${deleteTarget.name || deleteTarget.email}"? This action cannot be undone.`
   847→            : undefined
   848→        }
   849→        confirmLabel="Delete"
   850→        variant="destructive"
   851→        onConfirm={() => {
   852→          if (deleteTarget) deleteMutation.mutate(deleteTarget.id);
   853→        }}
   854→        isLoading={deleteMutation.isPending}
   855→      />
   856→
   857→      {/* Suspend/Activate Confirmation */}
   858→      <ConfirmDialog
   859→        open={!!suspendTarget}
   860→        onOpenChange={(open) => !open && setSuspendTarget(null)}
   861→        title={
   862→          suspendTarget?.status === 'SUSPENDED'
   863→            ? 'Activate User'
   864→            : 'Suspend User'
   865→        }
   866→        description={
   867→          suspendTarget
   868→            ? suspendTarget.status === 'SUSPENDED'
   869→              ? `Are you sure you want to activate "${suspendTarget.name || suspendTarget.email}"? They will regain access.`
   870→              : `Are you sure you want to suspend "${suspendTarget.name || suspendTarget.email}"? They will lose access immediately.`
   871→            : undefined
   872→        }
   873→        confirmLabel={
   874→          suspendTarget?.status === 'SUSPENDED' ? 'Activate' : 'Suspend'
   875→        }
   876→        variant={
   877→          suspendTarget?.status === 'SUSPENDED' ? 'default' : 'destructive'
   878→        }
   879→        onConfirm={() => {
   880→          if (suspendTarget) handleSuspendToggle(suspendTarget);
   881→        }}
   882→        isLoading={toggleStatusMutation.isPending}
   883→      />
   884→    </div>
   885→  );
   886→}
