     1→// ============================================================
     2→// PLATFORM DATA SERVICE — Centralized SaaS platform dataset
     3→// ============================================================
     4→// This is the SINGLE source of truth for the entire SaaS platform.
     5→// Both the Client Billing experience AND the Platform Admin
     6→// Dashboard read from and mutate this same in-memory dataset, so
     7→// a plan change, cancellation, or payment made on the client side
     8→// is immediately reflected on the admin side, and vice-versa.
     9→//
    10→// Consistency rules (enforced by derivation, never hardcoded):
    11→//  - totalCustomers  = customers.length (non-deleted)
    12→//  - activeSubscriptions = subscriptions where status === 'active'
    13→//  - mrr = sum of normalized monthly price over active PAID subs
    14→//  - totalSites = sites.length
    15→//  - planDistribution = subscriptions grouped by plan (all statuses)
    16→//  - statusCounts = subscriptions grouped by status
    17→//  - recentCustomers = customers sorted by createdAt desc
    18→//  - recentPayments = payments sorted by date desc
    19→//  - usage = aggregated from sites/content/media/AI records
    20→//  - alerts = derived from failed payments + past-due + storage
    21→//
    22→// Mock data is deterministic (no Math.random) so numbers are stable
    23→// across renders and never contradict each other.
    24→// ============================================================

   142→// -------------------- Plans (delegated to DB-backed plan-config) --------------------
   143→// The editable plan definitions live in the PlanConfig table and are cached
   144→// by ./plan-config — the SINGLE source of truth. The owner edits there via
   145→// the Platform Admin; this module (admin overview + client billing) and the
   146→// client billing API both read the same cached values, so an owner price
   147→// change propagates to the client billing page and to MRR on the next read.
   148→// The PLANS array below is a live view re-spliced whenever the config changes.
   149→
   150→import { getPlanConfigsSync, subscribe as subscribePlanConfig, type PlanConfigData } from './plan-config';
   151→
   152→function toPlan(d: PlanConfigData): Plan {
   153→  return {
   154→    id: d.planId as PlanId,
   155→    name: d.name,
   156→    price: d.priceMonthly,
   157→    priceMonthly: d.priceMonthly,
   158→    priceYearly: d.priceYearly,
   159→    currency: d.currency,
   160→    interval: d.interval,
   161→    isFree: d.isFree,
   162→    active: d.active,
   163→    features: d.features,
   164→    entitlements: d.entitlements,
   165→    limits: d.limits,
   166→  };
   167→}
   168→
   169→/** Live view of the DB-backed plan configs. Mutated in place on every
   170→ *  config refresh so existing references stay valid. */
   171→export const PLANS: Plan[] = [];
   172→
   173→function refreshPLANS(): void {
   174→  const data = getPlanConfigsSync();
   175→  PLANS.length = 0;
   176→  for (const d of data) PLANS.push(toPlan(d));
   177→}
   178→refreshPLANS();
   179→subscribePlanConfig(refreshPLANS);
   180→
   181→export function getPlan(id: PlanId): Plan {
   182→  return PLANS.find((p) => p.id === id) ?? PLANS[0];
   183→}
   184→
   185→/** Normalized monthly price. Yearly subscriptions contribute priceYearly/12
   186→ *  so MRR is a true monthly run-rate. */
   187→export function monthlyPrice(plan: Plan, interval: BillingInterval): number {
   188→  if (interval === 'yearly') return Math.round(plan.priceYearly / 12);
   189→  return plan.priceMonthly;
   190→}

   197→function isoDaysAgo(days: number): string {
   198→  const d = new Date();
   199→  d.setDate(d.getDate() - days);
   200→  d.setHours(9, 15, 0, 0);
   201→  return d.toISOString();
   202→}
   203→function isoDaysAhead(days: number): string {
   204→  const d = new Date();
   205→  d.setDate(d.getDate() + days);
   206→  d.setHours(9, 15, 0, 0);
   207→  return d.toISOString();
   208→}

   368→function buildRevenueSeries(currentMrr: number): RevenuePoint[] {
   369→  const trend = [0.41, 0.47, 0.54, 0.62, 0.71, 0.80, 0.89, 1.0];
   370→  const out: RevenuePoint[] = [];
   371→  const now = new Date();
   372→  for (let i = 7; i >= 0; i--) {
   373→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   374→    const factor = trend[7 - i];
   375→    out.push({ month: MONTHS[d.getMonth()], revenue: Math.round(currentMrr * factor) });
   376→  }
   377→  return out;
   378→}

   384→interface PlatformStore {
   385→  customers: Customer[];
   386→  sites: PlatformSite[];
   387→  payments: Payment[];
   388→  audit: AuditEntry[];
   389→}
   390→
   391→let _store: PlatformStore | null = null;
   392→let _auditCounter = 0;
   393→let _paymentCounter = 0;
   394→
   395→function createStore(): PlatformStore {
   396→  const customers: Customer[] = CUSTOMER_SEED.map((c, i) => ({
   397→    ...c,
   398→    id: `cus_${String(i + 1).padStart(3, '0')}`,
   399→  }));
   400→
   401→  const sites: PlatformSite[] = SITE_SEED.map((s, i) => ({
   402→    id: `site_p${String(i + 1).padStart(3, '0')}`,
   403→    customerId: customers[s.customerIdIdx].id,
   404→    name: s.name,
   405→    slug: s.slug,
   406→    domain: s.domain,
   407→    status: s.status,
   408→    articles: s.articles,
   409→    media: s.media,
   410→    storageBytes: s.storageBytes,
   411→    createdAt: isoDaysAgo(s.daysAgo),
   412→  }));
   413→
   414→  const payments: Payment[] = PAYMENT_SEED.map((p, i) => {
   415→    const c = customers[p.customerIdx];
   416→    return {
   417→      id: `pay_${String(i + 1).padStart(3, '0')}`,
   418→      customerId: c.id,
   419→      planId: p.planId,
   420→      amount: p.amount,
   421→      currency: 'CHF',
   422→      status: p.status,
   423→      method: p.method,
   424→      date: isoDaysAgo(p.daysAgo),
   425→      invoiceNumber: `INV-2026-${String(1000 + i)}`,
   426→    };
   427→  });
   428→
   429→  const audit: AuditEntry[] = INITIAL_AUDIT.map((a, i) => ({
   430→    ...a,
   431→    id: `aud_${String(i + 1).padStart(3, '0')}`,
   432→  }));
   433→
   434→  _auditCounter = audit.length;
   435→  _paymentCounter = payments.length;
   436→
   437→  return { customers, sites, payments, audit };
   438→}

   488→  const customers = s.customers;
   489→
   490→  const totalCustomers = customers.length;
   491→  const activeSubscriptions = customers.filter((c) => c.subscriptionStatus === 'active').length;
   492→
   493→  let mrr = 0;
   494→  for (const c of customers) {
   495→    if (c.subscriptionStatus === 'active' && c.planId !== 'free') {
   496→      mrr += monthlyPrice(getPlan(c.planId), c.billingInterval);
   497→    }
   498→  }
   499→
   500→  const totalSites = s.sites.length;
   501→
   502→  const planDistribution = PLANS.map((p) => ({
   503→    planId: p.id,
   504→    planName: p.name,
   505→    count: customers.filter((c) => c.planId === p.id).length,
   506→    color: p.id === 'free' ? '#10b981' : p.id === 'plus' ? '#f59e0b' : p.id === 'pro' ? '#8b5cf6' : '#ec4899',
   507→  }));
   508→
   509→  const statuses: SubscriptionStatus[] = ['active', 'trial', 'past_due', 'cancelled', 'expired'];
   510→  const statusCounts = statuses
   511→    .map((status) => ({ status, count: customers.filter((c) => c.subscriptionStatus === status).length }))
   512→    .filter((x) => x.count > 0);
   513→
   514→  const recentCustomers = [...customers]
   515→    .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   516→    .slice(0, 6)
   517→    .map((c) => ({ ...c, siteCount: s.sites.filter((si) => si.customerId === c.id).length }));
   518→
   519→  const recentPayments = [...s.payments]
   520→    .sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())
   521→    .slice(0, 6)
   522→    .map((p) => ({ ...p, customerName: customers.find((c) => c.id === p.customerId)?.name ?? '—' }));

   524→  const usage: PlatformUsage = {
   525→    totalSites,
   526→    totalArticles: s.sites.reduce((a, si) => a + si.articles, 0),
   527→    aiArticlesGenerated: Math.round(s.sites.reduce((a, si) => a + si.articles, 0) * 0.28),

   554→export function listCustomers(opts?: { search?: string; planId?: PlanId | 'all'; status?: SubscriptionStatus | 'all' }): (Customer & { siteCount: number })[] {

   559→    list = list.filter((c) => c.name.toLowerCase().includes(q) || c.email.toLowerCase().includes(q) || (c.company ?? '').toLowerCase().includes(q));
   560→  }
   561→  if (opts?.planId && opts.planId !== 'all') {
   562→    list = list.filter((c) => c.planId === opts.planId);
   563→  }
   564→  if (opts?.status && opts.status !== 'all') {
   565→    list = list.filter((c) => c.subscriptionStatus === opts.status);
   566→  }
   567→  return list
   568→    .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   569→    .map((c) => ({ ...c, siteCount: s.sites.filter((si) => si.customerId === c.id).length }));

   583→  const sites = s.sites.filter((si) => si.customerId === id);
   584→  const payments = s.payments
   585→    .filter((p) => p.customerId === id)

   588→  return { ...customer, siteCount: sites.length, sites, payments, recentActivity };

   621→    });
   622→  }
   623→  return list
   624→    .sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())
   625→    .map((p) => {
   626→      const c = s.customers.find((cu) => cu.id === p.customerId);

   670→    });
   671→  }

   673→  const pastDue = s.customers.filter((c) => c.subscriptionStatus === 'past_due');
   674→  if (pastDue.length > 0) {

   681→      // Customers page, which now is the customer-level subscription view —
   682→      // it shows Plan + Sub. Status + Account + Sites for every customer).
   683→      // Repoint this alert to the Customers page so the action stays useful.
   684→      action: { label: 'View Customers', module: 'platform-customers' },
   685→      time: 'recent',
   686→    });
   687→  }
   688→
   689→  const nearLimit = s.customers.filter((c) => {
   690→    const used = s.sites
   691→      .filter((si) => si.customerId === c.id)
   692→      .reduce((a, si) => a + si.storageBytes, 0);

   694→  });

   966→/**
   967→ * Async variant that prefers the DB Subscription row over the legacy
   968→ * in-memory customer. This is the version called by /api/platform/billing/me.
   969→ *
   970→ * The legacy in-memory customer (CUSTOMER_SEED) is still consulted for
   971→ * the Platform Admin Customers/Payments pages and for demo users who
   972→ * don't yet have a DB Subscription row.
   973→ *
   974→ * For users WITH a DB Subscription row:
   975→ *   - status, billingInterval, currentPeriodEnd, trialEnd come from the DB.
   976→ *   - freeTrialExpired is computed from trialEnd + plan.isFree + status.
   977→ *   - paymentHistory comes from the DB Payment table (real Stripe invoices
   978→ *     when Stripe is configured; empty for free plans).
   979→ *
   980→ * For users WITHOUT a DB Subscription row (legacy or fresh):
   981→ *   - Falls back to getClientBilling() (in-memory customer or Free plan).
   982→ */
   983→export async function getClientBillingAsync(user: BillingUser): Promise<ClientBillingState> {
   984→  // Bypass users: return the synthetic Internal plan (no DB lookup).
   985→  if (isBillingBypass(user)) return getClientBilling(user);
   986→
   987→  // No user id → can't look up DB Subscription. Fall back to legacy.
   988→  if (!user.id) return getClientBilling(user);
   989→
   990→  // Try the DB Subscription first.
   991→  const subState = await getClientSubscriptionState({
   992→    id: user.id,
   993→    email: user.email,
   994→    role: user.role ?? '',
   995→    billingMode: user.billingMode,
   996→  });
   997→
   998→  // No DB subscription row → fall back to the in-memory legacy state.
   999→  if (!subState.subscription) {
  1000→    return getClientBilling(user);
  1001→  }
  1002→
  1003→  // We have a real DB subscription. Build the ClientBillingState from it
  1004→  // (NOT from the in-memory customer — the DB is the source of truth).
  1005→  const plan = toPlan(subState.plan);
  1006→
  1007→  // Load real payment history from the DB Payment table.
  1008→  const dbPayments = await getUserPayments(user.id);
  1009→
  1010→  return {
  1011→    customer: null, // no in-memory customer when DB subscription is the source
  1012→    plan,
  1013→    allPlans: PLANS.filter((p) => p.active),
  1014→    status: subState.status,
  1015→    trialEnd: subState.trialEnd,
  1016→    nextBillingAt: subState.currentPeriodEnd,
  1017→    paymentHistory: dbPayments.map((p) => ({
  1018→      id: p.id,
  1019→      customerId: user.id as string,
  1020→      planId: p.planId as PlanId,
  1021→      amount: p.amount,
  1022→      currency: p.currency,
  1023→      status: p.status,
  1024→      method: p.method ?? '—',
  1025→      date: p.createdAt.toISOString(),
  1026→      invoiceNumber: p.invoiceNumber ?? '—',
  1027→    })),
  1028→    billingMode: (user.billingMode as 'EXTERNAL' | 'INTERNAL' | 'EXEMPT') ?? 'EXTERNAL',
  1029→    isInternal: false,
  1030→    billingInterval: subState.billingInterval,
  1031→    currentPeriodEnd: subState.currentPeriodEnd,
  1032→    freeTrialExpiresAt: subState.freeTrialExpiresAt,
  1033→    freeTrialExpired: subState.freeTrialExpired,
  1034→    stripeSubscriptionId: subState.subscription.stripeSubscriptionId,
  1035→    stripeCustomerId: subState.subscription.stripeCustomerId,
  1036→    requiresPlanAction: subState.requiresPlanAction,
  1037→    hasRealSubscription: true,
  1038→  };
  1039→}

  1060→ * if the user matches a CUSTOMER_SEED entry we ALSO mutate that entry

  1216→    });
  1217→
  1218→    if (s.subscriptionStatus === 'cancelled') {
  1219→      // Cancellation happened at some point after creation — there is no
  1220→      // dedicated timestamp in the dataset, so use subscriptionStart as a
  1221→      // stable proxy (still sorts in the past, like a historical event).
  1222→      events.push({
  1223→        id: `evt-sub-cancel-${s.id}`,
  1224→        type: 'WARNING',
  1225→        title: 'Subscription cancelled',
  1226→        message: `${s.name} cancelled the ${s.planName} plan subscription. Access has been revoked at the end of the current period.`,
  1227→        isRead: false,
  1228→        createdAt: s.subscriptionStart,
  1229→      });
  1230→    }
  1231→
  1232→    if (s.subscriptionStatus === 'trial') {
  1233→      const trialEnd = s.trialEnd ?? s.nextBillingAt ?? s.subscriptionStart;
  1234→      events.push({
  1235→        id: `evt-sub-trial-${s.id}`,
  1236→        type: 'ACTION_REQUIRED',
  1237→        title: 'Trial ending soon',
  1238→        message: `${s.name}'s ${s.planName} plan trial ends on ${trialEnd}. Convert to a paid subscription before then to retain access.`,
  1239→        isRead: false,
  1240→        createdAt: trialEnd,
  1241→      });
  1242→    }
  1243→  }
  1244→
  1245→  // 4. Audit log entries — INFO with action as title, detail as message
  1246→  for (const a of getAuditLog(50)) {
  1247→    events.push({
  1248→      id: `evt-aud-${a.id}`,
  1249→      type: 'INFO',
  1250→      title: a.action,
  1251→      message: a.detail && a.detail.length > 0 ? `${a.detail} — target: ${a.target}` : a.target,
  1252→      isRead: false,
  1253→      createdAt: a.timestamp,
  1254→    });
  1255→  }
  1256→
  1257→  // 5. Alerts — ERROR / WARNING / INFO by severity
  1258→  for (const alert of getAlerts()) {
  1259→    let type: PlatformEventType = 'INFO';
  1260→    if (alert.severity === 'critical') type = 'ERROR';
  1261→    else if (alert.severity === 'warning') type = 'WARNING';
  1262→    events.push({
  1263→      id: `evt-alert-${alert.id}`,
  1264→      type,
  1265→      title: alert.title,
  1266→      message: alert.message,
  1267→      isRead: false,
  1268→      createdAt: alertTimestamp(alert.id),
  1269→    });
  1270→  }
  1271→
  1272→  // Newest first — consistent with the Client Notifications feed.
  1273→  return events.sort(
  1274→    (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime(),
  1275→  );
  1276→}
