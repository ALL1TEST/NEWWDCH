   137→// -------------------- Schedules Page --------------------
   138→
   139→export function SchedulesPage({ scope = 'client' }: { scope?: 'client' | 'platform' } = {}) {
   140→  const queryClient = useQueryClient();
   141→  const isPlatform = scope === 'platform';
   142→  const [dialogOpen, setDialogOpen] = useState(false);
   143→  const [editingId, setEditingId] = useState<string | null>(null);
   144→  const [form, setForm] = useState<ScheduleForm>(initialForm);
   145→  const [deleteTarget, setDeleteTarget] = useState<ScheduleRow | null>(null);
   146→
   147→  const table = useDataTable({ initialSortField: 'createdAt', initialSortOrder: 'desc' });
   148→
   149→  const { data, isLoading } = useQuery({
   150→    queryKey: queryKeys.backupSchedules.list({
   151→      page: table.currentPage,
   152→      pageSize: table.pageSize,
   153→      sort: table.sortField,
   154→      order: table.sortOrder,
   155→      search: table.searchValue || undefined,
   156→      // Include scope in the cache key so client and platform entries do
   157→      // not collide. The value here is opaque to TanStack Query.
   158→      scope: isPlatform ? 'platform' : undefined,
   159→    }),
   160→    queryFn: () => getApi<ApiResponse<ScheduleRow[]>>('/api/backups/schedules', {
   161→      page: table.currentPage,
   162→      pageSize: table.pageSize,
   163→      sort: table.sortField,
   164→      order: table.sortOrder,
   165→      search: table.searchValue || undefined,
   166→      ...(isPlatform ? { scope: 'platform' } : {}),
   167→    }, { raw: true }),
   168→    staleTime: 10_000,
   169→  });
   170→
   171→  // Fetch configured storage destinations for the dropdown
   172→  const { data: storageData } = useQuery({
   173→    queryKey: ['backup-storage-destinations', isPlatform ? 'platform' : 'client'],
   174→    queryFn: () => getApi<{ id: string; name: string; provider: string; isActive: boolean }[]>(
   175→      '/api/backups/storage?pageSize=100',
   176→      isPlatform ? { scope: 'platform', pageSize: 100 } : { pageSize: 100 },
   177→    ),
   178→    staleTime: 30_000,
   179→  });
   180→  const storageDestinations = (storageData as unknown as { id: string; name: string; provider: string; isActive: boolean }[] | undefined)?.filter(s => s.isActive) ?? [];
   181→
   182→  const schedules = data?.data ?? [];
   183→  const pagination = data?.meta?.pagination;
   184→
   185→  // Dual empty-state logic (mirrors the Backups table):
   186→  // - isInitialEmpty: system genuinely has zero schedules AND no search is
   187→  //   active → show the full-page "No schedules configured" state.
   188→  // - isSearchEmpty: a search IS active but returns zero results → keep the
   189→  //   table card/headers/pagination visible and render the inline
   190→  //   "No schedules found" empty state inside the table body.
   191→  // These two states must NEVER be confused.
   192→  const hasSearch = !!table.searchValue?.trim();
   193→  const isInitialEmpty = !isLoading && schedules.length === 0 && !hasSearch;
   194→  const isSearchEmpty = !isLoading && schedules.length === 0 && hasSearch;
   195→
   196→  const createMutation = useMutation({
   197→    // For platform scope, send `scope: 'platform'` as a marker AND
   198→    // `backupScope: <BackupScope>` for the real data-scope choice. The
   199→    // /api/backups/schedules POST route peeks at the raw body BEFORE
   200→    // zod and rewrites scope -> backupScope (default FULL).
   201→    mutationFn: (body: ScheduleForm) => postApi(
   202→      '/api/backups/schedules',
   203→      isPlatform
   204→        ? { ...body, scope: 'platform', backupScope: body.scope }
   205→        : body,
   206→    ),
   207→    onSuccess: () => {
   208→      queryClient.invalidateQueries({ queryKey: queryKeys.backupSchedules.all });
   209→      toast.success('Schedule created');
   210→      closeDialog();
   211→    },
   212→    onError: (err: Error) => toast.error(err.message || 'Failed to create schedule'),
   213→  });
   214→
   215→  const updateMutation = useMutation({
   216→    // For platform scope, send `scope: 'platform'` marker + the real
   217→    // BackupScope under `backupScope`. The PATCH route peeks at the
   218→    // raw body BEFORE zod and rewrites scope.
   219→    mutationFn: ({ id, body }: { id: string; body: Partial<ScheduleForm> }) =>
   220→      patchApi(`/api/backups/schedules/${id}`, isPlatform
   221→        ? { ...body, scope: 'platform', ...(body.scope ? { backupScope: body.scope } : {}) }
   222→        : body,
   223→      ),
   224→    onSuccess: () => {
   225→      queryClient.invalidateQueries({ queryKey: queryKeys.backupSchedules.all });
   226→      toast.success('Schedule updated');
   227→      closeDialog();
   228→    },
   229→    onError: (err: Error) => toast.error(err.message || 'Failed to update schedule'),
   230→  });
   231→
   232→  const deleteMutation = useMutation({
   233→    // For platform scope, pass `?scope=platform` query param so the API
   234→    // gates the DELETE with requirePlatformAdmin.
   235→    mutationFn: (id: string) => deleteApi(`/api/backups/schedules/${id}${isPlatform ? '?scope=platform' : ''}`),
   236→    onSuccess: () => {
   237→      queryClient.invalidateQueries({ queryKey: queryKeys.backupSchedules.all });
   238→      setDeleteTarget(null);
   239→      toast.success('Schedule deleted');
   240→    },
   241→    onError: (err: Error) => {
   242→      setDeleteTarget(null);
   243→      toast.error(err.message || 'Failed to delete schedule');
   244→    },
   245→  });
   246→
   247→  const toggleActiveMutation = useMutation({
   248→    mutationFn: ({ id, isActive }: { id: string; isActive: boolean }) =>
   249→      patchApi(`/api/backups/schedules/${id}`, isPlatform
   250→        ? { isActive, scope: 'platform' }
   251→        : { isActive }
   252→      ),
   253→    onSuccess: () => {
   254→      queryClient.invalidateQueries({ queryKey: queryKeys.backupSchedules.all });
   255→    },
   256→  });
   257→
   258→  const openCreate = () => {
   259→    setEditingId(null);
   260→    setForm(initialForm);
   261→    setDialogOpen(true);
   262→  };
   263→
   264→  const openEdit = (row: ScheduleRow) => {
   265→    setEditingId(row.id);
   266→    setForm({
   267→      name: row.name,
   268→      description: row.description || '',
   269→      frequency: row.frequency,
   270→      cronExpression: row.cronExpression || '',
   271→      scope: row.scope,
   272→      storageId: row.storageId || '',
   273→      storageProvider: row.storageProvider,
   274→      encryptionEnabled: row.encryptionEnabled,
   275→      verificationEnabled: row.verificationEnabled,
   276→      retentionCount: row.retentionCount,
   277→    });
   278→    setDialogOpen(true);
   279→  };
   280→
   281→  const closeDialog = () => {
   282→    setDialogOpen(false);
   283→    setEditingId(null);
   284→    setForm(initialForm);
   285→  };
   286→
   287→  const handleSubmit = () => {
   288→    if (!form.name.trim()) return;
   289→    // Submit storageProvider (the API uses this); storageId is UI-only.
   290→    const { storageId: _omitStorageId, ...body } = form;
   291→    void _omitStorageId;
   292→    if (editingId) {
   293→      updateMutation.mutate({ id: editingId, body });
   294→    } else {
   295→      createMutation.mutate(body);
   296→    }
   297→  };
   298→
   299→  const updateForm = <K extends keyof ScheduleForm>(key: K, value: ScheduleForm[K]) => {
   300→    setForm((prev) => ({ ...prev, [key]: value }));
   301→  };
   302→
   303→  const isSaving = createMutation.isPending || updateMutation.isPending;
   304→
   305→  // Column min-widths are intentionally generous and the table uses
   306→  // `table-layout: auto` (NOT table-fixed) so cells NEVER truncate or clip —
   307→  // every value (Name, Frequency, Scope, Storage, Encrypt, Verify, Retention,
   308→  // Active, Next Run, Last Run, Actions) renders in full. Long schedule names
   309→  // simply make the table wider and trigger horizontal scroll inside the
   310→  // card. Mirrors the Backups table's reusable overflow/scroll pattern.
   311→  // Sum of min-widths: 220+110+130+130+90+90+110+80+150+150+60 = 1320px.
   312→  const columns: ColumnDef<ScheduleRow>[] = [
   313→      {
   314→        id: 'name',
   315→        header: 'Name',
   316→        accessorKey: 'name',
   317→        size: 220,
   318→        // No `truncate`/`overflow-hidden` — full schedule name always visible.
   319→        // `title` provides a hover tooltip for very long names without ever
   320→        // clipping the cell text.
   321→        cell: ({ getValue }) => {
   322→          const value = (getValue() as string) ?? '';
   323→          if (!value) return <span className="text-muted-foreground">—</span>;
   324→          return <span className="font-medium" title={value}>{value}</span>;
   325→        },
   326→      },
   327→      {
   328→        id: 'frequency',
   329→        header: 'Frequency',
   330→        accessorKey: 'frequency',
   331→        enableSorting: false,
   332→        size: 110,
   333→        cell: ({ row }) => {
   334→          const freq = row.original.frequency;
   335→          const cron = row.original.cronExpression;
   336→          return (
   337→            <span className="text-sm">
   338→              {labelize(freq)}
   339→              {freq === 'CUSTOM_CRON' && cron && (
   340→                <span className="block text-xs text-muted-foreground font-mono">{cron}</span>
   341→              )}
   342→            </span>
   343→          );
   344→        },
   345→      },
   346→      {
   347→        id: 'scope',
   348→        header: 'Scope',
   349→        accessorKey: 'scope',
   350→        enableSorting: false,
   351→        size: 130,
   352→        cell: ({ getValue }) => (
   353→          <Badge variant="outline" className="border-transparent bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400 font-medium">
   354→            {labelize(getValue() as string)}
   355→          </Badge>
   356→        ),
   357→      },
   358→      {
   359→        id: 'storage',
   360→        header: 'Storage',
   361→        accessorKey: 'storageProvider',
   362→        enableSorting: false,
   363→        size: 130,
   364→        // Full storage label visible (e.g. "Cloudflare R2", "Google Drive");
   365→        // never truncated.
   366→        cell: ({ getValue }) => (
   367→          <span className="text-xs text-muted-foreground" title={labelize(getValue() as string)}>
   368→            {labelize(getValue() as string)}
   369→          </span>
   370→        ),
   371→      },
   372→      {
   373→        id: 'encryptionEnabled',
   374→        header: 'Encrypt',
   375→        accessorKey: 'encryptionEnabled',
   376→        enableSorting: false,
   377→        size: 90,
   378→        cell: ({ getValue }) => (
   379→          <span className={cn(
   380→            'text-xs font-medium',
   381→            getValue() ? 'text-green-600 dark:text-green-400' : 'text-muted-foreground'
   382→          )}>
   383→            {getValue() ? 'Yes' : 'No'}
   384→          </span>
   385→        ),
   386→      },
   387→      {
   388→        id: 'verificationEnabled',
   389→        header: 'Verify',
   390→        accessorKey: 'verificationEnabled',
   391→        enableSorting: false,
   392→        size: 90,
   393→        cell: ({ getValue }) => (
   394→          <span className={cn(
   395→            'text-xs font-medium',
   396→            getValue() ? 'text-green-600 dark:text-green-400' : 'text-muted-foreground'
   397→          )}>
   398→            {getValue() ? 'Yes' : 'No'}
   399→          </span>
   400→        ),
   401→      },
   402→      {
   403→        id: 'retentionCount',
   404→        header: 'Retention',
   405→        accessorKey: 'retentionCount',
   406→        enableSorting: false,
   407→        size: 110,
   408→        cell: ({ getValue }) => (
   409→          <span className="tabular-nums text-sm">{getValue() as number} days</span>
   410→        ),
   411→      },
   412→      {
   413→        id: 'isActive',
   414→        header: 'Active',
   415→        accessorKey: 'isActive',
   416→        enableSorting: false,
   417→        size: 80,
   418→        cell: ({ row }) => (
   419→          <Switch
   420→            checked={row.original.isActive}
   421→            onCheckedChange={(checked) =>
   422→              toggleActiveMutation.mutate({ id: row.original.id, isActive: checked })
   423→            }
   424→          />
   425→        ),
   426→      },
   427→      {
   428→        id: 'nextRunAt',
   429→        header: 'Next Run',
   430→        accessorKey: 'nextRunAt',
   431→        size: 150,
   432→        // Full relative timestamp visible (e.g. "Tomorrow at 2:00 AM"); never
   433→        // truncated. The 150px min-width + auto layout guarantees realistic
   434→        // values fit; if a value is somehow wider the table scrolls.
   435→        cell: ({ getValue }) => {
   436→          const val = getValue() as string | null;
   437→          return (
   438→            <span className="text-xs text-muted-foreground" title={val ? formatRelativeTime(val) : undefined}>
   439→              {val ? formatRelativeTime(val) : '—'}
   440→            </span>
   441→          );
   442→        },
   443→      },
   444→      {
   445→        id: 'lastRunAt',
   446→        header: 'Last Run',
   447→        accessorKey: 'lastRunAt',
   448→        size: 150,
   449→        cell: ({ getValue }) => {
   450→          const val = getValue() as string | null;
   451→          return (
   452→            <span className="text-xs text-muted-foreground" title={val ? formatRelativeTime(val) : undefined}>
   453→              {val ? formatRelativeTime(val) : '—'}
   454→            </span>
   455→          );
   456→        },
   457→      },
   458→      ColumnDefHelper.actionColumn<ScheduleRow>({
   459→        id: 'actions',
   460→        size: 60,
   461→        render: (row) => (
   462→          <DropdownMenu>
   463→            <DropdownMenuTrigger asChild>
   464→              <Button variant="ghost" size="icon" className="h-8 w-8" onClick={(e) => e.stopPropagation()}>
   465→                <MoreHorizontal className="h-4 w-4" />
   466→              </Button>
   467→            </DropdownMenuTrigger>
   468→            <DropdownMenuContent align="end">
   469→              <DropdownMenuItem onClick={() => openEdit(row)}>
   470→                <Pencil className="h-4 w-4 mr-2" />
   471→                Edit
   472→              </DropdownMenuItem>
   473→              <DropdownMenuSeparator />
   474→              <DropdownMenuItem variant="destructive" onClick={() => setDeleteTarget(row)}>
   475→                <Trash2 className="h-4 w-4 mr-2" />
   476→                Delete
   477→              </DropdownMenuItem>
   478→            </DropdownMenuContent>
   479→          </DropdownMenu>
   480→        ),
   481→      }),
   482→    ];
   483→
   484→  return (
   485→    <div className="space-y-4">
   486→      {isPlatform ? (
   487→        <PlatformPageHeader
   488→          title="Backup Schedules"
   489→          subtitle="Platform-wide automated backup schedules across all customers and sites."
   490→          actions={
   491→            <Button size="sm" onClick={openCreate}>
   492→              <Plus className="h-4 w-4 mr-2" />
   493→              Create Schedule
   494→            </Button>
   495→          }
   496→        />
   497→      ) : (
   498→        <PageHeader
   499→          breadcrumbs={false}
   500→          title="Backup Schedules"
   501→          description="Configure automated backup schedules"
   502→          action={
   503→            <Button size="sm" onClick={openCreate}>
   504→              <Plus className="h-4 w-4 mr-2" />
   505→              Create Schedule
   506→            </Button>
   507→          }
   508→        />
   509→      )}
   510→
   511→      {isInitialEmpty ? (
   512→        <EmptyState
   513→          icon={CalendarClock}
   514→          title="No schedules configured"
   515→          description="Create a schedule to automate your backups."
   516→          action={{ label: 'Create Schedule', onClick: openCreate }}
   517→        />
   518→      ) : (
   519→        <DataTable
   520→          columns={columns}
   521→          data={schedules}
   522→          isLoading={isLoading}
   523→          totalItems={pagination?.total ?? 0}
   524→          pageSize={table.pageSize}
   525→          currentPage={table.currentPage}
   526→          onPageChange={(p) => table.setCurrentPage(p)}
   527→          onSortChange={(f, o) => table.setSortField(f, o)}
   528→          sortField={table.sortField}
   529→          sortOrder={table.sortOrder}
   530→          searchPlaceholder="Search schedules..."
   531→          searchValue={table.searchValue}
   532→          onSearch={(v) => {
   533→            table.setSearchValue(v);
   534→            table.setCurrentPage(1);
   535→          }}
   536→          getRowId={(row) => row.id}
   537→          // Auto layout (NOT table-fixed): cells never truncate/clip — every
   538→          // value renders in full. The table's min-width (1320px = sum of
   539→          // column min-widths: Name 220 + Frequency 110 + Scope 130 +
   540→          // Storage 130 + Encrypt 90 + Verify 90 + Retention 110 + Active 80
   541→          // + Next Run 150 + Last Run 150 + Actions 60) guarantees the table
   542→          // is only as narrow as its content needs; on viewports narrower
   543→          // than that, the table-container's overflow-x-auto provides a thin,
   544→          // subtle horizontal scrollbar that stays inside the card. The
   545→          // pagination/footer lives outside the scroll area and always
   546→          // aligns with the card edges. Same reusable pattern as the Backups
   547→          // table for consistency.
   548→          tableMinWidth={1320}
   549→          emptyMessage="No schedules found."
   550→          emptyState={
   551→            isSearchEmpty ? (
   552→              <NoSchedulesSearchResultsEmpty
   553→                onClear={() => {
   554→                  table.setSearchValue('');
   555→                  table.setCurrentPage(1);
   556→                }}
   557→              />
   558→            ) : undefined
   559→          }
   560→        />
   561→      )}
   562→
   563→      {/* Create/Edit Dialog */}
   564→      <Dialog open={dialogOpen} onOpenChange={(v) => !v && closeDialog()}>
   565→        <DialogContent className="sm:max-w-[520px]">
   566→          <DialogHeader>
   567→            <DialogTitle>{editingId ? 'Edit Schedule' : 'Create Schedule'}</DialogTitle>
   568→            <DialogDescription>
   569→              {editingId
   570→                ? 'Update backup schedule configuration.'
   571→                : 'Configure a new automated backup schedule.'}
   572→            </DialogDescription>
   573→          </DialogHeader>
   574→          <div className="space-y-4 py-2 max-h-[60vh] overflow-y-auto pr-1">
   575→            <div className="space-y-2">
   576→              <Label htmlFor="sched-name">Name</Label>
   577→              <Input
   578→                id="sched-name"
   579→                placeholder="e.g., Daily Database Backup"
   580→                value={form.name}
   581→                onChange={(e) => updateForm('name', e.target.value)}
   582→              />
   583→            </div>
   584→            <div className="space-y-2">
   585→              <Label htmlFor="sched-desc">Description</Label>
   586→              <Textarea
   587→                id="sched-desc"
   588→                placeholder="Optional description..."
   589→                value={form.description}
   590→                onChange={(e) => updateForm('description', e.target.value)}
   591→                rows={2}
   592→              />
   593→            </div>
   594→            <div className="space-y-2">
   595→              <Label htmlFor="sched-freq">Frequency</Label>
   596→              <Select
   597→                value={form.frequency}
   598→                onValueChange={(v) => updateForm('frequency', v as BackupScheduleFrequency)}
   599→              >
   600→                <SelectTrigger id="sched-freq">
   601→                  <SelectValue />
   602→                </SelectTrigger>
   603→                <SelectContent>
   604→                  {FREQUENCIES.map((f) => (
   605→                    <SelectItem key={f.value} value={f.value}>
   606→                      {f.label}
   607→                    </SelectItem>
   608→                  ))}
   609→                </SelectContent>
   610→              </Select>
   611→            </div>
   612→            {form.frequency === 'CUSTOM_CRON' && (
   613→              <div className="space-y-2">
   614→                <Label htmlFor="sched-cron">Custom Cron Expression</Label>
   615→                <Input
   616→                  id="sched-cron"
   617→                  placeholder="0 2 * * *"
   618→                  value={form.cronExpression}
   619→                  onChange={(e) => updateForm('cronExpression', e.target.value)}
   620→                  className="font-mono"
   621→                />
   622→                <p className="text-xs text-muted-foreground">
   623→                  Use standard 5-field cron format (minute hour day month weekday)
   624→                </p>
   625→              </div>
   626→            )}
   627→            <div className="grid grid-cols-2 gap-4">
   628→              <div className="space-y-2">
   629→                <Label htmlFor="sched-scope">Scope</Label>
   630→                <Select
   631→                  value={form.scope}
   632→                  onValueChange={(v) => updateForm('scope', v as BackupScope)}
   633→                >
   634→                  <SelectTrigger id="sched-scope">
   635→                    <SelectValue />
   636→                  </SelectTrigger>
   637→                  <SelectContent>
   638→                    {BACKUP_SCOPE_OPTIONS.map((opt) => (
   639→                      <SelectItem key={opt.value} value={opt.value}>{opt.label}</SelectItem>
   640→                    ))}
   641→                  </SelectContent>
   642→                </Select>
   643→              </div>
   644→              <div className="space-y-2">
   645→                <Label htmlFor="sched-storage">Storage Destination</Label>
   646→                <Select
   647→                  value={form.storageId}
   648→                  onValueChange={(v) => {
   649→                    const dest = storageDestinations.find((s) => s.id === v);
   650→                    updateForm('storageId', v);
   651→                    if (dest) updateForm('storageProvider', dest.provider as BackupStorageProvider);
   652→                  }}
   653→                >
   654→                  <SelectTrigger id="sched-storage">
   655→                    <SelectValue placeholder="Select a storage destination" />
   656→                  </SelectTrigger>
   657→                  <SelectContent>
   658→                    {storageDestinations.map((s) => (
   659→                      <SelectItem key={s.id} value={s.id}>
   660→                        <div className="flex items-center gap-2">
   661→                          <span>{s.name}</span>
   662→                          <span className="text-xs text-muted-foreground">{s.provider}</span>
   663→                        </div>
   664→                      </SelectItem>
   665→                    ))}
   666→                  </SelectContent>
   667→                </Select>
   668→                {storageDestinations.length === 0 && (
   669→                  <p className="text-xs text-amber-600">No storage destinations configured. Add one in Storage first.</p>
   670→                )}
   671→              </div>
   672→            </div>
   673→            <div className="space-y-2">
   674→              <Label htmlFor="sched-retention">Retention Count</Label>
   675→              <Input
   676→                id="sched-retention"
   677→                type="number"
   678→                min={1}
   679→                max={365}
   680→                value={form.retentionCount}
   681→                onChange={(e) => updateForm('retentionCount', Number(e.target.value))}
   682→              />
   683→              <p className="text-xs text-muted-foreground">Number of backups to keep</p>
   684→            </div>
   685→            <div className="flex items-center justify-between">
   686→              <div>
   687→                <Label htmlFor="sched-encrypt">Encryption</Label>
   688→                <p className="text-xs text-muted-foreground">Encrypt scheduled backups</p>
   689→              </div>
   690→              <Switch
   691→                id="sched-encrypt"
   692→                checked={form.encryptionEnabled}
   693→                onCheckedChange={(v) => updateForm('encryptionEnabled', v)}
   694→              />
   695→            </div>
   696→            <div className="flex items-center justify-between">
   697→              <div>
   698→                <Label htmlFor="sched-verify">Verification</Label>
   699→                <p className="text-xs text-muted-foreground">Verify backups after creation</p>
   700→              </div>
   701→              <Switch
   702→                id="sched-verify"
   703→                checked={form.verificationEnabled}
   704→                onCheckedChange={(v) => updateForm('verificationEnabled', v)}
   705→              />
   706→            </div>
   707→          </div>
   708→          <DialogFooter>
   709→            <Button variant="outline" onClick={closeDialog}>
   710→              Cancel
   711→            </Button>
   712→            <Button
   713→              onClick={handleSubmit}
   714→              disabled={isSaving || !form.name.trim()}
   715→            >
   716→              {isSaving && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   717→              {editingId ? 'Update' : 'Create'}
   718→            </Button>
   719→          </DialogFooter>
   720→        </DialogContent>
   721→      </Dialog>
   722→
   723→      {/* Delete Confirmation */}
   724→      <ConfirmDialog
   725→        open={!!deleteTarget}
   726→        onOpenChange={(v) => !v && setDeleteTarget(null)}
   727→        title="Delete Schedule"
   728→        description={
   729→          deleteTarget
   730→            ? `Are you sure you want to delete "${deleteTarget.name}"? Automated backups associated with this schedule will no longer run.`
   731→            : undefined
   732→        }
   733→        confirmLabel="Delete"
   734→        variant="destructive"
   735→        onConfirm={() => {
   736→          if (deleteTarget) deleteMutation.mutate(deleteTarget.id);
   737→        }}
   738→        isLoading={deleteMutation.isPending}
   739→      />
   740→    </div>
   741→  );
   742→}
