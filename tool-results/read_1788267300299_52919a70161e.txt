     3→// ============================================================
     4→// PLATFORM PLANS & PRICING — simplified admin UI.
     5→// ============================================================
     6→// Owner sees compact plan cards (Free / Plus / Pro / Max …) with a
     7→// quick Active toggle, a compact price + features + limits summary,
     8→// and an "Edit Plan" button. The Edit Plan modal has ONE base price
     9→// configuration per plan — NO per-currency matrix:
    10→//
    11→//   Basic Information: Name · Default/Fallback Currency (country/
    12→//   currency selector: 🇺🇸 United States — USD — $) · Auto Currency
    13→//   toggle · Monthly/Yearly Price (only for enabled periods) · Billing
    14→//   Periods checkboxes · Active
    15→//   then: Stripe Billing (sync + status) · Feature Access · Usage
    16→//   Limits.
    17→//
    18→// AUTO CURRENCY: each plan carries an autoCurrency flag. When ON,
    19→// the customer's currency is resolved SERVER-SIDE from their IP
    20→// (country → currency → plan price for that currency → fallback to
    21→// the plan default when unsupported) — the admin never picks the
    22→// customer's currency. When OFF, every customer is billed in the
    23→// plan's default currency. Per-currency prices (e.g. 90 DH) come
    24→// from the PLATFORM-level regional config (CountryPricing), not
    25→// from this modal.
    26→//
    27→// All writes go through /api/platform/admin/plans (GET list,
    28→// POST create) + /api/platform/admin/plans/[planId] (PUT update)
    29→// — owner-only — the same shared plan-config cache the Client
    30→// Billing page and MRR read. Server-side entitlement enforcement
    31→// (hasFeature) and usage-limit checks (checkLimit) consume the
    32→// same data, so toggling a feature off here denies it for clients
    33→// on the next request.
    34→// ============================================================

   222→/** Count the editor-visible enabled features (the 9 checkboxes —

   230→/** Toggle one feature key in the entitlements state. All editor
   231→ *  features are INDEPENDENT — there is no inter-feature dependency
   232→ *  anymore (API Access was removed as a separate feature; Client's
   233→ *  Own AI API already represents the client's own-API connectivity,

   235→function toggleFeature(cur: string[], key: string): string[] {
   236→  return cur.includes(key) ? cur.filter((k) => k !== key) : [...cur, key];

   451→/** Format the LARGE primary price on a plan card — integer amounts.
   452→ *  e.g. (90, 'MAD') → "90 DH"; (0, 'CHF') → "CHF 0". */
   453→function formatPriceSymbol(amount: number, currency: string): string {

   460→function formatPriceSymbolMonthlyEquiv(amount: number, currency: string): string {

   477→function PlanSummaryCard({
   478→  plan,
   479→  canEdit,
   480→  billingInterval,
   481→}: {
   482→  plan: PlanConfigData;
   483→  canEdit: boolean;
   484→  billingInterval: 'monthly' | 'yearly';
   485→}) {
   486→  const queryClient = useQueryClient();
   487→  const [editing, setEditing] = useState(false);
   488→
   489→  // Quick active toggle — auto-saves immediately (owner-only).
   490→  // Reuses the same PUT endpoint and the same invalidation chain
   491→  // as the full Edit modal so clients see the new status on the
   492→  // next read.
   493→  const activeMutation = useMutation({
   494→    mutationFn: (active: boolean) =>
   495→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, { active }),
   496→    onSuccess: (data) => {
   497→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   498→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   499→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   500→      toast.success(
   501→        `${data.name} ${data.active ? 'activated' : 'deactivated'} — changes are live for clients.`,
   502→      );
   503→    },
   504→    onError: (err: unknown) => {
   505→      const e = err as { error?: { message?: string }; message?: string };
   506→      toast.error(
   507→        e?.error?.message ?? e?.message ?? 'Unable to change plan status. OWNER access required.',
   508→      );
   509→    },
   510→  });
   511→
   512→  // Feature list shown on the card. Prefer the client-facing marketing
   513→  // copy (plan.features); fall back to the structured entitlement labels
   514→  // so a brand-new plan with no marketing copy still surfaces a readable
   515→  // feature list. Either way the data shown is real plan data — no mock.
   516→  const featureItems =
   517→    plan.features.length > 0
   518→      ? plan.features
   519→      : plan.entitlements.map((k) => ENTITLEMENT_LABELS[k as EntitlementKey] ?? k);
   520→
   521→  // The card shows the price for the plan's ENABLED billing periods:
   522→  //   - both periods   → follows the global Monthly/Yearly selector
   523→  //   - monthly only   → ALWAYS monthly (a disabled period never shows)
   524→  //   - yearly only    → ALWAYS yearly ("$X / year", no monthly option)
   525→  const effectiveInterval: 'monthly' | 'yearly' =
   526→    plan.billingMonthly && plan.billingYearly
   527→      ? billingInterval
   528→      : plan.billingMonthly
   529→        ? 'monthly'
   530→        : 'yearly';
   531→
   532→  return (
   533→    <>
   534→      <div
   535→        className={`group relative flex h-full flex-col overflow-hidden rounded-3xl border bg-card p-8 text-card-foreground transition-colors duration-300 hover:border-foreground/20 ${
   536→          plan.active ? '' : 'opacity-60'
   537→        }`}
   538→      >

   624→            </div>

   626→            // Monthly — plain amount, NO "/ month" suffix (the global
   627→            // Monthly/Yearly selector already communicates the period).

   666→            )}
   667→          </ul>
   668→        </div>
   669→
   670→        {/* Bottom row — Edit Plan (left, flex-1) + quick active toggle
   671→             (right, shrink-0), pinned to the card bottom. The toggle used
   672→             to live in the header's top-right; it was relocated here so the

   678→        <div className="mt-auto flex items-center gap-3 pt-8">
   679→          <Button
   680→            variant="outline"
   681→            size="sm"
   682→            className="flex-1 rounded-full"
   683→            disabled={!canEdit}
   684→            onClick={() => setEditing(true)}

   688→          </Button>
   689→          <div className="flex shrink-0 items-center gap-2">

   691→              htmlFor={`active-${plan.planId}`}

   694→              {plan.active ? 'Active' : 'Inactive'}
   695→            </Label>
   696→            <Switch
   697→              id={`active-${plan.planId}`}
   698→              checked={plan.active}
   699→              disabled={!canEdit || activeMutation.isPending}
   700→              onCheckedChange={(v) => activeMutation.mutate(v)}
   701→            />
   702→          </div>
   703→        </div>
   704→        {!canEdit && (
   705→          <p className="mt-2 text-center text-[10px] text-muted-foreground">
   706→            Editing is restricted to the OWNER role.
   707→          </p>
   708→        )}
   709→      </div>
   710→
   711→      {editing && (
   712→        <EditPlanDialog plan={plan} open={editing} onOpenChange={setEditing} />
   713→      )}
   714→    </>
   715→  );
   716→}
