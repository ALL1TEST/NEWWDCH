     3→// ============================================================
     4→// PLATFORM PLANS & PRICING — simplified admin UI.
     5→// ============================================================
     6→// Owner sees compact plan cards (Free / Plus / Pro / Max …) with a
     7→// quick Active toggle, a compact price + features + limits summary,
     8→// and an "Edit Plan" button. The Edit Plan modal has ONE base price
     9→// configuration per plan — NO per-currency matrix:
    10→//
    11→//   Basic Information: Name · Default/Fallback Currency (country/
    12→//   currency selector: 🇺🇸 United States — USD — $) · Auto Currency
    13→//   toggle · Monthly/Yearly Price (only for enabled periods) · Billing
    14→//   Periods checkboxes · Active
    15→//   then: Stripe Billing (sync + status) · Feature Access · Usage
    16→//   Limits.
    17→//
    18→// AUTO CURRENCY: each plan carries an autoCurrency flag. When ON,
    19→// the customer's currency is resolved SERVER-SIDE from their IP
    20→// (country → currency → plan price for that currency → fallback to
    21→// the plan default when unsupported) — the admin never picks the
    22→// customer's currency. When OFF, every customer is billed in the
    23→// plan's default currency. Per-currency prices (e.g. 90 DH) come
    24→// from the PLATFORM-level regional config (CountryPricing), not
    25→// from this modal.
    26→//
    27→// All writes go through /api/platform/admin/plans (GET list,
    28→// POST create) + /api/platform/admin/plans/[planId] (PUT update)
    29→// — owner-only — the same shared plan-config cache the Client
    30→// Billing page and MRR read. Server-side entitlement enforcement
    31→// (hasFeature) and usage-limit checks (checkLimit) consume the
    32→// same data, so toggling a feature off here denies it for clients
    33→// on the next request.
    34→// ============================================================

   231→/** Toggle one feature key in the entitlements state. All editor
   232→ *  features are INDEPENDENT — there is no inter-feature dependency
   233→ *  anymore (API Access was removed as a separate feature; Client's
   234→ *  Own AI API already represents the client's own-API connectivity,

   236→function toggleFeature(cur: string[], key: string): string[] {
   237→  return cur.includes(key) ? cur.filter((k) => k !== key) : [...cur, key];

   449→// formatMoney comes from the shared currency catalog (client-safe):
   450→//   (9, 'USD') → "$9" · (90, 'MAD') → "90 DH" · (9, 'CHF') → "CHF 9".

   452→/** Format the LARGE primary price on a plan card — integer amounts.
   453→ *  e.g. (90, 'MAD') → "90 DH"; (0, 'CHF') → "CHF 0". */
   454→function formatPriceSymbol(amount: number, currency: string): string {
   455→  return formatMoney(amount, currency, 0);

   458→/** Format with 2 decimals — used for the monthly-equivalent of a

   461→function formatPriceSymbolMonthlyEquiv(amount: number, currency: string): string {
   462→  return formatMoney(amount, currency, 2);

   478→function PlanSummaryCard({
   479→  plan,
   480→  canEdit,
   481→  billingInterval,
   482→}: {
   483→  plan: PlanConfigData;
   484→  canEdit: boolean;
   485→  billingInterval: 'monthly' | 'yearly';
   486→}) {
   487→  const queryClient = useQueryClient();
   488→  const [editing, setEditing] = useState(false);
   489→  // Delete confirmation — the icon button opens the dialog; the dialog
   490→  // confirms BEFORE the DELETE call fires (see DeletePlanDialog below).
   491→  const [confirmingDelete, setConfirmingDelete] = useState(false);
   492→
   493→  // Quick active toggle — auto-saves immediately (owner-only).
   494→  // Reuses the same PUT endpoint and the same invalidation chain
   495→  // as the full Edit modal so clients see the new status on the
   496→  // next read.
   497→  const activeMutation = useMutation({
   498→    mutationFn: (active: boolean) =>
   499→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, { active }),
   500→    onSuccess: (data) => {
   501→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   502→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   503→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   504→      toast.success(
   505→        `${data.name} ${data.active ? 'activated' : 'deactivated'} — changes are live for clients.`,
   506→      );
   507→    },
   508→    onError: (err: unknown) => {
   509→      const e = err as { error?: { message?: string }; message?: string };
   510→      toast.error(
   511→        e?.error?.message ?? e?.message ?? 'Unable to change plan status. OWNER access required.',
   512→      );
   513→    },
   514→  });
   515→
   516→  // Plan DELETION — reuses the EXISTING owner-only backend route
   517→  // (DELETE /api/platform/admin/plans/[planId] → deletePlanConfig),
   518→  // which already refuses to delete the last remaining plan or a plan
   519→  // still referenced by active subscriptions (409 DELETE_REFUSED).
   520→  // The same shared cache invalidation chain as the active toggle
   521→  // runs on success so clients see the removal on the next read.
   522→  const deleteMutation = useMutation({
   523→    mutationFn: () =>
   524→      deleteApi<{ deleted: boolean; planId: string }>(`/api/platform/admin/plans/${plan.planId}`),
   525→    onSuccess: () => {
   526→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   527→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   528→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });
   529→      toast.success(`${plan.name} deleted — the plan was removed for clients.`);
   530→      setConfirmingDelete(false);
   531→    },
   532→    onError: (err: unknown) => {
   533→      const e = err as { error?: { message?: string }; message?: string };
   534→      toast.error(

  1005→  return (
