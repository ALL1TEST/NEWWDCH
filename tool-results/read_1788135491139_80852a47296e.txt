     1→// ============================================================
     2→// PLATFORM DATA SERVICE — Centralized SaaS platform dataset
     3→// ============================================================
     4→// This is the SINGLE source of truth for the entire SaaS platform.
     5→// Both the Client Billing experience AND the Platform Admin
     6→// Dashboard read from and mutate this same in-memory dataset, so
     7→// a plan change, cancellation, or payment made on the client side
     8→// is immediately reflected on the admin side, and vice-versa.
     9→//
    10→// Consistency rules (enforced by derivation, never hardcoded):
    11→//  - totalCustomers  = customers.length (non-deleted)
    12→//  - activeSubscriptions = subscriptions where status === 'active'
    13→//  - mrr = sum of normalized monthly price over active PAID subs
    14→//  - totalSites = sites.length
    15→//  - planDistribution = subscriptions grouped by plan (all statuses)
    16→//  - statusCounts = subscriptions grouped by status
    17→//  - recentCustomers = customers sorted by createdAt desc
    18→//  - recentPayments = payments sorted by date desc
    19→//  - usage = aggregated from sites/content/media/AI records
    20→//  - alerts = derived from failed payments + past-due + storage
    21→//
    22→// Mock data is deterministic (no Math.random) so numbers are stable
    23→// across renders and never contradict each other.
    24→// ============================================================

    28→// PlanId is a plain string, NOT a fixed union. The PlanConfig table is the
    29→// source of truth — custom plans created via Platform Admin must flow
    30→// through every consumer (checkout, change-plan, admin change-plan,
    31→// billing page, MRR, entitlements). Callers validate plan ids against
    32→// the DB via `ensurePlanAssignable()` (subscription-data.ts), not
    33→// against a hardcoded TypeScript union.

    35→export type SubscriptionStatus = 'active' | 'trial' | 'past_due' | 'cancelled' | 'expired';
    36→export type BillingInterval = 'monthly' | 'yearly';
    37→export type CustomerStatus = 'ACTIVE' | 'SUSPENDED' | 'DEACTIVATED';
    38→export type PaymentStatus = 'paid' | 'pending' | 'failed' | 'refunded';

   160→// -------------------- Plans (delegated to DB-backed plan-config) --------------------
   161→// The editable plan definitions live in the PlanConfig table and are cached
   162→// by ./plan-config — the SINGLE source of truth. The owner edits there via
   163→// the Platform Admin; this module (admin overview + client billing) and the
   164→// client billing API both read the same cached values, so an owner price
   165→// change propagates to the client billing page and to MRR on the next read.
   166→// The PLANS array below is a live view re-spliced whenever the config changes.
   167→
   168→import { getPlanConfigsSync, getPlanConfigSync, subscribe as subscribePlanConfig, type PlanConfigData } from './plan-config';
   169→
   170→function toPlan(d: PlanConfigData): Plan {
   171→  return {
   172→    id: d.planId as PlanId,
   173→    name: d.name,
   174→    price: d.priceMonthly,
   175→    priceMonthly: d.priceMonthly,
   176→    priceYearly: d.priceYearly,
   177→    currency: d.currency,
   178→    pricesByCurrency: d.pricesByCurrency,
   179→    stripePriceIdsByCurrency: d.stripePriceIdsByCurrency,
   180→    interval: d.interval,
   181→    isFree: d.isFree,
   182→    active: d.active,
   183→    features: d.features,
   184→    entitlements: d.entitlements,
   185→    limits: d.limits,
   186→  };
   187→}
   188→
   189→/** Live view of the DB-backed plan configs. Mutated in place on every
   190→ *  config refresh so existing references stay valid. */
   191→export const PLANS: Plan[] = [];
   192→
   193→function refreshPLANS(): void {
   194→  const data = getPlanConfigsSync();
   195→  PLANS.length = 0;
   196→  for (const d of data) PLANS.push(toPlan(d));
   197→}
   198→refreshPLANS();
   199→subscribePlanConfig(refreshPLANS);
   200→
   201→export function getPlan(id: PlanId): Plan {
   202→  return PLANS.find((p) => p.id === id) ?? PLANS[0];
   203→}

   205→/** Normalized monthly price. Yearly subscriptions contribute priceYearly/12
   206→ *  so MRR is a true monthly run-rate. */
   207→export function monthlyPrice(plan: Plan, interval: BillingInterval): number {
   208→  if (interval === 'yearly') return Math.round(plan.priceYearly / 12);
   209→  return plan.priceMonthly;
   210→}

   217→function isoDaysAgo(days: number): string {
   218→  const d = new Date();
   219→  d.setDate(d.getDate() - days);
   220→  d.setHours(9, 15, 0, 0);
   221→  return d.toISOString();
   222→}
   223→function isoDaysAhead(days: number): string {
   224→  const d = new Date();
   225→  d.setDate(d.getDate() + days);
   226→  d.setHours(9, 15, 0, 0);
   227→  return d.toISOString();
   228→}

   402→function buildRevenueSeries(currentMrr: number): RevenuePoint[] {
   403→  const trend = [0.41, 0.47, 0.54, 0.62, 0.71, 0.80, 0.89, 1.0];
   404→  const out: RevenuePoint[] = [];
   405→  const now = new Date();
   406→  for (let i = 7; i >= 0; i--) {
   407→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   408→    const factor = trend[7 - i];
   409→    out.push({ month: MONTHS[d.getMonth()], revenue: Math.round(currentMrr * factor) });
   410→  }
   411→  return out;
   412→}

   429→function createStore(): PlatformStore {
   430→  const customers: Customer[] = CUSTOMER_SEED.map((c, i) => ({
   431→    ...c,
   432→    id: `cus_${String(i + 1).padStart(3, '0')}`,

   435→  const sites: PlatformSite[] = SITE_SEED.map((s, i) => ({

   438→    name: s.name,
   439→    slug: s.slug,
   440→    domain: s.domain,
   441→    status: s.status,
   442→    articles: s.articles,
   443→    media: s.media,
   444→    storageBytes: s.storageBytes,
   445→    createdAt: isoDaysAgo(s.daysAgo),
   446→  }));
   447→
   448→  const payments: Payment[] = PAYMENT_SEED.map((p, i) => {
   449→    const c = customers[p.customerIdx];
   450→    return {
   451→      id: `pay_${String(i + 1).padStart(3, '0')}`,
   452→      customerId: c.id,
   453→      planId: p.planId,
   454→      amount: p.amount,
   455→      currency: 'CHF',
   456→      status: p.status,
   457→      method: p.method,
   458→      date: isoDaysAgo(p.daysAgo),
   459→      invoiceNumber: `INV-2026-${String(1000 + i)}`,
   460→    };
   461→  });
   462→
   463→  const audit: AuditEntry[] = INITIAL_AUDIT.map((a, i) => ({
   464→    ...a,
   465→    id: `aud_${String(i + 1).padStart(3, '0')}`,
   466→  }));
   467→
   468→  _auditCounter = audit.length;
   469→  _paymentCounter = payments.length;
   470→
   471→  return { customers, sites, payments, audit };
   472→}

   504→/** Resolve the storageBytes limit for a plan id from the live plan
   505→ *  config cache. Returns Number.MAX_SAFE_INTEGER for unlimited (-1)

   507→function storageLimitBytesForPlan(planId: string): number {

   513→/** Map a real DB User (+ their Subscription row, when present) to the
   514→ *  public Customer interface the Platform Admin UI consumes. The shape
   515→ *  matches the legacy mock Customer exactly so the UI is unchanged —
   516→ *  only the data source switched from the in-memory seed to the DB.
   517→ *  Company / country default to placeholders since the User table has
   518→ *  no such columns (kept on the legacy mock customer only). */
   519→function mapUserToCustomer(
   520→  user: {
   521→    id: string;
   522→    name: string | null;
   523→    email: string;
   524→    status: string;
   525→    billingMode: string;
   526→    createdAt: Date;
   527→  },
   528→  sub: {
   529→    planId: string;
   530→    status: string;
   531→    billingInterval: string;
   532→    startDate: Date;
   533→    currentPeriodEnd: Date | null;
   534→    trialEnd: Date | null;
   535→  } | null,
   536→): Customer {
   537→  const planId = (sub && sub.planId ? sub.planId : 'free') as PlanId;
   538→  const subscriptionStatus = (sub && ['active', 'trial', 'past_due', 'cancelled', 'expired'].includes(sub.status)
   539→    ? sub.status
   540→    : 'active') as SubscriptionStatus;
   541→  const billingInterval: BillingInterval = sub && sub.billingInterval === 'yearly' ? 'yearly' : 'monthly';
   542→  const userStatus: CustomerStatus =
   543→    user.status === 'SUSPENDED' ? 'SUSPENDED' : user.status === 'DEACTIVATED' ? 'DEACTIVATED' : 'ACTIVE';
   544→  return {
   545→    id: user.id,
   546→    name: user.name ?? user.email,
   547→    email: user.email,
   548→    status: userStatus,
   549→    planId,
   550→    subscriptionStatus,
   551→    billingInterval,
   552→    createdAt: user.createdAt.toISOString(),
   553→    subscriptionStart: sub ? sub.startDate.toISOString() : user.createdAt.toISOString(),
   554→    nextBillingAt: sub?.currentPeriodEnd?.toISOString() ?? null,
   555→    trialEnd: sub?.trialEnd?.toISOString() ?? null,
   556→    company: null,
   557→    country: '—',
   558→    storageLimitBytes: storageLimitBytesForPlan(planId),
   559→  };
   560→}

  1024→/** Normalize a Prisma Payment row (with user included) to the public
  1025→ *  Payment interface the admin UI consumes. The shape is identical to
  1026→ *  the legacy mock Payment so the UI is unchanged; only the data source
  1027→ *  switched to the real DB. The `user` field is the resolved customer
  1028→ *  ({ id, name, email }) — passed in by listPayments / getRecentPayments-
  1029→ *  Real via Prisma's `include: { user: { select: ... } }`, and
  1030→ *  synthesized in `getCustomer` for the customer-detail Recent Payments
  1031→ *  list. Returns the enriched shape (Payment & { customerName,
  1032→ *  customerEmail }); CustomerDetail strips the extras when composing
  1033→ *  `payments: Payment[]`. */
  1034→function mapPaymentRowForCustomer(r: {
  1035→  id: string;
  1036→  userId: string;
  1037→  planId: string;
  1038→  amount: number;
  1039→  currency: string;
  1040→  status: string;
  1041→  method: string | null;
  1042→  invoiceNumber: string | null;
  1043→  stripeInvoiceId: string | null;
  1044→  stripePaymentIntentId: string | null;
  1045→  stripeChargeId: string | null;
  1046→  paymentMethodType: string | null;
  1047→  paymentMethodDetails: string | null;
  1048→  paidAt: Date | null;
  1049→  createdAt: Date;
  1050→  user?: { id: string; name: string | null; email: string } | null;
  1051→}): Payment & { customerName: string; customerEmail: string } {
  1052→  const planId = (r.planId ? r.planId : 'free') as PlanId;
  1053→  const status = (['paid', 'pending', 'failed', 'refunded'].includes(r.status)
  1054→    ? r.status
  1055→    : 'pending') as PaymentStatus;
  1056→  return {
  1057→    id: r.id,
  1058→    customerId: r.userId,
  1059→    planId,
  1060→    amount: r.amount,
  1061→    currency: r.currency,
  1062→    status,
  1063→    method: r.method ?? deriveMethodLabel(r.paymentMethodType, r.paymentMethodDetails) ?? '—',
  1064→    date: (r.paidAt ?? r.createdAt).toISOString(),
  1065→    invoiceNumber: r.invoiceNumber ?? r.stripeInvoiceId ?? '—',
  1066→    // Stripe transaction details — surfaced in the admin Payments table
  1067→    // so every row is verifiably tied to a real Stripe PaymentIntent /
  1068→    // Charge / Invoice. Null/undefined when Stripe isn't configured.
  1069→    stripeInvoiceId: r.stripeInvoiceId,
  1070→    stripePaymentIntentId: r.stripePaymentIntentId,
  1071→    stripeChargeId: r.stripeChargeId,
  1072→    customerName: r.user?.name ?? r.user?.email ?? '—',
  1073→    customerEmail: r.user?.email ?? '—',
  1074→  };
  1075→}

  1077→/** Build a human-readable payment-method label from the stored metadata,
  1078→ *  e.g. "Visa ••4242". Falls back to the capitalized type. */
  1079→function deriveMethodLabel(type: string | null, detailsJson: string | null): string | null {
  1080→  if (!type) return null;
  1081→  if (type === 'card' && detailsJson) {
  1082→    try {
  1083→      const d = JSON.parse(detailsJson) as { brand?: string; last4?: string };
  1084→      if (d.brand && d.last4) return `${capitalize(d.brand)} ••${d.last4}`;
  1085→      if (d.brand) return capitalize(d.brand);
  1086→    } catch {
  1087→      // ignore malformed JSON
  1088→    }
  1089→  }
  1090→  return capitalize(type);
  1091→}
  1092→
  1093→function capitalize(s: string): string {
  1094→  return s.charAt(0).toUpperCase() + s.slice(1);
  1095→}

  1681→export async function clientCancelSubscription(user: BillingUser): Promise<ClientBillingState | null> {
  1682→  if (isBillingBypass(user)) return getClientBilling(user); // bypass: nothing to cancel
  1683→  if (!user.id) return null;
  1684→
  1685→  // Cancel the DB Subscription (the authoritative record).
  1686→  await cancelDbSubscription(user.id);
  1687→
  1688→  // Also mutate the in-memory demo customer if one exists.
  1689→  const s = store();
  1690→  const c = s.customers.find((cu) => cu.email.toLowerCase() === user.email.toLowerCase());
  1691→  if (c) cancelSubscription(c.id, user.email);
  1692→
  1693→  return getClientBillingAsync(user);
  1694→}

  1699→// Aggregates platform-level events from the SAME centralized
  1700→// dataset that powers Platform Overview / Customers / Payments /
  1701→// Subscriptions / Audit / Alerts. PURE read-only function — no new
  1702→// state, no persistence. The /api/platform/admin/notifications

  1704→// are documented no-ops on the API side because the feed is derived
  1705→// fresh on every request (read-state would live in a separate
  1706→// table in a real implementation).
  1707→//
  1708→// Event types follow the same NotificationType shape the Client
  1709→// Notifications page uses: INFO | SUCCESS | WARNING | ERROR |
  1710→// ACTION_REQUIRED. This lets the Platform Notifications UI reuse
  1711→// the client page's NotificationCard / NOTIFICATION_TYPE_CONFIG
  1712→// verbatim.
  1713→// ============================================================
  1714→
  1715→export type PlatformEventType = 'INFO' | 'SUCCESS' | 'WARNING' | 'ERROR' | 'ACTION_REQUIRED';
  1716→
  1717→export interface PlatformEvent {
  1718→  id: string;
  1719→  type: PlatformEventType;
  1720→  title: string;
  1721→  message: string;
  1722→  isRead: boolean;
  1723→  createdAt: string;
  1724→}

  1726→function isoMinutesAgo(mins: number): string {
  1727→  return new Date(Date.now() - mins * 60_000).toISOString();
  1728→}
  1729→
  1730→/** Map alert id → a deterministic "recent" timestamp (no Math.random,
  1731→ *  stable across renders for a given wall-clock minute). */
  1732→function alertTimestamp(alertId: string): string {
  1733→  switch (alertId) {
  1734→    case 'alert-failed-payments': return isoMinutesAgo(30);
  1735→    case 'alert-past-due': return isoMinutesAgo(120);
  1736→    case 'alert-storage': return isoMinutesAgo(360);
  1737→    case 'alert-new-customers': return isoMinutesAgo(60 * 24);
  1738→    default: return isoMinutesAgo(60);
  1739→  }
  1740→}

  1742→/** Derive the unified platform events feed. Reads from the SYNC mock-store
  1743→ *  helpers (listCustomersMockSync / listPaymentsMockSync / listSubscriptionsMockSync)
  1744→ *  + getAuditLog + getAlerts. The public `listCustomers` / `listPayments` /
  1745→ *  `listSubscriptions` are now async + DB-backed (Task 78-D), but this
  1746→ *  function stays SYNC because /api/platform/admin/notifications calls it
  1747→ *  synchronously and isn't in this task's file list. The mock store is the
  1748→ *  deterministic seed (no Math.random) so the derived events are stable
  1749→ *  across renders. */
  1750→export function getPlatformEvents(): PlatformEvent[] {
  1751→  const events: PlatformEvent[] = [];
  1752→
  1753→  // 1. Customer registrations — INFO "New customer registered"
  1754→  for (const c of listCustomersMockSync()) {
  1755→    events.push({
  1756→      id: `evt-cust-${c.id}`,
  1757→      type: 'INFO',
  1758→      title: 'New customer registered',
  1759→      message: `${c.name} (${c.email}) signed up from ${c.country} on the ${c.planId.toUpperCase()} plan.`,
  1760→      isRead: false,
  1761→      createdAt: c.createdAt,
  1762→    });
  1763→  }
  1764→
  1765→  // 2. Payments — SUCCESS / ERROR / WARNING / INFO by status
  1766→  for (const p of listPaymentsMockSync()) {
  1767→    let type: PlatformEventType = 'INFO';
  1768→    let title = 'Payment update';
  1769→    if (p.status === 'paid') {
  1770→      type = 'SUCCESS';
  1771→      title = 'Successful payment';
  1772→    } else if (p.status === 'failed') {
  1773→      type = 'ERROR';
  1774→      title = 'Failed payment';
  1775→    } else if (p.status === 'refunded') {
  1776→      type = 'WARNING';
  1777→      title = 'Payment refunded';
  1778→    } else if (p.status === 'pending') {
  1779→      type = 'INFO';
  1780→      title = 'Payment pending';
  1781→    }
  1782→    events.push({
  1783→      id: `evt-pay-${p.id}`,
  1784→      type,
  1785→      title,
  1786→      message: `${p.customerName} — ${p.currency} ${p.amount} via ${p.method}. Invoice ${p.invoiceNumber}. Status: ${p.status}.`,
  1787→      isRead: false,
  1788→      createdAt: p.date,
  1789→    });
  1790→  }
  1791→
  1792→  // 3. Subscriptions — INFO created, WARNING cancelled, ACTION_REQUIRED trial ending
  1793→  for (const s of listSubscriptionsMockSync()) {
  1794→    const plan = getPlan(s.planId);
  1795→    events.push({
  1796→      id: `evt-sub-${s.id}`,
  1797→      type: 'INFO',
  1798→      title: 'Subscription created',
  1799→      message: `${s.name} subscribed to the ${s.planName} plan (${s.billingInterval} billing, ${plan.currency} ${s.monthlyPrice}/mo).`,
  1800→      isRead: false,
  1801→      createdAt: s.subscriptionStart,
  1802→    });
  1803→
  1804→    if (s.subscriptionStatus === 'cancelled') {
  1805→      // Cancellation happened at some point after creation — there is no
  1806→      // dedicated timestamp in the dataset, so use subscriptionStart as a
  1807→      // stable proxy (still sorts in the past, like a historical event).
  1808→      events.push({
  1809→        id: `evt-sub-cancel-${s.id}`,
  1810→        type: 'WARNING',
  1811→        title: 'Subscription cancelled',
  1812→        message: `${s.name} cancelled the ${s.planName} plan subscription. Access has been revoked at the end of the current period.`,
  1813→        isRead: false,
  1814→        createdAt: s.subscriptionStart,
  1815→      });
  1816→    }
  1817→
  1818→    if (s.subscriptionStatus === 'trial') {
  1819→      const trialEnd = s.trialEnd ?? s.nextBillingAt ?? s.subscriptionStart;
  1820→      events.push({
  1821→        id: `evt-sub-trial-${s.id}`,
  1822→        type: 'ACTION_REQUIRED',
  1823→        title: 'Trial ending soon',
  1824→        message: `${s.name}'s ${s.planName} plan trial ends on ${trialEnd}. Convert to a paid subscription before then to retain access.`,
  1825→        isRead: false,
  1826→        createdAt: trialEnd,
  1827→      });
  1828→    }
  1829→  }
  1830→
  1831→  // 4. Audit log entries — INFO with action as title, detail as message
  1832→  for (const a of getAuditLog(50)) {
  1833→    events.push({
  1834→      id: `evt-aud-${a.id}`,
  1835→      type: 'INFO',
  1836→      title: a.action,
  1837→      message: a.detail && a.detail.length > 0 ? `${a.detail} — target: ${a.target}` : a.target,
  1838→      isRead: false,
  1839→      createdAt: a.timestamp,
  1840→    });
  1841→  }
  1842→
  1843→  // 5. Alerts — ERROR / WARNING / INFO by severity
  1844→  for (const alert of getAlerts()) {
  1845→    let type: PlatformEventType = 'INFO';
  1846→    if (alert.severity === 'critical') type = 'ERROR';
  1847→    else if (alert.severity === 'warning') type = 'WARNING';
  1848→    events.push({
  1849→      id: `evt-alert-${alert.id}`,
  1850→      type,
  1851→      title: alert.title,
  1852→      message: alert.message,
  1853→      isRead: false,
  1854→      createdAt: alertTimestamp(alert.id),
  1855→    });
  1856→  }
  1857→
  1858→  // Newest first — consistent with the Client Notifications feed.
  1859→  return events.sort(
  1860→    (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime(),
  1861→  );
  1862→}
