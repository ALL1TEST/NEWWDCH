    83→// -------------------- Types --------------------
    84→
    85→interface AiProvider {
    86→  id: string;
    87→  name: string;
    88→  kind: AiProviderKind;
    89→  baseUrl: string;
    90→  apiKey: string | null;
    91→  apiVersion: string | null;
    92→  isActive: boolean;
    93→  isDefault: boolean;
    94→  connectionStatus: AiConnectionStatus;
    95→  latency: number | null;
    96→  lastSyncAt: string | null;
    97→  createdAt: string;
    98→  updatedAt: string;
    99→}
   100→
   101→interface ProviderFormData {
   102→  name: string;
   103→  kind: AiProviderKind;
   104→  baseUrl: string;
   105→  apiKey: string;
   106→  apiVersion: string;
   107→  isActive: boolean;
   108→}
   109→
   110→// -------------------- Constants --------------------
   111→
   112→// Provider kinds that can be selected when creating/editing a provider.
   113→// Legacy kinds (OPENROUTER, OLLAMA, AZURE_OPENAI) are kept in PROVIDER_CONFIGS
   114→// for display of existing rows but are no longer selectable.
   115→const PROVIDER_KINDS: AiProviderKind[] = [
   116→  'OPENAI', 'ANTHROPIC', 'GEMINI', 'GROQ', 'DEEPSEEK',
   117→];
   118→
   119→const PROVIDER_CONFIGS: Record<AiProviderKind, { label: string; defaultUrl: string; color: string }> = {
   120→  OPENAI: { label: 'OpenAI', defaultUrl: 'https://api.openai.com/v1', color: 'bg-emerald-100 text-emerald-700' },
   121→  ANTHROPIC: { label: 'Anthropic', defaultUrl: 'https://api.anthropic.com/v1', color: 'bg-orange-100 text-orange-700' },
   122→  GEMINI: { label: 'Gemini', defaultUrl: 'https://generativelanguage.googleapis.com/v1', color: 'bg-sky-100 text-sky-700' },
   123→  OPENROUTER: { label: 'OpenRouter', defaultUrl: 'https://openrouter.ai/api/v1', color: 'bg-violet-100 text-violet-700' },
   124→  GROQ: { label: 'Groq', defaultUrl: 'https://api.groq.com/openai/v1', color: 'bg-rose-100 text-rose-700' },
   125→  DEEPSEEK: { label: 'DeepSeek', defaultUrl: 'https://api.deepseek.com/v1', color: 'bg-cyan-100 text-cyan-700' },
   126→  OLLAMA: { label: 'Ollama', defaultUrl: 'http://localhost:11434/v1', color: 'bg-zinc-100 text-zinc-700' },
   127→  AZURE_OPENAI: { label: 'Azure OpenAI', defaultUrl: '', color: 'bg-teal-100 text-teal-700' },
   128→};
   129→
   130→function kindConfig(kind: string): { label: string; color: string } {
   131→  const cfg = (PROVIDER_CONFIGS as Record<string, { label: string; defaultUrl: string; color: string }>)[kind];
   132→  return cfg ?? { label: kind, color: 'bg-zinc-100 text-zinc-700' };
   133→}
   134→
   135→const CONNECTION_STATUS_CONFIG: Record<AiConnectionStatus, { color: string; label: string }> = {
   136→  CONNECTED: { color: 'bg-green-500', label: 'Connected' },
   137→  DISCONNECTED: { color: 'bg-zinc-400', label: 'Disconnected' },
   138→  ERROR: { color: 'bg-red-500', label: 'Error' },
   139→};
   140→
   141→const emptyForm: ProviderFormData = {
   142→  name: '',
   143→  kind: 'OPENAI',
   144→  baseUrl: PROVIDER_CONFIGS.OPENAI.defaultUrl,
   145→  apiKey: '',
   146→  apiVersion: '',
   147→  isActive: true,
   148→};

   150→// -------------------- Component --------------------
   151→
   152→export function ProvidersPage() {
   153→  const queryClient = useQueryClient();
   154→  const [page, setPage] = useState(1);
   155→  const [pageSize] = useState(25);
   156→  const [search, setSearch] = useState('');
   157→  const [kindFilter, setKindFilter] = useState<string>('all');
   158→  const [statusFilter, setStatusFilter] = useState<string>('all');
   159→  const [dialogOpen, setDialogOpen] = useState(false);
   160→  const [editingProvider, setEditingProvider] = useState<AiProvider | null>(null);
   161→  const [formData, setFormData] = useState<ProviderFormData>(emptyForm);
   162→  const [showApiKey, setShowApiKey] = useState(false);
   163→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   164→  const [deletingProvider, setDeletingProvider] = useState<AiProvider | null>(null);
   165→
   166→  // Fetch providers
   167→  const { data, isLoading, isError } = useQuery({
   168→    queryKey: queryKeys.aiProviders.list({ page, pageSize, search, kind: kindFilter, connectionStatus: statusFilter }),
   169→    queryFn: () => getApi<PaginatedResponse<AiProvider>>('/api/ai/providers', {
   170→      page,
   171→      pageSize,
   172→      search: search || undefined,
   173→      kind: kindFilter !== 'all' ? kindFilter : undefined,
   174→      connectionStatus: statusFilter !== 'all' ? statusFilter : undefined,
   175→    }),
   176→  });
   177→
   178→  const providers = data?.data ?? [];
   179→  const pagination = data?.pagination;
   180→
   181→  // Create / Update mutation
   182→  const saveMutation = useMutation({
   183→    mutationFn: (body: ProviderFormData) => {
   184→      if (editingProvider) {
   185→        const patchBody: Record<string, unknown> = { ...body };
   186→        if (!patchBody.apiKey) delete patchBody.apiKey;
   187→        return patchApi<AiProvider>(`/api/ai/providers/${editingProvider.id}`, patchBody);
   188→      }
   189→      return postApi<AiProvider>('/api/ai/providers', body);
   190→    },
   191→    onSuccess: () => {
   192→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   193→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   194→      toast.success(editingProvider ? 'Provider updated' : 'Provider created');
   195→      handleCloseDialog();
   196→    },
   197→    onError: (err: Error) => {
   198→      toast.error(err.message || 'Failed to save provider');
   199→    },
   200→  });
   201→
   202→  // Delete mutation
   203→  const deleteMutation = useMutation({
   204→    mutationFn: (id: string) => deleteApi(`/api/ai/providers/${id}`),
   205→    onSuccess: () => {
   206→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   207→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   208→      queryClient.invalidateQueries({ queryKey: queryKeys.aiSettings.all });
   209→      toast.success('Provider deleted');
   210→      setDeleteDialogOpen(false);
   211→      setDeletingProvider(null);
   212→    },
   213→    onError: (err: Error) => {
   214→      toast.error(err.message || 'Failed to delete provider');
   215→    },
   216→  });
   217→
   218→  // Test connection mutation
   219→  const testMutation = useMutation({
   220→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/test`),
   221→    onSuccess: (result: unknown) => {
   222→      const res = result as { success?: boolean; latency?: number; message?: string; status?: string };
   223→      if (res.success) {
   224→        toast.success(`Connection successful (${res.latency ?? 0}ms)`);
   225→      } else {
   226→        toast.error(res.message || `Connection ${res.status || 'failed'}`);
   227→      }
   228→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   229→    },
   230→    onError: (err: Error) => {
   231→      toast.error(err.message || 'Connection test failed');
   232→    },
   233→  });
   234→
   235→  // Sync models mutation
   236→  const syncMutation = useMutation({
   237→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/sync-models`),
   238→    onSuccess: (result: unknown) => {
   239→      const res = result as { syncedCount?: number; count?: number };
   240→      toast.success(`Synced ${res.syncedCount ?? res.count ?? 0} models`);
   241→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   242→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   243→    },
   244→    onError: (err: Error) => {
   245→      toast.error(err.message || 'Failed to sync models');
   246→    },
   247→  });
   248→
   249→  // Set default mutation
   250→  const setDefaultMutation = useMutation({
   251→    mutationFn: (id: string) => postApi(`/api/ai/providers/${id}/set-default`),
   252→    onSuccess: () => {
   253→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   254→      toast.success('Default provider updated');
   255→    },
   256→    onError: (err: Error) => {
   257→      toast.error(err.message || 'Failed to set default');
   258→    },
   259→  });
   260→
   261→  // Toggle active mutation
   262→  const toggleActiveMutation = useMutation({
   263→    mutationFn: ({ id, isActive }: { id: string; isActive: boolean }) =>
   264→      patchApi(`/api/ai/providers/${id}`, { isActive }),
   265→    onSuccess: () => {
   266→      queryClient.invalidateQueries({ queryKey: queryKeys.aiProviders.all });
   267→      queryClient.invalidateQueries({ queryKey: queryKeys.aiModels.all });
   268→    },
   269→    onError: (err: Error) => {
   270→      toast.error(err.message || 'Failed to update provider');
   271→    },
   272→  });
   273→
   274→  // Handlers
   275→  const handleOpenCreate = () => {
   276→    setEditingProvider(null);
   277→    setFormData(emptyForm);
   278→    setShowApiKey(false);
   279→    setDialogOpen(true);
   280→  };
   281→
   282→  const handleOpenEdit = (provider: AiProvider) => {
   283→    setEditingProvider(provider);
   284→    setFormData({
   285→      name: provider.name,
   286→      kind: provider.kind,
   287→      baseUrl: provider.baseUrl,
   288→      apiKey: '', // masked in edit
   289→      apiVersion: provider.apiVersion ?? '',
   290→      isActive: provider.isActive,
   291→    });
   292→    setShowApiKey(false);
   293→    setDialogOpen(true);
   294→  };
   295→
   296→  const handleCloseDialog = () => {
   297→    setDialogOpen(false);
   298→    setEditingProvider(null);
   299→    setFormData(emptyForm);
   300→    setShowApiKey(false);
   301→  };
   302→
   303→  const handleSave = () => {
   304→    if (!formData.name.trim()) {
   305→      toast.error('Name is required');
   306→      return;
   307→    }
   308→    saveMutation.mutate(formData);
   309→  };
   310→
   311→  const handleKindChange = (kind: string) => {
   312→    const providerKind = kind as AiProviderKind;
   313→    setFormData((prev) => ({
   314→      ...prev,
   315→      kind: providerKind,
   316→      baseUrl: PROVIDER_CONFIGS[providerKind].defaultUrl,
   317→    }));
   318→  };
   319→
   320→  // KPI calculations
   321→  const totalProviders = pagination?.total ?? 0;
   322→  const connectedCount = providers.filter((p) => p.connectionStatus === 'CONNECTED').length;
   323→  const defaultProvider = providers.find((p) => p.isDefault);
   324→
   325→  return (
   326→    <div className="space-y-6">
   327→      {/* KPI Cards */}
   328→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   329→        <Card>
   330→          <CardContent className="p-4 flex items-center gap-4">
   331→            <div className="p-2 rounded-lg bg-zinc-100">
   332→              <Server className="h-5 w-5 text-zinc-600" />
   333→            </div>
   334→            <div>
   335→              <p className="text-sm text-zinc-500">Total Providers</p>
   336→              <div className="text-2xl font-bold">{isLoading ? <Skeleton className="h-8 w-12 inline-block" /> : totalProviders}</div>
   337→            </div>
   338→          </CardContent>
   339→        </Card>
   340→        <Card>
   341→          <CardContent className="p-4 flex items-center gap-4">
   342→            <div className="p-2 rounded-lg bg-green-100">
   343→              <Wifi className="h-5 w-5 text-green-600" />
   344→            </div>
   345→            <div>
   346→              <p className="text-sm text-zinc-500">Connected</p>
   347→              <div className="text-2xl font-bold">{isLoading ? <Skeleton className="h-8 w-12 inline-block" /> : connectedCount}</div>
   348→            </div>
   349→          </CardContent>
   350→        </Card>
   351→        <Card>
   352→          <CardContent className="p-4 flex items-center gap-4">
   353→            <div className="p-2 rounded-lg bg-amber-100">
   354→              <Star className="h-5 w-5 text-amber-600" />
   355→            </div>
   356→            <div>
   357→              <p className="text-sm text-zinc-500">Default Provider</p>
   358→              <div className="text-lg font-bold truncate max-w-[200px]">{isLoading ? <Skeleton className="h-6 w-32 inline-block" /> : defaultProvider?.name ?? 'None'}</div>
   359→            </div>
   360→          </CardContent>
   361→        </Card>
   362→      </div>
   363→
   364→      {/* Filters & Actions */}
   365→      <Card>
   366→        <CardHeader className="pb-4">
   367→          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   368→            <CardTitle className="text-lg">AI Providers</CardTitle>
   369→            <Button onClick={handleOpenCreate} size="sm">
   370→              <Plus className="h-4 w-4 mr-2" />
   371→              Add Provider
   372→            </Button>
   373→          </div>
   374→          <div className="flex flex-col sm:flex-row gap-3 mt-2">
   375→            <div className="relative flex-1">
   376→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-zinc-400" />
   377→              <Input
   378→                placeholder="Search providers..."
   379→                value={search}
   380→                onChange={(e) => { setSearch(e.target.value); setPage(1); }}
   381→                className="pl-9"
   382→              />
   383→            </div>
   384→            <Select value={kindFilter} onValueChange={(v) => { setKindFilter(v); setPage(1); }}>
   385→              <SelectTrigger className="w-full sm:w-[180px]">
   386→                <SelectValue placeholder="Provider Kind" />
   387→              </SelectTrigger>
   388→              <SelectContent>
   389→                <SelectItem value="all">All Kinds</SelectItem>
   390→                {PROVIDER_KINDS.map((k) => (
   391→                  <SelectItem key={k} value={k}>{PROVIDER_CONFIGS[k].label}</SelectItem>
   392→                ))}
   393→              </SelectContent>
   394→            </Select>
   395→            <Select value={statusFilter} onValueChange={(v) => { setStatusFilter(v); setPage(1); }}>
   396→              <SelectTrigger className="w-full sm:w-[180px]">
   397→                <SelectValue placeholder="Status" />
   398→              </SelectTrigger>
   399→              <SelectContent>
   400→                <SelectItem value="all">All Status</SelectItem>
   401→                <SelectItem value="CONNECTED">Connected</SelectItem>
   402→                <SelectItem value="DISCONNECTED">Disconnected</SelectItem>
   403→                <SelectItem value="ERROR">Error</SelectItem>
   404→              </SelectContent>
   405→            </Select>
   406→          </div>
   407→        </CardHeader>
   408→        <CardContent className="p-0">
   409→          <ScrollArea className="max-h-[500px]">
   410→            <Table>
   411→              <TableHeader>
   412→                <TableRow>
   413→                  <TableHead>Name</TableHead>
   414→                  <TableHead>Kind</TableHead>
   415→                  <TableHead>Status</TableHead>
   416→                  <TableHead className="hidden md:table-cell">Latency</TableHead>
   417→                  <TableHead className="hidden lg:table-cell">Last Sync</TableHead>
   418→                  <TableHead>Default</TableHead>
   419→                  <TableHead>Active</TableHead>
   420→                  <TableHead className="text-right">Actions</TableHead>
   421→                </TableRow>
   422→              </TableHeader>
   423→              <TableBody>
   424→                {isLoading ? (
   425→                  Array.from({ length: 5 }).map((_, i) => (
   426→                    <TableRow key={i}>
   427→                      {Array.from({ length: 8 }).map((_, j) => (
   428→                        <TableCell key={j}><Skeleton className="h-5 w-20" /></TableCell>
   429→                      ))}
   430→                    </TableRow>
   431→                  ))
   432→                ) : isError ? (
   433→                  <TableRow>
   434→                    <TableCell colSpan={8} className="text-center py-8 text-zinc-500">
   435→                      <AlertCircle className="h-8 w-8 mx-auto mb-2 text-red-400" />
   436→                      Failed to load providers
   437→                    </TableCell>
   438→                  </TableRow>
   439→                ) : providers.length === 0 ? (
   440→                  <TableRow>
   441→                    <TableCell colSpan={8} className="text-center py-8 text-zinc-500">
   442→                      <Server className="h-8 w-8 mx-auto mb-2 text-zinc-300" />
   443→                      No providers found. Click &quot;Add Provider&quot; to get started.
   444→                    </TableCell>
   445→                  </TableRow>
   446→                ) : (
   447→                  providers.map((provider) => {
   448→                    const kc = kindConfig(provider.kind);
   449→                    const statusConfig = CONNECTION_STATUS_CONFIG[provider.connectionStatus];
   450→                    return (
   451→                      <TableRow key={provider.id}>
   452→                        <TableCell className="font-medium">{provider.name}</TableCell>
   453→                        <TableCell>
   454→                          <Badge variant="secondary" className={kc.color}>
   455→                            {kc.label}
   456→                          </Badge>
   457→                        </TableCell>
   458→                        <TableCell>
   459→                          <div className="flex items-center gap-2">
   460→                            <span className={`h-2.5 w-2.5 rounded-full ${statusConfig.color}`} />
   461→                            <span className="text-sm">{statusConfig.label}</span>
   462→                          </div>
   463→                        </TableCell>
   464→                        <TableCell className="hidden md:table-cell">
   465→                          {provider.latency != null ? `${provider.latency}ms` : '—'}
   466→                        </TableCell>
   467→                        <TableCell className="hidden lg:table-cell text-sm text-zinc-500">
   468→                          {provider.lastSyncAt
   469→                            ? new Date(provider.lastSyncAt).toLocaleDateString()
   470→                            : '—'}
   471→                        </TableCell>
   472→                        <TableCell>
   473→                          {provider.isDefault && (
   474→                            <Badge variant="secondary" className="bg-amber-100 text-amber-700">
   475→                              Default
   476→                            </Badge>
   477→                          )}
   478→                        </TableCell>
   479→                        <TableCell>
   480→                          <Switch
   481→                            checked={provider.isActive}
   482→                            onCheckedChange={(checked) =>
   483→                              toggleActiveMutation.mutate({ id: provider.id, isActive: checked })
   484→                            }
   485→                          />
   486→                        </TableCell>
   487→                        <TableCell className="text-right">
   488→                          <DropdownMenu>
   489→                            <DropdownMenuTrigger asChild>
   490→                              <Button variant="ghost" size="sm">
   491→                                <MoreHorizontal className="h-4 w-4" />
   492→                              </Button>
   493→                            </DropdownMenuTrigger>
   494→                            <DropdownMenuContent align="end">
   495→                              <DropdownMenuItem onClick={() => testMutation.mutate(provider.id)} disabled={testMutation.isPending}>
   496→                                <Zap className="h-4 w-4 mr-2" />
   497→                                Test Connection
   498→                              </DropdownMenuItem>
   499→                              <DropdownMenuItem onClick={() => syncMutation.mutate(provider.id)} disabled={syncMutation.isPending}>
   500→                                <RefreshCw className="h-4 w-4 mr-2" />
   501→                                Sync Models
   502→                              </DropdownMenuItem>
   503→                              {!provider.isDefault && (
   504→                                <DropdownMenuItem onClick={() => setDefaultMutation.mutate(provider.id)}>
   505→                                  <Star className="h-4 w-4 mr-2" />
   506→                                  Set Default
   507→                                </DropdownMenuItem>
   508→                              )}
   509→                              <DropdownMenuSeparator />
   510→                              <DropdownMenuItem onClick={() => handleOpenEdit(provider)}>
   511→                                <Pencil className="h-4 w-4 mr-2" />
   512→                                Edit
   513→                              </DropdownMenuItem>
   514→                              <DropdownMenuItem
   515→                                className="text-red-600"
   516→                                onClick={() => { setDeletingProvider(provider); setDeleteDialogOpen(true); }}
   517→                              >
   518→                                <Trash2 className="h-4 w-4 mr-2" />
   519→                                Delete
   520→                              </DropdownMenuItem>
   521→                            </DropdownMenuContent>
   522→                          </DropdownMenu>
   523→                        </TableCell>
   524→                      </TableRow>
   525→                    );
   526→                  })
   527→                )}
   528→              </TableBody>
   529→            </Table>
   530→            <ScrollBar />
   531→          </ScrollArea>
   532→
   533→          {/* Pagination */}
   534→          {pagination && pagination.totalPages > 1 && (
   535→            <div className="flex items-center justify-between px-4 py-3 border-t">
   536→              <p className="text-sm text-zinc-500">
   537→                Showing {(pagination.page - 1) * pagination.pageSize + 1}–
   538→                {Math.min(pagination.page * pagination.pageSize, pagination.total)} of {pagination.total}
   539→              </p>
   540→              <div className="flex items-center gap-2">
   541→                <Button
   542→                  variant="outline" size="sm"
   543→                  disabled={pagination.page <= 1}
   544→                  onClick={() => setPage((p) => p - 1)}
   545→                >
   546→                  <ChevronLeft className="h-4 w-4" />
   547→                </Button>
   548→                <span className="text-sm">Page {pagination.page} of {pagination.totalPages}</span>
   549→                <Button
   550→                  variant="outline" size="sm"
   551→                  disabled={pagination.page >= pagination.totalPages}
   552→                  onClick={() => setPage((p) => p + 1)}
   553→                >
   554→                  <ChevronRight className="h-4 w-4" />
   555→                </Button>
   556→              </div>
   557→            </div>
   558→          )}
   559→        </CardContent>
   560→      </Card>
   561→
   562→      {/* Add/Edit Dialog */}
   563→      <Dialog open={dialogOpen} onOpenChange={(open) => { if (!open) handleCloseDialog(); }}>
   564→        <DialogContent className="sm:max-w-[500px] max-h-[90vh] overflow-y-auto">
   565→          <DialogHeader>
   566→            <DialogTitle>{editingProvider ? 'Edit Provider' : 'Add Provider'}</DialogTitle>
   567→          </DialogHeader>
   568→          <div className="grid gap-4 py-4">
   569→            <div className="grid gap-2">
   570→              <Label htmlFor="provider-name">Name</Label>
   571→              <Input
   572→                id="provider-name"
   573→                placeholder="e.g. My OpenAI"
   574→                value={formData.name}
   575→                onChange={(e) => setFormData((p) => ({ ...p, name: e.target.value }))}
   576→              />
   577→            </div>
   578→            <div className="grid gap-2">
   579→              <Label htmlFor="provider-kind">Provider Kind</Label>
   580→              <Select value={formData.kind} onValueChange={handleKindChange}>
   581→                <SelectTrigger id="provider-kind">
   582→                  <SelectValue />
   583→                </SelectTrigger>
   584→                <SelectContent>
   585→                  {PROVIDER_KINDS.map((k) => (
   586→                    <SelectItem key={k} value={k}>{PROVIDER_CONFIGS[k].label}</SelectItem>
   587→                  ))}
   588→                </SelectContent>
   589→              </Select>
   590→            </div>
   591→            <div className="grid gap-2">
   592→              <Label htmlFor="provider-url">Base URL</Label>
   593→              <Input
   594→                id="provider-url"
   595→                placeholder="https://api.openai.com/v1"
   596→                value={formData.baseUrl}
   597→                onChange={(e) => setFormData((p) => ({ ...p, baseUrl: e.target.value }))}
   598→              />
   599→            </div>
   600→            <div className="grid gap-2">
   601→              <Label htmlFor="provider-key">API Key</Label>
   602→              <div className="relative">
   603→                <Input
   604→                  id="provider-key"
   605→                  type={showApiKey ? 'text' : 'password'}
   606→                  placeholder={editingProvider ? 'Leave blank to keep existing key' : 'sk-...'}
   607→                  value={formData.apiKey}
   608→                  onChange={(e) => setFormData((p) => ({ ...p, apiKey: e.target.value }))}
   609→                />
   610→                <Button
   611→                  type="button"
   612→                  variant="ghost"
   613→                  size="sm"
   614→                  className="absolute right-1 top-1/2 -translate-y-1/2 h-7 w-7 p-0"
   615→                  onClick={() => setShowApiKey(!showApiKey)}
   616→                >
   617→                  {showApiKey ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   618→                </Button>
   619→              </div>
   620→            </div>
   621→            {formData.kind === 'AZURE_OPENAI' && (
   622→              <p className="text-xs text-muted-foreground">
   623→                Note: Azure OpenAI is a legacy kind and no longer configurable from this form.
   624→              </p>
   625→            )}
   626→            <div className="flex items-center justify-between">
   627→              <Label htmlFor="provider-active">Active</Label>
   628→              <Switch
   629→                id="provider-active"
   630→                checked={formData.isActive}
   631→                onCheckedChange={(checked) => setFormData((p) => ({ ...p, isActive: checked }))}
   632→              />
   633→            </div>
   634→          </div>
   635→          <DialogFooter>
   636→            <Button variant="outline" onClick={handleCloseDialog}>Cancel</Button>
   637→            <Button onClick={handleSave} disabled={saveMutation.isPending}>
   638→              {saveMutation.isPending && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   639→              {editingProvider ? 'Update' : 'Create'}
   640→            </Button>
   641→          </DialogFooter>
   642→        </DialogContent>
   643→      </Dialog>
   644→
   645→      {/* Delete Confirmation */}
   646→      <AlertDialog open={deleteDialogOpen} onOpenChange={setDeleteDialogOpen}>
   647→        <AlertDialogContent>
   648→          <AlertDialogHeader>
   649→            <AlertDialogTitle>Delete Provider</AlertDialogTitle>
   650→            <AlertDialogDescription>
   651→              Are you sure you want to delete &quot;{deletingProvider?.name}&quot;? This action cannot be undone.
   652→            </AlertDialogDescription>
   653→          </AlertDialogHeader>
   654→          <AlertDialogFooter>
   655→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   656→            <AlertDialogAction
   657→              className="bg-red-600 hover:bg-red-700"
   658→              onClick={() => deletingProvider && deleteMutation.mutate(deletingProvider.id)}
   659→              disabled={deleteMutation.isPending}
   660→            >
   661→              {deleteMutation.isPending && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   662→              Delete
   663→            </AlertDialogAction>
   664→          </AlertDialogFooter>
   665→        </AlertDialogContent>
   666→      </AlertDialog>
   667→    </div>
   668→  );
   669→}
