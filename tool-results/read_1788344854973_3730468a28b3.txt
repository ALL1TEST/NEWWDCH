    87→    87→function formatCompactNumber(n: number | null | undefined): string {
    88→    88→  const num = n ?? 0;
    89→    89→  if (num >= 1_000_000) return `${(num / 1_000_000).toFixed(1)}M`;
    90→    90→  if (num >= 1_000) return `${(num / 1_000).toFixed(1)}K`;
    91→    91→  return num.toLocaleString();
    92→    92→}

   102→   102→// ==================== Health Score Ring ====================
   103→   103→
   104→   104→function ScoreRing({ score }: { score: number }) {
   105→   105→  const clampedScore = Math.max(0, Math.min(100, score));
   106→   106→  const radius = 54;
   107→   107→  const circumference = 2 * Math.PI * radius;
   108→   108→  const offset = circumference - (clampedScore / 100) * circumference;
   109→   109→  const color = clampedScore >= 80 ? 'text-green-500' : clampedScore >= 50 ? 'text-amber-500' : 'text-red-500';
   110→   110→  const bgColor = clampedScore >= 80 ? 'bg-green-100 dark:bg-green-900/20' : clampedScore >= 50 ? 'bg-amber-100 dark:bg-amber-900/20' : 'bg-red-100 dark:bg-red-900/20';
   111→   111→  const label = clampedScore >= 80 ? 'Healthy' : clampedScore >= 50 ? 'Needs Work' : 'Critical';
   112→   112→  const labelColor = clampedScore >= 80 ? 'text-green-600 dark:text-green-400' : clampedScore >= 50 ? 'text-amber-600 dark:text-amber-400' : 'text-red-600 dark:text-red-400';
   113→   113→
   114→   114→  return (
   115→   115→    <div className="flex flex-col items-center">
   116→   116→      <div className={cn('relative flex items-center justify-center w-36 h-36 rounded-full', bgColor)}>
   117→   117→        <svg className="w-full h-full -rotate-90" viewBox="0 0 120 120">
   118→   118→          <circle cx="60" cy="60" r={radius} fill="none" stroke="currentColor" strokeWidth="6" className="text-border/40" />
   119→   119→          <circle
   120→   120→            cx="60" cy="60" r={radius} fill="none" strokeWidth="6" strokeLinecap="round"
   121→   121→            className={color} strokeDasharray={circumference} strokeDashoffset={offset}
   122→   122→            style={{ transition: 'stroke-dashoffset 1s ease-in-out' }}
   123→   123→          />
   124→   124→        </svg>
   125→   125→        <div className="absolute flex flex-col items-center">
   126→   126→          <span className={cn('text-3xl font-bold tabular-nums', color)}>{clampedScore}</span>
   127→   127→          <span className="text-[10px] text-muted-foreground">/100</span>
   128→   128→        </div>
   129→   129→      </div>
   130→   130→      <p className="text-sm font-semibold mt-3">SEO Health</p>
   131→   131→      <span className={cn('text-xs font-medium mt-0.5', labelColor)}>{label}</span>
   132→   132→    </div>
   133→   133→  );
   134→   134→}
