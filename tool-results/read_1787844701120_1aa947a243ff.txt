   230→/**
   231→ * Derive the top-level section ID from the current browser hash.
   232→ * Uses EXPLICIT prefix matching — #settings/seo always returns 'Settings', never 'SEO'.
   233→ */
   234→function getSectionFromHash(hash: string): string | null {
   235→  const path = hash.replace(/^#\/?,?/, '').trim();
   236→  if (!path || path === 'dashboard') return null;
   237→
   238→  const firstSegment = path.split('/')[0];
   239→  return ROUTE_PREFIX_TO_SECTION[firstSegment] ?? null;
   240→}

   291→/**
   292→ * Collapsed-rail logo cell. Since the extra collapse-toggle icon was removed
   293→ * from below the logo, THIS cell is the explicit expand control:
   294→ *   • identical 32×32 grid geometry as every other rail icon
   295→ *   • click toggles the sidebar open (SidebarRail edge strip still works)
   296→ * No new icon is introduced and no empty clickable ghost remains.
   297→ *
   298→ * HOVER BEHAVIOR (collapsed state only — this button is rendered ONLY
   299→ * inside the collapsed-rail cluster, so the expanded LogoMark is untouched):
   300→ *   • at rest ............ shows the "C" logo mark on its bg-primary black
   301→ *                           box (NORMAL LOGO — background UNCHANGED, same
   302→ *                           as the expanded LogoMark).
   303→ *   • on mouse-enter ..... the "C" is replaced by a PanelLeftOpen icon AND
   304→ *                           the button's background becomes transparent so
   305→ *                           ONLY the Expand icon is visible with no box
   306→ *                           behind it. The icon color switches to
   307→ *                           muted-foreground (gray) so it stays visible
   308→ *                           against the transparent/page bg in both Light
   309→ *                           and Dark mode.
   310→ *   • on mouse-leave ..... restores the "C" logo mark AND its bg-primary box.
   311→ *   • click ............. calls toggleSidebar (existing functionality, unchanged).
   312→ *
   313→ * Background is removed ONLY from the temporary Expand-icon state; the
   314→ * normal logo (at rest) keeps its bg-primary background exactly as-is.
   315→ *
   316→ * Implementation: React state (useState + onMouseEnter/onMouseLeave) — NOT
   317→ * CSS :hover / group-hover variants. This is deliberate: Tailwind v4 wraps
   318→ * its hover:/group-hover: utilities inside @media (hover: hover), so on
   319→ * any browser that reports (hover: none) (headless, touch, some preview
   320→ * iframes) those utilities NEVER activate and the swap + bg-transparent
   321→ * would silently fail. React mouse events fire on ANY pointer input
   322→ * regardless of the hover media query, so the behavior is identical in
   323→ * every environment. The conditional className + conditional render swap
   324→ * the bg/color/glyph atomically per state. Works in both Light and Dark.
   325→ */
   326→function CollapsedLogoButton() {
   327→  const { toggleSidebar } = useSidebar();
   328→  // Local hover state drives the logo↔Expand-icon swap AND the per-state
   329→  // background/color. Using React state (not CSS :hover / group-hover)
   330→  // guarantees the swap works in ALL environments — including browsers
   331→  // that report (hover: none), where Tailwind's @media (hover: hover)-
   332→  // gated hover:/group-hover: variants would never activate.
   333→  const [hovered, setHovered] = useState(false);
   334→  return (
   335→    /* Hover-only "Expand" tooltip: instant (provider delayDuration=0),
   336→       appears on mouse-enter, disappears on mouse-leave. Radix closes it
   337→       on any pointer exit — disableHoverableContent guarantees it can
   338→       never linger or trap the pointer. */
   339→    <Tooltip disableHoverableContent>
   340→      <TooltipTrigger asChild>
   341→        <button
   342→          type="button"
   343→          onClick={toggleSidebar}
   344→          aria-label="Expand sidebar"
   345→          onMouseEnter={() => setHovered(true)}
   346→          onMouseLeave={() => setHovered(false)}
   347→          className={cn(
   348→            'flex h-8 w-8 shrink-0 cursor-pointer items-center justify-center rounded-lg font-bold text-sm outline-none transition-opacity hover:opacity-90 focus-visible:ring-2 focus-visible:ring-ring select-none',
   349→            // NORMAL LOGO STATE (at rest): bg-primary black box +
   350→            // text-primary-foreground white "C" — background UNCHANGED,
   351→            // same as the expanded LogoMark.
   352→            // EXPAND-ICON STATE (on hover): bg-transparent (NO background)
   353→            // + text-muted-foreground gray icon — ONLY this temporary
   354→            // state loses the background. The icon stays visible (gray
   355→            // on transparent/page bg) in both Light and Dark mode.
   356→            hovered
   357→              ? 'bg-transparent text-muted-foreground'
   358→              : 'bg-primary text-primary-foreground',
   359→          )}
   360→        >
   361→          {/* Conditional render: "C" logo at rest, PanelLeftOpen icon on
   362→              hover. h-4 w-4 matches the CollapseToggle's PanelLeftClose
   363→              icon size; the icon is the visual opposite of the collapse
   364→              icon so the collapsed logo reads as a clear "expand"
   365→              affordance when the pointer is over it. */}
   366→          {hovered ? (
   367→            <PanelLeftOpen className="h-4 w-4" />
   368→          ) : (
   369→            <span>C</span>
   370→          )}
   371→          <span className="sr-only">Expand sidebar</span>
   372→        </button>
   373→      </TooltipTrigger>
   374→      <TooltipContent side="right" sideOffset={8}>Expand</TooltipContent>
   375→    </Tooltip>
   376→  );
   377→}

   447→/**
   448→ * Settings parent item rendered on the COLLAPSED 48px rail.
   449→ * Clicking it opens a portal-based popover to the RIGHT of the icon —
   450→ * the sidebar itself NEVER expands, and the popover anchors to this
   451→ * exact button so the "star/sparkle alignment" grid stays intact.
   452→ */
   453→function CollapsedParentNavItem({
   454→  item,
   455→  currentModule,
   456→  currentSubPage,
   457→}: {
   458→  item: NavItem;
   459→  currentModule: string;
   460→  currentSubPage: string | null;
   461→}) {
   462→  const [floatOpen, setFloatOpen] = useState(false);
   463→  const mod = hrefToModule(item.href);
   464→  const isActive = currentModule === mod;
   465→
   466→  const handleChildNavigate = (
   467→    e: React.MouseEvent<HTMLAnchorElement>,
   468→    child: NavItem,
   469→  ) => {
   470→    e.preventDefault();
   471→    e.stopPropagation();
   472→    setFloatOpen(false); // close the floating submenu after navigation…
   473→    // …but do NOT touch sidebar expansion state (rule #7).
   474→    const hash = child.href.replace(/^#/, '');
   475→    const parts = hash.split('/');
   476→    const childSubPage = parts[1] || null;
   477→    useNavigationStore.getState().navigate(parts[0], null, childSubPage);
   478→  };
   479→
   480→  return (
   481→    <SidebarMenuItem>
   482→      <Popover open={floatOpen} onOpenChange={setFloatOpen}>
   483→        <PopoverTrigger asChild>
   484→          <SidebarMenuButton
   485→            isActive={isActive || floatOpen}
   486→            aria-expanded={floatOpen}
   487→            aria-haspopup="menu"
   488→            tooltip={floatOpen ? undefined : item.label}
   489→            onClick={(e: React.MouseEvent) => {
   490→              e.preventDefault();
   491→              setFloatOpen((o) => !o);
   492→            }}
   493→          >
   494→            <NavIcon name={item.icon} />
   495→            <span>{item.label}</span>
   496→          </SidebarMenuButton>
   497→        </PopoverTrigger>
   498→        {/* Portal → document.body: immune to sidebar overflow clipping.
   499→            Radix handles Esc + outside-click closing and re-anchors at any
   500→            viewport size/zoom via avoidCollisions. */}
   501→        <PopoverContent
   502→          side="right"
   503→          align="start"
   504→          sideOffset={6}
   505→          collisionPadding={12}
   506→          className="w-56 rounded-md border bg-popover p-1.5 shadow-md"
   507→          onEscapeKeyDown={() => setFloatOpen(false)}
   508→        >
   509→          <ul
   510→            role="menu"
   511→            aria-label={`${item.label} submenu`}
   512→            className="flex min-w-0 list-none flex-col gap-0.5 p-0 m-0"
   513→          >
   514→            {item.children!.map((child) => {
   515→              const hash = child.href.replace(/^#/, '');
   516→              const parts = hash.split('/');
   517→              const childSubPage = parts[1] || null;
   518→              const isChildActive =
   519→                currentModule === parts[0] &&
   520→                (!childSubPage || currentSubPage === childSubPage);
   521→              return (
   522→                <SidebarMenuSubItem key={child.label} className="list-none p-0">
   523→                  <SidebarMenuSubButton asChild isActive={isChildActive}>
   524→                    <a
   525→                      href={child.href}
   526→                      role="menuitem"
   527→                      onClick={(e) => handleChildNavigate(e, child)}
   528→                    >
   529→                      <NavIcon name={child.icon} />
   530→                      <span>{child.label}</span>
   531→                    </a>
   532→                  </SidebarMenuSubButton>
   533→                </SidebarMenuSubItem>
   534→              );
   535→            })}
   536→          </ul>
   537→        </PopoverContent>
   538→      </Popover>
   539→    </SidebarMenuItem>
   540→  );
   541→}

   543→function ExpandableNavItem({
   544→  item,
   545→  currentModule,
   546→  currentSubPage,
   547→  isExpanded,
   548→  onToggle,
   549→}: {
   550→  item: NavItem;
   551→  currentModule: string;
   552→  currentSubPage: string | null;
   553→  isExpanded: boolean;
   554→  onToggle: () => void;
   555→}) {
   556→  const mod = hrefToModule(item.href);
   557→  const isActive = currentModule === mod;
   558→  const sectionId = `submenu-${item.label.toLowerCase().replace(/\s+/g, '-')}`;
   559→
   560→  return (
   561→    <SidebarMenuItem>
   562→      <SidebarMenuButton
   563→        tooltip={item.label}
   564→        isActive={isActive}
   565→        onClick={(e: React.MouseEvent) => {
   566→          e.preventDefault();
   567→          onToggle();
   568→        }}
   569→        aria-expanded={isExpanded}
   570→        aria-controls={sectionId}
   571→      >
   572→        <NavIcon name={item.icon} />
   573→        <span>{item.label}</span>
   574→        <ChevronRight
   575→          className={cn(
   576→            'ml-auto h-4 w-4 shrink-0 transition-transform duration-200',
   577→            isExpanded && 'rotate-90',
   578→          )}
   579→        />
   580→      </SidebarMenuButton>
   581→
   582→      {item.badge != null && (
   583→        <SidebarMenuBadge>
   584→          {item.badge}
   585→        </SidebarMenuBadge>
   586→      )}
   587→
   588→      <AccordionSubmenu isOpen={isExpanded} sectionLabel={item.label}>
   589→        {item.children!.map((child) => {
   590→          const hash = child.href.replace(/^#/, '');
   591→          const parts = hash.split('/');
   592→          const childMod = parts[0];
   593→          const childSubPage = parts[1] || null;
   594→          const isChildActive =
   595→            currentModule === childMod &&
   596→            (!childSubPage || currentSubPage === childSubPage);
   597→
   598→          return (
   599→            <SidebarMenuSubItem key={child.label}>
   600→              <SidebarMenuSubButton
   601→                asChild
   602→                isActive={isChildActive}
   603→              >
   604→                <a
   605→                  href={child.href}
   606→                  onClick={(e) => {
   607→                    e.preventDefault();
   608→                    e.stopPropagation();
   609→                    useNavigationStore
   610→                      .getState()
   611→                      .navigate(childMod, null, childSubPage);
   612→                  }}
   613→                >
   614→                  <NavIcon name={child.icon} />
   615→                  <span>{child.label}</span>
   616→                </a>
   617→              </SidebarMenuSubButton>
   618→            </SidebarMenuSubItem>
   619→          );
   620→        })}
   621→      </AccordionSubmenu>

   626→function SimpleNavItem({
   627→  item,
   628→  currentModule,
   629→}: {
   630→  item: NavItem;
   631→  currentModule: string;
   632→}) {
   633→  const mod = hrefToModule(item.href);
   634→  const isActive = currentModule === mod;
   635→
   636→  return (
   637→    <SidebarMenuItem>
   638→      <SidebarMenuButton
   639→        asChild
   640→        isActive={isActive}
   641→        tooltip={item.label}
   642→      >
   643→        <a
   644→          href={item.href}
   645→          onClick={(e) => {
   646→            e.preventDefault();
   647→            useNavigationStore.getState().navigate(mod);
   648→          }}
   649→        >
   650→          <NavIcon name={item.icon} />
   651→          <span>{item.label}</span>
   652→        </a>
   653→      </SidebarMenuButton>
   654→
   655→      {item.badge != null && (
   656→        <SidebarMenuBadge>
   657→          {item.badge}
   658→        </SidebarMenuBadge>
   659→      )}
   660→    </SidebarMenuItem>
   661→  );
   662→}

   666→function NavGroupSection({
   667→  group,
   668→  currentModule,
   669→  currentSubPage,
   670→  openSection,
   671→  onToggleSection,
   672→}: {
   673→  group: NavGroup;
   674→  currentModule: string;
   675→  currentSubPage: string | null;
   676→  openSection: string | null;
   677→  onToggleSection: (label: string) => void;
   678→}) {
   679→  const isCollapsedRail = useCollapsedRail();
   680→
   681→  return (
   682→    <SidebarGroup>
   683→      <SidebarGroupContent>
   684→        <SidebarMenu>
   685→          {group.items.map((item) => {
   686→            const hasChildren = item.children && item.children.length > 0;
   687→
   688→            if (hasChildren && isCollapsedRail) {
   689→              /* Collapsed rail → parents with children become floating
   690→                 popovers anchored to their own icon. The rail NEVER
   691→                 auto-expands (#4/#5/#6). */
   692→              return (
   693→                <CollapsedParentNavItem
   694→                  key={item.label}
   695→                  item={item}
   696→                  currentModule={currentModule}
   697→                  currentSubPage={currentSubPage}
   698→                />
   699→              );
   700→            }
   701→
   702→            if (hasChildren) {
   703→              return (
   704→                <ExpandableNavItem
   705→                  key={item.label}
   706→                  item={item}
   707→                  currentModule={currentModule}
   708→                  currentSubPage={currentSubPage}
   709→                  isExpanded={openSection === item.label}
   710→                  onToggle={() => onToggleSection(item.label)}
   711→                />
   712→              );
   713→            }
   714→
   715→            return (
   716→              <SimpleNavItem
   717→                key={item.label}
   718→                item={item}
   719→                currentModule={currentModule}
   720→              />
   721→            );
   722→          })}
   723→        </SidebarMenu>
   724→      </SidebarGroupContent>
   725→    </SidebarGroup>
   726→  );
   727→}

   744→  const visibleItems = useMemo(() => {
   745→    if (!userRole) return [];
   746→    return getVisibleNavItems(userRole, NAV_ITEMS, pagePermissions);
   747→  }, [userRole, pagePermissions]);
   748→
   749→  /*
   750→   * SINGLE SOURCE OF TRUTH for the expanded top-level section.
   751→   *
   752→   * manualOverride is a tri-state:
   753→   *   null  → not overridden; use route-derived value
   754→   *   ""   → explicitly closed by user
   755→   *   "SEO" → explicitly opened by user
   756→   *
   757→   * When the module changes (navigation event), we reset manualOverride
   758→   * to null so the route-derived value takes over.
   759→   */
   760→  const [manualOverride, setManualOverride] = useState<string | null>(null);
   761→  // Track previous module to detect navigation changes synchronously during render.
   762→  // This is the React-approved pattern for "derived state from props".
   763→  const [prevModule, setPrevModule] = useState(currentModule);
   764→  if (currentModule !== prevModule) {
   765→    setPrevModule(currentModule);
   766→    setManualOverride(null);
   767→  }
   768→
   769→  // Derive the section from the current route hash (EXPLICIT prefix matching)
   770→  const routeDerivedSection = useMemo(() => {
   771→    return getSectionFromHash(window.location.hash);

   773→

   775→  const openSection = useMemo(() => {
   776→    if (manualOverride !== null) {
   777→      // User manually toggled: empty string = closed, non-empty = that section
   778→      return manualOverride || null;
   779→    }
   780→    return routeDerivedSection;
   781→  }, [manualOverride, routeDerivedSection]);

   785→    (label: string) => {
   786→      setManualOverride((prev) => {
   787→        // Determine what's currently open (considering route-derived if no override)
   788→        const currentlyOpen =
   789→          prev !== null ? (prev || null) : routeDerivedSection;
   790→        // If clicking the same section that's open, close it
   791→        if (currentlyOpen === label) {
   792→          return ''; // explicitly closed
   793→        }
   794→        // Otherwise, open the clicked section (closing any other)
   795→        return label;
   796→      });
   797→    },
   798→    [routeDerivedSection],
