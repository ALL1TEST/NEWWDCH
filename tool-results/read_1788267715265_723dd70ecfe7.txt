    45→    45→const GB_FACTOR = 1024 * 1024 * 1024;
    46→    46→
    47→    47→/** Server-side currency resolution context (from /api/platform/billing/me
    48→    48→ *  → customerCurrencyResolution). The customer never selects a currency —
    49→    49→ *  the server detects it from the request IP. */
    50→    50→interface CustomerCurrencyResolution {
    51→    51→  currency: string;
    52→    52→  countryCode: string;
    53→    53→  countryName: string;
    54→    54→  source: 'ip' | 'default' | 'local';
    55→    55→}
    56→    56→
    57→    57→/** Server-resolved FINAL price for one plan (from /api/platform/billing/me
    58→    58→ *  → planPricing[planId]). Computed by the same resolveCustomerPricing
    59→    59→ *  the checkout route uses — the price/currency displayed here is exactly
    60→    60→ *  what Stripe charges. */
    61→    61→interface ServerPlanPricing {
    62→    62→  planId: string;
    63→    63→  currency: string;
    64→    64→  monthly: number;
    65→    65→  yearly: number;
    66→    66→  source: 'ip' | 'default' | 'local' | 'plan';
    67→    67→  supported: boolean;
    68→    68→  detectedCurrency: string;
    69→    69→  countryCode: string;
    70→    70→  countryName: string;
    71→    71→  regional: boolean;
    72→    72→}
    73→    73→
    74→    74→type BillingStateWithCurrency = ClientBillingState & {
    75→    75→  customerCurrencyResolution?: CustomerCurrencyResolution;
    76→    76→  customerCurrency?: string;
    77→    77→  customerCountryCode?: string;
    78→    78→  customerCountryName?: string;
    79→    79→  currencySource?: 'ip' | 'default' | 'local';
    80→    80→  planPricing?: Record<string, ServerPlanPricing>;

    82→    82→
    83→    83→/** Resolve the price + currency to display for a plan. Prefers the
    84→    84→ *  SERVER-RESOLVED pricing (planPricing — the same values checkout
    85→    85→ *  charges); falls back to a local pricesByCurrency lookup for older
    86→    86→ *  API responses. */
    87→    87→function resolvePlanPricing(
    88→    88→  plan: {
    89→    89→    id: string;
    90→    90→    pricesByCurrency?: Record<string, { monthly: number; yearly: number }>;
    91→    91→    priceMonthly: number;
    92→    92→    priceYearly: number;
    93→    93→    currency: string;
    94→    94→    interval: 'monthly' | 'yearly';
    95→    95→  },
    96→    96→  serverPricing: ServerPlanPricing | undefined,
    97→    97→  customerCurrency: string,
    98→    98→): { monthly: number; yearly: number; currency: string } {
    99→    99→  if (serverPricing) {
   100→   100→    return {
   101→   101→      monthly: serverPricing.monthly,
   102→   102→      yearly: serverPricing.yearly,
   103→   103→      currency: serverPricing.currency,
   104→   104→    };
   105→   105→  }
   106→   106→  // Legacy fallback — local lookup (display only).
   107→   107→  const entry = plan.pricesByCurrency?.[customerCurrency];
   108→   108→  if (entry) return { monthly: entry.monthly, yearly: entry.yearly, currency: customerCurrency };
   109→   109→  return { monthly: plan.priceMonthly, yearly: plan.priceYearly, currency: plan.currency };
   110→   110→}
   111→   111→
   112→   112→/** One customer-facing plan feature: a label line (rendered with a ✓)
   113→   113→ *  plus optional indented sub-lines. Used for the Platform AI block —
   114→   114→ *  its description + the plan's AI usage limits sit NEXT TO the ✓
   115→   115→ *  Platform AI feature so the customer immediately understands what
   116→   116→ *  is included (never "✓ Platform AI" alone when Platform AI is
   117→   117→ *  enabled). */
   118→   118→interface PlanFeatureItem {
   119→   119→  label: string;
   120→   120→  subLines?: string[];
   121→   121→}
   122→   122→
   123→   123→/** Compact AI-limit number for the customer-facing plan cards:
   124→   124→ *  200000 → "200K", 25000 → "25K", 1500 → "1.5K", 100 → "100",
   125→   125→ *  -1 → "Unlimited". */
   126→   126→function formatAiCount(n: number): string {
   127→   127→  if (n === UNLIMITED) return 'Unlimited';
   128→   128→  if (n >= 1000) {
   129→   129→    const k = n / 1000;
   130→   130→    return `${k % 1 === 0 ? String(k) : k.toFixed(1)}K`;
   131→   131→  }
   132→   132→  return String(n);
   133→   133→}
   134→   134→
   135→   135→/** Derive the customer-facing feature list from the plan's entitlements +
   136→   136→ *  a few standard items (max sites, storage, support tier). Uses the
   137→   137→ *  legacy plan.features array when it has entries (legacy compat for
   138→   138→ *  plans that still have manually-maintained marketing copy, e.g. the
   139→   139→ *  synthetic Internal plan). The structured entitlements are the
   140→   140→ *  single source of truth — the admin no longer maintains a separate
   141→   141→ *  marketing list per plan.
   142→   142→ *
   143→   143→ *  PLATFORM AI: the ✓ Platform AI feature item carries its own
   144→   144→ *  description ("AI provided by the platform") plus the plan's AI
   145→   145→ *  usage limits (article + image generations per month — words/tokens
   146→   146→ *  are never metered) as nested sub-lines — shown ONLY while Platform
   147→   147→ *  AI is enabled (Client's Own AI API-only plans never show platform AI
   148→   148→ *  limits; own-API usage is not counted against them). */
   149→   149→function derivePlanFeatures(plan: {
   150→   150→  features: string[];
   151→   151→  entitlements: string[];
   152→   152→  isFree: boolean;
   153→   153→  limits: {
   154→   154→    maxSites: number;
   155→   155→    storageBytes: number;
   156→   156→    aiArticlesPerMonth?: number;
   157→   157→    aiImagesPerMonth?: number;
   158→   158→  };
   159→   159→}): PlanFeatureItem[] {
   160→   160→  if (plan.features.length > 0) {
   161→   161→    // Legacy marketing copy — plain items, no sub-lines.
   162→   162→    return plan.features.map((f) => ({ label: f }));
   163→   163→  }
   164→   164→  const items: PlanFeatureItem[] = [];
   165→   165→  items.push(
   166→   166→    plan.limits.maxSites === -1
   167→   167→      ? { label: 'Unlimited sites' }
   168→   168→      : { label: `Up to ${plan.limits.maxSites} sites` },
   169→   169→  );
   170→   170→  const gb = plan.limits.storageBytes / GB_FACTOR;
   171→   171→  if (gb >= 1) {
   172→   172→    items.push({ label: `${Math.floor(gb)} GB storage` });
   173→   173→  } else if (plan.limits.storageBytes > 0) {
   174→   174→    items.push({ label: `${plan.limits.storageBytes} bytes storage` });
   175→   175→  } else {
   176→   176→    items.push({ label: 'No storage' });
   177→   177→  }
   178→   178→  items.push({ label: plan.isFree ? 'Community support' : 'Priority support' });
   179→   179→  // Enabled features — ONLY the enabled ones are displayed (each
   180→   180→  // entitlement key renders its label). Platform AI nests its limits
   181→   181→  // directly next to the feature; Client's Own AI API states that
   182→   182→  // platform AI limits do not apply to it. The legacy 'ai_content'
   183→   183→  // alias is skipped (it duplicates the AI feature keys).
   184→   184→  for (const e of plan.entitlements) {
   185→   185→    if (e === 'ai_content') continue;
   186→   186→    const label = ENTITLEMENT_LABELS[e as EntitlementKey];
   187→   187→    if (!label) continue;
   188→   188→    if (e === 'ai_platform') {
   189→   189→      const articles = plan.limits.aiArticlesPerMonth ?? 0;
   190→   190→      const images = plan.limits.aiImagesPerMonth ?? 0;
   191→   191→      // Display-only: the monthly usage-limit logic is unchanged; the
   192→   192→      // customer-facing lines just drop the "/ month" suffix (they are
   193→   193→      // usage limits, not prices) and render as separate lines.
   194→   194→      items.push({
   195→   195→        label,
   196→   196→        subLines: [
   197→   197→          'AI provided by the platform',
   198→   198→          `${formatAiCount(articles)} AI articles`,
   199→   199→          `${formatAiCount(images)} AI images`,
   200→   200→        ],
   201→   201→      });
   202→   202→      continue;
   203→   203→    }
   204→   204→    if (e === 'ai_client') {
   205→   205→      items.push({
   206→   206→        label,
   207→   207→        subLines: ["Your own AI provider — platform AI usage limits don't apply"],
   208→   208→      });
   209→   209→      continue;
   210→   210→    }
   211→   211→    items.push({ label });
   212→   212→  }
   213→   213→  return items;
   214→   214→}
   215→   215→
   216→   216→/** Render the derived plan feature items — one ✓ row per item with
   217→   217→ *  optional indented sub-lines (the Platform AI limits next to the
   218→   218→ *  ✓ Platform AI feature). */
   219→   219→function PlanFeatureList({ items }: { items: PlanFeatureItem[] }) {
   220→   220→  return (
   221→   221→    <ul className="space-y-2">
   222→   222→      {items.map((item, idx) => (
   223→   223→        <li key={`${item.label}-${idx}`} className="flex items-start gap-2 text-sm">
   224→   224→          <Check className="h-4 w-4 text-muted-foreground shrink-0 mt-0.5" />
   225→   225→          <div className="min-w-0">
   226→   226→            <span className="block">{item.label}</span>
   227→   227→            {item.subLines?.map((s) => (
   228→   228→              <span key={s} className="block text-xs text-muted-foreground leading-snug mt-0.5">
   229→   229→                {s}
   230→   230→              </span>
   231→   231→            ))}
   232→   232→          </div>
   233→   233→        </li>
   234→   234→      ))}
   235→   235→    </ul>
   236→   236→  );
   237→   237→}
