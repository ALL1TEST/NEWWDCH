     5→     5→// ============================================================
     6→     6→// Owner sees compact plan cards (Beta / Pro / Max …) with a quick
     7→     7→// Active toggle, a compact price + features + limits summary,
     8→     8→// and an "Edit Plan" button. The Edit Plan button opens a Dialog
     9→     9→// with three compact sections: Basic Information, Feature Access,
    10→    10→// Usage Limits. An optional collapsed "Client Display" section
    11→    11→// holds the marketing feature list shown to clients on the
    12→    12→// Client Billing page. A "+ Create Plan" button at the top-right

    14→    14→// POSTs to /api/platform/admin/plans.

    16→    16→// All writes go through /api/platform/admin/plans (GET list,
    17→    17→// POST create) + /api/platform/admin/plans/[planId] (PUT update)
    18→    18→// — owner-only — the same shared plan-config cache the Client
    19→    19→// Billing page and MRR read. Server-side entitlement enforcement
    20→    20→// (hasFeature) and usage-limit checks (checkLimit) consume the
    21→    21→// same data, so toggling a feature off here denies it for clients
    22→    22→// on the next request. The Client Billing page is NOT modified
    23→    23→// and continues to read from /api/platform/billing/me.

    88→    88→/** Resolve the badge id for the shared PlanBadge component. Falls
    89→    89→ *  back to 'free' for any unknown plan id so future plans still
    90→    90→ *  render with a sensible neutral badge. */
    91→    91→function getPlanBadgeId(planId: string): 'free' | 'plus' | 'pro' | 'max' {
    92→    92→  if (planId === 'free' || planId === 'plus' || planId === 'pro' || planId === 'max') return planId;
    93→    93→  // Legacy 'beta' plan id (pre-migration) → render as 'plus' badge.
    94→    94→  if (planId === 'beta') return 'plus';
    95→    95→  return 'free';
    96→    96→}

   139→   139→      const e = err as { error?: { message?: string }; message?: string };

   141→   141→        e?.error?.message ?? e?.message ?? 'Unable to change plan status. OWNER access required.',

   146→   146→  // Feature list shown on the card. Prefer the client-facing marketing

   151→   151→    plan.features.length > 0

   156→   156→    <>

   158→   158→        className={`group relative flex h-full flex-col overflow-hidden rounded-3xl border bg-card p-8 text-card-foreground transition-colors duration-300 hover:border-foreground/20 ${
   159→   159→          plan.active ? '' : 'opacity-60'
   160→   160→        }`}
   161→   161→      >
   162→   162→        {/* Header: plan name (clean text label) + quick active toggle */}
   163→   163→        <div className="flex items-start justify-between gap-3">
   164→   164→          <div className="min-w-0">
   165→   165→            <h3 className="text-2xl font-semibold tracking-tight text-foreground">{plan.name}</h3>
   166→   166→          </div>
   167→   167→          <div className="flex shrink-0 items-center gap-2">
   168→   168→            <Label
   169→   169→              htmlFor={`active-${plan.planId}`}
   170→   170→              className="cursor-pointer text-xs text-muted-foreground"
   171→   171→            >
   172→   172→              {plan.active ? 'Active' : 'Inactive'}
   173→   173→            </Label>
   174→   174→            <Switch
   175→   175→              id={`active-${plan.planId}`}
   176→   176→              checked={plan.active}
   177→   177→              disabled={!canEdit || activeMutation.isPending}
   178→   178→              onCheckedChange={(v) => activeMutation.mutate(v)}
   179→   179→            />
   180→   180→          </div>
   181→   181→        </div>
   182→   182→
   183→   183→        {/* Price — reflects the global billing-interval selector (Monthly / Yearly).
   184→   184→             Free plan always shows "Free" regardless of the selector. */}
   185→   185→        <div className="mt-8">
   186→   186→          {plan.isFree ? (
   187→   187→            <div className="flex items-baseline gap-2">
   188→   188→              <span className="text-5xl font-semibold tracking-tight text-foreground">Free</span>
   189→   189→            </div>
   190→   190→          ) : billingInterval === 'yearly' ? (
   191→   191→            <div className="flex flex-col gap-1">
   192→   192→              <div className="flex items-baseline gap-2">
   193→   193→                <span className="text-5xl font-semibold tracking-tight text-foreground">
   194→   194→                  {formatCurrency(plan.priceYearly, plan.currency)}
   195→   195→                </span>
   196→   196→                <span className="text-sm text-muted-foreground">/ year</span>
   197→   197→              </div>
   198→   198→              {plan.priceYearly > 0 && (
   199→   199→                <span className="text-xs text-muted-foreground">
   200→   200→                  ≈ {formatCurrency(Math.round(plan.priceYearly / 12), plan.currency)} / month
   201→   201→                </span>
   202→   202→              )}
   203→   203→            </div>
   204→   204→          ) : (
   205→   205→            <div className="flex items-baseline gap-2">
   206→   206→              <span className="text-5xl font-semibold tracking-tight text-foreground">
   207→   207→                {formatCurrency(plan.priceMonthly, plan.currency)}
   208→   208→              </span>
   209→   209→              <span className="text-sm text-muted-foreground">/ month</span>
   210→   210→            </div>
   211→   211→          )}
   212→   212→        </div>
   213→   213→
   214→   214→        {/* Thin horizontal divider between the price section and the features section */}
   215→   215→        <div className="mt-8 h-px bg-border" aria-hidden />
   216→   216→
   217→   217→        {/* Feature items (section label omitted) */}
   218→   218→        <div className="mt-6">
   219→   219→          <ul className="space-y-2.5">
   220→   220→            {featureItems.length === 0 ? (
   221→   221→              <li className="text-sm text-muted-foreground">No features configured</li>
   222→   222→            ) : (
   223→   223→              featureItems.map((f, i) => (
   224→   224→                <li key={`${f}-${i}`} className="flex items-start gap-2.5 text-sm text-card-foreground">
   225→   225→                  <Check className="mt-0.5 h-4 w-4 shrink-0 text-muted-foreground" />
   226→   226→                  <span className="leading-snug">{f}</span>
   227→   227→                </li>
   228→   228→              ))
   229→   229→            )}
   230→   230→          </ul>
   231→   231→        </div>
   232→   232→
   233→   233→        {/* Edit button — pinned to card bottom */}
   234→   234→        <div className="mt-auto pt-8">
   235→   235→          <Button
   236→   236→            variant="outline"
   237→   237→            size="sm"
   238→   238→            className="w-full rounded-full"
   239→   239→            disabled={!canEdit}
   240→   240→            onClick={() => setEditing(true)}
   241→   241→          >
   242→   242→            <Pencil className="h-4 w-4 mr-2" />
   243→   243→            Edit Plan
   244→   244→          </Button>
   245→   245→          {!canEdit && (
   246→   246→            <p className="mt-2 text-center text-[10px] text-muted-foreground">
   247→   247→              Editing is restricted to the OWNER role.
   248→   248→            </p>
   249→   249→          )}
   250→   250→        </div>
   251→   251→      </div>
   252→   252→
   253→   253→      {editing && (
   254→   254→        <EditPlanDialog plan={plan} open={editing} onOpenChange={setEditing} />
   255→   255→      )}
   256→   256→    </>
   257→   257→  );
   258→   258→}

   262→   262→function EditPlanDialog({
   263→   263→  plan,
   264→   264→  open,
   265→   265→  onOpenChange,
   266→   266→}: {
   267→   267→  plan: PlanConfigData;
   268→   268→  open: boolean;
   269→   269→  onOpenChange: (open: boolean) => void;
   270→   270→}) {

   273→   273→  const [priceMonthly, setPriceMonthly] = useState(String(plan.priceMonthly));

   278→   278→  const [features, setFeatures] = useState(plan.features.join('\n'));
   279→   279→  const [entitlements, setEntitlements] = useState<string[]>(plan.entitlements);
   280→   280→  const [limits, setLimits] = useState<PlanLimits>(plan.limits);
   281→   281→  // New (Task 64): free-trial duration + Stripe Price IDs.
   282→   282→  const [freePlanDurationDays, setFreePlanDurationDays] = useState<string>(
   283→   283→    plan.freePlanDurationDays == null ? '' : String(plan.freePlanDurationDays),
   284→   284→  );
   285→   285→  const [stripePriceIdMonthly, setStripePriceIdMonthly] = useState(plan.stripePriceIdMonthly ?? '');
   286→   286→  const [stripePriceIdYearly, setStripePriceIdYearly] = useState(plan.stripePriceIdYearly ?? '');
   287→   287→  const [clientDisplayOpen, setClientDisplayOpen] = useState(false);
   288→   288→  const [stripeOpen, setStripeOpen] = useState(false);
   289→   289→
   290→   290→  // Fetch the supported currency list from CountryPricing so the
   291→   291→  // currency field is a Select (not a free-text input).
   292→   292→  const currenciesQuery = useQuery({
   293→   293→    queryKey: ['platform-currencies'],
   294→   294→    queryFn: async () => {
   295→   295→      const res = await fetch('/api/platform/admin/countries', { credentials: 'include' });
   296→   296→      const json = await res.json();
   297→   297→      const rows = Array.isArray(json) ? json : (json.data ?? []);
   298→   298→      const list = (rows as { currency?: string }[])
   299→   299→        .map((r) => r.currency)
   300→   300→        .filter((c): c is string => Boolean(c) && typeof c === 'string');
   301→   301→      const unique = Array.from(new Set(list));
   302→   302→      return unique.length > 0 ? unique : ['CHF', 'USD', 'EUR', 'MAD'];
   303→   303→    },
   304→   304→    staleTime: 5 * 60 * 1000,
   305→   305→  });
   306→   306→  const currencies = currenciesQuery.data ?? ['CHF'];
   307→   307→
   308→   308→  // EditPlanDialog is conditionally rendered by the parent (mounted
   309→   309→  // fresh each time the user opens it), so the useState initializers
   310→   310→  // above already seed local state from the latest server snapshot.
   311→   311→  // No useEffect sync is needed — that would just trigger cascading
   312→   312→  // renders (and the React Compiler correctly flags it).
   313→   313→
   314→   314→  const reset = () => {
   315→   315→    setName(plan.name);
   316→   316→    setPriceMonthly(String(plan.priceMonthly));
   317→   317→    setPriceYearly(String(plan.priceYearly));
   318→   318→    setCurrency(plan.currency);
   319→   319→    setInterval(plan.interval);
   320→   320→    setActive(plan.active);
   321→   321→    setFeatures(plan.features.join('\n'));
   322→   322→    setEntitlements(plan.entitlements);
   323→   323→    setLimits(plan.limits);
   324→   324→    setFreePlanDurationDays(plan.freePlanDurationDays == null ? '' : String(plan.freePlanDurationDays));
   325→   325→    setStripePriceIdMonthly(plan.stripePriceIdMonthly ?? '');
   326→   326→    setStripePriceIdYearly(plan.stripePriceIdYearly ?? '');
   327→   327→  };
   328→   328→
   329→   329→  const saveMutation = useMutation({
   330→   330→    mutationFn: (patch: PlanPatch) =>
   331→   331→      putApi<PlanConfigData>(`/api/platform/admin/plans/${plan.planId}`, patch),
   332→   332→    onSuccess: (data) => {
   333→   333→      queryClient.invalidateQueries({ queryKey: ['platform-plans'] });
   334→   334→      queryClient.invalidateQueries({ queryKey: ['platform-billing-me'] });
   335→   335→      queryClient.invalidateQueries({ queryKey: ['platform-overview'] });

   337→   337→      onOpenChange(false);
   338→   338→    },
   339→   339→    onError: (err: unknown) => {
   340→   340→      const e = err as { error?: { message?: string }; message?: string };
   341→   341→      toast.error(
   342→   342→        e?.error?.message ?? e?.message ?? 'Unable to save plan. You may need OWNER access.',
   343→   343→      );
   344→   344→    },
   345→   345→  });
   346→   346→
   347→   347→  const buildPatch = (): PlanPatch => {
   348→   348→    const monthly = Number(priceMonthly) || 0;
   349→   349→    const yearly = Number(priceYearly) || 0;
   350→   350→    const isFreeDerived = monthly === 0 && yearly === 0;
   351→   351→    return {
   352→   352→      name,
   353→   353→      priceMonthly: monthly,
   354→   354→      priceYearly: yearly,
   355→   355→      currency,
   356→   356→      interval,
   357→   357→      isFree: isFreeDerived,
   358→   358→      freePlanDurationDays: isFreeDerived ? (freePlanDurationDays.trim() === '' ? null : Number(freePlanDurationDays) || null) : null,
   359→   359→      stripePriceIdMonthly: stripePriceIdMonthly.trim() === '' ? null : stripePriceIdMonthly.trim(),
   360→   360→      stripePriceIdYearly: stripePriceIdYearly.trim() === '' ? null : stripePriceIdYearly.trim(),
   361→   361→      active,
   362→   362→      features: features
   363→   363→        .split('\n')
   364→   364→        .map((f) => f.trim())
   365→   365→        .filter(Boolean),
   366→   366→      entitlements,
   367→   367→      limits: { ...limits, storageBytes: Number(limits.storageBytes) || 0 },
   368→   368→    };
   369→   369→  };
   370→   370→
   371→   371→  return (
   372→   372→    <Dialog open={open} onOpenChange={onOpenChange}>
   373→   373→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   374→   374→        <DialogHeader>
   375→   375→          <div className="flex items-center gap-2">
   376→   376→            <DialogTitle className="text-base">Edit {plan.name}</DialogTitle>
   377→   377→            <PlanBadge planId={getPlanBadgeId(plan.planId)} />
   378→   378→          </div>
   379→   379→          <DialogDescription className="text-xs">
   380→   380→            Single source of truth — changes propagate to the Client Billing page, MRR and
   381→   381→            server-side entitlement enforcement.
   382→   382→          </DialogDescription>
   383→   383→        </DialogHeader>
   384→   384→
   385→   385→        <div className="space-y-5">
   386→   386→          {/* -------------------- Basic Information -------------------- */}
   387→   387→          <section className="space-y-3">
   388→   388→            <h4 className="text-sm font-semibold">Basic Information</h4>
