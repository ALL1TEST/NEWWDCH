   226→   226→/**
   227→   227→ * Derive the top-level section ID from the current browser hash.
   228→   228→ * Uses EXPLICIT prefix matching — #settings/seo always returns 'Settings', never 'SEO'.

   230→   230→function getSectionFromHash(hash: string): string | null {
   231→   231→  const path = hash.replace(/^#\/?,?/, '').trim();
   232→   232→  if (!path || path === 'dashboard') return null;
   233→   233→
   234→   234→  const firstSegment = path.split('/')[0];
   235→   235→  return ROUTE_PREFIX_TO_SECTION[firstSegment] ?? null;
   236→   236→}

   377→   377→/**
   378→   378→ * Settings parent item rendered on the COLLAPSED 48px rail.
   379→   379→ * Clicking it opens a portal-based popover to the RIGHT of the icon —
   380→   380→ * the sidebar itself NEVER expands, and the popover anchors to this
   381→   381→ * exact button so the "star/sparkle alignment" grid stays intact.
   382→   382→ */
   383→   383→function CollapsedParentNavItem({
   384→   384→  item,
   385→   385→  currentModule,
   386→   386→  currentSubPage,
   387→   387→}: {
   388→   388→  item: NavItem;
   389→   389→  currentModule: string;
   390→   390→  currentSubPage: string | null;
   391→   391→}) {
   392→   392→  const [floatOpen, setFloatOpen] = useState(false);
   393→   393→  const mod = hrefToModule(item.href);
   394→   394→  const isActive = currentModule === mod;
   395→   395→
   396→   396→  const handleChildNavigate = (
   397→   397→    e: React.MouseEvent<HTMLAnchorElement>,
   398→   398→    child: NavItem,
   399→   399→  ) => {
   400→   400→    e.preventDefault();
   401→   401→    e.stopPropagation();
   402→   402→    setFloatOpen(false); // close the floating submenu after navigation…
   403→   403→    // …but do NOT touch sidebar expansion state (rule #7).
   404→   404→    const hash = child.href.replace(/^#/, '');
   405→   405→    const parts = hash.split('/');
   406→   406→    const childSubPage = parts[1] || null;
   407→   407→    useNavigationStore.getState().navigate(parts[0], null, childSubPage);
   408→   408→  };
   409→   409→
   410→   410→  return (
   411→   411→    <SidebarMenuItem>
   412→   412→      <Popover open={floatOpen} onOpenChange={setFloatOpen}>
   413→   413→        <PopoverTrigger asChild>
   414→   414→          <SidebarMenuButton
   415→   415→            isActive={isActive || floatOpen}
   416→   416→            aria-expanded={floatOpen}
   417→   417→            aria-haspopup="menu"
   418→   418→            tooltip={floatOpen ? undefined : item.label}
   419→   419→            onClick={(e: React.MouseEvent) => {
   420→   420→              e.preventDefault();
   421→   421→              setFloatOpen((o) => !o);
   422→   422→            }}
   423→   423→          >
   424→   424→            <NavIcon name={item.icon} />
   425→   425→            <span>{item.label}</span>
   426→   426→          </SidebarMenuButton>
   427→   427→        </PopoverTrigger>
   428→   428→        {/* Portal → document.body: immune to sidebar overflow clipping.
   429→   429→            Radix handles Esc + outside-click closing and re-anchors at any
   430→   430→            viewport size/zoom via avoidCollisions. */}
   431→   431→        <PopoverContent
   432→   432→          side="right"
   433→   433→          align="start"
   434→   434→          sideOffset={6}
   435→   435→          collisionPadding={12}
   436→   436→          className="w-56 rounded-md border bg-popover p-1.5 shadow-md"
   437→   437→          onEscapeKeyDown={() => setFloatOpen(false)}
   438→   438→        >
   439→   439→          <ul
   440→   440→            role="menu"
   441→   441→            aria-label={`${item.label} submenu`}
   442→   442→            className="flex min-w-0 list-none flex-col gap-0.5 p-0 m-0"
   443→   443→          >
   444→   444→            {item.children!.map((child) => {
   445→   445→              const hash = child.href.replace(/^#/, '');
   446→   446→              const parts = hash.split('/');
   447→   447→              const childSubPage = parts[1] || null;
   448→   448→              const isChildActive =
   449→   449→                currentModule === parts[0] &&
   450→   450→                (!childSubPage || currentSubPage === childSubPage);
   451→   451→              return (
   452→   452→                <SidebarMenuSubItem key={child.label} className="list-none p-0">
   453→   453→                  <SidebarMenuSubButton asChild isActive={isChildActive}>
   454→   454→                    <a
   455→   455→                      href={child.href}
   456→   456→                      role="menuitem"
   457→   457→                      onClick={(e) => handleChildNavigate(e, child)}
   458→   458→                    >
   459→   459→                      <NavIcon name={child.icon} />
   460→   460→                      <span>{child.label}</span>
   461→   461→                    </a>
   462→   462→                  </SidebarMenuSubButton>
   463→   463→                </SidebarMenuSubItem>
   464→   464→              );
   465→   465→            })}
   466→   466→          </ul>
   467→   467→        </PopoverContent>
   468→   468→      </Popover>
   469→   469→    </SidebarMenuItem>
   470→   470→  );
   471→   471→}

   473→   473→function ExpandableNavItem({
   474→   474→  item,
   475→   475→  currentModule,
   476→   476→  currentSubPage,
   477→   477→  isExpanded,
   478→   478→  onToggle,
   479→   479→}: {
   480→   480→  item: NavItem;
   481→   481→  currentModule: string;
   482→   482→  currentSubPage: string | null;
   483→   483→  isExpanded: boolean;
   484→   484→  onToggle: () => void;
   485→   485→}) {
   486→   486→  const mod = hrefToModule(item.href);
   487→   487→  const isActive = currentModule === mod;
   488→   488→  const sectionId = `submenu-${item.label.toLowerCase().replace(/\s+/g, '-')}`;
   489→   489→
   490→   490→  return (
   491→   491→    <SidebarMenuItem>
   492→   492→      <SidebarMenuButton
   493→   493→        tooltip={item.label}
   494→   494→        isActive={isActive}
   495→   495→        onClick={(e: React.MouseEvent) => {
   496→   496→          e.preventDefault();
   497→   497→          onToggle();
   498→   498→        }}
   499→   499→        aria-expanded={isExpanded}
   500→   500→        aria-controls={sectionId}
   501→   501→      >
   502→   502→        <NavIcon name={item.icon} />
   503→   503→        <span>{item.label}</span>
   504→   504→        <ChevronRight
   505→   505→          className={cn(
   506→   506→            'ml-auto h-4 w-4 shrink-0 transition-transform duration-200',
   507→   507→            isExpanded && 'rotate-90',
   508→   508→          )}
   509→   509→        />
   510→   510→      </SidebarMenuButton>
   511→   511→
   512→   512→      {item.badge != null && (
   513→   513→        <SidebarMenuBadge>
   514→   514→          {item.badge}
   515→   515→        </SidebarMenuBadge>
   516→   516→      )}
   517→   517→
   518→   518→      <AccordionSubmenu isOpen={isExpanded} sectionLabel={item.label}>
   519→   519→        {item.children!.map((child) => {
   520→   520→          const hash = child.href.replace(/^#/, '');
   521→   521→          const parts = hash.split('/');
   522→   522→          const childMod = parts[0];
   523→   523→          const childSubPage = parts[1] || null;
   524→   524→          const isChildActive =
   525→   525→            currentModule === childMod &&
   526→   526→            (!childSubPage || currentSubPage === childSubPage);
   527→   527→
   528→   528→          return (
   529→   529→            <SidebarMenuSubItem key={child.label}>
   530→   530→              <SidebarMenuSubButton
   531→   531→                asChild
   532→   532→                isActive={isChildActive}
   533→   533→              >
   534→   534→                <a
   535→   535→                  href={child.href}
   536→   536→                  onClick={(e) => {
   537→   537→                    e.preventDefault();
   538→   538→                    e.stopPropagation();
   539→   539→                    useNavigationStore
   540→   540→                      .getState()
   541→   541→                      .navigate(childMod, null, childSubPage);
   542→   542→                  }}
   543→   543→                >
   544→   544→                  <NavIcon name={child.icon} />
   545→   545→                  <span>{child.label}</span>
   546→   546→                </a>
   547→   547→              </SidebarMenuSubButton>
   548→   548→            </SidebarMenuSubItem>
   549→   549→          );
   550→   550→        })}
   551→   551→      </AccordionSubmenu>
   552→   552→    </SidebarMenuItem>
   553→   553→  );
   554→   554→}

   556→   556→function SimpleNavItem({
   557→   557→  item,
   558→   558→  currentModule,
   559→   559→}: {
   560→   560→  item: NavItem;
   561→   561→  currentModule: string;

   564→   564→  const isActive = currentModule === mod;
   565→   565→
   566→   566→  return (
   567→   567→    <SidebarMenuItem>
   568→   568→      <SidebarMenuButton
   569→   569→        asChild
   570→   570→        isActive={isActive}
   571→   571→        tooltip={item.label}
   572→   572→      >
   573→   573→        <a
   574→   574→          href={item.href}
   575→   575→          onClick={(e) => {
   576→   576→            e.preventDefault();
   577→   577→            useNavigationStore.getState().navigate(mod);
   578→   578→          }}
   579→   579→        >
   580→   580→          <NavIcon name={item.icon} />
   581→   581→          <span>{item.label}</span>
   582→   582→        </a>

   584→   584→
   585→   585→      {item.badge != null && (

   596→   596→function NavGroupSection({
   597→   597→  group,
   598→   598→  currentModule,
   599→   599→  currentSubPage,
   600→   600→  openSection,
   601→   601→  onToggleSection,
   602→   602→}: {
   603→   603→  group: NavGroup;
   604→   604→  currentModule: string;
   605→   605→  currentSubPage: string | null;
   606→   606→  openSection: string | null;
   607→   607→  onToggleSection: (label: string) => void;
   608→   608→}) {
   609→   609→  const isCollapsedRail = useCollapsedRail();
   610→   610→
   611→   611→  return (
   612→   612→    <SidebarGroup>
   613→   613→      <SidebarGroupContent>
   614→   614→        <SidebarMenu>
   615→   615→          {group.items.map((item) => {
   616→   616→            const hasChildren = item.children && item.children.length > 0;
   617→   617→
   618→   618→            if (hasChildren && isCollapsedRail) {
   619→   619→              /* Collapsed rail → parents with children become floating
   620→   620→                 popovers anchored to their own icon. The rail NEVER
   621→   621→                 auto-expands (#4/#5/#6). */
   622→   622→              return (
   623→   623→                <CollapsedParentNavItem
   624→   624→                  key={item.label}
   625→   625→                  item={item}
   626→   626→                  currentModule={currentModule}
   627→   627→                  currentSubPage={currentSubPage}
   628→   628→                />
   629→   629→              );
   630→   630→            }
   631→   631→
   632→   632→            if (hasChildren) {
   633→   633→              return (
   634→   634→                <ExpandableNavItem
   635→   635→                  key={item.label}
   636→   636→                  item={item}
   637→   637→                  currentModule={currentModule}
   638→   638→                  currentSubPage={currentSubPage}
   639→   639→                  isExpanded={openSection === item.label}
   640→   640→                  onToggle={() => onToggleSection(item.label)}
   641→   641→                />
   642→   642→              );
   643→   643→            }
   644→   644→
   645→   645→            return (
   646→   646→              <SimpleNavItem
   647→   647→                key={item.label}
   648→   648→                item={item}
   649→   649→                currentModule={currentModule}
   650→   650→              />
   651→   651→            );
   652→   652→          })}
   653→   653→        </SidebarMenu>
   654→   654→      </SidebarGroupContent>
   655→   655→    </SidebarGroup>
   656→   656→  );
   657→   657→}

   661→   661→export function AppSidebar() {
   662→   662→  const user = useAuthStore((s) => s.user);
   663→   663→  const logout = useAuthStore((s) => s.logout);
   664→   664→  const currentModule = useNavigationStore((s) => s.currentModule);
   665→   665→  const currentSubPage = useNavigationStore((s) => s.currentSubPage);
   666→   666→  // Sidebar context hook — MUST run before any early return.
   667→   667→  const sidebarCtx = useSidebar();
   668→   668→
   669→   669→  const userRole = user?.role;
   670→   670→  const pagePermissions = user?.pagePermissions ?? null;
   671→   671→  const visibleItems = useMemo(() => {
   672→   672→    if (!userRole) return [];
   673→   673→    return getVisibleNavItems(userRole, NAV_ITEMS, pagePermissions);
   674→   674→  }, [userRole, pagePermissions]);
   675→   675→
   676→   676→  /*
   677→   677→   * SINGLE SOURCE OF TRUTH for the expanded top-level section.
   678→   678→   *
   679→   679→   * manualOverride is a tri-state:
   680→   680→   *   null  → not overridden; use route-derived value
   681→   681→   *   ""   → explicitly closed by user
   682→   682→   *   "SEO" → explicitly opened by user
   683→   683→   *
   684→   684→   * When the module changes (navigation event), we reset manualOverride
   685→   685→   * to null so the route-derived value takes over.
   686→   686→   */
   687→   687→  const [manualOverride, setManualOverride] = useState<string | null>(null);
   688→   688→  // Track previous module to detect navigation changes synchronously during render.
   689→   689→  // This is the React-approved pattern for "derived state from props".
   690→   690→  const [prevModule, setPrevModule] = useState(currentModule);
   691→   691→  if (currentModule !== prevModule) {
   692→   692→    setPrevModule(currentModule);
   693→   693→    setManualOverride(null);
   694→   694→  }
   695→   695→
   696→   696→  // Derive the section from the current route hash (EXPLICIT prefix matching)
   697→   697→  const routeDerivedSection = useMemo(() => {
   698→   698→    return getSectionFromHash(window.location.hash);
   699→   699→  }, [currentModule, currentSubPage]);
   700→   700→
   701→   701→  // Compute the actual open section
   702→   702→  const openSection = useMemo(() => {
   703→   703→    if (manualOverride !== null) {
   704→   704→      // User manually toggled: empty string = closed, non-empty = that section
   705→   705→      return manualOverride || null;
   706→   706→    }
   707→   707→    return routeDerivedSection;
   708→   708→  }, [manualOverride, routeDerivedSection]);
   709→   709→
   710→   710→  // Toggle a section open/closed
   711→   711→  const handleToggleSection = useCallback(
   712→   712→    (label: string) => {
   713→   713→      setManualOverride((prev) => {
   714→   714→        // Determine what's currently open (considering route-derived if no override)
   715→   715→        const currentlyOpen =
   716→   716→          prev !== null ? (prev || null) : routeDerivedSection;
   717→   717→        // If clicking the same section that's open, close it
   718→   718→        if (currentlyOpen === label) {
   719→   719→          return ''; // explicitly closed
   720→   720→        }
   721→   721→        // Otherwise, open the clicked section (closing any other)
   722→   722→        return label;
   723→   723→      });
   724→   724→    },
   725→   725→    [routeDerivedSection],
   726→   726→  );
   727→   727→
   728→   728→  if (!user) return null;
   729→   729→
   730→   730→  const groups = buildNavGroups(visibleItems);
   731→   731→
   732→   732→  return (
   733→   733→    <Sidebar collapsible="icon">
   734→   734→      {/* ---- Header: one 32px icon cell per row, all centered at x=24px ---- */}
   735→   735→      <SidebarHeader className="px-2 py-3 shrink-0">
   736→   736→        {/* Expanded: [logo][title][collapse control] single row */}
   737→   737→        <div className="flex h-8 items-center gap-2 group-data-[collapsible=icon]:hidden">
   738→   738→          <LogoMark />
   739→   739→          <span className="truncate font-semibold text-sm tracking-tight whitespace-nowrap text-text-primary">
   740→   740→            CMS Admin
   741→   741→          </span>
   742→   742→          <div className="ml-auto shrink-0">
   743→   743→            <CollapseToggle side="bottom" />
   744→   744→          </div>
   745→   745→        </div>
   746→   746→        {/* Collapsed rail: ONLY the "C" logo. The previously stacked
   747→   747→            collapse-toggle icon was removed on purpose — no extra icon may
   748→   748→            appear under/next to the logo.
   749→   749→            Expand affordance: the logo cell itself toggles the sidebar
   750→   750→            (same 32px grid cell, same x=24 center-line), and the invisible
   751→   751→            SidebarRail edge strip keeps its native toggle behavior too. */}
   752→   752→        <div className="hidden flex-col items-center group-data-[collapsible=icon]:flex">
   753→   753→          <CollapsedLogoButton />
   754→   754→        </div>
   755→   755→      </SidebarHeader>
   756→   756→
   757→   757→      <SidebarSeparator className="mx-0" />
   758→   758→
   759→   759→      <SidebarContent className="sidebar-thin-scroll">
   760→   760→        {groups.map((group) => (
   761→   761→          <NavGroupSection
   762→   762→            key={group.label}
   763→   763→            group={group}
   764→   764→            currentModule={currentModule}
   765→   765→            currentSubPage={currentSubPage}
   766→   766→            openSection={openSection}
   767→   767→            onToggleSection={handleToggleSection}
   768→   768→          />
   769→   769→        ))}
   770→   770→      </SidebarContent>
   771→   771→
   772→   772→      {/* Hide the bottom separator while collapsed — with the footer
   773→   773→          intentionally empty in that state there is nothing to separate,
   774→   774→          so the rail bottom stays clean (no orphan hairline). */}
   775→   775→      <SidebarSeparator className="mx-0 group-data-[collapsible=icon]:hidden" />
   776→   776→
   777→   777→      <SidebarFooter className="shrink-0 group-data-[collapsible=icon]:p-0">
