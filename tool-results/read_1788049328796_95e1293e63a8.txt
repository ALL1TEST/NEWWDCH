     1→// ============================================================
     2→// PLATFORM DATA SERVICE — Centralized SaaS platform dataset
     3→// ============================================================
     4→// This is the SINGLE source of truth for the entire SaaS platform.
     5→// Both the Client Billing experience AND the Platform Admin
     6→// Dashboard read from and mutate this same in-memory dataset, so
     7→// a plan change, cancellation, or payment made on the client side
     8→// is immediately reflected on the admin side, and vice-versa.
     9→//
    10→// Consistency rules (enforced by derivation, never hardcoded):
    11→//  - totalCustomers  = customers.length (non-deleted)
    12→//  - activeSubscriptions = subscriptions where status === 'active'
    13→//  - mrr = sum of normalized monthly price over active PAID subs
    14→//  - totalSites = sites.length
    15→//  - planDistribution = subscriptions grouped by plan (all statuses)
    16→//  - statusCounts = subscriptions grouped by status
    17→//  - recentCustomers = customers sorted by createdAt desc
    18→//  - recentPayments = payments sorted by date desc
    19→//  - usage = aggregated from sites/content/media/AI records
    20→//  - alerts = derived from failed payments + past-due + storage
    21→//
    22→// Mock data is deterministic (no Math.random) so numbers are stable
    23→// across renders and never contradict each other.
    24→// ============================================================

   150→// -------------------- Plans (delegated to DB-backed plan-config) --------------------
   151→// The editable plan definitions live in the PlanConfig table and are cached
   152→// by ./plan-config — the SINGLE source of truth. The owner edits there via
   153→// the Platform Admin; this module (admin overview + client billing) and the
   154→// client billing API both read the same cached values, so an owner price
   155→// change propagates to the client billing page and to MRR on the next read.
   156→// The PLANS array below is a live view re-spliced whenever the config changes.
   157→
   158→import { getPlanConfigsSync, getPlanConfigSync, subscribe as subscribePlanConfig, type PlanConfigData } from './plan-config';
   159→
   160→function toPlan(d: PlanConfigData): Plan {
   161→  return {
   162→    id: d.planId as PlanId,
   163→    name: d.name,
   164→    price: d.priceMonthly,
   165→    priceMonthly: d.priceMonthly,
   166→    priceYearly: d.priceYearly,
   167→    currency: d.currency,
   168→    interval: d.interval,
   169→    isFree: d.isFree,
   170→    active: d.active,
   171→    features: d.features,
   172→    entitlements: d.entitlements,
   173→    limits: d.limits,
   174→  };
   175→}
   176→
   177→/** Live view of the DB-backed plan configs. Mutated in place on every
   178→ *  config refresh so existing references stay valid. */
   179→export const PLANS: Plan[] = [];
   180→
   181→function refreshPLANS(): void {
   182→  const data = getPlanConfigsSync();
   183→  PLANS.length = 0;
   184→  for (const d of data) PLANS.push(toPlan(d));
   185→}
   186→refreshPLANS();
   187→subscribePlanConfig(refreshPLANS);
   188→
   189→export function getPlan(id: PlanId): Plan {
   190→  return PLANS.find((p) => p.id === id) ?? PLANS[0];
   191→}
   192→
   193→/** Normalized monthly price. Yearly subscriptions contribute priceYearly/12
   194→ *  so MRR is a true monthly run-rate. */
   195→export function monthlyPrice(plan: Plan, interval: BillingInterval): number {
   196→  if (interval === 'yearly') return Math.round(plan.priceYearly / 12);
   197→  return plan.priceMonthly;
   198→}

   205→function isoDaysAgo(days: number): string {
   206→  const d = new Date();
   207→  d.setDate(d.getDate() - days);
   208→  d.setHours(9, 15, 0, 0);
   209→  return d.toISOString();
   210→}
   211→function isoDaysAhead(days: number): string {
   212→  const d = new Date();
   213→  d.setDate(d.getDate() + days);
   214→  d.setHours(9, 15, 0, 0);
   215→  return d.toISOString();
   216→}

   390→function buildRevenueSeries(currentMrr: number): RevenuePoint[] {
   391→  const trend = [0.41, 0.47, 0.54, 0.62, 0.71, 0.80, 0.89, 1.0];
   392→  const out: RevenuePoint[] = [];
   393→  const now = new Date();
   394→  for (let i = 7; i >= 0; i--) {
   395→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   396→    const factor = trend[7 - i];
   397→    out.push({ month: MONTHS[d.getMonth()], revenue: Math.round(currentMrr * factor) });
   398→  }
   399→  return out;
   400→}

   406→interface PlatformStore {
   407→  customers: Customer[];
   408→  sites: PlatformSite[];
   409→  payments: Payment[];
   410→  audit: AuditEntry[];
   411→}
   412→
   413→let _store: PlatformStore | null = null;
   414→let _auditCounter = 0;
   415→let _paymentCounter = 0;
   416→
   417→function createStore(): PlatformStore {
   418→  const customers: Customer[] = CUSTOMER_SEED.map((c, i) => ({
   419→    ...c,
   420→    id: `cus_${String(i + 1).padStart(3, '0')}`,
   421→  }));
   422→
   423→  const sites: PlatformSite[] = SITE_SEED.map((s, i) => ({
   424→    id: `site_p${String(i + 1).padStart(3, '0')}`,
   425→    customerId: customers[s.customerIdIdx].id,
   426→    name: s.name,
   427→    slug: s.slug,
   428→    domain: s.domain,
   429→    status: s.status,
   430→    articles: s.articles,
   431→    media: s.media,
   432→    storageBytes: s.storageBytes,
   433→    createdAt: isoDaysAgo(s.daysAgo),
   434→  }));
   435→
   436→  const payments: Payment[] = PAYMENT_SEED.map((p, i) => {
   437→    const c = customers[p.customerIdx];
   438→    return {
   439→      id: `pay_${String(i + 1).padStart(3, '0')}`,
   440→      customerId: c.id,
   441→      planId: p.planId,
   442→      amount: p.amount,
   443→      currency: 'CHF',
   444→      status: p.status,
   445→      method: p.method,
   446→      date: isoDaysAgo(p.daysAgo),
   447→      invoiceNumber: `INV-2026-${String(1000 + i)}`,
   448→    };
   449→  });
   450→
   451→  const audit: AuditEntry[] = INITIAL_AUDIT.map((a, i) => ({
   452→    ...a,
   453→    id: `aud_${String(i + 1).padStart(3, '0')}`,
   454→  }));
   455→
   456→  _auditCounter = audit.length;
   457→  _paymentCounter = payments.length;
   458→
   459→  return { customers, sites, payments, audit };
   460→}

   492→/** Resolve the storageBytes limit for a plan id from the live plan
   493→ *  config cache. Returns Number.MAX_SAFE_INTEGER for unlimited (-1)
   494→ *  so the UI's progress-bar math never divides by zero. */
   495→function storageLimitBytesForPlan(planId: string): number {
   496→  const cfg = getPlanConfigSync(planId);
   497→  const v = cfg?.limits?.storageBytes ?? 0;
   498→  return v === -1 ? Number.MAX_SAFE_INTEGER : v;
   499→}

   501→/** Map a real DB User (+ their Subscription row, when present) to the
   502→ *  public Customer interface the Platform Admin UI consumes. The shape
   503→ *  matches the legacy mock Customer exactly so the UI is unchanged —
   504→ *  only the data source switched from the in-memory seed to the DB.
   505→ *  Company / country default to placeholders since the User table has
   506→ *  no such columns (kept on the legacy mock customer only). */
   507→function mapUserToCustomer(
   508→  user: {
   509→    id: string;
   510→    name: string | null;
   511→    email: string;
   512→    status: string;
   513→    billingMode: string;
   514→    createdAt: Date;
   515→  },
   516→  sub: {
   517→    planId: string;
   518→    status: string;
   519→    billingInterval: string;
   520→    startDate: Date;
   521→    currentPeriodEnd: Date | null;
   522→    trialEnd: Date | null;
   523→  } | null,
   524→): Customer {
   525→  const planId = (sub && ['free', 'plus', 'pro', 'max'].includes(sub.planId) ? sub.planId : 'free') as PlanId;
   526→  const subscriptionStatus = (sub && ['active', 'trial', 'past_due', 'cancelled', 'expired'].includes(sub.status)
   527→    ? sub.status
   528→    : 'active') as SubscriptionStatus;
   529→  const billingInterval: BillingInterval = sub && sub.billingInterval === 'yearly' ? 'yearly' : 'monthly';
   530→  const userStatus: CustomerStatus =
   531→    user.status === 'SUSPENDED' ? 'SUSPENDED' : user.status === 'DEACTIVATED' ? 'DEACTIVATED' : 'ACTIVE';
   532→  return {
   533→    id: user.id,
   534→    name: user.name ?? user.email,
   535→    email: user.email,
   536→    status: userStatus,
   537→    planId,
   538→    subscriptionStatus,
   539→    billingInterval,
   540→    createdAt: user.createdAt.toISOString(),
   541→    subscriptionStart: sub ? sub.startDate.toISOString() : user.createdAt.toISOString(),
   542→    nextBillingAt: sub?.currentPeriodEnd?.toISOString() ?? null,
   543→    trialEnd: sub?.trialEnd?.toISOString() ?? null,
   544→    company: null,
   545→    country: '—',
   546→    storageLimitBytes: storageLimitBytesForPlan(planId),
   547→  };
   548→}

   552→  activeSubscriptions: number;

   566→// REAL DB-backed (replaces the in-memory store() derivation). Reads
   567→//  the live User + Subscription + Payment tables so the Platform Admin
   568→//  Dashboard overview always agrees with the Customers / Subscriptions /
   569→//  Payments pages — same rows, same counts, same MRR. The legacy mock
   570→//  store() is now ONLY used by the sync `getPlatformEvents()` (which
   571→//  derives the platform notifications feed from mock data) and by the
   572→//  sync `getClientBilling()` fallback path (legacy demo customers).
   573→export async function getOverview(): Promise<PlatformOverview> {
   574→  const { db } = await import('@/lib/db');
   575→
   576→  // Customers = EXTERNAL users WITH a subscription row (paying customers).
   577→  const customerUsers = await db.user.findMany({
   578→    where: {
   579→      billingMode: 'EXTERNAL',
   580→      deletedAt: null,
   581→      subscription: { isNot: null },
   582→    },
   583→    include: { subscription: true },
   584→  });
   585→  const totalCustomers = customerUsers.length;
   586→
   587→  // All subs — used for plan distribution + status counts + MRR.
   588→  const allSubs = await db.subscription.findMany();
   589→  const activeSubs = allSubs.filter((s) => s.status === 'active');
   590→  const activeSubscriptions = activeSubs.length;
   591→
   592→  let mrr = 0;
   593→  for (const s of activeSubs) {
   594→    if (s.planId !== 'free') {
   595→      mrr += monthlyPrice(
   596→        getPlan(s.planId as PlanId),
   597→        s.billingInterval === 'yearly' ? 'yearly' : 'monthly',
   598→      );
   599→    }
   600→  }
   601→
   602→  // No real Sites-per-customer table populated; keep 0 (UI handles).
   603→  const totalSites = 0;
   604→
   605→  const planDistribution = PLANS.map((p) => ({
   606→    planId: p.id,
   607→    planName: p.name,
   608→    count: allSubs.filter((s) => s.planId === p.id).length,
   609→    color: p.id === 'free' ? '#10b981' : p.id === 'plus' ? '#f59e0b' : p.id === 'pro' ? '#8b5cf6' : '#ec4899',
   610→  }));
   611→
   612→  const statuses: SubscriptionStatus[] = ['active', 'trial', 'past_due', 'cancelled', 'expired'];
   613→  const statusCounts = statuses
   614→    .map((status) => ({ status, count: allSubs.filter((s) => s.status === status).length }))
   615→    .filter((x) => x.count > 0);
   616→
   617→  const recentCustomers = [...customerUsers]
   618→    .sort((a, b) => b.createdAt.getTime() - a.createdAt.getTime())
   619→    .slice(0, 6)
   620→    .map((u) => ({ ...mapUserToCustomer(u, u.subscription), siteCount: 0 }));
   621→
   622→  // Real recent payments — already DB-backed (Task 75).
   623→  const recentPayments = await getRecentPaymentsReal(6);
   624→
   625→  const usage: PlatformUsage = {
   626→    totalSites,
   627→    totalArticles: 0,
   628→    aiArticlesGenerated: 0,
   629→    aiWordsGenerated: 0,
   630→    mediaStorageBytes: 0,
   631→    automationRuns: 1284, // mock — no real automation-runs table populated
   632→  };
   633→
   634→  // Alerts — derived from the real DB state (failed payments, past-due
   635→  // subs, new customers this week). The legacy storage-limit alert is
   636→  // skipped because no real storage tracking exists yet.
   637→  const alerts: PlatformAlert[] = [];
   638→  const failedCount = await db.payment.count({ where: { status: 'failed' } });
   639→  if (failedCount > 0) {
   640→    alerts.push({
   641→      id: 'alert-failed-payments',
   642→      severity: 'critical',
   643→      title: 'Failed payments',
   644→      message: `${failedCount} payment${failedCount === 1 ? '' : 's'} failed and need attention`,
   645→      action: { label: 'View Payments', module: 'platform-payments' },
   646→      time: 'recent',
   647→    });
   648→  }
   649→  const pastDueCount = await db.subscription.count({ where: { status: 'past_due' } });
   650→  if (pastDueCount > 0) {
   651→    alerts.push({
   652→      id: 'alert-past-due',
   653→      severity: 'warning',
   654→      title: 'Past-due subscriptions',
   655→      message: `${pastDueCount} subscription${pastDueCount === 1 ? '' : 's'} past due`,
   656→      action: { label: 'View Customers', module: 'platform-customers' },
   657→      time: 'recent',
   658→    });
   659→  }
   660→  const weekAgo = new Date(Date.now() - 7 * 24 * 3600_000);
   661→  const newCustomersCount = await db.user.count({
   662→    where: { billingMode: 'EXTERNAL', createdAt: { gt: weekAgo } },
   663→  });
   664→  if (newCustomersCount > 0) {
   665→    alerts.push({
   666→      id: 'alert-new-customers',
   667→      severity: 'info',
   668→      title: 'New customers',
   669→      message: `${newCustomersCount} new customer${newCustomersCount === 1 ? '' : 's'} registered this week`,
   670→      action: { label: 'View Customers', module: 'platform-customers' },
   671→      time: 'recent',
   672→    });
   673→  }
   674→
   675→  return {
   676→    totalCustomers,
   677→    activeSubscriptions,
   678→    mrr,
   679→    currency: 'CHF',
   680→    totalSites,
   681→    planDistribution,
   682→    statusCounts,
   683→    revenueSeries: buildRevenueSeries(mrr),
   684→    recentCustomers,
   685→    recentPayments,
   686→    usage,
   687→    alerts,
   688→    // Placeholder — the overview API route overlays the real
   689→    // (live) health summary here before returning, so that this
   690→    // module does not pull server-only deps (fs, db) into the
   691→    // client bundle via platform-settings.tsx.
   692→    systemHealth: [],
   693→  };

   984→/** Normalize a Prisma Payment row (with user included) to the public
   985→ *  Payment interface the admin UI consumes. The shape is identical to
   986→ *  the legacy mock Payment so the UI is unchanged; only the data source
   987→ *  switched to the real DB. The `user` field is the resolved customer
   988→ *  ({ id, name, email }) — passed in by listPayments / getRecentPayments-
   989→ *  Real via Prisma's `include: { user: { select: ... } }`, and
   990→ *  synthesized in `getCustomer` for the customer-detail Recent Payments
   991→ *  list. Returns the enriched shape (Payment & { customerName,
   992→ *  customerEmail }); CustomerDetail strips the extras when composing
   993→ *  `payments: Payment[]`. */
   994→function mapPaymentRowForCustomer(r: {
   995→  id: string;
   996→  userId: string;
   997→  planId: string;
   998→  amount: number;
   999→  currency: string;
  1000→  status: string;
  1001→  method: string | null;
  1002→  invoiceNumber: string | null;
  1003→  stripeInvoiceId: string | null;
  1004→  stripePaymentIntentId: string | null;
  1005→  stripeChargeId: string | null;
  1006→  paymentMethodType: string | null;
  1007→  paymentMethodDetails: string | null;
  1008→  paidAt: Date | null;
  1009→  createdAt: Date;
  1010→  user?: { id: string; name: string | null; email: string } | null;
  1011→}): Payment & { customerName: string; customerEmail: string } {
  1012→  const planId = (['free', 'plus', 'pro', 'max'].includes(r.planId) ? r.planId : 'free') as PlanId;
  1013→  const status = (['paid', 'pending', 'failed', 'refunded'].includes(r.status)
  1014→    ? r.status
  1015→    : 'pending') as PaymentStatus;
  1016→  return {
  1017→    id: r.id,
  1018→    customerId: r.userId,
  1019→    planId,
  1020→    amount: r.amount,
  1021→    currency: r.currency,
  1022→    status,
  1023→    method: r.method ?? deriveMethodLabel(r.paymentMethodType, r.paymentMethodDetails) ?? '—',
  1024→    date: (r.paidAt ?? r.createdAt).toISOString(),
  1025→    invoiceNumber: r.invoiceNumber ?? r.stripeInvoiceId ?? '—',
  1026→    // Stripe transaction details — surfaced in the admin Payments table
  1027→    // so every row is verifiably tied to a real Stripe PaymentIntent /
  1028→    // Charge / Invoice. Null/undefined when Stripe isn't configured.
  1029→    stripeInvoiceId: r.stripeInvoiceId,
  1030→    stripePaymentIntentId: r.stripePaymentIntentId,
  1031→    stripeChargeId: r.stripeChargeId,
  1032→    customerName: r.user?.name ?? r.user?.email ?? '—',
  1033→    customerEmail: r.user?.email ?? '—',
  1034→  };
  1035→}

  1037→/** Build a human-readable payment-method label from the stored metadata,
  1038→ *  e.g. "Visa ••4242". Falls back to the capitalized type. */
  1039→function deriveMethodLabel(type: string | null, detailsJson: string | null): string | null {
  1040→  if (!type) return null;
  1041→  if (type === 'card' && detailsJson) {
  1042→    try {
  1043→      const d = JSON.parse(detailsJson) as { brand?: string; last4?: string };
  1044→      if (d.brand && d.last4) return `${capitalize(d.brand)} ••${d.last4}`;
  1045→      if (d.brand) return capitalize(d.brand);
  1046→    } catch {
  1047→      // ignore malformed JSON
  1048→    }
  1049→  }
  1050→  return capitalize(type);
  1051→}
  1052→
  1053→function capitalize(s: string): string {
  1054→  return s.charAt(0).toUpperCase() + s.slice(1);
  1055→}

  1199→  if (!user || !user.subscription) return null;

  1688→/** Map alert id → a deterministic "recent" timestamp (no Math.random,
  1689→ *  stable across renders for a given wall-clock minute). */
  1690→function alertTimestamp(alertId: string): string {
  1691→  switch (alertId) {
  1692→    case 'alert-failed-payments': return isoMinutesAgo(30);
  1693→    case 'alert-past-due': return isoMinutesAgo(120);
  1694→    case 'alert-storage': return isoMinutesAgo(360);
  1695→    case 'alert-new-customers': return isoMinutesAgo(60 * 24);
  1696→    default: return isoMinutesAgo(60);
  1697→  }
  1698→}

  1700→/** Derive the unified platform events feed. Reads from the SYNC mock-store
  1701→ *  helpers (listCustomersMockSync / listPaymentsMockSync / listSubscriptionsMockSync)
  1702→ *  + getAuditLog + getAlerts. The public `listCustomers` / `listPayments` /
  1703→ *  `listSubscriptions` are now async + DB-backed (Task 78-D), but this
  1704→ *  function stays SYNC because /api/platform/admin/notifications calls it
  1705→ *  synchronously and isn't in this task's file list. The mock store is the
  1706→ *  deterministic seed (no Math.random) so the derived events are stable
  1707→ *  across renders. */
  1708→export function getPlatformEvents(): PlatformEvent[] {
  1709→  const events: PlatformEvent[] = [];
  1710→
  1711→  // 1. Customer registrations — INFO "New customer registered"
  1712→  for (const c of listCustomersMockSync()) {
  1713→    events.push({
  1714→      id: `evt-cust-${c.id}`,
  1715→      type: 'INFO',
  1716→      title: 'New customer registered',
  1717→      message: `${c.name} (${c.email}) signed up from ${c.country} on the ${c.planId.toUpperCase()} plan.`,
  1718→      isRead: false,
  1719→      createdAt: c.createdAt,
  1720→    });
  1721→  }
  1722→
  1723→  // 2. Payments — SUCCESS / ERROR / WARNING / INFO by status
  1724→  for (const p of listPaymentsMockSync()) {
  1725→    let type: PlatformEventType = 'INFO';
  1726→    let title = 'Payment update';
  1727→    if (p.status === 'paid') {
  1728→      type = 'SUCCESS';
  1729→      title = 'Successful payment';
  1730→    } else if (p.status === 'failed') {
  1731→      type = 'ERROR';
  1732→      title = 'Failed payment';
  1733→    } else if (p.status === 'refunded') {
  1734→      type = 'WARNING';
  1735→      title = 'Payment refunded';
  1736→    } else if (p.status === 'pending') {
  1737→      type = 'INFO';
  1738→      title = 'Payment pending';
  1739→    }
  1740→    events.push({
  1741→      id: `evt-pay-${p.id}`,
  1742→      type,
  1743→      title,
  1744→      message: `${p.customerName} — ${p.currency} ${p.amount} via ${p.method}. Invoice ${p.invoiceNumber}. Status: ${p.status}.`,
  1745→      isRead: false,
  1746→      createdAt: p.date,
  1747→    });
  1748→  }
  1749→
  1750→  // 3. Subscriptions — INFO created, WARNING cancelled, ACTION_REQUIRED trial ending
  1751→  for (const s of listSubscriptionsMockSync()) {
  1752→    const plan = getPlan(s.planId);
  1753→    events.push({
  1754→      id: `evt-sub-${s.id}`,
  1755→      type: 'INFO',
  1756→      title: 'Subscription created',
  1757→      message: `${s.name} subscribed to the ${s.planName} plan (${s.billingInterval} billing, ${plan.currency} ${s.monthlyPrice}/mo).`,
  1758→      isRead: false,
  1759→      createdAt: s.subscriptionStart,
  1760→    });
  1761→
  1762→    if (s.subscriptionStatus === 'cancelled') {
  1763→      // Cancellation happened at some point after creation — there is no
  1764→      // dedicated timestamp in the dataset, so use subscriptionStart as a
  1765→      // stable proxy (still sorts in the past, like a historical event).
  1766→      events.push({
  1767→        id: `evt-sub-cancel-${s.id}`,
  1768→        type: 'WARNING',
  1769→        title: 'Subscription cancelled',
  1770→        message: `${s.name} cancelled the ${s.planName} plan subscription. Access has been revoked at the end of the current period.`,
  1771→        isRead: false,
  1772→        createdAt: s.subscriptionStart,
  1773→      });
  1774→    }
  1775→
  1776→    if (s.subscriptionStatus === 'trial') {
  1777→      const trialEnd = s.trialEnd ?? s.nextBillingAt ?? s.subscriptionStart;
  1778→      events.push({
  1779→        id: `evt-sub-trial-${s.id}`,
  1780→        type: 'ACTION_REQUIRED',
  1781→        title: 'Trial ending soon',
  1782→        message: `${s.name}'s ${s.planName} plan trial ends on ${trialEnd}. Convert to a paid subscription before then to retain access.`,
  1783→        isRead: false,
  1784→        createdAt: trialEnd,
  1785→      });
  1786→    }
  1787→  }
  1788→
  1789→  // 4. Audit log entries — INFO with action as title, detail as message
  1790→  for (const a of getAuditLog(50)) {
  1791→    events.push({
  1792→      id: `evt-aud-${a.id}`,
  1793→      type: 'INFO',
  1794→      title: a.action,
  1795→      message: a.detail && a.detail.length > 0 ? `${a.detail} — target: ${a.target}` : a.target,
  1796→      isRead: false,
  1797→      createdAt: a.timestamp,
  1798→    });
  1799→  }
  1800→
  1801→  // 5. Alerts — ERROR / WARNING / INFO by severity
  1802→  for (const alert of getAlerts()) {
  1803→    let type: PlatformEventType = 'INFO';
  1804→    if (alert.severity === 'critical') type = 'ERROR';
  1805→    else if (alert.severity === 'warning') type = 'WARNING';
  1806→    events.push({
  1807→      id: `evt-alert-${alert.id}`,
  1808→      type,
  1809→      title: alert.title,
  1810→      message: alert.message,
  1811→      isRead: false,
  1812→      createdAt: alertTimestamp(alert.id),
  1813→    });
  1814→  }
  1815→
  1816→  // Newest first — consistent with the Client Notifications feed.
  1817→  return events.sort(
  1818→    (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime(),
  1819→  );
  1820→}
