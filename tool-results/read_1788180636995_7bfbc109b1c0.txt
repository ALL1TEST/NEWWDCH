     1→// ============================================================
     2→// PLAN CONFIG — DB-backed, cached single source of truth for plans,
     3→// pricing, entitlements and usage limits.
     4→// ============================================================
     5→// The Client Billing API and the Platform Admin both read from (and
     6→// the admin mutates) this same service. `platform-data.ts` delegates
     7→// its `PLANS` / `getPlan` / `monthlyPrice` to the synchronous cache
     8→// exposed here, so an owner edit (savePlanConfig) propagates to the
     9→// client billing experience and to MRR on the next read.
    10→//
    11→// MULTI-CURRENCY: the plan carries ONE base price configuration —
    12→// priceMonthly / priceYearly denominated in the plan's default /
    13→// fallback `currency` (both edited directly in the Edit Plan modal).
    14→// The customer's currency is resolved SERVER-SIDE from their IP
    15→// geolocation (see country-pricing.ts:resolveCustomerPricing) when
    16→// the plan's `autoCurrency` flag is on; otherwise every customer is
    17→// billed in the plan default currency.
    18→//
    19→// pricesByCurrency is a DERIVED store (NOT edited per-plan in the
    20→// modal): the default currency's entry mirrors the base price on
    21→// every save, and the other entries hold the platform-level
    22→// regional prices (CountryPricing.regionalPrices + legacy entries)
    23→// — preserved across saves so customers in other currencies keep
    24→// seeing their configured price (e.g. "90 MAD / month").
    25→//
    26→// STRIPE: for each paid plan × supported currency × interval, a real
    27→// Stripe Price is created on sync and its ID persisted in
    28→// stripePriceIdsByCurrency. The default currency's IDs are mirrored
    29→// in stripePriceIdMonthly/Yearly for legacy callers. Checkout always
    30→// resolves the per-currency Stripe Price ID server-side — the
    31→// customer's currency determines which Price is charged.
    32→//
    33→// The cache is hydrated from the PlanConfig table on startup and
    34→// after every mutation. If the table is empty on first hydrate, the
    35→// DEFAULT configs are seeded.
    36→// ============================================================

   271→function notify() {
   272→  for (const cb of _subscribers) {
   273→    try {
   274→      cb();
   275→    } catch {
   276→      // ignore subscriber errors
   277→    }
   278→  }

   281→/** Register a callback fired after every cache refresh (hydrate / mutation). */
   282→export function subscribe(cb: () => void): () => void {
   283→  _subscribers.add(cb);
   284→  return () => _subscribers.delete(cb);
   285→}

   341→function rowToData(row: PlanConfigRow): PlanConfigData {
   342→  let features: string[] = [];
   343→  let entitlements: string[] = [];
   344→  try {
   345→    features = JSON.parse(row.features || '[]');
   346→  } catch {
   347→    features = [];
   348→  }
   349→  try {
   350→    entitlements = JSON.parse(row.entitlements || '[]');
   351→  } catch {
   352→    entitlements = [];
   353→  }
   354→  // Explicitly pick only the known limit fields. Older DB rows may still
   355→  // carry stale `aiWords` / `aiArticles` / `automationRuns` keys in the
   356→  // limits JSON (those were removed because no real enforcement exists).
   357→  // Picking only the known fields prevents the stale keys from leaking
   358→  // back into the cache and being re-serialized on the next save.
   359→  let limits: PlanLimits = { maxSites: 0, storageBytes: 0 };
   360→  try {
   361→    const parsed = JSON.parse(row.limits || '{}') as Partial<PlanLimits>;
   362→    limits = {
   363→      maxSites: typeof parsed.maxSites === 'number' ? parsed.maxSites : 0,
   364→      storageBytes: typeof parsed.storageBytes === 'number' ? parsed.storageBytes : 0,
   365→    };
   366→  } catch {
   367→    // keep defaults
   368→  }
   369→  let pricesByCurrency: PricesByCurrency = {};
   370→  try {
   371→    const parsed = JSON.parse(row.pricesByCurrency || '{}') as Partial<PricesByCurrency>;
   372→    if (parsed && typeof parsed === 'object') {
   373→      for (const [cur, val] of Object.entries(parsed)) {
   374→        if (val && typeof val === 'object') {
   375→          const m = typeof (val as CurrencyPrice).monthly === 'number' ? (val as CurrencyPrice).monthly : 0;
   376→          const y = typeof (val as CurrencyPrice).yearly === 'number' ? (val as CurrencyPrice).yearly : 0;
   377→          pricesByCurrency[cur.toUpperCase()] = { monthly: m, yearly: y };
   378→        }
   379→      }
   380→    }
   381→  } catch {
   382→    pricesByCurrency = {};
   383→  }
   384→  let stripePriceIdsByCurrency: StripePriceIdsByCurrency = {};
   385→  try {
   386→    const parsed = JSON.parse(row.stripePriceIdsByCurrency || '{}') as Partial<StripePriceIdsByCurrency>;
   387→    if (parsed && typeof parsed === 'object') {
   388→      for (const [cur, val] of Object.entries(parsed)) {
   389→        if (val && typeof val === 'object') {
   390→          const m = typeof (val as CurrencyStripeIds).monthly === 'string' ? (val as CurrencyStripeIds).monthly : null;
   391→          const y = typeof (val as CurrencyStripeIds).yearly === 'string' ? (val as CurrencyStripeIds).yearly : null;
   392→          stripePriceIdsByCurrency[cur.toUpperCase()] = { monthly: m, yearly: y };
   393→        }
   394→      }
   395→    }
   396→  } catch {
   397→    stripePriceIdsByCurrency = {};
   398→  }
   399→  return {
   400→    planId: row.planId,
   401→    name: row.name,
   402→    priceMonthly: row.priceMonthly,
   403→    priceYearly: row.priceYearly,
   404→    currency: row.currency,
   405→    autoCurrency: row.autoCurrency ?? true,
   406→    pricesByCurrency,
   407→    stripePriceIdsByCurrency,
   408→    interval: (row.interval === 'yearly' ? 'yearly' : 'monthly') as BillingInterval,
   409→    isFree: row.isFree,
   410→    freePlanDurationDays: row.freePlanDurationDays,
   411→    stripePriceIdMonthly: row.stripePriceIdMonthly,
   412→    stripePriceIdYearly: row.stripePriceIdYearly,
   413→    active: row.active,
   414→    features,
   415→    entitlements,
   416→    limits,
   417→    badgeVariant: row.badgeVariant,
   418→    sortOrder: row.sortOrder,
   419→  };
   420→}

   446→/**
   447→ * Migrate the legacy catalog to the new Free/Plus/Pro/Max shape. Idempotent.
   448→ *
   449→ * Legacy catalog (created by older bootstrap):
   450→ *   beta (free, active=false), pro, max, enterprise (stub)
   451→ *
   452→ * Migration steps (each guarded, can be re-run safely):
   453→ *  1. If `enterprise` exists AND no active subscription references it → delete it.
   454→ *     (Active subs that reference enterprise are preserved — the migration will
   455→ *      not break them; the plan row stays if needed. In practice, the demo
   456→ *      dataset has no DB subscriptions yet, so this is a clean delete.)
   457→ *  2. If `beta` exists → rename to `plus` (planId, name, badgeVariant) and
   458→ *     re-enable it (legacy `beta` was active=false in the demo DB). Also
   459→ *     shift sortOrder to 1 (Free takes 0).
   460→ *  3. If `free` does NOT exist → insert the Free plan at sortOrder 0.
   461→ *  4. Renumber sortOrder to match the canonical order: free=0, plus=1, pro=2, max=3.
   462→ *
   463→ * Returns a short report string for the bootstrap log.
   464→ */
   465→export async function migrateLegacyPlans(): Promise<string> {
   466→  const notes: string[] = [];
   467→
   468→  try {
   469→    // Step 1: delete Enterprise if safe
   470→    const enterprise = await db.planConfig.findUnique({ where: { planId: 'enterprise' } });
   471→    if (enterprise) {
   472→      const subCount = await db.subscription.count({ where: { planId: 'enterprise' } });
   473→      if (subCount === 0) {
   474→        await db.planConfig.delete({ where: { planId: 'enterprise' } });
   475→        notes.push('deleted enterprise plan (0 active subscriptions)');
   476→      } else {
   477→        notes.push(`kept enterprise plan (${subCount} active subscriptions reference it)`);
   478→      }
   479→    }
   480→
   481→    // Step 2: rename beta → plus (and refresh to a real paid tier)
   482→    const beta = await db.planConfig.findUnique({ where: { planId: 'beta' } });
   483→    if (beta) {
   484→      // Don't overwrite an existing plus row — if both exist, prefer the
   485→      // plus row and just delete the beta stub.
   486→      const plus = await db.planConfig.findUnique({ where: { planId: 'plus' } });
   487→      if (!plus) {
   488→        // Rename beta → plus AND refresh the data to the real DEFAULT
   489→        // Plus tier (paid plan with real entitlements + limits, not the
   490→        // legacy free-tier Beta stub). The old Beta was active=false and
   491→        // had no real customers, so this is a safe refresh.
   492→        const plusDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'plus')!;
   493→        await db.planConfig.update({
   494→          where: { planId: 'beta' },
   495→          data: {
   496→            ...dataToRow(plusDefault),
   497→            // Keep the original createdAt for continuity.
   498→            // (updatedAt is auto.)
   499→          },
   500→        });
   501→        notes.push('renamed + refreshed beta → plus (real paid tier)');
   502→      } else {
   503→        // Both beta and plus exist — delete the beta stub (migrate any
   504→        // legacy subscriptions on beta to plus first).
   505→        const betaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   506→        if (betaSubs > 0) {
   507→          await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   508→          notes.push(`migrated ${betaSubs} beta subscription(s) → plus`);
   509→        }
   510→        await db.planConfig.delete({ where: { planId: 'beta' } });
   511→        notes.push('deleted legacy beta stub (plus already exists)');
   512→      }
   513→    }
   514→
   515→    // Step 3: insert Free if missing
   516→    const free = await db.planConfig.findUnique({ where: { planId: 'free' } });
   517→    if (!free) {
   518→      const freeDefault = DEFAULT_PLAN_CONFIGS.find((p) => p.planId === 'free')!;
   519→      await db.planConfig.create({ data: dataToRow(freeDefault) });
   520→      notes.push('inserted free plan');
   521→    } else {
   522→      // Ensure free plan is properly flagged
   523→      if (!free.isFree || free.badgeVariant !== 'free' || free.sortOrder !== 0) {
   524→        await db.planConfig.update({
   525→          where: { planId: 'free' },
   526→          data: { isFree: true, badgeVariant: 'free', sortOrder: 0 },
   527→        });
   528→        notes.push('normalized free plan flags');
   529→      }
   530→    }
   531→
   532→    // Step 4: normalize sortOrder for all canonical plans
   533→    const canonicalOrder: Record<string, number> = { free: 0, plus: 1, pro: 2, max: 3 };
   534→    for (const [pid, order] of Object.entries(canonicalOrder)) {
   535→      const row = await db.planConfig.findUnique({ where: { planId: pid } });
   536→      if (row && row.sortOrder !== order) {
   537→        await db.planConfig.update({ where: { planId: pid }, data: { sortOrder: order } });
   538→      }
   539→    }
   540→
   541→    // Also migrate legacy subscriptions on 'beta' or 'enterprise' planId
   542→    // to the new ids (beta → plus, enterprise → max) so historical subs
   543→    // don't reference a missing plan.
   544→    const legacyBetaSubs = await db.subscription.count({ where: { planId: 'beta' } });
   545→    if (legacyBetaSubs > 0) {
   546→      await db.subscription.updateMany({ where: { planId: 'beta' }, data: { planId: 'plus' } });
   547→      notes.push(`migrated ${legacyBetaSubs} legacy beta subscription(s) → plus`);
   548→    }
   549→    const legacyEnterpriseSubs = await db.subscription.count({ where: { planId: 'enterprise' } });
   550→    if (legacyEnterpriseSubs > 0) {
   551→      await db.subscription.updateMany({ where: { planId: 'enterprise' }, data: { planId: 'max' } });
   552→      notes.push(`migrated ${legacyEnterpriseSubs} legacy enterprise subscription(s) → max`);
   553→    }
   554→  } catch (err) {
   555→    notes.push(`migration error (non-fatal): ${(err as Error).message}`);
   556→  }
   557→
   558→  return notes.length > 0 ? notes.join('; ') : 'no migration needed';
   559→}

   561→/** Load all plan configs from the DB into the cache. Self-seeds the
   562→ *  DEFAULT_PLAN_CONFIGS when the table is empty. Runs the legacy
   563→ *  migration when the catalog still contains 'beta' or 'enterprise'.
   564→ *  Safe to call repeatedly. */
   565→export async function hydrate(): Promise<void> {
   566→  if (_hydrating) return _hydrating;
   567→  _hydrating = (async () => {
   568→    try {
   569→      const rows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   570→      if (rows.length === 0) {
   571→        // Self-seed so the platform is never without plans.
   572→        await db.planConfig.createMany({ data: DEFAULT_PLAN_CONFIGS.map(dataToRow) });
   573→        _cache = DEFAULT_PLAN_CONFIGS.map((p) => ({
   574→          ...p,
   575→          features: [...p.features],
   576→          entitlements: [...p.entitlements],
   577→          limits: { ...p.limits },
   578→        }));
   579→      } else {
   580→        // Check for legacy catalog (beta / enterprise) and migrate.
   581→        const hasLegacy =
   582→          rows.some((r) => r.planId === 'beta' || r.planId === 'enterprise') ||
   583→          !rows.some((r) => r.planId === 'free');
   584→        if (hasLegacy) {
   585→          await migrateLegacyPlans();
   586→          const migratedRows = await db.planConfig.findMany({ orderBy: { sortOrder: 'asc' } });
   587→          _cache = migratedRows.map(rowToData);
   588→        } else {
   589→          _cache = rows.map(rowToData);
   590→        }
   591→      }
   592→      _hydrated = true;
   593→      notify();
   594→    } catch (err) {
   595→      // DB unavailable — keep defaults (still functional, just not persisted).
   596→      _hydrated = true;
   597→      notify();
   598→    } finally {
   599→      _hydrating = null;
   600→    }
   601→  })();
   602→  return _hydrating;
   603→}

   607→export interface PlanConfigInput {
   608→  name?: string;
   609→  /** Base monthly price in the plan default currency (modal-edited). */
   610→  priceMonthly?: number;
   611→  /** Base yearly price in the plan default currency (modal-edited). */
   612→  priceYearly?: number;
   613→  /** Plan default / fallback currency (admin-selected). When provided
   614→   *  together with priceMonthly/priceYearly, the default currency's
   615→   *  pricesByCurrency entry is DERIVED from the base price. */
   616→  currency?: string;
   617→  /** Auto-detect the customer's currency from their IP (fallback to
   618→   *  `currency` when the detected currency is unsupported). */
   619→  autoCurrency?: boolean;
   620→  /** Per-currency prices. Rarely sent directly — the modal sends the
   621→   *  base price + currency and the backend derives the default entry. */
   622→  pricesByCurrency?: PricesByCurrency;
   623→  /** Stripe Price ID snapshots for the default currency. */
   624→  stripePriceIdMonthly?: string | null;
   625→  stripePriceIdYearly?: string | null;
   626→  /** Authoritative per-currency Stripe Price IDs: { USD: { monthly, yearly }, ... }. */
   627→  stripePriceIdsByCurrency?: StripePriceIdsByCurrency;
   628→  interval?: BillingInterval;
   629→  isFree?: boolean;
   630→  freePlanDurationDays?: number | null;
   631→  active?: boolean;
   632→  /** Marketing copy. Auto-derived from entitlements when omitted. */
   633→  features?: string[];
   634→  entitlements?: string[];
   635→  limits?: Partial<PlanLimits>;
   636→  badgeVariant?: string;
   637→  sortOrder?: number;

   640→/** Merge a patch (Partial<PlanConfigData>) into the current data.
   641→ *  The Edit Plan modal sends the ONE base price configuration
   642→ *  (priceMonthly + priceYearly + currency + autoCurrency) — NOT a
   643→ *  per-currency matrix. Derivation rules:
   644→ *  - pricesByCurrency: starts from the current map (or the explicit
   645→ *    patch map when provided); whenever the base price / currency is
   646→ *    in the patch, the DEFAULT currency's entry is re-derived as
   647→ *    { monthly: priceMonthly, yearly: priceYearly }. Other currency
   648→ *    entries (platform regional / legacy prices) are PRESERVED.
   649→ *  - stripePriceIdsByCurrency: kept as-is when omitted (the Stripe
   650→ *    sync maintains it); the default-currency snapshots mirror the
   651→ *    default entry.
   652→ */
   653→function mergePlanPatch(current: PlanConfigData, patch: PlanConfigInput, defaultCurrency: string): PlanConfigData {
   654→  const curUpper = defaultCurrency.toUpperCase();
   655→
   656→  // pricesByCurrency — clone (never mutate the cache's object in place).
   657→  const sourcePrices = patch.pricesByCurrency ?? current.pricesByCurrency ?? {};
   658→  const pricesByCurrency: PricesByCurrency = { ...sourcePrices };
   659→
   660→  // Base price precedence:
   661→  //   1. Explicit patch values (the modal always sends these).
   662→  //   2. Legacy mirror: an explicit pricesByCurrency map with an entry
   663→  //      for the final default currency (backward-compat callers).
   664→  //   3. Current snapshot.
   665→  const mirrored =
   666→    patch.priceMonthly === undefined && patch.pricesByCurrency !== undefined
   667→      ? pricesByCurrency[curUpper]
   668→      : undefined;
   669→  const priceMonthly =
   670→    patch.priceMonthly !== undefined ? patch.priceMonthly : mirrored?.monthly ?? current.priceMonthly;
   671→  const priceYearly =
   672→    patch.priceYearly !== undefined ? patch.priceYearly : mirrored?.yearly ?? current.priceYearly;
   673→
   674→  // The base price configuration was touched (or a legacy map was sent)
   675→  // → re-derive the DEFAULT currency's entry from the base price. All
   676→  // OTHER currency entries (platform regional / legacy prices) are
   677→  // preserved untouched.
   678→  const baseTouched =
   679→    patch.priceMonthly !== undefined ||
   680→    patch.priceYearly !== undefined ||
   681→    patch.currency !== undefined ||
   682→    patch.pricesByCurrency !== undefined;
   683→  if (baseTouched) {
   684→    pricesByCurrency[curUpper] = { monthly: priceMonthly, yearly: priceYearly };
   685→  }
   686→
   687→  const stripePriceIdsByCurrency: StripePriceIdsByCurrency = {
   688→    ...(patch.stripePriceIdsByCurrency ?? current.stripePriceIdsByCurrency ?? {}),
   689→  };
   690→  // Default-currency snapshot — keep the legacy fields in sync so legacy
   691→  // callers that read stripePriceIdMonthly/Yearly see the right values.
   692→  const defIds = stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
   693→  const stripePriceIdMonthly =
   694→    patch.stripePriceIdMonthly !== undefined
   695→      ? patch.stripePriceIdMonthly
   696→      : patch.stripePriceIdsByCurrency !== undefined
   697→        ? defIds.monthly
   698→        : current.stripePriceIdMonthly;
   699→  const stripePriceIdYearly =
   700→    patch.stripePriceIdYearly !== undefined
   701→      ? patch.stripePriceIdYearly
   702→      : patch.stripePriceIdsByCurrency !== undefined
   703→        ? defIds.yearly
   704→        : current.stripePriceIdYearly;
   705→  const isFree = patch.isFree ?? current.isFree;
   706→  return {
   707→    planId: current.planId,
   708→    name: patch.name ?? current.name,
   709→    priceMonthly,
   710→    priceYearly,
   711→    currency: curUpper,
   712→    autoCurrency: patch.autoCurrency ?? current.autoCurrency,
   713→    pricesByCurrency,
   714→    stripePriceIdsByCurrency,
   715→    interval: patch.interval ?? current.interval,
   716→    isFree,
   717→    freePlanDurationDays: patch.freePlanDurationDays ?? current.freePlanDurationDays,
   718→    stripePriceIdMonthly,
   719→    stripePriceIdYearly,
   720→    active: patch.active ?? current.active,
   721→    features: patch.features ?? current.features,
   722→    entitlements: patch.entitlements ?? current.entitlements,
   723→    limits: { ...current.limits, ...(patch.limits ?? {}) },
   724→    badgeVariant: patch.badgeVariant ?? current.badgeVariant,
   725→    sortOrder: patch.sortOrder ?? current.sortOrder,
   726→  };
   727→}

   729→/** Update an existing plan config. Writes DB, refreshes cache.
   730→ *
   731→ *  AUTO-SYNC TO STRIPE: when Stripe is configured AND the plan is
   732→ *  PAID (any currency has a positive price) AND the admin did NOT
   733→ *  explicitly set the Stripe Price IDs in this patch, the backend calls
   734→ *  syncPlanToStripe (multi-currency) to ensure a Stripe Product +
   735→ *  per-currency monthly+yearly Prices exist and writes the resolved IDs
   736→ *  into stripePriceIdsByCurrency + the default-currency snapshot fields.
   737→ */
   738→export async function savePlanConfig(planId: string, patch: PlanConfigInput): Promise<PlanConfigData | null> {
   739→  const existing = await db.planConfig.findUnique({ where: { planId } });
   740→  if (!existing) return null;
   741→  const current = rowToData(existing);
   742→
   743→  // When the admin omitted currency, resolve the platform default so the
   744→  // legacy priceMonthly/priceYearly snapshot stays in the right currency.
   745→  if (!patch.currency && !current.currency) {
   746→    patch = { ...patch, currency: await resolvePlatformDefaultCurrency() };
   747→  }
   748→  const finalCurrency = (patch.currency ?? current.currency).toUpperCase();
   749→
   750→  // Distinguish "field omitted" from "field set". When the admin omits
   751→  // both stripePriceId* AND stripePriceIdsByCurrency, they want the
   752→  // backend to auto-sync. When either is explicitly provided, manual.
   753→  const adminTouchedStripePriceIds =
   754→    'stripePriceIdMonthly' in patch ||
   755→    'stripePriceIdYearly' in patch ||
   756→    'stripePriceIdsByCurrency' in patch;
   757→
   758→  const next = mergePlanPatch(current, patch, finalCurrency);
   759→  await db.planConfig.update({ where: { planId }, data: dataToRow(next) });
   760→  await hydrate();
   761→
   762→  // ---- AUTO-SYNC TO STRIPE (multi-currency) ----
   763→  // Best-effort — never throws.
   764→  const hasAnyPaidCurrency = Object.values(next.pricesByCurrency).some(
   765→    (p) => p.monthly > 0 || p.yearly > 0,
   766→  );
   767→  if (!adminTouchedStripePriceIds && !next.isFree && hasAnyPaidCurrency) {
   768→    try {
   769→      const { isStripeConfiguredAsync, getStripeClient, syncPlanToStripeMulti } = await import('@/lib/stripe');
   770→      if (await isStripeConfiguredAsync()) {
   771→        const stripe = await getStripeClient();
   772→        const syncResult = await syncPlanToStripeMulti(stripe, {
   773→          planId: next.planId,
   774→          name: next.name,
   775→          defaultCurrency: finalCurrency,
   776→          pricesByCurrency: next.pricesByCurrency,
   777→          stripePriceIdsByCurrency: next.stripePriceIdsByCurrency,
   778→        });
   779→        // Persist the resolved Stripe Price IDs back onto the row when they changed.
   780→        const snapshot = syncResult.stripePriceIdsByCurrency[finalCurrency] ?? { monthly: null, yearly: null };
   781→        const changed =
   782→          JSON.stringify(syncResult.stripePriceIdsByCurrency) !==
   783→            JSON.stringify(next.stripePriceIdsByCurrency) ||
   784→          snapshot.monthly !== next.stripePriceIdMonthly ||
   785→          snapshot.yearly !== next.stripePriceIdYearly;
   786→        if (changed) {
   787→          await db.planConfig.update({
   788→            where: { planId },
   789→            data: {
   790→              stripePriceIdsByCurrency: JSON.stringify(syncResult.stripePriceIdsByCurrency),
   791→              stripePriceIdMonthly: snapshot.monthly,
   792→              stripePriceIdYearly: snapshot.yearly,
   793→            },
   794→          });
   795→          await hydrate();
   796→        }
   797→      }
   798→    } catch {
   799→      // best-effort — never block the save on a Stripe error
   800→    }
   801→  }
   802→
   803→  return getPlanConfigSync(planId);
   804→}

   806→/** Create a new plan config. AUTO-SYNC TO STRIPE: when Stripe is
   807→ *  configured AND the new plan is PAID AND the admin did NOT
   808→ *  explicitly provide Stripe Price IDs in the input, the backend
   809→ *  calls syncPlanToStripeMulti (multi-currency) to create a Stripe
   810→ *  Product + per-currency monthly+yearly Prices and writes the
   811→ *  resolved IDs back onto the new plan row.
   812→ *
   813→ *  The Create Plan modal sends the ONE base price configuration
   814→ *  (priceMonthly + priceYearly + currency + autoCurrency); the default
   815→ *  currency's pricesByCurrency entry is derived from the base price. */
   816→export async function createPlanConfig(
   817→  input: PlanConfigInput & { planId: string; name: string },
   818→): Promise<PlanConfigData> {
   819→  const maxOrder = await db.planConfig.count();
   820→  const currency =
   821→    input.currency && input.currency.trim().length > 0
   822→      ? input.currency
   823→      : await resolvePlatformDefaultCurrency();
   824→  const curUpper = currency.toUpperCase();
   825→  const pricesByCurrency: PricesByCurrency = { ...(input.pricesByCurrency ?? {}) };
   826→  // Base price precedence: explicit input → legacy mirror from an
   827→  // explicit pricesByCurrency map → 0.
   828→  const mirrored = input.pricesByCurrency ? pricesByCurrency[curUpper] : undefined;
   829→  const priceMonthly = input.priceMonthly ?? mirrored?.monthly ?? 0;
   830→  const priceYearly = input.priceYearly ?? mirrored?.yearly ?? 0;
   831→  // Derive the default currency entry from the base price.
   832→  pricesByCurrency[curUpper] = { monthly: priceMonthly, yearly: priceYearly };
   833→  const stripePriceIdsByCurrency: StripePriceIdsByCurrency = input.stripePriceIdsByCurrency ?? {};
   834→  const defIds = stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
   835→  const data: PlanConfigData = {
   836→    planId: input.planId,
   837→    name: input.name,
   838→    priceMonthly,
   839→    priceYearly,
   840→    currency: curUpper,
   841→    autoCurrency: input.autoCurrency ?? true,
   842→    pricesByCurrency,
   843→    stripePriceIdsByCurrency,
   844→    interval: input.interval ?? 'monthly',
   845→    isFree: input.isFree ?? false,
   846→    freePlanDurationDays: input.freePlanDurationDays ?? null,
   847→    stripePriceIdMonthly: input.stripePriceIdMonthly !== undefined ? input.stripePriceIdMonthly : defIds.monthly,
   848→    stripePriceIdYearly: input.stripePriceIdYearly !== undefined ? input.stripePriceIdYearly : defIds.yearly,
   849→    active: input.active ?? true,
   850→    features: input.features ?? [],
   851→    entitlements: input.entitlements ?? [],
   852→    limits: {
   853→      maxSites: 0,
   854→      storageBytes: 0,
   855→      ...(input.limits ?? {}),
   856→    },
   857→    badgeVariant: input.badgeVariant ?? input.planId,
   858→    sortOrder: input.sortOrder ?? maxOrder,
   859→  };
   860→  await db.planConfig.create({ data: dataToRow(data) });
   861→  await hydrate();
   862→
   863→  // ---- AUTO-SYNC TO STRIPE (multi-currency) ----
   864→  // Best-effort — never throws. The plan row is already created; if
   865→  // the Stripe side fails, the admin can use the explicit "Sync to
   866→  // Stripe" route to surface errors.
   867→  const adminTouchedStripePriceIds =
   868→    'stripePriceIdMonthly' in input ||
   869→    'stripePriceIdYearly' in input ||
   870→    'stripePriceIdsByCurrency' in input;
   871→  const hasAnyPaidCurrency = Object.values(data.pricesByCurrency).some(
   872→    (p) => p.monthly > 0 || p.yearly > 0,
   873→  );
   874→  if (!adminTouchedStripePriceIds && !data.isFree && hasAnyPaidCurrency) {
   875→    try {
   876→      const { isStripeConfiguredAsync, getStripeClient, syncPlanToStripeMulti } = await import('@/lib/stripe');
   877→      if (await isStripeConfiguredAsync()) {
   878→        const stripe = await getStripeClient();
   879→        const syncResult = await syncPlanToStripeMulti(stripe, {
   880→          planId: data.planId,
   881→          name: data.name,
   882→          defaultCurrency: curUpper,
   883→          pricesByCurrency: data.pricesByCurrency,
   884→          stripePriceIdsByCurrency: data.stripePriceIdsByCurrency,
   885→        });
   886→        const snapshot = syncResult.stripePriceIdsByCurrency[curUpper] ?? { monthly: null, yearly: null };
   887→        const changed =
   888→          JSON.stringify(syncResult.stripePriceIdsByCurrency) !==
   889→            JSON.stringify(data.stripePriceIdsByCurrency) ||
   890→          snapshot.monthly !== data.stripePriceIdMonthly ||
   891→          snapshot.yearly !== data.stripePriceIdYearly;
   892→        if (changed) {
   893→          await db.planConfig.update({
   894→            where: { planId: data.planId },
   895→            data: {
   896→              stripePriceIdsByCurrency: JSON.stringify(syncResult.stripePriceIdsByCurrency),
   897→              stripePriceIdMonthly: snapshot.monthly,
   898→              stripePriceIdYearly: snapshot.yearly,
   899→            },
   900→          });
   901→          await hydrate();
   902→          data.stripePriceIdsByCurrency = syncResult.stripePriceIdsByCurrency;
   903→          data.stripePriceIdMonthly = snapshot.monthly;
   904→          data.stripePriceIdYearly = snapshot.yearly;
   905→        }
   906→      }
   907→    } catch {
   908→      // best-effort — never block the create on a Stripe error
   909→    }
   910→  }
   911→
   912→  return data;
   913→}

   915→/**
   916→ * Delete a plan config. Refuses if:
   917→ *  - it's the last remaining plan, OR
   918→ *  - any Subscription still references this planId (would orphan the sub).
   919→ *
   920→ * Returns `{ ok: boolean, reason?: string }` describing the outcome.
   921→ */
   922→export async function deletePlanConfig(
   923→  planId: string,
   924→): Promise<{ ok: boolean; reason?: string }> {
   925→  const count = await db.planConfig.count();
   926→  if (count <= 1) return { ok: false, reason: 'Cannot delete the last plan.' };
   927→  const subCount = await db.subscription.count({ where: { planId } });
   928→  if (subCount > 0) {
   929→    return {
   930→      ok: false,
   931→      reason: `Cannot delete plan "${planId}" — ${subCount} active subscription(s) reference it. Migrate them first.`,
   932→    };
   933→  }
   934→  try {
   935→    await db.planConfig.delete({ where: { planId } });
   936→    await hydrate();
   937→    return { ok: true };
   938→  } catch (err) {
   939→    return { ok: false, reason: (err as Error).message };
   940→  }
   941→}
