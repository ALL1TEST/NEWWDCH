     2→     2→// PLATFORM DATA SERVICE — Centralized SaaS platform dataset
     3→     3→// ============================================================
     4→     4→// This is the SINGLE source of truth for the entire SaaS platform.
     5→     5→// Both the Client Billing experience AND the Platform Admin
     6→     6→// Dashboard read from and mutate this same in-memory dataset, so
     7→     7→// a plan change, cancellation, or payment made on the client side
     8→     8→// is immediately reflected on the admin side, and vice-versa.
     9→     9→//
    10→    10→// Consistency rules (enforced by derivation, never hardcoded):
    11→    11→//  - totalCustomers  = customers.length (non-deleted)
    12→    12→//  - activeSubscriptions = subscriptions where status === 'active'
    13→    13→//  - mrr = sum of normalized monthly price over active PAID subs
    14→    14→//  - totalSites = sites.length
    15→    15→//  - planDistribution = subscriptions grouped by plan (all statuses)
    16→    16→//  - statusCounts = subscriptions grouped by status
    17→    17→//  - recentCustomers = customers sorted by createdAt desc
    18→    18→//  - recentPayments = payments sorted by date desc
    19→    19→//  - usage = aggregated from sites/content/media/AI records
    20→    20→//  - alerts = derived from failed payments + past-due + storage
    21→    21→//
    22→    22→// Mock data is deterministic (no Math.random) so numbers are stable
    23→    23→// across renders and never contradict each other.

   175→   175→  const d = new Date();
   176→   176→  d.setDate(d.getDate() - days);
   177→   177→  d.setHours(9, 15, 0, 0);
   178→   178→  return d.toISOString();

   183→   183→  d.setHours(9, 15, 0, 0);
   184→   184→  return d.toISOString();
   185→   185→}

   345→   345→function buildRevenueSeries(currentMrr: number): RevenuePoint[] {
   346→   346→  const trend = [0.41, 0.47, 0.54, 0.62, 0.71, 0.80, 0.89, 1.0];
   347→   347→  const out: RevenuePoint[] = [];
   348→   348→  const now = new Date();
   349→   349→  for (let i = 7; i >= 0; i--) {
   350→   350→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   351→   351→    const factor = trend[7 - i];
   352→   352→    out.push({ month: MONTHS[d.getMonth()], revenue: Math.round(currentMrr * factor) });
   353→   353→  }
   354→   354→  return out;
   355→   355→}

   589→   589→        p.id.toLowerCase().includes(q) ||
   590→   590→        p.invoiceNumber.toLowerCase().includes(q) ||
   591→   591→        (c?.name.toLowerCase().includes(q) ?? false) ||
   592→   592→        (c?.email.toLowerCase().includes(q) ?? false)
   593→   593→      );
   594→   594→    });
   595→   595→  }
   596→   596→  return list
   597→   597→    .sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())
   598→   598→    .map((p) => {
   599→   599→      const c = s.customers.find((cu) => cu.id === p.customerId);
   600→   600→      return { ...p, customerName: c?.name ?? '—', customerEmail: c?.email ?? '—' };

   602→   602→}

   635→   635→export function getAlerts(): PlatformAlert[] {
   636→   636→  const s = store();
   637→   637→  const alerts: PlatformAlert[] = [];
   638→   638→
   639→   639→  const failed = s.payments.filter((p) => p.status === 'failed');
   640→   640→  if (failed.length > 0) {
   641→   641→    alerts.push({
   642→   642→      id: 'alert-failed-payments',
   643→   643→      severity: 'critical',
   644→   644→      title: 'Failed payments',
   645→   645→      message: `${failed.length} payment${failed.length === 1 ? '' : 's'} failed and need attention`,
   646→   646→      action: { label: 'View Payments', module: 'platform-payments' },
   647→   647→      time: 'recent',
   648→   648→    });
   649→   649→  }
   650→   650→
   651→   651→  const pastDue = s.customers.filter((c) => c.subscriptionStatus === 'past_due');
   652→   652→  if (pastDue.length > 0) {
   653→   653→    alerts.push({
   654→   654→      id: 'alert-past-due',
   655→   655→      severity: 'warning',
   656→   656→      title: 'Past-due subscriptions',
   657→   657→      message: `${pastDue.length} subscription${pastDue.length === 1 ? '' : 's'} past due`,
   658→   658→      action: { label: 'View Subscriptions', module: 'platform-subscriptions' },
   659→   659→      time: 'recent',
   660→   660→    });
   661→   661→  }
   662→   662→
   663→   663→  const nearLimit = s.customers.filter((c) => {
   664→   664→    const used = s.sites
   665→   665→      .filter((si) => si.customerId === c.id)
   666→   666→      .reduce((a, si) => a + si.storageBytes, 0);
   667→   667→    return c.storageLimitBytes > 0 && used / c.storageLimitBytes > 0.8;
   668→   668→  });
   669→   669→  if (nearLimit.length > 0) {
   670→   670→    alerts.push({
   671→   671→      id: 'alert-storage',
   672→   672→      severity: 'warning',
   673→   673→      title: 'Storage limit',
   674→   674→      message: `${nearLimit.length} customer${nearLimit.length === 1 ? '' : 's'} approaching storage limit`,
   675→   675→      action: { label: 'View Customers', module: 'platform-customers' },
   676→   676→      time: 'recent',
   677→   677→    });
   678→   678→  }
   679→   679→
   680→   680→  const weekAgo = Date.now() - 7 * 24 * 3600_000;
   681→   681→  const newCustomers = s.customers.filter((c) => new Date(c.createdAt).getTime() > weekAgo);
   682→   682→  if (newCustomers.length > 0) {
   683→   683→    alerts.push({
   684→   684→      id: 'alert-new-customers',
   685→   685→      severity: 'info',
   686→   686→      title: 'New customers',
   687→   687→      message: `${newCustomers.length} new customer${newCustomers.length === 1 ? '' : 's'} registered this week`,
   688→   688→      action: { label: 'View Customers', module: 'platform-customers' },
   689→   689→      time: 'recent',
   690→   690→    });
   691→   691→  }
   692→   692→
   693→   693→  return alerts;
   694→   694→}
