   146→/*
   147→ * SINGLE source of truth for the positioning of every tooltip/label that
   148→ * appears when the sidebar is collapsed into the 48px icon rail.
   149→ *
   150→ *   side="right"         → opens to the RIGHT of the 48px collapsed rail,
   151→ *                          fully inside the main viewport (never overlaps
   152→ *                          the rail, never opens to the left/outside).
   153→ *   align="center"       → vertically centers the bubble on the icon's
   154→ *                          32×32 hover target so labels line up across
   155→ *                          every row of the rail (logo, theme, nav items,
   156→ *                          parent items with submenu).
   157→ *   sideOffset=8         → ~8px visible gap from the trigger button's right
   158→ *                          edge (button right edge ~x=40 inside the 48px
   159→ *                          rail, tooltip left edge ~x=48 = the rail's
   160→ *                          right edge — flush with the RAIL, NOT with the
   161→ *                          icon, so the bubble never touches the glyph).
   162→ *   collisionPadding=12 → 12px viewport-edge collision padding so the
   163→ *                          bubble can never be clipped at any viewport
   164→ *                          edge or scroll past the visible area.
   165→ *
   166→ * Portal-based rendering (Radix TooltipPrimitive.Portal → document.body,
   167→ * wired inside src/components/ui/tooltip.tsx) means every tooltip is
   168→ * rendered OUTSIDE the sidebar DOM tree — so the sidebar's
   169→ * overflow-hidden / overflow-x-hidden on the rail container can NEVER
   170→ * clip or hide the bubble. The tooltip floats above every other
   171→ * element at z-50.
   172→ *
   173→ * Used by:
   174→ *   • CollapsedLogoButton         (header "Expand" tooltip)
   175→ *   • SimpleNavItem              (leaf nav items: Dashboard, Articles,
   176→ *                                  Calendar, Media, Users, Comments,
   177→ *                                  Newsletter, SEO, AI, Automation)
   178→ *   • ExpandableNavItem          (parent nav items in EXPANDED state —
   179→ *                                  tooltip is hidden by SidebarMenuButton
   180→ *                                  when state!=="collapsed", but the
   181→ *                                  positioning object is still applied
   182→ *                                  so the prop stays consistent across
   183→ *                                  every nav item renderer)
   184→ *   • CollapsedParentNavItem     (parent nav items in COLLAPSED state —
   185→ *                                  label tooltip shows when the floating
   186→ *                                  popover is CLOSED; when the popover is
   187→ *                                  open, no tooltip shows to avoid visual
   188→ *                                  conflict)
   189→ *
   190→ * The ThemeToggle component (src/components/layout/theme-toggle.tsx) lives
   191→ * in a separate file and inlines the SAME four values — see the comment
   192→ * there for the rationale.

   287→ * Derive the top-level section ID from the current browser hash.

   290→function getSectionFromHash(hash: string): string | null {
   291→  const path = hash.replace(/^#\/?,?/, '').trim();
   292→  if (!path || path === 'dashboard') return null;
   293→
   294→  const firstSegment = path.split('/')[0];
   295→  return ROUTE_PREFIX_TO_SECTION[firstSegment] ?? null;

   347→/**
   348→ * Collapsed-rail logo cell. Since the extra collapse-toggle icon was removed
   349→ * from below the logo, THIS cell is the explicit expand control:
   350→ *   • identical 32×32 grid geometry as every other rail icon
   351→ *   • click toggles the sidebar open (SidebarRail edge strip still works)
   352→ * No new icon is introduced and no empty clickable ghost remains.
   353→ *
   354→ * HOVER BEHAVIOR (collapsed state only — this button is rendered ONLY
   355→ * inside the collapsed-rail cluster, so the expanded LogoMark is untouched):
   356→ *   • at rest ............ shows the "C" logo mark on its bg-primary black
   357→ *                           box (NORMAL LOGO — background UNCHANGED, same
   358→ *                           as the expanded LogoMark).
   359→ *   • on mouse-enter ..... the "C" is replaced by a PanelLeftOpen icon AND
   360→ *                           the button's background becomes transparent so
   361→ *                           ONLY the Expand icon is visible with no box
   362→ *                           behind it. The icon color switches to
   363→ *                           muted-foreground (gray) so it stays visible
   364→ *                           against the transparent/page bg in both Light
   365→ *                           and Dark mode.
   366→ *   • on mouse-leave ..... restores the "C" logo mark AND its bg-primary box.
   367→ *   • click ............. calls toggleSidebar (existing functionality, unchanged).
   368→ *
   369→ * Background is removed ONLY from the temporary Expand-icon state; the
   370→ * normal logo (at rest) keeps its bg-primary background exactly as-is.
   371→ *
   372→ * Implementation: React state (useState + onMouseEnter/onMouseLeave) — NOT
   373→ * CSS :hover / group-hover variants. This is deliberate: Tailwind v4 wraps
   374→ * its hover:/group-hover: utilities inside @media (hover: hover), so on
   375→ * any browser that reports (hover: none) (headless, touch, some preview
   376→ * iframes) those utilities NEVER activate and the swap + bg-transparent
   377→ * would silently fail. React mouse events fire on ANY pointer input
   378→ * regardless of the hover media query, so the behavior is identical in
   379→ * every environment. The conditional className + conditional render swap
   380→ * the bg/color/glyph atomically per state. Works in both Light and Dark.
   381→ */
   382→function CollapsedLogoButton() {
   383→  const { toggleSidebar } = useSidebar();
   384→  // Local hover state drives the logo↔Expand-icon swap AND the per-state
   385→  // background/color. Using React state (not CSS :hover / group-hover)
   386→  // guarantees the swap works in ALL environments — including browsers
   387→  // that report (hover: none), where Tailwind's @media (hover: hover)-
   388→  // gated hover:/group-hover: variants would never activate.
   389→  const [hovered, setHovered] = useState(false);
   390→  return (
   391→    /* Hover-only "Expand" tooltip: instant (provider delayDuration=0),
   392→       appears on mouse-enter, disappears on mouse-leave. Radix closes it
   393→       on any pointer exit — disableHoverableContent guarantees it can
   394→       never linger or trap the pointer. */
   395→    <Tooltip disableHoverableContent>
   396→      <TooltipTrigger asChild>
   397→        <button
   398→          type="button"
   399→          onClick={toggleSidebar}
   400→          aria-label="Expand sidebar"
   401→          onMouseEnter={() => setHovered(true)}
   402→          onMouseLeave={() => setHovered(false)}
   403→          className={cn(
   404→            'flex h-8 w-8 shrink-0 cursor-pointer items-center justify-center rounded-lg font-bold text-sm outline-none transition-opacity hover:opacity-90 focus-visible:ring-2 focus-visible:ring-ring select-none',
   405→            // NORMAL LOGO STATE (at rest): bg-primary black box +
   406→            // text-primary-foreground white "C" — background UNCHANGED,
   407→            // same as the expanded LogoMark.
   408→            // EXPAND-ICON STATE (on hover): bg-transparent (NO background)
   409→            // + text-muted-foreground gray icon — ONLY this temporary
   410→            // state loses the background. The icon stays visible (gray
   411→            // on transparent/page bg) in both Light and Dark mode.
   412→            hovered
   413→              ? 'bg-transparent text-muted-foreground'
   414→              : 'bg-primary text-primary-foreground',
   415→          )}
   416→        >
   417→          {/* Conditional render: "C" logo at rest, PanelLeftOpen icon on
   418→              hover. h-4 w-4 matches the CollapseToggle's PanelLeftClose
   419→              icon size; the icon is the visual opposite of the collapse
   420→              icon so the collapsed logo reads as a clear "expand"
   421→              affordance when the pointer is over it. */}
   422→          {hovered ? (
   423→            <PanelLeftOpen className="h-4 w-4" />
   424→          ) : (
   425→            <span>C</span>
   426→          )}
   427→          <span className="sr-only">Expand sidebar</span>
   428→        </button>
   429→      </TooltipTrigger>
   430→      {/* Collapsed-rail tooltip — uses the SHARED COLLAPSED_TOOLTIP_PROPS
   431→          constant so the "Expand" label has IDENTICAL positioning (side,
   432→          align, sideOffset, collisionPadding) to every other collapsed-
   433→          rail tooltip (ThemeToggle, SimpleNavItem, ExpandableNavItem,
   434→          CollapsedParentNavItem). Portal-based rendering means the bubble
   435→          floats outside the sidebar DOM at z-50 — never clipped by the
   436→          rail's overflow-hidden. */}
   437→      <TooltipContent {...COLLAPSED_TOOLTIP_PROPS}>Expand</TooltipContent>
   438→    </Tooltip>
   439→  );
   440→}

   510→/**
   511→ * Settings parent item rendered on the COLLAPSED 48px rail.
   512→ * Clicking it opens a portal-based popover to the RIGHT of the icon —
   513→ * the sidebar itself NEVER expands, and the popover anchors to this
   514→ * exact button so the "star/sparkle alignment" grid stays intact.
   515→ */
   516→function CollapsedParentNavItem({
   517→  item,
   518→  currentModule,
   519→  currentSubPage,
   520→}: {
   521→  item: NavItem;
   522→  currentModule: string;
   523→  currentSubPage: string | null;
   524→}) {
   525→  const [floatOpen, setFloatOpen] = useState(false);
   526→  const mod = hrefToModule(item.href);
   527→  const isActive = currentModule === mod;
   528→
   529→  const handleChildNavigate = (
   530→    e: React.MouseEvent<HTMLAnchorElement>,
   531→    child: NavItem,
   532→  ) => {
   533→    e.preventDefault();
   534→    e.stopPropagation();
   535→    setFloatOpen(false); // close the floating submenu after navigation…
   536→    // …but do NOT touch sidebar expansion state (rule #7).
   537→    const hash = child.href.replace(/^#/, '');
   538→    const parts = hash.split('/');
   539→    const childSubPage = parts[1] || null;
   540→    useNavigationStore.getState().navigate(parts[0], null, childSubPage);
   541→  };
   542→
   543→  return (
   544→    <SidebarMenuItem>
   545→      <Popover open={floatOpen} onOpenChange={setFloatOpen}>
   546→        <PopoverTrigger asChild>
   547→          <SidebarMenuButton
   548→            isActive={isActive || floatOpen}
   549→            aria-expanded={floatOpen}
   550→            aria-haspopup="menu"
   551→            // Collapsed-rail tooltip — uses the SHARED COLLAPSED_TOOLTIP_PROPS
   552→            // so the parent item's label has IDENTICAL positioning (side,
   553→            // align, sideOffset, collisionPadding) to every other collapsed-
   554→            // rail tooltip (CollapsedLogoButton, ThemeToggle, SimpleNavItem,
   555→            // ExpandableNavItem). Suppressed when the popover is open so the
   556→            // floating submenu never visually conflicts with a lingering
   557→            // label bubble. Portal-based rendering means the bubble floats
   558→            // outside the sidebar DOM at z-50 — never clipped by the rail's
   559→            // overflow-hidden.
   560→            tooltip={
   561→              floatOpen
   562→                ? undefined
   563→                : { ...COLLAPSED_TOOLTIP_PROPS, children: item.label }
   564→            }
   565→            onClick={(e: React.MouseEvent) => {
   566→              e.preventDefault();
   567→              setFloatOpen((o) => !o);
   568→            }}
   569→          >
   570→            <NavIcon name={item.icon} />
   571→            <span>{item.label}</span>
   572→          </SidebarMenuButton>
   573→        </PopoverTrigger>
   574→        {/* Portal → document.body: immune to sidebar overflow clipping.
   575→            Radix handles Esc + outside-click closing and re-anchors at any
   576→            viewport size/zoom via avoidCollisions. */}
   577→        <PopoverContent
   578→          side="right"
   579→          align="start"
   580→          sideOffset={6}
   581→          collisionPadding={12}
   582→          className="w-56 rounded-md border bg-popover p-1.5 shadow-md"
   583→          onEscapeKeyDown={() => setFloatOpen(false)}
   584→        >
   585→          <ul
   586→            role="menu"
   587→            aria-label={`${item.label} submenu`}
   588→            className="flex min-w-0 list-none flex-col gap-0.5 p-0 m-0"
   589→          >
   590→            {item.children!.map((child) => {
   591→              const hash = child.href.replace(/^#/, '');

   593→              const childSubPage = parts[1] || null;

   604→                    >

   619→function ExpandableNavItem({
   620→  item,
   621→  currentModule,
   622→  currentSubPage,
   623→  isExpanded,
   624→  onToggle,
   625→}: {
   626→  item: NavItem;
   627→  currentModule: string;
   628→  currentSubPage: string | null;
   629→  isExpanded: boolean;
   630→  onToggle: () => void;
   631→}) {
   632→  const mod = hrefToModule(item.href);
   633→  const isActive = currentModule === mod;
   634→  const sectionId = `submenu-${item.label.toLowerCase().replace(/\s+/g, '-')}`;
   635→
   636→  return (
   637→    <SidebarMenuItem>
   638→      <SidebarMenuButton
   639→        // Expandable parent item rendered in EXPANDED state. SidebarMenuButton
   640→        // hides its tooltip whenever state!=="collapsed" (see ui/sidebar.tsx),
   641→        // so this label never shows while expanded — but passing the SAME
   642→        // COLLAPSED_TOOLTIP_PROPS object as every other nav-item renderer
   643→        // keeps the prop consistent across the codebase. If the sidebar
   644→        // toggles between expanded and collapsed while this row stays
   645→        // mounted, the tooltip that appears is identical in positioning to
   646→        // SimpleNavItem / CollapsedParentNavItem / CollapsedLogoButton.
   647→        tooltip={{ ...COLLAPSED_TOOLTIP_PROPS, children: item.label }}
   648→        isActive={isActive}
   649→        onClick={(e: React.MouseEvent) => {
   650→          e.preventDefault();
   651→          onToggle();
   652→        }}
   653→        aria-expanded={isExpanded}
   654→        aria-controls={sectionId}
   655→      >
   656→        <NavIcon name={item.icon} />
   657→        <span>{item.label}</span>
   658→        <ChevronRight
   659→          className={cn(
   660→            'ml-auto h-4 w-4 shrink-0 transition-transform duration-200',
   661→            isExpanded && 'rotate-90',
   662→          )}
   663→        />
   664→      </SidebarMenuButton>
   665→
   666→      {item.badge != null && (
   667→        <SidebarMenuBadge>
   668→          {item.badge}
   669→        </SidebarMenuBadge>
   670→      )}
   671→
   672→      <AccordionSubmenu isOpen={isExpanded} sectionLabel={item.label}>
   673→        {item.children!.map((child) => {
   674→          const hash = child.href.replace(/^#/, '');
   675→          const parts = hash.split('/');
   676→          const childMod = parts[0];
   677→          const childSubPage = parts[1] || null;
   678→          const isChildActive =
   679→            currentModule === childMod &&
   680→            (!childSubPage || currentSubPage === childSubPage);
   681→
   682→          return (
   683→            <SidebarMenuSubItem key={child.label}>
   684→              <SidebarMenuSubButton
   685→                asChild
   686→                isActive={isChildActive}
   687→              >
   688→                <a
   689→                  href={child.href}
   690→                  onClick={(e) => {
   691→                    e.preventDefault();
   692→                    e.stopPropagation();
   693→                    useNavigationStore
   694→                      .getState()
   695→                      .navigate(childMod, null, childSubPage);
   696→                  }}
   697→                >
   698→                  <NavIcon name={child.icon} />
   699→                  <span>{child.label}</span>
   700→                </a>
   701→              </SidebarMenuSubButton>
   702→            </SidebarMenuSubItem>
   703→          );
   704→        })}
   705→      </AccordionSubmenu>
   706→    </SidebarMenuItem>
   707→  );

   748→      {item.badge != null && (

   759→function NavGroupSection({
   760→  group,
   761→  currentModule,
   762→  currentSubPage,
   763→  openSection,
   764→  onToggleSection,
   765→}: {
   766→  group: NavGroup;
   767→  currentModule: string;
   768→  currentSubPage: string | null;
   769→  openSection: string | null;
   770→  onToggleSection: (label: string) => void;
   771→}) {
   772→  const isCollapsedRail = useCollapsedRail();
   773→
   774→  return (
   775→    <SidebarGroup>
   776→      <SidebarGroupContent>
   777→        <SidebarMenu>
   778→          {group.items.map((item) => {
   779→            const hasChildren = item.children && item.children.length > 0;
   780→
   781→            if (hasChildren && isCollapsedRail) {
   782→              /* Collapsed rail → parents with children become floating
   783→                 popovers anchored to their own icon. The rail NEVER
   784→                 auto-expands (#4/#5/#6). */
   785→              return (
   786→                <CollapsedParentNavItem
   787→                  key={item.label}
   788→                  item={item}
   789→                  currentModule={currentModule}
   790→                  currentSubPage={currentSubPage}
   791→                />
   792→              );
   793→            }
   794→
   795→            if (hasChildren) {
   796→              return (
   797→                <ExpandableNavItem
   798→                  key={item.label}
   799→                  item={item}
   800→                  currentModule={currentModule}
   801→                  currentSubPage={currentSubPage}
   802→                  isExpanded={openSection === item.label}
   803→                  onToggle={() => onToggleSection(item.label)}
   804→                />
   805→              );
   806→            }
   807→
   808→            return (
   809→              <SimpleNavItem
   810→                key={item.label}
   811→                item={item}
   812→                currentModule={currentModule}
   813→              />
   814→            );
   815→          })}
   816→        </SidebarMenu>
   817→      </SidebarGroupContent>
   818→    </SidebarGroup>
   819→  );
   820→}
