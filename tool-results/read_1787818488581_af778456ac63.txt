   226→/**
   227→ * Derive the top-level section ID from the current browser hash.
   228→ * Uses EXPLICIT prefix matching — #settings/seo always returns 'Settings', never 'SEO'.
   229→ */
   230→function getSectionFromHash(hash: string): string | null {
   231→  const path = hash.replace(/^#\/?,?/, '').trim();
   232→  if (!path || path === 'dashboard') return null;
   233→
   234→  const firstSegment = path.split('/')[0];
   235→  return ROUTE_PREFIX_TO_SECTION[firstSegment] ?? null;
   236→}

   364→/*

   377→/**
   378→ * Settings parent item rendered on the COLLAPSED 48px rail.
   379→ * Clicking it opens a portal-based popover to the RIGHT of the icon —
   380→ * the sidebar itself NEVER expands, and the popover anchors to this
   381→ * exact button so the "star/sparkle alignment" grid stays intact.
   382→ */
   383→function CollapsedParentNavItem({
   384→  item,
   385→  currentModule,
   386→  currentSubPage,
   387→}: {
   388→  item: NavItem;
   389→  currentModule: string;
   390→  currentSubPage: string | null;
   391→}) {
   392→  const [floatOpen, setFloatOpen] = useState(false);
   393→  const mod = hrefToModule(item.href);
   394→  const isActive = currentModule === mod;
   395→
   396→  const handleChildNavigate = (
   397→    e: React.MouseEvent<HTMLAnchorElement>,
   398→    child: NavItem,
   399→  ) => {
   400→    e.preventDefault();
   401→    e.stopPropagation();
   402→    setFloatOpen(false); // close the floating submenu after navigation…
   403→    // …but do NOT touch sidebar expansion state (rule #7).
   404→    const hash = child.href.replace(/^#/, '');
   405→    const parts = hash.split('/');
   406→    const childSubPage = parts[1] || null;
   407→    useNavigationStore.getState().navigate(parts[0], null, childSubPage);
   408→  };
   409→
   410→  return (
   411→    <SidebarMenuItem>
   412→      <Popover open={floatOpen} onOpenChange={setFloatOpen}>
   413→        <PopoverTrigger asChild>
   414→          <SidebarMenuButton
   415→            isActive={isActive || floatOpen}
   416→            aria-expanded={floatOpen}
   417→            aria-haspopup="menu"
   418→            tooltip={floatOpen ? undefined : item.label}
   419→            onClick={(e: React.MouseEvent) => {
   420→              e.preventDefault();
   421→              setFloatOpen((o) => !o);
   422→            }}
   423→          >
   424→            <NavIcon name={item.icon} />
   425→            <span>{item.label}</span>
   426→          </SidebarMenuButton>
   427→        </PopoverTrigger>
   428→        {/* Portal → document.body: immune to sidebar overflow clipping.
   429→            Radix handles Esc + outside-click closing and re-anchors at any
   430→            viewport size/zoom via avoidCollisions. */}
   431→        <PopoverContent
   432→          side="right"
   433→          align="start"
   434→          sideOffset={6}
   435→          collisionPadding={12}
   436→          className="w-56 rounded-md border bg-popover p-1.5 shadow-md"
   437→          onEscapeKeyDown={() => setFloatOpen(false)}
   438→        >
   439→          <ul
   440→            role="menu"
   441→            aria-label={`${item.label} submenu`}
   442→            className="flex min-w-0 list-none flex-col gap-0.5 p-0 m-0"
   443→          >
   444→            {item.children!.map((child) => {
   445→              const hash = child.href.replace(/^#/, '');
   446→              const parts = hash.split('/');
   447→              const childSubPage = parts[1] || null;
   448→              const isChildActive =
   449→                currentModule === parts[0] &&
   450→                (!childSubPage || currentSubPage === childSubPage);
   451→              return (
   452→                <SidebarMenuSubItem key={child.label} className="list-none p-0">
   453→                  <SidebarMenuSubButton asChild isActive={isChildActive}>
   454→                    <a
   455→                      href={child.href}
   456→                      role="menuitem"
   457→                      onClick={(e) => handleChildNavigate(e, child)}
   458→                    >
   459→                      <NavIcon name={child.icon} />
   460→                      <span>{child.label}</span>
   461→                    </a>
   462→                  </SidebarMenuSubButton>
   463→                </SidebarMenuSubItem>
   464→              );
   465→            })}
   466→          </ul>
   467→        </PopoverContent>
   468→      </Popover>
   469→    </SidebarMenuItem>
   470→  );
   471→}

   473→function ExpandableNavItem({
   474→  item,
   475→  currentModule,
   476→  currentSubPage,
   477→  isExpanded,
   478→  onToggle,
   479→}: {
   480→  item: NavItem;
   481→  currentModule: string;
   482→  currentSubPage: string | null;
   483→  isExpanded: boolean;
   484→  onToggle: () => void;
   485→}) {
   486→  const mod = hrefToModule(item.href);
   487→  const isActive = currentModule === mod;
   488→  const sectionId = `submenu-${item.label.toLowerCase().replace(/\s+/g, '-')}`;
   489→
   490→  return (
   491→    <SidebarMenuItem>
   492→      <SidebarMenuButton
   493→        tooltip={item.label}
   494→        isActive={isActive}
   495→        onClick={(e: React.MouseEvent) => {
   496→          e.preventDefault();
   497→          onToggle();
   498→        }}
   499→        aria-expanded={isExpanded}
   500→        aria-controls={sectionId}
   501→      >
   502→        <NavIcon name={item.icon} />
   503→        <span>{item.label}</span>
   504→        <ChevronRight
   505→          className={cn(
   506→            'ml-auto h-4 w-4 shrink-0 transition-transform duration-200',
   507→            isExpanded && 'rotate-90',
   508→          )}
   509→        />
   510→      </SidebarMenuButton>
   511→
   512→      {item.badge != null && (
   513→        <SidebarMenuBadge>
   514→          {item.badge}
   515→        </SidebarMenuBadge>
   516→      )}
   517→
   518→      <AccordionSubmenu isOpen={isExpanded} sectionLabel={item.label}>
   519→        {item.children!.map((child) => {
   520→          const hash = child.href.replace(/^#/, '');
   521→          const parts = hash.split('/');
   522→          const childMod = parts[0];
   523→          const childSubPage = parts[1] || null;
   524→          const isChildActive =
   525→            currentModule === childMod &&
   526→            (!childSubPage || currentSubPage === childSubPage);
   527→
   528→          return (
   529→            <SidebarMenuSubItem key={child.label}>
   530→              <SidebarMenuSubButton
   531→                asChild
   532→                isActive={isChildActive}
   533→              >
   534→                <a
   535→                  href={child.href}
   536→                  onClick={(e) => {
   537→                    e.preventDefault();
   538→                    e.stopPropagation();
   539→                    useNavigationStore
   540→                      .getState()
   541→                      .navigate(childMod, null, childSubPage);
   542→                  }}
   543→                >
   544→                  <NavIcon name={child.icon} />
   545→                  <span>{child.label}</span>
   546→                </a>
   547→              </SidebarMenuSubButton>
   548→            </SidebarMenuSubItem>
   549→          );
   550→        })}
   551→      </AccordionSubmenu>
   552→    </SidebarMenuItem>
   553→  );
   554→}

   596→function NavGroupSection({
   597→  group,
   598→  currentModule,
   599→  currentSubPage,
   600→  openSection,
   601→  onToggleSection,
   602→}: {
   603→  group: NavGroup;
   604→  currentModule: string;
   605→  currentSubPage: string | null;
   606→  openSection: string | null;
   607→  onToggleSection: (label: string) => void;
   608→}) {
   609→  const isCollapsedRail = useCollapsedRail();
   610→
   611→  return (
   612→    <SidebarGroup>
   613→      <SidebarGroupContent>
   614→        <SidebarMenu>
   615→          {group.items.map((item) => {
   616→            const hasChildren = item.children && item.children.length > 0;
   617→
   618→            if (hasChildren && isCollapsedRail) {
   619→              /* Collapsed rail → parents with children become floating
   620→                 popovers anchored to their own icon. The rail NEVER
   621→                 auto-expands (#4/#5/#6). */
   622→              return (
   623→                <CollapsedParentNavItem
   624→                  key={item.label}
   625→                  item={item}
   626→                  currentModule={currentModule}
   627→                  currentSubPage={currentSubPage}
   628→                />
   629→              );
   630→            }
   631→
   632→            if (hasChildren) {
   633→              return (
   634→                <ExpandableNavItem
   635→                  key={item.label}
   636→                  item={item}
   637→                  currentModule={currentModule}
   638→                  currentSubPage={currentSubPage}
   639→                  isExpanded={openSection === item.label}
   640→                  onToggle={() => onToggleSection(item.label)}
   641→                />
   642→              );
   643→            }
   644→
   645→            return (
   646→              <SimpleNavItem
   647→                key={item.label}
   648→                item={item}
   649→                currentModule={currentModule}
   650→              />
   651→            );
   652→          })}
   653→        </SidebarMenu>
   654→      </SidebarGroupContent>
   655→    </SidebarGroup>
   656→  );
   657→}

   668→
   669→  const userRole = user?.role;
   670→  const pagePermissions = user?.pagePermissions ?? null;
   671→  const visibleItems = useMemo(() => {
   672→    if (!userRole) return [];
   673→    return getVisibleNavItems(userRole, NAV_ITEMS, pagePermissions);
   674→  }, [userRole, pagePermissions]);
   675→
   676→  /*
   677→   * SINGLE SOURCE OF TRUTH for the expanded top-level section.
   678→   *
   679→   * manualOverride is a tri-state:
   680→   *   null  → not overridden; use route-derived value
   681→   *   ""   → explicitly closed by user
   682→   *   "SEO" → explicitly opened by user
   683→   *
   684→   * When the module changes (navigation event), we reset manualOverride
   685→   * to null so the route-derived value takes over.
   686→   */
   687→  const [manualOverride, setManualOverride] = useState<string | null>(null);
   688→  // Track previous module to detect navigation changes synchronously during render.
   689→  // This is the React-approved pattern for "derived state from props".
   690→  const [prevModule, setPrevModule] = useState(currentModule);
   691→  if (currentModule !== prevModule) {
   692→    setPrevModule(currentModule);
   693→    setManualOverride(null);
   694→  }
   695→
   696→  // Derive the section from the current route hash (EXPLICIT prefix matching)
   697→  const routeDerivedSection = useMemo(() => {
   698→    return getSectionFromHash(window.location.hash);
   699→  }, [currentModule, currentSubPage]);
   700→
   701→  // Compute the actual open section
   702→  const openSection = useMemo(() => {
   703→    if (manualOverride !== null) {
   704→      // User manually toggled: empty string = closed, non-empty = that section
   705→      return manualOverride || null;
   706→    }
   707→    return routeDerivedSection;
   708→  }, [manualOverride, routeDerivedSection]);
   709→
   710→  // Toggle a section open/closed
   711→  const handleToggleSection = useCallback(
   712→    (label: string) => {
   713→      setManualOverride((prev) => {
   714→        // Determine what's currently open (considering route-derived if no override)
   715→        const currentlyOpen =
   716→          prev !== null ? (prev || null) : routeDerivedSection;
   717→        // If clicking the same section that's open, close it
   718→        if (currentlyOpen === label) {
   719→          return ''; // explicitly closed
   720→        }
   721→        // Otherwise, open the clicked section (closing any other)
   722→        return label;
   723→      });
   724→    },
   725→    [routeDerivedSection],
   726→  );
   727→
   728→  if (!user) return null;
   729→
   730→  const groups = buildNavGroups(visibleItems);
   731→
   732→  return (
   733→    <Sidebar collapsible="icon">
   734→      {/* ---- Header: one 32px icon cell per row, all centered at x=24px ---- */}
   735→      <SidebarHeader className="px-2 py-3 shrink-0">
   736→        {/* Expanded: [logo][title][collapse control] single row */}
   737→        <div className="flex h-8 items-center gap-2 group-data-[collapsible=icon]:hidden">
   738→          <LogoMark />
   739→          <span className="truncate font-semibold text-sm tracking-tight whitespace-nowrap text-text-primary">
   740→            CMS Admin
   741→          </span>
   742→          <div className="ml-auto shrink-0">
   743→            <CollapseToggle side="bottom" />
   744→          </div>
   745→        </div>
   746→        {/* Collapsed rail: ONLY the "C" logo. The previously stacked
   747→            collapse-toggle icon was removed on purpose — no extra icon may
   748→            appear under/next to the logo.
   749→            Expand affordance: the logo cell itself toggles the sidebar
   750→            (same 32px grid cell, same x=24 center-line), and the invisible
   751→            SidebarRail edge strip keeps its native toggle behavior too. */}
   752→        <div className="hidden flex-col items-center group-data-[collapsible=icon]:flex">
   753→          <CollapsedLogoButton />
   754→        </div>
   755→      </SidebarHeader>
   756→
   757→      <SidebarSeparator className="mx-0" />
   758→
   759→      <SidebarContent className="sidebar-thin-scroll">
   760→        {groups.map((group) => (
   761→          <NavGroupSection
   762→            key={group.label}
   763→            group={group}
   764→            currentModule={currentModule}
   765→            currentSubPage={currentSubPage}
   766→            openSection={openSection}
   767→            onToggleSection={handleToggleSection}
   768→          />
   769→        ))}
   770→      </SidebarContent>
   771→
   772→      {/* Hide the bottom separator while collapsed — with the footer
   773→          intentionally empty in that state there is nothing to separate,
   774→          so the rail bottom stays clean (no orphan hairline). */}
   775→      <SidebarSeparator className="mx-0 group-data-[collapsible=icon]:hidden" />
   776→
   777→      <SidebarFooter className="shrink-0 group-data-[collapsible=icon]:p-0">
   778→        {/* Expanded: unchanged row layout [avatar][name/role][logout] */}
   779→        <div className="flex items-center gap-3 px-2 py-2 group-data-[collapsible=icon]:hidden">
   780→          <Avatar className="h-8 w-8 shrink-0">
   781→            <AvatarImage src={user.avatarUrl ?? undefined} alt={user.name} />
   782→            <AvatarFallback className="text-xs">
   783→              {getInitials(user.name)}
   784→            </AvatarFallback>
   785→          </Avatar>
   786→          <div className="flex flex-1 flex-col overflow-hidden">
   787→            <span className="truncate text-sm font-medium leading-tight">
   788→              {user.name}
   789→            </span>
   790→            <Badge variant="secondary" className="mt-0.5 h-4 w-fit text-[10px] px-1.5">
   791→              {user.role.replace(/_/g, ' ')}
   792→            </Badge>
   793→          </div>
   794→          <Tooltip>
   795→            <TooltipTrigger asChild>
   796→              <Button
   797→                variant="ghost"
   798→                size="icon"
   799→                className="h-8 w-8 shrink-0 rounded-md"
   800→                onClick={() => void logout()}
   801→                aria-label="Log out"
   802→              >
   803→                <LogOut className="h-4 w-4" />
   804→                <span className="sr-only">Log out</span>
   805→              </Button>
   806→            </TooltipTrigger>
   807→            <TooltipContent side="top">Log out</TooltipContent>
   808→          </Tooltip>
   809→        </div>
   810→
   811→        {/* Collapsed rail: NO admin profile block is rendered here BY DESIGN.
   812→            Avatar / name / role tooltip and the logout icon only exist in
   813→            the expanded layout above; expanding the sidebar restores them.
   814→            This keeps the bottom of the 48px rail clean per spec. */}
   815→      </SidebarFooter>
   816→
   817→      <SidebarRail />
   818→    </Sidebar>
   819→  );
   820→}
